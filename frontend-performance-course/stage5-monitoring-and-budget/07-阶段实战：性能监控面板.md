### 步骤 3：实现 Web Vitals 采集

```javascript
// lib/vitals.js
import { onLCP, onINP, onCLS, onFCP, onTTFB } from 'web-vitals';

const VITALS_ENDPOINT = '/api/vitals';

function sendMetric(metric) {
  const body = JSON.stringify({
    name: metric.name,
    value: metric.value,
    rating: metric.rating,
    delta: metric.delta,
    id: metric.id,
    page: window.location.pathname,
    timestamp: Date.now(),
    userAgent: navigator.userAgent,
    connection: navigator.connection?.effectiveType,
    deviceMemory: navigator.deviceMemory,
    viewport: `${window.innerWidth}x${window.innerHeight}`,
  });

  if (navigator.sendBeacon) {
    navigator.sendBeacon(VITALS_ENDPOINT, body);
  } else {
    fetch(VITALS_ENDPOINT, {
      method: 'POST',
      body,
      headers: { 'Content-Type': 'application/json' },
      keepalive: true,
    });
  }
}

export function initVitals() {
  onLCP(sendMetric);
  onINP(sendMetric);
  onCLS(sendMetric);
  onFCP(sendMetric);
  onTTFB(sendMetric);
}
```

import { NextResponse } from 'next/server';
import { db } from '@/lib/database';

export async function POST(request) {
  try {
    const metric = await request.json();

    await db.insert('web_vitals', {
      name: metric.name,
      value: metric.value,
      rating: metric.rating,
      page: metric.page,
      timestamp: new Date(metric.timestamp),
      user_agent: metric.userAgent,
      connection: metric.connection,
      device_memory: metric.deviceMemory,
      viewport: metric.viewport,
    });

    return NextResponse.json({ success: true });
  } catch (error) {
    console.error('Failed to save metric:', error);
    return NextResponse.json({ error: 'Failed to save' }, { status: 500 });
  }
}

### 步骤 5：构建监控看板

```jsx
// app/dashboard/performance/page.tsx
import { db } from '@/lib/database';

export default async function PerformanceDashboard() {
  // 获取最近 7 天的数据
  const metrics = await db.query(`
    SELECT
      name,
      PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY value) as p50,
      PERCENTILE_CONT(0.75) WITHIN GROUP (ORDER BY value) as p75,
      PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY value) as p95,
      COUNT(*) as sample_count
    FROM web_vitals
    WHERE timestamp > NOW() - INTERVAL '7 days'
    GROUP BY name
  `);

  // 获取按页面分组的数据
  const byPage = await db.query(`
    SELECT
      page,
      name,
      PERCENTILE_CONT(0.75) WITHIN GROUP (ORDER BY value) as p75
    FROM web_vitals
    WHERE timestamp > NOW() - INTERVAL '7 days'
    GROUP BY page, name
    ORDER BY page, name
  `);

  // 获取趋势数据
  const trends = await db.query(`
    SELECT
      DATE(timestamp) as date,
      name,
      PERCENTILE_CONT(0.75) WITHIN GROUP (ORDER BY value) as p75
    FROM web_vitals
    WHERE timestamp > NOW() - INTERVAL '30 days'
    GROUP BY DATE(timestamp), name
    ORDER BY date, name
  `);

  return (
    <div className="p-8">
      <h1 className="text-2xl font-bold mb-8">性能监控面板</h1>

      {/* 核心指标卡片 */}
      <div className="grid grid-cols-3 gap-4 mb-8">
        {metrics.map((metric) => (
          <MetricCard key={metric.name} metric={metric} />
        ))}
      </div>

      {/* 按页面分组 */}
      <div className="mb-8">
        <h2 className="text-xl font-semibold mb-4">按页面分组</h2>
        <PageMetricsTable data={byPage} />
      </div>

      {/* 趋势图 */}
      <div>
        <h2 className="text-xl font-semibold mb-4">30 天趋势</h2>
        <TrendChart data={trends} />
      </div>
    </div>
  );
}

function MetricCard({ metric }) {
  const thresholds = {
    LCP: { good: 2500, poor: 4000 },
    INP: { good: 200, poor: 500 },
    CLS: { good: 0.1, poor: 0.25 },
    FCP: { good: 1800, poor: 3000 },
    TTFB: { good: 800, poor: 1800 },
  };

  const threshold = thresholds[metric.name] || {};
  const isGood = metric.p75 <= threshold.good;
  const isPoor = metric.p75 > threshold.poor;

  return (
    <div className={`p-4 rounded-lg border ${
      isGood ? 'border-green-200 bg-green-50' :
      isPoor ? 'border-red-200 bg-red-50' :
      'border-yellow-200 bg-yellow-50'
    }`}>
      <h3 className="font-medium text-gray-600">{metric.name}</h3>
      <p className="text-3xl font-bold mt-2">
        {metric.name === 'CLS' ? metric.p75.toFixed(3) : Math.round(metric.p75)}
        <span className="text-sm font-normal ml-1">
          {metric.name === 'CLS' ? '' : 'ms'}
        </span>
      </p>
      <p className="text-sm text-gray-500 mt-1">
        P75 · {metric.sample_count} 样本
      </p>
    </div>
  );
}
```

### 步骤 6：设置告警

```javascript
// lib/alerts.js
const ALERT_THRESHOLDS = {
  LCP: { warning: 2500, critical: 4000 },
  INP: { warning: 200, critical: 500 },
  CLS: { warning: 0.1, critical: 0.25 },
};

export async function checkAlerts() {
  for (const [metric, thresholds] of Object.entries(ALERT_THRESHOLDS)) {
    // 获取最近 1 小时的 P75
    const result = await db.query(`
      SELECT PERCENTILE_CONT(0.75) WITHIN GROUP (ORDER BY value) as p75
      FROM web_vitals
      WHERE name = $1 AND timestamp > NOW() - INTERVAL '1 hour'
    `, [metric]);

    const p75 = result[0]?.p75;
    if (!p75) continue;

    if (p75 > thresholds.critical) {
      await sendAlert({
        severity: 'critical',
        metric,
        value: p75,
        threshold: thresholds.critical,
        message: `${metric} P75 严重超标: ${p75} > ${thresholds.critical}`,
      });
    } else if (p75 > thresholds.warning) {
      await sendAlert({
        severity: 'warning',
        metric,
        value: p75,
        threshold: thresholds.warning,
        message: `${metric} P75 超标: ${p75} > ${thresholds.warning}`,
      });
    }
  }
}

async function sendAlert(alert) {
  // 发送到 Slack / 钉钉 / 邮件
  console.warn('Performance Alert:', alert);

  // 示例：发送到 Slack
  await fetch(process.env.SLACK_WEBHOOK_URL, {
    method: 'POST',
    body: JSON.stringify({
      text: `🚨 ${alert.message}`,
      attachments: [{
        color: alert.severity === 'critical' ? 'danger' : 'warning',
        fields: [
          { title: 'Metric', value: alert.metric, short: true },
          { title: 'Value', value: String(alert.value), short: true },
          { title: 'Threshold', value: String(alert.threshold), short: true },
        ],
      }],
    }),
  });
}
```
