### 2.1 规则类型

```python
from enum import Enum
from typing import Any, List, Optional, Callable

class RuleType(Enum):
    """规则类型"""
    NOT_NULL = "not_null"              # 非空检查
    UNIQUE = "unique"                  # 唯一性检查
    RANGE = "range"                    # 范围检查
    ENUM = "enum"                      # 枚举检查
    FORMAT = "format"                  # 格式检查
    CUSTOM = "custom"                  # 自定义检查
    REFERENTIAL = "referential"        # 引用完整性检查
    AGGREGATE = "aggregate"            # 聚合检查
```

### 2.2 规则定义

```python
from dataclasses import dataclass, field
from typing import Any, List, Optional, Callable, Dict

@dataclass
class QualityRule:
    """数据质量规则"""
    name: str                           # 规则名称
    rule_type: RuleType                 # 规则类型
    column: str                         # 检查的列
    description: str                    # 规则描述
    
    # 规则参数
    params: Dict[str, Any] = field(default_factory=dict)
    
    # 严重程度
    severity: str = "error"             # error, warning, info
    
    # 自定义检查函数
    check_func: Optional[Callable] = None

# 规则工厂
class RuleFactory:
    """规则工厂"""
    
    @staticmethod
    def not_null(column: str, severity: str = "error") -> QualityRule:
        """非空规则"""
        return QualityRule(
            name=f"{column}_not_null",
            rule_type=RuleType.NOT_NULL,
            column=column,
            description=f"{column} 不能为空",
            severity=severity
        )
    
    @staticmethod
    def unique(column: str, severity: str = "error") -> QualityRule:
        """唯一性规则"""
        return QualityRule(
            name=f"{column}_unique",
            rule_type=RuleType.UNIQUE,
            column=column,
            description=f"{column} 必须唯一",
            severity=severity
        )
    
    @staticmethod
    def range_check(
        column: str,
        min_val: Optional[float] = None,
        max_val: Optional[float] = None,
        severity: str = "error"
    ) -> QualityRule:
        """范围检查规则"""
        return QualityRule(
            name=f"{column}_range",
            rule_type=RuleType.RANGE,
            column=column,
            description=f"{column} 必须在 {min_val}-{max_val} 范围内",
            params={"min": min_val, "max": max_val},
            severity=severity
        )
    
    @staticmethod
    def enum_check(
        column: str,
        allowed_values: List[Any],
        severity: str = "error"
    ) -> QualityRule:
        """枚举检查规则"""
        return QualityRule(
            name=f"{column}_enum",
            rule_type=RuleType.ENUM,
            column=column,
            description=f"{column} 必须是 {allowed_values} 之一",
            params={"allowed_values": allowed_values},
            severity=severity
        )
    
    @staticmethod
    def format_check(
        column: str,
        pattern: str,
        severity: str = "error"
    ) -> QualityRule:
        """格式检查规则"""
        return QualityRule(
            name=f"{column}_format",
            rule_type=RuleType.FORMAT,
            column=column,
            description=f"{column} 格式不正确",
            params={"pattern": pattern},
            severity=severity
        )
```

## 三、质量检查器实现

### 3.1 基本检查器

```python
import pandas as pd
import re
from typing import List, Dict, Any, Optional

class QualityChecker:
    """数据质量检查器"""
    
    def __init__(self, rules: List[QualityRule]):
        self.rules = rules
    
    def check(self, data: pd.DataFrame) -> Dict[str, Any]:
        """执行所有规则检查"""
        results = {
            "total_records": len(data),
            "passed": 0,
            "failed": 0,
            "checks": []
        }
        
        for rule in self.rules:
            check_result = self._execute_rule(data, rule)
            results["checks"].append(check_result)
            
            if check_result["passed"]:
                results["passed"] += 1
            else:
                results["failed"] += 1
        
        return results
    
    def _execute_rule(self, data: pd.DataFrame, rule: QualityRule) -> Dict[str, Any]:
        """执行单个规则"""
        
        if rule.rule_type == RuleType.NOT_NULL:
            return self._check_not_null(data, rule)
        elif rule.rule_type == RuleType.UNIQUE:
            return self._check_unique(data, rule)
        elif rule.rule_type == RuleType.RANGE:
            return self._check_range(data, rule)
        elif rule.rule_type == RuleType.ENUM:
            return self._check_enum(data, rule)
        elif rule.rule_type == RuleType.FORMAT:
            return self._check_format(data, rule)
        elif rule.rule_type == RuleType.CUSTOM and rule.check_func:
            return self._check_custom(data, rule)
        else:
            return {
                "rule": rule.name,
                "passed": True,
                "message": "规则类型未实现"
            }
    
    def _check_not_null(self, data: pd.DataFrame, rule: QualityRule) -> Dict[str, Any]:
        """检查非空"""
        if rule.column not in data.columns:
            return {
                "rule": rule.name,
                "passed": False,
                "message": f"列 {rule.column} 不存在"
            }
        
        null_count = data[rule.column].isnull().sum()
        null_rate = null_count / len(data)
        
        return {
            "rule": rule.name,
            "passed": null_count == 0,
            "severity": rule.severity,
            "message": f"{rule.column} 空值数量: {null_count} ({null_rate:.2%})",
            "details": {
                "null_count": int(null_count),
                "null_rate": float(null_rate)
            }
        }
    
    def _check_unique(self, data: pd.DataFrame, rule: QualityRule) -> Dict[str, Any]:
        """检查唯一性"""
        if rule.column not in data.columns:
            return {
                "rule": rule.name,
                "passed": False,
                "message": f"列 {rule.column} 不存在"
            }
        
        duplicate_count = data[rule.column].duplicated().sum()
        
        return {
            "rule": rule.name,
            "passed": duplicate_count == 0,
            "severity": rule.severity,
            "message": f"{rule.column} 重复数量: {duplicate_count}",
            "details": {
                "duplicate_count": int(duplicate_count)
            }
        }
    
    def _check_range(self, data: pd.DataFrame, rule: QualityRule) -> Dict[str, Any]:
        """检查范围"""
        if rule.column not in data.columns:
            return {
                "rule": rule.name,
                "passed": False,
                "message": f"列 {rule.column} 不存在"
            }
        
        min_val = rule.params.get("min")
        max_val = rule.params.get("max")
        
        series = data[rule.column].dropna()
        violations = pd.Series(False, index=series.index)
        
        if min_val is not None:
            violations = violations | (series < min_val)
        
        if max_val is not None:
            violations = violations | (series > max_val)
        
        violation_count = violations.sum()
        
        return {
            "rule": rule.name,
            "passed": violation_count == 0,
            "severity": rule.severity,
            "message": f"{rule.column} 范围违规数量: {violation_count}",
            "details": {
                "violation_count": int(violation_count),
                "min_val": min_val,
                "max_val": max_val
            }
        }
    
    def _check_enum(self, data: pd.DataFrame, rule: QualityRule) -> Dict[str, Any]:
        """检查枚举值"""
        if rule.column not in data.columns:
            return {
                "rule": rule.name,
                "passed": False,
                "message": f"列 {rule.column} 不存在"
            }
        
        allowed_values = rule.params.get("allowed_values", [])
        series = data[rule.column].dropna()
        
        invalid_mask = ~series.isin(allowed_values)
        invalid_count = invalid_mask.sum()
        
        invalid_values = series[invalid_mask].unique().tolist()
        
        return {
            "rule": rule.name,
            "passed": invalid_count == 0,
            "severity": rule.severity,
            "message": f"{rule.column} 无效值数量: {invalid_count}",
            "details": {
                "invalid_count": int(invalid_count),
                "invalid_values": invalid_values[:10]  # 最多显示10个
            }
        }
    
    def _check_format(self, data: pd.DataFrame, rule: QualityRule) -> Dict[str, Any]:
        """检查格式"""
        if rule.column not in data.columns:
            return {
                "rule": rule.name,
                "passed": False,
                "message": f"列 {rule.column} 不存在"
            }
        
        pattern = rule.params.get("pattern", "")
        series = data[rule.column].dropna().astype(str)
        
        invalid_mask = ~series.str.match(pattern)
        invalid_count = invalid_mask.sum()
        
        return {
            "rule": rule.name,
            "passed": invalid_count == 0,
            "severity": rule.severity,
            "message": f"{rule.column} 格式违规数量: {invalid_count}",
            "details": {
                "invalid_count": int(invalid_count)
            }
        }
    
    def _check_custom(self, data: pd.DataFrame, rule: QualityRule) -> Dict[str, Any]:
        """自定义检查"""
        try:
            result = rule.check_func(data)
            return {
                "rule": rule.name,
                "passed": result["passed"],
                "severity": rule.severity,
                "message": result.get("message", ""),
                "details": result.get("details", {})
            }
        except Exception as e:
            return {
                "rule": rule.name,
                "passed": False,
                "severity": "error",
                "message": f"自定义检查异常: {str(e)}"
            }
```

## 四、不合格数据处理

### 4.1 处理策略

```
┌──────────────────────────────────────────────────────────────┐
│                    不合格数据处理策略                          │
├──────────────────────────────────────────────────────────────┤
│                                                              │
│  策略 1：隔离（QUARANTINE）                                   │
│  ├── 将不合格数据移到隔离表                                  │
│  ├── 不影响正常数据                                          │
│  └── 后续人工处理或修复                                      │
│                                                              │
│  策略 2：修复（FIX）                                          │
│  ├── 自动修复可修复的问题                                    │
│  ├── 填充默认值                                              │
│  └── 格式标准化                                              │
│                                                              │
│  策略 3：丢弃（DISCARD）                                      │
│  ├── 直接删除不合格数据                                      │
│  ├── 适用：数据量大、问题严重                                │
│  └── 注意：可能丢失有用信息                                  │
│                                                              │
│  策略 4：标记（MARK）                                         │
│  ├── 保留数据，添加质量标记                                  │
│  ├── 下游根据标记处理                                        │
│  └── 适用：需要保留原始数据                                  │
│                                                              │
└──────────────────────────────────────────────────────────────┘
```

### 4.2 隔离实现

```python
class QualityDataHandler:
    """质量数据处理器"""
    
    def __init__(self, engine):
        self.engine = engine
    
    def quarantine(
        self,
        data: pd.DataFrame,
        quality_result: Dict[str, Any],
        quarantine_table: str = "quarantine_data"
    ) -> Dict[str, Any]:
        """隔离不合格数据"""
        
        # 找出有问题的记录
        invalid_records = self._find_invalid_records(data, quality_result)
        
        if invalid_records.empty:
            return {"quarantined": 0, "message": "没有不合格数据"}
        
        # 添加元信息
        invalid_records["quarantine_time"] = pd.Timestamp.now()
        invalid_records["quarantine_reason"] = self._get_quarantine_reason(quality_result)
        
        # 写入隔离表
        invalid_records.to_sql(
            quarantine_table,
            self.engine,
            if_exists="append",
            index=False
        )
        
        return {
            "quarantined": len(invalid_records),
            "message": f"已隔离 {len(invalid_records)} 条记录"
        }
    
    def _find_invalid_records(
        self,
        data: pd.DataFrame,
        quality_result: Dict[str, Any]
    ) -> pd.DataFrame:
        """找出不合格记录"""
        
        invalid_mask = pd.Series(False, index=data.index)
        
        for check in quality_result.get("checks", []):
            if not check["passed"] and check.get("severity") == "error":
                # 根据规则类型标记不合格记录
                rule_name = check["rule"]
                
                if "not_null" in rule_name:
                    column = rule_name.replace("_not_null", "")
                    invalid_mask = invalid_mask | data[column].isnull()
                
                elif "unique" in rule_name:
                    column = rule_name.replace("_unique", "")
                    invalid_mask = invalid_mask | data[column].duplicated(keep=False)
        
        return data[invalid_mask]
    
    def _get_quarantine_reason(self, quality_result: Dict[str, Any]) -> str:
        """获取隔离原因"""
        failed_rules = [
            check["rule"] for check in quality_result.get("checks", [])
            if not check["passed"]
        ]
        return ", ".join(failed_rules)
```

## 五、数据质量报告

### 5.1 报告生成

```python
from datetime import datetime
from typing import Dict, Any

class QualityReporter:
    """数据质量报告生成器"""
    
    def generate_report(
        self,
        data: pd.DataFrame,
        quality_result: Dict[str, Any],
        batch_id: str = None
    ) -> Dict[str, Any]:
        """生成质量报告"""
        
        report = {
            "report_time": datetime.now().isoformat(),
            "batch_id": batch_id,
            "summary": {
                "total_records": quality_result["total_records"],
                "passed_rules": quality_result["passed"],
                "failed_rules": quality_result["failed"],
                "pass_rate": quality_result["passed"] / (quality_result["passed"] + quality_result["failed"])
            },
            "details": quality_result["checks"],
            "column_stats": self._calculate_column_stats(data),
            "recommendations": self._generate_recommendations(quality_result)
        }
        
        return report
    
    def _calculate_column_stats(self, data: pd.DataFrame) -> Dict[str, Any]:
        """计算列统计信息"""
        stats = {}
        
        for column in data.columns:
            col_stats = {
                "total_count": len(data),
                "null_count": int(data[column].isnull().sum()),
                "null_rate": float(data[column].isnull().mean()),
                "unique_count": int(data[column].nunique())
            }
            
            # 数值列额外统计
            if pd.api.types.is_numeric_dtype(data[column]):
                col_stats.update({
                    "mean": float(data[column].mean()) if not data[column].isnull().all() else None,
                    "min": float(data[column].min()) if not data[column].isnull().all() else None,
                    "max": float(data[column].max()) if not data[column].isnull().all() else None,
                })
            
            stats[column] = col_stats
        
        return stats
    
    def _generate_recommendations(self, quality_result: Dict[str, Any]) -> List[str]:
        """生成改进建议"""
        recommendations = []
        
        for check in quality_result.get("checks", []):
            if not check["passed"]:
                if check["severity"] == "error":
                    recommendations.append(f"必须修复: {check['message']}")
                elif check["severity"] == "warning":
                    recommendations.append(f"建议修复: {check['message']}")
        
        return recommendations
    
    def print_report(self, report: Dict[str, Any]):
        """打印报告"""
        print("=" * 60)
        print("数据质量报告")
        print("=" * 60)
        print(f"报告时间: {report['report_time']}")
        if report.get('batch_id'):
            print(f"批次号: {report['batch_id']}")
        print()
        
        summary = report["summary"]
        print("【摘要】")
        print(f"  总记录数: {summary['total_records']}")
        print(f"  通过规则: {summary['passed_rules']}")
        print(f"  失败规则: {summary['failed_rules']}")
        print(f"  通过率: {summary['pass_rate']:.2%}")
        print()
        
        print("【规则检查详情】")
        for check in report["details"]:
            status = "✓" if check["passed"] else "✗"
            severity = check.get("severity", "").upper()
            print(f"  {status} [{severity}] {check['rule']}: {check['message']}")
        print()
        
        if report["recommendations"]:
            print("【改进建议】")
            for rec in report["recommendations"]:
                print(f"  - {rec}")
        
        print("=" * 60)
```

## 六、完整的质量检查流程

```python
class DataQualityPipeline:
    """数据质量检查管道"""
    
    def __init__(self, engine, rules: List[QualityRule]):
        self.engine = engine
        self.checker = QualityChecker(rules)
        self.handler = QualityDataHandler(engine)
        self.reporter = QualityReporter()
    
    def run(
        self,
        data: pd.DataFrame,
        batch_id: str = None,
        quarantine: bool = True
    ) -> Dict[str, Any]:
        """运行质量检查"""
        
        # 1. 执行检查
        quality_result = self.checker.check(data)
        
        # 2. 隔离不合格数据
        quarantine_result = None
        if quarantine and quality_result["failed"] > 0:
            quarantine_result = self.handler.quarantine(data, quality_result)
        
        # 3. 生成报告
        report = self.reporter.generate_report(data, quality_result, batch_id)
        
        # 4. 返回结果
        return {
            "quality_result": quality_result,
            "quarantine_result": quarantine_result,
            "report": report,
            "passed": quality_result["failed"] == 0
        }

    RuleFactory.not_null("job_id"),
    RuleFactory.not_null("title"),
    RuleFactory.unique("job_id"),
    RuleFactory.range_check("salary_min", min_val=0),
```

## 常见误区

- **质量规则越多越好**：规则太多会导致大量误报，真正的问题被淹没在告警噪音中。应该先覆盖关键字段和高频问题，逐步迭代。
- **所有规则都是 error 级别**：把每条规则都设为 error，一个小问题就阻断整个流程。应该区分 critical（阻断）、error（隔离）、warning（记录）、info（统计），让管道有一定的容错能力。
- **只检查不处理**：检查出问题数据后只是打个日志，不隔离、不修复、不告警，等于没检查。质量检查必须配合处理策略（隔离/修复/丢弃）才有意义。
- **规则硬编码在业务逻辑里**：质量规则应该和业务代码解耦，用配置文件或数据库管理，方便非开发人员（如数据分析师）维护和调整。

## 工程建议

1. **质量规则按维度分类管理**：按完整性、准确性、一致性等维度组织规则，方便按维度统计质量得分，也方便后续扩展。
2. **检查结果要结构化存储**：每条检查结果记录规则名、通过/失败、严重程度、违规数量、违规样例，方便生成报告和趋势分析。
3. **先跑 dry-run 再正式执行**：新规则上线前先跑 dry-run 模式，统计有多少数据会被拦截，评估规则的合理性，避免误杀正常数据。
4. **质量报告定期生成并推送**：每天/每批次生成质量报告，推送到钉钉/飞书/邮件，让团队对数据质量有持续感知，而不是等问题爆发才发现。
