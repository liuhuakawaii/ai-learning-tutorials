## 二、自动化流程

```python
# scheduler.py
from apscheduler.schedulers.background import BackgroundScheduler

scheduler = BackgroundScheduler()

# 每天早上 8 点执行 ETL
@scheduler.scheduled_job('cron', hour=8, minute=0)
def daily_etl():
    """每日 ETL 任务"""
    pipeline.run()
    check_quality()
    aggregate_metrics()

# 每小时增量更新
@scheduler.scheduled_job('interval', hours=1)
def hourly_update():
    """每小时增量更新"""
    incremental_crawl()

scheduler.start()
```

## 三、监控面板

```python
# 监控 API
@router.get("/api/monitor/status")
async def get_system_status():
    """系统状态"""
    return {
        "database": check_db_connection(),
        "redis": check_redis_connection(),
        "last_etl": get_last_etl_status(),
        "data_freshness": get_data_freshness(),
        "alert_count": get_recent_alerts_count()
    }

@router.get("/api/monitor/tasks")
async def get_task_overview():
    """任务概览"""
    return {
        "running": count_tasks("running"),
        "completed_today": count_tasks("completed", today=True),
        "failed_today": count_tasks("failed", today=True)
    }
```
