## 一、完整 CI/CD 流水线

```yaml
name: CI/CD Pipeline

on:
  push:
    branches: [main]
    tags: ['v*']
  pull_request:
    branches: [main]

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}

jobs:
  # ---- CI：检查和测试 ----
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 18
          cache: npm
      - run: npm ci
      - run: npm run lint
      - run: npm test
      - run: npm run build

  # ---- CD：构建和推送镜像 ----
  build:
    needs: test
    runs-on: ubuntu-latest
    if: github.event_name == 'push'
    
    permissions:
      contents: read
      packages: write
    
    outputs:
      image-tag: ${{ steps.meta.outputs.version }}
    
    steps:
      - uses: actions/checkout@v4
      
      - name: Login to GHCR
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}
      
      - name: Extract metadata
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}
          tags: |
            type=ref,event=branch
            type=semver,pattern={{version}}
            type=sha
      
      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3
      
      - name: Build and push
        uses: docker/build-push-action@v5
        with:
          context: .
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          cache-from: type=gha
          cache-to: type=gha,mode=max

  # ---- CD：部署 ----
  deploy:
    needs: build
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'
    environment: production
    
    steps:
      - name: Deploy to server
        uses: appleboy/ssh-action@v1
        with:
          host: ${{ secrets.SERVER_HOST }}
          username: ${{ secrets.SERVER_USER }}
          key: ${{ secrets.SERVER_SSH_KEY }}
          script: |
            cd /opt/my-app
            docker compose pull
            docker compose up -d
```

## 二、部署方式

### 2.1 SSH 部署到 VPS

```yaml
deploy:
  needs: build
  runs-on: ubuntu-latest
  steps:
    - name: Deploy via SSH
      uses: appleboy/ssh-action@v1
      with:
        host: ${{ secrets.SERVER_HOST }}
        username: ${{ secrets.SERVER_USER }}
        key: ${{ secrets.SERVER_SSH_KEY }}
        script: |
          cd /opt/my-app
          git pull
          docker compose pull
          docker compose up -d --build
          docker compose ps
```

### 2.2 使用部署脚本

```yaml
deploy:
  needs: build
  runs-on: ubuntu-latest
  steps:
    - uses: actions/checkout@v4
    
    - name: Deploy
      env:
        SSH_PRIVATE_KEY: ${{ secrets.SSH_PRIVATE_KEY }}
        SERVER_HOST: ${{ secrets.SERVER_HOST }}
      run: |
        chmod +x scripts/deploy.sh
        ./scripts/deploy.sh
```

```bash
#!/bin/bash
# scripts/deploy.sh

set -e

echo "Deploying to production..."

# 配置 SSH
mkdir -p ~/.ssh
echo "$SSH_PRIVATE_KEY" > ~/.ssh/id_rsa
chmod 600 ~/.ssh/id_rsa
ssh-keyscan -H $SERVER_HOST >> ~/.ssh/known_hosts

# 部署
ssh deploy@$SERVER_HOST << 'EOF'
  cd /opt/my-app
  docker compose pull
  docker compose up -d
  docker compose ps
EOF

echo "Deployment completed!"
```

## 三、环境和审批

```yaml
deploy-staging:
  needs: build
  runs-on: ubuntu-latest
  environment: staging
  steps:
    - run: echo "Deploy to staging"

deploy-production:
  needs: deploy-staging
  runs-on: ubuntu-latest
  environment:
    name: production
    url: https://my-app.com
  steps:
    - run: echo "Deploy to production"
```

```
在 GitHub 中配置环境：

  1. 进入仓库 Settings → Environments
  2. 创建 environment（如 production）
  3. 设置保护规则：
     - 需要审批
     - 限制部署分支
     - 等待定时器
```

## 四、回滚策略

### 4.1 使用 Git 标签回滚

```bash
# 回滚到上一个版本
git revert HEAD
git push

# 或者使用标签
git checkout v1.0.0
git push origin main --force  # 危险！不推荐
```

### 4.2 使用 Docker 镜像回滚

```bash
# 在服务器上回滚到指定版本
docker compose pull my-app:v1.0.0
# 修改 compose.yml 使用旧版本
docker compose up -d
```

### 4.3 自动回滚脚本

```bash
#!/bin/bash
# scripts/rollback.sh

PREVIOUS_VERSION=$1

if [ -z "$PREVIOUS_VERSION" ]; then
  echo "Usage: ./rollback.sh <version>"
  exit 1
fi

echo "Rolling back to version $PREVIOUS_VERSION..."

# 更新镜像版本
export IMAGE_TAG=$PREVIOUS_VERSION
docker compose pull
docker compose up -d

echo "Rollback completed!"
```

## 五、通知

```yaml
notify:
  needs: [build, deploy]
  runs-on: ubuntu-latest
  if: always()
  steps:
    - name: Notify on success
      if: needs.deploy.result == 'success'
      run: echo "Deployment successful!"
    
    - name: Notify on failure
      if: needs.deploy.result == 'failure'
      run: echo "Deployment failed!"
```

```yaml

          context: .

```

## 小结

1. **完整流水线**：test → build → deploy
2. **部署方式**：SSH、deploy 脚本、云平台 API
3. **环境保护**：使用 environment 控制部署权限
4. **回滚**：使用 Git 标签或 Docker 镜像版本
