### 3.2 攻击树构建工具

```python
from dataclasses import dataclass, field
from typing import List, Optional, Dict
from enum import Enum

class NodeType(Enum):
    AND = "and"  # 所有子节点都需要成功
    OR = "or"    # 任一子节点成功即可

@dataclass
class AttackNode:
    name: str
    node_type: NodeType
    probability: float  # 0-1
    impact: float       # 0-1
    children: List['AttackNode'] = field(default_factory=list)
    description: str = ""
    mitigation: str = ""

    @property
    def risk_score(self) -> float:
        return self.probability * self.impact

class AttackTree:
    def __init__(self, root: AttackNode):
        self.root = root

    def calculate_success_probability(self, node: Optional[AttackNode] = None) -> float:
        if node is None:
            node = self.root

        if not node.children:
            return node.probability

        child_probs = [self.calculate_success_probability(c) for c in node.children]

        if node.node_type == NodeType.AND:
            # 所有都需要成功
            result = 1.0
            for p in child_probs:
                result *= p
            return result
        else:
            # 任一成功即可
            fail_prob = 1.0
            for p in child_probs:
                fail_prob *= (1 - p)
            return 1 - fail_prob

    def find_critical_paths(self, node: Optional[AttackNode] = None, path: List[str] = None) -> List[Dict]:
        if node is None:
            node = self.root
        if path is None:
            path = []

        current_path = path + [node.name]
        results = []

        if not node.children:
            results.append({
                "path": current_path,
                "probability": self.calculate_success_probability(node),
                "risk_score": node.risk_score
            })
        else:
            for child in node.children:
                results.extend(self.find_critical_paths(child, current_path))

        return sorted(results, key=lambda x: x["risk_score"], reverse=True)

    def generate_report(self) -> str:
        report = "# 攻击树分析报告\n\n"
        report += f"## 根目标: {self.root.name}\n"
        report += f"## 整体成功概率: {self.calculate_success_probability():.2%}\n\n"

        report += "## 关键攻击路径:\n"
        for i, path in enumerate(self.find_critical_paths()[:5], 1):
            report += f"\n### 路径 {i}\n"
            report += f"- 路径: {' → '.join(path['path'])}\n"
            report += f"- 成功概率: {path['probability']:.2%}\n"
            report += f"- 风险评分: {path['risk_score']:.2f}\n"

        return report


# 使用示例
tree = AttackTree(
    AttackNode(
        name="获取系统 Prompt",
        node_type=NodeType.OR,
        probability=0.8,
        impact=0.9,
        children=[
            AttackNode(
                name="直接询问",
                node_type=NodeType.AND,
                probability=0.1,
                impact=0.9,
                children=[
                    AttackNode("发送直接请求", NodeType.AND, 1.0, 0.9),
                    AttackNode("绕过防御", NodeType.AND, 0.1, 0.9)
                ]
            ),
            AttackNode(
                name="Prompt 注入",
                node_type=NodeType.AND,
                probability=0.6,
                impact=0.9,
                children=[
                    AttackNode("构造注入 payload", NodeType.AND, 0.8, 0.9),
                    AttackNode("绕过输入过滤", NodeType.AND, 0.7, 0.9),
                    AttackNode("模型执行注入指令", NodeType.AND, 0.8, 0.9)
                ]
            ),
            AttackNode(
                name="间接泄露",
                node_type=NodeType.AND,
                probability=0.3,
                impact=0.9,
                children=[
                    AttackNode("RAG 包含 Prompt", NodeType.AND, 0.4, 0.9),
                    AttackNode("触发检索", NodeType.AND, 0.7, 0.9)
                ]
            )
        ]
    )

from typing import Dict, List

class STRIDEThreatModeler:
    """基于 STRIDE 的 AI 威胁建模"""

    def __init__(self):
        self.threat_categories = {

                    "能否冒充合法用户?",

                    "能否欺骗 RAG 检索?"

            },
            "tampering": {
                "name": "篡改",

                    "模型权重能否被修改?",
                    "Prompt 能否被篡改?",
                    "输出能否被中间人修改?"
                ]
            },
            "repudiation": {
                "name": "抵赖",
                "questions": [
                    "AI 生成内容能否溯源?",
                    "操作是否有完整审计?",
                    "用户行为能否否认?",
                    "决策依据是否可追溯?"
                ]
            },
            "information_disclosure": {
                "name": "信息泄露",
                "questions": [
                    "训练数据能否被提取?",
                    "系统 Prompt 能否泄露?",
                    "用户隐私能否暴露?",
                    "模型架构能否被逆向?"
                ]
            },
            "denial_of_service": {
                "name": "拒绝服务",
                "questions": [
                    "能否耗尽计算资源?",
                    "能否触发无限循环?",
                    "能否超出 API 配额?",
                    "能否让模型拒绝正常请求?"
                ]
            },
            "elevation_of_privilege": {
                "name": "权限提升",
                "questions": [
                    "能否绕过访问控制?",
                    "能否获取未授权功能?",
                    "能否执行系统命令?",
                    "能否访问其他用户数据?"
                ]
            }
        }

    def analyze_component(self, component_name: str, component_type: str) -> Dict:
        """分析单个组件的 STRIDE 威胁"""
        results = {}

        for category, info in self.threat_categories.items():
            threats = []
            for question in info["questions"]:
                # 这里可以接入实际的检查逻辑
                threat = {
                    "question": question,
                    "applicable": True,  # 简化示例
                    "risk_level": "MEDIUM",
                    "mitigation": "需要具体分析"
                }
                threats.append(threat)

            results[category] = {
                "name": info["name"],
                "threats": threats
            }

        return {
            "component": component_name,
            "type": component_type,
            "threats": results
        }

    def generate_report(self, analysis: Dict) -> str:
        report = f"# STRIDE 威胁分析: {analysis['component']}\n\n"
        report += f"组件类型: {analysis['type']}\n\n"

        for category, data in analysis["threats"].items():
            report += f"## {data['name']} ({category})\n"
            for threat in data["threats"]:
                status = "⚠️" if threat["applicable"] else "✅"
                report += f"- {status} {threat['question']}\n"
                if threat["applicable"]:
                    report += f"  - 风险等级: {threat['risk_level']}\n"
                    report += f"  - 缓解措施: {threat['mitigation']}\n"
            report += "\n"

        return report
```

### 4.2 自动化攻击面扫描

```python
import re
from typing import List, Dict
from dataclasses import dataclass

@dataclass
class Vulnerability:
    name: str
    severity: str
    location: str
    description: str
    recommendation: str

class AttackSurfaceScanner:
    """攻击面自动扫描器"""

    def __init__(self):
        self.vulnerabilities: List[Vulnerability] = []

    def scan_code(self, code: str, file_path: str) -> List[Vulnerability]:
        """扫描代码中的安全问题"""
        self.vulnerabilities = []

        # 检测不安全的 prompt 构造
        if re.search(r'f["\'].*\{.*\}.*user_input', code):
            self.vulnerabilities.append(Vulnerability(
                name="不安全的 Prompt 构造",
                severity="HIGH",
                location=file_path,
                description="检测到使用 f-string 直接拼接用户输入到 Prompt",
                recommendation="使用结构化 Prompt 模板，分离指令和数据"
            ))

        # 检测 SQL 注入风险
        if re.search(r'execute\(f["\'].*\{', code):
            self.vulnerabilities.append(Vulnerability(
                name="SQL 注入风险",
                severity="CRITICAL",
                location=file_path,
                description="检测到字符串拼接的 SQL 查询",
                recommendation="使用参数化查询"
            ))

        # 检测命令注入风险
        if re.search(r'os\.(system|popen|exec)', code):
            self.vulnerabilities.append(Vulnerability(
                name="命令注入风险",
                severity="CRITICAL",
                location=file_path,
                description="检测到直接执行系统命令",
                recommendation="避免执行用户可控的命令，使用白名单验证"
            ))

        # 检测硬编码密钥
        if re.search(r'(api_key|secret|password)\s*=\s*["\'][^"\']+["\']', code, re.I):
            self.vulnerabilities.append(Vulnerability(
                name="硬编码凭证",
                severity="HIGH",
                location=file_path,
                description="检测到硬编码的 API 密钥或密码",
                recommendation="使用环境变量或密钥管理服务"
            ))

        # 检测缺少输入验证
        if 'def ' in code and 'request.' in code and 'validate' not in code.lower():
            self.vulnerabilities.append(Vulnerability(
                name="缺少输入验证",
                severity="MEDIUM",
                location=file_path,
                description="检测到处理用户请求但缺少输入验证",
                recommendation="添加输入验证和清理逻辑"
            ))

        return self.vulnerabilities

    def generate_report(self) -> str:
        report = "# 攻击面扫描报告\n\n"
        report += f"发现 {len(self.vulnerabilities)} 个潜在问题\n\n"

        by_severity = {}
        for v in self.vulnerabilities:
            by_severity.setdefault(v.severity, []).append(v)

        for severity in ["CRITICAL", "HIGH", "MEDIUM", "LOW"]:
            vulns = by_severity.get(severity, [])
            if vulns:
                report += f"## {severity} 级别 ({len(vulns)})\n"
                for v in vulns:
                    report += f"\n### {v.name}\n"
                    report += f"- 位置: {v.location}\n"
                    report += f"- 描述: {v.description}\n"
                    report += f"- 建议: {v.recommendation}\n"

        return report
```

## 6. 常见误区

1. **"只关注 LLM 本身"**: 应用层、数据层同样重要
2. **"攻击面是静态的"**: 随着功能迭代，攻击面持续变化
3. **"威胁建模一次就够"**: 需要在每个迭代周期重新评估
4. **"自动化工具能覆盖一切"**: 需要结合人工分析和专家经验
5. **"只关注技术风险"**: 流程和人员也是攻击面的一部分

## 总结

- 攻击面分析是 AI 安全的基础工作
- 需要系统性地识别各层的攻击入口点
- 攻击树和 STRIDE 模型是有效的分析工具
- 需要持续更新和重新评估攻击面

## 练习

### 练习 1: 攻击面识别
为你正在开发的 AI 应用画出完整的攻击面图，标注每个入口点的风险等级。

### 练习 2: 攻击树构建
选择一个攻击目标（如"获取用户数据"），构建完整的攻击树。

### 练习 3: 威胁建模
使用 STRIDE 模型对一个 RAG 应用进行威胁建模，输出分析报告。
