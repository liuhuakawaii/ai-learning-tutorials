### 2.2 合规差距分析工具

```python
from typing import Dict, List
from dataclasses import dataclass

@dataclass
class ComplianceGap:
    requirement: str
    current_status: str
    gap_description: str
    risk_level: str
    remediation: str
    deadline: str

class ComplianceAnalyzer:
    """合规差距分析器"""

    def __init__(self):
        self.requirements = self._load_requirements()

    def _load_requirements(self) -> Dict:
        return {
            "生成式AI": {
                "R001": {
                    "name": "算法备案",
                    "description": "具有舆论属性或者社会动员能力的算法推荐服务提供者应当办理备案",
                    "check_method": "检查是否完成算法备案",
                    "evidence": "备案编号"
                },
                "R002": {
                    "name": "内容安全审核",
                    "description": "建立内容审核机制，防止生成违法不良信息",
                    "check_method": "检查是否部署内容审核系统",
                    "evidence": "审核系统配置和日志"
                },
                "R003": {
                    "name": "用户身份认证",
                    "description": "对使用者进行真实身份信息认证",
                    "check_method": "检查是否实施实名认证",
                    "evidence": "认证流程文档"
                },
                "R004": {
                    "name": "数据来源合法",
                    "description": "训练数据应当具有合法来源",
                    "check_method": "检查数据来源审查机制",
                    "evidence": "数据来源证明文件"
                },
                "R005": {
                    "name": "内容标识",
                    "description": "对 AI 生成内容进行标识",
                    "check_method": "检查是否实施内容标识",
                    "evidence": "标识实现代码和示例"
                }
            }
        }

    def analyze(self, system_config: Dict) -> List[ComplianceGap]:
        gaps = []

        for category, reqs in self.requirements.items():
            for req_id, req in reqs.items():
                # 检查是否满足要求
                is_compliant = self._check_compliance(req_id, system_config)

                if not is_compliant:
                    gap = ComplianceGap(
                        requirement=f"[{req_id}] {req['name']}",
                        current_status="未满足",
                        gap_description=req["description"],
                        risk_level=self._assess_risk(req_id),
                        remediation=self._suggest_remediation(req_id),
                        deadline=self._suggest_deadline(req_id)
                    )
                    gaps.append(gap)

        return gaps

    def _check_compliance(self, req_id: str, config: Dict) -> bool:
        # 简化的合规检查
        compliance_status = config.get("compliance", {})
        return compliance_status.get(req_id, False)

    def _assess_risk(self, req_id: str) -> str:
        high_risk = ["R001", "R002", "R003"]
        if req_id in high_risk:
            return "HIGH"
        return "MEDIUM"

    def _suggest_remediation(self, req_id: str) -> str:
        remediations = {
            "R001": "联系当地网信部门完成算法备案",
            "R002": "部署内容审核系统，配置敏感词库和分类模型",
            "R003": "接入实名认证服务，实施用户身份验证",
            "R004": "建立数据来源审查机制，保留数据来源证明",
            "R005": "在 AI 生成内容中添加标识水印"
        }
        return remediations.get(req_id, "待评估")

    def _suggest_deadline(self, req_id: str) -> str:
        return "建议 30 天内完成"

    def generate_report(self, gaps: List[ComplianceGap]) -> str:
        report = "# 合规差距分析报告\n\n"
        report += f"## 发现 {len(gaps)} 个合规差距\n\n"

        for gap in gaps:
            report += f"### {gap.requirement}\n"
            report += f"- 风险等级: {gap.risk_level}\n"
            report += f"- 差距描述: {gap.gap_description}\n"
            report += f"- 整改建议: {gap.remediation}\n"
            report += f"- 建议完成: {gap.deadline}\n\n"

        return report
```

```python
                ]
            }

        for category, check in self.checks.items():
            results[category] = {
                "name": check["name"],
                "questions": []
            }

            for question in check["questions"]:
                # 这里可以接入实际检查逻辑
                passed = self._check_question(question, system_config)
                results[category]["questions"].append({

                    "passed": passed,

        # 简化的检查逻辑

            for q in data["questions"]:

                report += f"- {status} {q['question']}\n"
                if not q["passed"] and q["recommendation"]:
                    report += f"  - 建议: {q['recommendation']}\n"
            report += "\n"

        return report
```

### 4.2 合规自动化工具

```python
from datetime import datetime, timedelta
from typing import List, Dict
from dataclasses import dataclass

@dataclass
class ComplianceTask:
    task_id: str
    regulation: str
    requirement: str
    deadline: datetime
    status: str
    assignee: str
    evidence: List[str]

class ComplianceManager:
    """合规管理工具"""

    def __init__(self):
        self.tasks: List[ComplianceTask] = []
        self.regulations = {}
        self.audit_log = []

    def add_regulation(self, name: str, requirements: List[Dict]):
        self.regulations[name] = requirements

    def create_compliance_task(self, regulation: str, requirement: str,
                                deadline_days: int = 30) -> ComplianceTask:
        task = ComplianceTask(
            task_id=f"COMP-{len(self.tasks)+1:04d}",
            regulation=regulation,
            requirement=requirement,
            deadline=datetime.now() + timedelta(days=deadline_days),
            status="OPEN",
            assignee="",
            evidence=[]
        )
        self.tasks.append(task)
        self.audit_log.append({
            "timestamp": datetime.now(),
            "action": "task_created",
            "task_id": task.task_id
        })
        return task

    def update_task_status(self, task_id: str, status: str, evidence: List[str] = None):
        for task in self.tasks:
            if task.task_id == task_id:
                task.status = status
                if evidence:
                    task.evidence.extend(evidence)
                self.audit_log.append({
                    "timestamp": datetime.now(),
                    "action": "task_updated",
                    "task_id": task_id,
                    "new_status": status
                })
                break

    def get_overdue_tasks(self) -> List[ComplianceTask]:
        now = datetime.now()
        return [t for t in self.tasks if t.deadline < now and t.status != "COMPLETED"]

    def generate_compliance_dashboard(self) -> Dict:
        total = len(self.tasks)
        completed = sum(1 for t in self.tasks if t.status == "COMPLETED")
        in_progress = sum(1 for t in self.tasks if t.status == "IN_PROGRESS")
        overdue = len(self.get_overdue_tasks())

        return {
            "total_tasks": total,
            "completed": completed,
            "in_progress": in_progress,
            "overdue": overdue,
            "compliance_rate": (completed / total * 100) if total > 0 else 0,
            "by_regulation": self._group_by_regulation()
        }

    def _group_by_regulation(self) -> Dict:
        groups = {}
        for task in self.tasks:
            if task.regulation not in groups:
                groups[task.regulation] = {"total": 0, "completed": 0}
            groups[task.regulation]["total"] += 1
            if task.status == "COMPLETED":
                groups[task.regulation]["completed"] += 1
        return groups

    def generate_report(self) -> str:
        dashboard = self.generate_compliance_dashboard()

        report = "# 合规管理报告\n\n"
        report += f"## 总体状况\n"
        report += f"- 总任务数: {dashboard['total_tasks']}\n"
        report += f"- 已完成: {dashboard['completed']}\n"
        report += f"- 进行中: {dashboard['in_progress']}\n"
        report += f"- 已逾期: {dashboard['overdue']}\n"
        report += f"- 合规率: {dashboard['compliance_rate']:.1f}%\n\n"

        report += "## 按法规分类\n"
        for reg, stats in dashboard['by_regulation'].items():
            rate = stats['completed'] / stats['total'] * 100 if stats['total'] > 0 else 0
            report += f"- {reg}: {stats['completed']}/{stats['total']} ({rate:.0f}%)\n"

        report += "\n## 逾期任务\n"
        for task in self.get_overdue_tasks():
            report += f"- [{task.task_id}] {task.requirement} "
            report += f"(截止: {task.deadline.strftime('%Y-%m-%d')})\n"

        return report
```

```python
            "是否完成算法备案",

            "是否实施用户实名认证",
            "是否对 AI 生成内容进行标识",
            "是否建立投诉举报机制",
            "是否完成安全评估",
            "是否建立数据安全管理制度",
            "是否签订数据处理协议"
        ],
        "数据处理": [
            "是否确定数据处理合法性基础",
            "是否发布隐私政策",
            "是否获得用户同意",
            "是否实施数据加密",
            "是否建立数据删除机制",
            "是否进行数据保护影响评估",
            "是否设置数据保护官"
        ]
    }

    results = {
        "system_type": system_type,
        "checks": [],
        "passed": 0,
        "failed": 0
    }

    for check in checklist.get(system_type, []):
        # 这里应该接入实际检查逻辑
        passed = False  # 默认未通过
        results["checks"].append({
            "item": check,
            "passed": passed,
            "recommendation": "需要实施" if not passed else "已满足"
        })
        if passed:
            results["passed"] += 1
        else:
            results["failed"] += 1

    total = results["passed"] + results["failed"]
    results["compliance_rate"] = (results["passed"] / total * 100) if total > 0 else 0

    return results
```

## 6. 常见误区

1. **"合规等于安全"**: 合规是最低要求，不等于全面安全
2. **"一次合规足够"**: 法规持续更新，需要持续合规
3. **"只关注技术合规"**: 组织流程和人员培训同样重要
4. **"小公司不需要合规"**: 法规适用于所有规模的企业
5. **"海外服务不受中国法规管辖"**: 面向境内用户就需要合规

## 总结

- 全球 AI 法规框架日趋完善，企业需要主动合规
- 中国 AI 法规强调内容安全、算法透明和数据保护
- GDPR 对 AI 系统有严格的数据处理要求
- 合规管理体系需要覆盖治理、制度、执行、监督四个层次
