```python
                ]
            },
            "audit_logging": {
                "name": "审计日志",
                "requirements": [
                    "操作日志记录",
                    "安全事件日志",
                    "日志完整性保护",
                    "日志保留策略"
                ]
            }
        }

    def analyze_project(self, project_description: str) -> dict:
        """分析项目安全需求"""
        requirements = {}

        for category, info in self.requirement_categories.items():
            requirements[category] = {
                "name": info["name"],
                "items": []
            }

            for req in info["requirements"]:
                requirement = {
                    "description": req,
                    "priority": self._assess_priority(req, project_description),
                    "implementation": self._suggest_implementation(req),
                    "verification": self._suggest_verification(req)
                }
                requirements[category]["items"].append(requirement)

        return requirements

    def _assess_priority(self, requirement: str, project_desc: str) -> str:
        # 简化的优先级评估
        high_priority = ["身份验证", "加密", "注入防护", "访问控制"]
        for keyword in high_priority:
            if keyword in requirement:
                return "HIGH"
        return "MEDIUM"

    def _suggest_implementation(self, requirement: str) -> str:
        implementations = {
            "用户身份验证机制": "使用 OAuth 2.0 / JWT",
            "API 密钥管理": "使用密钥管理服务 (KMS)",
            "数据加密": "传输使用 TLS 1.3，存储使用 AES-256",
            "Prompt 注入防护": "实现多层输入验证和指令隔离"
        }
        for key, value in implementations.items():
            if key in requirement:
                return value
        return "待定"

    def _suggest_verification(self, requirement: str) -> str:
        return "安全测试 + 代码审查"

    def generate_document(self, requirements: dict) -> str:
        doc = "# 安全需求规格说明书\n\n"

        for category, data in requirements.items():
            doc += f"## {data['name']}\n\n"
            doc += "| 需求 | 优先级 | 实现方案 | 验证方法 |\n"
            doc += "|------|--------|----------|----------|\n"

            for item in data["items"]:
                doc += f"| {item['description']} | {item['priority']} "
                doc += f"| {item['implementation']} | {item['verification']} |\n"

            doc += "\n"

        return doc
```

### 2.3 安全代理实现

```python
from typing import Optional, Dict, Callable
import time
import hashlib

class SecurityProxy:
    """AI 系统安全代理"""

    def __init__(self, llm_client):
        self.llm_client = llm_client
        self.input_validators = []
        self.output_filters = []
        self.audit_logger = AuditLogger()
        self.rate_limiter = RateLimiter()

    def add_input_validator(self, validator: Callable):
        self.input_validators.append(validator)

    def add_output_filter(self, filter_fn: Callable):
        self.output_filters.append(filter_fn)

    async def process_request(self, request: Dict) -> Dict:
        start_time = time.time()

        # 1. 身份验证
        user = self._authenticate(request)
        if not user:
            return {"error": "Authentication failed"}

        # 2. 权限检查
        if not self._authorize(user, request):
            return {"error": "Authorization failed"}

        # 3. 速率限制
        if not self.rate_limiter.is_allowed(user.id):
            return {"error": "Rate limit exceeded"}

        # 4. 输入验证
        user_input = request.get("input", "")
        for validator in self.input_validators:
            is_valid, error_msg = validator(user_input)
            if not is_valid:
                self.audit_logger.log_validation_failure(user, request, error_msg)
                return {"error": f"Input validation failed: {error_msg}"}

        # 5. 调用 LLM
        try:
            llm_response = await self.llm_client.generate(user_input)
        except Exception as e:
            self.audit_logger.log_error(user, request, str(e))
            return {"error": "Service temporarily unavailable"}

        # 6. 输出过滤
        filtered_output = llm_response
        for filter_fn in self.output_filters:
            filtered_output = filter_fn(filtered_output)

        # 7. 审计日志
        processing_time = time.time() - start_time
        self.audit_logger.log_request(
            user=user,
            request=request,
            response=filtered_output,
            processing_time=processing_time
        )

        return {"output": filtered_output}

    def _authenticate(self, request: Dict) -> Optional[Dict]:
        token = request.get("auth_token")
        if not token:
            return None
        # 验证 token 逻辑
        return {"id": "user_123", "role": "user"}

    def _authorize(self, user: Dict, request: Dict) -> bool:
        # 基于角色的权限检查
        return True


class AuditLogger:
    """审计日志记录器"""

    def log_request(self, user, request, response, processing_time):
        log_entry = {
            "timestamp": time.time(),
            "user_id": user.get("id"),
            "action": "llm_request",
            "input_hash": hashlib.sha256(str(request).encode()).hexdigest(),
            "output_length": len(str(response)),
            "processing_time": processing_time
        }
        # 持久化日志
        self._persist_log(log_entry)

    def log_validation_failure(self, user, request, reason):
        log_entry = {
            "timestamp": time.time(),
            "user_id": user.get("id"),
            "action": "validation_failure",
            "reason": reason
        }
        self._persist_log(log_entry)

    def log_error(self, user, request, error):
        log_entry = {
            "timestamp": time.time(),
            "user_id": user.get("id"),
            "action": "error",
            "error": error
        }
        self._persist_log(log_entry)

    def _persist_log(self, entry):
        # 写入日志存储
        pass


class RateLimiter:
    """速率限制器"""

    def __init__(self, max_requests=100, window=60):
        self.max_requests = max_requests
        self.window = window
        self.requests = {}

    def is_allowed(self, user_id: str) -> bool:
        now = time.time()
        if user_id not in self.requests:
            self.requests[user_id] = []

        # 清理过期记录
        self.requests[user_id] = [
            t for t in self.requests[user_id]
            if now - t < self.window
        ]

        if len(self.requests[user_id]) >= self.max_requests:
            return False

        self.requests[user_id].append(now)
        return True
```

```python
from typing import List, Dict, Callable
from dataclasses import dataclass
from enum import Enum

class TestSeverity(Enum):
    INFO = "info"
    LOW = "low"
    MEDIUM = "medium"
    HIGH = "high"
    CRITICAL = "critical"

@dataclass
class SecurityTestResult:
    test_name: str
    passed: bool
    severity: TestSeverity
    description: str
    details: str = ""
    recommendation: str = ""

class AISecurityTestSuite:
    """AI 安全测试套件"""

    def __init__(self):
        self.tests: List[Dict] = []
        self.results: List[SecurityTestResult] = []

    def add_test(self, name: str, category: str, test_fn: Callable, severity: TestSeverity):
        self.tests.append({
            "name": name,
            "category": category,
            "test_fn": test_fn,
            "severity": severity
        })

    def run_all(self) -> List[SecurityTestResult]:
        self.results = []
        for test in self.tests:
            try:
                result = test["test_fn"]()
                self.results.append(SecurityTestResult(
                    test_name=test["name"],
                    passed=result,
                    severity=test["severity"],
                    description=f"测试 {test['name']}"
                ))
            except Exception as e:
                self.results.append(SecurityTestResult(
                    test_name=test["name"],
                    passed=False,
                    severity=test["severity"],
                    description=f"测试 {test['name']} 执行失败",
                    details=str(e)
                ))
        return self.results

    def generate_report(self) -> str:
        report = "# AI 安全测试报告\n\n"

        passed = sum(1 for r in self.results if r.passed)
        failed = sum(1 for r in self.results if not r.passed)

        report += f"## 总体结果\n"
        report += f"- 通过: {passed}\n"
        report += f"- 失败: {failed}\n"
        report += f"- 通过率: {passed / len(self.results) * 100:.1f}%\n\n"

        report += "## 失败项详情\n"
        for result in self.results:
            if not result.passed:
                report += f"\n### {result.test_name}\n"
                report += f"- 严重程度: {result.severity.value}\n"
                report += f"- 描述: {result.description}\n"
                if result.details:
                    report += f"- 详情: {result.details}\n"
                if result.recommendation:
                    report += f"- 建议: {result.recommendation}\n"

        return report
```

```python
class SecurityChecklist:
    """安全开发检查清单"""

    def __init__(self):
        self.checklists = {

                ("是否进行了威胁建模", "HIGH"),

            "model": [

                ("是否检测了后门", "MEDIUM"),
                ("是否限制了模型信息暴露", "MEDIUM"),
            ],
            "application": [
                ("是否实现了输入验证", "CRITICAL"),
                ("是否实施了 Prompt 注入防护", "CRITICAL"),
                ("是否实现了输出过滤", "HIGH"),
                ("是否实施了访问控制", "HIGH"),
                ("是否记录了审计日志", "MEDIUM"),
            ],
            "deployment": [
                ("是否使用了安全配置", "HIGH"),
                ("是否启用了 HTTPS", "HIGH"),
                ("是否配置了速率限制", "MEDIUM"),
                ("是否设置了监控告警", "MEDIUM"),
            ]
        }

    def evaluate(self, phase: str, implementations: dict) -> dict:
        checklist = self.checklists.get(phase, [])
        results = {
            "phase": phase,
            "total": len(checklist),
            "passed": 0,
            "failed": 0,
            "details": []
        }

        for item, priority in checklist:
            implemented = implementations.get(item, False)
            results["details"].append({
                "item": item,
                "priority": priority,
                "implemented": implemented
            })
            if implemented:
                results["passed"] += 1
            else:
                results["failed"] += 1

        results["score"] = results["passed"] / results["total"] * 100
        return results

    def generate_report(self, all_results: list) -> str:
        report = "# 安全检查报告\n\n"

        total_score = sum(r["score"] for r in all_results) / len(all_results)
        report += f"## 总体评分: {total_score:.1f}/100\n\n"

        for result in all_results:
            report += f"### {result['phase']} ({result['score']:.0f}%)\n"
            for detail in result["details"]:
                status = "✅" if detail["implemented"] else "❌"
                report += f"- {status} [{detail['priority']}] {detail['item']}\n"
            report += "\n"

        return report
```

## 5. 常见误区

1. **"安全是最后阶段的事"**: 安全应贯穿整个生命周期
2. **"安全测试等同于功能测试"**: 需要专门的安全测试方法
3. **"一次安全评估足够"**: 需要持续的安全验证
4. **"安全会降低开发效率"**: 前期投入能减少后期成本
5. **"只关注技术安全"**: 流程和人员同样重要

## 总结

- AI 安全开发生命周期需要覆盖从需求到运维的全过程
- 每个阶段都有特定的安全活动和检查点
- 安全代理模式是实现纵深防御的有效方式
- 持续的安全测试和评估是保障安全的关键
