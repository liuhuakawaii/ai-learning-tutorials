// src/utils/validator.ts
import { AppError } from '../middlewares/errorHandler';

export function validateTitle(title: unknown): string {
  if (typeof title !== 'string') {
    throw new AppError('Title must be a string', 400);
  }
  const trimmed = title.trim();
  if (trimmed.length === 0) {
    throw new AppError('Title cannot be empty', 400);
  }
  if (trimmed.length > 200) {
    throw new AppError('Title cannot exceed 200 characters', 400);
  }
  return trimmed;
}

export function validatePriority(priority: unknown): 'low' | 'medium' | 'high' {
  const valid = ['low', 'medium', 'high'];
  if (!valid.includes(priority as string)) {
    throw new AppError(`Priority must be one of: ${valid.join(', ')}`, 400);
  }
  return priority as 'low' | 'medium' | 'high';
}

export function validateStatus(status: unknown): 'todo' | 'in_progress' | 'done' {
  const valid = ['todo', 'in_progress', 'done'];
  if (!valid.includes(status as string)) {
    throw new AppError(`Status must be one of: ${valid.join(', ')}`, 400);
  }
  return status as 'todo' | 'in_progress' | 'done';
}

export function validatePagination(page?: string, pageSize?: string) {
  const p = Math.max(1, parseInt(page || '1', 10) || 1);
  const ps = Math.min(100, Math.max(1, parseInt(pageSize || '20', 10) || 20));
  return { page: p, pageSize: ps };
}

export function validateDate(dateStr: unknown): Date | null {
  if (dateStr === null || dateStr === undefined || dateStr === '') {
    return null;
  }
  const date = new Date(dateStr as string);
  if (isNaN(date.getTime())) {
    throw new AppError('Invalid date format', 400);
  }
  return date;
}

// src/repositories/taskRepository.ts
import { Task, CreateTaskDTO, UpdateTaskDTO, TaskFilter } from '../types/task';
import { v4 as uuidv4 } from 'uuid';

// In-memory store for demonstration
let tasks: Task[] = [];

export function findAll(filter: TaskFilter): { tasks: Task[]; total: number } {
  let filtered = [...tasks];

  if (filter.status) {
    filtered = filtered.filter(t => t.status === filter.status);
  }
  if (filter.priority) {
    filtered = filtered.filter(t => t.priority === filter.priority);
  }
  if (filter.assigneeId) {
    filtered = filtered.filter(t => t.assigneeId === filter.assigneeId);
  }

  const sortBy = filter.sortBy || 'createdAt';
  const sortOrder = filter.sortOrder || 'desc';
  filtered.sort((a, b) => {
    const aVal = (a as any)[sortBy];
    const bVal = (b as any)[sortBy];
    return sortOrder === 'asc' ? (aVal > bVal ? 1 : -1) : (aVal < bVal ? 1 : -1);
  });

  const page = filter.page || 1;
  const pageSize = filter.pageSize || 20;
  const total = filtered.length;
  const start = (page - 1) * pageSize;
  const paged = filtered.slice(start, start + pageSize);

  return { tasks: paged, total };
}

export function findById(id: string): Task | undefined {
  return tasks.find(t => t.id === id);
}

export function create(dto: CreateTaskDTO): Task {
  const now = new Date();
  const task: Task = {
    id: uuidv4(),
    title: dto.title,
    description: dto.description || '',
    status: 'todo',
    priority: dto.priority || 'medium',
    assigneeId: dto.assigneeId || null,
    dueDate: dto.dueDate ? new Date(dto.dueDate) : null,
    createdAt: now,
    updatedAt: now,
  };
  tasks.push(task);
  return task;
}

export function update(id: string, dto: UpdateTaskDTO): Task | undefined {
  const index = tasks.findIndex(t => t.id === id);
  if (index === -1) return undefined;

  const task = tasks[index];
  if (dto.title !== undefined) task.title = dto.title;
  if (dto.description !== undefined) task.description = dto.description;
  if (dto.status !== undefined) task.status = dto.status;
  if (dto.priority !== undefined) task.priority = dto.priority;
  if (dto.assigneeId !== undefined) task.assigneeId = dto.assigneeId;
  if (dto.dueDate !== undefined) {
    task.dueDate = dto.dueDate ? new Date(dto.dueDate) : null;
  }
  task.updatedAt = new Date();
  tasks[index] = task;
  return task;
}

export function remove(id: string): boolean {
  const index = tasks.findIndex(t => t.id === id);
  if (index === -1) return false;
  tasks.splice(index, 1);
  return true;
}

export function count(): number {
  return tasks.length;
}

export function reset(): void {
  tasks = [];
}

// src/services/taskService.ts
import * as taskRepo from '../repositories/taskRepository';
import { CreateTaskDTO, UpdateTaskDTO, TaskFilter } from '../types/task';
import { validateTitle, validatePriority, validateStatus, validateDate } from '../utils/validator';
import { AppError } from '../middlewares/errorHandler';

export function getTasks(filter: TaskFilter) {
  return taskRepo.findAll(filter);
}

export function getTaskById(id: string) {
  const task = taskRepo.findById(id);
  if (!task) {
    throw new AppError('Task not found', 404);
  }
  return task;
}

export function createTask(dto: CreateTaskDTO) {
  const title = validateTitle(dto.title);
  const priority = dto.priority ? validatePriority(dto.priority) : 'medium';
  const dueDate = dto.dueDate ? validateDate(dto.dueDate) : undefined;

  if (dueDate && dueDate < new Date()) {
    throw new AppError('Due date cannot be in the past', 400);
  }

  return taskRepo.create({
    ...dto,
    title,
    priority,
    dueDate: dueDate?.toISOString(),
  });
}

export function updateTask(id: string, dto: UpdateTaskDTO) {
  const existing = taskRepo.findById(id);
  if (!existing) {
    throw new AppError('Task not found', 404);
  }

  if (dto.title !== undefined) {
    dto.title = validateTitle(dto.title);
  }
  if (dto.priority !== undefined) {
    dto.priority = validatePriority(dto.priority);
  }
  if (dto.status !== undefined) {
    dto.status = validateStatus(dto.status);
  }
  if (dto.dueDate !== undefined && dto.dueDate !== null) {
    const date = validateDate(dto.dueDate);
    if (date && date < new Date()) {
      throw new AppError('Due date cannot be in the past', 400);
    }
  }

  return taskRepo.update(id, dto);
}

export function deleteTask(id: string) {
  const existing = taskRepo.findById(id);
  if (!existing) {
    throw new AppError('Task not found', 404);
  }
  return taskRepo.remove(id);
}

export function getStats() {
  const { tasks } = taskRepo.findAll({ pageSize: 9999 });
  return {
    total: tasks.length,
    todo: tasks.filter(t => t.status === 'todo').length,
    inProgress: tasks.filter(t => t.status === 'in_progress').length,
    done: tasks.filter(t => t.status === 'done').length,
    highPriority: tasks.filter(t => t.priority === 'high').length,
  };
}

// src/middlewares/errorHandler.ts
import { Request, Response, NextFunction } from 'express';

export class AppError extends Error {
  public statusCode: number;

  constructor(message: string, statusCode: number) {
    super(message);
    this.statusCode = statusCode;
    this.name = 'AppError';
  }
}

export function errorHandler(err: Error, req: Request, res: Response, _next: NextFunction) {
  if (err instanceof AppError) {
    res.status(err.statusCode).json({
      code: err.statusCode,
      data: null,
      message: err.message,
    });
    return;
  }

  console.error('Unexpected error:', err);
  res.status(500).json({
    code: 500,
    data: null,
    message: 'Internal server error',
  });
}

// src/middlewares/auth.ts
import { Request, Response, NextFunction } from 'express';
import jwt from 'jsonwebtoken';
import { AppError } from './errorHandler';

const JWT_SECRET = process.env.JWT_SECRET || 'test-secret';

export interface AuthRequest extends Request {
  user?: { id: string; email: string; role: string };
}

export function authenticate(req: AuthRequest, res: Response, next: NextFunction) {
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    throw new AppError('No token provided', 401);
  }

  const token = authHeader.split(' ')[1];
  try {
    const payload = jwt.verify(token, JWT_SECRET) as { id: string; email: string; role: string };
    req.user = payload;
    next();
  } catch {
    throw new AppError('Invalid token', 401);
  }
}

export function authorize(...roles: string[]) {
  return (req: AuthRequest, res: Response, next: NextFunction) => {
    if (!req.user) {
      throw new AppError('Not authenticated', 401);
    }
    if (!roles.includes(req.user.role)) {
      throw new AppError('Insufficient permissions', 403);
    }
    next();
  };
}

## 七、工程建议

1. **从底层往上逐层测试**：先测工具函数（validator），再测 Repository，再测 Service，最后测 Controller，每层测试通过后再进入下一层，避免底层 Bug 干扰上层调试。
2. **Mock 粒度要适中**：Service 测试 Mock Repository，Controller 测试用真实 Service + 内存 Repository，不要 Mock 到"测试变成了验证 Mock 实现"的程度。
3. **测试数据工厂化**：封装 `createTestTask()` 等工厂函数，统一管理测试数据的创建，避免每个测试用例重复写 10 行 setup 代码，也方便后续调整数据结构。
4. **覆盖率补充要聚焦高风险代码**：当覆盖率到 85% 后，剩余的 15% 提升难度很大，优先补充支付、认证、数据修改等高风险路径的测试，低风险的配置读取可以暂时跳过。
