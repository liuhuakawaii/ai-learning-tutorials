### 2.1 准备被测函数

```typescript
// src/utils/string.ts

export function slugify(text: string): string {
  return text
    .toLowerCase()
    .trim()
    .replace(/[^\w\s-]/g, '')
    .replace(/[\s_]+/g, '-')
    .replace(/-+/g, '-')
    .replace(/^-+|-+$/g, '');
}

export function truncate(text: string, maxLength: number, suffix = '...'): string {
  if (text.length <= maxLength) return text;
  return text.slice(0, maxLength - suffix.length) + suffix;
}

export function parseFullName(fullName: string): {
  firstName: string;
  lastName: string;
  middleName?: string;
} {
  const parts = fullName.trim().split(/\s+/);
  if (parts.length === 1) {
    return { firstName: parts[0], lastName: '' };
  }
  if (parts.length === 2) {
    return { firstName: parts[0], lastName: parts[1] };
  }
  return {
    firstName: parts[0],
    lastName: parts[parts.length - 1],
    middleName: parts.slice(1, -1).join(' '),
  };
}
```

## 3. 用 AI 生成 pytest 测试

### 3.1 被测函数

```python
# src/utils/text.py

import re


def slugify(text: str) -> str:
    """将文本转换为 URL 友好的 slug。"""
    text = text.lower().strip()
    text = re.sub(r"[^\w\s-]", "", text)
    text = re.sub(r"[\s_]+", "-", text)
    text = re.sub(r"-+", "-", text)
    text = text.strip("-")
    return text


def truncate(text: str, max_length: int, suffix: str = "...") -> str:
    """截断文本到指定长度，超出部分用 suffix 替代。"""
    if len(text) <= max_length:
        return text
    return text[: max_length - len(suffix)] + suffix


def count_words(text: str) -> dict[str, int]:
    """统计文本中各单词出现次数，忽略大小写。"""
    words = re.findall(r"[a-zA-Z]+", text.lower())
    result: dict[str, int] = {}
    for word in words:
        result[word] = result.get(word, 0) + 1
    return result
```

## 4. AI 生成测试的常见问题

### 4.1 问题清单

```
┌─────────────────────────────────────────────────────────────────────┐
│                 AI 生成测试的常见问题                                │
├──────────────────┬──────────────────────────────────────────────────┤
│ 问题             │ 表现                                             │
├──────────────────┼──────────────────────────────────────────────────┤
│ 测试太浅         │ 只测 happy path，缺少异常和边界                    │
│ 断言太弱         │ 只 assert 不抛异常，不验证具体值                   │
│ 测试耦合实现     │ 测试内部实现细节，重构即失败                       │
│ Mock 滥用       │ 过度 mock，测试通过但实际代码有 Bug                │
│ 测试名称模糊     │ "test1", "test2" 无法从失败信息定位问题            │
│ 忽略异步处理     │ 未 await 异步函数，测试假通过                      │
│ 数据硬编码       │ 测试数据写死，可读性差                             │
└──────────────────┴──────────────────────────────────────────────────┘
```

### 4.2 如何改进 AI 生成的测试

**改进前（AI 默认生成）：**

```typescript
it('测试加法', () => {
  expect(add(1, 2)).toBe(3);
});
```

**改进后（你告诉 AI 修正）：**

```typescript
describe('add', () => {
  it('两个正数相加返回正确结果', () => {
    expect(add(2, 3)).toBe(5);
  });

  it('正数与负数相加', () => {
    expect(add(5, -3)).toBe(2);
  });

  it('两个负数相加', () => {
    expect(add(-2, -3)).toBe(-5);
  });

  it('与零相加返回原值', () => {
    expect(add(7, 0)).toBe(7);
  });

  it('浮点数相加处理精度', () => {
    expect(add(0.1, 0.2)).toBeCloseTo(0.3);
  });

  it('大数相加不溢出', () => {
    expect(add(Number.MAX_SAFE_INTEGER, 1)).toBe(Number.MAX_SAFE_INTEGER + 1);
  });
});
```

## 6. 实战：为 Express 路由生成测试

### 6.1 被测代码

```typescript
// src/routes/auth.ts
import { Router, Request, Response } from 'express';
import bcrypt from 'bcrypt';
import jwt from 'jsonwebtoken';
import { db } from '../database';

const router = Router();

router.post('/register', async (req: Request, res: Response) => {
  const { email, password, name } = req.body;

  if (!email || !password) {
    return res.status(400).json({ error: '邮箱和密码不能为空' });
  }

  if (password.length < 8) {
    return res.status(400).json({ error: '密码长度不能少于 8 位' });
  }

  const existing = await db.user.findUnique({ where: { email } });
  if (existing) {
    return res.status(409).json({ error: '邮箱已注册' });
  }

  const hashedPassword = await bcrypt.hash(password, 10);
  const user = await db.user.create({
    data: { email, password: hashedPassword, name },
  });

  const token = jwt.sign({ userId: user.id }, process.env.JWT_SECRET!);
  res.status(201).json({ token, userId: user.id });
});

export default router;
```

## 常见错误

### 错误 1：直接使用 AI 生成的测试而不审查

```bash
# ❌ 错误做法：AI 生成后直接提交
# AI 可能生成看似正确但逻辑有误的测试
# 例如：测试通过了，但断言的值本身就是错的

# ✅ 正确做法：仔细审查每个断言
# 确认 expected 值是你真正期望的
# 运行测试确认全部通过
# 人为检查边界值是否合理
```

### 错误 2：测试名称不够描述性

```typescript
// ❌ AI 有时生成这种模糊命名
it('test1', () => { ... });
it('works correctly', () => { ... });

// ✅ 改为描述具体行为
it('空邮箱输入返回 400 错误', () => { ... });
it('密码少于 8 位时返回验证错误', () => { ... });
```

### 错误 3：忽略异步测试的 await

```typescript
// ❌ 忘记 await，测试假通过
it('获取用户', () => {
  const user = fetchUser('123'); // 返回 Promise！
  expect(user.name).toBe('Alice'); // 实际上永远不报错
});

// ✅ 正确使用 async/await
it('获取用户', async () => {
  const user = await fetchUser('123');
  expect(user.name).toBe('Alice');
});
```

### 错误 4：Mock 过深导致测试无意义

```typescript
// ❌ Mock 了要测试的逻辑本身
jest.mock('../service', () => ({
  calculate: jest.fn().mockReturnValue(42), // 这还测什么？
}));

// ✅ Mock 外部依赖，测试核心逻辑
jest.mock('../database', () => ({
  db: { user: { findUnique: jest.fn() } },
}));
```

### 练习 2：为 Python 数据处理函数生成 pytest 测试

```python
def merge_dicts(*dicts: dict) -> dict:
    """合并多个字典，后面的覆盖前面的。"""
    result = {}
    for d in dicts:
        result.update(d)
    return result

def flatten(nested_list: list) -> list:
    """递归展平嵌套列表。"""
    result = []
    for item in nested_list:
        if isinstance(item, list):
            result.extend(flatten(item))
        else:
            result.append(item)
    return result
```

要求：
- 使用 `@pytest.mark.parametrize` 参数化
- 覆盖空输入、嵌套、类型混合等边界情况
- 每个函数至少 6 组测试数据
