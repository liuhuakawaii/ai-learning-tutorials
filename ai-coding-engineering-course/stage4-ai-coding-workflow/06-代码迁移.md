**原始 JavaScript 代码**:
```javascript
// src/services/taskService.js
const { db } = require('../db');
const { tasks } = require('../db/schema');
const { eq, and, count } = require('drizzle-orm');

const STATUS_MAP = {
  TODO: 'todo',
  IN_PROGRESS: 'in_progress',
  DONE: 'done',
};

const PRIORITY_MAP = {
  LOW: 'low',
  MEDIUM: 'medium',
  HIGH: 'high',
};

async function createTask(data) {
  const { title, description, assigneeId, dueDate, priority, tags } = data;

  if (!title || title.length > 200) {
    throw new Error('Invalid title');
  }

  const [task] = await db
    .insert(tasks)
    .values({
      title,
      description: description || null,
      assigneeId: assigneeId || null,
      dueDate: dueDate ? new Date(dueDate) : null,
      priority: priority || PRIORITY_MAP.MEDIUM,
      tags: tags || [],
      status: STATUS_MAP.TODO,
    })
    .returning();

  return task;
}

async function listTasks(filters = {}) {
  const { status, assigneeId, page = 1, pageSize = 20 } = filters;
  const conditions = [];

  if (status) conditions.push(eq(tasks.status, status));
  if (assigneeId) conditions.push(eq(tasks.assigneeId, assigneeId));

  const where = conditions.length > 0 ? and(...conditions) : undefined;

  const [{ total }] = await db
    .select({ total: count() })
    .from(tasks)
    .where(where);

  const data = await db
    .select()
    .from(tasks)
    .where(where)
    .limit(pageSize)
    .offset((page - 1) * pageSize);

  return {
    data,
    pagination: { page, pageSize, total, totalPages: Math.ceil(total / pageSize) },
  };
}

module.exports = { createTask, listTasks, STATUS_MAP, PRIORITY_MAP };
```

```typescript
  pagination: {
    page: number;
    pageSize: number;
    total: number;
    totalPages: number;
  };
}

export interface TaskRecord {
  id: string;
  title: string;
  description: string | null;
  status: TaskStatus;
  priority: TaskPriority;
  assigneeId: string | null;
  dueDate: Date | null;
  tags: string[];

  title: z.string().min(1).max(200),
  description: z.string().optional(),
  assigneeId: z.string().uuid().optional(),
  dueDate: z.string().datetime().optional(),
  priority: z.nativeEnum(TaskPriority).default(TaskPriority.MEDIUM),

export async function createTask(data: CreateTaskData): Promise<TaskRecord> {

  return task as TaskRecord;
}

  const { status, assigneeId, page = 1, pageSize = 20 } = filters;

  const conditions: SQL[] = [];
  if (status) conditions.push(eq(tasks.status, status));
  if (assigneeId) conditions.push(eq(tasks.assigneeId, assigneeId));

  const where = conditions.length > 0 ? and(...conditions) : undefined;

  const [{ total }] = await db
    .select({ total: count() })
    .from(tasks)
    .where(where);

  const data = await db
    .select()
    .from(tasks)
    .where(where)
    .limit(pageSize)
    .offset((page - 1) * pageSize);

  return {
    data: data as TaskRecord[],
    pagination: {
      page,
      pageSize,
      total,
      totalPages: Math.ceil(total / pageSize),

  };
```

```typescript
function getJsFiles(dir: string): string[] {
  const files: string[] = [];
  for (const entry of readdirSync(dir, { withFileTypes: true })) {
    const fullPath = join(dir, entry.name);
    if (entry.isDirectory()) {
```

```jsx
class TaskList extends React.Component {
  constructor(props) {
    super(props);
    this.state = {
      tasks: [],
      loading: false,
      error: null,
      filters: { status: 'all', search: '' },
    };
  }

  componentDidMount() {
    this.fetchTasks();
  }

  componentDidUpdate(prevProps, prevState) {
    if (prevState.filters !== this.state.filters) {
      this.fetchTasks();
    }
  }

  fetchTasks = async () => {
    this.setState({ loading: true, error: null });
    try {
      const response = await fetch(`/api/tasks?${new URLSearchParams(this.state.filters)}`);
      if (!response.ok) throw new Error('Failed to fetch');
      const data = await response.json();
      this.setState({ tasks: data.data, loading: false });
    } catch (error) {
      this.setState({ error: error.message, loading: false });
    }
  };

  handleFilterChange = (key, value) => {
    this.setState(prev => ({
      filters: { ...prev.filters, [key]: value },
    }));
  };

  render() {
    const { tasks, loading, error, filters } = this.state;

    if (loading) return <Spinner />;
    if (error) return <ErrorMessage message={error} />;

    return (
      <div>
        <Filters value={filters} onChange={this.handleFilterChange} />
        <TaskItems tasks={tasks} />
      </div>
    );
  }
```

**Hooks 组件（AI 转换后）**:
```tsx
import { useState, useEffect, useCallback } from 'react';

interface TaskFilters {
  status: string;
  search: string;
}

interface Task {
  id: string;
  title: string;
  status: string;
  // ...
}

// 提取自定义 Hook
function useTasks(filters: TaskFilters) {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const fetchTasks = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const params = new URLSearchParams(filters);
      const response = await fetch(`/api/tasks?${params}`);
      if (!response.ok) throw new Error('Failed to fetch');
      const data = await response.json();
      setTasks(data.data);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
    } finally {
      setLoading(false);
    }
  }, [filters]);

  useEffect(() => {
    fetchTasks();
  }, [fetchTasks]);

  return { tasks, loading, error, refetch: fetchTasks };
}

// 主组件
function TaskList() {
  const [filters, setFilters] = useState<TaskFilters>({
    status: 'all',
    search: '',
  });

  const { tasks, loading, error } = useTasks(filters);

  const handleFilterChange = useCallback((key: string, value: string) => {
    setFilters(prev => ({ ...prev, [key]: value }));
  }, []);

  if (loading) return <Spinner />;
  if (error) return <ErrorMessage message={error} />;

  return (
    <div>
      <Filters value={filters} onChange={handleFilterChange} />
      <TaskItems tasks={tasks} />
    </div>
  );
}
```

## 7. 常见错误

### ❌ 错误 1：一次性迁移所有代码
大规模迁移应渐进式进行。**分模块迁移，每步验证**。

### ❌ 错误 2：迁移前没有补充测试
没有测试就无法验证迁移是否正确。**先补测试再迁移**。

### ❌ 错误 3：完全信任 AI 的转换结果
AI 可能误解业务逻辑。**逐行审查转换后的代码**。

### ❌ 错误 4：忽略性能影响
新代码可能引入性能问题。**迁移前后做性能对比**。

### ❌ 错误 5：没有回滚计划
迁移失败时无法快速恢复。**始终保持可回滚状态**。

---

## 8. 总结

本课学习了 AI 辅助的代码迁移策略：

1. **JS → TS** — 渐进式类型迁移，从叶子模块开始
2. **REST → GraphQL** — 并行运行，逐步迁移
3. **Class → Hooks** — 功能等价转换，提取自定义 Hooks
4. **安全策略** — 测试先行，渐进迁移，随时可回滚
5. **验证方法** — 功能、性能、代码质量三重验证

**核心原则**: 迁移是手段不是目的，确保每次迁移都带来实际价值。
