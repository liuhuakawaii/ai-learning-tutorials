```typescript
// src/db/schema.ts
import { pgTable, text, timestamp, integer, bigint, boolean, index, uniqueIndex } from 'drizzle-orm/pg-core';
import { createId } from '@paralleldrive/cuid2';

export const links = pgTable('links', {
  id: text('id').primaryKey().$defaultFn(() => createId()),
  code: text('code').notNull(),
  originalUrl: text('original_url').notNull(),
  title: text('title'),
  expiresAt: timestamp('expires_at'),
  isCustomCode: boolean('is_custom_code').default(false),
  apiKeyId: text('api_key_id').references(() => apiKeys.id),
  clickCount: bigint('click_count', { mode: 'number' }).default(0),
  createdAt: timestamp('created_at').defaultNow().notNull(),
  updatedAt: timestamp('updated_at').defaultNow().notNull(),
  deletedAt: timestamp('deleted_at'),
}, (table) => ({
  codeIdx: uniqueIndex('code_idx').on(table.code),
  apiKeyIdx: index('api_key_idx').on(table.apiKeyId),
  expiresIdx: index('expires_idx').on(table.expiresAt),
}));

export const clicks = pgTable('clicks', {
  id: text('id').primaryKey().$defaultFn(() => createId()),
  linkId: text('link_id').references(() => links.id).notNull(),
  ip: text('ip'),
  userAgent: text('user_agent'),
  referer: text('referer'),
  country: text('country'),
  createdAt: timestamp('created_at').defaultNow().notNull(),
}, (table) => ({
  linkIdx: index('click_link_idx').on(table.linkId),
  createdAtIdx: index('click_created_at_idx').on(table.createdAt),
}));

export const apiKeys = pgTable('api_keys', {
  id: text('id').primaryKey().$defaultFn(() => createId()),
  key: text('key').notNull(),
  name: text('name').notNull(),
  isActive: boolean('is_active').default(true),
  lastUsedAt: timestamp('last_used_at'),
  createdAt: timestamp('created_at').defaultNow().notNull(),
  deletedAt: timestamp('deleted_at'),
}, (table) => ({
  keyIdx: uniqueIndex('api_key_idx').on(table.key),
}));
```

```typescript
// src/services/codeGenerator.ts
import { nanoid } from 'nanoid';
import { customAlphabet } from 'nanoid';

const alphabet = '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz';
const generate = customAlphabet(alphabet, 7);

const MAX_RETRIES = 5;

export function generateCode(customCode?: string): string {
  if (customCode) {
    validateCustomCode(customCode);
    return customCode;
  }
  return generate();
}

function validateCustomCode(code: string): void {
  if (code.length < 3 || code.length < 20) {
    throw new Error('Custom code must be between 3 and 20 characters');
  }
  if (!/^[a-zA-Z0-9_-]+$/.test(code)) {
    throw new Error('Custom code can only contain letters, numbers, hyphens, and underscores');
  }
  const reserved = ['api', 'admin', 'health', 'metrics', 'docs'];
  if (reserved.includes(code.toLowerCase())) {
    throw new Error('This code is reserved and cannot be used');
  }
}

export function generateApiKey(): string {
  return `sk_${nanoid(48)}`;
}
```

### 4.2 短链接 Service

```typescript
// src/services/linkService.ts
import { db } from '../db';
import { links, clicks } from '../db/schema';
import { eq, and, isNull, gt, desc, sql } from 'drizzle-orm';
import { generateCode } from './codeGenerator';
import { validateUrl } from '../utils/urlValidator';
import { NotFoundError, ConflictError, ValidationError } from '../utils/errors';

interface CreateLinkInput {
  url: string;
  title?: string;
  customCode?: string;
  expiresAt?: string;
  apiKeyId?: string;
}

interface LinkListQuery {
  page?: number;
  pageSize?: number;
  apiKeyId?: string;
}

export async function createLink(input: CreateLinkInput) {
  validateUrl(input.url);

  const code = generateCode(input.customCode);

  const existing = await db.select().from(links).where(eq(links.code, code)).limit(1);
  if (existing.length > 0) {
    throw new ConflictError(`Code "${code}" already exists`);
  }

  const [link] = await db.insert(links).values({
    code,
    originalUrl: input.url,
    title: input.title || null,
    expiresAt: input.expiresAt ? new Date(input.expiresAt) : null,
    isCustomCode: !!input.customCode,
    apiKeyId: input.apiKeyId || null,
  }).returning();

  return link;
}

export async function getLinkByCode(code: string) {
  const [link] = await db.select().from(links).where(
    and(
      eq(links.code, code),
      isNull(links.deletedAt)
    )
  ).limit(1);

  if (!link) {
    throw new NotFoundError('Link not found');
  }

  if (link.expiresAt && link.expiresAt < new Date()) {
    throw new NotFoundError('Link has expired');
  }

  return link;
}

export async function listLinks(query: LinkListQuery) {
  const page = query.page || 1;
  const pageSize = Math.min(query.pageSize || 20, 100);
  const offset = (page - 1) * pageSize;

  const conditions = [isNull(links.deletedAt)];
  if (query.apiKeyId) {
    conditions.push(eq(links.apiKeyId, query.apiKeyId));
  }

  const [countResult] = await db.select({ count: sql<number>`count(*)` })
    .from(links)
    .where(and(...conditions));

  const data = await db.select().from(links)
    .where(and(...conditions))
    .orderBy(desc(links.createdAt))
    .limit(pageSize)
    .offset(offset);

  return {
    data,
    pagination: {
      page,
      pageSize,
      total: countResult.count,
      totalPages: Math.ceil(countResult.count / pageSize),
    },
  };
}

export async function deleteLink(code: string) {
  const [link] = await db.select().from(links).where(eq(links.code, code)).limit(1);
  if (!link) {
    throw new NotFoundError('Link not found');
  }

  await db.update(links)
    .set({ deletedAt: new Date(), updatedAt: new Date() })
    .where(eq(links.code, code));

  return { success: true };
}
```

### 4.3 点击统计 Service

```typescript
// src/services/clickService.ts
import { db } from '../db';
import { clicks, links } from '../db/schema';
import { eq, and, sql, gte, lte } from 'drizzle-orm';

export async function recordClick(linkId: string, data: {
  ip?: string;
  userAgent?: string;
  referer?: string;
}) {
  await db.insert(clicks).values({
    linkId,
    ip: data.ip || null,
    userAgent: data.userAgent || null,
    referer: data.referer || null,
  });

  await db.update(links)
    .set({ clickCount: sql`${links.clickCount} + 1` })
    .where(eq(links.id, linkId));
}

export async function getStats(linkId: string, days: number = 7) {
  const startDate = new Date();
  startDate.setDate(startDate.getDate() - days);

  const [totalClicks] = await db.select({ count: sql<number>`count(*)` })
    .from(clicks)
    .where(and(
      eq(clicks.linkId, linkId),
      gte(clicks.createdAt, startDate)
    ));

  const dailyClicks = await db.select({
    date: sql<string>`date(${clicks.createdAt})`,
    count: sql<number>`count(*)`,
  })
    .from(clicks)
    .where(and(
      eq(clicks.linkId, linkId),
      gte(clicks.createdAt, startDate)
    ))
    .groupBy(sql`date(${clicks.createdAt})`)
    .orderBy(sql`date(${clicks.createdAt})`);

  const topReferers = await db.select({
    referer: clicks.referer,
    count: sql<number>`count(*)`,
  })
    .from(clicks)
    .where(and(
      eq(clicks.linkId, linkId),
      gte(clicks.createdAt, startDate)
    ))
    .groupBy(clicks.referer)
    .orderBy(sql`count(*) desc`)
    .limit(10);

  return {
    totalClicks: totalClicks.count,
    dailyClicks,
    topReferers,
  };
}
```

### 4.4 路由实现

```typescript
// src/routes/links.ts
import { FastifyInstance, FastifyRequest, FastifyReply } from 'fastify';
import { z } from 'zod';
import * as linkService from '../services/linkService';
import * as clickService from '../services/clickService';

const createLinkSchema = z.object({
  url: z.string().url('Invalid URL format'),
  title: z.string().max(200).optional(),
  customCode: z.string().min(3).max(20).regex(/^[a-zA-Z0-9_-]+$/).optional(),
  expiresAt: z.string().datetime().optional(),
});

export async function linkRoutes(app: FastifyInstance) {
  // 创建短链接
  app.post('/api/v1/links', async (req: FastifyRequest, reply: FastifyReply) => {
    const body = createLinkSchema.parse(req.body);
    const apiKeyId = (req as any).apiKeyId;
    const link = await linkService.createLink({ ...body, apiKeyId });
    return reply.status(201).send({ code: 0, data: link, message: 'created' });
  });

  // 获取列表
  app.get('/api/v1/links', async (req: FastifyRequest, reply: FastifyReply) => {
    const query = req.query as any;
    const apiKeyId = (req as any).apiKeyId;
    const result = await linkService.listLinks({
      page: parseInt(query.page) || 1,
      pageSize: parseInt(query.pageSize) || 20,
      apiKeyId,
    });
    return reply.send({ code: 0, data: result, message: 'success' });
  });

  // 获取详情
  app.get('/api/v1/links/:code', async (req: FastifyRequest, reply: FastifyReply) => {
    const { code } = req.params as { code: string };
    const link = await linkService.getLinkByCode(code);
    return reply.send({ code: 0, data: link, message: 'success' });
  });

  // 删除
  app.delete('/api/v1/links/:code', async (req: FastifyRequest, reply: FastifyReply) => {
    const { code } = req.params as { code: string };
    await linkService.deleteLink(code);
    return reply.send({ code: 0, data: null, message: 'deleted' });
  });

  // 获取统计
  app.get('/api/v1/links/:code/stats', async (req: FastifyRequest, reply: FastifyReply) => {
    const { code } = req.params as { code: string };
    const query = req.query as any;
    const link = await linkService.getLinkByCode(code);
    const stats = await clickService.getStats(link.id, parseInt(query.days) || 7);
    return reply.send({ code: 0, data: stats, message: 'success' });
  });
}
```

### 4.5 重定向路由

```typescript
// src/routes/redirect.ts
import { FastifyInstance, FastifyRequest, FastifyReply } from 'fastify';
import * as linkService from '../services/linkService';
import * as clickService from '../services/clickService';

export async function redirectRoutes(app: FastifyInstance) {
  app.get('/:code', async (req: FastifyRequest, reply: FastifyReply) => {
    const { code } = req.params as { code: string };

    // 跳过 API 路由
    if (code === 'api') return;

    const link = await linkService.getLinkByCode(code);

    // 异步记录点击（不阻塞重定向）
    clickService.recordClick(link.id, {
      ip: req.ip,
      userAgent: req.headers['user-agent'],
      referer: req.headers.referer,
    }).catch(console.error);

    return reply.redirect(302, link.originalUrl);
  });
}
```

```typescript
import Fastify from 'fastify';

export async function buildApp() {
```

**POST** `/api/v1/links`

## 常见误区

### ❌ 误区 1：跳过需求分析直接写代码
觉得需求简单就直接开写，结果中途发现遗漏了关键功能，不得不推倒重来。**即使用 AI 辅助，也要先完成需求结构化再动手。**

### ❌ 误区 2：每个阶段只和 AI 交互一次
把整个需求一次性丢给 AI 让它"全搞定"，生成的代码往往质量低下。**每个阶段应该多轮对话，逐步细化和验证。**

### ❌ 误区 3：测试和文档留到最后补
先写完所有功能再补测试和文档，结果测试覆盖不到关键路径，文档和代码已经脱节。**测试和文档应该与代码同步生成。**

### ❌ 误区 4：过度依赖 AI 生成的架构决策
AI 不了解你的团队规模、业务约束和技术债务。**架构决策必须由人来做，AI 只提供参考方案。**

## 工程建议

1. **建立阶段检查点**：每个 Phase 结束前，花 5 分钟回顾产出物是否满足验收标准，再进入下一阶段。不要因为"AI 已经生成了"就跳过审查。

2. **保持可运行状态**：每完成一个模块的实现，确保项目能正常启动并通过已有测试。增量式开发比"全部写完再跑"更安全。

3. **记录 AI 的贡献边界**：标注哪些代码是 AI 生成的、哪些是人工修改的。这不仅方便代码审查，也有助于团队评估 AI 辅助开发的实际效果。

4. **从核心功能开始**：优先实现 P0 需求（短链接创建和重定向），确保核心链路跑通后再扩展 P1/P2 功能。避免一开始就陷入 QR 码生成等次要功能。
