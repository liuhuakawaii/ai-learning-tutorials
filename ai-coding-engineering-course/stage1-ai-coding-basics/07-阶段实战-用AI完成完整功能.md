**AI 生成的代码**：

```typescript
import jwt, { JwtPayload } from 'jsonwebtoken';

export class AuthError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'AuthError';
  }
}

interface TokenPayload {
  id: string;
  role: string;
}

export function generateToken(payload: TokenPayload): string {
  const secret = process.env.JWT_SECRET;
  if (!secret) {
    throw new Error('JWT_SECRET environment variable is not set');
  }

  return jwt.sign(payload, secret, {
    algorithm: 'HS256',
    expiresIn: '24h',
  });
}

export function verifyToken(token: string): TokenPayload & JwtPayload {
  const secret = process.env.JWT_SECRET;
  if (!secret) {
    throw new Error('JWT_SECRET environment variable is not set');
  }

  try {
    const decoded = jwt.verify(token, secret, { algorithms: ['HS256'] });
    return decoded as TokenPayload & JwtPayload;
  } catch (error) {
    if (error instanceof jwt.TokenExpiredError) {
      throw new AuthError('Token 已过期');
    }
    if (error instanceof jwt.JsonWebTokenError) {
      throw new AuthError('无效的 Token');
    }
    throw error;
  }
}
```

### 步骤 5：认证服务层

```
Chat 提示词：

创建 src/auth/auth.service.ts，实现以下功能：

1. register(input: RegisterInput): Promise<{ user: User; token: string }>
   - 检查邮箱唯一性
   - bcrypt 加密密码（salt rounds: 12）
   - 创建用户记录
   - 生成 JWT token
   - 返回用户信息（排除 passwordHash）和 token

2. login(input: LoginInput): Promise<{ user: User; token: string }>
   - 根据邮箱查找用户
   - bcrypt 比较密码
   - 生成 JWT token
   - 返回用户信息和 token

使用 Prisma，包含完整的错误处理。
```

**AI 生成的代码**：

```typescript
import bcrypt from 'bcrypt';
import { prisma } from '../lib/prisma.js';
import { generateToken, AuthError } from '../lib/jwt.js';
import type { RegisterInput, LoginInput } from './auth.schema.js';

const SALT_ROUNDS = 12;

export class AuthService {
  async register(input: RegisterInput) {
    // 检查邮箱唯一性
    const existingUser = await prisma.user.findUnique({
      where: { email: input.email },
    });

    if (existingUser) {
      throw new AuthError('该邮箱已被注册');
    }

    // 加密密码
    const passwordHash = await bcrypt.hash(input.password, SALT_ROUNDS);

    // 创建用户
    const user = await prisma.user.create({
      data: {
        email: input.email,
        name: input.name,
        passwordHash,
      },
    });

    // 生成 token
    const token = generateToken({ id: user.id, role: user.role });

    // 返回用户信息（排除 passwordHash）
    const { passwordHash: _, ...safeUser } = user;
    return { user: safeUser, token };
  }

  async login(input: LoginInput) {
    // 查找用户
    const user = await prisma.user.findUnique({
      where: { email: input.email },
    });

    if (!user) {
      throw new AuthError('邮箱或密码错误');
    }

    // 比较密码
    const isValidPassword = await bcrypt.compare(input.password, user.passwordHash);

    if (!isValidPassword) {
      throw new AuthError('邮箱或密码错误');
    }

    // 生成 token
    const token = generateToken({ id: user.id, role: user.role });

    // 返回用户信息（排除 passwordHash）
    const { passwordHash: _, ...safeUser } = user;
    return { user: safeUser, token };
  }
}

export const authService = new AuthService();
```

> **审查要点**：
> - 邮箱唯一性检查？✅
> - bcrypt salt rounds 是否足够（12）？✅
> - 错误信息是否泄露敏感信息（统一"邮箱或密码错误"）？✅
> - 返回时是否排除了 passwordHash？✅
> - 是否有并发问题？（邮箱唯一性有 DB 约束兜底）✅

### 步骤 6：Express 中间件

```
Chat 提示词：

创建 src/auth/auth.middleware.ts，实现两个中间件：

1. authenticate — JWT 认证中间件
   - 从 Authorization header 提取 Bearer token
   - 验证 token
   - 将用户信息附加到 req.user
   - 无效 token 返回 401

2. requireRole(...roles: string[]) — 角色授权中间件
   - 检查 req.user.role 是否在允许的角色列表中
   - 不满足返回 403

需要扩展 Express 的 Request 类型以支持 req.user。
```

**AI 生成的代码**：

```typescript
import { Request, Response, NextFunction } from 'express';
import { verifyToken, AuthError } from '../lib/jwt.js';

// 扩展 Express Request 类型
declare global {
  namespace Express {
    interface Request {
      user?: {
        id: string;
        role: string;
      };
    }
  }
}

export function authenticate(req: Request, res: Response, next: NextFunction) {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return res.status(401).json({ error: '未提供认证 Token' });
    }

    const token = authHeader.split(' ')[1];
    const payload = verifyToken(token);

    req.user = {
      id: payload.id,
      role: payload.role,
    };

    next();
  } catch (error) {
    if (error instanceof AuthError) {
      return res.status(401).json({ error: error.message });
    }
    return res.status(500).json({ error: '认证服务异常' });
  }
}

export function requireRole(...roles: string[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.user) {
      return res.status(401).json({ error: '未认证' });
    }

    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ error: '权限不足' });
    }

    next();
  };
}
```

> **审查要点**：
> - 是否正确提取 Bearer token？✅
> - 错误响应是否一致？✅
> - 角色检查是否在认证之后？✅
> - 类型扩展是否正确？✅


**AI 生成的代码**：

```typescript
import { Router, Request, Response } from 'express';
import { ZodError } from 'zod';
import { authService } from './auth.service.js';
import { RegisterSchema, LoginSchema } from './auth.schema.js';
import { authenticate, requireRole } from './auth.middleware.js';
import { AuthError } from '../lib/jwt.js';
import { prisma } from '../lib/prisma.js';

export const authRouter = Router();

// POST /auth/register
authRouter.post('/register', async (req: Request, res: Response) => {
  try {
    const input = RegisterSchema.parse(req.body);
    const result = await authService.register(input);
    return res.status(201).json(result);
  } catch (error) {
    if (error instanceof ZodError) {
      return res.status(400).json({
        error: '请求参数验证失败',
        details: error.errors,
      });
    }
    if (error instanceof AuthError) {
      return res.status(409).json({ error: error.message });
    }
    console.error('Registration error:', error);
    return res.status(500).json({ error: '服务器内部错误' });
  }
});

// POST /auth/login
authRouter.post('/login', async (req: Request, res: Response) => {
  try {
    const input = LoginSchema.parse(req.body);
    const result = await authService.login(input);
    return res.status(200).json(result);
  } catch (error) {
    if (error instanceof ZodError) {
      return res.status(400).json({
        error: '请求参数验证失败',
        details: error.errors,
      });
    }
    if (error instanceof AuthError) {
      return res.status(401).json({ error: error.message });
    }
    console.error('Login error:', error);
    return res.status(500).json({ error: '服务器内部错误' });
  }
});

// GET /auth/me
authRouter.get('/me', authenticate, async (req: Request, res: Response) => {
  try {
    const user = await prisma.user.findUnique({
      where: { id: req.user!.id },
      select: {
        id: true,
        email: true,
        name: true,
        role: true,
        createdAt: true,
      },
    });

    if (!user) {
      return res.status(404).json({ error: '用户不存在' });
    }

    return res.status(200).json(user);
  } catch (error) {
    console.error('Get user error:', error);
    return res.status(500).json({ error: '服务器内部错误' });
  }
});

// GET /auth/admin-only — 需要 ADMIN 角色
authRouter.get(
  '/admin-only',
  authenticate,
  requireRole('ADMIN'),
  async (req: Request, res: Response) => {
    return res.status(200).json({ message: '欢迎，管理员！' });
  }
);
```

> - 敏感信息是否在日志中暴露？（没有打印密码）✅

> - 测试用例是否覆盖了成功和失败路径？✅
> - beforeEach 是否清理了 mock 状态？✅
> - 断言是否精确？✅

## 常见错误

| 错误 | 正确做法 |
|------|----------|
| 让 AI 一次性生成所有文件 | 分步骤生成，每步审查后再继续 |
| 不看 AI 代码直接运行 | 按审查清单逐项检查 |
| 忽略错误处理 | 每个端点都要有完整的 try-catch |
| 不写测试 | 至少覆盖核心路径的测试 |
| 硬编码密钥 | 使用环境变量 |
| 返回密码字段 | 明确排除 passwordHash |

## 总结

本课要点：

1. **完整流程**：从项目初始化到测试，AI 覆盖了所有步骤

3. **审查实践**：每一步都有具体的审查要点
4. **安全优先**：密码加密、参数验证、token 安全是核心

### 练习 3：代码质量审计（20 分钟）

列出所有发现的问题，按严重程度排序，逐一修复。
