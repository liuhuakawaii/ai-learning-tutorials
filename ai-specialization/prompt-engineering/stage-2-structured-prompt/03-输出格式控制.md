# 输出格式控制

> **课程定位**：Stage 2 - 结构化 Prompt 工程 · 第 3 课
> **前置要求**：完成前两课，了解 JSON 基础语法
> **预计用时**：50-60 分钟

---

## 场景引入

你让 GPT-4 分析一段用户评论并输出 JSON，结果它返回了一段带 Markdown 代码块的文本，里面混着中文引号和尾部逗号，`json.loads()` 直接报错。你加了"请只输出 JSON"的指令，偶尔还是会多出一句"以下是分析结果"。输出格式不可靠是 LLM 应用最常见的工程痛点——仅靠 Prompt 约束远远不够，需要 API 参数、Schema 校验和格式修复三重保障。

---

1. 掌握 JSON Mode 和 Structured Output 的使用方法
2. 学会用 Schema 约束 LLM 输出格式
3. 理解格式修复（Format Repair）策略
4. 构建基于 Pydantic 的输出解析与校验系统

## 4. 正则引导的输出解析

```python
import re
from typing import Optional

class OutputParser:
    """通用输出解析器"""

    @staticmethod
    def extract_json(text: str) -> Optional[dict]:
        """从文本中提取 JSON"""
        # 尝试直接解析
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            pass

        # 尝试从代码块中提取
        patterns = [
            r'```json\s*(.*?)\s*```',
            r'```\s*(.*?)\s*```',
            r'\{[^{}]*(?:\{[^{}]*\}[^{}]*)*\}',
        ]

        for pattern in patterns:
            match = re.search(pattern, text, re.DOTALL)
            if match:
                try:
                    candidate = match.group(1) if '```' in pattern else match.group(0)
                    return json.loads(candidate)
                except (json.JSONDecodeError, IndexError):
                    continue

        return None

    @staticmethod
    def extract_xml_tags(text: str, tag: str) -> list[str]:
        """提取指定 XML 标签的内容"""
        pattern = f'<{tag}>(.*?)</{tag}>'
        return re.findall(pattern, text, re.DOTALL)

    @staticmethod
    def extract_table(text: str) -> list[list[str]]:
        """提取 Markdown 表格数据"""
        rows = []
        for line in text.strip().split('\n'):
            line = line.strip()
            if line.startswith('|') and line.endswith('|'):
                cells = [c.strip() for c in line.split('|')[1:-1]]
                if not all(set(c) <= {'-', ' ', ':'} for c in cells):
                    rows.append(cells)
        return rows
```

### 5.1 格式修复实现

```python
class FormatRepairer:
    """格式修复器"""

    @staticmethod
    def repair_json(text: str) -> Optional[str]:
        """尝试修复常见的 JSON 格式问题"""
        # 移除 BOM 和零宽字符
        text = text.strip().lstrip('\ufeff')

        # 修复中文引号
        text = text.replace('\u201c', '"').replace('\u201d', '"')
        text = text.replace('\u2018', "'").replace('\u2019', "'")

        # 修复尾部逗号
        text = re.sub(r',\s*([}\]])', r'\1', text)

        # 修复缺失的引号（简单场景）
        text = re.sub(
            r'(?<=[\{,])\s*(\w+)\s*:',
            r' "\1":',
            text
        )

        # 尝试补全未闭合的括号
        open_braces = text.count('{') - text.count('}')
        if open_braces > 0:
            text += '}' * open_braces

        open_brackets = text.count('[') - text.count(']')
        if open_brackets > 0:
            text += ']' * open_brackets

        return text

    @staticmethod
    def extract_and_repair(text: str) -> Optional[dict]:
        """提取并修复 JSON"""
        parser = OutputParser()

        # 先尝试直接解析
        result = parser.extract_json(text)
        if result:
            return result

        # 尝试修复
        repaired = FormatRepairer.repair_json(text)
        if repaired:
            try:
                return json.loads(repaired)
            except json.JSONDecodeError:
                pass

        # 从代码块中提取后修复
        for pattern in [r'```json\s*(.*?)\s*```', r'```\s*(.*?)\s*```']:
            match = re.search(pattern, text, re.DOTALL)
            if match:
                repaired = FormatRepairer.repair_json(match.group(1))
                try:
                    return json.loads(repaired)
                except json.JSONDecodeError:
                    continue

        return None
```

## 6. 代码实战：完整的输出控制系统

### 6.1 Pydantic Schema 驱动的输出校验

```python
from pydantic import BaseModel, Field, field_validator
from typing import TypeVar, Generic
from openai import OpenAI

T = TypeVar('T', bound=BaseModel)

class StructuredOutputEngine(Generic[T]):
    """基于 Pydantic 的结构化输出引擎"""

    def __init__(self, model: str = "gpt-4o"):
        self.client = OpenAI()
        self.model = model
        self.parser = OutputParser()
        self.repairer = FormatRepairer()

    def extract_schema(self, pydantic_model: type[T]) -> dict:
        """从 Pydantic Model 提取 JSON Schema"""
        return pydantic_model.model_json_schema()

    def build_prompt(self, task: str, schema: dict) -> str:
        return f"""{task}

请严格按照以下 JSON Schema 输出：
```json
{json.dumps(schema, ensure_ascii=False, indent=2)}
```

要求：
1. 输出合法 JSON
2. 包含所有 required 字段
3. 遵守类型和约束限制
4. 只输出 JSON，无其他内容"""

    def call_with_retry(
        self,
        messages: list[dict],
        pydantic_model: type[T],
        max_retries: int = 3,
    ) -> T:
        """带重试的结构化调用"""
        schema = self.extract_schema(pydantic_model)
        last_error = None
        last_output = None

        for attempt in range(max_retries):
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                temperature=0.1 if attempt > 0 else 0.3,
            )
            raw_output = response.choices[0].message.content
            last_output = raw_output

            # 解析
            data = self.parser.extract_json(raw_output)

            # 修复
            if data is None:
                data = self.repairer.extract_and_repair(raw_output)

            if data is None:
                last_error = "无法从输出中提取有效 JSON"
                messages.append({"role": "assistant", "content": raw_output})
                messages.append({
                    "role": "user",
                    "content": "你的输出不是合法 JSON，请重新输出，只输出 JSON，无其他内容。",
                })
                continue

            # 校验
            try:
                validated = pydantic_model.model_validate(data)
                return validated
            except Exception as e:
                last_error = f"Schema 校验失败: {e}"
                messages.append({"role": "assistant", "content": raw_output})
                messages.append({
                    "role": "user",
                    "content": f"JSON 校验失败，错误：{e}\n请修正后重新输出。",
                })
                continue

        raise RuntimeError(
            f"重试 {max_retries} 次后仍失败。最后错误: {last_error}\n最后输出: {last_output}"
        )
```

### 6.2 使用示例：产品评测分析

```python
class ProductReview(BaseModel):
    """产品评测结果"""
    product_name: str = Field(..., description="产品名称")
    overall_score: float = Field(..., ge=0, le=10, description="综合评分 0-10")
    pros: list[str] = Field(..., min_length=1, max_length=5, description="优点列表")
    cons: list[str] = Field(..., min_length=1, max_length=5, description="缺点列表")
    recommendation: Literal["强烈推荐", "推荐", "一般", "不推荐"] = Field(
        ..., description="推荐等级"
    )
    target_audience: str = Field(..., description="适合的目标用户群")

    @field_validator("overall_score")
    @classmethod
    def score_precision(cls, v):
        return round(v, 1)


def demo_structured_output():
    engine = StructuredOutputEngine[ProductReview](model="gpt-4o")

    messages = [
        {
            "role": "user",
            "content": engine.build_prompt(
                task="请对 MacBook Pro M4 进行产品评测",
                schema=engine.extract_schema(ProductReview),
            ),
        }
    ]

    review = engine.call_with_retry(messages, ProductReview)

    print(f"产品：{review.product_name}")
    print(f"评分：{review.overall_score}/10")
    print(f"推荐：{review.recommendation}")
    print(f"优点：{', '.join(review.pros)}")
    print(f"缺点：{', '.join(review.cons)}")
    print(f"适合：{review.target_audience}")
```

## 7. 常见误区

### ❌ 错误 1：过度依赖 Prompt 约束格式

```python
# 不可靠：仅靠 Prompt 要求 JSON
prompt = "输出JSON格式"  # LLM 可能输出多余文字

# 可靠：Prompt + API 参数 + 后处理三重保障
response = client.chat.completions.create(
    model="gpt-4o",
    messages=[...],
    response_format={"type": "json_object"},  # API 层保证
)
data = OutputParser.extract_json(response.choices[0].message.content)  # 后处理兜底
```

### ❌ 错误 2：Schema 过于复杂

```python
# 错误：嵌套层级过深，LLM 容易出错
schema = {
    "type": "object",
    "properties": {
        "level1": {
            "type": "object",
            "properties": {
                "level2": {
                    "type": "object",
                    "properties": {
                        "level3": {  # 嵌套太深！
                            ...
                        }
                    }
                }
            }
        }
    }
}

# 正确：扁平化 schema，或拆分为多次调用
```

### ❌ 错误 3：不处理格式修复就重试

```python
# 错误：直接重试，浪费 token
for i in range(3):
    response = call_llm(prompt)  # 每次输出可能有同样问题

# 正确：先修复，修复失败再重试
data = repairer.extract_and_repair(response)
if data is None:
    # 重试时追加错误反馈
    messages.append({"role": "user", "content": f"错误：{error}"})
```

## 8. 工程建议

1. **Structured Output 优先**：OpenAI 的 Structured Output 功能从 API 层保证输出符合 Schema，可靠性远高于纯 Prompt 约束，生产环境应优先使用。
2. **格式修复做兜底，但限制重试次数**：格式修复器能处理大部分边界情况（中文引号、尾部逗号、未闭合括号），但重试不应超过 3 次，避免无限循环。
3. **Schema 嵌套不超过 3 层**：深层嵌套的 JSON Schema 会让 LLM 出错率显著上升，复杂结构应拆分为多次调用或扁平化处理。
4. **为输出解析写单元测试**：用各种边界 case（空值、特殊字符、超长文本）测试解析器，确保生产环境中不会因格式问题导致 pipeline 中断。

## 9. 格式控制策略对比

| 策略 | 可靠性 | 延迟 | Token 消耗 | 适用场景 |
|------|--------|------|-----------|---------|
| Prompt 约束 | 低 | 无额外 | 少 | 简单任务 |
| JSON Mode | 高 | 无额外 | 少 | 通用 JSON 输出 |
| Structured Output | 最高 | 无额外 | 少 | 严格 Schema |
| 格式修复 | 中 | 重试时增加 | 中 | 兜底方案 |
| Regex 解析 | 中 | 低 | 无 | 自由格式提取 |

---

## 10. 总结

- 输出格式控制是 LLM 应用可靠性的关键环节
- Structured Output 是最可靠的方案，推荐优先使用
- 格式修复作为兜底方案，能处理大部分边界情况
- Pydantic 提供了从 Schema 定义到运行时校验的完整链路

## 参考答案

### 练习 1：多格式输出器

**思路**：核心思路是构造一个统一的任务描述，分别用 JSON Schema、XML 标签约束和 Markdown 表格格式三种 Prompt 调用 LLM，然后对每种格式的输出用对应的解析器提取和校验。比较准确率时，需要定义统一的校验标准（如字段完整性、类型正确性），用相同的 Pydantic Model 分别校验三种格式的解析结果。

**答案**：

class AnalysisResult(BaseModel):

    sentiment: str
    key_points: list[str]

@dataclass
class FormatTestResult:
    """单次格式测试结果"""
    format_name: str
    raw_output: str
    parsed_data: Optional[dict] = None
    parse_success: bool = False
    validation_success: bool = False
    error: str = ""


class MultiFormatOutputter:
    """多格式输出器：比较 JSON/XML/Markdown 的输出准确率"""

    def __init__(self, client=None, model: str = "gpt-4o"):
        self.client = client
        self.model = model

    def build_json_prompt(self, task: str, schema: dict) -> str:
        return f"""{task}

请严格按照以下 JSON Schema 输出：
```json
{json.dumps(schema, ensure_ascii=False, indent=2)}
```

要求：
1. 输出合法 JSON
2. 包含所有 required 字段
3. 只输出 JSON，无其他内容"""

    def build_xml_prompt(self, task: str) -> str:
        return f"""{task}

请使用以下 XML 格式输出：

<analysis>
  <summary>一句话总结</summary>
  <sentiment>positive/negative/neutral</sentiment>
  <key_points>
    <point>要点1</point>
    <point>要点2</point>
  </key_points>
  <confidence>0.0-1.0</confidence>
</analysis>

只输出 XML，不要添加其他说明。"""

    def build_markdown_prompt(self, task: str) -> str:
        return f"""{task}

请使用以下 Markdown 格式输出：

## Summary
（一句话总结）

## Sentiment
（positive/negative/neutral）

## Key Points
- 要点1
- 要点2

## Confidence
（0.0-1.0 的数字）

只输出 Markdown，不要添加其他说明。"""

    def parse_json_output(self, text: str) -> Optional[dict]:
        """解析 JSON 格式输出"""
        text = text.strip()
        # 尝试直接解析
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            pass
        # 从代码块提取
        match = re.search(r'```(?:json)?\s*(.*?)\s*```', text, re.DOTALL)
        if match:
            try:
                return json.loads(match.group(1))
            except json.JSONDecodeError:
                pass
        # 修复常见问题后重试
        text = text.replace('\u201c', '"').replace('\u201d', '"')
        text = re.sub(r',\s*([}\]])', r'\1', text)
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            return None

    def parse_xml_output(self, text: str) -> Optional[dict]:
        """解析 XML 格式输出"""
        def extract_tag(tag: str) -> Optional[str]:
            match = re.search(f'<{tag}>(.*?)</{tag}>', text, re.DOTALL)
            return match.group(1).strip() if match else None

        summary = extract_tag("summary")
        sentiment = extract_tag("sentiment")
        confidence_str = extract_tag("confidence")

        points = re.findall(r'<point>(.*?)</point>', text, re.DOTALL)

        if not all([summary, sentiment, confidence_str]):
            return None

        try:
            return {
                "summary": summary,
                "sentiment": sentiment,
                "key_points": [p.strip() for p in points],
                "confidence": float(confidence_str),
            }
        except (ValueError, TypeError):
            return None

    def parse_markdown_output(self, text: str) -> Optional[dict]:
        """解析 Markdown 格式输出"""
        sections = {}
        current_section = None
        current_content = []

        for line in text.strip().split('\n'):
            if line.startswith('## '):
                if current_section:
                    sections[current_section] = '\n'.join(current_content).strip()
                current_section = line[3:].strip().lower()
                current_content = []
            else:
                current_content.append(line)
        if current_section:
            sections[current_section] = '\n'.join(current_content).strip()

        try:
            key_points = [
                line.lstrip('- ').strip()
                for line in sections.get('key points', '').split('\n')
                if line.strip().startswith('- ')
            ]
            return {
                "summary": sections.get('summary', ''),
                "sentiment": sections.get('sentiment', '').strip(),
                "key_points": key_points,
                "confidence": float(sections.get('confidence', '0').strip()),
            }
        except (ValueError, KeyError):
            return None

    def validate_result(self, data: Optional[dict]) -> tuple[bool, str]:
        """用 Pydantic 校验解析结果"""
        if data is None:
            return False, "解析结果为 None"
        try:
            AnalysisResult(**data)
            return True, ""
        except Exception as e:
            return False, str(e)

    def test_format(self, task: str, format_name: str) -> FormatTestResult:
        """测试单种格式的输出准确率"""
        result = FormatTestResult(format_name=format_name, raw_output="")

        if not self.client:
            # 无客户端时返回空结果
            result.error = "未配置 OpenAI 客户端"
            return result

        # 构建 Prompt
        schema = AnalysisResult.model_json_schema()
        if format_name == "json":
            prompt = self.build_json_prompt(task, schema)
        elif format_name == "xml":
            prompt = self.build_xml_prompt(task)
        else:
            prompt = self.build_markdown_prompt(task)

        # 调用 LLM
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[{"role": "user", "content": prompt}],
            temperature=0.1,
        )
        result.raw_output = response.choices[0].message.content

        # 解析
        if format_name == "json":
            result.parsed_data = self.parse_json_output(result.raw_output)
        elif format_name == "xml":
            result.parsed_data = self.parse_xml_output(result.raw_output)
        else:
            result.parsed_data = self.parse_markdown_output(result.raw_output)

        result.parse_success = result.parsed_data is not None

        # 校验
        if result.parse_success:
            result.validation_success, result.error = self.validate_result(result.parsed_data)
        else:
            result.error = "解析失败"

        return result

    def compare_formats(self, task: str, rounds: int = 5) -> dict:
        """比较三种格式的准确率

        Args:
            task: 测试任务描述
            rounds: 测试轮数

        Returns:
            各格式的准确率统计
        """
        stats = {
            "json": {"parse_ok": 0, "valid_ok": 0, "total": rounds},
            "xml": {"parse_ok": 0, "valid_ok": 0, "total": rounds},
            "markdown": {"parse_ok": 0, "valid_ok": 0, "total": rounds},
        }

        for i in range(rounds):
            for fmt in ["json", "xml", "markdown"]:
                result = self.test_format(task, fmt)
                if result.parse_success:
                    stats[fmt]["parse_ok"] += 1
                if result.validation_success:
                    stats[fmt]["valid_ok"] += 1

        # 计算准确率
        summary = {}
        for fmt, data in stats.items():
            summary[fmt] = {
                "parse_rate": data["parse_ok"] / data["total"],
                "validation_rate": data["valid_ok"] / data["total"],
            }

        return summary
```

```python

    print(f"校验通过: {result.validation_success}")
    if result.parsed_data:
        print(f"结果: {json.dumps(result.parsed_data, ensure_ascii=False, indent=2)}")

**要点**：

- `temperature` 设为 0.1 而非 0，因为过低的温度可能导致某些模型输出模式固化，反而不利于测试泛化能力

### 练习 2：智能格式修复器

**思路**：在原有 `FormatRepairer` 的基础上扩展三个修复维度：日期格式统一化（将各种日期写法规范化为 ISO 格式）、数字精度对齐（统一小数位数）、缺失字段补默认值（根据 Schema 定义自动填充）。关键设计点是每种修复策略独立实现为一个方法，通过管道模式依次执行。

**答案**：

```python
import re
import json
from datetime import datetime
from typing import Optional, Any
from pydantic import BaseModel, Field


class SmartFormatRepairer:
    """智能格式修复器：修复日期、数字精度和缺失字段"""

    # 常见日期格式模式
    DATE_PATTERNS = [
        (r'(\d{4})/(\d{1,2})/(\d{1,2})', '{0}-{1:02d}-{2:02d}'),
        (r'(\d{4})\.(\d{1,2})\.(\d{1,2})', '{0}-{1:02d}-{2:02d}'),
        (r'(\d{4})年(\d{1,2})月(\d{1,2})日', '{0}-{1:02d}-{2:02d}'),
        (r'(\d{1,2})/(\d{1,2})/(\d{4})', '{2}-{0:02d}-{1:02d}'),
    ]

    def repair(self, text: str, schema: Optional[type[BaseModel]] = None,
               precision: int = 2) -> str:
        """执行完整的修复管道

        Args:
            text: 待修复的 JSON 字符串
            schema: Pydantic Schema（用于补全缺失字段）
            precision: 数字统一精度

        Returns:
            修复后的 JSON 字符串
        """
        # Step 1: 基础清理（复用原有逻辑）
        text = self._basic_cleanup(text)

        # Step 2: 日期格式统一
        text = self._normalize_dates(text)

        # Step 3: 数字精度对齐
        text = self._normalize_numbers(text, precision)

        # Step 4: 缺失字段补默认值
        if schema:
            text = self._fill_missing_fields(text, schema)

        return text

    def _basic_cleanup(self, text: str) -> str:
        """基础清理：去除 BOM、修复引号、修复尾部逗号"""
        text = text.strip().lstrip('\ufeff')
        text = text.replace('\u201c', '"').replace('\u201d', '"')
        text = text.replace('\u2018', "'").replace('\u2019', "'")
        text = re.sub(r',\s*([}\]])', r'\1', text)

        # 补全未闭合的括号
        open_braces = text.count('{') - text.count('}')
        if open_braces > 0:
            text += '}' * open_braces
        open_brackets = text.count('[') - text.count(']')
        if open_brackets > 0:
            text += ']' * open_brackets

        return text

    def _normalize_dates(self, text: str) -> str:
        """统一日期格式为 YYYY-MM-DD"""
        for pattern, replacement in self.DATE_PATTERNS:
            def replacer(match):
                groups = match.groups()
                try:
                    year = int(groups[0]) if len(groups[0]) == 4 else int(groups[2])
                    month = int(groups[1]) if len(groups[0]) == 4 else int(groups[0])
                    day = int(groups[2]) if len(groups[0]) == 4 else int(groups[1])
                    # 验证日期合法性
                    datetime(year, month, day)
                    return replacement.format(year, month, day)
                except (ValueError, IndexError):
                    return match.group(0)
            text = re.sub(pattern, replacer, text)
        return text

    def _normalize_numbers(self, text: str, precision: int = 2) -> str:
        """统一数字精度

        将 JSON 中的浮点数统一为指定小数位数。
        只处理值中的数字，不影响键名。
        """
        # 匹配 JSON 值中的浮点数（排除键名中的数字）
        # 匹配模式：冒号后面的数字值，或数组中的数字
        def normalize_match(match):
            num_str = match.group(0)
            try:
                num = float(num_str)
                # 只处理浮点数，整数保持不变
                if '.' in num_str:
                    return f"{num:.{precision}f}"
                return num_str
            except ValueError:
                return num_str

        # 匹配浮点数（包括负数）
        text = re.sub(r'-?\d+\.\d+', normalize_match, text)
        return text

    def _fill_missing_fields(self, text: str, schema: type[BaseModel]) -> str:
        """根据 Schema 自动补全缺失字段"""
        try:
            data = json.loads(text)
        except json.JSONDecodeError:
            return text

        if not isinstance(data, dict):
            return text

        # 获取 Schema 的默认值
        for field_name, field_info in schema.model_fields.items():
            if field_name not in data:
                if field_info.default is not None and field_info.default != ...:
                    data[field_name] = field_info.default
                elif field_info.default_factory is not None:
                    data[field_name] = field_info.default_factory()
                # 必填字段无默认值时不补全，让后续校验报错

        return json.dumps(data, ensure_ascii=False, indent=2)

    def repair_and_parse(self, text: str, schema: type[BaseModel],
                         precision: int = 2) -> tuple[Optional[BaseModel], str]:
        """修复并解析为 Pydantic Model

        Returns:
            (解析结果, 修复后的原始文本)
        """
        repaired_text = self.repair(text, schema=schema, precision=precision)
        try:
            data = json.loads(repaired_text)
            result = schema.model_validate(data)
            return result, repaired_text
        except Exception:
            return None, repaired_text
```

- 数字精度修复只处理浮点数，不改变整数值，且只匹配 JSON 值中的数字（通过正则避免误改键名）
- 缺失字段补全依赖 Pydantic Model 的默认值定义，必填字段没有默认值时不补全，留给后续校验器报错

### 练习 3：嵌套 Schema 分解器

**思路**：当 JSON Schema 嵌套超过 3 层时，将深层嵌套的对象类型提取为独立的子 Schema，用 `$ref` 引用关系记录它们之间的关联。每个扁平子 Schema 单独调用 LLM 获取结构化输出，最后根据引用关系将子结果合并为完整的嵌套结构。核心挑战是正确处理数组类型中嵌套对象的拆分和合并。

**答案**：

```python
import json
import copy
from dataclasses import dataclass, field
from typing import Any, Optional
from pydantic import BaseModel


@dataclass
class SubSchema:
    """拆分后的子 Schema"""
    name: str
    path: list[str]  # 在原始 Schema 中的路径
    schema: dict
    result: Optional[dict] = None


class SchemaDecomposer:
    """嵌套 Schema 分解器：将深层嵌套 Schema 拆分为扁平子 Schema"""

    MAX_DEPTH = 3

    def __init__(self, client=None, model: str = "gpt-4o"):
        self.client = client
        self.model = model

    def analyze_depth(self, schema: dict, current_depth: int = 0) -> int:
        """分析 Schema 的最大嵌套深度"""
        max_depth = current_depth
        if schema.get("type") == "object" and "properties" in schema:
            for prop_schema in schema["properties"].values():
                prop_depth = self.analyze_depth(prop_schema, current_depth + 1)
                max_depth = max(max_depth, prop_depth)
        elif schema.get("type") == "array" and "items" in schema:
            max_depth = max(max_depth, self.analyze_depth(schema["items"], current_depth + 1))
        return max_depth

    def decompose(self, schema: dict) -> list[SubSchema]:
        """将嵌套 Schema 拆分为扁平子 Schema 列表

        Args:
            schema: 原始 JSON Schema

        Returns:
            子 Schema 列表，每个子 Schema 嵌套不超过 MAX_DEPTH 层
        """
        sub_schemas = []
        schema = copy.deepcopy(schema)
        self._extract_nested(schema, [], sub_schemas)
        return sub_schemas

    def _extract_nested(self, schema: dict, path: list[str],
                        sub_schemas: list[SubSchema]) -> dict:
        """递归提取深层嵌套为子 Schema"""
        if schema.get("type") != "object" or "properties" not in schema:
            return schema

        current_depth = len(path)
        if current_depth < self.MAX_DEPTH:
            # 深度未超限，递归处理子属性
            for prop_name, prop_schema in schema["properties"].items():
                if prop_schema.get("type") == "object" and "properties" in prop_schema:
                    schema["properties"][prop_name] = self._extract_nested(
                        prop_schema, path + [prop_name], sub_schemas
                    )
                elif prop_schema.get("type") == "array" and "items" in prop_schema:
                    items = prop_schema["items"]
                    if items.get("type") == "object" and "properties" in items:
                        schema["properties"][prop_name]["items"] = self._extract_nested(
                            items, path + [prop_name, "items"], sub_schemas
                        )
            return schema

        # 深度超限，提取为子 Schema
        sub_name = "_".join(path) if path else "root"
        sub_schema = SubSchema(
            name=sub_name,
            path=path,
            schema=schema,
        )
        sub_schemas.append(sub_schema)

        # 返回引用标记
        return {
            "type": "object",
            "description": f"引用子 Schema: {sub_name}",
            "_ref": sub_name,
        }

    def build_sub_prompt(self, task: str, sub_schema: SubSchema) -> str:
        """为子 Schema 构建独立的 Prompt"""
        return f"""{task}

请提取以下结构的信息，严格按照 JSON Schema 输出：
```json
{json.dumps(sub_schema.schema, ensure_ascii=False, indent=2)}
```

要求：
1. 输出合法 JSON
2. 包含所有 required 字段
3. 只输出 JSON，无其他内容"""

    def call_llm(self, prompt: str) -> Optional[dict]:
        """调用 LLM 并解析 JSON 输出"""
        if not self.client:
            return None

        response = self.client.chat.completions.create(
            model=self.model,
            messages=[{"role": "user", "content": prompt}],
            temperature=0.1,
        )
        text = response.choices[0].message.content.strip()

        # 解析 JSON
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            # 尝试从代码块提取
            import re
            match = re.search(r'```(?:json)?\s*(.*?)\s*```', text, re.DOTALL)
            if match:
                try:
                    return json.loads(match.group(1))
                except json.JSONDecodeError:
                    pass
        return None

    def merge_results(self, original_schema: dict,
                      sub_schemas: list[SubSchema]) -> dict:
        """将子 Schema 的结果合并回原始结构

        Args:
            original_schema: 原始 Schema（含 _ref 标记）
            sub_schemas: 子 Schema 列表（含结果）

        Returns:
            合并后的完整数据
        """
        result_map = {s.name: s.result for s in sub_schemas if s.result}
        return self._resolve_refs(original_schema, result_map)

    def _resolve_refs(self, schema: dict, result_map: dict) -> dict:
        """递归解析 _ref 引用并替换为实际结果"""
        if "_ref" in schema:
            ref_name = schema["_ref"]
            return result_map.get(ref_name, {})

        if schema.get("type") == "object" and "properties" in schema:
            result = {}
            for prop_name, prop_schema in schema["properties"].items():
                if "_ref" in prop_schema:
                    result[prop_name] = result_map.get(prop_schema["_ref"], {})
                elif prop_schema.get("type") == "array" and "items" in prop_schema:
                    items = prop_schema["items"]
                    if "_ref" in items:
                        result[prop_name] = result_map.get(items["_ref"], [])
                    else:
                        result[prop_name] = self._resolve_refs(items, result_map)
                elif prop_schema.get("type") == "object":
                    result[prop_name] = self._resolve_refs(prop_schema, result_map)
            return result

        return {}

    def process(self, task: str, schema: dict) -> dict:
        """完整的分解-调用-合并流程

        Args:
            task: 任务描述
            schema: 原始 JSON Schema

        Returns:
            合并后的完整结果
        """
        # 1. 分析嵌套深度
        max_depth = self.analyze_depth(schema)
        print(f"Schema 最大嵌套深度: {max_depth}")

        if max_depth <= self.MAX_DEPTH:
            print("嵌套深度未超限，无需分解")
            if self.client:
                prompt = self.build_sub_prompt(task, SubSchema(
                    name="root", path=[], schema=schema
                ))
                return self.call_llm(prompt) or {}
            return {}

        # 2. 分解 Schema
        decomposed_schema = copy.deepcopy(schema)
        sub_schemas = []
        self._extract_nested(decomposed_schema, [], sub_schemas)
        print(f"分解为 {len(sub_schemas)} 个子 Schema:")
        for s in sub_schemas:
            print(f"  - {s.name} (路径: {' -> '.join(s.path)})")

        # 3. 分别调用 LLM
        for sub in sub_schemas:
            prompt = self.build_sub_prompt(
                f"{task}\n请提取 '{sub.name}' 相关的结构化信息。",
                sub,
            )
            sub.result = self.call_llm(prompt)
            print(f"  {s.name}: {'成功' if sub.result else '失败'}")

        # 4. 合并结果
        return self.merge_results(decomposed_schema, sub_schemas)
```

**要点**：
- 递归提取时用 `_ref` 标记替代深层嵌套，合并时根据标记回填实际结果，这是 JSON Schema `$ref` 机制的简化实现
- 数组类型的 `items` 如果是对象也需要递归拆分，否则嵌套在数组中的深层结构会被遗漏
- `analyze_depth` 用于预判断是否需要分解，避免对浅层 Schema 做不必要的拆分开销
