## 场景引入

你的 Prompt 已经有了版本管理，团队协作也顺畅了。但有一天，同事改了一个"优化中文情感识别"的 Prompt，提交后本地测了几条都通过了，于是直接合并到主分支。结果线上客服机器人突然不会回答"如何重置密码"了——新 Prompt 虽然提升了情感识别准确率，却意外破坏了问答能力。你意识到，光有版本管理还不够，必须在每次合并前自动运行回归测试，确保新改动不会破坏已有功能。

### 1.3 回归检测器实现

```python
import json
import numpy as np
from pathlib import Path
from dataclasses import dataclass
from typing import Optional

@dataclass
class RegressionCase:
    """回归测试用例"""
    id: str
    input_text: str
    baseline_response: str
    baseline_score: float
    must_contain: list[str]
    must_not_contain: list[str]
    min_score: float = 0.7
    category: str = "general"
    created_from: str = ""  # 来源（bug ID 等）

@dataclass
class RegressionResult:
    """回归测试结果"""
    case_id: str
    passed: bool
    current_score: float
    baseline_score: float
    regression_detected: bool
    details: dict

class RegressionDetector:
    """Prompt 回归检测器"""
    
    def __init__(self, regression_dir: str = "./regression_cases"):
        self.regression_dir = Path(regression_dir)
        self.regression_dir.mkdir(parents=True, exist_ok=True)
        self.cases: list[RegressionCase] = []
        self._load_cases()
    
    def _load_cases(self):
        """加载回归测试用例"""
        cases_file = self.regression_dir / "cases.json"
        if cases_file.exists():
            with open(cases_file, "r", encoding="utf-8") as f:
                data = json.load(f)
            self.cases = [RegressionCase(**c) for c in data]
    
    def save_cases(self):
        """保存回归测试用例"""
        cases_file = self.regression_dir / "cases.json"
        with open(cases_file, "w", encoding="utf-8") as f:
            json.dump(
                [c.__dict__ for c in self.cases],
                f, ensure_ascii=False, indent=2
            )
    
    def add_case(self, case: RegressionCase):
        """添加回归测试用例"""
        self.cases.append(case)
        self.save_cases()
    
    def add_from_incident(self, case_id: str, input_text: str,
                          expected_response: str, actual_response: str,
                          description: str = ""):
        """从线上事故创建回归用例"""
        case = RegressionCase(
            id=case_id,
            input_text=input_text,
            baseline_response=expected_response,
            baseline_score=1.0,
            must_contain=self._extract_keywords(expected_response),
            must_not_contain=self._extract_negative_keywords(actual_response),
            min_score=0.8,
            category="incident",
            created_from=description
        )
        self.add_case(case)
    
    def _extract_keywords(self, text: str) -> list[str]:
        """提取文本中的关键词（简化实现）"""
        # 实际项目中应使用 NLP 技术提取
        words = text.split()
        return [w for w in words if len(w) > 2][:5]
    
    def _extract_negative_keywords(self, text: str) -> list[str]:
        """从错误回答中提取应避免的关键词"""
        return []
    
    def check_keyword_regression(self, case: RegressionCase,
                                  current_response: str) -> dict:
        """关键词级别回归检查"""
        results = {
            "keyword_check": True,
            "missing_keywords": [],
            "forbidden_keywords": []
        }
        
        for kw in case.must_contain:
            if kw not in current_response:
                results["keyword_check"] = False
                results["missing_keywords"].append(kw)
        
        for kw in case.must_not_contain:
            if kw in current_response:
                results["keyword_check"] = False
                results["forbidden_keywords"].append(kw)
        
        return results
    
    def run_regression_test(self, prompt_fn, 
                            llm_judge_fn=None) -> list[RegressionResult]:
        """运行完整的回归测试"""
        results = []
        
        for case in self.cases:
            # 调用当前 Prompt
            current_response = prompt_fn(case.input_text)
            
            # 关键词检查
            keyword_result = self.check_keyword_regression(case, current_response)
            
            # 语义检查（如果提供了 judge）
            semantic_score = None
            if llm_judge_fn:
                semantic_score = llm_judge_fn(
                    case.input_text, current_response, case.baseline_response
                )
            
            # 判断是否回归
            current_score = semantic_score if semantic_score else (
                1.0 if keyword_result["keyword_check"] else 0.5
            )
            
            regression_detected = (
                current_score < case.min_score or
                (case.baseline_score - current_score) > 0.2  # 分数下降超过 20%
            )
            
            result = RegressionResult(
                case_id=case.id,
                passed=keyword_result["keyword_check"] and not regression_detected,
                current_score=current_score,
                baseline_score=case.baseline_score,
                regression_detected=regression_detected,
                details={
                    "keyword_result": keyword_result,
                    "semantic_score": semantic_score,
                    "current_response": current_response[:200]
                }
            )
            results.append(result)
        
        return results
    
    def generate_report(self, results: list[RegressionResult]) -> str:
        """生成回归测试报告"""
        total = len(results)
        passed = sum(1 for r in results if r.passed)
        failed = total - passed
        regressions = sum(1 for r in results if r.regression_detected)
        
        report = [
            "# 回归测试报告",
            "",
            f"**总用例数**: {total}",
            f"**通过**: {passed}",
            f"**失败**: {failed}",
            f"**检测到回归**: {regressions}",
            "",
            "## 详细结果",
            "",
            "| 用例ID | 通过 | 基线分数 | 当前分数 | 回归 |",
            "|--------|------|---------|---------|------|",
        ]
        
        for r in results:
            status = "✓" if r.passed else "✗"
            regression = "⚠️" if r.regression_detected else "✓"
            report.append(
                f"| {r.case_id} | {status} | {r.baseline_score:.2f} | "
                f"{r.current_score:.2f} | {regression} |"
            )
        
        if regressions > 0:
            report.extend([
                "",
                "## ⚠️ 回归详情",
                "",
            ])
            for r in results:
                if r.regression_detected:
                    report.extend([
                        f"### {r.case_id}",
                        f"- 基线分数: {r.baseline_score:.2f}",
                        f"- 当前分数: {r.current_score:.2f}",
                        f"- 分数下降: {r.baseline_score - r.current_score:.2f}",
                        "",
                    ])
        
        return "\n".join(report)
```

### 2.3 质量门禁脚本

```python
"""
scripts/quality_gate.py - 质量门禁检查脚本
"""
import sys
import json
import argparse
import xml.etree.ElementTree as ET
from pathlib import Path
from dataclasses import dataclass
from datetime import datetime

@dataclass
class TestSuiteResult:
    """测试套件结果"""
    name: str
    tests: int
    failures: int
    errors: int
    skipped: int
    
    @property
    def pass_rate(self) -> float:
        if self.tests == 0:
            return 0.0
        return (self.tests - self.failures - self.errors) / self.tests

@dataclass
class QualityGateResult:
    """质量门禁结果"""
    passed: bool
    unit_test_result: TestSuiteResult
    regression_result: TestSuiteResult
    checks: list[dict]
    recommendations: list[str]

class QualityGate:
    """质量门禁检查器"""
    
    def __init__(self, min_pass_rate: float = 0.95, 
                 max_regression_count: int = 0,
                 min_total_tests: int = 10):
        self.min_pass_rate = min_pass_rate
        self.max_regression_count = max_regression_count
        self.min_total_tests = min_total_tests
    
    def parse_junit_xml(self, xml_path: str) -> TestSuiteResult:
        """解析 JUnit XML 测试报告"""
        tree = ET.parse(xml_path)
        root = tree.getroot()
        
        testsuites = root.findall('.//testsuite')
        if not testsuites:
            # 可能根元素就是 testsuite
            if root.tag == 'testsuite':
                testsuites = [root]
        
        total_tests = 0
        total_failures = 0
        total_errors = 0
        total_skipped = 0
        
        for suite in testsuites:
            total_tests += int(suite.get('tests', 0))
            total_failures += int(suite.get('failures', 0))
            total_errors += int(suite.get('errors', 0))
            total_skipped += int(suite.get('skipped', 0))
        
        return TestSuiteResult(
            name="test_suite",
            tests=total_tests,
            failures=total_failures,
            errors=total_errors,
            skipped=total_skipped
        )
    
    def evaluate(self, unit_results_path: str,
                 regression_results_path: str) -> QualityGateResult:
        """评估质量门禁"""
        unit_result = self.parse_junit_xml(unit_results_path)
        regression_result = self.parse_junit_xml(regression_results_path)
        
        checks = []
        recommendations = []
        all_passed = True
        
        # 检查1：单元测试通过率
        unit_pass = unit_result.pass_rate >= self.min_pass_rate
        checks.append({
            "name": "单元测试通过率",
            "passed": unit_pass,
            "value": f"{unit_result.pass_rate:.1%}",
            "threshold": f">= {self.min_pass_rate:.1%}"
        })
        if not unit_pass:
            all_passed = False
            recommendations.append(
                f"单元测试通过率 {unit_result.pass_rate:.1%} 低于阈值 "
                f"{self.min_pass_rate:.1%}，请修复失败的测试"
            )
        
        # 检查2：回归测试通过率
        regression_pass = regression_result.pass_rate >= 1.0
        checks.append({
            "name": "回归测试通过率",
            "passed": regression_pass,
            "value": f"{regression_result.pass_rate:.1%}",
            "threshold": "100%"
        })
        if not regression_pass:
            all_passed = False
            recommendations.append(
                f"存在 {regression_result.failures} 个回归问题，必须全部修复"
            )
        
        # 检查3：最小测试数量
        min_tests_pass = unit_result.tests >= self.min_total_tests
        checks.append({
            "name": "测试覆盖率",
            "passed": min_tests_pass,
            "value": f"{unit_result.tests} 个测试",
            "threshold": f">= {self.min_total_tests} 个"
        })
        if not min_tests_pass:
            all_passed = False
            recommendations.append(
                f"测试数量不足，请添加更多测试用例"
            )
        
        return QualityGateResult(
            passed=all_passed,
            unit_test_result=unit_result,
            regression_result=regression_result,
            checks=checks,
            recommendations=recommendations
        )
    
    def generate_report(self, result: QualityGateResult) -> str:
        """生成质量门禁报告"""
        status = "✅ 通过" if result.passed else "❌ 未通过"
        
        report = [
            f"# Prompt 质量门禁报告",
            f"",
            f"**状态**: {status}",
            f"**检查时间**: {datetime.now().isoformat()}",
            f"",
            f"## 检查项",
            f"",
            f"| 检查项 | 结果 | 当前值 | 阈值 |",
            f"|--------|------|--------|------|",
        ]
        
        for check in result.checks:
            status_icon = "✅" if check["passed"] else "❌"
            report.append(
                f"| {check['name']} | {status_icon} | {check['value']} | {check['threshold']} |"
            )
        
        if result.recommendations:
            report.extend([
                "",
                "## 改进建议",
                "",
            ])
            for rec in result.recommendations:
                report.append(f"- {rec}")
        
        return "\n".join(report)


def main():
    parser = argparse.ArgumentParser(description="Prompt 质量门禁检查")
    parser.add_argument("--unit-results", required=True, help="单元测试结果 XML 路径")
    parser.add_argument("--regression-results", required=True, help="回归测试结果 XML 路径")
    parser.add_argument("--min-pass-rate", type=float, default=0.95, help="最低通过率")
    parser.add_argument("--max-regression-count", type=int, default=0, help="最大回归数")
    
    args = parser.parse_args()
    
    gate = QualityGate(
        min_pass_rate=args.min_pass_rate,
        max_regression_count=args.max_regression_count
    )
    
    result = gate.evaluate(args.unit_results, args.regression_results)
    report = gate.generate_report(result)
    
    # 保存报告
    Path("reports").mkdir(exist_ok=True)
    Path("reports/quality-report.md").write_text(report, encoding="utf-8")
    
    print(report)
    
    # 输出结果
    if not result.passed:
        print("\n❌ 质量门禁未通过，请修复上述问题后重试")
        sys.exit(1)
    else:
        print("\n✅ 质量门禁通过")


if __name__ == "__main__":
    main()
```

### 4.1 生产环境监控

```python
"""
monitoring/prompt_monitor.py - Prompt 生产环境监控
"""
import json
import time
from datetime import datetime
from pathlib import Path
from collections import defaultdict
from dataclasses import dataclass
from typing import Optional

@dataclass
class MonitorAlert:
    """监控告警"""
    timestamp: str
    prompt_name: str
    alert_type: str
    severity: str  # low, medium, high, critical
    message: str
    metrics: dict

class PromptMonitor:
    """Prompt 生产环境监控器"""
    
    def __init__(self, metrics_dir: str = "./metrics",
                 alert_thresholds: Optional[dict] = None):
        self.metrics_dir = Path(metrics_dir)
        self.metrics_dir.mkdir(parents=True, exist_ok=True)
        
        self.thresholds = alert_thresholds or {
            "latency_p99_ms": 5000,
            "error_rate": 0.05,
            "quality_score_min": 0.7,
            "consistency_score_min": 0.8
        }
        
        self.metrics_buffer: dict[str, list] = defaultdict(list)
        self.alerts: list[MonitorAlert] = []
    
    def record_metric(self, prompt_name: str, metric_name: str, 
                      value: float, metadata: Optional[dict] = None):
        """记录指标"""
        entry = {
            "timestamp": datetime.now().isoformat(),
            "prompt_name": prompt_name,
            "metric": metric_name,
            "value": value,
            "metadata": metadata or {}
        }
        
        self.metrics_buffer[prompt_name].append(entry)
        
        # 检查是否触发告警
        self._check_alert(prompt_name, metric_name, value)
    
    def _check_alert(self, prompt_name: str, metric_name: str, value: float):
        """检查是否需要告警"""
        threshold_key = f"{metric_name}_min"
        max_key = f"{metric_name}_max"
        
        alert = None
        
        if threshold_key in self.thresholds and value < self.thresholds[threshold_key]:
            alert = MonitorAlert(
                timestamp=datetime.now().isoformat(),
                prompt_name=prompt_name,
                alert_type="threshold_breach",
                severity="high",
                message=f"{metric_name} 低于阈值: {value} < {self.thresholds[threshold_key]}",
                metrics={metric_name: value}
            )
        elif max_key in self.thresholds and value > self.thresholds[max_key]:
            alert = MonitorAlert(
                timestamp=datetime.now().isoformat(),
                prompt_name=prompt_name,
                alert_type="threshold_breach",
                severity="high",
                message=f"{metric_name} 超过阈值: {value} > {self.thresholds[max_key]}",
                metrics={metric_name: value}
            )
        
        if alert:
            self.alerts.append(alert)
            self._send_alert(alert)
    
    def _send_alert(self, alert: MonitorAlert):
        """发送告警（简化实现）"""
        print(f"⚠️ ALERT [{alert.severity}]: {alert.message}")
        
        # 保存告警记录
        alerts_file = self.metrics_dir / "alerts.jsonl"
        with open(alerts_file, "a", encoding="utf-8") as f:
            f.write(json.dumps({
                "timestamp": alert.timestamp,
                "prompt_name": alert.prompt_name,
                "type": alert.alert_type,
                "severity": alert.severity,
                "message": alert.message,
                "metrics": alert.metrics
            }, ensure_ascii=False) + "\n")
    
    def flush_metrics(self):
        """将缓冲区的指标写入磁盘"""
        for prompt_name, metrics in self.metrics_buffer.items():
            if not metrics:
                continue
            
            metrics_file = self.metrics_dir / f"{prompt_name}_metrics.jsonl"
            with open(metrics_file, "a", encoding="utf-8") as f:
                for entry in metrics:
                    f.write(json.dumps(entry, ensure_ascii=False) + "\n")
            
            self.metrics_buffer[prompt_name] = []
    
    def get_summary(self, prompt_name: str, 
                    hours: int = 24) -> dict:
        """获取指标摘要"""
        metrics_file = self.metrics_dir / f"{prompt_name}_metrics.jsonl"
        
        if not metrics_file.exists():
            return {"error": "无数据"}
        
        # 读取最近 N 小时的数据
        cutoff = datetime.now().timestamp() - hours * 3600
        metrics = []
        
        with open(metrics_file, "r", encoding="utf-8") as f:
            for line in f:
                entry = json.loads(line)
                entry_time = datetime.fromisoformat(entry["timestamp"]).timestamp()
                if entry_time >= cutoff:
                    metrics.append(entry)
        
        if not metrics:
            return {"error": "指定时间范围内无数据"}
        
        # 按指标分组统计
        by_metric = defaultdict(list)
        for m in metrics:
            by_metric[m["metric"]].append(m["value"])
        
        summary = {}
        for metric_name, values in by_metric.items():
            import numpy as np
            summary[metric_name] = {
                "count": len(values),
                "mean": float(np.mean(values)),
                "std": float(np.std(values)),
                "min": float(np.min(values)),
                "max": float(np.max(values)),
                "p50": float(np.percentile(values, 50)),
                "p95": float(np.percentile(values, 95)),
                "p99": float(np.percentile(values, 99))
            }
        
        return summary
```

```python
"""
monitoring/prometheus_exporter.py - Prometheus 指标导出

from prometheus_client import start_http_server, Gauge, Counter, Histogram
import time

# 定义指标
PROMPT_LATENCY = Histogram(
    'prompt_latency_seconds',
    'Prompt 响应延迟',
    ['prompt_name', 'model']
)

PROMPT_QUALITY = Gauge(
    'prompt_quality_score',
    'Prompt 质量评分',
    ['prompt_name', 'metric']
)

PROMPT_ERRORS = Counter(
    'prompt_errors_total',
    'Prompt 错误总数',
    ['prompt_name', 'error_type']
)

PROMPT_REQUESTS = Counter(
    'prompt_requests_total',
    'Prompt 请求总数',
    ['prompt_name', 'status']
)

class PrometheusExporter:
    """Prometheus 指标导出器"""
    
    def __init__(self, port: int = 8000):
        self.port = port
    
    def start(self):
        """启动 HTTP 服务器暴露指标"""
        start_http_server(self.port)
        print(f"Prometheus 指标服务器启动在 :{self.port}")
    
    def record_latency(self, prompt_name: str, model: str, 
                       latency_seconds: float):
        PROMPT_LATENCY.labels(prompt_name=prompt_name, model=model).observe(latency_seconds)
    
    def record_quality(self, prompt_name: str, metric: str, score: float):
        PROMPT_QUALITY.labels(prompt_name=prompt_name, metric=metric).set(score)
    
    def record_error(self, prompt_name: str, error_type: str):
        PROMPT_ERRORS.labels(prompt_name=prompt_name, error_type=error_type).inc()
    
    def record_request(self, prompt_name: str, status: str):
        PROMPT_REQUESTS.labels(prompt_name=prompt_name, status=status).inc()
```

### ❌ 错误1：回归用例太少

```python

    {"id": "BUG-003", "input": "密码", "must_contain": ["重置"]},
    # ... 从线上事故持续积累
]
```

### ❌ 错误2：CI 中每次调用真实 API

```python
# 错误：每次都调用 OpenAI API（慢且贵）
def test_prompt():
    result = call_openai(prompt)  # 每次 CI 都花钱

# 正确：使用 mock 或缓存
@pytest.fixture
def mock_llm():
    def _mock(system, user):
        return "这是一个测试回复"
    return _mock

# 或使用缓存
@pytest.fixture
def cached_llm():
    cache = {}
    def _call(system, user):
        key = hash((system, user))
        if key not in cache:
            cache[key] = call_openai(system, user)
        return cache[key]
    return _call
```

### ❌ 错误3：质量门禁阈值设置不合理

```yaml
# 错误：阈值太严格，导致 CI 总是失败
min-pass-rate: 1.0  # 不允许任何失败

# 错误：阈值太宽松，失去了门禁意义
min-pass-rate: 0.5  # 一半失败也能通过

# 正确：根据实际情况设置合理的阈值
min-pass-rate: 0.95  # 允许 5% 的失败率
```

### ❌ 错误4：不区分测试类型和执行时机

```yaml
# 错误：所有测试都在每次提交时运行
on: [push, pull_request]  # 包含慢速的集成测试

# 正确：分层执行
on:
  push:
    # 快速测试每次推送都运行

    # 完整测试在 PR 时运行

  schedule:
    # 定期运行完整测试
    - cron: '0 2 * * *'
```

## 6. 工程建议

1. **回归用例必须从线上事故中积累**：每修一个 bug，就把问题输入和期望输出加入回归集——这是最有价值的测试资产，比你自己编造的测试用例更能覆盖真实场景。
2. **CI 中使用 LLM 响应缓存**：在开发和 CI 环境中缓存 LLM 的响应，避免每次提交都调用真实 API——既省成本又保证测试结果可复现。
3. **质量门禁阈值要渐进式收紧**：一开始设 90% 通过率，随着测试用例的成熟逐步提高到 95% 甚至更高——太严格会导致 CI 频繁失败，团队会绕过门禁。
4. **监控指标必须持久化**：内存中的指标在进程重启后就丢失了，使用 JSONL 或时序数据库持久化存储，才能支持趋势分析和异常回溯。

## 7. 总结

1. **回归测试**确保 Prompt 修改不会破坏已有的正确行为
2. **GitHub Actions** 提供了免费且强大的 CI/CD 基础设施
3. **质量门禁**是防止低质量 Prompt 进入生产环境的最后一道防线
4. **分层测试**策略：快速单元测试 → 全面回归测试 → 深度质量测试
5. **持续监控**让问题在影响用户之前被发现

### 练习2：GitHub Actions 配置

**思路**：完整的 CI 工作流应包含四个阶段：代码质量检查（lint + type check）→ 单元测试 → 回归测试 → 质量门禁。每个阶段失败都应阻止后续阶段，最终结果应自动评论到 PR 上。

**答案**：

```yaml
# .github/workflows/prompt-quality.yml

on:
  push:
    branches: [main, develop]
    paths:
      - 'prompts/**'
      - 'tests/prompts/**'
      - 'regression_cases/**'
  pull_request:
    branches: [main]
    paths:
      - 'prompts/**'
      - 'tests/prompts/**'
      - 'regression_cases/**'

  TEST_MODEL: gpt-4o-mini

    name: 代码质量检查

    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
      - name: 安装依赖
        run: |
```

```python
# scripts/check_metadata.py - 元数据格式检查
import sys
import yaml
from pathlib import Path

def check_metadata():
    prompts_dir = Path("prompts")
    if not prompts_dir.exists():
        print("prompts 目录不存在，跳过检查")
        return True

    all_passed = True
    for metadata_path in prompts_dir.glob("*/metadata.yaml"):
        try:
            with open(metadata_path) as f:
                metadata = yaml.safe_load(f)

            required = ["name", "version", "status", "author"]
            for field in required:
                if field not in metadata:
                    print(f"错误: {metadata_path} 缺少必需字段 '{field}'")
                    all_passed = False

            version = metadata.get("version", "")
            parts = version.split(".")
            if len(parts) != 3 or not all(p.isdigit() for p in parts):
                print(f"错误: {metadata_path} 版本号格式不正确: {version}")
                all_passed = False

            print(f"  ✓ {metadata_path}")
        except Exception as e:
            print(f"错误: 解析 {metadata_path} 失败: {e}")
            all_passed = False

    return all_passed

if __name__ == "__main__":
    if not check_metadata():
        sys.exit(1)
    print("所有元数据检查通过")
```

- 质量门禁的阈值应渐进式收紧：一开始设 90% 通过率，随着测试成熟逐步提高到 95%
- PR 评论让审查者一目了然地看到测试结果，无需跳转到 CI 页面

### 练习3：监控系统

**思路**：生产环境监控系统需要实时采集 Prompt 的质量指标（响应时间、评分、错误率），当指标异常时自动告警，并支持在严重退化时触发自动回滚。核心是指标采集层 + 异常检测层 + 告警/回滚层。

**答案**：

```python
"""
monitoring/prompt_monitor.py - 完整的 Prompt 监控系统
"""
import json
import time
import statistics
from datetime import datetime, timedelta
from pathlib import Path
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Optional, Callable

@dataclass
class MetricPoint:
    timestamp: str
    prompt_name: str
    metric_name: str
    value: float
    metadata: dict = field(default_factory=dict)

@dataclass
class Alert:
    timestamp: str
    prompt_name: str
    alert_type: str
    severity: str  # low, medium, high, critical
    message: str
    current_value: float
    threshold: float
    auto_action: str = ""  # rollback, notify, scale

class PromptMonitor:
    """Prompt 生产环境监控系统"""

    def __init__(self, metrics_dir: str = "./metrics",
                 thresholds: Optional[dict] = None,
                 on_alert: Optional[Callable] = None):
        self.metrics_dir = Path(metrics_dir)
        self.metrics_dir.mkdir(parents=True, exist_ok=True)
        self.on_alert = on_alert

        self.thresholds = thresholds or {
            "latency_p99_ms": {"max": 5000, "severity": "high"},
            "error_rate": {"max": 0.05, "severity": "critical"},
            "quality_score": {"min": 0.7, "severity": "high"},
            "consistency_score": {"min": 0.8, "severity": "medium"},
        }

        self.metrics_buffer: dict[str, list[MetricPoint]] = defaultdict(list)
        self.alerts: list[Alert] = []
        self.rollback_history: list[dict] = []

    def record(self, prompt_name: str, metric_name: str, value: float,
               metadata: Optional[dict] = None):
        """记录指标"""
        point = MetricPoint(
            timestamp=datetime.now().isoformat(),
            prompt_name=prompt_name,
            metric_name=metric_name,
            value=value,
            metadata=metadata or {}
        )
        self.metrics_buffer[prompt_name].append(point)
        self._check_threshold(prompt_name, metric_name, value)

    def _check_threshold(self, prompt_name: str, metric_name: str, value: float):
        """检查阈值并触发告警"""
        if metric_name not in self.thresholds:
            return

        threshold = self.thresholds[metric_name]
        alert = None

        if "max" in threshold and value > threshold["max"]:
            alert = Alert(
                timestamp=datetime.now().isoformat(),
                prompt_name=prompt_name,
                alert_type="threshold_breach",
                severity=threshold["severity"],
                message=f"{metric_name} 超过阈值: {value:.4f} > {threshold['max']}",
                current_value=value,
                threshold=threshold["max"]
            )
        elif "min" in threshold and value < threshold["min"]:
            alert = Alert(
                timestamp=datetime.now().isoformat(),
                prompt_name=prompt_name,
                alert_type="threshold_breach",
                severity=threshold["severity"],
                message=f"{metric_name} 低于阈值: {value:.4f} < {threshold['min']}",
                current_value=value,
                threshold=threshold["min"]
            )

        if alert:
            self.alerts.append(alert)
            self._handle_alert(alert)

    def _handle_alert(self, alert: Alert):
        """处理告警"""
        print(f"⚠️ [{alert.severity.upper()}] {alert.message}")

        # 持久化告警
        alerts_file = self.metrics_dir / "alerts.jsonl"
        with open(alerts_file, "a", encoding="utf-8") as f:
            f.write(json.dumps({
                "timestamp": alert.timestamp,
                "prompt_name": alert.prompt_name,
                "severity": alert.severity,
                "message": alert.message,
                "current_value": alert.current_value,
                "threshold": alert.threshold
            }, ensure_ascii=False) + "\n")

        # 严重告警触发自动回滚
        if alert.severity == "critical":
            self._auto_rollback(alert)

        # 外部回调
        if self.on_alert:
            self.on_alert(alert)

    def _auto_rollback(self, alert: Alert):
        """自动回滚到上一个稳定版本"""
        rollback_record = {
            "timestamp": datetime.now().isoformat(),
            "prompt_name": alert.prompt_name,
            "reason": alert.message,
            "action": "rollback_to_previous_version"
        }
        self.rollback_history.append(rollback_record)

        # 保存回滚记录
        rollback_file = self.metrics_dir / "rollbacks.jsonl"
        with open(rollback_file, "a", encoding="utf-8") as f:
            f.write(json.dumps(rollback_record, ensure_ascii=False) + "\n")

        print(f"🔄 自动回滚已触发: {alert.prompt_name}")

    def flush(self):
        """将缓冲区数据写入磁盘"""
        for prompt_name, points in self.metrics_buffer.items():
            if not points:
                continue
            metrics_file = self.metrics_dir / f"{prompt_name}_metrics.jsonl"
            with open(metrics_file, "a", encoding="utf-8") as f:
                for p in points:
                    f.write(json.dumps({
                        "timestamp": p.timestamp,
                        "metric": p.metric_name,
                        "value": p.value,
                        **p.metadata
                    }, ensure_ascii=False) + "\n")
            self.metrics_buffer[prompt_name] = []

    def get_summary(self, prompt_name: str, hours: int = 24) -> dict:
        """获取指标摘要"""
        metrics_file = self.metrics_dir / f"{prompt_name}_metrics.jsonl"
        if not metrics_file.exists():
            return {"error": "无数据"}

        cutoff = (datetime.now() - timedelta(hours=hours)).timestamp()
        by_metric = defaultdict(list)

        with open(metrics_file, "r", encoding="utf-8") as f:
            for line in f:
                entry = json.loads(line)
                entry_time = datetime.fromisoformat(entry["timestamp"]).timestamp()
                if entry_time >= cutoff:
                    by_metric[entry["metric"]].append(entry["value"])

        summary = {}
        for metric, values in by_metric.items():
            summary[metric] = {
                "count": len(values),
                "mean": statistics.mean(values),
                "stdev": statistics.stdev(values) if len(values) > 1 else 0,
                "min": min(values),
                "max": max(values),
                "p50": statistics.median(values),
            }
        return summary

if __name__ == "__main__":
    def alert_handler(alert: Alert):
        """告警回调：可接入钉钉/飞书/Slack"""
        print(f"  → 告警已推送到通知渠道: {alert.severity}")

    monitor = PromptMonitor(
        metrics_dir="./metrics",
        on_alert=alert_handler
    )

        latency = 200 + (i * 10 if i > 80 else 0)
        monitor.record("classifier", "latency_ms", latency)
        monitor.record("classifier", "quality_score", 0.9 - (0.02 * max(0, i - 85)))

    monitor.flush()

    summary = monitor.get_summary("classifier")

    print(f"\n告警数: {len(monitor.alerts)}")
    print(f"自动回滚数: {len(monitor.rollback_history)}")
```

**要点**：
- 监控指标必须持久化——内存中的数据在进程重启后就丢失，使用 JSONL 或时序数据库存储
- 自动回滚应仅在 critical 级别告警时触发，且需要记录详细的回滚原因和时间，便于事后复盘
- Grafana 仪表盘应包含延迟趋势、质量评分趋势、错误率和告警历史四个核心面板
