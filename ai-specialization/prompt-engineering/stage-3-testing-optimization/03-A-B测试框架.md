### 1.3 A/B 测试 vs 其他评估方式

| 方式 | 优点 | 缺点 | 适用场景 |
|------|------|------|---------|
| 人工评估 | 高质量判断 | 成本高、速度慢、主观 | 预研阶段 |
| 自动化测试 | 快速、可重复 | 可能遗漏细节 | 持续集成 |

| Shadow 测试 | 不影响用户 | 结果分析复杂 | 高风险变更 |

### 2.2 样本量计算

```python
import math
from scipy import stats

def calculate_sample_size(baseline_rate: float, 
                          minimum_detectable_effect: float,
                          alpha: float = 0.05, 
                          power: float = 0.8) -> int:
    """
    计算 A/B 测试所需的最小样本量
    
    Args:
        baseline_rate: 基线指标值（如当前通过率 0.7）
        minimum_detectable_effect: 最小可检测效应（如 0.05 表示 5% 提升）
        alpha: 显著性水平（通常 0.05）
        power: 统计功效（通常 0.8）
    
    Returns:
        每组所需的最小样本量
    """
    effect_size = minimum_detectable_effect / math.sqrt(baseline_rate * (1 - baseline_rate))
    
    z_alpha = stats.norm.ppf(1 - alpha / 2)
    z_beta = stats.norm.ppf(power)
    
    n = ((z_alpha + z_beta) / effect_size) ** 2
    return math.ceil(n)


# 示例：基线通过率 70%，想检测 5% 的提升
n = calculate_sample_size(
    baseline_rate=0.70,
    minimum_detectable_effect=0.05
)
print(f"每组需要至少 {n} 个样本")  # 约 1500+
```

### 2.3 流量分配策略

```python
import hashlib
import random

class TrafficSplitter:
    """流量分配器（基于用户 ID 的确定性分配）"""
    
    def __init__(self, control_ratio: float = 0.5, salt: str = ""):
        """
        Args:
            control_ratio: 对照组流量比例（0.5 = 50%）
            salt: 盐值，用于生成不同的分配方案
        """
        self.control_ratio = control_ratio
        self.salt = salt
    
    def assign(self, user_id: str) -> str:
        """
        根据用户 ID 确定性地分配到对照组或实验组
        同一用户始终在同一个组
        """
        hash_input = f"{user_id}:{self.salt}".encode()
        hash_value = int(hashlib.md5(hash_input).hexdigest(), 16)
        normalized = (hash_value % 10000) / 10000
        
        return "control" if normalized < self.control_ratio else "variant"
    
    def assign_batch(self, user_ids: list[str]) -> dict[str, list[str]]:
        """批量分配"""
        groups = {"control": [], "variant": []}
        for uid in user_ids:
            groups[self.assign(uid)].append(uid)
        return groups


# 示例
splitter = TrafficSplitter(control_ratio=0.5, salt="experiment_001")
assignment = splitter.assign("user_12345")
print(f"用户 user_12345 被分配到: {assignment}")
```

## 3. 统计显著性分析

### 3.1 假设检验基础

```
A/B 测试的假设检验：

H₀ (零假设): Prompt A 和 Prompt B 的效果没有差异
H₁ (备择假设): Prompt B 的效果优于 Prompt A

┌─────────────────────────────────────────────────────┐
│                    决策矩阵                          │
├─────────────────────┬───────────────┬───────────────┤
│                     │  实际有差异    │  实际无差异    │
├─────────────────────┼───────────────┼───────────────┤
│  判定有差异          │  ✓ 正确决策   │  ✗ 第一类错误  │
│  (p < α)            │  (True +)     │  (False +)    │
├─────────────────────┼───────────────┼───────────────┤
│  判定无差异          │  ✗ 第二类错误  │  ✓ 正确决策   │
│  (p ≥ α)            │  (False -)    │  (True -)     │
└─────────────────────┴───────────────┴───────────────┘
```

### 3.2 实现统计检验

```python
import numpy as np
from scipy import stats
from dataclasses import dataclass

@dataclass
class ABTestResult:
    """A/B 测试结果"""
    control_mean: float
    variant_mean: float
    control_std: float
    variant_std: float
    p_value: float
    confidence_interval: tuple[float, float]
    is_significant: bool
    effect_size: float
    recommendation: str

def run_t_test(control_scores: list[float], 
               variant_scores: list[float],
               alpha: float = 0.05) -> ABTestResult:
    """
    执行独立样本 t 检验
    
    Args:
        control_scores: 对照组的指标分数
        variant_scores: 实验组的指标分数
        alpha: 显著性水平
    """
    control = np.array(control_scores)
    variant = np.array(variant_scores)
    
    # t 检验
    t_stat, p_value = stats.ttest_ind(control, variant, equal_var=False)
    
    # 效应量（Cohen's d）
    pooled_std = np.sqrt((control.std()**2 + variant.std()**2) / 2)
    effect_size = (variant.mean() - control.mean()) / pooled_std if pooled_std > 0 else 0
    
    # 置信区间
    diff_mean = variant.mean() - control.mean()
    se = np.sqrt(control.var()/len(control) + variant.var()/len(variant))
    ci = stats.t.interval(1-alpha, df=min(len(control),len(variant))-1, 
                          loc=diff_mean, scale=se)
    
    is_significant = p_value < alpha
    
    # 生成建议
    if not is_significant:
        recommendation = "结果不显著，建议继续收集数据或增大样本量"
    elif effect_size > 0.8:
        recommendation = "效果显著且效应量大，建议采用实验组 Prompt"
    elif effect_size > 0.2:
        recommendation = "效果显著但效应量中等，建议结合业务判断"
    else:
        recommendation = "效果显著但效应量小，改进的实际意义有限"
    
    return ABTestResult(
        control_mean=float(control.mean()),
        variant_mean=float(variant.mean()),
        control_std=float(control.std()),
        variant_std=float(variant.std()),
        p_value=float(p_value),
        confidence_interval=ci,
        is_significant=is_significant,
        effect_size=float(effect_size),
        recommendation=recommendation
    )
```

### 3.3 贝叶斯 A/B 测试

```python
import numpy as np

def bayesian_ab_test(control_scores: list[float], 
                     variant_scores: list[float],
                     num_simulations: int = 100000) -> dict:
    """
    贝叶斯 A/B 测试
    返回实验组优于对照组的概率
    """
    control = np.array(control_scores)
    variant = np.array(variant_scores)
    
    # 使用 Bootstrap 重采样
    control_means = []
    variant_means = []
    
    for _ in range(num_simulations):
        control_sample = np.random.choice(control, size=len(control), replace=True)
        variant_sample = np.random.choice(variant, size=len(variant), replace=True)
        control_means.append(control_sample.mean())
        variant_means.append(variant_sample.mean())
    
    control_means = np.array(control_means)
    variant_means = np.array(variant_means)
    
    # 实验组优于对照组的概率
    prob_variant_better = float(np.mean(variant_means > control_means))
    
    # 期望损失（Expected Loss）
    loss_if_choose_control = float(np.mean(np.maximum(variant_means - control_means, 0)))
    loss_if_choose_variant = float(np.mean(np.maximum(control_means - variant_means, 0)))
    
    return {
        "prob_variant_better": prob_variant_better,
        "prob_control_better": 1 - prob_variant_better,
        "expected_loss_choose_control": loss_if_choose_control,
        "expected_loss_choose_variant": loss_if_choose_variant,
        "recommendation": (
            "采用实验组" if prob_variant_better > 0.95 
            else "继续测试" if prob_variant_better > 0.8 
            else "采用对照组" if prob_variant_better < 0.05 
            else "差异不明显"
        )
    }
```

## 4. 偏差识别与缓解

### 4.1 常见偏差类型

```
┌──────────────────────────────────────────────────────────┐
│                    A/B 测试常见偏差                        │
├──────────────┬───────────────────────────────────────────┤
│  选择偏差     │  样本不代表目标用户群体                     │
│  (Selection) │  缓解：随机分配 + 大样本                    │
├──────────────┼───────────────────────────────────────────┤
│  新奇效应     │  用户对新功能的短暂热情                     │
│  (Novelty)   │  缓解：延长测试时间，观察趋势               │
├──────────────┼───────────────────────────────────────────┤
│  时间偏差     │  不同时段的用户行为不同                     │
│  (Temporal)  │  缓解：覆盖完整周期（至少一周）             │
├──────────────┼───────────────────────────────────────────┤
│  幸存者偏差   │  只看到成功案例                            │
│  (Survivor)  │  缓解：记录所有请求，包括失败的             │
├──────────────┼───────────────────────────────────────────┤
│  辛普森悖论   │  整体趋势与分组趋势相反                     │
│  (Simpson's) │  缓解：分层分析 + 整体分析                  │
└──────────────┴───────────────────────────────────────────┘
```

### 4.2 偏差检测代码

```python
from datetime import datetime
from collections import defaultdict
import numpy as np

class BiasDetector:
    """A/B 测试偏差检测器"""
    
    def check_temporal_bias(self, timestamps: list[datetime], 
                            scores: list[float]) -> dict:
        """检测时间偏差"""
        # 按小时分组
        hourly = defaultdict(list)
        for ts, score in zip(timestamps, scores):
            hourly[ts.hour].append(score)
        
        hourly_means = {h: np.mean(s) for h, s in hourly.items()}
        
        # 检查不同时间段的分数是否有显著差异
        all_means = list(hourly_means.values())
        variance = np.var(all_means)
        
        return {
            "has_bias": variance > 0.01,  # 阈值可调
            "hourly_means": hourly_means,
            "variance": float(variance),
            "recommendation": (
                "检测到时间偏差，建议延长测试覆盖更多时段"
                if variance > 0.01
                else "未检测到明显时间偏差"
            )
        }
    
    def check_sample_balance(self, control_users: list[str],
                             variant_users: list[str],
                             user_features: dict[str, dict]) -> dict:
        """检测样本平衡性"""
        # 检查两组用户的特征分布是否相似
        control_features = [user_features.get(uid, {}) for uid in control_users]
        variant_features = [user_features.get(uid, {}) for uid in variant_users]
        
        # 简化：检查用户 ID 的哈希分布
        control_hash_mean = np.mean([hash(uid) % 1000 for uid in control_users])
        variant_hash_mean = np.mean([hash(uid) % 1000 for uid in variant_users])
        
        balance_ratio = min(control_hash_mean, variant_hash_mean) / max(
            control_hash_mean, variant_hash_mean)
        
        return {
            "is_balanced": balance_ratio > 0.9,
            "control_size": len(control_users),
            "variant_size": len(variant_users),
            "size_ratio": len(control_users) / max(1, len(variant_users)),
            "recommendation": (
                "样本分布均衡"
                if balance_ratio > 0.9
                else "样本分布不均衡，建议检查分配逻辑"
            )
        }
    
    def check_novelty_effect(self, scores_over_time: list[list[float]],
                             window_size: int = 100) -> dict:
        """检测新奇效应"""
        # 计算滑动窗口平均分
        windowed_means = []
        for i in range(0, len(scores_over_time) - window_size + 1, window_size):
            window = scores_over_time[i:i + window_size]
            windowed_means.append(np.mean(window))
        
        if len(windowed_means) < 3:
            return {"has_novelty": False, "note": "数据不足以检测"}
        
        # 检查前期分数是否显著高于后期
        first_third = np.mean(windowed_means[:len(windowed_means)//3])
        last_third = np.mean(windowed_means[-len(windowed_means)//3:])
        
        decline = first_third - last_third
        
        return {
            "has_novelty": decline > 0.05,
            "first_period_mean": float(first_third),
            "last_period_mean": float(last_third),
            "decline": float(decline),
            "recommendation": (
                "检测到新奇效应，建议延长测试直到指标稳定"
                if decline > 0.05
                else "未检测到明显新奇效应"
            )
        }
```

### 5.2 核心实现

```python
"""
A/B 测试框架 - 核心实现
"""
import json
import time
import hashlib
import numpy as np
from datetime import datetime
from pathlib import Path
from dataclasses import dataclass, field, asdict
from typing import Optional, Callable
from openai import OpenAI

# ============================================================
# 数据模型
# ============================================================

@dataclass
class PromptVariant:
    """Prompt 变体"""
    name: str
    system_prompt: str
    user_prompt_template: str
    temperature: float = 0.7
    max_tokens: int = 1000
    metadata: dict = field(default_factory=dict)

@dataclass
class ExperimentConfig:
    """实验配置"""
    experiment_id: str
    control: PromptVariant
    variant: PromptVariant
    traffic_split: float = 0.5  # 对照组比例
    min_sample_size: int = 100
    metrics: list[str] = field(default_factory=lambda: ["accuracy", "relevance"])
    alpha: float = 0.05
    salt: str = ""

@dataclass
class TestRecord:
    """单次测试记录"""
    experiment_id: str
    group: str  # control / variant
    user_id: str
    input_text: str
    response: str
    metrics: dict
    timestamp: str
    latency_ms: float

# ============================================================
# A/B 测试引擎
# ============================================================

class ABTestEngine:
    """A/B 测试引擎"""
    
    def __init__(self, results_dir: str = "./ab_results"):
        self.client = OpenAI()
        self.results_dir = Path(results_dir)
        self.results_dir.mkdir(parents=True, exist_ok=True)
        self.experiments: dict[str, ExperimentConfig] = {}
        self.records: dict[str, list[TestRecord]] = {}
    
    def create_experiment(self, config: ExperimentConfig) -> str:
        """创建新实验"""
        self.experiments[config.experiment_id] = config
        self.records[config.experiment_id] = []
        
        # 保存实验配置
        config_path = self.results_dir / f"{config.experiment_id}_config.json"
        with open(config_path, "w", encoding="utf-8") as f:
            json.dump(asdict(config), f, ensure_ascii=False, indent=2)
        
        return config.experiment_id
    
    def assign_group(self, experiment_id: str, user_id: str) -> str:
        """确定用户所属分组"""
        config = self.experiments[experiment_id]
        hash_input = f"{user_id}:{experiment_id}:{config.salt}".encode()
        hash_value = int(hashlib.md5(hash_input).hexdigest(), 16)
        normalized = (hash_value % 10000) / 10000
        
        return "control" if normalized < config.traffic_split else "variant"
    
    def run_test(self, experiment_id: str, user_id: str, 
                 input_text: str) -> TestRecord:
        """执行单次测试"""
        config = self.experiments[experiment_id]
        group = self.assign_group(experiment_id, user_id)
        variant = config.control if group == "control" else config.variant
        
        # 构建 prompt
        system_prompt = variant.system_prompt
        user_prompt = variant.user_prompt_template.replace("{input}", input_text)
        
        # 调用 LLM
        start_time = time.time()
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            temperature=variant.temperature,
            max_tokens=variant.max_tokens
        )
        latency_ms = (time.time() - start_time) * 1000
        
        result_text = response.choices[0].message.content
        
        # 计算指标
        metrics = self._compute_metrics(input_text, result_text, config.metrics)
        
        # 创建记录
        record = TestRecord(
            experiment_id=experiment_id,
            group=group,
            user_id=user_id,
            input_text=input_text,
            response=result_text,
            metrics=metrics,
            timestamp=datetime.now().isoformat(),
            latency_ms=latency_ms
        )
        
        self.records[experiment_id].append(record)
        self._save_record(record)
        
        return record
    
    def _compute_metrics(self, input_text: str, response: str, 
                         metric_names: list[str]) -> dict:
        """计算评估指标"""
        metrics = {}
        
        for name in metric_names:
            if name == "accuracy":
                # 简化的准确性评估
                eval_prompt = f"""评估以下回答的准确性（0-1分）：
                问题：{input_text}
                回答：{response}
                只输出数字分数。"""
                
                result = self.client.chat.completions.create(
                    model="gpt-4o-mini",
                    messages=[
                        {"role": "system", "content": "你是评估专家，只输出0-1的数字。"},
                        {"role": "user", "content": eval_prompt}
                    ],
                    temperature=0.0
                )
                try:
                    metrics[name] = float(result.choices[0].message.content.strip())
                except ValueError:
                    metrics[name] = 0.5
            
            elif name == "relevance":
                eval_prompt = f"""评估以下回答与问题的相关性（0-1分）：
                问题：{input_text}
                回答：{response}
                只输出数字分数。"""
                
                result = self.client.chat.completions.create(
                    model="gpt-4o-mini",
                    messages=[
                        {"role": "system", "content": "你是评估专家，只输出0-1的数字。"},
                        {"role": "user", "content": eval_prompt}
                    ],
                    temperature=0.0
                )
                try:
                    metrics[name] = float(result.choices[0].message.content.strip())
                except ValueError:
                    metrics[name] = 0.5
            
            elif name == "latency":
                metrics[name] = 0.0  # 在 run_test 中单独处理
        
        return metrics
    
    def _save_record(self, record: TestRecord):
        """保存测试记录"""
        records_path = self.results_dir / f"{record.experiment_id}_records.jsonl"
        with open(records_path, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(record), ensure_ascii=False) + "\n")
    
    def analyze(self, experiment_id: str) -> dict:
        """分析实验结果"""
        records = self.records[experiment_id]
        config = self.experiments[experiment_id]
        
        control_records = [r for r in records if r.group == "control"]
        variant_records = [r for r in records if r.group == "variant"]
        
        results = {
            "experiment_id": experiment_id,
            "total_records": len(records),
            "control_size": len(control_records),
            "variant_size": len(variant_records),
            "metrics": {}
        }
        
        # 对每个指标进行统计检验
        for metric_name in config.metrics:
            control_scores = [r.metrics.get(metric_name, 0) for r in control_records]
            variant_scores = [r.metrics.get(metric_name, 0) for r in variant_records]
            
            if len(control_scores) < 2 or len(variant_scores) < 2:
                results["metrics"][metric_name] = {"error": "样本量不足"}
                continue
            
            # t 检验
            t_stat, p_value = stats.ttest_ind(control_scores, variant_scores)
            
            # 效应量
            control_mean = np.mean(control_scores)
            variant_mean = np.mean(variant_scores)
            pooled_std = np.sqrt(
                (np.std(control_scores)**2 + np.std(variant_scores)**2) / 2
            )
            effect_size = (variant_mean - control_mean) / pooled_std if pooled_std > 0 else 0
            
            results["metrics"][metric_name] = {
                "control_mean": float(control_mean),
                "variant_mean": float(variant_mean),
                "improvement": float(variant_mean - control_mean),
                "improvement_pct": float((variant_mean - control_mean) / control_mean * 100) if control_mean > 0 else 0,
                "p_value": float(p_value),
                "effect_size": float(effect_size),
                "is_significant": p_value < config.alpha
            }
        
        # 总体结论
        significant_improvements = sum(
            1 for m in results["metrics"].values() 
            if isinstance(m, dict) and m.get("is_significant") and m.get("improvement", 0) > 0
        )
        
        if significant_improvements > len(config.metrics) / 2:
            results["conclusion"] = "实验组显著优于对照组，建议采用新 Prompt"
        elif significant_improvements > 0:
            results["conclusion"] = "部分指标有显著改进，需综合评估"
        else:
            results["conclusion"] = "无显著差异，建议继续测试或保持现有 Prompt"
        
        return results
    
    def generate_report(self, experiment_id: str) -> str:
        """生成实验报告"""
        results = self.analyze(experiment_id)
        
        report_lines = [
            f"# A/B 测试报告",
            f"",
            f"**实验 ID**: {results['experiment_id']}",
            f"**总样本数**: {results['total_records']}",
            f"**对照组**: {results['control_size']} 条",
            f"**实验组**: {results['variant_size']} 条",
            f"",
            f"## 指标分析",
            f"",
            f"| 指标 | 对照组均值 | 实验组均值 | 提升 | p值 | 显著性 |",
            f"|------|-----------|-----------|------|-----|--------|",
        ]
        
        for name, metric in results["metrics"].items():
            if "error" in metric:
                report_lines.append(f"| {name} | - | - | - | - | {metric['error']} |")
            else:
                sig = "✓" if metric["is_significant"] else "✗"
                report_lines.append(
                    f"| {name} | {metric['control_mean']:.3f} | "
                    f"{metric['variant_mean']:.3f} | "
                    f"{metric['improvement_pct']:+.1f}% | "
                    f"{metric['p_value']:.4f} | {sig} |"
                )
        
        report_lines.extend([
            f"",
            f"## 结论",
            f"",
            f"{results['conclusion']}",
        ])
        
        report = "\n".join(report_lines)
        
        # 保存报告
        report_path = self.results_dir / f"{experiment_id}_report.md"
        with open(report_path, "w", encoding="utf-8") as f:
            f.write(report)
        
        return report
```

## 6. 常见误区

### ❌ 错误1：样本量不足就下结论

```python
# 错误：只测试 10 个样本就做决定
if variant_mean > control_mean:
    adopt_variant()

# 正确：确保样本量足够，且通过统计检验
if len(scores) >= min_sample_size and p_value < 0.05:
    adopt_variant()
```

### ❌ 错误2：测试中途查看结果并提前终止

```python
# 错误：看到暂时领先就停止测试
if current_variant_wins > current_control_wins:
    stop_test()

# 正确：预设测试时长和样本量，到点再分析
if total_samples >= planned_sample_size:
    analyze()
```

### ❌ 错误3：同时运行多个互相影响的实验

```python
# 错误：同一用户同时参与多个实验
user -> experiment_A (测试 system prompt)
     -> experiment_B (测试 temperature)
     -> 无法区分改进来源

# 正确：使用正交实验设计或串行执行
```

### ❌ 错误4：忽略多重比较问题

```python
# 错误：测试 20 个指标，只报告显著的那几个
for metric in all_metrics:
    if p_value < 0.05:
        report(metric)  # 多重比较会增加假阳性

# 正真：使用 Bonferroni 校正
adjusted_alpha = 0.05 / len(all_metrics)  # 20个指标时 α = 0.0025
```

### ❌ 错误5：混淆统计显著性和实际显著性

```python
# 错误：p < 0.05 就采用
if p_value < 0.05:
    adopt_variant()

# 正确：同时考虑效应量
if p_value < 0.05 and effect_size > 0.2:  # 中等以上效应
    adopt_variant()
```

## 7. 工程建议

1. **先算样本量再开始测试**：用 `calculate_sample_size()` 预估所需样本量，避免测了 50 个样本就下结论——小样本的 A/B 测试比不做测试更危险，因为它会给你虚假的信心。
2. **测试周期至少覆盖一个完整业务周期**：如果用户行为有周内波动（工作日 vs 周末），测试时长至少一周；如果有月度波动，至少跑满一个月。
3. **同时关注统计显著性和效应量**：p < 0.05 只说明差异不是随机的，Cohen's d > 0.2 才说明差异有实际意义——统计显著但效应量微小的改进不值得上线。
4. **建立实验看板，避免偷看数据**：预设好测试时长和样本量，在测试结束前不要反复查看中间结果——每次偷看都会增加假阳性风险。

### 练习1：设计实验

**思路**：客服机器人 A/B 测试的核心是用科学方法对比两个 Prompt 版本在真实场景下的表现差异。需要明确控制变量（只改 system prompt），选择有业务意义的指标（用户满意度、问题解决率），并计算足够大的样本量以保证统计显著性。

**答案**：

**对照组 Prompt（版本 A - 专业简洁）：**
```
你是一个专业的客服助手。请简洁、准确地回答用户的问题。
规则：
1. 直接回答问题，不要多余寒暄
2. 如果无法回答，说明原因并建议联系人工客服
3. 保持专业和礼貌
```

**实验组 Prompt（版本 B - 温暖亲切）：**
```
你是一个温暖贴心的客服助手。请用亲切友好的方式帮助用户解决问题。
规则：
1. 先表达理解用户的感受，再回答问题
2. 如果无法回答，表达歉意并提供替代方案
3. 语气温暖但不失专业
```

**监控指标：**
- 主要指标：用户满意度评分（1-5 分，由用户在对话结束后评价）
- 次要指标：问题解决率（用户确认问题已解决的比例）、平均对话轮数、对话时长
- 护栏指标：投诉率（不能因优化满意度而增加投诉）

**样本量计算：**
```python
import math
from scipy import stats

def calculate_sample_size(baseline_rate, mde, alpha=0.05, power=0.8):
    effect_size = mde / math.sqrt(baseline_rate * (1 - baseline_rate))
    z_alpha = stats.norm.ppf(1 - alpha / 2)
    z_beta = stats.norm.ppf(power)
    n = ((z_alpha + z_beta) / effect_size) ** 2
    return math.ceil(n)

# 假设当前满意度基线为 3.5/5（即 70%），想检测 5% 的提升
n = calculate_sample_size(baseline_rate=0.70, minimum_detectable_effect=0.05)
print(f"每组需要至少 {n} 个样本")  # 约 1500+
```

每组至少需要约 1500 个有效对话样本，两组共需 3000+。假设日均 200 个客服对话，50% 流量分配给实验组，测试至少需要 30 天。

**要点**：
- 只改变 system prompt 这一个变量，temperature、模型版本等参数必须保持一致
- 测试周期至少覆盖一个完整业务周期（含工作日和周末），避免时间偏差
- 除了满意度，必须监控投诉率作为护栏指标——防止出现"满意度提升但投诉也增加"的情况

### 练习2：统计分析

**思路**：对 A/B 测试数据同时执行频率学派的 t 检验和贝叶斯分析，从两个角度验证结论的可靠性。t 检验关注 p 值和效应量，贝叶斯分析关注实验组优于对照组的概率。

**答案**：

```python
import numpy as np
from scipy import stats

# 模拟测试数据
np.random.seed(42)
control_scores = np.random.normal(loc=0.72, scale=0.15, size=200)  # 对照组
control_scores = np.clip(control_scores, 0, 1)
variant_scores = np.random.normal(loc=0.76, scale=0.14, size=200)  # 实验组
variant_scores = np.clip(variant_scores, 0, 1)

# === 1. t 检验 ===
t_stat, p_value = stats.ttest_ind(control_scores, variant_scores, equal_var=False)

# Cohen's d 效应量
pooled_std = np.sqrt((np.std(control_scores)**2 + np.std(variant_scores)**2) / 2)
cohens_d = (np.mean(variant_scores) - np.mean(control_scores)) / pooled_std

# 置信区间
diff_mean = np.mean(variant_scores) - np.mean(control_scores)
se = np.sqrt(np.var(control_scores)/len(control_scores) + np.var(variant_scores)/len(variant_scores))
ci = stats.t.interval(0.95, df=min(len(control_scores), len(variant_scores))-1,
                       loc=diff_mean, scale=se)

print("=== t 检验结果 ===")
print(f"对照组均值: {np.mean(control_scores):.4f}")
print(f"实验组均值: {np.mean(variant_scores):.4f}")
print(f"差异: {diff_mean:.4f}")
print(f"t 统计量: {t_stat:.4f}")
print(f"p 值: {p_value:.6f}")
print(f"Cohen's d: {cohens_d:.4f}")
print(f"95% 置信区间: ({ci[0]:.4f}, {ci[1]:.4f})")
print(f"显著性 (α=0.05): {'是' if p_value < 0.05 else '否'}")

# === 2. 贝叶斯分析 ===
num_simulations = 100000
control_means = []
variant_means = []

for _ in range(num_simulations):
    c_sample = np.random.choice(control_scores, size=len(control_scores), replace=True)
    v_sample = np.random.choice(variant_scores, size=len(variant_scores), replace=True)
    control_means.append(c_sample.mean())
    variant_means.append(v_sample.mean())

control_means = np.array(control_means)
variant_means = np.array(variant_means)

prob_variant_better = float(np.mean(variant_means > control_means))
loss_choose_control = float(np.mean(np.maximum(variant_means - control_means, 0)))
loss_choose_variant = float(np.mean(np.maximum(control_means - variant_means, 0)))

print("\n=== 贝叶斯分析结果 ===")
print(f"实验组优于对照组的概率: {prob_variant_better:.4f}")
print(f"选择对照组的期望损失: {loss_choose_control:.6f}")
print(f"选择实验组的期望损失: {loss_choose_variant:.6f}")

# === 3. 综合建议 ===
print("\n=== 综合建议 ===")
if p_value < 0.05 and cohens_d > 0.2 and prob_variant_better > 0.95:
    print("强烈建议采用实验组：统计显著、效应量中等以上、贝叶斯概率高")
elif p_value < 0.05 and cohens_d > 0.2:
    print("建议采用实验组：统计显著且效应量有实际意义")
elif p_value < 0.05 and cohens_d <= 0.2:
    print("谨慎决策：统计显著但效应量小，改进的实际意义有限")
elif prob_variant_better > 0.9:
    print("贝叶斯分析支持实验组，但频率学检验未达显著，建议增加样本量")
else:
    print("暂不采用：两种方法均未给出强烈信号，建议继续测试或调整方案")
```

**要点**：
- 频率学派和贝叶斯方法应同时使用——t 检验告诉你"差异是否显著"，贝叶斯告诉你"实验组更好的概率是多少"
- Cohen's d > 0.8 为大效应，0.2-0.8 为中等效应，< 0.2 为小效应——统计显著但效应量微小的改进不值得上线
- 贝叶斯分析的"期望损失"是更实用的决策指标——它能量化"选错方案的代价有多大"

### 练习3：实现完整框架

**思路**：在课程代码基础上扩展三个能力：多变体支持（A/B/C/...）、自动化评估集成、偏差检测。核心改动是将双组设计改为 N 组设计，流量分配从二分改为多路分配。

**答案**：

```python
import json
import time
import hashlib
import numpy as np
from datetime import datetime
from pathlib import Path
from dataclasses import dataclass, field, asdict
from typing import Optional
from scipy import stats
from openai import OpenAI

@dataclass
class PromptVariant:
    name: str
    system_prompt: str
    user_prompt_template: str
    temperature: float = 0.7
    max_tokens: int = 1000

@dataclass
class MultiVariantConfig:
    experiment_id: str
    variants: list[PromptVariant]
    traffic_weights: list[float]  # 每个变体的流量比例，总和为 1
    min_sample_size: int = 100
    metrics: list[str] = field(default_factory=lambda: ["accuracy", "relevance"])
    alpha: float = 0.05

    def __post_init__(self):
        if len(self.variants) != len(self.traffic_weights):
            raise ValueError("变体数量与流量权重数量不匹配")
        if abs(sum(self.traffic_weights) - 1.0) > 0.01:
            raise ValueError("流量权重总和必须为 1.0")


class MultiVariantTestEngine:
    """多变体 A/B/C/... 测试引擎"""

    def __init__(self, results_dir: str = "./ab_results"):
        self.client = OpenAI()
        self.results_dir = Path(results_dir)
        self.results_dir.mkdir(parents=True, exist_ok=True)
        self.experiments: dict[str, MultiVariantConfig] = {}
        self.records: dict[str, list[dict]] = {}

    def create_experiment(self, config: MultiVariantConfig) -> str:
        self.experiments[config.experiment_id] = config
        self.records[config.experiment_id] = []
        return config.experiment_id

    def assign_variant(self, experiment_id: str, user_id: str) -> str:
        config = self.experiments[experiment_id]
        hash_input = f"{user_id}:{experiment_id}".encode()
        hash_value = int(hashlib.md5(hash_input).hexdigest(), 16)
        normalized = (hash_value % 10000) / 10000

        cumulative = 0.0
        for i, weight in enumerate(config.traffic_weights):
            cumulative += weight
            if normalized < cumulative:
                return config.variants[i].name
        return config.variants[-1].name

    def run_test(self, experiment_id: str, user_id: str,
                 input_text: str) -> dict:
        config = self.experiments[experiment_id]
        variant_name = self.assign_variant(experiment_id, user_id)
        variant = next(v for v in config.variants if v.name == variant_name)

        user_prompt = variant.user_prompt_template.replace("{input}", input_text)

        start_time = time.time()
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": variant.system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            temperature=variant.temperature,
            max_tokens=variant.max_tokens
        )
        latency_ms = (time.time() - start_time) * 1000

        result_text = response.choices[0].message.content

        record = {
            "experiment_id": experiment_id,
            "variant": variant_name,
            "user_id": user_id,
            "input": input_text,
            "response": result_text,
            "latency_ms": latency_ms,
            "timestamp": datetime.now().isoformat()
        }

        self.records[experiment_id].append(record)
        return record

    def analyze(self, experiment_id: str) -> dict:
        config = self.experiments[experiment_id]
        records = self.records[experiment_id]

        # 按变体分组
        by_variant = {}
        for r in records:
            if r["variant"] not in by_variant:
                by_variant[r["variant"]] = []
            by_variant[r["variant"]].append(r)

        # 统计每个变体的样本量和延迟
        summary = {}
        for vname, vrecords in by_variant.items():
            latencies = [r["latency_ms"] for r in vrecords]
            summary[vname] = {
                "count": len(vrecords),
                "avg_latency_ms": float(np.mean(latencies)),
                "p95_latency_ms": float(np.percentile(latencies, 95))
            }

        # 多组比较（ANOVA，如果有评分数据的话）
        return {
            "experiment_id": experiment_id,
            "total_records": len(records),
            "variant_summary": summary,
            "variants": list(by_variant.keys())
        }

    def generate_report(self, experiment_id: str) -> str:
        results = self.analyze(experiment_id)

        lines = [
            "# 多变体 A/B 测试报告",
            "",
            f"**实验 ID**: {results['experiment_id']}",
            f"**总样本数**: {results['total_records']}",
            "",
            "## 变体统计",
            "",
            "| 变体 | 样本数 | 平均延迟 | P95 延迟 |",
            "|------|--------|---------|---------|",
        ]

        for vname, stats in results["variant_summary"].items():
            lines.append(
                f"| {vname} | {stats['count']} | "
                f"{stats['avg_latency_ms']:.0f}ms | "
                f"{stats['p95_latency_ms']:.0f}ms |"
            )

        return "\n".join(lines)
```

**要点**：
- 多变体测试的流量分配必须使用累积概率法，而不是简单的取模分配——否则变体数量变化时分配逻辑会出错
- 多组比较时应使用 ANOVA 而非多次 t 检验——多次两两比较会显著增加假阳性率
- 变体数量不宜超过 5 个——每个变体都需要足够的样本量，变体越多总样本需求越大
