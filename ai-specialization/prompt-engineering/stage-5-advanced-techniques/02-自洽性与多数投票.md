## 1. 什么是自洽性（Self-Consistency）？

Self-Consistency 是一种提升 LLM 推理准确率的技术。核心思想：

> **对同一问题生成多条推理路径，然后通过多数投票选择最一致的答案。**

```
╔══════════════════════════════════════════════════════════════════════════════╗
║                    Self-Consistency 工作流程                                 ║
╠══════════════════════════════════════════════════════════════════════════════╣
║                                                                            ║
║                        ┌─────────────────┐                                 ║
║                        │    用户问题     │                                 ║
║                        └────────┬────────┘                                 ║
║                                 │                                          ║
║              ┌──────────────────┼──────────────────┐                       ║
║              ▼                  ▼                  ▼                       ║
║     ┌────────────────┐ ┌────────────────┐ ┌────────────────┐              ║
║     │  推理路径 1    │ │  推理路径 2    │ │  推理路径 3    │              ║
║     │  (CoT + 答案) │ │  (CoT + 答案) │ │  (CoT + 答案) │              ║
║     └───────┬────────┘ └───────┬────────┘ └───────┬────────┘              ║
║             │                  │                  │                        ║
║             ▼                  ▼                  ▼                        ║
║     ┌─────────────────────────────────────────────────────┐               ║
║     │                   多数投票                           │               ║
║     │   答案A: 2票  │  答案B: 1票  │  答案C: 0票        │               ║
║     └─────────────────────────────────────────────────────┘               ║
║                                 │                                          ║
║                                 ▼                                          ║
║                        ┌─────────────────┐                                 ║
║                        │   最终答案：A   │                                 ║
║                        │   置信度：66.7% │                                 ║
║                        └─────────────────┘                                 ║
║                                                                            ║
╚══════════════════════════════════════════════════════════════════════════════╝
```

### 1.1 为什么 Self-Consistency 有效？

| 特性 | 单次 CoT | Self-Consistency |
|------|----------|------------------|
| 准确率 | 基准 | +10-20% 提升 |
| 容错性 | 单点故障 | 多路径容错 |
| 成本 | 低 | N 倍（N 条路径） |
| 适用场景 | 简单问题 | 复杂推理任务 |

## 2. 采样策略

### 2.1 温度采样

```python
import openai
import json
from typing import List, Dict, Tuple
from collections import Counter

client = openai.OpenAI()

class SelfConsistencyChain:
    """自洽性推理链"""

    def __init__(
        self,
        model: str = "gpt-4",
        num_paths: int = 5,
        temperature: float = 0.7
    ):
        self.model = model
        self.num_paths = num_paths
        self.temperature = temperature

    def generate_single_path(
        self,
        question: str,
        system_prompt: str = None
    ) -> Dict[str, str]:
        """生成单条推理路径"""
        if system_prompt is None:
            system_prompt = """你是一个逻辑推理专家。请一步一步思考问题，最后给出答案。

输出格式：
推理过程：[详细的推理步骤]
最终答案：[简洁的答案]"""

        response = client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": question}
            ],
            temperature=self.temperature
        )

        content = response.choices[0].message.content
        return self._parse_reasoning(content)

    def _parse_reasoning(self, content: str) -> Dict[str, str]:
        """解析推理路径"""
        result = {
            "reasoning": "",
            "answer": "",
            "raw": content
        }

        lines = content.strip().split("\n")
        for line in lines:
            if line.startswith("推理过程：") or line.startswith("推理过程:"):
                result["reasoning"] = line.split("：", 1)[-1].strip()
            elif line.startswith("最终答案：") or line.startswith("最终答案:"):
                result["answer"] = line.split("：", 1)[-1].strip()

        # 如果没有明确标记，尝试从最后提取答案
        if not result["answer"]:
            result["answer"] = lines[-1].strip() if lines else ""

        return result

    def generate_multiple_paths(
        self,
        question: str,
        system_prompt: str = None
    ) -> List[Dict[str, str]]:
        """生成多条推理路径"""
        paths = []
        for i in range(self.num_paths):
            print(f"生成路径 {i+1}/{self.num_paths}...")
            path = self.generate_single_path(question, system_prompt)
            path["path_id"] = i + 1
            paths.append(path)
        return paths

    def majority_vote(self, paths: List[Dict[str, str]]) -> Tuple[str, float, Dict]:
        """多数投票"""
        answers = [p["answer"] for p in paths]
        counter = Counter(answers)

        # 获取最常见的答案
        most_common = counter.most_common(1)[0]
        best_answer = most_common[0]
        vote_count = most_common[1]
        confidence = vote_count / len(paths)

        # 统计详情
        vote_details = {
            "total_paths": len(paths),
            "unique_answers": len(counter),
            "vote_distribution": dict(counter),
            "confidence": confidence
        }

        return best_answer, confidence, vote_details

    def solve(
        self,
        question: str,
        system_prompt: str = None
    ) -> Dict[str, any]:
        """完整的 Self-Consistency 求解流程"""
        print(f"\n问题: {question}")
        print(f"采样路径数: {self.num_paths}")
        print("-" * 50)

        # 生成多条路径
        paths = self.generate_multiple_paths(question, system_prompt)

        # 多数投票
        answer, confidence, details = self.majority_vote(paths)

        return {
            "question": question,
            "answer": answer,
            "confidence": confidence,
            "details": details,
            "paths": paths
        }


# 使用示例
if __name__ == "__main__":
    sc = SelfConsistencyChain(
        model="gpt-4",
        num_paths=5,
        temperature=0.7
    )

    question = "一个商店打8折促销，原价200元的商品，先涨价25%再打8折，实际价格是多少？"

    result = sc.solve(question)

    print(f"\n最终答案: {result['answer']}")
    print(f"置信度: {result['confidence']:.2%}")
    print(f"\n投票分布:")
    for answer, count in result['details']['vote_distribution'].items():
        print(f"  {answer}: {count}票")
```

### 3.1 投票策略对比

```
║                       投票策略对比                                           ║
╠══════════════════════════════════════════════════════════════════════════════╣
║                                                                            ║
║   1. 简单多数投票 (Simple Majority)                                         ║
║      ┌─────────────────────────────────┐                                   ║
║      │ 答案A: 3票  答案B: 2票  答案C: 1票 │ → 选择 A                       ║
║      └─────────────────────────────────┘                                   ║
║                                                                            ║
║   2. 加权投票 (Weighted Voting)                                             ║
║      ┌─────────────────────────────────┐                                   ║
║      │ 答案A: 0.8+0.7+0.6 = 2.1       │                                   ║
║      │ 答案B: 0.9+0.5      = 1.4       │ → 选择 A（按置信度加权）           ║
║      │ 答案C: 0.3          = 0.3       │                                   ║
║      └─────────────────────────────────┘                                   ║
║                                                                            ║
║   3. 阈值投票 (Threshold Voting)                                            ║
║      ┌─────────────────────────────────┐                                   ║
║      │ 阈值: 60%                        │                                   ║
║      │ 答案A: 5/8 = 62.5% ✓            │ → 选择 A（超过阈值）              ║
║      │ 答案B: 3/8 = 37.5%              │                                   ║
```

### 3.2 实现多种投票机制

```python
from typing import Callable
import numpy as np

class VotingMechanism:
    """投票机制集合"""

    @staticmethod
    def simple_majority(paths: List[Dict]) -> Tuple[str, float]:
        """简单多数投票"""
        answers = [p["answer"] for p in paths]
        counter = Counter(answers)
        most_common = counter.most_common(1)[0]
        return most_common[0], most_common[1] / len(paths)

    @staticmethod
    def weighted_voting(paths: List[Dict]) -> Tuple[str, float]:
        """加权投票（基于推理路径的置信度）"""
        weight_map = {}

        for path in paths:
            answer = path["answer"]
            # 从推理过程中提取置信度（简化：基于推理长度和关键词）
            weight = VotingMechanism._calculate_path_weight(path)

            if answer not in weight_map:
                weight_map[answer] = 0
            weight_map[answer] += weight

        # 归一化
        total_weight = sum(weight_map.values())
        normalized = {k: v/total_weight for k, v in weight_map.items()}

        best_answer = max(normalized, key=normalized.get)
        return best_answer, normalized[best_answer]

    @staticmethod
    def _calculate_path_weight(path: Dict) -> float:
        """计算路径权重"""
        reasoning = path.get("reasoning", "")
        weight = 1.0

        # 基于推理长度
        if len(reasoning) > 100:
            weight += 0.2

        # 基于关键词
        positive_keywords = ["因此", "所以", "综上", "计算得出", "验证"]
        for keyword in positive_keywords:
            if keyword in reasoning:
                weight += 0.1

        return weight

    @staticmethod
    def threshold_voting(
        paths: List[Dict],
        threshold: float = 0.6
    ) -> Tuple[str, float, bool]:
        """阈值投票"""
        answers = [p["answer"] for p in paths]
        counter = Counter(answers)

        most_common = counter.most_common(1)[0]
        confidence = most_common[1] / len(paths)

        if confidence >= threshold:
            return most_common[0], confidence, True
        else:
            return most_common[0], confidence, False

    @staticmethod
    def bayesian_voting(paths: List[Dict]) -> Tuple[str, float]:
        """贝叶斯投票（考虑先验概率）"""
        # 假设先验：较短的答案通常更可能是正确答案
        answer_scores = {}

        for path in paths:
            answer = path["answer"]
            # 先验分数（基于答案长度的启发式）
            prior = 1.0 / (1.0 + len(answer) / 100)

            # 似然（基于推理质量）
            likelihood = VotingMechanism._calculate_path_weight(path)

            # 后验 = 先验 * 似然
            posterior = prior * likelihood

            if answer not in answer_scores:
                answer_scores[answer] = 0
            answer_scores[answer] += posterior

        # 归一化
        total = sum(answer_scores.values())
        normalized = {k: v/total for k, v in answer_scores.items()}

        best_answer = max(normalized, key=normalized.get)
        return best_answer, normalized[best_answer]


class AdvancedSelfConsistency(SelfConsistencyChain):
    """高级自洽性推理"""

    def __init__(
        self,
        voting_strategy: str = "majority",
        threshold: float = 0.6,
        **kwargs
    ):
        super().__init__(**kwargs)
        self.voting_strategy = voting_strategy
        self.threshold = threshold
        self.voting = VotingMechanism()

    def solve(self, question: str, system_prompt: str = None) -> Dict:
        """使用指定投票策略求解"""
        # 生成多条路径
        paths = self.generate_multiple_paths(question, system_prompt)

        # 应用投票策略
        if self.voting_strategy == "majority":
            answer, confidence = self.voting.simple_majority(paths)
            return {
                "question": question,
                "answer": answer,
                "confidence": confidence,
                "strategy": "simple_majority",
                "paths": paths
            }

        elif self.voting_strategy == "weighted":
            answer, confidence = self.voting.weighted_voting(paths)
            return {
                "question": question,
                "answer": answer,
                "confidence": confidence,
                "strategy": "weighted_voting",
                "paths": paths
            }

        elif self.voting_strategy == "threshold":
            answer, confidence, reached = self.voting.threshold_voting(
                paths, self.threshold
            )
            return {
                "question": question,
                "answer": answer,
                "confidence": confidence,
                "threshold_reached": reached,
                "strategy": "threshold_voting",
                "paths": paths
            }

        elif self.voting_strategy == "bayesian":
            answer, confidence = self.voting.bayesian_voting(paths)
            return {
                "question": question,
                "answer": answer,
                "confidence": confidence,
                "strategy": "bayesian_voting",
                "paths": paths
            }

        else:
            raise ValueError(f"未知投票策略: {self.voting_strategy}")
```

## 4. 置信度校准

### 4.1 为什么需要校准？

模型输出的置信度往往不准确。校准的目的是让置信度反映真实准确率。

```
╔══════════════════════════════════════════════════════════════════════════════╗
║                       置信度校准                                             ║
╠══════════════════════════════════════════════════════════════════════════════╣
║                                                                            ║
║   校准前：                                                                  ║
║   ┌─────────────────────────────────────────────────────┐                  ║
║   │ 模型置信度: 90%  →  实际准确率: 70%  (过度自信)     │                  ║
║   │ 模型置信度: 60%  →  实际准确率: 55%  (略过度自信)   │                  ║
║   └─────────────────────────────────────────────────────┘                  ║
║                                                                            ║
║   校准后：                                                                  ║
║   ┌─────────────────────────────────────────────────────┐                  ║
║   │ 校准置信度: 75%  →  实际准确率: 75%  (完美校准)     │                  ║
║   │ 校准置信度: 55%  →  实际准确率: 55%  (完美校准)     │                  ║
║   └─────────────────────────────────────────────────────┘                  ║
```

### 4.2 校准实现

```python
class ConfidenceCalibrator:
    """置信度校准器"""

    def __init__(self):
        self.calibration_data = []  # (predicted_confidence, actual_correct)

    def add_sample(self, predicted_confidence: float, actual_correct: bool):
        """添加校准样本"""
        self.calibration_data.append((predicted_confidence, actual_correct))

    def calibrate(self, num_bins: int = 10) -> Dict[str, List]:
        """执行校准，返回校准曲线数据"""
        if not self.calibration_data:
            return {"bin_centers": [], "bin_accuracies": [], "bin_counts": []}

        # 分箱
        bins = np.linspace(0, 1, num_bins + 1)
        bin_centers = []
        bin_accuracies = []
        bin_counts = []

        for i in range(num_bins):
            low, high = bins[i], bins[i+1]
            bin_samples = [
                (conf, correct)
                for conf, correct in self.calibration_data
                if low <= conf < high
            ]

            if bin_samples:
                bin_center = (low + high) / 2
                bin_accuracy = sum(c for _, c in bin_samples) / len(bin_samples)
                bin_count = len(bin_samples)

                bin_centers.append(bin_center)
                bin_accuracies.append(bin_accuracy)
                bin_counts.append(bin_count)

        return {
            "bin_centers": bin_centers,
            "bin_accuracies": bin_accuracies,
            "bin_counts": bin_counts
        }

    def expected_calibration_error(self) -> float:
        """计算期望校准误差 (ECE)"""
        calibration = self.calibrate()
        ece = 0
        total_samples = sum(calibration["bin_counts"])

        for center, accuracy, count in zip(
            calibration["bin_centers"],
            calibration["bin_accuracies"],
            calibration["bin_counts"]
        ):
            ece += (count / total_samples) * abs(center - accuracy)

        return ece

    def temperature_scale(self, logits: List[float], temperature: float = 1.5) -> List[float]:
        """温度缩放校准"""
        # 简化的温度缩放
        scaled = [l / temperature for l in logits]
        # Softmax
        exp_scaled = [np.exp(l) for l in scaled]
        total = sum(exp_scaled)
        return [e / total for e in exp_scaled]


class CalibratedSelfConsistency(AdvancedSelfConsistency):
    """带校准的自洽性推理"""

    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.calibrator = ConfidenceCalibrator()

    def solve_with_calibration(
        self,
        question: str,
        expected_answer: str = None
    ) -> Dict:
        """求解并进行校准"""
        result = self.solve(question)

        # 如果有期望答案，用于校准
        if expected_answer:
            is_correct = expected_answer.lower() in result["answer"].lower()
            self.calibrator.add_sample(result["confidence"], is_correct)
            result["is_correct"] = is_correct

        return result

    def get_calibration_stats(self) -> Dict:
        """获取校准统计"""
        return {
            "ece": self.calibrator.expected_calibration_error(),
            "calibration_curve": self.calibrator.calibrate(),
            "sample_count": len(self.calibrator.calibration_data)
        }
```

## 5. 完整 Pipeline 实现

```python
class SelfConsistencyPipeline:
    """完整的自洽性推理 Pipeline"""

    def __init__(
        self,
        model: str = "gpt-4",
        num_paths: int = 5,
        temperature: float = 0.7,
        voting_strategy: str = "majority",
        enable_calibration: bool = True
    ):
        self.solver = CalibratedSelfConsistency(
            model=model,
            num_paths=num_paths,
            temperature=temperature,
            voting_strategy=voting_strategy
        )
        self.enable_calibration = enable_calibration
        self.results_history = []

    def solve_batch(
        self,
        questions: List[Dict[str, str]]
    ) -> List[Dict]:
        """批量求解"""
        results = []

        for i, q in enumerate(questions):
            print(f"\n{'='*60}")
            print(f"问题 {i+1}/{len(questions)}")
            print(f"{'='*60}")

            question = q["question"]
            expected = q.get("expected_answer")

            if self.enable_calibration:
                result = self.solver.solve_with_calibration(question, expected)
            else:
                result = self.solver.solve(question)

            results.append(result)
            self.results_history.append(result)

        return results

    def evaluate(self, results: List[Dict]) -> Dict:
        """评估批量结果"""
        if not results:
            return {}

        # 计算准确率
        correct = sum(
            1 for r in results
            if r.get("is_correct", False)
        )
        accuracy = correct / len(results)

        # 计算平均置信度
        avg_confidence = sum(
            r["confidence"] for r in results
        ) / len(results)

        # 置信度分布
        confidence_distribution = {
            "high (>0.8)": sum(1 for r in results if r["confidence"] > 0.8),
            "medium (0.5-0.8)": sum(1 for r in results if 0.5 <= r["confidence"] <= 0.8),
            "low (<0.5)": sum(1 for r in results if r["confidence"] < 0.5)
        }

        eval_result = {
            "total_questions": len(results),
            "accuracy": accuracy,
            "average_confidence": avg_confidence,
            "confidence_distribution": confidence_distribution
        }

        # 校准统计
        if self.enable_calibration:
            eval_result["calibration"] = self.solver.get_calibration_stats()

        return eval_result

    def print_report(self, results: List[Dict]):
        """打印评估报告"""
        eval_result = self.evaluate(results)

        print("\n" + "="*60)
        print("自洽性推理评估报告")
        print("="*60)
        print(f"总问题数: {eval_result['total_questions']}")
        print(f"准确率: {eval_result['accuracy']:.2%}")
        print(f"平均置信度: {eval_result['average_confidence']:.2%}")
        print(f"\n置信度分布:")
        for level, count in eval_result['confidence_distribution'].items():
            print(f"  {level}: {count}")

        if "calibration" in eval_result:
            print(f"\n校准误差 (ECE): {eval_result['calibration']['ece']:.4f}")


# 使用示例
if __name__ == "__main__":
    pipeline = SelfConsistencyPipeline(
        model="gpt-4",
        num_paths=5,
        temperature=0.7,
        voting_strategy="majority",
        enable_calibration=True
    )

    questions = [
        {
            "question": "小明有 15 个苹果，给了小红 3 个，又买了 8 个，现在有多少个？",
            "expected_answer": "20"
        },
        {
            "question": "一个正方形的周长是 20 厘米，面积是多少？",
            "expected_answer": "25"
        },
        {
            "question": "如果 3x + 7 = 22，那么 x 等于多少？",
            "expected_answer": "5"
        }
    ]

    results = pipeline.solve_batch(questions)
    pipeline.print_report(results)
```

## 6. 常见误区

| 错误 | 正确做法 |
|------|----------|
| 采样路径太少（<3） | 至少 5 条路径以获得可靠投票 |
| 温度过低（<0.5） | 温度 0.7-1.0 确保路径多样性 |
| 忽略置信度校准 | 在验证集上校准置信度 |
| 所有路径用相同 Prompt | 可变化 Prompt 增加多样性 |
| 不分析失败案例 | 记录并分析投票失败的情况 |

---

## 7. 工程建议

1. 路径数量选择需权衡成本与效果，生产环境建议 5-7 条路径，关键决策可增加到 10 条
2. 对于需要高可靠性的场景，设置置信度阈值——低于阈值时自动触发人工审核
3. 使用不同温度参数混合采样（如 0.5、0.7、1.0），兼顾推理质量和路径多样性
4. 记录每次投票的详细分布，用于分析模型的薄弱环节和优化方向

## 8. 本节小结

Self-Consistency 是提升 LLM 推理准确率的强大技术：

1. **多路径采样**：通过温度/Top-p 采样生成多样化的推理路径
2. **投票机制**：简单多数、加权、阈值、贝叶斯等多种策略
3. **置信度校准**：让置信度真实反映准确率
4. **完整 Pipeline**：批量求解 + 评估 + 报告

## 参考答案

### 练习 1

**思路**：基础 Self-Consistency 的实现分三步：对同一问题用较高温度（0.7-1.0）采样 N 次生成不同推理路径；从每条路径中提取最终答案；对答案进行多数投票，票数最多的即为最终结果。关键是温度不能太低，否则路径缺乏多样性，投票就失去意义。

**答案**：

```python
import openai
from typing import List, Dict, Tuple
from collections import Counter

client = openai.OpenAI()

class BasicSelfConsistency:
    """基础 Self-Consistency 系统"""

    def __init__(self, model: str = "gpt-4", num_paths: int = 5, temperature: float = 0.7):
        self.model = model
        self.num_paths = num_paths
        self.temperature = temperature

    def generate_path(self, question: str) -> Dict[str, str]:
        """生成单条推理路径"""
        response = client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": """你是一个数学推理专家。请一步一步思考问题，最后给出答案。

输出格式：
推理过程：[详细的推理步骤]
最终答案：[简洁的数字或表达式]"""},
                {"role": "user", "content": question}
            ],
            temperature=self.temperature
        )
        content = response.choices[0].message.content
        return self._parse(content)

    def _parse(self, content: str) -> Dict[str, str]:
        """解析推理路径，提取推理过程和答案"""
        result = {"reasoning": "", "answer": "", "raw": content}
        for line in content.strip().split("\n"):
            if "推理过程" in line:
                result["reasoning"] = line.split("：", 1)[-1].strip()
            elif "最终答案" in line:
                result["answer"] = line.split("：", 1)[-1].strip()
        if not result["answer"]:
            lines = content.strip().split("\n")
            result["answer"] = lines[-1].strip() if lines else ""
        return result

    def solve(self, question: str) -> Dict:
        """完整求解流程：多路径采样 + 多数投票"""
        paths = []
        for i in range(self.num_paths):
            path = self.generate_path(question)
            path["path_id"] = i + 1
            paths.append(path)
            print(f"路径 {i+1}: 答案={path['answer']}")

        # 多数投票
        answers = [p["answer"] for p in paths]
        counter = Counter(answers)
        most_common = counter.most_common(1)[0]

        return {
            "question": question,
            "answer": most_common[0],
            "confidence": most_common[1] / self.num_paths,
            "vote_distribution": dict(counter),
            "paths": paths
        }


if __name__ == "__main__":
    sc = BasicSelfConsistency(model="gpt-4", num_paths=5, temperature=0.7)

    question = "小明有 15 个苹果，给了小红 3 个，又买了 8 个，现在有多少个？"
    result = sc.solve(question)

    print(f"\n最终答案: {result['answer']}")
    print(f"置信度: {result['confidence']:.2%}")
    print(f"投票分布: {result['vote_distribution']}")
```

**要点**：
- 温度是 Self-Consistency 的关键参数——太低（<0.5）路径缺乏多样性，太高（>1.0）路径质量下降，0.7 是常用起点
- 路径数量建议 5 条以上，少于 3 条时投票结果不够可靠
- 答案提取逻辑需要健壮处理各种格式，否则投票时会因为格式差异导致同一答案被当作不同答案

### 练习 2

**思路**：对比三种投票策略需要在同一组问题上分别运行，使用相同的采样路径（而非重新采样），以确保公平对比。简单多数直接数票数，加权投票根据推理质量给不同路径赋权重，贝叶斯投票结合先验概率和推理质量。通过准确率和校准误差来评判哪种策略更优。

**答案**：

```python
import openai
import json
from typing import List, Dict, Tuple
from collections import Counter

client = openai.OpenAI()

class VotingComparison:
    """投票策略对比系统"""

    def __init__(self, model: str = "gpt-4", num_paths: int = 5):
        self.model = model
        self.num_paths = num_paths

    def generate_paths(self, question: str) -> List[Dict]:
        """为一个问题生成多条推理路径"""
        paths = []
        for i in range(self.num_paths):
            response = client.chat.completions.create(
                model=self.model,
                messages=[
                    {"role": "system", "content": "请一步步推理，最后给出答案。格式：\n推理过程：...\n最终答案：..."},
                    {"role": "user", "content": question}
                ],
                temperature=0.7
            )
            content = response.choices[0].message.content
            answer = ""
            reasoning = ""
            for line in content.split("\n"):
                if "推理过程" in line:
                    reasoning = line.split("：", 1)[-1].strip()
                elif "最终答案" in line:
                    answer = line.split("：", 1)[-1].strip()
            if not answer:
                answer = content.split("\n")[-1].strip()
            paths.append({"answer": answer, "reasoning": reasoning, "raw": content})
        return paths

    def simple_majority(self, paths: List[Dict]) -> Tuple[str, float]:
        """简单多数投票"""
        answers = [p["answer"] for p in paths]
        counter = Counter(answers)
        best = counter.most_common(1)[0]
        return best[0], best[1] / len(paths)

    def weighted_voting(self, paths: List[Dict]) -> Tuple[str, float]:
        """加权投票：根据推理质量赋权"""
        weight_map = {}
        for path in paths:
            weight = 1.0
            reasoning = path.get("reasoning", "")
            if len(reasoning) > 50:
                weight += 0.3
            for kw in ["因此", "所以", "计算得出", "验证"]:
                if kw in reasoning:
                    weight += 0.1
            answer = path["answer"]
            weight_map[answer] = weight_map.get(answer, 0) + weight

        total = sum(weight_map.values())
        normalized = {k: v / total for k, v in weight_map.items()}
        best = max(normalized, key=normalized.get)
        return best, normalized[best]

    def bayesian_voting(self, paths: List[Dict]) -> Tuple[str, float]:
        """贝叶斯投票：结合先验（答案简洁度）和似然（推理质量）"""
        score_map = {}
        for path in paths:
            prior = 1.0 / (1.0 + len(path["answer"]) / 50)
            likelihood = 1.0
            for kw in ["因此", "所以", "计算得出", "验证"]:
                if kw in path.get("reasoning", ""):
                    likelihood += 0.15
            posterior = prior * likelihood
            answer = path["answer"]
            score_map[answer] = score_map.get(answer, 0) + posterior

        total = sum(score_map.values())
        normalized = {k: v / total for k, v in score_map.items()}
        best = max(normalized, key=normalized.get)
        return best, normalized[best]

    def compare(self, questions: List[Dict[str, str]]) -> Dict:
        """对比三种投票策略"""
        strategies = {
            "简单多数": self.simple_majority,
            "加权投票": self.weighted_voting,
            "贝叶斯投票": self.bayesian_voting,
        }
        results = {name: {"correct": 0, "total": 0} for name in strategies}

        for q in questions:
            question = q["question"]
            expected = q["expected_answer"]
            paths = self.generate_paths(question)

            for name, strategy in strategies.items():
                answer, confidence = strategy(paths)
                is_correct = expected in answer
                results[name]["total"] += 1
                if is_correct:
                    results[name]["correct"] += 1
                print(f"  [{name}] 答案={answer}, 正确={is_correct}")

        print("\n对比结果：")
        for name, data in results.items():
            accuracy = data["correct"] / data["total"] if data["total"] else 0
            print(f"  {name}: 准确率 {accuracy:.2%} ({data['correct']}/{data['total']})")

        return results


if __name__ == "__main__":
    vc = VotingComparison(num_paths=5)
    questions = [
        {"question": "一个正方形的周长是 20 厘米，面积是多少？", "expected_answer": "25"},
        {"question": "如果 3x + 7 = 22，那么 x 等于多少？", "expected_answer": "5"},
        {"question": "小明有 15 个苹果，给了小红 3 个，又买了 8 个，现在有多少个？", "expected_answer": "20"},
    ]
    vc.compare(questions)
```

**要点**：
- 对比必须使用相同的采样路径，否则结果差异可能来自随机性而非策略本身
- 加权和贝叶斯策略依赖推理质量的启发式判断，这些启发式需要根据具体任务调优
- 在数学推理场景中，简单多数投票通常已经足够好；加权策略在推理质量差异大的场景中更有优势

### 练习 3

**思路**：温度缩放校准的核心思想是：模型输出的置信度往往偏高（过度自信），通过一个温度参数 T 对 logits 进行缩放（logits / T），使置信度分布更平滑。校准曲线将预测置信度分箱，比较每个箱内的平均置信度和实际准确率。ECE（期望校准误差）是各箱误差的加权平均。

**答案**：

```python
import numpy as np
from typing import List, Dict

class TemperatureScalingCalibrator:
    """温度缩放置信度校准器"""

    def __init__(self):
        self.calibration_data = []  # (predicted_confidence, is_correct)
        self.temperature = 1.0

    def add_sample(self, confidence: float, is_correct: bool):
        """添加校准样本"""
        self.calibration_data.append((confidence, is_correct))

    def find_optimal_temperature(self, logits: List[float], labels: List[bool]) -> float:
        """通过网格搜索找到最优温度参数"""
        best_temp = 1.0
        best_ece = float("inf")

        for temp in np.arange(0.5, 5.0, 0.1):
            scaled = self._temperature_scale(logits, temp)
            ece = self._compute_ece_from_logits(scaled, labels)
            if ece < best_ece:
                best_ece = ece
                best_temp = temp

        self.temperature = best_temp
        print(f"最优温度: {best_temp:.2f}, ECE: {best_ece:.4f}")
        return best_temp

    def _temperature_scale(self, logits: List[float], temperature: float) -> List[float]:
        """温度缩放"""
        scaled = [l / temperature for l in logits]
        exp_scaled = [np.exp(l) for l in scaled]
        total = sum(exp_scaled)
        return [e / total for e in exp_scaled]

    def _compute_ece_from_logits(self, probabilities: List[float], labels: List[bool], num_bins: int = 10) -> float:
        """计算 ECE"""
        bins = np.linspace(0, 1, num_bins + 1)
        ece = 0
        total = len(probabilities)

        for i in range(num_bins):
            low, high = bins[i], bins[i + 1]
            indices = [j for j, p in enumerate(probabilities) if low <= p < high]
            if not indices:
                continue
            bin_probs = [probabilities[j] for j in indices]
            bin_labels = [labels[j] for j in indices]
            avg_conf = np.mean(bin_probs)
            avg_acc = np.mean([1.0 if l else 0.0 for l in bin_labels])
            ece += len(indices) / total * abs(avg_conf - avg_acc)

        return ece

    def calibrate(self, num_bins: int = 10) -> Dict:
        """执行校准并返回校准曲线数据"""
        if not self.calibration_data:
            return {"bin_centers": [], "bin_accuracies": [], "bin_counts": []}

        bins = np.linspace(0, 1, num_bins + 1)
        bin_centers, bin_accuracies, bin_counts = [], [], []

        for i in range(num_bins):
            low, high = bins[i], bins[i + 1]
            samples = [(c, ok) for c, ok in self.calibration_data if low <= c < high]
            if samples:
                bin_centers.append((low + high) / 2)
                bin_accuracies.append(sum(ok for _, ok in samples) / len(samples))
                bin_counts.append(len(samples))

        return {
            "bin_centers": bin_centers,
            "bin_accuracies": bin_accuracies,
            "bin_counts": bin_counts
        }

    def expected_calibration_error(self) -> float:
        """计算 ECE"""
        cal = self.calibrate()
        total = sum(cal["bin_counts"])
        ece = 0
        for center, acc, count in zip(cal["bin_centers"], cal["bin_accuracies"], cal["bin_counts"]):
            ece += (count / total) * abs(center - acc)
        return ece

    def plot_calibration_curve(self):
        """绘制校准曲线（使用 matplotlib）"""
        try:
            import matplotlib.pyplot as plt
        except ImportError:
            print("需要安装 matplotlib: pip install matplotlib")
            return

        cal = self.calibrate()
        if not cal["bin_centers"]:
            print("没有校准数据。")
            return

        fig, ax = plt.subplots(1, 1, figsize=(6, 6))
        ax.plot([0, 1], [0, 1], "k--", label="完美校准")
        ax.plot(cal["bin_centers"], cal["bin_accuracies"], "bo-", label="模型校准曲线")
        ax.set_xlabel("预测置信度")
        ax.set_ylabel("实际准确率")
        ax.set_title(f"校准曲线 (ECE={self.expected_calibration_error():.4f})")
        ax.legend()
        ax.grid(True)
        plt.tight_layout()
        plt.savefig("calibration_curve.png", dpi=150)
        print("校准曲线已保存为 calibration_curve.png")


if __name__ == "__main__":
    calibrator = TemperatureScalingCalibrator()

    # 模拟校准样本：模型置信度偏高
    np.random.seed(42)
    for _ in range(100):
        confidence = np.random.uniform(0.6, 1.0)
        is_correct = np.random.random() < (confidence - 0.15)  # 实际准确率比置信度低
        calibrator.add_sample(confidence, is_correct)

    print(f"校准前 ECE: {calibrator.expected_calibration_error():.4f}")
    calibrator.plot_calibration_curve()
```

**要点**：
- 温度 T > 1 会使置信度分布更平滑（降低过度自信），T < 1 会使分布更尖锐
- ECE 越小越好，0 表示完美校准——预测置信度 80% 时，实际准确率也应该是 80%
- 校准数据必须来自验证集而非训练集，否则会过拟合校准参数
