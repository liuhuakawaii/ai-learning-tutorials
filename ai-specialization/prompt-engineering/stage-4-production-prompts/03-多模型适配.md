## 场景引入

你的团队用 GPT-4o 精心调试了一套 Prompt，效果非常好。老板说："Claude 更便宜，能不能换？"你把同一套 Prompt 原封不动发给 Claude，结果输出格式乱了、JSON 解析失败、甚至连角色设定都不遵守。你又试了 Gemini，发现它对 XML 标签结构完全无感，反而在超长文档分析上碾压其他模型。同一个业务需求，三个模型三套表现——你需要一套系统化的多模型适配方案。

## 3. 实现模型适配器

```python
"""
model_adapter.py
多模型适配器 - 统一接口，自动翻译 Prompt
"""

import os
import json
from abc import ABC, abstractmethod
from typing import List, Dict, Optional, Any, Iterator
from dataclasses import dataclass, field
from enum import Enum


class ModelProvider(Enum):
    OPENAI = "openai"
    ANTHROPIC = "anthropic"
    GOOGLE = "google"
    QWEN = "qwen"


@dataclass
class UnifiedRequest:
    """统一请求格式"""
    messages: List[Dict[str, str]]
    model: str
    temperature: float = 0.7
    max_tokens: int = 2000
    system_prompt: Optional[str] = None
    json_mode: bool = False
    tools: Optional[List[Dict]] = None
    stream: bool = False


@dataclass
class UnifiedResponse:
    """统一响应格式"""
    content: str
    model: str
    provider: ModelProvider
    input_tokens: int = 0
    output_tokens: int = 0
    finish_reason: str = "stop"
    raw_response: Any = None


class BaseModelAdapter(ABC):
    """模型适配器基类"""

    @abstractmethod
    def translate_messages(self, request: UnifiedRequest) -> List[Dict]:
        """翻译消息格式"""
        pass

    @abstractmethod
    def call_api(self, request: UnifiedRequest) -> UnifiedResponse:
        """调用 API"""
        pass

    @abstractmethod
    def stream_api(self, request: UnifiedRequest) -> Iterator[str]:
        """流式调用 API"""
        pass

    def adapt_prompt_for_model(self, prompt: str) -> str:
        """针对模型特性调整 Prompt"""
        return prompt


class OpenAIAdapter(BaseModelAdapter):
    """OpenAI GPT 系列适配器"""

    def __init__(self, api_key: Optional[str] = None):
        from openai import OpenAI
        self.client = OpenAI(api_key=api_key or os.getenv("OPENAI_API_KEY"))

    def translate_messages(self, request: UnifiedRequest) -> List[Dict]:
        messages = []
        if request.system_prompt:
            messages.append({"role": "system", "content": request.system_prompt})
        messages.extend(request.messages)
        return messages

    def call_api(self, request: UnifiedRequest) -> UnifiedResponse:
        messages = self.translate_messages(request)
        kwargs = {
            "model": request.model,
            "messages": messages,
            "temperature": request.temperature,
            "max_tokens": request.max_tokens,
        }
        if request.json_mode:
            kwargs["response_format"] = {"type": "json_object"}
        if request.tools:
            kwargs["tools"] = request.tools

        response = self.client.chat.completions.create(**kwargs)
        choice = response.choices[0]

        return UnifiedResponse(
            content=choice.message.content or "",
            model=response.model,
            provider=ModelProvider.OPENAI,
            input_tokens=response.usage.prompt_tokens,
            output_tokens=response.usage.completion_tokens,
            finish_reason=choice.finish_reason,
            raw_response=response,
        )

    def stream_api(self, request: UnifiedRequest) -> Iterator[str]:
        messages = self.translate_messages(request)
        stream = self.client.chat.completions.create(
            model=request.model,
            messages=messages,
            temperature=request.temperature,
            max_tokens=request.max_tokens,
            stream=True,
        )
        for chunk in stream:
            if chunk.choices[0].delta.content:
                yield chunk.choices[0].delta.content

    def adapt_prompt_for_model(self, prompt: str) -> str:
        """GPT 系列偏好明确的指令格式"""
        return prompt


class AnthropicAdapter(BaseModelAdapter):
    """Anthropic Claude 适配器"""

    def __init__(self, api_key: Optional[str] = None):
        import anthropic
        self.client = anthropic.Anthropic(
            api_key=api_key or os.getenv("ANTHROPIC_API_KEY")
        )

    def translate_messages(self, request: UnifiedRequest) -> List[Dict]:
        """Claude 的 system 是独立参数，消息中不包含 system"""
        messages = []
        for msg in request.messages:
            if msg["role"] != "system":
                messages.append(msg)
        return messages

    def call_api(self, request: UnifiedRequest) -> UnifiedResponse:
        messages = self.translate_messages(request)
        response = self.client.messages.create(
            model=request.model,
            max_tokens=request.max_tokens,
            system=request.system_prompt or "",
            messages=messages,
            temperature=request.temperature,
        )
        return UnifiedResponse(
            content=response.content[0].text,
            model=response.model,
            provider=ModelProvider.ANTHROPIC,
            input_tokens=response.usage.input_tokens,
            output_tokens=response.usage.output_tokens,
            finish_reason=response.stop_reason,
            raw_response=response,
        )

    def stream_api(self, request: UnifiedRequest) -> Iterator[str]:
        messages = self.translate_messages(request)
        with self.client.messages.stream(
            model=request.model,
            max_tokens=request.max_tokens,
            system=request.system_prompt or "",
            messages=messages,
            temperature=request.temperature,
        ) as stream:
            for text in stream.text_stream:
                yield text

    def adapt_prompt_for_model(self, prompt: str) -> str:
        """Claude 偏好 XML 标签结构化"""
        if "<instructions>" not in prompt:
            prompt = f"<instructions>\n{prompt}\n</instructions>"
        return prompt


class GoogleAdapter(BaseModelAdapter):
    """Google Gemini 适配器"""

    def __init__(self, api_key: Optional[str] = None):
        import google.generativeai as genai
        genai.configure(api_key=api_key or os.getenv("GOOGLE_API_KEY"))
        self.genai = genai

    def translate_messages(self, request: UnifiedRequest) -> List[Dict]:
        messages = []
        for msg in request.messages:
            role = "user" if msg["role"] == "user" else "model"
            messages.append({"role": role, "parts": [msg["content"]]})
        return messages

    def call_api(self, request: UnifiedRequest) -> UnifiedResponse:
        model = self.genai.GenerativeModel(
            model_name=request.model,
            system_instruction=request.system_prompt,
        )
        messages = self.translate_messages(request)
        chat = model.start_chat(history=messages[:-1] if len(messages) > 1 else [])
        response = chat.send_message(messages[-1]["parts"][0])

        return UnifiedResponse(
            content=response.text,
            model=request.model,
            provider=ModelProvider.GOOGLE,
            input_tokens=response.usage_metadata.prompt_token_count,
            output_tokens=response.usage_metadata.candidates_token_count,
            finish_reason="stop",
            raw_response=response,
        )

    def stream_api(self, request: UnifiedRequest) -> Iterator[str]:
        model = self.genai.GenerativeModel(
            model_name=request.model,
            system_instruction=request.system_prompt,
        )
        messages = self.translate_messages(request)
        chat = model.start_chat(history=messages[:-1] if len(messages) > 1 else [])
        response = chat.send_message(messages[-1]["parts"][0], stream=True)
        for chunk in response:
            if chunk.text:
                yield chunk.text

    def adapt_prompt_for_model(self, prompt: str) -> str:
        """Gemini 偏好直接、简洁的指令"""
        prompt = prompt.replace("<instructions>", "").replace("</instructions>", "")
        return prompt


class ModelAdapterFactory:
    """适配器工厂"""

    _adapters = {
        ModelProvider.OPENAI: OpenAIAdapter,
        ModelProvider.ANTHROPIC: AnthropicAdapter,
        ModelProvider.GOOGLE: GoogleAdapter,
    }

    @classmethod
    def create(cls, provider: ModelProvider, **kwargs) -> BaseModelAdapter:
        adapter_class = cls._adapters.get(provider)
        if not adapter_class:
            raise ValueError(f"Unsupported provider: {provider}")
        return adapter_class(**kwargs)

    @classmethod
    def from_model_name(cls, model: str, **kwargs) -> BaseModelAdapter:
        """根据模型名自动选择适配器"""
        if model.startswith("gpt"):
            return cls.create(ModelProvider.OPENAI, **kwargs)
        elif model.startswith("claude"):
            return cls.create(ModelProvider.ANTHROPIC, **kwargs)
        elif model.startswith("gemini"):
            return cls.create(ModelProvider.GOOGLE, **kwargs)
        raise ValueError(f"Unknown model: {model}")


# 统一调用接口
class UnifiedLLM:
    """统一 LLM 调用接口"""

    def __init__(self):
        self._adapters: Dict[str, BaseModelAdapter] = {}

    def register_model(self, model: str, adapter: BaseModelAdapter):
        """注册模型适配器"""
        self._adapters[model] = adapter

    def call(self, request: UnifiedRequest) -> UnifiedResponse:
        adapter = self._adapters.get(request.model)
        if not adapter:
            adapter = ModelAdapterFactory.from_model_name(request.model)
            self._adapters[request.model] = adapter
        return adapter.call_api(request)

    def stream(self, request: UnifiedRequest) -> Iterator[str]:
        adapter = self._adapters.get(request.model)
        if not adapter:
            adapter = ModelAdapterFactory.from_model_name(request.model)
            self._adapters[request.model] = adapter
        return adapter.stream_api(request)


# 使用示例
if __name__ == "__main__":
    llm = UnifiedLLM()

    request = UnifiedRequest(
        messages=[{"role": "user", "content": "用一句话解释量子计算"}],
        model="gpt-4o-mini",
        system_prompt="你是一个科普专家",
        max_tokens=100,
    )

    # 统一接口调用不同模型
    # response = llm.call(request)
    # request.model = "claude-3-5-sonnet-20241022"
    # response = llm.call(request)

    print("适配器注册完成，可统一调用不同模型")
```

## 4. Prompt 翻译策略

```python
"""
prompt_translator.py
Prompt 跨模型翻译器
"""

from typing import Dict, List, Optional
from dataclasses import dataclass


@dataclass
class PromptTemplate:
    """Prompt 模板"""
    name: str
    content: str
    variables: List[str]
    model_preferences: Dict[str, str]  # model -> variant


class PromptTranslator:
    """Prompt 跨模型翻译器"""

    def __init__(self):
        self.model_styles = {
            "gpt-4o": {
                "instruction_format": "direct",
                "prefers_examples": True,
                "json_reliable": True,
                "max_effective_length": 8000,
            },
            "claude-3-5-sonnet": {
                "instruction_format": "xml_structured",
                "prefers_examples": True,
                "json_reliable": False,
                "max_effective_length": 12000,
            },
            "gemini-1.5-pro": {
                "instruction_format": "concise",
                "prefers_examples": False,
                "json_reliable": True,
                "max_effective_length": 16000,
            },
        }

    def translate(self, prompt: str, source_model: str,
                  target_model: str) -> str:
        """将 Prompt 从一个模型风格翻译到另一个"""
        source_style = self.model_styles.get(source_model, {})
        target_style = self.model_styles.get(target_model, {})

        result = prompt

        # 格式转换
        if source_style.get("instruction_format") == "xml_structured":
            result = self._strip_xml_tags(result)
        if target_style.get("instruction_format") == "xml_structured":
            result = self._add_xml_structure(result)

        # 长度适配
        max_len = target_style.get("max_effective_length", 8000)
        if len(result) > max_len:
            result = self._compress_for_model(result, max_len)

        return result

    def _strip_xml_tags(self, text: str) -> str:
        """移除 XML 标签"""
        import re
        return re.sub(r'</?[^>]+>', '', text).strip()

    def _add_xml_structure(self, text: str) -> str:
        """添加 XML 结构"""
        lines = text.split('\n')
        structured = []
        for line in lines:
            if line.strip().startswith(('1.', '2.', '3.', '-', '*')):
                structured.append(f"<item>{line}</item>")
            else:
                structured.append(line)
        return f"<instructions>\n{''.join(structured)}\n</instructions>"

    def _compress_for_model(self, text: str, max_length: int) -> str:
        """压缩到目标长度"""
        if len(text) <= max_length:
            return text
        return text[:max_length - 20] + "\n\n[内容已截断]"

    def generate_model_specific(self, template: PromptTemplate,
                                 model: str) -> str:
        """根据模型生成特定版本"""
        variant = template.model_preferences.get(model)
        if variant:
            return variant

        content = template.content
        style = self.model_styles.get(model, {})

        if style.get("instruction_format") == "xml_structured":
            content = self._add_xml_structure(content)
        if not style.get("prefers_examples", True):
            content = self._remove_examples(content)

        return content

    def _remove_examples(self, text: str) -> str:
        """移除示例段落"""
        import re
        text = re.sub(r'(?:例如|示例|Example).*?(?=\n\n|\Z)', '', text, flags=re.DOTALL)
        return text.strip()


# 使用示例
if __name__ == "__main__":
    translator = PromptTranslator()

    claude_prompt = """<instructions>
请分析以下文本的情感倾向。
<constraints>
- 只输出正面、负面或中性
- 给出置信度分数
</constraints>
</instructions>"""

    gpt_version = translator.translate(claude_prompt, "claude-3-5-sonnet", "gpt-4o")
    print("GPT 版本:")
    print(gpt_version)
```

## 6. 常见误区

### ❌ 错误 1：假设所有模型行为一致

```python
# 错误：同一 Prompt 直接用于不同模型
for model in ["gpt-4o", "claude-3-5-sonnet", "gemini-1.5-pro"]:
    response = call(model, same_prompt)  # 结果可能差异巨大

# 正确：针对模型特性调整 Prompt
for model in models:
    adapted = translator.translate(prompt, "gpt-4o", model)
    response = call(model, adapted)
```

### ❌ 错误 2：忽略模型的 System Prompt 差异

```python
# 错误：Claude 的 system 是独立参数，不是消息
response = claude.messages.create(
    messages=[{"role": "system", "content": "..."}]  # 不推荐
)
# 正确：使用 system 参数
response = claude.messages.create(system="...", messages=[...])
```

### ❌ 错误 3：硬编码模型特定的输出格式

```python
# 错误：假设所有模型都能可靠输出 JSON
prompt = "输出 JSON 格式"
# 某些模型可能输出带 markdown 代码块的 JSON

# 正确：添加解析容错
try:
    result = json.loads(response)
except json.JSONDecodeError:
    # 尝试提取代码块中的 JSON
    match = re.search(r'```json\n(.*?)\n```', response, re.DOTALL)
    result = json.loads(match.group(1))
```

## 7. 模型特性适配对照表

| Prompt 特性 | GPT-4o | Claude | Gemini | 适配建议 |
|------------|--------|--------|--------|---------|
| 指令格式 | 直接描述 | XML 标签 | 简洁直接 | 按目标模型调整 |
| Few-shot | 效果好 | 效果好 | 一般 | 复杂任务加示例 |
| JSON 输出 | 原生支持 | 需引导 | 原生支持 | 添加解析容错 |
| 长文本 | 128K | 200K | 1M | 按模型截断 |
| 角色扮演 | 良好 | 优秀 | 一般 | Claude 更灵活 |
| 代码生成 | 优秀 | 优秀 | 良好 | 复杂代码用前两者 |

## 8. 工程建议

1. **每次切换模型都要重新测试 Prompt**：不要假设同一 Prompt 在不同模型上效果一致。建立标准化测试集，切换模型前跑一遍回归测试，确保质量不下降。

2. **为每个模型维护独立的 Prompt 变体**：通过适配器模式统一管理，`adapt_prompt_for_model()` 方法封装模型特定的格式调整，业务代码不感知底层差异。

3. **JSON 输出必须添加解析容错**：不同模型的 JSON 可靠性差异很大，Claude 可能输出带 XML 标签的 JSON，Gemini 可能省略引号。解析层要做好正则提取和降级处理。

4. **建立模型选择决策矩阵**：按任务类型（简单问答/复杂推理/代码生成）、成本预算、延迟要求三个维度，预定义模型选择规则，避免每次都要人工判断。

## 总结

多模型适配的核心是**抽象 + 翻译**：通过统一接口屏蔽底层差异，通过 Prompt 翻译适配不同模型的偏好。选择模型时需要在质量、成本、延迟之间找到平衡点。

```python
"""

阿里 Qwen-Max 模型适配器
"""

import os

@dataclass
class UnifiedRequest:
    messages: List[Dict[str, str]]
    model: str
    temperature: float = 0.7
    max_tokens: int = 2000
    system_prompt: Optional[str] = None
    json_mode: bool = False
    stream: bool = False


@dataclass
class UnifiedResponse:
    content: str
    model: str
    provider: str
    input_tokens: int = 0
    output_tokens: int = 0
    finish_reason: str = "stop"


class QwenAdapter:
    """阿里 Qwen-Max 适配器"""

    def __init__(self, api_key: Optional[str] = None,
                 base_url: str = "https://dashscope.aliyuncs.com/compatible-mode/v1"):
        from openai import OpenAI
        self.client = OpenAI(
            api_key=api_key or os.getenv("DASHSCOPE_API_KEY"),
            base_url=base_url,
        )

    def translate_messages(self, request: UnifiedRequest) -> List[Dict]:
        """翻译消息格式 - Qwen 兼容 OpenAI 格式"""
        messages = []
        if request.system_prompt:
            adapted_system = self._adapt_system_prompt(request.system_prompt)
            messages.append({"role": "system", "content": adapted_system})
        messages.extend(request.messages)
        return messages

    def _adapt_system_prompt(self, prompt: str) -> str:
        """适配系统提示词 - 增强中文指令遵循"""
        if "请用中文" not in prompt and "用中文回答" not in prompt:
            prompt = prompt + "\n请始终使用中文回答。"
        return prompt

    def call_api(self, request: UnifiedRequest) -> UnifiedResponse:
        messages = self.translate_messages(request)
        kwargs = {
            "model": request.model,
            "messages": messages,
            "temperature": request.temperature,
            "max_tokens": request.max_tokens,
        }
        if request.json_mode:
            kwargs["response_format"] = {"type": "json_object"}

        response = self.client.chat.completions.create(**kwargs)
        choice = response.choices[0]

        return UnifiedResponse(
            content=choice.message.content or "",
            model=response.model,
            provider="qwen",
            input_tokens=response.usage.prompt_tokens,
            output_tokens=response.usage.completion_tokens,
            finish_reason=choice.finish_reason,
        )

    def stream_api(self, request: UnifiedRequest) -> Iterator[str]:
        messages = self.translate_messages(request)
        stream = self.client.chat.completions.create(
            model=request.model,
            messages=messages,
            temperature=request.temperature,
            max_tokens=request.max_tokens,
            stream=True,
        )
        for chunk in stream:
            if chunk.choices[0].delta.content:
                yield chunk.choices[0].delta.content

    def adapt_prompt_for_model(self, prompt: str) -> str:
        """Qwen 中文优化适配"""
        # Qwen 对中文指令理解更好，无需 XML 标签
        # 移除其他模型的格式标记
        prompt = prompt.replace("<instructions>", "").replace("</instructions>", "")
        # 添加中文格式引导
        if "JSON" in prompt and "json" not in prompt.lower():
            prompt += "\n请直接输出 JSON，不要添加 markdown 代码块标记。"
        return prompt
```

### 练习 2：A/B 测试框架

**思路**：构建一个 `ModelABTester`，将同一组测试用例分发到多个模型，记录每个模型的输出质量、延迟、Token 用量，最后生成对比报告。

**答案**：

```python
"""
model_ab_tester.py
模型 A/B 测试框架
"""

import time
import json
from typing import List, Dict, Callable, Optional
from dataclasses import dataclass, field
from collections import defaultdict


@dataclass
class TestCase:
    """测试用例"""
    id: str
    input_text: str
    expected_keywords: List[str] = field(default_factory=list)
    category: str = "general"


@dataclass
class ModelResult:
    """单个模型的测试结果"""
    model: str
    test_id: str
    output: str
    latency_ms: float
    input_tokens: int
    output_tokens: int
    cost: float
    contains_keywords: bool = False
    quality_score: float = 0.0


class ModelABTester:
    """模型 A/B 测试框架"""

    def __init__(self):
        self._models: Dict[str, Callable] = {}
        self._results: Dict[str, List[ModelResult]] = defaultdict(list)

    def register_model(self, name: str, call_fn: Callable):
        """注册模型调用函数
        
        call_fn 签名: (messages: List[Dict]) -> Dict
        返回: {"content": str, "input_tokens": int, "output_tokens": int, "cost": float}
        """
        self._models[name] = call_fn

    def run_test(self, test_cases: List[TestCase],
                 system_prompt: str = "") -> Dict:
        """对所有注册模型运行测试"""
        for test in test_cases:
            messages = []
            if system_prompt:
                messages.append({"role": "system", "content": system_prompt})
            messages.append({"role": "user", "content": test.input_text})

            for model_name, call_fn in self._models.items():
                start = time.perf_counter()
                try:
                    response = call_fn(messages)
                    latency = (time.perf_counter() - start) * 1000

                    output = response.get("content", "")
                    contains_kw = all(
                        kw.lower() in output.lower()
                        for kw in test.expected_keywords
                    ) if test.expected_keywords else True

                    quality = self._calculate_quality(
                        output, test.expected_keywords
                    )

                    result = ModelResult(
                        model=model_name,
                        test_id=test.id,
                        output=output,
                        latency_ms=latency,
                        input_tokens=response.get("input_tokens", 0),
                        output_tokens=response.get("output_tokens", 0),
                        cost=response.get("cost", 0.0),
                        contains_keywords=contains_kw,
                        quality_score=quality,
                    )
                except Exception as e:
                    result = ModelResult(
                        model=model_name,
                        test_id=test.id,
                        output=f"ERROR: {str(e)}",
                        latency_ms=(time.perf_counter() - start) * 1000,
                        input_tokens=0,
                        output_tokens=0,
                        cost=0.0,
                        quality_score=0.0,
                    )

                self._results[model_name].append(result)

        return self.generate_report()

    def _calculate_quality(self, output: str,
                           expected_keywords: List[str]) -> float:
        """计算质量分"""
        if not expected_keywords:
            return 0.8

        matched = sum(
            1 for kw in expected_keywords
            if kw.lower() in output.lower()
        )
        keyword_score = matched / len(expected_keywords)

        length_score = min(len(output) / 100, 1.0)

        return keyword_score * 0.7 + length_score * 0.3

    def generate_report(self) -> Dict:
        """生成对比报告"""
        report = {}
        for model_name, results in self._results.items():
            latencies = [r.latency_ms for r in results]
            costs = [r.cost for r in results]
            qualities = [r.quality_score for r in results]
            keyword_hits = sum(1 for r in results if r.contains_keywords)

            report[model_name] = {
                "total_tests": len(results),
                "keyword_hit_rate": f"{keyword_hits / len(results):.1%}" if results else "N/A",
                "avg_quality": f"{sum(qualities) / len(qualities):.3f}" if results else "N/A",
                "avg_latency_ms": f"{sum(latencies) / len(latencies):.1f}" if results else "N/A",
                "p95_latency_ms": f"{sorted(latencies)[int(len(latencies) * 0.95)]:.1f}" if len(latencies) > 1 else "N/A",
                "total_cost": f"${sum(costs):.4f}",
                "avg_cost": f"${sum(costs) / len(costs):.6f}" if results else "N/A",
            }

        return report

    def print_comparison(self):
        """打印对比结果"""
        report = self.generate_report()
        print("=" * 70)
        print("模型 A/B 测试对比报告")
        print("=" * 70)
        for model, stats in report.items():
            print(f"\n[{model}]")
            for key, value in stats.items():
                print(f"  {key}: {value}")

    tester = ModelABTester()

    def mock_gpt4o_mini(messages):
        return {
            "content": "机器学习是AI的分支，让计算机从数据中学习。",
            "input_tokens": 50, "output_tokens": 30, "cost": 0.0001,
        }

    def mock_claude(messages):
        return {
            "content": "机器学习（Machine Learning）是人工智能的一个核心分支，通过算法让计算机从数据中自动学习规律。",
            "input_tokens": 50, "output_tokens": 50, "cost": 0.0002,
        }

    tester.register_model("gpt-4o-mini", mock_gpt4o_mini)
    tester.register_model("claude-3-5-sonnet", mock_claude)

    test_cases = [
        TestCase(id="t1", input_text="什么是机器学习？",
                 expected_keywords=["机器学习", "学习", "数据"]),
        TestCase(id="t2", input_text="解释深度学习",
                 expected_keywords=["深度学习", "神经网络"]),
    ]

    tester.run_test(test_cases, system_prompt="你是一个AI技术专家")
    tester.print_comparison()
```

**要点**：
- A/B 测试框架的核心是统一接口：同一组测试用例、同样的调用方式，只换模型
- 质量评估应该多维度：关键词命中率、输出长度、格式正确性
- 报告需要包含成本对比，因为模型选择不只是质量问题，还有经济考量

### 练习 3：智能路由

**思路**：根据任务类型（简单问答、复杂推理、代码生成）自动选择最合适的模型。使用关键词匹配和规则引擎判断任务类型，结合成本预算和延迟要求做最终决策。

**答案**：

```python
"""
smart_router.py
智能路由器 - 根据任务类型自动选择模型
"""

import re
from typing import Dict, List, Optional
from dataclasses import dataclass
from enum import Enum


class TaskType(Enum):
    SIMPLE_QA = "simple_qa"
    COMPLEX_REASONING = "complex_reasoning"
    CODE_GENERATION = "code_generation"
    LONG_DOCUMENT = "long_document"
    CHINESE_OPTIMIZED = "chinese_optimized"


@dataclass
class ModelProfile:
    """模型能力画像"""
    name: str
    strengths: List[TaskType]
    cost_per_1k_input: float
    cost_per_1k_output: float
    avg_latency_ms: float
    max_context_tokens: int
    quality_score: float  # 0-1


class SmartRouter:
    """智能路由器"""

    def __init__(self):
        self._models: Dict[str, ModelProfile] = {}
        self._rules: List[Dict] = []

    def register_model(self, profile: ModelProfile):
        """注册模型"""
        self._models[profile.name] = profile

    def classify_task(self, user_input: str,
                      context_length: int = 0) -> TaskType:
        """分类任务类型"""
        input_lower = user_input.lower()

        code_keywords = [
            "代码", "函数", "编程", "实现", "代码审查", "bug", "修复",
            "code", "function", "implement", "debug", "refactor",
            "python", "javascript", "typescript", "java", "go",
        ]
        if any(kw in input_lower for kw in code_keywords):
            return TaskType.CODE_GENERATION

        reasoning_keywords = [
            "分析", "推理", "比较", "评估", "论证", "为什么", "如何",
            "analyze", "reason", "compare", "evaluate", "why", "how",
            "策略", "方案", "设计", "架构",
        ]
        if any(kw in input_lower for kw in reasoning_keywords):
            return TaskType.COMPLEX_REASONING

        if context_length > 50000 or "文档" in input_lower or "长文" in input_lower:
            return TaskType.LONG_DOCUMENT

        chinese_chars = len(re.findall(r'[\u4e00-\u9fff]', user_input))
        if chinese_chars / max(len(user_input), 1) > 0.5:
            return TaskType.CHINESE_OPTIMIZED

        return TaskType.SIMPLE_QA

    def route(self, user_input: str, context_length: int = 0,
              budget_priority: str = "balanced",
              latency_priority: str = "normal") -> Dict:
        """
        路由到最合适的模型

        Args:
            budget_priority: "low" | "balanced" | "quality"
            latency_priority: "fast" | "normal" | "quality"
        """
        task_type = self.classify_task(user_input, context_length)

        candidates = []
        for name, profile in self._models.items():
            score = self._score_model(profile, task_type,
                                      budget_priority, latency_priority)
            candidates.append((name, score, profile))

        candidates.sort(key=lambda x: x[1], reverse=True)

        if not candidates:
            return {"error": "no suitable model found"}

        best_name, best_score, best_profile = candidates[0]

        return {
            "selected_model": best_name,
            "task_type": task_type.value,
            "score": best_score,
            "alternatives": [
                {"model": name, "score": score}
                for name, score, _ in candidates[1:3]
            ],
            "model_info": {
                "cost_input": best_profile.cost_per_1k_input,
                "cost_output": best_profile.cost_per_1k_output,
                "avg_latency_ms": best_profile.avg_latency_ms,
                "max_context": best_profile.max_context_tokens,
            },
        }

    def _score_model(self, profile: ModelProfile, task_type: TaskType,
                     budget_priority: str, latency_priority: str) -> float:
        """计算模型匹配分数"""
        score = 0.0

        if task_type in profile.strengths:
            score += 0.4

        score += profile.quality_score * 0.2

        if budget_priority == "low":
            cost_score = 1.0 - min(profile.cost_per_1k_input / 0.01, 1.0)
            score += cost_score * 0.3
        elif budget_priority == "quality":
            score += profile.quality_score * 0.2
        else:
            cost_score = 1.0 - min(profile.cost_per_1k_input / 0.01, 1.0)
            score += cost_score * 0.15
            score += profile.quality_score * 0.1

        if latency_priority == "fast":
            latency_score = 1.0 - min(profile.avg_latency_ms / 5000, 1.0)
            score += latency_score * 0.2
        elif latency_priority == "quality":
            score += profile.quality_score * 0.1

        return score
```

- 返回备选模型便于在首选模型不可用时快速降级
