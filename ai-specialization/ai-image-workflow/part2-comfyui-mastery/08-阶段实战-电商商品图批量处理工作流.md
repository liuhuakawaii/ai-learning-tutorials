# 第 14 课：阶段实战 — 电商商品图批量处理工作流

某服装电商每天要处理 200+ 张商品图：白底主图、场景效果图、模特展示图。人工处理每张 15 分钟（拍摄 + 抠图 + 调色 + 合成），200 张需要 50 小时人工工时。

目标：把单张处理时间从 15 分钟降到 2 分钟，总工时从 50 小时降到 7 小时（含人工审核）。这节课从需求分析开始，一步步搭建完整的工作流。

## 需求规格

```
三类商品图的规格：

类型      分辨率        背景          用途              质量要求
────────────────────────────────────────────────────────────────
白底图    800×800      纯白 #FFF     主图/搜索结果      边缘锐利，无锯齿
场景图    1200×1200    生活场景      社交媒体/广告      自然融合，风格统一
模特图    1024×1536    任意          详情页/品牌宣传    人脸一致，姿态自然

约束：
- 单张处理 < 3 分钟
- 失败率 < 5%
- 显存占用 < 16GB（RTX 4060 Ti）
```

## 完整 Pipeline 设计

```
输入队列 → 图像读取 → 质量预检 → 分割抠图 → 背景替换 → 超分修复 → 质量门控 → 输出
                │          │                      │
                │     不合格 → 人工队列              │
                │                                   │
                └───────────────────────────────────┘
```

每个模块对应一个或多个 ComfyUI 节点，通过 Python 脚本串联批量执行。

## 模块一：图像分割

电商场景的分割需求是"前景商品 + 背景分离"。选择 RMBG-1.4：专门为商品和人像优化，速度快且边缘质量好。

```python
import torch
import numpy as np
from PIL import Image, ImageFilter

class ProductSegmentation:
    """商品图像分割节点"""

    def __init__(self, model_type="rmbg"):
        self.model = self._load_model(model_type)

    def _load_model(self, model_type):
        if model_type == "rmbg":
            from transformers import AutoModelForImageSegmentation
            return AutoModelForImageSegmentation.from_pretrained(
                "briaai/RMBG-1.4", trust_remote_code=True
            )
        raise ValueError(f"Unknown model: {model_type}")

    def segment(self, image: Image.Image, edge_blur=2):
        """分割图像，返回 (前景, 蒙版)"""
        mask = self.model.predict(image)

        if edge_blur > 0:
            mask = mask.filter(ImageFilter.GaussianBlur(radius=edge_blur))

        mask_np = np.array(mask).astype(np.float32) / 255.0
        img_np = np.array(image).astype(np.float32) / 255.0

        foreground = img_np * mask_np[..., None]
        return Image.fromarray((foreground * 255).astype(np.uint8)), mask_np


# 使用示例
seg = ProductSegmentation("rmbg")
img = Image.open("product_photo.jpg").convert("RGB")
foreground, mask = seg.segment(img, edge_blur=2)
```

## 模块二：背景替换

三类背景的生成策略不同：

```python
class BackgroundGenerator:
    """背景生成器"""

    @staticmethod
    def white_background(foreground: Image.Image, mask: np.ndarray,
                         color="#FFFFFF", shadow=True, shadow_opacity=0.15):
        """白底图：直接替换背景色"""
        bg_color = tuple(int(color.lstrip('#')[i:i+2], 16) for i in (0, 2, 4))
        w, h = foreground.size
        background = Image.new("RGB", (w, h), bg_color)

        if shadow:
            shadow_mask = np.zeros_like(mask)
            offset = 5
            shadow_mask[offset:, :] = mask[:-offset, :]
            shadow_arr = (shadow_mask * shadow_opacity * 255).astype(np.uint8)
            shadow_img = Image.fromarray(shadow_arr, mode='L')
            background.paste(Image.new("RGB", (w, h), (200, 200, 200)),
                           mask=shadow_img)

        mask_img = Image.fromarray((mask * 255).astype(np.uint8), mode='L')
        background.paste(foreground, mask=mask_img)
        return background

    @staticmethod
    def scene_background_comfyui(image_path: str, mask_path: str,
                                  prompt: str, output_path: str):
        """场景图：通过 ComfyUI API 调用 Inpainting"""
        import requests

        workflow = {
            "1": {"class_type": "LoadImage", "inputs": {"image": image_path}},
            "2": {"class_type": "LoadImage", "inputs": {"image": mask_path}},
            "3": {"class_type": "CheckpointLoaderSimple",
                  "inputs": {"ckpt_name": "sd_xl_base_1.0.safetensors"}},
            "4": {"class_type": "CLIPTextEncode",
                  "inputs": {"text": prompt, "clip": ["3", 1]}},
            "5": {"class_type": "CLIPTextEncode",
                  "inputs": {"text": "blurry, low quality", "clip": ["3", 1]}},
            "6": {"class_type": "SetLatentNoiseMask",
                  "inputs": {"samples": ["8", 0], "mask": ["2", 1]}},
            "7": {"class_type": "KSampler",
                  "inputs": {"model": ["3", 0], "positive": ["4", 0],
                             "negative": ["5", 0], "latent_image": ["6", 0],
                             "seed": 42, "steps": 25, "cfg": 7.0,
                             "sampler_name": "dpmpp_2m", "denoise": 0.75}},
            "8": {"class_type": "VAEEncode",
                  "inputs": {"pixels": ["1", 0], "vae": ["3", 2]}},
            "9": {"class_type": "VAEDecode",
                  "inputs": {"samples": ["7", 0], "vae": ["3", 2]}},
            "10": {"class_type": "SaveImage",
                   "inputs": {"images": ["9", 0], "filename_prefix": "scene"}},
        }

        resp = requests.post("http://127.0.0.1:8188/prompt",
                           json={"prompt": workflow})
        return resp.json()["prompt_id"]
```

## 模块三：批量处理引擎

```python
import os
import json
import time
import logging
from pathlib import Path
from dataclasses import dataclass, field
from enum import Enum

logging.basicConfig(level=logging.INFO,
                    format='%(asctime)s %(levelname)s %(message)s')
logger = logging.getLogger(__name__)


class TaskStatus(Enum):
    PENDING = "pending"
    PROCESSING = "processing"
    SUCCESS = "success"
    FAILED = "failed"
    REVIEW = "manual_review"


class OutputType(Enum):
    WHITE_BG = "white_bg"
    SCENE = "scene"
    MODEL = "model"


@dataclass
class ProcessingTask:
    task_id: str
    input_path: str
    output_type: OutputType
    status: TaskStatus = TaskStatus.PENDING
    output_path: str = ""
    error: str = ""
    processing_time: float = 0.0
    quality_score: float = 0.0


@dataclass
class BatchConfig:
    input_dir: str
    output_dir: str
    output_type: OutputType
    quality_threshold: float = 0.7
    scene_prompt: str = "modern living room, natural lighting"
    background_color: str = "#FFFFFF"


class BatchProcessor:
    def __init__(self, config: BatchConfig):
        self.config = config
        self.tasks: list[ProcessingTask] = []
        self.stats = {"total": 0, "success": 0, "failed": 0, "review": 0}
        os.makedirs(config.output_dir, exist_ok=True)
        os.makedirs(os.path.join(config.output_dir, "review"), exist_ok=True)

    def scan_input(self):
        exts = {'.jpg', '.jpeg', '.png', '.webp'}
        for idx, f in enumerate(sorted(Path(self.config.input_dir).iterdir())):
            if f.suffix.lower() in exts:
                self.tasks.append(ProcessingTask(
                    task_id=f"task_{idx:04d}",
                    input_path=str(f),
                    output_type=self.config.output_type,
                ))
        self.stats["total"] = len(self.tasks)
        logger.info(f"扫描完成: {len(self.tasks)} 个文件")

    def process_all(self):
        start = time.time()
        for task in self.tasks:
            self._process(task)
        elapsed = time.time() - start
        logger.info(f"完成: 成功={self.stats['success']}, "
                    f"失败={self.stats['failed']}, "
                    f"审核={self.stats['review']}, "
                    f"耗时={elapsed:.1f}s")
        self._save_report()

    def _process(self, task: ProcessingTask):
        task.status = TaskStatus.PROCESSING
        start = time.time()

        try:
            img = Image.open(task.input_path).convert("RGB")

            # 质量预检
            ok, reason = self._quality_check(img)
            if not ok:
                task.status = TaskStatus.REVIEW
                task.error = reason
                self.stats["review"] += 1
                return

            # 分割
            foreground, mask = self._segment(img)

            # 背景替换
            if task.output_type == OutputType.WHITE_BG:
                result = BackgroundGenerator.white_background(
                    foreground, mask, self.config.background_color)
            elif task.output_type == OutputType.SCENE:
                result = self._scene_generate(img, mask)
            else:
                result = foreground  # 模特图需要额外处理

            # 超分
            result = self._upscale(result, task.output_type)

            # 质量门控
            passed, gate_msg = self._quality_gate(result)
            if not passed:
                task.status = TaskStatus.REVIEW
                task.error = gate_msg
                self.stats["review"] += 1
                return

            # 保存
            out_name = f"{task.task_id}_{task.output_type.value}.png"
            task.output_path = os.path.join(self.config.output_dir, out_name)
            result.save(task.output_path, quality=95)
            task.status = TaskStatus.SUCCESS
            self.stats["success"] += 1

        except Exception as e:
            task.status = TaskStatus.FAILED
            task.error = str(e)
            self.stats["failed"] += 1
            logger.error(f"[{task.task_id}] 失败: {e}")

        finally:
            task.processing_time = time.time() - start

    def _quality_check(self, img):
        w, h = img.size
        if min(w, h) < 256:
            return False, f"分辨率过低: {w}×{h}"

        gray = np.array(img.convert('L'), dtype=np.float32)
        brightness = gray.mean() / 255
        if brightness < 0.05:
            return False, "图像过暗"
        if brightness > 0.95:
            return False, "图像过亮"

        return True, "通过"

    def _segment(self, img):
        seg = ProductSegmentation("rmbg")
        return seg.segment(img)

    def _scene_generate(self, img, mask):
        # 简化：实际应调用 ComfyUI API
        return img

    def _upscale(self, img, output_type):
        sizes = {OutputType.WHITE_BG: 800, OutputType.SCENE: 1200, OutputType.MODEL: 1024}
        target = sizes.get(output_type, 800)
        w, h = img.size
        scale = target / max(w, h)
        return img.resize((int(w * scale), int(h * scale)), Image.LANCZOS)

    def _quality_gate(self, img):
        w, h = img.size
        if min(w, h) < 800:
            return False, f"输出分辨率不达标: {w}×{h}"
        return True, "通过"

    def _save_report(self):
        report = {
            "stats": self.stats,
            "tasks": [{"id": t.task_id, "status": t.status.value,
                       "time": round(t.processing_time, 2),
                       "error": t.error} for t in self.tasks],
        }
        path = os.path.join(self.config.output_dir, "report.json")
        with open(path, "w", encoding="utf-8") as f:
            json.dump(report, f, ensure_ascii=False, indent=2)
        logger.info(f"报告已保存: {path}")
```

## 参考答案

### 练习一

节点连接：
1. LoadImage → 加载输入
2. RMBGSegmentation (rmbg, edge_blur=2) → 分割
3. ImageBlurMask → 边缘优化
4. WhiteBackgroundGenerator (color=#FAFAFA, shadow=true, opacity=0.12)
5. RealESRGAN x4plus → 放大到 800×800
6. ImageSharpen (amount=0.2) → 轻微锐化
7. SaveImage

注意：白底用 `#FAFAFA` 而非纯白 `#FFF`，因为电商平台压缩后纯白可能偏灰。浅色商品需要降低分割阈值（0.6），避免误删商品边缘。

```python
def load_from_csv(self, csv_path):
    import csv
    with open(csv_path, 'r', encoding='utf-8') as f:
        for idx, row in enumerate(csv.DictReader(f)):
            self.tasks.append(ProcessingTask(
                task_id=f"task_{idx:04d}",
                input_path=row['input_path'],
                output_type=OutputType(row['output_type']),
            ))

def resume_from_checkpoint(self):
    cp_path = os.path.join(self.config.output_dir, "checkpoint.json")
    if os.path.exists(cp_path):
        with open(cp_path) as f:
            done = set(json.load(f).get("completed", []))
        self.tasks = [t for t in self.tasks if t.task_id not in done]
        logger.info(f"断点续传: 跳过 {len(done)} 个已完成任务")

def save_checkpoint(self, task_id):
    cp_path = os.path.join(self.config.output_dir, "checkpoint.json")
    done = []
    if os.path.exists(cp_path):
        with open(cp_path) as f:
            done = json.load(f).get("completed", [])
    done.append(task_id)
    with open(cp_path, "w") as f:
        json.dump({"completed": done}, f)
```
