你刚加入一家跨境电商公司，运营总监给你发了一份需求：公司有 1000 个 SKU，每个 SKU 需要白底图、场景图、模特图各 5 张，总共 15000 张图像。传统拍摄流程是找摄影团队、租棚、搭建场景、拍摄、修图、交付。每个 SKU 的拍摄成本在 200-500 元，1000 个 SKU 就是 20-50 万，周期至少 2 个月。

## 一、产品图类型与技术要求

### 1.1 三种产品图类型

电商产品图不是一种图，而是三种完全不同的图，每种的技术路线和质量标准都不一样：

```
┌─────────────────────────────────────────────────────────────────┐
│                     电商产品图类型对比                            │
├──────────────┬─────────────────┬─────────────────┬──────────────┤
│              │ 白底图           │ 场景图           │ 模特图        │
├──────────────┼─────────────────┼─────────────────┼──────────────┤
│ 定义         │ 纯白背景         │ 产品置于真实     │ 产品由真人    │
│              │ 产品居中         │ 或虚拟环境中     │ 模特穿戴/展示 │
├──────────────┼─────────────────┼─────────────────┼──────────────┤
│ 用途         │ 主图、搜索结果   │ 详情页、广告     │ 服装、配饰    │
│              │ SKU 标准化       │ 场景化营销       │ 穿搭展示      │
├──────────────┼─────────────────┼─────────────────┼──────────────┤
│ 技术难点     │ 背景去除干净     │ 产品形状不变     │ 产品外观不变  │
│              │ 阴影自然         │ 光照方向一致     │ 人体比例合理  │
├──────────────┼─────────────────┼─────────────────┼──────────────┤
│ 平台规范     │ 背景纯白         │ 无硬性背景要求   │ 人脸清晰      │
│              │ 产品居中         │ 但需高分辨率     │ 无畸变        │
├──────────────┼─────────────────┼─────────────────┼──────────────┤
│ AI 技术路线  │ 分割模型 + 合成  │ Inpainting +     │ IP-Adapter +  │
│              │                  │ ControlNet       │ 虚拟试穿      │
└──────────────┴─────────────────┴─────────────────┴──────────────┘
```

## 二、产品分割与白底图自动生成

### 2.1 整体架构

白底图生成是所有产品图自动化的第一步。核心流程是"分割 → 去背景 → 合成白底 → 加阴影"：

```
┌──────────────┐     ┌──────────────┐     ┌──────────────┐     ┌──────────────┐
│  原始产品图   │────▶│  语义分割     │────▶│  背景去除     │────▶│  白底合成     │
│  (任意背景)   │     │  SAM/RMBG    │     │  Alpha Matte │     │  + 投影阴影   │
└──────────────┘     └──────────────┘     └──────────────┘     └──────────────┘
                                                                      │
                                                                      ▼
                                                              ┌──────────────┐
                                                              │  规范校验     │
                                                              │  尺寸/纯白/居中│
                                                              └──────────────┘
```

### 2.2 使用 RMBG-2.0 进行产品分割

RMBG-2.0 是 BRIA AI 开源的背景去除模型，专门针对产品图优化，比通用分割模型在边缘细节上表现更好：

```python
import torch
import numpy as np
from PIL import Image, ImageFilter
from transformers import AutoModelForImageSegmentation, AutoConfig
from torchvision import transforms


class ProductBackgroundRemover:
    """产品图背景去除器，基于 RMBG-2.0 模型"""

    def __init__(self, model_path: str = "briaai/RMBG-2.0"):
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        config = AutoConfig.from_pretrained(model_path, trust_remote_code=True)
        self.model = AutoModelForImageSegmentation.from_pretrained(
            model_path, config=config, trust_remote_code=True
        ).to(self.device).eval()

        self.transform = transforms.Compose([
            transforms.Resize((1024, 1024)),
            transforms.ToTensor(),
            transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225]),
        ])

    def remove_background(self, image: Image.Image) -> Image.Image:
        """去除背景，返回带 alpha 通道的 RGBA 图像"""
        original_size = image.size
        input_tensor = self.transform(image).unsqueeze(0).to(self.device)

        with torch.no_grad():
            mask_logits = self.model(input_tensor)[-1].sigmoid()

        mask_np = mask_logits.squeeze().cpu().numpy()
        mask_pil = Image.fromarray((mask_np * 255).astype(np.uint8), mode="L")
        mask_pil = mask_pil.resize(original_size, Image.LANCZOS)
        mask_pil = mask_pil.filter(ImageFilter.GaussianBlur(radius=1))

        result = image.convert("RGBA")
        result.putalpha(mask_pil)
        return result

    def refine_edges(self, rgba_image: Image.Image) -> Image.Image:
        """优化边缘：去除半透明像素的色溢（color bleeding）"""
        np_image = np.array(rgba_image)
        alpha = np_image[:, :, 3]
        rgb = np_image[:, :, :3].astype(float)

        white_bg = np.array([255, 255, 255], dtype=float)
        alpha_float = alpha[:, :, np.newaxis] / 255.0
        blended = rgb * alpha_float + white_bg * (1 - alpha_float)

        result = np_image.copy()
        result[:, :, :3] = blended.astype(np.uint8)
        result[:, :, 3] = alpha
        return Image.fromarray(result)


if __name__ == "__main__":
    remover = ProductBackgroundRemover()

    product_photo = Image.open("product_raw.jpg").convert("RGB")
    cutout = remover.remove_background(product_photo)
    refined = remover.refine_edges(cutout)
    refined.save("product_cutout.png")
    print(f"分割完成，输出尺寸: {refined.size}")
```

### 2.3 白底图合成与阴影生成

分割只是第一步，真正的白底图还需要自然的投影阴影，否则产品看起来像"贴"在白纸上：

```python
import numpy as np
from PIL import Image, ImageFilter


class WhiteBackgroundComposer:
    """将抠出的产品图合成到纯白背景上，添加自然投影"""

    ECOMMERCE_SPECS = {
        "taobao": {"size": (800, 800), "product_ratio": 0.75},
        "jd": {"size": (800, 800), "product_ratio": 0.75},
        "amazon": {"size": (1000, 1000), "product_ratio": 0.80},
        "shopify": {"size": (2048, 2048), "product_ratio": 0.75},
    }

    def compose_white_background(
        self,
        cutout: Image.Image,
        platform: str = "taobao",
        shadow_intensity: int = 40,
        shadow_blur_radius: int = 15,
    ) -> Image.Image:
        """将产品抠图放到纯白背景上"""
        spec = self.ECOMMERCE_SPECS.get(platform, self.ECOMMERCE_SPECS["taobao"])
        canvas_size = spec["size"]
        product_ratio = spec["product_ratio"]

        product_fitted = self._fit_product(cutout, canvas_size, product_ratio)
        canvas = Image.new("RGB", canvas_size, (255, 255, 255))

        shadow = self._generate_shadow(product_fitted, shadow_intensity, shadow_blur_radius)
        shadow_offset = (shadow_blur_radius // 2, shadow_blur_radius // 2)
        canvas.paste(shadow, shadow_offset, shadow.split()[3])

        product_position = self._center_position(product_fitted.size, canvas_size)
        canvas.paste(product_fitted, product_position, product_fitted.split()[3])

        return canvas

    def _fit_product(self, cutout, canvas_size, ratio):
        max_w = int(canvas_size[0] * ratio)
        max_h = int(canvas_size[1] * ratio)
        orig_w, orig_h = cutout.size
        scale = min(max_w / orig_w, max_h / orig_h)
        new_size = (int(orig_w * scale), int(orig_h * scale))
        return cutout.resize(new_size, Image.LANCZOS)

    def _center_position(self, product_size, canvas_size):
        x = (canvas_size[0] - product_size[0]) // 2
        y = (canvas_size[1] - product_size[1]) // 2
        return (x, y)

    def _generate_shadow(self, cutout, intensity, blur_radius):
        shadow = Image.new("RGBA", cutout.size, (0, 0, 0, 0))
        alpha = cutout.split()[3]
        shadow_alpha = alpha.point(lambda x: min(int(x * intensity / 255), intensity))
        shadow.putalpha(shadow_alpha)
        return shadow.filter(ImageFilter.GaussianBlur(radius=blur_radius))


if __name__ == "__main__":
    composer = WhiteBackgroundComposer()
    cutout = Image.open("product_cutout.png").convert("RGBA")

    for platform in ["taobao", "amazon", "shopify"]:
        result = composer.compose_white_background(cutout, platform=platform)
        result.save(f"product_white_{platform}.jpg", quality=95)
        print(f"{platform} 白底图: {result.size}")
```

## 三、场景图自动生成

### 3.1 技术路线：Inpainting + ControlNet

场景图的核心挑战是：把产品"放"到一个场景里，但产品本身不能有任何变化。这需要两个技术配合：

```
┌────────────────────────────────────────────────────────────────┐
│                    场景图生成 Pipeline                           │
│                                                                │
│  ┌──────────┐    ┌──────────┐    ┌──────────┐    ┌──────────┐ │
│  │ 产品抠图  │───▶│ 场景模板  │───▶│ Inpaint  │───▶│ 后处理   │ │
│  │ (Section2)│    │ + Mask   │    │ + Control │    │ 颜色校正 │ │
│  └──────────┘    └──────────┘    │   Net     │    │ 光照统一 │ │
│                                   └──────────┘    └──────────┘ │
│                                                                │
│  关键：产品区域用 ControlNet Canny/Depth 锁定形状               │
│       背景区域用 Inpainting 自由生成                            │
└────────────────────────────────────────────────────────────────┘
```

### 3.2 场景模板管理与 Inpainting 生成

```python
import os
import json
import torch
import numpy as np
from PIL import Image, ImageDraw
from diffusers import StableDiffusionXLInpaintPipeline, ControlNetModel
from diffusers.utils import load_image


SCENE_TEMPLATES = {
    "kitchen_counter": {
        "prompt": "modern kitchen marble counter, natural daylight, clean minimal style, product photography",
        "negative": "cluttered, dirty, dark, low quality",
        "placement": {"x_ratio": 0.5, "y_ratio": 0.65, "scale_ratio": 0.35},
    },
    "living_room_table": {
        "prompt": "elegant living room wooden table, soft warm lighting, cozy interior, product shot",
        "negative": "messy, dark, blurry, distorted",
        "placement": {"x_ratio": 0.5, "y_ratio": 0.6, "scale_ratio": 0.4},
    },
    "outdoor_garden": {
        "prompt": "beautiful garden table with flowers, natural sunlight, green plants background, product photography",
        "negative": "dark, indoor, low quality, artifacts",
        "placement": {"x_ratio": 0.5, "y_ratio": 0.55, "scale_ratio": 0.45},
    },
    "studio_gradient": {
        "prompt": "professional studio background, soft gradient, clean product photography, high end",
        "negative": "cluttered, low quality, artifacts",
        "placement": {"x_ratio": 0.5, "y_ratio": 0.55, "scale_ratio": 0.5},
    },
}


class SceneImageGenerator:
    """基于 Inpainting 的产品场景图生成器"""

    def __init__(self, model_path: str = "stabilityai/stable-diffusion-xl-base-1.0"):
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.pipe = StableDiffusionXLInpaintPipeline.from_pretrained(
            model_path,
            torch_dtype=torch.float16 if self.device == "cuda" else torch.float32,
        ).to(self.device)

    def generate_scene(
        self,
        product_cutout: Image.Image,
        scene_name: str,
        output_size: tuple = (1024, 1024),
        num_variants: int = 3,
        seed_base: int = 42,
    ) -> list[Image.Image]:
        """为一个产品生成多个场景图变体"""
        template = SCENE_TEMPLATES[scene_name]
        placement = template["placement"]

        bg_canvas = Image.new("RGB", output_size, (240, 240, 240))
        product_resized = self._resize_product(
            product_cutout, output_size, placement["scale_ratio"]
        )
        paste_x = int(output_size[0] * placement["x_ratio"]) - product_resized.width // 2
        paste_y = int(output_size[1] * placement["y_ratio"]) - product_resized.height // 2

        product_position = (paste_x, paste_y)
        bg_canvas.paste(product_resized, product_position, product_resized.split()[3])

        inpaint_mask = self._create_inpaint_mask(output_size, product_resized.size, product_position)

        results = []
        for i in range(num_variants):
            generator = torch.Generator(self.device).manual_seed(seed_base + i)
            scene_image = self.pipe(
                prompt=template["prompt"],
                negative_prompt=template["negative"],
                image=bg_canvas,
                mask_image=inpaint_mask,
                num_inference_steps=30,
                generator=generator,
                strength=0.75,
                guidance_scale=7.5,
            ).images[0]

            scene_image.paste(product_resized, product_position, product_resized.split()[3])
            results.append(scene_image)

        return results

    def _resize_product(self, cutout, canvas_size, scale_ratio):
        max_dim = int(min(canvas_size) * scale_ratio)
        orig_w, orig_h = cutout.size
        scale = max_dim / max(orig_w, orig_h)
        new_size = (int(orig_w * scale), int(orig_h * scale))
        return cutout.resize(new_size, Image.LANCZOS)

    def _create_inpaint_mask(self, canvas_size, product_size, position):
        mask = Image.new("L", canvas_size, 255)
        draw = ImageDraw.Draw(mask)
        margin = 20
        x0 = position[0] - margin
        y0 = position[1] - margin
        x1 = position[0] + product_size[0] + margin
        y1 = position[1] + product_size[1] + margin
        draw.rectangle([x0, y0, x1, y1], fill=0)
        return mask


if __name__ == "__main__":
    generator = SceneImageGenerator()
    cutout = Image.open("product_cutout.png").convert("RGBA")

    for scene in ["kitchen_counter", "studio_gradient"]:
        variants = generator.generate_scene(cutout, scene_name=scene, num_variants=2)
        for idx, img in enumerate(variants):
            img.save(f"scene_{scene}_v{idx:02d}.jpg", quality=95)
            print(f"已生成: scene_{scene}_v{idx:02d}.jpg")
```

## 四、模特图自动生成

### 4.1 IP-Adapter 实现产品一致性控制

模特图的核心难点是"产品一致性"——生成的模特可以变姿势、变场景，但穿戴的产品外观必须和实物完全一致。IP-Adapter 通过图像特征注入实现这一点：

```
┌────────────────────────────────────────────────────────────────┐
│                   模特图生成 Pipeline                           │
│                                                                │
│  ┌──────────┐    ┌──────────┐    ┌──────────┐    ┌──────────┐ │
│  │ 产品照片  │───▶│IP-Adapter│───▶│ SDXL     │───▶│ 人脸修复 │ │
│  │ (特征源)  │    │特征提取  │    │ + Pose   │    │ GFPGAN   │ │
│  └──────────┘    └──────────┘    │ ControlNet│    └──────────┘ │
│                                   └──────────┘                 │
│  ┌──────────┐                                                  │
│  │ 姿势模板  │──── OpenPose 骨骼图控制模特姿势                   │
│  └──────────┘                                                  │
└────────────────────────────────────────────────────────────────┘
```

### 4.2 模特图生成器实现

```python
import torch
import numpy as np
from PIL import Image
from diffusers import StableDiffusionXLPipeline, StableDiffusionXLControlNetPipeline
from diffusers import ControlNetModel
from ip_adapter import IPAdapterXL


POSE_TEMPLATES = {
    "standing_front": "a fashion model standing upright, front view, professional studio lighting",
    "standing_side": "a fashion model standing, side angle, elegant pose, studio lighting",
    "walking": "a fashion model walking, dynamic pose, street style photography",
    "seated": "a fashion model seated on a chair, relaxed pose, lifestyle photography",
}


class ModelImageGenerator:
    """基于 IP-Adapter 的模特图生成器，保持产品外观一致性"""

    def __init__(
        self,
        base_model: str = "stabilityai/stable-diffusion-xl-base-1.0",
        ip_adapter_path: str = "h94/IP-Adapter",
        image_encoder_path: str = "h94/IP-Adapter/models/image_encoder",
    ):
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        controlnet = ControlNetModel.from_pretrained(
            "thibaudi/controlnet-openpose-sdxl-1.0",
            torch_dtype=torch.float16,
        )
        self.pipe = StableDiffusionXLControlNetPipeline.from_pretrained(
            base_model,
            controlnet=controlnet,
            torch_dtype=torch.float16,
        ).to(self.device)

        self.ip_adapter = IPAdapterXL(
            self.pipe,
            image_encoder_path,
            ip_adapter_path,
            device=self.device,
        )

    def generate_model_image(
        self,
        product_image: Image.Image,
        pose_name: str = "standing_front",
        gender: str = "female",
        style_prompt: str = "",
        num_variants: int = 3,
        seed_base: int = 100,
    ) -> list[Image.Image]:
        """生成模特穿戴/展示产品的图像"""
        pose_desc = POSE_TEMPLATES.get(pose_name, POSE_TEMPLATES["standing_front"])

        base_prompt = (
            f"professional e-commerce product photography, "
            f"a {gender} model wearing/using the product, "
            f"{pose_desc}, high quality, detailed, 8k"
        )
        if style_prompt:
            base_prompt += f", {style_prompt}"

        negative_prompt = (
            "deformed, distorted, ugly, blurry, low quality, "
            "extra limbs, extra fingers, watermark, text"
        )

        results = []
        for i in range(num_variants):
            generator = torch.Generator(self.device).manual_seed(seed_base + i)
            result = self.ip_adapter.generate(
                prompt=base_prompt,
                negative_prompt=negative_prompt,
                pil_image=product_image,
                scale=0.8,
                num_samples=1,
                num_inference_steps=30,
                generator=generator,
                guidance_scale=7.0,
            )
            results.extend(result)

        return results


class BatchModelGenerator:
    """批量模特图生成，支持多产品 × 多姿势组合"""

    def __init__(self, generator: ModelImageGenerator):
        self.generator = generator

    def generate_product_catalog(
        self,
        product_images: list[dict],
        poses: list[str] = None,
        gender: str = "female",
    ) -> dict:
        """为一批产品生成完整的模特图目录"""
        if poses is None:
            poses = ["standing_front", "standing_side"]

        catalog = {}
        for product in product_images:
            sku_id = product["sku_id"]
            image = Image.open(product["image_path"]).convert("RGB")
            catalog[sku_id] = {}

            for pose in poses:
                variants = self.generator.generate_model_image(
                    product_image=image,
                    pose_name=pose,
                    gender=gender,
                    num_variants=2,
                )
                catalog[sku_id][pose] = variants

        return catalog


if __name__ == "__main__":
    generator = ModelImageGenerator()
    product = Image.open("product_cutout.png").convert("RGB")

    model_images = generator.generate_model_image(
        product_image=product,
        pose_name="standing_front",
        gender="female",
        num_variants=2,
    )
    for idx, img in enumerate(model_images):
        img.save(f"model_standing_v{idx:02d}.jpg", quality=95)
        print(f"已生成模特图: model_standing_v{idx:02d}.jpg")
```

## 五、多角度产品图生成

### 5.1 单张图到多角度的挑战

很多 SKU 只有一张正面照片，但电商详情页需要正面、侧面、背面、俯视等多个角度。直接用 3D 重建成本太高，更实用的方案是用 Inpainting 在保持产品一致性的前提下"想象"其他角度：

```python
import torch
from PIL import Image
from diffusers import StableDiffusionXLInpaintPipeline


ANGLE_PROMPTS = {
    "front": "product front view, centered, professional product photography",
    "side_left": "product left side view, 45 degree angle, professional product photography",
    "side_right": "product right side view, 45 degree angle, professional product photography",
    "back": "product back view, centered, professional product photography",
    "top_down": "product top-down view, bird's eye angle, flat lay product photography",
    "close_up": "product close-up detail shot, macro photography, showing texture and details",
}


class MultiAngleGenerator:
    """多角度产品图生成器"""

    def __init__(self, model_path: str = "stabilityai/stable-diffusion-xl-base-1.0"):
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.pipe = StableDiffusionXLInpaintPipeline.from_pretrained(
            model_path,
            torch_dtype=torch.float16 if self.device == "cuda" else torch.float32,
        ).to(self.device)

    def generate_angles(
        self,
        product_image: Image.Image,
        angles: list[str] = None,
        consistency_seed: int = 42,
    ) -> dict[str, Image.Image]:
        """从一张产品图生成多个角度的变体"""
        if angles is None:
            angles = ["front", "side_left", "side_right", "top_down"]

        results = {}
        for angle in angles:
            prompt = ANGLE_PROMPTS.get(angle, ANGLE_PROMPTS["front"])
            full_prompt = (
                f"{prompt}, white background, studio lighting, "
                f"clean minimal style, high quality product photo"
            )

            generator = torch.Generator(self.device).manual_seed(consistency_seed)
            result = self.pipe(
                prompt=full_prompt,
                image=product_image,
                mask_image=self._create_center_mask(product_image.size, ratio=0.7),
                num_inference_steps=30,
                generator=generator,
                strength=0.6,
                guidance_scale=7.0,
            ).images[0]

            results[angle] = result

        return results

    def _create_center_mask(self, size, ratio=0.7):
        from PIL import ImageDraw
        mask = Image.new("L", size, 255)
        draw = ImageDraw.Draw(mask)
        w, h = size
        margin_x = int(w * (1 - ratio) / 2)
        margin_y = int(h * (1 - ratio) / 2)
        draw.rectangle([margin_x, margin_y, w - margin_x, h - margin_y], fill=0)
        return mask


if __name__ == "__main__":
    generator = MultiAngleGenerator()
    product = Image.open("product_white_taobao.jpg").convert("RGB")

    angles = generator.generate_angles(product, angles=["front", "side_left", "top_down"])
    for angle_name, img in angles.items():
        img.save(f"product_angle_{angle_name}.jpg", quality=95)
        print(f"已生成: product_angle_{angle_name}.jpg")
```

### 6.2 批量 Pipeline 完整实现

```python
import os
import time
import json
import logging
from dataclasses import dataclass, field
from enum import Enum
from concurrent.futures import ThreadPoolExecutor, as_completed
from PIL import Image

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")
logger = logging.getLogger(__name__)


class ImageType(Enum):
    WHITE_BACKGROUND = "white_bg"
    SCENE = "scene"
    MODEL = "model"


class PipelineStatus(Enum):
    PENDING = "pending"
    PROCESSING = "processing"
    COMPLETED = "completed"
    FAILED = "failed"


@dataclass
class ProductTask:
    sku_id: str
    source_path: str
    image_types: list[ImageType] = field(default_factory=lambda: [ImageType.WHITE_BACKGROUND])
    scene_names: list[str] = field(default_factory=lambda: ["studio_gradient"])
    status: PipelineStatus = PipelineStatus.PENDING
    output_paths: dict = field(default_factory=dict)
    error_message: str = ""
    processing_time: float = 0.0


class ProductImagePipeline:
    """产品图自动化批量处理 Pipeline"""

    def __init__(
        self,
        output_dir: str = "./output",
        max_workers: int = 2,
        platform: str = "taobao",
    ):
        self.output_dir = output_dir
        self.max_workers = max_workers
        self.platform = platform
        os.makedirs(output_dir, exist_ok=True)

        self.remover = ProductBackgroundRemover()
        self.white_composer = WhiteBackgroundComposer()
        self.scene_generator = SceneImageGenerator()

    def process_batch(self, tasks: list[ProductTask]) -> list[ProductTask]:
        """批量处理多个 SKU 的产品图生成任务"""
        logger.info(f"开始批量处理 {len(tasks)} 个 SKU，并发数: {self.max_workers}")
        start_time = time.time()

        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            future_to_task = {
                executor.submit(self._process_single_task, task): task
                for task in tasks
            }

            for future in as_completed(future_to_task):
                task = future_to_task[future]
                try:
                    completed_task = future.result()
                    logger.info(
                        f"SKU {completed_task.sku_id} 处理完成，"
                        f"耗时 {completed_task.processing_time:.1f}s，"
                        f"输出 {len(completed_task.output_paths)} 张图"
                    )
                except Exception as exc:
                    task.status = PipelineStatus.FAILED
                    task.error_message = str(exc)
                    logger.error(f"SKU {task.sku_id} 处理失败: {exc}")

        elapsed = time.time() - start_time
        completed = sum(1 for t in tasks if t.status == PipelineStatus.COMPLETED)
        failed = sum(1 for t in tasks if t.status == PipelineStatus.FAILED)
        logger.info(f"批量处理完成: {completed} 成功, {failed} 失败, 总耗时 {elapsed:.1f}s")

        return tasks

    def _process_single_task(self, task: ProductTask) -> ProductTask:
        """处理单个 SKU 的全部图片生成任务"""
        task.status = PipelineStatus.PROCESSING
        task_start = time.time()

        try:
            source_image = Image.open(task.source_path).convert("RGB")
            cutout = self.remover.remove_background(source_image)
            cutout = self.remover.refine_edges(cutout)

            sku_output_dir = os.path.join(self.output_dir, task.sku_id)
            os.makedirs(sku_output_dir, exist_ok=True)

            if ImageType.WHITE_BACKGROUND in task.image_types:
                white_img = self.white_composer.compose_white_background(
                    cutout, platform=self.platform
                )
                output_path = os.path.join(sku_output_dir, f"{task.sku_id}_white_bg.jpg")
                white_img.save(output_path, quality=95)
                task.output_paths["white_bg"] = output_path

            if ImageType.SCENE in task.image_types:
                for scene_name in task.scene_names:
                    scene_variants = self.scene_generator.generate_scene(
                        cutout, scene_name=scene_name, num_variants=2
                    )
                    for idx, scene_img in enumerate(scene_variants):
                        output_path = os.path.join(
                            sku_output_dir, f"{task.sku_id}_scene_{scene_name}_v{idx:02d}.jpg"
                        )
                        scene_img.save(output_path, quality=95)
                        task.output_paths[f"scene_{scene_name}_v{idx:02d}"] = output_path

            task.status = PipelineStatus.COMPLETED
            task.processing_time = time.time() - task_start

        except Exception as e:
            task.status = PipelineStatus.FAILED
            task.error_message = str(e)
            task.processing_time = time.time() - task_start
            raise

        return task

    def generate_report(self, tasks: list[ProductTask]) -> dict:
        """生成处理报告"""
        total = len(tasks)
        completed = sum(1 for t in tasks if t.status == PipelineStatus.COMPLETED)
        total_images = sum(len(t.output_paths) for t in tasks)
        total_time = sum(t.processing_time for t in tasks)

        report = {
            "total_skus": total,
            "completed": completed,
            "failed": total - completed,
            "total_images_generated": total_images,
            "total_processing_time_seconds": round(total_time, 1),
            "avg_time_per_sku": round(total_time / max(total, 1), 1),
            "estimated_cost_per_image": 0.002,
            "total_estimated_cost": round(total_images * 0.002, 2),
        }

        report_path = os.path.join(self.output_dir, "pipeline_report.json")
        with open(report_path, "w", encoding="utf-8") as f:
            json.dump(report, f, ensure_ascii=False, indent=2)

        return report


if __name__ == "__main__":
    pipeline = ProductImagePipeline(
        output_dir="./product_output",
        max_workers=2,
        platform="taobao",
    )

    tasks = [
        ProductTask(
            sku_id="SKU001",
            source_path="./photos/SKU001_raw.jpg",
            image_types=[ImageType.WHITE_BACKGROUND, ImageType.SCENE],
            scene_names=["studio_gradient", "kitchen_counter"],
        ),
        ProductTask(
            sku_id="SKU002",
            source_path="./photos/SKU002_raw.jpg",
            image_types=[ImageType.WHITE_BACKGROUND, ImageType.SCENE, ImageType.MODEL],
            scene_names=["living_room_table"],
        ),
    ]

    completed_tasks = pipeline.process_batch(tasks)
    report = pipeline.generate_report(completed_tasks)
    print(json.dumps(report, ensure_ascii=False, indent=2))
```

## 七、常见误区

**误区一：分割模型越贵越好。** 很多人直接用 SAM2 做产品分割，但通用分割模型在产品边缘（尤其是毛绒玩具、透明玻璃瓶）上并不比专门的 RMBG 模型好。选模型要看你的产品品类，而不是看参数量。

**误区二：Inpainting strength 越高生成越自由。** strength 设太高（>0.85）会导致场景和产品风格割裂，看起来像 PS 合成的。一般建议 0.6-0.75，让模型在保持整体一致性的同时生成背景。

**误区三：忽略颜色校正。** 不同光照条件下拍摄的产品图颜色偏差很大，直接喂给模型会导致输出颜色不一致。pipeline 前端应该加一步白平衡校正。

**误区四：不做质量校验直接上线。** AI 生成的图偶尔会出现畸变、多指、文字乱码等问题。批量 pipeline 必须加质量过滤环节（可以用 CLIP Score 或人工抽检）。

## 小结

本课从电商产品图的实际需求出发，构建了一套完整的自动化 pipeline：

1. **白底图生成**：RMBG-2.0 分割 + 边缘优化 + 白底合成 + 自然阴影，满足各电商平台规范
2. **场景图生成**：Inpainting + 场景模板管理，产品区域用 mask 保护不变形
3. **模特图生成**：IP-Adapter 注入产品特征，ControlNet 控制姿势，保持产品外观一致
4. **多角度生成**：从单张图生成多视角变体，扩展详情页素材
5. **批量 Pipeline**：并发处理 + 状态追踪 + 质量校验 + 成本核算

核心原则是"产品一致性优先"——宁可背景简单，也不能让产品外观走样。

### 练习一

**思路**：白底图校验的核心是检查四个维度——尺寸、背景白度、产品居中度、文件格式。背景白度可以通过统计边缘像素的 RGB 值来判断，居中度可以通过产品 bounding box 计算偏移量。

**答案**：

```python
from PIL import Image
import numpy as np


def validate_white_background(image_path: str) -> dict:
    """校验白底图是否符合电商平台规范"""
    result = {
        "path": image_path,
        "valid": True,
        "issues": [],
    }

    image = Image.open(image_path).convert("RGB")
    np_image = np.array(image)
    width, height = image.size

    if (width, height) != (800, 800):
        result["issues"].append(f"尺寸不符: {width}x{height}, 期望 800x800")
        result["valid"] = False

    border_pixels = np.concatenate([
        np_image[0, :, :].reshape(-1, 3),
        np_image[-1, :, :].reshape(-1, 3),
        np_image[:, 0, :].reshape(-1, 3),
        np_image[:, -1, :].reshape(-1, 3),
    ])
    avg_rgb = border_pixels.mean(axis=0)
    if any(avg_rgb[i] < 250 for i in range(3)):
        result["issues"].append(f"背景不够白: 平均 RGB={avg_rgb.astype(int).tolist()}")
        result["valid"] = False

    gray = np.mean(np_image, axis=2)
    product_mask = gray < 240
    if product_mask.sum() > 0:
        rows = np.where(product_mask.any(axis=1))[0]
        cols = np.where(product_mask.any(axis=0))[0]
        bbox_center_x = (cols[0] + cols[-1]) / 2
        bbox_center_y = (rows[0] + rows[-1]) / 2
        offset_x = abs(bbox_center_x - width / 2) / width
        offset_y = abs(bbox_center_y - height / 2) / height

        if offset_x > 0.1 or offset_y > 0.1:
            result["issues"].append(
                f"产品未居中: 偏移 x={offset_x:.1%}, y={offset_y:.1%}"
            )
            result["valid"] = False

    return result
```

**要点**：
- 边缘像素采样比全图遍历快得多
- 用灰度阈值 240 区分产品和白色背景
- 居中偏移容忍度设 10%，因为有些产品本身形状不规则

### 练习二

**思路**：场景模板需要包含生成参数和布局参数两部分。生成参数控制 AI 生成风格，布局参数控制产品在画布上的位置和大小。配置需要有合理的默认值和校验逻辑。

**答案**：

```python
import json
from dataclasses import dataclass, field
from pathlib import Path


@dataclass
class PlacementConfig:
    x_ratio: float = 0.5
    y_ratio: float = 0.6
    scale_ratio: float = 0.4

    def validate(self):
        for name, val in [("x_ratio", self.x_ratio), ("y_ratio", self.y_ratio)]:
            if not 0.0 <= val <= 1.0:
                raise ValueError(f"{name} 必须在 0-1 之间，当前值: {val}")
        if not 0.1 <= self.scale_ratio <= 0.9:
            raise ValueError(f"scale_ratio 必须在 0.1-0.9 之间，当前值: {self.scale_ratio}")


@dataclass
class SceneTemplate:
    name: str
    prompt: str
    negative: str = ""
    canvas_size: tuple = (1024, 1024)
    placement: PlacementConfig = field(default_factory=PlacementConfig)
    num_inference_steps: int = 30
    guidance_scale: float = 7.5

    def validate(self):
        if not self.prompt.strip():
            raise ValueError(f"场景 '{self.name}' 的 prompt 不能为空")
        if self.num_inference_steps < 10 or self.num_inference_steps > 100:
            raise ValueError(f"num_inference_steps 应在 10-100 之间")
        self.placement.validate()


def load_scene_templates(config_path: str) -> dict[str, SceneTemplate]:
    """从 JSON 文件加载场景模板配置"""
    with open(config_path, "r", encoding="utf-8") as f:
        raw = json.load(f)

    templates = {}
    for name, config in raw.items():
        placement_data = config.get("placement", {})
        placement = PlacementConfig(**placement_data)
        template = SceneTemplate(
            name=name,
            prompt=config["prompt"],
            negative=config.get("negative", ""),
            canvas_size=tuple(config.get("canvas_size", [1024, 1024])),
            placement=placement,
            num_inference_steps=config.get("num_inference_steps", 30),
            guidance_scale=config.get("guidance_scale", 7.5),
        )
        template.validate()
        templates[name] = template

    return templates
```

**要点**：
- dataclass 比 dict 更安全，有类型提示和校验
- validate() 在加载时就捕获配置错误，避免运行时才发现
- canvas_size 用 tuple 存储，JSON 里用 list 表示

### 练习三

**思路**：断点续跑的关键是把已完成 SKU 的状态持久化到文件，下次启动时先读取进度文件，跳过已完成的 SKU。

**答案**：

```python
import json
import os
from pathlib import Path


class PipelineProgressTracker:
    """Pipeline 进度追踪器，支持断点续跑"""

    def __init__(self, progress_path: str = "./output/progress.json"):
        self.progress_path = progress_path
        self.completed_skus: dict[str, dict] = {}
        self._load_progress()

    def _load_progress(self):
        if os.path.exists(self.progress_path):
            with open(self.progress_path, "r", encoding="utf-8") as f:
                data = json.load(f)
                self.completed_skus = data.get("completed", {})
            print(f"加载已有进度: {len(self.completed_skus)} 个 SKU 已完成")

    def _save_progress(self):
        os.makedirs(os.path.dirname(self.progress_path), exist_ok=True)
        data = {
            "completed": self.completed_skus,
            "total_completed": len(self.completed_skus),
        }
        with open(self.progress_path, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def is_completed(self, sku_id: str) -> bool:
        return sku_id in self.completed_skus

    def mark_completed(self, sku_id: str, output_paths: dict, processing_time: float):
        self.completed_skus[sku_id] = {
            "output_paths": output_paths,
            "processing_time": processing_time,
        }
        self._save_progress()

    def filter_pending_tasks(self, tasks: list) -> list:
        """过滤掉已完成的任务，返回待处理列表"""
        pending = [t for t in tasks if not self.is_completed(t.sku_id)]
        skipped = len(tasks) - len(pending)
        if skipped > 0:
            print(f"跳过 {skipped} 个已完成的 SKU，剩余 {len(pending)} 个待处理")
        return pending


def process_batch_with_checkpoint(pipeline, tasks, tracker):
    """带断点续跑的批量处理"""
    pending_tasks = tracker.filter_pending_tasks(tasks)

    for task in pending_tasks:
        try:
            completed = pipeline._process_single_task(task)
            tracker.mark_completed(
                sku_id=completed.sku_id,
                output_paths=completed.output_paths,
                processing_time=completed.processing_time,
            )
            print(f"✓ {completed.sku_id} 完成 ({completed.processing_time:.1f}s)")
        except Exception as e:
            print(f"✗ {task.sku_id} 失败: {e}")

    return tracker.completed_skus
```
