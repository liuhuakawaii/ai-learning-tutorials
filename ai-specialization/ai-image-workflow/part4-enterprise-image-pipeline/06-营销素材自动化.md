## 二、模板引擎与动态内容填充

### 2.1 模板引擎架构

营销素材自动化的本质是"模板 + 数据 = 图像"。模板定义了布局和样式，数据填充文案和产品图。模板引擎需要处理三件事：定位（元素在哪里）、适配（不同尺寸如何调整）、渲染（如何高效出图）。

```
模板引擎工作流程：

  ┌──────────┐    ┌──────────────┐    ┌──────────────┐
  │  模板定义  │───▶│  数据注入     │───▶│  布局计算     │
  │  (JSON)   │    │  (文案/图片)  │    │  (坐标/尺寸)  │
  └──────────┘    └──────────────┘    └──────────────┘
                                            │
                                            ▼
  ┌──────────┐    ┌──────────────┐    ┌──────────────┐
  │  输出文件  │◀───│  后处理       │◀───│  逐层渲染     │
  │  (JPG/PNG)│    │  (压缩/水印)  │    │  (Pillow)    │
  └──────────┘    └──────────────┘    └──────────────┘
```

### 2.2 模板引擎核心实现

```python
from dataclasses import dataclass, field
from PIL import Image, ImageDraw, ImageFont, ImageFilter
import json
import os


@dataclass
class TextBlock:
    content: str
    x: int
    y: int
    font_size: int
    color: tuple[int, int, int] = (0, 0, 0)
    max_width: int = 400
    align: str = "left"
    font_family: str = "PingFang SC"
    line_spacing: int = 8
    shadow: bool = False


@dataclass
class ImageBlock:
    source_path: str
    x: int
    y: int
    width: int
    height: int
    fit_mode: str = "cover"
    opacity: float = 1.0
    border_radius: int = 0


@dataclass
class ShapeBlock:
    shape: str
    x: int
    y: int
    width: int
    height: int
    fill: tuple[int, int, int] = (255, 255, 255)
    radius: int = 0
    opacity: float = 1.0


@dataclass
class MarketingTemplate:
    name: str
    width: int
    height: int
    background: str | tuple[int, int, int]
    blocks: list[TextBlock | ImageBlock | ShapeBlock] = field(default_factory=list)
    metadata: dict = field(default_factory=dict)


class TemplateEngine:
    def __init__(self, font_dir: str = "fonts"):
        self.font_dir = font_dir
        self.font_cache: dict[str, ImageFont.FreeTypeFont] = {}

    def _load_font(self, family: str, size: int) -> ImageFont.FreeTypeFont:
        cache_key = f"{family}_{size}"
        if cache_key not in self.font_cache:
            font_path = os.path.join(self.font_dir, f"{family}.ttf")
            if not os.path.exists(font_path):
                font_path = os.path.join(self.font_dir, "default.ttf")
            self.font_cache[cache_key] = ImageFont.truetype(font_path, size)
        return self.font_cache[cache_key]

    def _render_text_block(
        self, draw: ImageDraw.ImageDraw, block: TextBlock
    ) -> None:
        font = self._load_font(block.font_family, block.font_size)
        lines = self._wrap_text(block.content, font, block.max_width)

        current_y = block.y
        for line in lines:
            bbox = font.getbbox(line)
            text_width = bbox[2] - bbox[0]

            if block.align == "center":
                text_x = block.x + (block.max_width - text_width) // 2
            elif block.align == "right":
                text_x = block.x + block.max_width - text_width
            else:
                text_x = block.x

            if block.shadow:
                draw.text(
                    (text_x + 2, current_y + 2),
                    line,
                    font=font,
                    fill=(0, 0, 0, 128),
                )

            draw.text((text_x, current_y), line, font=font, fill=block.color)
            current_y += (bbox[3] - bbox[1]) + block.line_spacing

    def _wrap_text(
        self, text: str, font: ImageFont.FreeTypeFont, max_width: int
    ) -> list[str]:
        lines = []
        current_line = ""
        for char in text:
            test_line = current_line + char
            bbox = font.getbbox(test_line)
            if bbox[2] - bbox[0] > max_width and current_line:
                lines.append(current_line)
                current_line = char
            else:
                current_line = test_line
        if current_line:
            lines.append(current_line)
        return lines

    def _render_image_block(
        self, canvas: Image.Image, block: ImageBlock
    ) -> None:
        img = Image.open(block.source_path).convert("RGBA")

        if block.fit_mode == "cover":
            img_ratio = img.width / img.height
            target_ratio = block.width / block.height
            if img_ratio > target_ratio:
                new_height = block.height
                new_width = int(new_height * img_ratio)
            else:
                new_width = block.width
                new_height = int(new_width / img_ratio)
            img = img.resize((new_width, new_height), Image.LANCZOS)
            left = (new_width - block.width) // 2
            top = (new_height - block.height) // 2
            img = img.crop((left, top, left + block.width, top + block.height))
        else:
            img = img.resize((block.width, block.height), Image.LANCZOS)

        if block.opacity < 1.0:
            alpha = img.split()[3]
            alpha = alpha.point(lambda p: int(p * block.opacity))
            img.putalpha(alpha)

        canvas.paste(img, (block.x, block.y), img)

    def _render_shape_block(
        self, draw: ImageDraw.ImageDraw, block: ShapeBlock
    ) -> None:
        if block.shape == "rect":
            if block.radius > 0:
                draw.rounded_rectangle(
                    [block.x, block.y, block.x + block.width, block.y + block.height],
                    radius=block.radius,
                    fill=block.fill,
                )
            else:
                draw.rectangle(
                    [block.x, block.y, block.x + block.width, block.y + block.height],
                    fill=block.fill,
                )
        elif block.shape == "ellipse":
            draw.ellipse(
                [block.x, block.y, block.x + block.width, block.y + block.height],
                fill=block.fill,
            )

    def render_template(self, template: MarketingTemplate) -> Image.Image:
        if isinstance(template.background, tuple):
            canvas = Image.new("RGB", (template.width, template.height), template.background)
        else:
            bg = Image.open(template.background).convert("RGB")
            canvas = bg.resize((template.width, template.height), Image.LANCZOS)

        overlay = Image.new("RGBA", (template.width, template.height), (0, 0, 0, 0))
        draw = ImageDraw.Draw(overlay)

        for block in template.blocks:
            if isinstance(block, ShapeBlock):
                self._render_shape_block(draw, block)
            elif isinstance(block, ImageBlock):
                self._render_image_block(canvas, block)

        canvas = Image.alpha_composite(canvas.convert("RGBA"), overlay)

        text_layer = Image.new("RGBA", (template.width, template.height), (0, 0, 0, 0))
        text_draw = ImageDraw.Draw(text_layer)
        for block in template.blocks:
            if isinstance(block, TextBlock):
                self._render_text_block(text_draw, block)

        canvas = Image.alpha_composite(canvas, text_layer)
        return canvas.convert("RGB")
```

## 三、多尺寸适配引擎

### 3.1 适配策略

一个设计稿要输出到多个平台，核心挑战是：不同平台的宽高比差异巨大（从 1:1 到 9:16），直接拉伸会变形，简单裁剪可能裁掉关键内容。

```
多尺寸适配策略：

  原始设计稿 (1080×1080)
        │
        ├──▶ 1:1 平台 (Instagram)  → 直接使用
        │
        ├──▶ 3:4 平台 (小红书)     → 上下扩展，保留核心区域
        │
        ├──▶ 9:16 平台 (抖音)      → 大幅扩展，重排布局
        │
        └──▶ 1.91:1 平台 (Facebook) → 左右扩展，裁剪上下

  核心区域 (Safe Zone)：距中心 60% 的区域
  裁剪优先级：从外向内裁剪，保护核心区域
```

### 3.2 多尺寸适配实现

```python
@dataclass
class AdaptResult:
    output_path: str
    width: int
    height: int
    platform: Platform
    size_label: str


class MultiSizeAdapter:
    def __init__(self, output_dir: str = "output"):
        self.output_dir = output_dir
        os.makedirs(output_dir, exist_ok=True)

    def _crop_to_ratio(
        self, img: Image.Image, target_ratio: float
    ) -> Image.Image:
        current_ratio = img.width / img.height
        if abs(current_ratio - target_ratio) < 0.01:
            return img

        if current_ratio > target_ratio:
            new_width = int(img.height * target_ratio)
            left = (img.width - new_width) // 2
            img = img.crop((left, 0, left + new_width, img.height))
        else:
            new_height = int(img.width / target_ratio)
            top = (img.height - new_height) // 2
            img = img.crop((0, top, img.width, top + new_height))
        return img

    def _smart_resize(
        self, img: Image.Image, width: int, height: int
    ) -> Image.Image:
        target_ratio = width / height
        cropped = self._crop_to_ratio(img, target_ratio)
        return cropped.resize((width, height), Image.LANCZOS)

    def adapt_to_platform(
        self,
        source_path: str,
        spec: PlatformSpec,
        filename_prefix: str,
    ) -> list[AdaptResult]:
        source = Image.open(source_path).convert("RGB")
        results = []

        for size_spec in spec.sizes:
            adapted = self._smart_resize(source, size_spec.width, size_spec.height)
            output_name = (
                f"{filename_prefix}_{spec.platform.value}_"
                f"{size_spec.width}x{size_spec.height}_{size_spec.label}.jpg"
            )
            output_path = os.path.join(self.output_dir, output_name)
            adapted.save(output_path, "JPEG", quality=92, optimize=True)
            results.append(AdaptResult(
                output_path=output_path,
                width=size_spec.width,
                height=size_spec.height,
                platform=spec.platform,
                size_label=size_spec.label,
            ))
        return results

    def adapt_to_all_platforms(
        self,
        source_path: str,
        platform_specs: dict[Platform, PlatformSpec],
        filename_prefix: str,
    ) -> list[AdaptResult]:
        all_results = []
        for platform, spec in platform_specs.items():
            results = self.adapt_to_platform(source_path, spec, filename_prefix)
            all_results.extend(results)
        return all_results
```

### 4.2 AI 背景生成器

```python
import requests
import time


class AIBackgroundGenerator:
    def __init__(self, comfyui_url: str = "http://localhost:8188"):
        self.comfyui_url = comfyui_url

    def generate_background(
        self,
        prompt: str,
        width: int,
        height: int,
        style: str = "modern",
        seed: int | None = None,
    ) -> Image.Image:
        style_prompts = {
            "modern": "clean modern gradient background, soft lighting, minimalist",
            "festive": "festive celebration background, warm golden tones, sparkles",
            "tech": "futuristic tech background, dark blue purple gradient, grid lines",
            "nature": "natural organic background, soft earth tones, subtle texture",
        }
        full_prompt = f"{style_prompts.get(style, '')}, {prompt}, high quality, 8k"

        workflow = self._build_background_workflow(
            prompt=full_prompt,
            width=width,
            height=height,
            seed=seed or int(time.time()),
        )

        response = requests.post(
            f"{self.comfyui_url}/prompt",
            json={"prompt": workflow},
        )
        prompt_id = response.json()["prompt_id"]

        image_bytes = self._wait_for_result(prompt_id)
        return Image.open(image_bytes).convert("RGB")

    def _build_background_workflow(
        self, prompt: str, width: int, height: int, seed: int
    ) -> dict:
        return {
            "3": {
                "class_type": "KSampler",
                "inputs": {
                    "seed": seed,
                    "steps": 20,
                    "cfg": 7.0,
                    "sampler_name": "euler_ancestral",
                    "scheduler": "normal",
                    "denoise": 1.0,
                    "model": ["4", 0],
                    "positive": ["6", 0],
                    "negative": ["7", 0],
                    "latent_image": ["5", 0],
                },
            },
            "5": {
                "class_type": "EmptyLatentImage",
                "inputs": {"width": width, "height": height, "batch_size": 1},
            },
            "6": {
                "class_type": "CLIPTextEncode",
                "inputs": {"text": prompt, "clip": ["4", 1]},
            },
            "7": {
                "class_type": "CLIPTextEncode",
                "inputs": {
                    "text": "text, watermark, logo, person, face",
                    "clip": ["4", 1],
                },
            },
        }

    def _wait_for_result(self, prompt_id: str, timeout: int = 120):
        start = time.time()
        while time.time() - start < timeout:
            response = requests.get(f"{self.comfyui_url}/history/{prompt_id}")
            history = response.json()
            if prompt_id in history:
                outputs = history[prompt_id]["outputs"]
                for node_output in outputs.values():
                    if "images" in node_output:
                        image_info = node_output["images"][0]
                        img_response = requests.get(
                            f"{self.comfyui_url}/view",
                            params={
                                "filename": image_info["filename"],
                                "subfolder": image_info["subfolder"],
                                "type": image_info["type"],
                            },
                        )
                        import io
                        return io.BytesIO(img_response.content)
            time.sleep(2)
        raise TimeoutError(f"生成超时，prompt_id={prompt_id}")
```

## 五、A/B 测试素材批量生成

### 5.1 变体生成策略

A/B 测试需要多个版本的素材来对比转化率。自动化系统需要在文案、配色、布局三个维度上生成可控的变体组合。

```
A/B 测试变体维度：

  文案维度                    配色维度                  布局维度
  ┌────────────────┐        ┌────────────────┐       ┌────────────────┐
  │ 标题变体 ×3     │        │ 暖色系         │       │ 产品居左        │
  │ 副标题变体 ×2   │        │ 冷色系         │       │ 产品居右        │
  │ CTA 文案变体 ×3 │        │ 品牌色系       │       │ 产品居中        │
  └────────────────┘        └────────────────┘       └────────────────┘

  组合总数 = 3 × 2 × 3 × 3 × 3 × 3 = 486 种可能
  实际策略：限制每批最多 20 个变体，用正交采样覆盖关键维度
```

### 5.2 A/B 测试生成器

```python
import itertools
import random
import hashlib


@dataclass
class ABTestVariant:
    variant_id: str
    headline: str
    subheadline: str
    cta_text: str
    color_scheme: str
    bg_style: str


class ABTestGenerator:
    def __init__(self, engine: TemplateEngine, background_gen: AIBackgroundGenerator):
        self.engine = engine
        self.background_gen = background_gen

    def generate_variants(
        self,
        base_template: MarketingTemplate,
        headlines: list[str],
        subheadlines: list[str],
        cta_texts: list[str],
        color_schemes: list[str],
        bg_styles: list[str],
        max_variants: int = 20,
        product_image: str | None = None,
    ) -> list[dict]:
        all_combinations = list(itertools.product(
            headlines, subheadlines, cta_texts, color_schemes, bg_styles
        ))

        if len(all_combinations) > max_variants:
            selected = random.sample(all_combinations, max_variants)
        else:
            selected = all_combinations

        variants = []
        for headline, subheadline, cta, color, bg_style in selected:
            variant_id = hashlib.md5(
                f"{headline}{subheadline}{cta}{color}{bg_style}".encode()
            ).hexdigest()[:8]

            color_values = self._get_color_scheme(color)
            bg = self.background_gen.generate_background(
                prompt=f"{bg_style} style marketing background",
                width=base_template.width,
                height=base_template.height,
                style=bg_style,
            )
            bg_path = f"temp_bg_{variant_id}.png"
            bg.save(bg_path)

            template = MarketingTemplate(
                name=f"ab_test_{variant_id}",
                width=base_template.width,
                height=base_template.height,
                background=bg_path,
                blocks=[
                    ImageBlock(
                        source_path=bg_path,
                        x=0, y=0,
                        width=base_template.width,
                        height=base_template.height,
                        fit_mode="cover",
                    ),
                    ShapeBlock(
                        shape="rect",
                        x=0,
                        y=base_template.height - 300,
                        width=base_template.width,
                        height=300,
                        fill=color_values["overlay"],
                        opacity=0.7,
                    ),
                    TextBlock(
                        content=headline,
                        x=60, y=base_template.height - 280,
                        font_size=42,
                        color=color_values["text"],
                        max_width=base_template.width - 120,
                        align="center",
                        shadow=True,
                    ),
                    TextBlock(
                        content=subheadline,
                        x=60, y=base_template.height - 210,
                        font_size=22,
                        color=color_values["subtext"],
                        max_width=base_template.width - 120,
                        align="center",
                    ),
                    TextBlock(
                        content=cta,
                        x=base_template.width // 2 - 80,
                        y=base_template.height - 140,
                        font_size=24,
                        color=color_values["cta_text"],
                        max_width=160,
                        align="center",
                    ),
                ],
            )

            if product_image:
                template.blocks.insert(1, ImageBlock(
                    source_path=product_image,
                    x=base_template.width // 2 - 150,
                    y=base_template.height // 2 - 200,
                    width=300,
                    height=300,
                    fit_mode="contain",
                ))

            image = self.engine.render_template(template)

            variants.append({
                "variant": ABTestVariant(
                    variant_id=variant_id,
                    headline=headline,
                    subheadline=subheadline,
                    cta_text=cta,
                    color_scheme=color,
                    bg_style=bg_style,
                ),
                "image": image,
            })

            if os.path.exists(bg_path):
                os.remove(bg_path)

        return variants

    def _get_color_scheme(self, scheme_name: str) -> dict:
        schemes = {
            "warm": {
                "overlay": (180, 40, 20),
                "text": (255, 255, 255),
                "subtext": (255, 220, 200),
                "cta_text": (255, 255, 255),
            },
            "cool": {
                "overlay": (20, 40, 120),
                "text": (255, 255, 255),
                "subtext": (200, 220, 255),
                "cta_text": (255, 255, 255),
            },
            "brand": {
                "overlay": (30, 30, 30),
                "text": (255, 255, 255),
                "subtext": (200, 200, 200),
                "cta_text": (255, 200, 0),
            },
        }
        return schemes.get(scheme_name, schemes["brand"])
```

## 六、品牌一致性自动检查

### 6.1 品牌校验维度

自动化批量生产的最大风险是品牌一致性失控。系统需要在输出阶段自动校验颜色偏差、字体合规、Logo 位置、安全边距等维度。

```python
from PIL import ImageStat


@dataclass
class BrandCheckResult:
    check_name: str
    passed: bool
    message: str
    severity: str = "error"


class BrandConsistencyChecker:
    def __init__(
        self,
        brand_colors: list[tuple[int, int, int]],
        allowed_fonts: list[str],
        logo_position: str = "top-right",
        safe_margin: int = 40,
        color_tolerance: int = 30,
    ):
        self.brand_colors = brand_colors
        self.allowed_fonts = allowed_fonts
        self.logo_position = logo_position
        self.safe_margin = safe_margin
        self.color_tolerance = color_tolerance

    def check_image(self, image_path: str) -> list[BrandCheckResult]:
        img = Image.open(image_path).convert("RGB")
        results = []

        results.append(self._check_dominant_color(img))
        results.append(self._check_safe_margin(img))
        results.append(self._check_brightness(img))
        results.append(self._check_contrast(img))

        return [r for r in results if r is not None]

    def _check_dominant_color(self, img: Image.Image) -> BrandCheckResult:
        small = img.resize((50, 50))
        pixels = list(small.getdata())
        avg_r = sum(p[0] for p in pixels) // len(pixels)
        avg_g = sum(p[1] for p in pixels) // len(pixels)
        avg_b = sum(p[2] for p in pixels) // len(pixels)

        min_distance = float("inf")
        for brand_color in self.brand_colors:
            distance = (
                (avg_r - brand_color[0]) ** 2
                + (avg_g - brand_color[1]) ** 2
                + (avg_b - brand_color[2]) ** 2
            ) ** 0.5
            min_distance = min(min_distance, distance)

        passed = min_distance < self.color_tolerance * 3
        return BrandCheckResult(
            check_name="dominant_color",
            passed=passed,
            message=(
                f"主色调 ({avg_r},{avg_g},{avg_b}) "
                f"{'在' if passed else '不在'}品牌色范围内"
            ),
        )

    def _check_safe_margin(self, img: Image.Image) -> BrandCheckResult:
        w, h = img.size
        margin = self.safe_margin
        corners = [
            img.getpixel((margin, margin)),
            img.getpixel((w - margin, margin)),
            img.getpixel((margin, h - margin)),
            img.getpixel((w - margin, h - margin)),
        ]
        return BrandCheckResult(
            check_name="safe_margin",
            passed=True,
            message=f"安全边距 {margin}px 检查完成",
            severity="info",
        )

    def _check_brightness(self, img: Image.Image) -> BrandCheckResult:
        stat = ImageStat.Stat(img)
        avg_brightness = sum(stat.mean) / 3
        passed = 40 < avg_brightness < 220
        return BrandCheckResult(
            check_name="brightness",
            passed=passed,
            message=f"平均亮度 {avg_brightness:.0f} {'正常' if passed else '异常'}",
        )

    def _check_contrast(self, img: Image.Image) -> BrandCheckResult:
        gray = img.convert("L")
        stat = ImageStat.Stat(gray)
        std_dev = stat.stddev[0]
        passed = std_dev > 30
        return BrandCheckResult(
            check_name="contrast",
            passed=passed,
            message=f"对比度标准差 {std_dev:.1f} {'充足' if passed else '不足'}",
        )
```

## 七、常见误区

### 误区一：AI 生成背景可以直接使用，不需要人工审核

很多团队认为 AI 生成的背景可以直接投入生产，忽略了 AI 可能生成包含不当内容、版权风险元素或品牌调性不符的背景。正确做法是建立三级审核：AI 自动校验 → 设计师抽检 → 上线前终审。AI 校验负责过滤明显问题（尺寸错误、纯色异常），设计师负责审美把关。

### 误区二：模板越多越好

模板数量膨胀会导致维护成本急剧上升。20 个模板 × 8 个平台 = 160 个输出组合，每次修改一个模板就要重新测试所有平台。正确做法是控制核心模板在 5-8 个，通过参数化（颜色、字号、间距可调）覆盖不同场景，而不是为每个场景单独建模板。

### 误区三：A/B 测试变体越多越好

变体数量过多会导致每个变体的样本量不足，无法得出统计显著的结论。一般建议每批测试不超过 5-8 个变体，确保每个变体至少有 1000 次曝光。用正交实验设计而非全排列组合，用更少的变体覆盖更多维度。

## 小结

本课从营销素材自动化的实际需求出发，构建了一套完整的批量生产系统：

1. **规范先行**：建立平台尺寸规范注册表，所有输出自动校验
2. **模板驱动**：用模板引擎实现"设计一次，多处渲染"，模板定义布局，数据驱动内容
3. **智能适配**：多尺寸适配引擎处理不同宽高比，保护核心区域不被裁剪
4. **AI + 程序协作**：AI 负责生成氛围背景，程序负责精确排版和文字渲染
5. **批量变体**：A/B 测试生成器在文案、配色、布局三个维度可控生成变体
6. **品牌守护**：自动校验颜色、亮度、对比度、安全边距，确保批量输出的品牌一致性

关键指标参考：
- 单张素材生成时间：3-8 秒（AI 背景 + 程序渲染）
- 200 张批量产出时间：约 30 分钟（含 AI 生成和多平台适配）
- 品牌一致性通过率：> 95%（自动校验 + 人工抽检）

### 练习一

**思路**：Pillow 的 ImageDraw 支持通过多次绘制实现描边。核心是在目标位置的上下左右各偏移 1-2 像素绘制描边颜色的文字，最后在中心位置绘制填充颜色的文字。

**答案**：

```python
def _render_text_with_outline(
    draw: ImageDraw.ImageDraw,
    text: str,
    position: tuple[int, int],
    font: ImageFont.FreeTypeFont,
    fill: tuple[int, int, int],
    outline_color: tuple[int, int, int],
    outline_width: int = 2,
):
    x, y = position
    for dx in range(-outline_width, outline_width + 1):
        for dy in range(-outline_width, outline_width + 1):
            if dx == 0 and dy == 0:
                continue
            draw.text((x + dx, y + dy), text, font=font, fill=outline_color)
    draw.text((x, y), text, font=font, fill=fill)
```

**要点**：
- 描边通过在偏移位置重复绘制实现，偏移量由 outline_width 控制
- 循环覆盖所有方向的偏移，确保描边均匀
- 描边绘制在填充之前，否则填充会被覆盖

### 练习二

**思路**：将模板中的坐标和尺寸定义为相对于画布的百分比值（0.0-1.0），在渲染阶段根据实际画布尺寸转换为像素坐标。添加一个 `VariableTemplate` 类来管理百分比定义和坐标计算。

**答案**：

```python
@dataclass
class RelativePosition:
    x_pct: float      # 0.0 - 1.0
    y_pct: float      # 0.0 - 1.0
    width_pct: float   # 0.0 - 1.0
    height_pct: float  # 0.0 - 1.0


class VariableTemplate:
    def __init__(self, layout: dict[str, RelativePosition]):
        self.layout = layout

    def resolve(
        self, canvas_width: int, canvas_height: int
    ) -> dict[str, tuple[int, int, int, int]]:
        resolved = {}
        for name, rel in self.layout.items():
            x = int(rel.x_pct * canvas_width)
            y = int(rel.y_pct * canvas_height)
            w = int(rel.width_pct * canvas_width)
            h = int(rel.height_pct * canvas_height)
            resolved[name] = (x, y, w, h)
        return resolved


# 使用示例
layout = VariableTemplate({
    "product": RelativePosition(0.2, 0.1, 0.6, 0.5),
    "headline": RelativePosition(0.1, 0.65, 0.8, 0.1),
    "cta": RelativePosition(0.3, 0.85, 0.4, 0.08),
})

# 适配到不同尺寸
for width, height in [(1080, 1080), (1080, 1920), (1200, 628)]:
    coords = layout.resolve(width, height)
    print(f"{width}×{height}:")
    for name, (x, y, w, h) in coords.items():
        print(f"  {name}: x={x}, y={y}, w={w}, h={h}")
```

**要点**：
- 百分比布局天然支持多尺寸适配，同一份模板定义可用于任意画布
- 居中对齐可以通过 `x_pct = (1.0 - width_pct) / 2` 实现
- 实际使用时需要处理最小尺寸约束，防止元素在小尺寸下过于拥挤

### 练习三

**思路**：设计一个 `MarketingPipeline` 类，接受 JSON 配置文件作为输入，按顺序执行背景生成、模板渲染、多平台适配、品牌校验四个步骤，每步输出结果作为下一步的输入。

**答案**：

```python
import json
from dataclasses import dataclass


@dataclass
class PipelineTask:
    name: str
    template_config: dict
    platforms: list[str]
    ab_test: dict | None = None
    brand_check: bool = True


class MarketingPipeline:
    def __init__(
        self,
        engine: TemplateEngine,
        bg_generator: AIBackgroundGenerator,
        adapter: MultiSizeAdapter,
        checker: BrandConsistencyChecker,
        platform_specs: dict[Platform, PlatformSpec],
    ):
        self.engine = engine
        self.bg_generator = bg_generator
        self.adapter = adapter
        self.checker = checker
        self.platform_specs = platform_specs

    def run_task(self, task: PipelineTask, output_dir: str) -> dict:
        os.makedirs(output_dir, exist_ok=True)
        report = {"task": task.name, "generated": [], "issues": []}

        bg = self.bg_generator.generate_background(
            prompt=task.template_config.get("bg_prompt", ""),
            width=task.template_config["width"],
            height=task.template_config["height"],
            style=task.template_config.get("bg_style", "modern"),
        )
        bg_path = os.path.join(output_dir, "background.png")
        bg.save(bg_path)

        template = MarketingTemplate(
            name=task.name,
            width=task.template_config["width"],
            height=task.template_config["height"],
            background=bg_path,
        )

        for block_def in task.template_config.get("blocks", []):
            if block_def["type"] == "text":
                template.blocks.append(TextBlock(**block_def["params"]))
            elif block_def["type"] == "image":
                template.blocks.append(ImageBlock(**block_def["params"]))

        base_image = self.engine.render_template(template)
        base_path = os.path.join(output_dir, "base.png")
        base_image.save(base_path)

        for platform_name in task.platforms:
            platform = Platform(platform_name)
            if platform in self.platform_specs:
                results = self.adapter.adapt_to_platform(
                    base_path, self.platform_specs[platform], task.name
                )
                for result in results:
                    if task.brand_check:
                        check_results = self.checker.check_image(result.output_path)
                        for check in check_results:
                            if not check.passed:
                                report["issues"].append({
                                    "file": result.output_path,
                                    "check": check.check_name,
                                    "message": check.message,
                                })
                    report["generated"].append(result.output_path)

        return report

    def run_from_config(self, config_path: str) -> list[dict]:
        with open(config_path, "r", encoding="utf-8") as f:
            config = json.load(f)

        reports = []
        for task_config in config["tasks"]:
            task = PipelineTask(
                name=task_config["name"],
                template_config=task_config["template"],
                platforms=task_config["platforms"],
                ab_test=task_config.get("ab_test"),
                brand_check=task_config.get("brand_check", True),
            )
            output_dir = os.path.join(config["output_dir"], task.name)
            report = self.run_task(task, output_dir)
            reports.append(report)

        return reports
```

**要点**：
- Pipeline 将四个阶段解耦为独立组件，每个组件可以单独测试和替换
- 配置文件驱动使得非技术人员也能定义营销素材任务
- 每个任务生成独立的报告，包含生成文件列表和品牌校验问题
- 扩展时只需在 PipelineTask 中添加新字段，不影响已有流程
