# 第 21 课：批量生产架构 — 从单张生成到企业级流水线

你是一个电商团队的技术负责人。5000 个 SKU，每个需要 4-6 张不同场景的产品图——白底图、场景图、节日主题图、社交媒体图。总共 20000+ 张，每月上新 200 个 SKU 意味着持续生产。

单张生成效率是每小时 20 张（含 prompt 调整、参数调试、质量检查）。20000 张需要 1000 小时。你需要的不是更快的生成方式，而是一套自动接收任务、调度 GPU 资源、处理失败重试、监控进度的架构。

## 架构选型：队列式

选择队列式架构而不是简单并行脚本，原因：

1. **解耦**：任务提交和执行分离，提交方不需要关心有多少 GPU
2. **可靠性**：队列保证任务不丢失，worker 崩溃后任务重新回到队列
3. **可扩展**：增加 GPU 只需要启动新 worker
4. **优先级**：紧急任务可以插队

```
任务提交端                    任务执行端
─────────                    ─────────
                             ┌──────────┐
┌──────────┐    ┌────────┐  │ Worker 1 │  (GPU 0)
│ 批量任务  │───►│ Redis  │──│ Worker 2 │  (GPU 1)
│ 提交器   │    │ Queue  │  │ Worker 3 │  (GPU 2)
└──────────┘    └────────┘  └──────────┘
      │               │          │
      ▼               ▼          ▼
┌──────────┐    ┌────────┐  ┌──────────┐
│ SQLite   │    │ Redis  │  │ 文件系统  │
│ 任务状态  │    │ Cache  │  │ 结果存储  │
└──────────┘    └────────┘  └──────────┘
```

## 任务定义

```python
import json, uuid, time
from enum import Enum
from dataclasses import dataclass, field, asdict
from datetime import datetime


class TaskStatus(str, Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    RETRYING = "retrying"


class TaskPriority(int, Enum):
    LOW = 0
    NORMAL = 1
    HIGH = 2
    URGENT = 3


@dataclass
class GenerationTask:
    task_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    sku_id: str = ""
    scene_type: str = ""
    prompt: str = ""
    negative_prompt: str = ""
    width: int = 1024
    height: int = 1024
    num_images: int = 1
    seed: int = -1
    checkpoint: str = ""
    lora_path: str = ""
    lora_strength: float = 0.8
    priority: TaskPriority = TaskPriority.NORMAL
    status: TaskStatus = TaskStatus.PENDING
    retry_count: int = 0
    max_retries: int = 3
    created_at: str = field(default_factory=lambda: datetime.now().isoformat())
    error_message: str = ""
    output_paths: list = field(default_factory=list)

    def to_json(self):
        return json.dumps(asdict(self), ensure_ascii=False, default=str)

    @classmethod
    def from_json(cls, s):
        d = json.loads(s)
        d["priority"] = TaskPriority(d["priority"])
        d["status"] = TaskStatus(d["status"])
        return cls(**d)

    def mark_failed(self, error):
        self.error_message = error
        if self.retry_count < self.max_retries:
            self.status = TaskStatus.RETRYING
            self.retry_count += 1
        else:
            self.status = TaskStatus.FAILED
```

## 任务队列（Redis）

```python
import redis

class TaskQueue:
    QUEUE_KEY = "gen:queue"
    PROCESSING_KEY = "gen:processing"
    COMPLETED_KEY = "gen:completed"
    FAILED_KEY = "gen:failed"
    TASK_PREFIX = "gen:task:"

    def __init__(self, redis_url="redis://localhost:6379/0"):
        self.r = redis.from_url(redis_url, decode_responses=True)

    def enqueue(self, task: GenerationTask):
        self.r.set(f"{self.TASK_PREFIX}{task.task_id}", task.to_json())
        self.r.zadd(self.QUEUE_KEY, {task.task_id: task.priority.value})

    def dequeue(self):
        result = self.r.zpopmax(self.QUEUE_KEY)
        if not result:
            return None
        task_id, _ = result[0]
        data = self.r.get(f"{self.TASK_PREFIX}{task_id}")
        if not data:
            return None
        task = GenerationTask.from_json(data)
        task.status = TaskStatus.RUNNING
        self.r.set(f"{self.TASK_PREFIX}{task.task_id}", task.to_json())
        self.r.sadd(self.PROCESSING_KEY, task.task_id)
        return task

    def complete(self, task):
        task.status = TaskStatus.COMPLETED
        self.r.set(f"{self.TASK_PREFIX}{task.task_id}", task.to_json())
        self.r.srem(self.PROCESSING_KEY, task.task_id)
        self.r.sadd(self.COMPLETED_KEY, task.task_id)

    def fail(self, task, error):
        task.mark_failed(error)
        self.r.set(f"{self.TASK_PREFIX}{task.task_id}", task.to_json())
        self.r.srem(self.PROCESSING_KEY, task.task_id)
        if task.status == TaskStatus.RETRYING:
            self.r.zadd(self.QUEUE_KEY, {task.task_id: task.priority.value})
        else:
            self.r.sadd(self.FAILED_KEY, task.task_id)

    def stats(self):
        return {
            "pending": self.r.zcard(self.QUEUE_KEY),
            "processing": self.r.scard(self.PROCESSING_KEY),
            "completed": self.r.scard(self.COMPLETED_KEY),
            "failed": self.r.scard(self.FAILED_KEY),
        }
```

## GPU 资源管理

```python
import subprocess
from threading import Lock

class GPUManager:
    def __init__(self, min_free_mb=4000):
        self.min_free_mb = min_free_mb
        self._lock = Lock()
        self._allocated: dict[int, str] = {}

    def query_gpus(self):
        try:
            r = subprocess.run(
                ["nvidia-smi", "--query-gpu=index,name,memory.free,temperature.gpu",
                 "--format=csv,noheader,nounits"],
                capture_output=True, text=True, timeout=10)
            gpus = []
            for line in r.stdout.strip().split("\n"):
                parts = [p.strip() for p in line.split(",")]
                if len(parts) >= 4:
                    free = int(parts[2])
                    gpus.append({"id": int(parts[0]), "name": parts[1],
                                "free_mb": free, "temp": int(parts[3]),
                                "available": free >= self.min_free_mb})
            return gpus
        except Exception:
            return []

    def allocate(self, worker_id):
        with self._lock:
            gpus = self.query_gpus()
            available = [g for g in gpus
                        if g["available"] and g["id"] not in self._allocated]
            if not available:
                return None
            best = max(available, key=lambda g: g["free_mb"])
            self._allocated[best["id"]] = worker_id
            return best["id"]

    def release(self, device_id):
        with self._lock:
            self._allocated.pop(device_id, None)
```

## Worker

```python
import os, time, torch

class Worker:
    def __init__(self, worker_id, queue, gpu_mgr, output_dir):
        self.id = worker_id
        self.queue = queue
        self.gpu_mgr = gpu_mgr
        self.output_dir = output_dir
        self.gpu = None
        self.running = False
        self.stats = {"completed": 0, "failed": 0, "images": 0}

    def start(self):
        self.running = True
        self.gpu = self.gpu_mgr.allocate(self.id)
        while not self.gpu and self.running:
            time.sleep(5)
            self.gpu = self.gpu_mgr.allocate(self.id)
        if not self.running:
            return

        print(f"[{self.id}] 启动, GPU {self.gpu}")
        while self.running:
            task = self.queue.dequeue()
            if task is None:
                time.sleep(1)
                continue
            self._process(task)

    def stop(self):
        self.running = False
        if self.gpu is not None:
            self.gpu_mgr.release(self.gpu)

    def _process(self, task):
        print(f"[{self.id}] 处理 {task.task_id} (SKU: {task.sku_id})")
        try:
            os.environ["CUDA_VISIBLE_DEVICES"] = str(self.gpu)
            paths = self._generate(task)
            task.output_paths = paths
            self.queue.complete(task)
            self.stats["completed"] += 1
            self.stats["images"] += len(paths)
        except Exception as e:
            self.queue.fail(task, f"{type(e).__name__}: {e}")
            self.stats["failed"] += 1

    def _generate(self, task):
        from diffusers import StableDiffusionXLPipeline, EulerAncestralDiscreteScheduler

        pipe = StableDiffusionXLPipeline.from_pretrained(
            task.checkpoint, torch_dtype=torch.float16).to("cuda")
        pipe.scheduler = EulerAncestralDiscreteScheduler.from_config(pipe.scheduler.config)

        if task.lora_path:
            pipe.load_lora_weights(task.lora_path)
            pipe.fuse_lora(lora_scale=task.lora_strength)

        paths = []
        for i in range(task.num_images):
            seed = task.seed if task.seed >= 0 else int(time.time()) + i
            gen = torch.Generator("cuda").manual_seed(seed)
            img = pipe(prompt=task.prompt, negative_prompt=task.negative_prompt,
                      width=task.width, height=task.height,
                      num_inference_steps=30, guidance_scale=7.0,
                      generator=gen).images[0]
            path = os.path.join(self.output_dir, f"{task.sku_id}_{task.scene_type}_{seed}.png")
            img.save(path)
            paths.append(path)

        del pipe
        torch.cuda.empty_cache()
        return paths
```

## 错误处理

```python
class ErrorHandler:
    RECOVERABLE = {"CUDA out of memory", "TimeoutError", "ConnectionError"}
    FATAL = {"FileNotFoundError", "PermissionError", "SafetensorError"}

    def classify(self, error):
        msg = str(error)
        for pattern in self.FATAL:
            if pattern in msg:
                return "fatal"
        return "recoverable"

    def handle(self, task, error):
        category = self.classify(error)
        if category == "fatal":
            return {"action": "skip", "reason": str(error)}
        if task.retry_count < task.max_retries:
            delay = min(5 * (2 ** task.retry_count), 60)
            return {"action": "retry", "delay": delay}
        return {"action": "fail", "reason": f"重试耗尽: {error}"}

    def adjust_for_oom(self, task):
        """OOM 时降低分辨率重试"""
        if task.width > 512:
            task.width = max(512, task.width // 2)
        if task.height > 512:
            task.height = max(512, task.height // 2)
        task.num_images = 1
```

## 并发控制与监控

```python
import threading
from concurrent.futures import ThreadPoolExecutor

class Pipeline:
    def __init__(self, redis_url, output_dir, max_workers=4):
        self.queue = TaskQueue(redis_url)
        self.gpu_mgr = GPUManager()
        self.output_dir = output_dir
        self.workers = []
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
        self._shutdown = False

    def start(self):
        for i in range(self.executor._max_workers):
            w = Worker(f"worker_{i}", self.queue, self.gpu_mgr, self.output_dir)
            self.workers.append(w)
            self.executor.submit(w.start)
        threading.Thread(target=self._monitor, daemon=True).start()

    def stop(self):
        self._shutdown = True
        for w in self.workers:
            w.stop()
        self.executor.shutdown(wait=True)

    def submit_batch(self, tasks):
        for t in tasks:
            self.queue.enqueue(t)
        print(f"提交 {len(tasks)} 个任务")

    def _monitor(self):
        while not self._shutdown:
            time.sleep(10)
            stats = self.queue.stats()
            print(f"队列: 待处理={stats['pending']}, "
                  f"处理中={stats['processing']}, "
                  f"完成={stats['completed']}, 失败={stats['failed']}")
```

def create_tasks(sku_list, scenes, checkpoint, lora=""):
    prompts = {
        "white_bg": "product photography, white background, studio lighting",
        "lifestyle": "product in lifestyle setting, warm lighting",
        "social": "social media style product photo, trendy",
    }
    tasks = []
    for sku in sku_list:
        for scene in scenes:
            tasks.append(GenerationTask(
                sku_id=sku["id"], scene_type=scene,
                prompt=f"{sku['prefix']}, {prompts[scene]}, {sku['name']}",
                checkpoint=checkpoint, lora_path=lora,
                priority=TaskPriority.HIGH if sku.get("new") else TaskPriority.NORMAL,
            ))
    return tasks


pipeline = Pipeline("redis://localhost:6379", "./output", max_workers=4)
pipeline.start()

tasks = create_tasks(
    [{"id": "SKU001", "name": "白色陶瓷花瓶", "prefix": "brand", "new": True}],
    ["white_bg", "lifestyle"],
    "models/sdxl_base.safetensors",
)
pipeline.submit_batch(tasks)

# 等待完成...
pipeline.stop()

## 练习

### 练习一：架构设计

3 块 GPU（2 块 A100 80GB + 1 块 RTX 4090 24GB），每天 5000 张产品图。设计 worker 分配、优先级策略、GPU 资源方案。

### 练习二：OOM 诊断

前 1000 张生成顺利，之后频繁 CUDA OOM，但 GPU 使用率不高。分析原因并给出方案。

---

## 参考答案

### 练习一

A100 × 2 各分配 2 个 worker 处理 1024×1024，4090 分配 1 个 worker 处理 768×768。优先级：新品 URGENT > 营销 HIGH > 常规 NORMAL > 补图 LOW。5 个 worker 并行，每张约 15 秒，5000 张约 4.2 小时。

### 练习二

最可能原因：显存碎片化。长时间运行后 PyTorch 显存分配器产生碎片，虽然总空闲显存足够但无法分配连续大块。解决方案：每个任务后 `del pipeline; torch.cuda.empty_cache()`，设置 `PYTORCH_CUDA_ALLOC_CONF=max_split_size_mb:512`，每 100 个任务重启 worker。
