# 第 2 课：质量控制系统 — 让每张输出都达标

## 场景引入

你的批量生产流水线已经跑起来了，5 个 worker 每小时输出 1200 张图片。但问题随之而来——交付给客户的第一批 500 张图片中，有 30 张出现了人物手指畸变，15 张的色彩与品牌风格偏差明显，还有 5 张因为 prompt 理解错误生成了完全不相关的内容。

客户退回了这 50 张图片，要求重新生成。更糟糕的是，客户开始质疑整批图片的质量，要求对所有 500 张进行人工审核。一个设计师花了两天时间逐张检查，发现其中还有 20 张存在细微的质量问题——背景有伪影、文字区域模糊、光影方向不一致。

问题的根源不是生成技术不行，而是缺少一套系统化的质量控制流程。人工逐张检查不可持续，你需要一套**自动化的质量控制系统**：在图片生成后立即进行自动检测，不合格的自动重做，合格的进入人工抽检流程，最终输出的每一张图都经过了质量验证。

本课将构建这样一套质量控制系统，覆盖美学评分、人脸检测、内容安全、CLIP 匹配度检测和人工审核流程。

### 1.2 检测管线实现

```python
import numpy as np
from PIL import Image
from pathlib import Path
from dataclasses import dataclass, field
from enum import Enum
from datetime import datetime


class QualityVerdict(str, Enum):
    PASS = "pass"
    FAIL = "fail"
    WARNING = "warning"
    BLOCK = "block"


@dataclass
class QualityReport:
    """单张图片的质量报告"""
    image_path: str
    verdict: QualityVerdict = QualityVerdict.PASS
    aesthetic_score: float = 0.0
    clip_score: float = 0.0
    face_quality_score: float = 0.0
    nsfw_score: float = 0.0
    technical_score: float = 0.0
    overall_score: float = 0.0
    issues: list = field(default_factory=list)
    checked_at: str = field(default_factory=lambda: datetime.now().isoformat())
    
    def to_dict(self) -> dict:
        return {
            "image_path": self.image_path,
            "verdict": self.verdict.value,
            "scores": {
                "aesthetic": round(self.aesthetic_score, 3),
                "clip": round(self.clip_score, 3),
                "face_quality": round(self.face_quality_score, 3),
                "nsfw": round(self.nsfw_score, 3),
                "technical": round(self.technical_score, 3),
                "overall": round(self.overall_score, 3),
            },
            "issues": self.issues,
            "checked_at": self.checked_at,
        }


class QualityChecker:
    """质量检测管线"""
    
    def __init__(self, config: dict = None):
        self.config = config or self._default_config()
        self.checks = []
        self._init_checks()
    
    def _default_config(self) -> dict:
        return {
            "aesthetic_threshold": 5.5,
            "clip_threshold": 0.25,
            "face_quality_threshold": 0.8,
            "nsfw_threshold": 0.5,
            "technical_threshold": 7.0,
            "overall_threshold": 6.0,
            "weights": {
                "aesthetic": 0.30,
                "clip": 0.25,
                "face_quality": 0.20,
                "nsfw": 0.15,
                "technical": 0.10,
            },
        }
    
    def _init_checks(self):
        """初始化检测模块"""
        self.checks = [
            ("aesthetic", self._check_aesthetic),
            ("clip", self._check_clip_score),
            ("face", self._check_face_quality),
            ("nsfw", self._check_nsfw),
            ("technical", self._check_technical),
        ]
    
    def check(self, image_path: str, prompt: str = "") -> QualityReport:
        """执行完整的质量检测"""
        report = QualityReport(image_path=image_path)
        
        try:
            img = Image.open(image_path).convert("RGB")
        except Exception as e:
            report.verdict = QualityVerdict.BLOCK
            report.issues.append(f"无法打开图片: {e}")
            return report
        
        for check_name, check_fn in self.checks:
            try:
                result = check_fn(img, image_path, prompt)
                self._apply_result(report, check_name, result)
            except Exception as e:
                report.issues.append(f"{check_name} 检测失败: {e}")
        
        self._calculate_overall(report)
        self._make_verdict(report)
        
        return report
    
    def _apply_result(self, report: QualityReport, check_name: str, result: dict):
        """应用检测结果"""
        if check_name == "aesthetic":
            report.aesthetic_score = result["score"]
        elif check_name == "clip":
            report.clip_score = result["score"]
        elif check_name == "face":
            report.face_quality_score = result["score"]
        elif check_name == "nsfw":
            report.nsfw_score = result["score"]
        elif check_name == "technical":
            report.technical_score = result["score"]
        
        if result.get("issues"):
            report.issues.extend(result["issues"])
    
    def _calculate_overall(self, report: QualityReport):
        """计算综合分数"""
        weights = self.config["weights"]
        
        scores = {
            "aesthetic": min(report.aesthetic_score / 10.0, 1.0),
            "clip": min(report.clip_score / 0.5, 1.0),
            "face_quality": report.face_quality_score,
            "nsfw": 1.0 - report.nsfw_score,
            "technical": min(report.technical_score / 10.0, 1.0),
        }
        
        report.overall_score = sum(
            scores[dim] * weight for dim, weight in weights.items()
        ) * 10
    
    def _make_verdict(self, report: QualityReport):
        """做出最终判断"""
        if report.nsfw_score >= self.config["nsfw_threshold"]:
            report.verdict = QualityVerdict.BLOCK
            report.issues.append(f"NSFW 内容分数过高: {report.nsfw_score:.3f}")
            return
        
        if report.aesthetic_score < self.config["aesthetic_threshold"]:
            report.verdict = QualityVerdict.FAIL
            report.issues.append(f"美学评分过低: {report.aesthetic_score:.3f}")
        
        if report.clip_score < self.config["clip_threshold"]:
            report.verdict = QualityVerdict.FAIL
            report.issues.append(f"CLIP 匹配度过低: {report.clip_score:.3f}")
        
        if report.face_quality_score < self.config["face_quality_threshold"] and report.face_quality_score > 0:
            report.verdict = QualityVerdict.FAIL
            report.issues.append(f"人脸质量不佳: {report.face_quality_score:.3f}")
        
        if report.overall_score >= self.config["overall_threshold"] and not report.issues:
            report.verdict = QualityVerdict.PASS
        elif report.overall_score >= self.config["overall_threshold"] * 0.8:
            if report.verdict != QualityVerdict.FAIL:
                report.verdict = QualityVerdict.WARNING
    
    def _check_aesthetic(self, img: Image.Image, path: str, prompt: str) -> dict:
        """美学评分检测"""
        return AestheticScorer.score(img)
    
    def _check_clip_score(self, img: Image.Image, path: str, prompt: str) -> dict:
        """CLIP 匹配度检测"""
        if not prompt:
            return {"score": 1.0, "issues": []}
        return CLIPScorer.score(img, prompt)
    
    def _check_face_quality(self, img: Image.Image, path: str, prompt: str) -> dict:
        """人脸质量检测"""
        return FaceQualityChecker.check(img)
    
    def _check_nsfw(self, img: Image.Image, path: str, prompt: str) -> dict:
        """NSFW 内容检测"""
        return NSFWChecker.check(img)
    
    def _check_technical(self, img: Image.Image, path: str, prompt: str) -> dict:
        """技术质量检测"""
        return TechnicalChecker.check(img)
```

## 二、美学评分系统

### 2.1 美学预测模型

美学评分使用 LAION Aesthetics Predictor，它是一个基于 CLIP 的回归模型，预测人类对图片美学感受的 1-10 分评分。

```python
import torch
import numpy as np
from PIL import Image
from transformers import CLIPModel, CLIPProcessor


class AestheticScorer:
    """美学评分器"""
    
    _model = None
    _processor = None
    _aesthetic_head = None
    
    @classmethod
    def _load_model(cls):
        if cls._model is not None:
            return
        
        cls._processor = CLIPProcessor.from_pretrained("openai/clip-vit-large-patch14")
        cls._model = CLIPModel.from_pretrained("openai/clip-vit-large-patch14")
        
        cls._aesthetic_head = torch.nn.Linear(768, 1)
        
        cls._model.eval()
        cls._aesthetic_head.eval()
    
    @classmethod
    def score(cls, img: Image.Image) -> dict:
        """计算美学评分"""
        cls._load_model()
        
        inputs = cls._processor(images=img, return_tensors="pt")
        
        with torch.no_grad():
            image_features = cls._model.get_image_features(**inputs)
            image_features = image_features / image_features.norm(dim=-1, keepdim=True)
            
            aesthetic_score = cls._aesthetic_head(image_features.float()).item()
        
        issues = []
        if aesthetic_score < 4.0:
            issues.append(f"美学评分极低 ({aesthetic_score:.2f})，图片质量可能很差")
        elif aesthetic_score < 5.5:
            issues.append(f"美学评分偏低 ({aesthetic_score:.2f})，建议检查构图和色彩")
        
        return {
            "score": aesthetic_score,
            "issues": issues,
        }
```

### 2.2 基于图像特征的快速美学评估

当无法加载大模型时，可以使用基于图像统计特征的快速评估：

```python
import numpy as np
from PIL import Image, ImageStat


class FastAestheticScorer:
    """基于图像统计特征的快速美学评估"""
    
    @staticmethod
    def score(img: Image.Image) -> dict:
        """快速美学评分（不需要大模型）"""
        img_array = np.array(img).astype(float)
        
        scores = {}
        
        scores["brightness"] = FastAestheticScorer._score_brightness(img_array)
        scores["contrast"] = FastAestheticScorer._score_contrast(img_array)
        scores["saturation"] = FastAestheticScorer._score_saturation(img_array)
        scores["sharpness"] = FastAestheticScorer._score_sharpness(img_array)
        scores["color_harmony"] = FastAestheticScorer._score_color_harmony(img_array)
        
        weights = {
            "brightness": 0.15,
            "contrast": 0.20,
            "saturation": 0.20,
            "sharpness": 0.25,
            "color_harmony": 0.20,
        }
        
        overall = sum(scores[k] * weights[k] for k in weights)
        
        issues = []
        if scores["sharpness"] < 5.0:
            issues.append("图片可能模糊")
        if scores["contrast"] < 4.0:
            issues.append("对比度过低")
        if scores["brightness"] < 3.0 or scores["brightness"] > 8.0:
            issues.append("亮度异常")
        
        return {
            "score": overall,
            "sub_scores": scores,
            "issues": issues,
        }
    
    @staticmethod
    def _score_brightness(img_array: np.ndarray) -> float:
        """亮度评分（1-10）"""
        brightness = img_array.mean() / 255.0
        if 0.3 <= brightness <= 0.7:
            return 8.0 + (0.5 - abs(brightness - 0.5)) * 4
        return max(1.0, 10.0 - abs(brightness - 0.5) * 20)
    
    @staticmethod
    def _score_contrast(img_array: np.ndarray) -> float:
        """对比度评分（1-10）"""
        contrast = img_array.std() / 255.0
        if 0.15 <= contrast <= 0.35:
            return 8.0
        return max(1.0, 10.0 - abs(contrast - 0.25) * 30)
    
    @staticmethod
    def _score_saturation(img_array: np.ndarray) -> float:
        """饱和度评分（1-10）"""
        r, g, b = img_array[:,:,0], img_array[:,:,1], img_array[:,:,2]
        max_c = np.maximum(np.maximum(r, g), b)
        min_c = np.minimum(np.minimum(r, g), b)
        diff = (max_c - min_c) / (max_c + 1e-6)
        saturation = diff.mean()
        
        if 0.2 <= saturation <= 0.6:
            return 8.0
        return max(1.0, 10.0 - abs(saturation - 0.4) * 15)
    
    @staticmethod
    def _score_sharpness(img_array: np.ndarray) -> float:
        """清晰度评分（1-10）"""
        gray = img_array.mean(axis=2)
        laplacian = np.zeros_like(gray)
        laplacian[1:-1, 1:-1] = (
            gray[2:, 1:-1] + gray[:-2, 1:-1] +
            gray[1:-1, 2:] + gray[1:-1, :-2] -
            4 * gray[1:-1, 1:-1]
        )
        sharpness = laplacian.var()
        
        score = min(10.0, sharpness / 500.0)
        return max(1.0, score)
    
    @staticmethod
    def _score_color_harmony(img_array: np.ndarray) -> float:
        """色彩和谐度评分（1-10）"""
        from collections import Counter
        
        quantized = (img_array / 32).astype(int)
        color_keys = [
            f"{r},{g},{b}"
            for r, g, b in zip(
                quantized[:,:,0].flatten(),
                quantized[:,:,1].flatten(),
                quantized[:,:,2].flatten(),
            )
        ]
        
        color_counts = Counter(color_keys)
        total_pixels = len(color_keys)
        top_colors_count = sum(count for _, count in color_counts.most_common(5))
        dominance_ratio = top_colors_count / total_pixels
        
        if dominance_ratio > 0.7:
            return 7.0
        elif dominance_ratio > 0.5:
            return 8.0
        else:
            return 6.0
```

### 3.1 人脸质量检测

```python
import numpy as np
from PIL import Image


class FaceQualityChecker:
    """人脸质量检测器"""
    
    _face_detector = None
    
    @classmethod
    def _load_detector(cls):
        if cls._face_detector is not None:
            return
        
        try:
            from insightface.app import FaceAnalysis
            cls._face_detector = FaceAnalysis(
                name="buffalo_l",
                providers=["CUDAExecutionProvider", "CPUExecutionProvider"],
            )
            cls._face_detector.prepare(ctx_id=0, det_size=(640, 640))
        except ImportError:
            cls._face_detector = "unavailable"
    
    @classmethod
    def check(cls, img: Image.Image) -> dict:
        """检测图片中的人脸质量"""
        cls._load_detector()
        
        if cls._face_detector == "unavailable":
            return {"score": 1.0, "issues": []}
        
        img_array = np.array(img)
        faces = cls._face_detector.get(img_array)
        
        if not faces:
            return {"score": 1.0, "issues": []}
        
        issues = []
        min_score = 1.0
        
        for i, face in enumerate(faces):
            face_score = float(face.det_score)
            min_score = min(min_score, face_score)
            
            if face_score < 0.8:
                issues.append(f"人脸 {i+1} 置信度低: {face_score:.3f}")
            
            if cls._check_face_distortion(face):
                issues.append(f"人脸 {i+1} 可能存在畸变")
            
            bbox = face.bbox
            face_width = bbox[2] - bbox[0]
            face_height = bbox[3] - bbox[1]
            
            if face_width < 30 or face_height < 30:
                issues.append(f"人脸 {i+1} 过小，可能不清晰")
        
        return {
            "score": min_score,
            "num_faces": len(faces),
            "issues": issues,
        }
    
    @classmethod
    def _check_face_distortion(cls, face) -> bool:
        """检测人脸畸变"""
        if hasattr(face, "kps") and face.kps is not None:
            kps = face.kps
            
            left_eye = kps[0]
            right_eye = kps[1]
            nose = kps[2]
            
            eye_distance = np.linalg.norm(right_eye - left_eye)
            
            nose_to_left = np.linalg.norm(nose - left_eye)
            nose_to_right = np.linalg.norm(nose - right_eye)
            
            symmetry_ratio = min(nose_to_left, nose_to_right) / max(nose_to_left, nose_to_right)
            
            if symmetry_ratio < 0.7:
                return True
        
        return False
```

```python
class SimpleFaceChecker:
    """简化版人脸检测（基于 OpenCV）"""
    
    @staticmethod
    def check(img: Image.Image) -> dict:
        """使用 OpenCV Haar 级联检测人脸"""
        import cv2
        
        gray = cv2.cvtColor(np.array(img), cv2.COLOR_RGB2GRAY)
        
        cascade_path = cv2.data.haarcascades + "haarcascade_frontalface_default.xml"
        face_cascade = cv2.CascadeClassifier(cascade_path)
        
        faces = face_cascade.detectMultiScale(
            gray,
            scaleFactor=1.1,
            minNeighbors=5,
            minSize=(30, 30),
        )
        
        if len(faces) == 0:
            return {"score": 1.0, "num_faces": 0, "issues": []}
        
        issues = []
        for i, (x, y, w, h) in enumerate(faces):
            face_region = gray[y:y+h, x:x+w]
            blur_score = cv2.Laplacian(face_region, cv2.CV_64F).var()
            
            if blur_score < 100:
                issues.append(f"人脸 {i+1} 可能模糊 (blur_score={blur_score:.0f})")
        
        overall_score = 1.0 if not issues else 0.6
        
        return {
            "score": overall_score,
            "num_faces": len(faces),
            "issues": issues,
        }
```

```python
import torch
from PIL import Image

            issues.append(f"检测到 NSFW 内容 (置信度: {nsfw_prob:.3f})")
        elif nsfw_prob >= 0.3:
            issues.append(f"可能存在不当内容 (置信度: {nsfw_prob:.3f})")
        
        return {
            "score": nsfw_prob,
            "issues": issues,
        }
```

## 五、CLIP Score 与内容匹配

### 5.1 CLIP 匹配度检测

```python
import torch
from PIL import Image
from transformers import CLIPProcessor, CLIPModel


class CLIPScorer:
    """CLIP 匹配度评分器"""
    
    _model = None
    _processor = None
    
    @classmethod
    def _load_model(cls):
        if cls._model is not None:
            return
        
        cls._processor = CLIPProcessor.from_pretrained("openai/clip-vit-large-patch14")
        cls._model = CLIPModel.from_pretrained("openai/clip-vit-large-patch14")
        cls._model.eval()
    
    @classmethod
    def score(cls, img: Image.Image, prompt: str) -> dict:
        """计算图片与 prompt 的 CLIP Score"""
        cls._load_model()
        
        inputs = cls._processor(
            text=[prompt],
            images=img,
            return_tensors="pt",
            padding=True,
        )
        
        with torch.no_grad():
            outputs = cls._model(**inputs)
            logits_per_image = outputs.logits_per_image
            clip_score = logits_per_image.item() / 100.0
        
        issues = []
        if clip_score < 0.20:
            issues.append(f"图片与 prompt 严重不匹配 (CLIP Score: {clip_score:.3f})")
        elif clip_score < 0.25:
            issues.append(f"图片与 prompt 匹配度偏低 (CLIP Score: {clip_score:.3f})")
        
        return {
            "score": clip_score,
            "issues": issues,
        }
```

```python
    def check(img: Image.Image) -> dict:
        """检测图片的技术质量"""
        img_array = np.array(img).astype(float)
        
        scores = {}
        issues = []
        
        scores["blur"] = TechnicalChecker._detect_blur(img_array)
        scores["noise"] = TechnicalChecker._detect_noise(img_array)
        scores["artifacts"] = TechnicalChecker._detect_artifacts(img_array)
        scores["exposure"] = TechnicalChecker._check_exposure(img_array)
        
        if scores["blur"] < 5.0:
            issues.append(f"图片模糊 (blur_score={scores['blur']:.1f})")
        if scores["noise"] < 5.0:
            issues.append(f"噪点过多 (noise_score={scores['noise']:.1f})")
        if scores["artifacts"] < 5.0:
            issues.append(f"存在伪影 (artifact_score={scores['artifacts']:.1f})")
        if scores["exposure"] < 5.0:
            issues.append(f"曝光异常 (exposure_score={scores['exposure']:.1f})")
        
        overall = sum(scores.values()) / len(scores)
        
        return {
            "score": overall,
            "sub_scores": scores,
            "issues": issues,
        }
    
    @staticmethod
    def _detect_blur(img_array: np.ndarray) -> float:
        """检测模糊程度"""
        gray = img_array.mean(axis=2)
        
        laplacian = np.zeros_like(gray)
        laplacian[1:-1, 1:-1] = (
            gray[2:, 1:-1] + gray[:-2, 1:-1] +
            gray[1:-1, 2:] + gray[1:-1, :-2] -
            4 * gray[1:-1, 1:-1]
        )
        
        variance = laplacian.var()
        score = min(10.0, variance / 300.0)
        return max(1.0, score)
    
    @staticmethod
    def _detect_noise(img_array: np.ndarray) -> float:
        """检测噪点"""
        from scipy.ndimage import uniform_filter
        
        gray = img_array.mean(axis=2)
        local_mean = uniform_filter(gray, size=5)
        local_var = uniform_filter(gray ** 2, size=5) - local_mean ** 2
        
        noise_level = np.sqrt(np.maximum(local_var, 0)).mean()
        
        if noise_level < 5:
            return 9.0
        elif noise_level < 15:
            return 7.0
        elif noise_level < 30:
            return 5.0
        else:
            return 3.0
    
    @staticmethod
    def _detect_artifacts(img_array: np.ndarray) -> float:
        """检测 JPEG 伪影"""
        gray = img_array.mean(axis=2)
        
        block_size = 8
        h, w = gray.shape
        block_diffs = []
        
        for i in range(0, h - block_size, block_size):
            for j in range(0, w - block_size, block_size):
                block = gray[i:i+block_size, j:j+block_size]
                
                if j + block_size < w:
                    right_block = gray[i:i+block_size, j+block_size:j+2*block_size]
                    diff = np.abs(block[:, -1] - right_block[:, 0]).mean()
                    block_diffs.append(diff)
        
        if not block_diffs:
            return 8.0
        
        avg_diff = np.mean(block_diffs)
        
        if avg_diff < 3:
            return 9.0
        elif avg_diff < 8:
            return 7.0
        elif avg_diff < 15:
            return 5.0
        else:
            return 3.0
    
    @staticmethod
    def _check_exposure(img_array: np.ndarray) -> float:
        """检查曝光"""
        brightness = img_array.mean(axis=2)
        
        dark_ratio = (brightness < 30).mean()
        bright_ratio = (brightness > 225).mean()
        
        if dark_ratio > 0.5:
            return max(1.0, 5.0 - dark_ratio * 5)
        elif bright_ratio > 0.3:
            return max(1.0, 5.0 - bright_ratio * 5)
        else:
            return 8.0
```

```python
import json
from pathlib import Path
from datetime import datetime
from enum import Enum


class ReviewDecision(str, Enum):
    APPROVED = "approved"
    REJECTED = "rejected"
    NEEDS_REVISION = "needs_revision"


class HumanReviewManager:
    """人工审核管理"""
    
    def __init__(self, review_dir: str):
        self.review_dir = Path(review_dir)
        self.review_dir.mkdir(parents=True, exist_ok=True)
    
    def create_review_batch(
        self,
        images: list[dict],
        reviewer: str,
        batch_name: str = "",
    ) -> str:
        """创建审核批次"""
        batch_id = f"review_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
        
        review_items = []
        for img_info in images:
            review_items.append({
                "image_path": img_info["path"],
                "auto_report": img_info.get("auto_report", {}),
                "decision": None,
                "reviewer_notes": "",
                "reviewed_at": None,
            })
        
        batch = {
            "batch_id": batch_id,
            "batch_name": batch_name or batch_id,
            "reviewer": reviewer,
            "created_at": datetime.now().isoformat(),
            "status": "pending",
            "items": review_items,
            "progress": {
                "total": len(review_items),
                "reviewed": 0,
                "approved": 0,
                "rejected": 0,
                "needs_revision": 0,
            },
        }
        
        batch_path = self.review_dir / f"{batch_id}.json"
        with open(batch_path, "w", encoding="utf-8") as f:
            json.dump(batch, f, ensure_ascii=False, indent=2)
        
        return batch_id
    
    def submit_decision(
        self,
        batch_id: str,
        image_index: int,
        decision: ReviewDecision,
        notes: str = "",
    ) -> dict:
        """提交审核决定"""
        batch_path = self.review_dir / f"{batch_id}.json"
        
        with open(batch_path, "r", encoding="utf-8") as f:
            batch = json.load(f)
        
        item = batch["items"][image_index]
        item["decision"] = decision.value
        item["reviewer_notes"] = notes
        item["reviewed_at"] = datetime.now().isoformat()
        
        batch["progress"]["reviewed"] += 1
        batch["progress"][decision.value] += 1
        
        if batch["progress"]["reviewed"] >= batch["progress"]["total"]:
            batch["status"] = "completed"
        
        with open(batch_path, "w", encoding="utf-8") as f:
            json.dump(batch, f, ensure_ascii=False, indent=2)
        
        return batch["progress"]
    
    def get_review_summary(self, batch_id: str) -> dict:
        """获取审核摘要"""
        batch_path = self.review_dir / f"{batch_id}.json"
        
        with open(batch_path, "r", encoding="utf-8") as f:
            batch = json.load(f)
        
        return {
            "batch_id": batch_id,
            "batch_name": batch["batch_name"],
            "reviewer": batch["reviewer"],
            "status": batch["status"],
            "progress": batch["progress"],
            "approval_rate": (
                batch["progress"]["approved"] / max(batch["progress"]["reviewed"], 1)
            ),
            "rejection_reasons": self._collect_rejection_reasons(batch),
        }
    
    def _collect_rejection_reasons(self, batch: dict) -> dict:
        """收集拒绝原因"""
        from collections import Counter
        
        reasons = []
        for item in batch["items"]:
            if item["decision"] == "rejected" and item["reviewer_notes"]:
                reasons.append(item["reviewer_notes"])
        
        return dict(Counter(reasons).most_common(10))
```

```python
class QualityGate:
    """质量门控：决定图片是否可以通过"""
    
    def __init__(self, config: dict = None):
        self.config = config or {
            "auto_pass_threshold": 7.5,
            "auto_fail_threshold": 4.0,
            "human_review_range": (4.0, 7.5),
            "nsfw_block_threshold": 0.5,
            "require_human_review_ratio": 0.1,
        }
    
    def evaluate(self, report: QualityReport) -> dict:
        """评估质量报告，决定处理方式"""
        if report.nsfw_score >= self.config["nsfw_block_threshold"]:
            return {
                "action": "block",
                "reason": "NSFW 内容",
                "auto_decided": True,
            }
        
        if report.overall_score >= self.config["auto_pass_threshold"]:
            return {
                "action": "pass",
                "reason": f"自动通过 (综合评分: {report.overall_score:.2f})",
                "auto_decided": True,
            }
        
        if report.overall_score < self.config["auto_fail_threshold"]:
            return {
                "action": "reject",
                "reason": f"自动拒绝 (综合评分: {report.overall_score:.2f})",
                "auto_decided": True,
            }
        
        return {
            "action": "human_review",
            "reason": f"需要人工审核 (综合评分: {report.overall_score:.2f})",
            "auto_decided": False,
        }
```

```python
import json
from pathlib import Path
from datetime import datetime
from collections import Counter


class QualityReporter:
    """质量报告生成器"""
    
    def __init__(self, report_dir: str):
        self.report_dir = Path(report_dir)
        self.report_dir.mkdir(parents=True, exist_ok=True)
    
    def generate_batch_report(
        self,
        batch_name: str,
        reports: list[QualityReport],
        gate_results: list[dict],
    ) -> dict:
        """生成批量质量报告"""
        total = len(reports)
        
        verdict_counts = Counter(r.verdict.value for r in reports)
        action_counts = Counter(g["action"] for g in gate_results)
        
        scores = {
            "aesthetic": [r.aesthetic_score for r in reports],
            "clip": [r.clip_score for r in reports],
            "overall": [r.overall_score for r in reports],
        }
        
        all_issues = []
        for r in reports:
            all_issues.extend(r.issues)
        issue_counts = Counter(all_issues)
        
        report = {
            "batch_name": batch_name,
            "generated_at": datetime.now().isoformat(),
            "summary": {
                "total_images": total,
                "pass_rate": verdict_counts.get("pass", 0) / max(total, 1),
                "fail_rate": verdict_counts.get("fail", 0) / max(total, 1),
                "block_rate": verdict_counts.get("block", 0) / max(total, 1),
            },
            "verdict_breakdown": dict(verdict_counts),
            "action_breakdown": dict(action_counts),
            "score_statistics": {
                dim: {
                    "mean": float(np.mean(vals)),
                    "std": float(np.std(vals)),
                    "min": float(np.min(vals)),
                    "max": float(np.max(vals)),
                    "median": float(np.median(vals)),
                }
                for dim, vals in scores.items()
            },
            "top_issues": dict(issue_counts.most_common(10)),
            "recommendations": self._generate_recommendations(reports, gate_results),
        }
        
        report_path = self.report_dir / f"quality_report_{batch_name}_{datetime.now().strftime('%Y%m%d')}.json"
        with open(report_path, "w", encoding="utf-8") as f:
            json.dump(report, f, ensure_ascii=False, indent=2)
        
        return report
    
    def _generate_recommendations(
        self,
        reports: list[QualityReport],
        gate_results: list[dict],
    ) -> list[str]:
        """生成改进建议"""
        recommendations = []
        
        aesthetic_scores = [r.aesthetic_score for r in reports]
        if np.mean(aesthetic_scores) < 5.5:
            recommendations.append(
                "整体美学评分偏低，建议调整 prompt 中的风格描述词，或更换基础模型"
            )
        
        clip_scores = [r.clip_score for r in reports if r.clip_score > 0]
        if clip_scores and np.mean(clip_scores) < 0.25:
            recommendations.append(
                "CLIP 匹配度偏低，建议优化 prompt 结构，确保关键描述词在前"
            )
        
        fail_count = sum(1 for g in gate_results if g["action"] == "reject")
        if fail_count / len(gate_results) > 0.2:
            recommendations.append(
                "失败率超过 20%，建议检查训练数据质量和模型配置"
            )
        
        nsfw_blocks = sum(1 for r in reports if r.nsfw_score >= 0.5)
        if nsfw_blocks > 0:
            recommendations.append(
                f"有 {nsfw_blocks} 张图片被 NSFW 过滤器拦截，建议检查 prompt 措辞"
            )
        
        all_issues = []
        for r in reports:
            all_issues.extend(r.issues)
        issue_counts = Counter(all_issues)
        
        if issue_counts:
            top_issue = issue_counts.most_common(1)[0]
            recommendations.append(
                f"最常见的问题是「{top_issue[0]}」(出现 {top_issue[1]} 次)，建议针对性优化"
            )
        
        return recommendations
    
    def print_summary(self, report: dict):

        print("\n" + "=" * 60)
        print(f"质量报告: {report['batch_name']}")
        print(f"生成时间: {report['generated_at']}")
        print("=" * 60)
        
        summary = report["summary"]
        print(f"总图片数: {summary['total_images']}")
        print(f"通过率: {summary['pass_rate']:.1%}")
        print(f"失败率: {summary['fail_rate']:.1%}")
        print(f"拦截率: {summary['block_rate']:.1%}")
        
        print("\n分数统计:")
        for dim, stats in report["score_statistics"].items():
            print(f"  {dim}: 均值={stats['mean']:.3f}, "
                  f"标准差={stats['std']:.3f}, "
                  f"范围=[{stats['min']:.3f}, {stats['max']:.3f}]")
        
        if report["top_issues"]:
            print("\n常见问题:")
            for issue, count in list(report["top_issues"].items())[:5]:
                print(f"  - {issue} ({count}次)")
        
        if report["recommendations"]:
            print("\n改进建议:")
            for i, rec in enumerate(report["recommendations"], 1):
                print(f"  {i}. {rec}")
        
        print("=" * 60)
```

### 8.3 完整的使用流程

```python
def run_quality_control_pipeline(
    image_dir: str,
    prompt: str,
    output_dir: str,
    human_review_ratio: float = 0.1,
):
    """运行完整的质量控制流水线"""
    import random
    
    checker = QualityChecker()
    gate = QualityGate()
    reporter = QualityReporter(output_dir)
    review_manager = HumanReviewManager(output_dir)
    
    image_dir_path = Path(image_dir)
    image_files = list(image_dir_path.glob("*.png")) + list(image_dir_path.glob("*.jpg"))
    
    print(f"开始质量检测: {len(image_files)} 张图片")
    
    reports = []
    gate_results = []
    
    for img_path in image_files:
        report = checker.check(str(img_path), prompt)
        gate_result = gate.evaluate(report)
        
        reports.append(report)
        gate_results.append(gate_result)
    
    batch_report = reporter.generate_batch_report(
        batch_name="product_images_batch_001",
        reports=reports,
        gate_results=gate_results,
    )
    
    reporter.print_summary(batch_report)
    
    human_review_needed = [
        {"path": reports[i].image_path, "auto_report": reports[i].to_dict()}
        for i, gr in enumerate(gate_results)
        if gr["action"] == "human_review"
    ]
    
    sampled_for_review = random.sample(
        [r for r, g in zip(reports, gate_results) if g["action"] == "pass"],
        k=min(int(len(reports) * human_review_ratio), len(reports)),
    )
    
    human_review_needed.extend(
        {"path": r.image_path, "auto_report": r.to_dict()}
        for r in sampled_for_review
    )
    
    if human_review_needed:
        review_batch_id = review_manager.create_review_batch(
            images=human_review_needed,
            reviewer="pending",
            batch_name="product_images_review_001",
        )
        print(f"\n人工审核批次已创建: {review_batch_id}")
        print(f"需要人工审核: {len(human_review_needed)} 张图片")
    
    return batch_report
```

## 常见误区

**误区一：只看美学评分就够了。** 美学评分只衡量"好不好看"，不衡量"对不对"。一张美学评分 8.0 的图片可能与 prompt 完全不匹配。必须结合 CLIP Score 来确保内容正确性。

**误区二：NSFW 检测器可以完全信任。** 任何 NSFW 检测器都有误报和漏报。误报会误杀正常图片，漏报会放过不当内容。建议将 NSFW 阈值设置得稍微宽松一些（如 0.7），然后通过人工抽检来捕获漏网之鱼。

**误区三：自动化可以完全替代人工。** 自动化检测能处理 80-90% 的质量判断，但最终的品牌一致性、创意质量、微妙的风格偏差仍然需要人工判断。建议保留 10-20% 的人工抽检比例。

**误区四：质量检测在生产后才做。** 最佳实践是在生成阶段就嵌入质量控制——在 prompt 中加入质量描述词（如 "high quality, detailed"），在 negative prompt 中排除常见问题（如 "blurry, distorted"）。这样可以从源头减少不合格图片的产生。

**误区五：所有图片用同一套阈值。** 不同场景类型的质量标准可能不同。白底产品图要求更高的技术质量（清晰度、无伪影），而创意海报可以容忍更多的艺术性"瑕疵"。应该为不同场景类型配置不同的检测阈值。

## 小结

本课构建了一套完整的 AI 图像质量控制系统：

1. **美学评分**使用 LAION Aesthetics Predictor 或基于图像统计特征的快速评估，量化图片的视觉美感。
2. **人脸质量检测**通过 InsightFace 或 OpenCV 检测人脸畸变、模糊和不对称问题。
3. **NSFW 内容过滤**使用预训练的分类模型拦截不当内容，设置合理的阈值平衡误报和漏报。
4. **CLIP Score** 检测图片与 prompt 的语义匹配度，确保生成内容符合预期。
5. **技术质量检测**通过拉普拉斯方差、噪点分析、伪影检测等手段评估图片的技术指标。
6. **质量门控**将综合评分映射为"通过/拒绝/人工审核"三种处理方式。
7. **人工审核流程**管理审核批次、追踪审核进度、收集拒绝原因。
8. **统计报告**自动生成质量分析报告和改进建议。

质量控制的核心理念是：**分层检测，逐步过滤**。每一层检测解决一类问题，最终输出的图片经过了多层质量验证。

- 15% 的人脸畸变率过高，需要从 prompt 和模型两个层面解决
- ADetailer 是解决人脸问题的有效工具

### 练习三

**思路**：使用 CLIP 计算图片与品牌风格描述的相似度

**答案**：

```python
class BrandStyleChecker:
    """品牌风格一致性检测"""
    
    _model = None
    _processor = None
    
    def __init__(self, brand_style_descriptions: list[str], threshold: float = 0.25):
        self.brand_descriptions = brand_style_descriptions
        self.threshold = threshold
        self._load_model()
    
    @classmethod
    def _load_model(cls):
        if cls._model is not None:
            return
        cls._processor = CLIPProcessor.from_pretrained("openai/clip-vit-large-patch14")
        cls._model = CLIPModel.from_pretrained("openai/clip-vit-large-patch14")
        cls._model.eval()
    
    def check(self, img: Image.Image) -> dict:
        """检测图片与品牌风格的一致性"""
        inputs = self._processor(
            text=self.brand_descriptions,
            images=img,
            return_tensors="pt",
            padding=True,
        )
        
        with torch.no_grad():
            outputs = self._model(**inputs)
            logits_per_image = outputs.logits_per_image
            scores = logits_per_image[0].numpy() / 100.0
        
        best_score = float(scores.max())
        best_match = self.brand_descriptions[scores.argmax()]
        
        issues = []
        if best_score < self.threshold:
            issues.append(
                f"品牌风格匹配度低 ({best_score:.3f})，"
                f"最接近的描述: {best_match}"
            )
        
        return {
            "score": best_score,
            "best_match": best_match,
            "all_scores": {
                desc: float(score)
                for desc, score in zip(self.brand_descriptions, scores)
            },
            "issues": issues,
        }
```

**要点**：
- 品牌风格描述应该包含多个维度（色彩、构图、质感）
- 取所有描述中的最高分作为匹配度
- 阈值需要根据实际品牌风格的特异性来调整
