## 2. N+1 查询问题

### 2.1 什么是 N+1 查询

```
N+1 查询问题：

查询用户列表 + 每个用户的订单

❌ N+1 方式（1 + 100 = 101 次查询）：

  查询 1: SELECT * FROM users LIMIT 100
  查询 2: SELECT * FROM orders WHERE userId = 1
  查询 3: SELECT * FROM orders WHERE userId = 2
  ...
  查询 101: SELECT * FROM orders WHERE userId = 100

✅ 优化方式（2 次查询）：

  查询 1: SELECT * FROM users LIMIT 100
  查询 2: SELECT * FROM orders WHERE userId IN (1, 2, ..., 100)

性能对比：
  N+1: ~500ms（100 次网络往返）
  优化后: ~15ms（2 次网络往返）
```

### 2.3 N+1 修复示例

**Before（N+1 查询）：**

```typescript
// ❌ 循环中逐个查询
async function getUserOrders(userIds: string[]) {
  const results = [];
  for (const userId of userIds) {
    const user = await db.user.findUnique({ where: { id: userId } });
    const orders = await db.order.findMany({ where: { userId } });
    results.push({ ...user, orders });
  }
  return results;
}
```

**After（批量查询）：**

```typescript
// ✅ 批量查询 + 内存关联
async function getUserOrders(userIds: string[]) {
  // 2 次查询代替 N+1 次
  const [users, orders] = await Promise.all([
    db.user.findMany({ where: { id: { in: userIds } } }),
    db.order.findMany({ where: { userId: { in: userIds } } }),
  ]);

  // 内存中关联
  const ordersByUser = new Map<string, typeof orders>();
  for (const order of orders) {
    const list = ordersByUser.get(order.userId) || [];
    list.push(order);
    ordersByUser.set(order.userId, list);
  }

  return users.map((user) => ({
    ...user,
    orders: ordersByUser.get(user.id) || [],
  }));
}
```

### 3.2 缓存策略对比

| 策略 | 适用场景 | 一致性 | 复杂度 |
|------|---------|--------|--------|
| Cache-Aside | 读多写少 | 最终一致 | 低 |
| Write-Through | 写频繁 | 强一致 | 中 |
| Write-Behind | 写密集 | 最终一致 | 高 |
| Read-Through | 读密集 | 强一致 | 中 |

### 3.3 Cache-Aside 实现

```typescript
// 通用缓存包装器
class CacheService {
  private cache = new Map<string, { data: unknown; expiry: number }>();

  async get<T>(key: string): Promise<T | null> {
    const entry = this.cache.get(key);
    if (!entry) return null;
    if (Date.now() > entry.expiry) {
      this.cache.delete(key);
      return null;
    }
    return entry.data as T;
  }

  set(key: string, data: unknown, ttlSeconds: number): void {
    this.cache.set(key, {
      data,
      expiry: Date.now() + ttlSeconds * 1000,
    });
  }

  invalidate(key: string): void {
    this.cache.delete(key);
  }

  invalidatePattern(pattern: string): void {
    const regex = new RegExp(pattern);
    for (const key of this.cache.keys()) {
      if (regex.test(key)) {
        this.cache.delete(key);
      }
    }
  }
}

// 使用示例
const cache = new CacheService();

async function getUserById(id: string): Promise<User> {
  const cacheKey = `user:${id}`;
  
  // 1. 先查缓存
  const cached = await cache.get<User>(cacheKey);
  if (cached) return cached;

  // 2. 缓存未命中，查数据库
  const user = await db.user.findUnique({ where: { id } });
  if (!user) throw new NotFoundError('User not found');

  // 3. 写入缓存（TTL 5 分钟）
  cache.set(cacheKey, user, 300);

  return user;
}

async function updateUser(id: string, data: Partial<User>): Promise<User> {
  const user = await db.user.update({ where: { id }, data });
  
  // 更新后清除缓存
  cache.invalidate(`user:${id}`);
  
  return user;
}
```

```typescript
// 原生 Intersection Observer
function setupLazyLoading() {
  const observer = new IntersectionObserver(
    (entries) => {
      entries.forEach((entry) => {
        if (entry.isIntersecting) {
          const img = entry.target as HTMLImageElement;
          img.src = img.dataset.src!;
          observer.unobserve(img);
        }
      });
    },
    { rootMargin: '200px' }  // 提前 200px 开始加载
  );

  document.querySelectorAll('img[data-src]').forEach((img) => {
    observer.observe(img);
  });
}

// React Hook
function useLazyLoad(ref: React.RefObject<HTMLElement>) {
  const [isVisible, setIsVisible] = useState(false);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          observer.disconnect();
        }
      },
      { rootMargin: '200px' }
    );

    if (ref.current) {
      observer.observe(ref.current);
    }

    return () => observer.disconnect();
  }, [ref]);

  return isVisible;
}
```

### 4.4 数据懒加载（无限滚动）

```typescript
function useInfiniteScroll<T>(
  fetchMore: (page: number) => Promise<T[]>
) {
  const [data, setData] = useState<T[]>([]);
  const [page, setPage] = useState(1);
  const [loading, setLoading] = useState(false);
  const [hasMore, setHasMore] = useState(true);

  const loadMore = useCallback(async () => {
    if (loading || !hasMore) return;
    setLoading(true);
    const newItems = await fetchMore(page);
    setData((prev) => [...prev, ...newItems]);
    setHasMore(newItems.length > 0);
    setPage((p) => p + 1);
    setLoading(false);
  }, [page, loading, hasMore, fetchMore]);

  // 监听滚动到底部
  useEffect(() => {
    const handleScroll = () => {
      if (
        window.innerHeight + window.scrollY >=
        document.body.offsetHeight - 500
      ) {
        loadMore();
      }
    };

    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, [loadMore]);

  return { data, loading, hasMore, loadMore };
}

// 使用
function UserList() {
  const { data: users, loading } = useInfiniteScroll(async (page) => {
    const res = await fetch(`/api/users?page=${page}`);
    return res.json();
  });

  return (
    <div>
      {users.map((user) => (
        <UserCard key={user.id} user={user} />
      ))}
      {loading && <Spinner />}
    </div>
  );
}
```

## 5. 其他常见优化

### 5.1 防抖与节流

```typescript
// 防抖：连续触发只执行最后一次
function debounce<T extends (...args: unknown[]) => unknown>(
  fn: T,
  delay: number
): (...args: Parameters<T>) => void {
  let timer: NodeJS.Timeout;
  return (...args: Parameters<T>) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
}

// 节流：固定时间间隔执行一次
function throttle<T extends (...args: unknown[]) => unknown>(
  fn: T,
  limit: number
): (...args: Parameters<T>) => void {
  let inThrottle = false;
  return (...args: Parameters<T>) => {
    if (!inThrottle) {
      fn(...args);
      inThrottle = true;
      setTimeout(() => (inThrottle = false), limit);
    }
  };
}

// 使用场景
const searchHandler = debounce(async (query: string) => {
  const results = await fetch(`/api/search?q=${query}`);
  renderResults(results);
}, 300);

const scrollHandler = throttle(() => {
  updateScrollIndicator();
}, 100);
```

### 5.2 批量处理

```typescript
// ❌ 逐个处理
for (const item of items) {
  await processItem(item);  // 100 个 item = 100 次等待
}

// ✅ 批量并发（控制并发数）
async function processBatch<T>(
  items: T[],
  processor: (item: T) => Promise<void>,
  concurrency = 10
) {
  const chunks = [];
  for (let i = 0; i < items.length; i += concurrency) {
    chunks.push(items.slice(i, i + concurrency));
  }

  for (const chunk of chunks) {
    await Promise.all(chunk.map(processor));
  }
}

await processBatch(items, processItem, 10); // 每批 10 个并发
```

### 6.1 基准测试代码

```typescript
// 性能测试工具
async function benchmark(
  name: string,
  fn: () => Promise<void>,
  iterations = 100
) {
  const times: number[] = [];
  
  for (let i = 0; i < iterations; i++) {
    const start = performance.now();
    await fn();
    times.push(performance.now() - start);
  }

  const avg = times.reduce((a, b) => a + b) / times.length;
  const min = Math.min(...times);
  const max = Math.max(...times);
  const p95 = times.sort((a, b) => a - b)[Math.floor(times.length * 0.95)];

  console.log(`
    ${name}:
      平均: ${avg.toFixed(2)}ms
      最小: ${min.toFixed(2)}ms
      最大: ${max.toFixed(2)}ms
      P95:  ${p95.toFixed(2)}ms
  `);
}

// 使用
await benchmark('N+1 查询', async () => {
  await getUserOrdersNPlusOne(userIds);
});

await benchmark('批量查询', async () => {
  await getUserOrdersBatch(userIds);
});
```

## 常见误区

### 错误 1：过早优化

```bash
# ❌ 还没度量就开始优化
# "我觉得这里应该用缓存"
# 可能瓶颈根本不在这里

# ✅ 先度量再优化
# 用 benchmark/Profiler 找到真正的瓶颈
# 80% 的性能问题通常在 20% 的代码中
```

### 错误 2：缓存一致性问题

```typescript
// ❌ 更新数据库后忘记清除缓存
async function updateUser(id: string, data: Partial<User>) {
  await db.user.update({ where: { id }, data });
  // 忘了清除缓存！读到的还是旧数据
}

// ✅ 更新后清除缓存
async function updateUser(id: string, data: Partial<User>) {
  await db.user.update({ where: { id }, data });
  await cache.invalidate(`user:${id}`);
}
```

### 错误 3：缓存穿透

```typescript
// ❌ 查询不存在的数据，每次都穿透到数据库
await getCachedUser('nonexistent-id'); // 缓存未命中 → 查 DB → 不存在 → 不缓存
await getCachedUser('nonexistent-id'); // 又穿透！

// ✅ 缓存空值（短 TTL）
async function getCachedUser(id: string): Promise<User | null> {
  const cached = await cache.get<User | null>(`user:${id}`);
  if (cached !== undefined) return cached; // 命中缓存（包括 null）

  const user = await db.user.findUnique({ where: { id } });
  cache.set(`user:${id}`, user ?? null, user ? 300 : 60); // 空值缓存 60s
  return user;
}
```

## 工程建议

1. **先度量再优化，用数据说话**：在优化前后都要跑 benchmark，记录 P50/P95/P99 响应时间和吞吐量，避免"感觉快了"的主观判断，也防止过早优化浪费精力。
2. **缓存策略要设计失效机制**：缓存不是加了就行，必须规划好 TTL、主动失效、缓存穿透/击穿/雪崩的防护策略，否则缓存一致性问题比没有缓存更糟糕。
3. **N+1 问题作为 Code Review 的必查项**：在团队 Review Checklist 中加入"循环内是否有数据库查询或 API 调用"，让 AI 在每次 PR Review 时自动检查，从源头杜绝 N+1。
4. **性能优化要关注用户体验指标**：不要只看后端响应时间，还要关注 LCP（最大内容绘制）、FID（首次输入延迟）、CLS（累积布局偏移）等 Core Web Vitals，这些才是用户真正感知到的性能。

## 参考答案

### 练习 1：修复 N+1 查询

**思路**：原代码在 `Promise.all` 的 `map` 回调中对每个 post 单独查询 author，这是典型的 N+1 问题。应先批量查出所有 author，再在内存中关联。

**答案**：

```typescript
// ❌ 原始代码：N+1 查询（1 次查 posts + N 次查 author）
async function getPostsWithAuthors(postIds: string[]) {
  const posts = await db.post.findMany({ where: { id: { in: postIds } } });
  return Promise.all(
    posts.map(async (post) => {
      const author = await db.user.findUnique({ where: { id: post.authorId } });
      return { ...post, author };
    })
  );
}

// ✅ 修复方案 1：批量查询 + 内存关联（2 次查询）
async function getPostsWithAuthors(postIds: string[]) {
  const posts = await db.post.findMany({ where: { id: { in: postIds } } });

  const authorIds = [...new Set(posts.map((p) => p.authorId))];
  const authors = await db.user.findMany({
    where: { id: { in: authorIds } },
  });

  const authorMap = new Map(authors.map((a) => [a.id, a]));

  return posts.map((post) => ({
    ...post,
    author: authorMap.get(post.authorId) ?? null,
  }));
}

// ✅ 修复方案 2：使用 ORM 的 include/JOIN（1 次查询，推荐）
async function getPostsWithAuthors(postIds: string[]) {
  return db.post.findMany({
    where: { id: { in: postIds } },
    include: {
      author: true, // 自动 JOIN users 表
    },
  });
}
```

**要点**：
- N+1 问题的本质是"循环内发起数据库查询"，修复方式是批量查询 + 内存关联，或者使用 ORM 的 `include`/`JOIN` 功能
- 批量查询时用 `Set` 去重 authorId，避免查询重复的用户
- ORM 的 `include` 底层生成 JOIN 查询，性能最优且代码最简洁，优先使用

### 练习 2：实现缓存层

**思路**：为 `getProductCatalog` 添加 Cache-Aside 策略：先查缓存、未命中则查数据库并写入缓存、产品更新时清除对应分类缓存、空结果也缓存以防止缓存穿透。

**答案**：

```typescript
import { CacheService } from './cacheService'; // 假设已有缓存服务

const cache = new CacheService();
const CACHE_TTL = 300; // 5 分钟
const NULL_CACHE_TTL = 60; // 空结果缓存 60 秒

// ✅ 带缓存的产品目录查询
async function getProductCatalog(category: string): Promise<Product[]> {
  const cacheKey = `products:category:${category}`;

  // 1. 先查缓存
  const cached = await cache.get<Product[]>(cacheKey);
  if (cached !== undefined) {
    return cached; // 命中缓存（包括空数组）
  }

  // 2. 缓存未命中，查数据库
  const products = await db.product.findMany({
    where: { category, isActive: true },
    orderBy: { createdAt: 'desc' },
  });

  // 3. 写入缓存（空结果也缓存，防止缓存穿透）
  if (products.length > 0) {
    cache.set(cacheKey, products, CACHE_TTL);
  } else {
    cache.set(cacheKey, [], NULL_CACHE_TTL); // 空结果短 TTL
  }

  return products;
}

// ✅ 产品更新时清除对应分类缓存
async function updateProduct(productId: string, data: Partial<Product>) {
  const existing = await db.product.findUnique({ where: { id: productId } });
  if (!existing) {
    throw new NotFoundError(`Product ${productId} not found`);
  }

  const updated = await db.product.update({
    where: { id: productId },
    data,
  });

  // 清除旧分类和新分类的缓存
  cache.invalidate(`products:category:${existing.category}`);
  if (data.category && data.category !== existing.category) {
    cache.invalidate(`products:category:${data.category}`);
  }

  return updated;
}

// ✅ 批量清除分类缓存（用于产品批量操作）
async function invalidateCategoryCache(...categories: string[]) {
  for (const category of categories) {
    cache.invalidate(`products:category:${category}`);
  }
}
```

**要点**：
- Cache-Aside 模式的核心流程是"查缓存 → 未命中查 DB → 写缓存"，缓存失效由写操作主动触发
- 缓存穿透防护：对查询结果为空的情况也写入缓存，但使用较短的 TTL（如 60 秒），避免长时间缓存"不存在"
- 更新数据时必须清除相关缓存，且要考虑分类变更的情况（旧分类和新分类都要清除）

### 练习 3：实现防抖搜索

**思路**：组合防抖函数、loading 状态管理和 AbortController 来实现一个完整的搜索组件。关键点是每次新请求前取消上一次未完成的请求，避免竞态条件。

**答案**：

```typescript
import { useState, useEffect, useRef, useCallback } from 'react';

interface SearchResult {
  id: string;
  title: string;
  description: string;
}

function useDebouncedSearch(delay = 300) {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<SearchResult[]>([]);
  const [loading, setLoading] = useState(false);
  const abortControllerRef = useRef<AbortController | null>(null);
  const timerRef = useRef<NodeJS.Timeout | null>(null);

  const search = useCallback(
    async (searchQuery: string) => {
      // 取消上一次未完成的请求
      if (abortControllerRef.current) {
        abortControllerRef.current.abort();
      }

      if (!searchQuery.trim()) {
        setResults([]);
        setLoading(false);
        return;
      }

      // 创建新的 AbortController
      const controller = new AbortController();
      abortControllerRef.current = controller;

      setLoading(true);
      try {
        const response = await fetch(
          `/api/search?q=${encodeURIComponent(searchQuery)}`,
          { signal: controller.signal }
        );

        if (!controller.signal.aborted) {
          const data = await response.json();
          setResults(data);
        }
      } catch (error) {
        if (error instanceof DOMException && error.name === 'AbortError') {
          return; // 请求被取消，忽略
        }
        console.error('Search failed:', error);
        if (!controller.signal.aborted) {
          setResults([]);
        }
      } finally {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      }
    },
    []
  );

  // 防抖效果：query 变化后延迟执行搜索
  useEffect(() => {
    if (timerRef.current) {
      clearTimeout(timerRef.current);
    }

    timerRef.current = setTimeout(() => {
      search(query);
    }, delay);

    return () => {
      if (timerRef.current) {
        clearTimeout(timerRef.current);
      }
    };
  }, [query, delay, search]);

  // 组件卸载时取消请求
  useEffect(() => {
    return () => {
      if (abortControllerRef.current) {
        abortControllerRef.current.abort();
      }
    };
  }, []);

  return { query, setQuery, results, loading };
}

// 使用示例
function SearchComponent() {
  const { query, setQuery, results, loading } = useDebouncedSearch(300);

  return (
    <div>
      <input
        type="text"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        placeholder="搜索..."
      />
      {loading && <div className="spinner">搜索中...</div>}
      <ul>
        {results.map((item) => (
          <li key={item.id}>
            <strong>{item.title}</strong>
            <p>{item.description}</p>
          </li>
        ))}
      </ul>
      {!loading && query && results.length === 0 && <div>无搜索结果</div>}
    </div>
  );
}
```

**要点**：
- 防抖的核心是用 `setTimeout` 延迟执行，每次输入变化时清除上一个定时器，只有停止输入 300ms 后才真正发起请求
- `AbortController` 用于取消浏览器的 `fetch` 请求，防止快速输入时旧请求的响应覆盖新请求的结果（竞态条件）
- 组件卸载时必须取消未完成的请求和定时器，避免内存泄漏和对已卸载组件的 setState 警告
