```typescript
// validator.ts
export function validateTitle(title: unknown): string {
  if (typeof title !== 'string') throw new AppError('Title must be a string', 400);
  const trimmed = title.trim();
  if (trimmed.length === 0) throw new AppError('Title cannot be empty', 400);
  if (trimmed.length > 200) throw new AppError('Title cannot exceed 200 characters', 400);
  return trimmed;
}

// taskService.ts
export function createTask(dto: CreateTaskDTO) {
  const title = validateTitle(dto.title);
  const priority = dto.priority ? validatePriority(dto.priority) : 'medium';
  if (dto.dueDate && new Date(dto.dueDate) < new Date()) {
    throw new AppError('Due date cannot be in the past', 400);
  }
  return taskRepo.create({ ...dto, title, priority });
}
```

## 测试策略：从底层往上

```
Layer 1: 工具函数 (validator)     ← 最先测，纯函数无依赖
Layer 2: Repository               ← 内存存储，用 reset() 清理
Layer 3: Service                  ← Mock Repository
Layer 4: Controller               ← 用 supertest
Layer 5: Middleware               ← 用真实 JWT
```

## 实战：逐层生成测试

### Layer 1：工具函数

```typescript
describe('validateTitle', () => {
  it('接受有效的标题', () => {
    expect(validateTitle('买菜')).toBe('买菜');
  });
  it('去除首尾空格', () => {
    expect(validateTitle('  有空格  ')).toBe('有空格');
  });
  it('拒绝非字符串输入', () => {
    expect(() => validateTitle(123)).toThrow('Title must be a string');
  });
  it('拒绝空字符串', () => {
    expect(() => validateTitle('')).toThrow('Title cannot be empty');
  });
  it('拒绝超过 200 字符', () => {
    expect(() => validateTitle('a'.repeat(201))).toThrow('200 characters');
  });
});
```

### Layer 2：Repository

```typescript
describe('taskRepository', () => {
  beforeEach(() => { taskRepo.reset(); });

  it('创建任务并返回完整对象', () => {
    const task = taskRepo.create({ title: '测试任务' });
    expect(task.id).toBeDefined();
    expect(task.status).toBe('todo');
  });
  it('按优先级过滤', () => {
    taskRepo.create({ title: 'A', priority: 'high' });
    taskRepo.create({ title: 'B', priority: 'low' });
    expect(taskRepo.findAll({ priority: 'high' }).tasks).toHaveLength(1);
  });
  it('对不存在的 ID 返回 undefined', () => {
    expect(taskRepo.findById('nonexistent')).toBeUndefined();
  });
});
```

### Layer 3：Service（含 Mock）

```typescript
vi.mock('../../repositories/taskRepository');

describe('taskService', () => {
  beforeEach(() => { vi.clearAllMocks(); });

  it('任务不存在时抛出 404', () => {
    vi.mocked(taskRepo.findById).mockReturnValue(undefined);
    expect(() => taskService.getTaskById('x')).toThrow('Task not found');
  });
  it('拒绝无效的标题', () => {
    expect(() => taskService.createTask({ title: '' })).toThrow('Title cannot be empty');
  });
  it('拒绝过去的截止日期', () => {
    expect(() => taskService.createTask({
      title: '任务', dueDate: '2020-01-01'
    })).toThrow('Due date cannot be in the past');
  });
});
```

### Layer 4：Controller 集成测试

```typescript
const app = express();
app.use(express.json());
app.use('/api/v1/tasks', taskRouter);
app.use(errorHandler);

describe('TaskController', () => {
  beforeEach(() => { taskRepo.reset(); });

  it('创建任务返回 201', async () => {
    const res = await request(app).post('/api/v1/tasks').send({ title: '新任务' });
    expect(res.status).toBe(201);
    expect(res.body.data.title).toBe('新任务');
  });
  it('空标题返回 400', async () => {
    const res = await request(app).post('/api/v1/tasks').send({ title: '' });
    expect(res.status).toBe(400);
  });
});
```
