## Phase 3：核心实现（60 分钟）

短码生成器：

```typescript
import { customAlphabet } from 'nanoid';
const generate = customAlphabet('0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz', 7);

export function generateCode(customCode?: string): string {
  if (customCode) {
    if (customCode.length < 3 || customCode.length > 20) throw new Error('Code must be 3-20 chars');
    if (!/^[a-zA-Z0-9_-]+$/.test(customCode)) throw new Error('Invalid characters');
    return customCode;
  }
  return generate();
}
```

短链接 Service：

```typescript
export async function createLink(input: CreateLinkInput) {
  validateUrl(input.url);
  const code = generateCode(input.customCode);
  const existing = await db.select().from(links).where(eq(links.code, code)).limit(1);
  if (existing.length > 0) throw new ConflictError(`Code "${code}" already exists`);
  const [link] = await db.insert(links).values({
    code, originalUrl: input.url, expiresAt: input.expiresAt ? new Date(input.expiresAt) : null,
  }).returning();
  return link;
}

export async function getLinkByCode(code: string) {
  const [link] = await db.select().from(links).where(
    and(eq(links.code, code), isNull(links.deletedAt))
  ).limit(1);
  if (!link) throw new NotFoundError('Link not found');
  if (link.expiresAt && link.expiresAt < new Date()) throw new NotFoundError('Link expired');
  return link;
}
```

重定向路由：

```typescript
app.get('/:code', async (req, reply) => {
  const { code } = req.params as { code: string };
  if (code === 'api') return;
  const link = await linkService.getLinkByCode(code);
  clickService.recordClick(link.id, { ip: req.ip, userAgent: req.headers['user-agent'] }).catch(console.error);
  return reply.redirect(302, link.originalUrl);
});
```

## Phase 4：测试（30 分钟）

```typescript
describe('codeGenerator', () => {
  it('生成 7 位随机码', () => {
    const code = generateCode();
    expect(code).toHaveLength(7);
    expect(code).toMatch(/^[a-zA-Z0-9]+$/);
  });
  it('使用自定义短码', () => { expect(generateCode('my-link')).toBe('my-link'); });
  it('拒绝保留字', () => { expect(() => generateCode('api')).toThrow('reserved'); });
  it('生成不重复的码', () => {
    const codes = new Set(Array.from({ length: 1000 }, () => generateCode()));
    expect(codes.size).toBe(1000);
  });
});
```

### 练习二

```typescript
import QRCode from 'qrcode';
export async function generateQRCode(code: string) {
  const link = await getLinkByCode(code);
  const shortUrl = `${process.env.BASE_URL}/${link.code}`;
  const buffer = await QRCode.toBuffer(shortUrl, { type: 'png', width: 300 });
  return { data: buffer, contentType: 'image/png' };
}
```
