# 05 - 效果度量与 ROI 分析

> 用数据说话，量化 AI 编程带来的效率提升和投资回报

## 场景引入

季度预算会上，CTO 问你："团队用了半年 AI 编程工具，每人每月 20 美元的订阅费花得值吗？"你凭直觉回答"感觉效率提高了不少"，但拿不出任何数据。CTO 继续追问："提高了多少？Bug 率有没有变化？哪些 Prompt 真正好用？"你哑口无言。会后，预算被砍了一半。你意识到：没有数据支撑的 AI 投入，在管理层眼中就是一笔"说不清的开销"。

## 学习目标

- 理解 AI 编程效果度量的重要性
- 掌握多维度指标体系的设计方法
- 学会计算 AI 编程的投资回报率（ROI）
- 建立持续监控和优化机制

## 2. 指标体系设计

### 2.1 四维指标框架

```
                    效率指标
                      │
                      │
    质量指标 ──────────┼────────── 经济指标
                      │
                      │
                    能力指标
```

### 2.2 详细指标定义

| 维度 | 指标 | 定义 | 计算方式 |
|------|------|------|---------|
| **效率** | 代码产出速度 | 单位时间代码行数 | 行/小时 |
| | 任务完成时间 | 从开始到完成的时长 | 小时 |
| | Prompt 响应时间 | AI 返回结果的速度 | 秒 |
| | 代码补全接受率 | AI 建议被采纳的比率 | 接受数/建议数 |
| **质量** | Bug 引入率 | AI 生成代码的 bug 比例 | bugs/千行 |
| | 代码审查通过率 | PR 一次通过的比率 | 通过数/总数 |
| | 测试覆盖率 | 代码测试覆盖程度 | 百分比 |
| | 技术债务变化 | 代码质量趋势 | SonarQube 评分 |
| **经济** | 人力成本节约 | 减少的开发工时 | 元 |
| | 工具成本 | AI 工具订阅费用 | 元/月 |
| | ROI | 投资回报率 | (收益-成本)/成本 |
| **能力** | Prompt 质量评分 | Prompt 的平均效果 | 1-5 分 |
| | 知识库使用率 | 知识资产被引用次数 | 次/周 |
| | 学习曲线 | 新人上手速度 | 天 |

### 2.3 指标收集实现

```typescript
// src/metrics/collector.ts
interface MetricEntry {
  id: string;
  timestamp: string;
  userId: string;
  metricType: 'efficiency' | 'quality' | 'economic' | 'capability';
  metricName: string;
  value: number;
  unit: string;
  context: Record<string, unknown>;
}

class MetricsCollector {
  private metrics: MetricEntry[] = [];
  private flushInterval: NodeJS.Timeout | null = null;

  constructor(
    private endpoint: string,
    private batchSize: number = 50,
    private flushIntervalMs: number = 60000,
  ) {
    this.startAutoFlush();
  }

  record(
    userId: string,
    metricType: MetricEntry['metricType'],
    metricName: string,
    value: number,
    unit: string,
    context: Record<string, unknown> = {},
  ): void {
    this.metrics.push({
      id: crypto.randomUUID(),
      timestamp: new Date().toISOString(),
      userId,
      metricType,
      metricName,
      value,
      unit,
      context,
    });

    if (this.metrics.length >= this.batchSize) {
      this.flush();
    }
  }

  // 便捷方法
  recordCodeGeneration(userId: string, linesGenerated: number, accepted: boolean): void {
    this.record(userId, 'efficiency', 'code_generation', linesGenerated, 'lines', { accepted });
  }

  recordBugFound(userId: string, source: 'ai' | 'human', severity: string): void {
    this.record(userId, 'quality', 'bug_found', 1, 'count', { source, severity });
  }

  recordTaskCompletion(userId: string, taskId: string, durationMinutes: number): void {
    this.record(userId, 'efficiency', 'task_completion', durationMinutes, 'minutes', { taskId });
  }

  private async flush(): Promise<void> {
    if (this.metrics.length === 0) return;

    const batch = this.metrics.splice(0);
    try {
      await fetch(this.endpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ metrics: batch }),
      });
    } catch (error) {
      this.metrics.unshift(...batch);
      console.error('Metrics flush failed:', error);
    }
  }

  private startAutoFlush(): void {
    this.flushInterval = setInterval(() => this.flush(), this.flushIntervalMs);
  }

  destroy(): void {
    if (this.flushInterval) clearInterval(this.flushInterval);
    this.flush();
  }
}
```

## 3. ROI 计算模型

### 3.1 ROI 计算公式

```
AI编程ROI = (效率提升收益 + 质量提升收益 - 工具成本 - 学习成本) / 总成本 × 100%

其中：
  效率提升收益 = 节省工时 × 平均时薪
  质量提升收益 = 减少的 bug 修复成本 + 减少的线上事故成本
  工具成本 = 订阅费 + 基础设施费用
  学习成本 = 培训时间 × 平均时薪
```

### 3.2 ROI 计算器

```python
# scripts/roi_calculator.py
from dataclasses import dataclass
from typing import Optional

@dataclass
class TeamConfig:
    team_size: int
    avg_hourly_rate: float  # 元/小时
    working_hours_per_month: float = 160.0

@dataclass
class AIToolConfig:
    monthly_subscription: float  # 元/人/月
    setup_cost: float = 0.0  # 一次性设置成本
    training_hours: float = 8.0  # 培训时长

@dataclass
class EfficiencyGains:
    code_generation_speedup: float = 1.5  # 代码生成提速倍数
    code_review_speedup: float = 2.0  # 代码审查提速倍数
    debugging_speedup: float = 1.3  # 调试提速倍数
    documentation_speedup: float = 3.0  # 文档编写提速倍数

@dataclass
class QualityGains:
    bug_reduction_rate: float = 0.2  # bug 减少比率
    avg_bug_fix_hours: float = 4.0  # 平均 bug 修复时长
    bugs_per_month: float = 20.0  # 每月平均 bug 数
    incident_reduction_rate: float = 0.15  # 线上事故减少比率
    avg_incident_cost: float = 5000.0  # 平均事故成本

class ROICalculator:
    def __init__(
        self,
        team: TeamConfig,
        ai_tool: AIToolConfig,
        efficiency: EfficiencyGains,
        quality: QualityGains,
    ):
        self.team = team
        self.ai_tool = ai_tool
        self.efficiency = efficiency
        self.quality = quality

    def calculate_monthly_roi(self) -> dict:
        # 成本计算
        tool_cost = self.team.team_size * self.ai_tool.monthly_subscription
        training_cost = (
            self.team.team_size
            * self.ai_tool.training_hours
            * self.team.avg_hourly_rate
        ) / 12  # 分摊到12个月
        monthly_cost = tool_cost + training_cost

        # 效率提升收益
        # 假设 60% 时间在编码，20% 在 review，10% 在调试，10% 在写文档
        coding_hours = self.team.working_hours_per_month * 0.6
        review_hours = self.team.working_hours_per_month * 0.2
        debug_hours = self.team.working_hours_per_month * 0.1
        doc_hours = self.team.working_hours_per_month * 0.1

        efficiency_hours_saved = (
            coding_hours * (1 - 1/self.efficiency.code_generation_speedup)
            + review_hours * (1 - 1/self.efficiency.code_review_speedup)
            + debug_hours * (1 - 1/self.efficiency.debugging_speedup)
            + doc_hours * (1 - 1/self.efficiency.documentation_speedup)
        )
        efficiency_savings = (
            efficiency_hours_saved
            * self.team.avg_hourly_rate
            * self.team.team_size
        )

        # 质量提升收益
        bug_savings = (
            self.quality.bugs_per_month
            * self.quality.bug_reduction_rate
            * self.quality.avg_bug_fix_hours
            * self.team.avg_hourly_rate
        )
        incident_savings = (
            self.quality.incident_reduction_rate
            * self.quality.avg_incident_cost
        )
        quality_savings = bug_savings + incident_savings

        # 总收益
        total_benefit = efficiency_savings + quality_savings

        # ROI
        roi = (total_benefit - monthly_cost) / monthly_cost * 100

        return {
            'monthly_cost': round(monthly_cost, 2),
            'tool_cost': round(tool_cost, 2),
            'training_cost_monthly': round(training_cost, 2),
            'efficiency_savings': round(efficiency_savings, 2),
            'quality_savings': round(quality_savings, 2),
            'total_benefit': round(total_benefit, 2),
            'net_benefit': round(total_benefit - monthly_cost, 2),
            'roi_percentage': round(roi, 1),
            'payback_period_months': round(
                self.ai_tool.setup_cost / max(total_benefit - monthly_cost, 1), 1
            ) if self.ai_tool.setup_cost > 0 else 0,
        }

    def generate_report(self) -> str:
        results = self.calculate_monthly_roi()

        return f"""
# AI 编程投资回报分析报告

## 团队配置
- 团队规模: {self.team.team_size} 人
- 平均时薪: ¥{self.team.avg_hourly_rate}
- AI 工具月费: ¥{self.ai_tool.monthly_subscription}/人

## 月度成本分析
| 项目 | 金额 |
|------|------|
| 工具订阅费 | ¥{results['tool_cost']:,.2f} |
| 培训成本（月均） | ¥{results['training_cost_monthly']:,.2f} |
| **月度总成本** | **¥{results['monthly_cost']:,.2f}** |

## 月度收益分析
| 来源 | 金额 |
|------|------|
| 效率提升收益 | ¥{results['efficiency_savings']:,.2f} |
| 质量提升收益 | ¥{results['quality_savings']:,.2f} |
| **月度总收益** | **¥{results['total_benefit']:,.2f}** |

## ROI 分析
| 指标 | 数值 |
|------|------|
| 月净收益 | ¥{results['net_benefit']:,.2f} |
| **ROI** | **{results['roi_percentage']}%** |
| 投资回收期 | {results['payback_period_months']} 个月 |

## 结论
{'✅ 投资回报率为正，建议推进 AI 编程工具部署' if results['roi_percentage'] > 0 else '⚠️ 当前配置 ROI 为负，需要优化使用方式或调整预期'}
"""


if __name__ == '__main__':
    calculator = ROICalculator(
        team=TeamConfig(team_size=10, avg_hourly_rate=200),
        ai_tool=AIToolConfig(monthly_subscription=150, training_hours=8),
        efficiency=EfficiencyGains(),
        quality=QualityGains(),
    )
    print(calculator.generate_report())
```

### 4.2 看板生成脚本

```python
# scripts/generate_dashboard.py
import json
from datetime import datetime, timedelta
from typing import Dict, List

class DashboardGenerator:
    def __init__(self, metrics_data: List[Dict]):
        self.metrics = metrics_data

    def calculate_period_metrics(self, days: int = 30) -> Dict:
        cutoff = datetime.now() - timedelta(days=days)
        period_metrics = [
            m for m in self.metrics
            if datetime.fromisoformat(m['timestamp']) > cutoff
        ]

        return {
            'efficiency': self._calc_efficiency(period_metrics),
            'quality': self._calc_quality(period_metrics),
            'economic': self._calc_economic(period_metrics),
        }

    def _calc_efficiency(self, metrics: List[Dict]) -> Dict:
        gen_metrics = [m for m in metrics if m['metricName'] == 'code_generation']
        task_metrics = [m for m in metrics if m['metricName'] == 'task_completion']

        return {
            'total_lines_generated': sum(m['value'] for m in gen_metrics),
            'acceptance_rate': (
                len([m for m in gen_metrics if m['context'].get('accepted')])
                / max(len(gen_metrics), 1)
            ),
            'avg_task_hours': (
                sum(m['value'] for m in task_metrics)
                / max(len(task_metrics), 1)
            ),
        }

    def _calc_quality(self, metrics: List[Dict]) -> Dict:
        bug_metrics = [m for m in metrics if m['metricName'] == 'bug_found']
        ai_bugs = [m for m in bug_metrics if m['context'].get('source') == 'ai']
        human_bugs = [m for m in bug_metrics if m['context'].get('source') == 'human']

        return {
            'total_bugs': len(bug_metrics),
            'ai_bug_rate': len(ai_bugs) / max(len(bug_metrics), 1),
            'human_bug_rate': len(human_bugs) / max(len(bug_metrics), 1),
        }

    def _calc_economic(self, metrics: List[Dict]) -> Dict:
        efficiency = self._calc_efficiency(metrics)
        # 简化的经济计算
        return {
            'estimated_hours_saved': efficiency['total_lines_generated'] * 0.01,
            'estimated_savings': efficiency['total_lines_generated'] * 0.01 * 200,
        }

    def generate_html_dashboard(self) -> str:
        data = self.calculate_period_metrics()

        return f"""
<!DOCTYPE html>
<html>
<head>
    <title>AI 编程效果看板</title>
    <style>
        body {{ font-family: -apple-system, sans-serif; padding: 20px; }}
        .card {{ background: #f5f5f5; border-radius: 8px; padding: 20px; margin: 10px; }}
        .metric {{ font-size: 2em; font-weight: bold; color: #2563eb; }}
        .trend-up {{ color: #10b981; }}
        .trend-down {{ color: #ef4444; }}
    </style>
</head>
<body>
    <h1>AI 编程效果看板</h1>
    <div style="display: grid; grid-template-columns: repeat(3, 1fr);">
        <div class="card">
            <h3>效率指标</h3>
            <div class="metric">{data['efficiency']['total_lines_generated']}</div>
            <p>本月 AI 生成代码行数</p>
        </div>
        <div class="card">
            <h3>质量指标</h3>
            <div class="metric">{data['quality']['ai_bug_rate']:.1%}</div>
            <p>AI 代码 bug 率</p>
        </div>
        <div class="card">
            <h3>经济效益</h3>
            <div class="metric">¥{data['economic']['estimated_savings']:,.0f}</div>
            <p>本月预估节约</p>
        </div>
```

## 6. 常见误区

### ❌ 错误 1：只看代码行数

```
问题：用代码行数衡量 AI 价值，导致生成冗余代码

解决：综合指标评估
- 代码行数 × 质量权重
- 功能点完成数
- bug 率变化
- 用户满意度
```

### ❌ 错误 2：不区分 AI 贡献和人工贡献

```
问题：把所有改进都归功于 AI

解决：建立对照组或前后对比
- A/B 测试：部分用 AI，部分不用
- 时间序列对比：引入 AI 前后的变化
```

### ❌ 错误 3：忽略隐性成本

```
问题：只计算工具订阅费，忽略隐性成本

隐性成本包括：
- 培训时间成本
- Prompt 调试时间
- AI 输出审查时间
- 安全检查成本
```

### ❌ 错误 4：短期数据下结论

```
问题：用一周数据就得出 ROI 结论

解决：
- 至少收集 1-3 个月数据
- 考虑学习曲线因素
- 季度性评估
```

## 7. 工程建议

1. **从 3 个核心指标开始**：不要一开始就设计 20 个指标，先从"代码补全接受率""任务完成时间变化""AI 代码 Bug 率"这三个最容易收集的指标开始，用数据说话比完美的指标体系更重要。
2. **ROI 计算要考虑隐性成本**：工具订阅费只是冰山一角，培训时间、Prompt 调试时间、AI 输出审查时间都是成本。忽略隐性成本会高估 ROI，导致管理层期望过高。
3. **至少收集 1-3 个月数据再下结论**：AI 工具有学习曲线，第一周的数据不代表长期效果。建议至少收集一个完整季度的数据，再做 ROI 评估和工具选型决策。
4. **度量是为了优化而非考核**：把度量数据用于发现优化机会（哪些 Prompt 需要改进、哪些场景不适合用 AI），而不是用于考核个人绩效，否则会导致数据造假和抵触情绪。

## 参考答案

### 练习 1：设计指标体系

**思路**：指标体系应覆盖效率、质量、经济、能力四个维度，每个指标需要定义收集方法、基线值和目标值。关键是选择容易收集且能直接反映 AI 价值的指标，避免过度设计。

**答案**：

```markdown
# AI 编程效果指标体系

## 指标定义

| # | 维度 | 指标名称 | 定义 | 收集方式 | 基线值 | 目标值 |
|---|------|---------|------|---------|--------|--------|

## 数据可视化方案

### 效率看板
- 折线图：代码补全接受率趋势（按周）
- 柱状图：任务完成时间对比（使用 AI 前 vs 后）
- 数字卡片：本周代码生成速度

### 质量看板
- 折线图：AI 代码 Bug 率趋势（按月）
- 饼图：PR 一次通过率 vs 需修改率
- 堆叠柱状图：Bug 来源分布（AI vs Human）

### 经济看板
- 数字卡片：本月 ROI
- 面积图：成本 vs 收益趋势
- 表格：各团队/项目 ROI 排名
```

**要点**：
- 基线值必须在引入 AI 工具前测量，否则无法计算提升幅度
- 代码补全接受率是最容易收集的指标，IDE 插件自动上报，适合作为第一个落地指标
- 经济指标需要至少 1-3 个月数据才能有意义，短期数据波动大
- 指标体系不是一成不变的，应根据实际使用情况每季度调整

### 练习 2：ROI 计算实践

**思路**：使用课程中的 ROI 计算器，代入具体参数计算。关键是考虑隐性成本（培训、审查时间），不能只看工具订阅费。

**答案**：

```python
# scripts/roi_practice.py
from dataclasses import dataclass

@dataclass
class TeamConfig:
    team_size: int = 15
    avg_hourly_rate: float = 250  # 元/小时
    working_hours_per_month: float = 160.0

@dataclass
class AIToolConfig:
    monthly_subscription: float = 19 * 7.2  # $19 转人民币 ≈ ¥137/人/月
    setup_cost: float = 5000  # 初始配置和培训成本
    training_hours: float = 10  # 每人培训时长

@dataclass
class EfficiencyGains:
    code_generation_speedup: float = 1.4  # 代码生成提速 40%
    code_review_speedup: float = 1.3  # review 提速 30%
    debugging_speedup: float = 1.2  # 调试提速 20%
    documentation_speedup: float = 2.0  # 文档编写提速 100%

@dataclass
class QualityGains:
    bug_reduction_rate: float = 0.25  # bug 减少 25%
    avg_bug_fix_hours: float = 4.0
    bugs_per_month: float = 30  # 每月 30 个 bug
    incident_reduction_rate: float = 0.15
    avg_incident_cost: float = 5000.0

def calculate_roi():
    team = TeamConfig()
    ai_tool = AIToolConfig()
    efficiency = EfficiencyGains()
    quality = QualityGains()

    # 月度成本
    tool_cost = team.team_size * ai_tool.monthly_subscription
    training_cost = (team.team_size * ai_tool.training_hours * team.avg_hourly_rate) / 12
    monthly_cost = tool_cost + training_cost

    # 效率收益
    coding_hours = team.working_hours_per_month * 0.6
    review_hours = team.working_hours_per_month * 0.2
    debug_hours = team.working_hours_per_month * 0.1
    doc_hours = team.working_hours_per_month * 0.1

    hours_saved = (
        coding_hours * (1 - 1/efficiency.code_generation_speedup)
        + review_hours * (1 - 1/efficiency.code_review_speedup)
        + debug_hours * (1 - 1/efficiency.debugging_speedup)
        + doc_hours * (1 - 1/efficiency.documentation_speedup)
    )
    efficiency_savings = hours_saved * team.avg_hourly_rate * team.team_size

    # 质量收益
    bug_savings = (
        quality.bugs_per_month * quality.bug_reduction_rate
        * quality.avg_bug_fix_hours * team.avg_hourly_rate
    )
    incident_savings = quality.incident_reduction_rate * quality.avg_incident_cost
    quality_savings = bug_savings + incident_savings

    total_benefit = efficiency_savings + quality_savings
    roi = (total_benefit - monthly_cost) / monthly_cost * 100

    print("=" * 50)
    print("AI 编程投资回报分析报告")
    print("=" * 50)
    print(f"团队规模: {team.team_size} 人")
    print(f"平均时薪: ¥{team.avg_hourly_rate}")
    print(f"工具月费: ¥{ai_tool.monthly_subscription}/人")
    print()
    print("--- 月度成本 ---")
    print(f"工具订阅费: ¥{tool_cost:,.2f}")
    print(f"培训成本（月均）: ¥{training_cost:,.2f}")
    print(f"月度总成本: ¥{monthly_cost:,.2f}")
    print()
    print("--- 月度收益 ---")
    print(f"效率提升收益: ¥{efficiency_savings:,.2f}")
    print(f"质量提升收益: ¥{quality_savings:,.2f}")
    print(f"月度总收益: ¥{total_benefit:,.2f}")
    print()
    print("--- ROI 分析 ---")
    print(f"月净收益: ¥{total_benefit - monthly_cost:,.2f}")
    print(f"ROI: {roi:.1f}%")
    payback = ai_tool.setup_cost / max(total_benefit - monthly_cost, 1)
    print(f"投资回收期: {payback:.1f} 个月")
    print()
    print("结论: ✅ 投资回报率为正，建议推进 AI 编程工具部署" if roi > 0 else "结论: ⚠️ ROI 为负，需优化")

calculate_roi()
```

### 练习 3：监控看板设计

**思路**：选择 5 个最能反映 AI 编程效果的核心指标，设计直观的可视化布局，设置合理的告警阈值。

**答案**：

```python

ALERT_RULES = [
    AlertRule("代码补全接受率", "below", 25, "接受率低于 25%，建议优化 Prompt 模板", "warning"),
    AlertRule("AI 代码 Bug 率", "above", 5, "Bug 率超过 5/千行，建议加强人工 Review", "critical"),
    AlertRule("月度 ROI", "below", 100, "ROI 低于 100%，成本已超过收益", "warning"),
    AlertRule("工具利用率", "below", 50, "工具利用率低于 50%，需要培训推广", "warning"),
]

def check_alerts(metrics: Dict[str, float]) -> List[AlertRule]:
    """检查指标是否触发告警"""
    triggered = []
    for rule in ALERT_RULES:
        value = metrics.get(rule.metric)
        if value is None:
            continue
        if rule.condition == "below" and value < rule.threshold:
            triggered.append(rule)
        elif rule.condition == "above" and value > rule.threshold:
            triggered.append(rule)
    return triggered

            for a in alerts
```
