## 场景引入

数据采集是量化交易中最"脏"的活——没有人愿意做，但所有人都绕不开。**数据质量直接决定策略的上限**，一个再精妙的 Alpha 模型，喂进去错误的数据，输出的也是垃圾信号。

### 1.2 各类数据详解

**行情数据（Market Data）**

行情数据是最基础的数据类型，记录资产在不同时间粒度上的价格和成交量信息：

| 字段 | 含义 | 示例 |
|------|------|------|
| open | 开盘价 | 10.50 |
| high | 最高价 | 10.85 |
| low | 最低价 | 10.32 |
| close | 收盘价 | 10.70 |
| volume | 成交量（股） | 15,000,000 |
| amount | 成交额（元） | 160,500,000 |
| turnover_rate | 换手率 | 1.23% |

**基本面数据（Fundamental Data）**

```
基本面数据层级：

  公司层面
    ├── 利润表（营收、净利润、毛利率）
    ├── 资产负债表（总资产、负债率、流动比率）
    ├── 现金流量表（经营/投资/筹资现金流）
    └── 估值指标（PE、PB、PS、EV/EBITDA）

  宏观层面
    ├── GDP、CPI、PPI、PMI
    ├── 货币供应量（M1、M2）
    ├── 利率（LPR、SHIBOR、国债收益率）
    └── 汇率（人民币中间价）
```

**另类数据（Alternative Data）**

另类数据是近年来量化领域的热门方向，来源于传统金融渠道之外：

- **舆情数据**：新闻标题情绪分析、社交媒体讨论热度
- **分析师数据**：盈利预测修正、目标价变化
- **资金流向**：主力资金净流入、北向资金持仓变化
- **产业链数据**：上游原材料价格、下游需求指标

class RateLimiter:
    """
    令牌桶限流器
    控制 API 调用频率，避免触发数据源的限流机制
    """
    def __init__(self, calls_per_minute: int = 60):
        self.calls_per_minute = calls_per_minute
        self.interval = 60.0 / calls_per_minute
        self.last_call_time = 0.0
        self.call_count = 0
        self.window_start = time.time()

    def wait(self):
        """等待直到可以发起下一次调用"""
        now = time.time()
        elapsed = now - self.window_start

        # 如果超过 1 分钟窗口，重置计数器
        if elapsed >= 60:
            self.window_start = now
            self.call_count = 0

        # 如果当前窗口内调用次数已达上限，等待到下一个窗口
        if self.call_count >= self.calls_per_minute:
            wait_time = 60 - elapsed + 0.1
            logger.debug(f"限流等待 {wait_time:.1f} 秒")
            time.sleep(wait_time)
            self.window_start = time.time()
            self.call_count = 0
        else:
            # 确保两次调用之间有最小间隔
            time_since_last = now - self.last_call_time
            if time_since_last < self.interval:
                time.sleep(self.interval - time_since_last)

        self.last_call_time = time.time()
        self.call_count += 1


def retry_with_backoff(
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    exceptions: tuple = (Exception,)
):
    """
    指数退避重试装饰器
    当 API 调用失败时，自动重试并逐步增加等待时间
    """
    def decorator(func: Callable):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    if attempt == max_retries:
                        logger.error(
                            f"{func.__name__} 在 {max_retries} 次重试后仍然失败: {e}"
                        )
                        raise
                    delay = min(base_delay * (2 ** attempt), max_delay)
                    logger.warning(
                        f"{func.__name__} 第 {attempt + 1} 次失败，"
                        f"{delay:.1f} 秒后重试: {e}"
                    )
                    time.sleep(delay)
            raise last_exception
        return wrapper
    return decorator


class DataCollector:
    """
    A 股数据采集器
    集成限流、重试、缓存、质量检查等功能
    """
    def __init__(
        self,
        cache_dir: str = "./data_cache",
        calls_per_minute: int = 60,
        max_retries: int = 3,
    ):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self.rate_limiter = RateLimiter(calls_per_minute)
        self.max_retries = max_retries

    def _get_cache_path(self, key: str) -> Path:
        """根据缓存 key 生成文件路径"""
        hash_val = hashlib.md5(key.encode()).hexdigest()[:12]
        return self.cache_dir / f"{key}_{hash_val}.parquet"

    def _load_cache(self, key: str, max_age_hours: int = 24) -> Optional[pd.DataFrame]:
        """从缓存加载数据"""
        cache_path = self._get_cache_path(key)
        if cache_path.exists():
            mtime = datetime.fromtimestamp(cache_path.stat().st_mtime)
            age_hours = (datetime.now() - mtime).total_seconds() / 3600
            if age_hours < max_age_hours:
                logger.info(f"从缓存加载: {key} (已缓存 {age_hours:.1f} 小时)")
                return pd.read_parquet(cache_path)
        return None

    def _save_cache(self, key: str, df: pd.DataFrame):
        """保存数据到缓存"""
        cache_path = self._get_cache_path(key)
        df.to_parquet(cache_path, index=False)
        logger.info(f"数据已缓存: {key}")

    @retry_with_backoff(max_retries=3, base_delay=2.0)
    def fetch_stock_daily(
        self,
        symbol: str,
        start_date: str,
        end_date: str,
        adjust: str = "qfq"  # qfq=前复权, hfq=后复权, ""=不复权
    ) -> pd.DataFrame:
        """
        获取个股日线数据

        Args:
            symbol: 股票代码，如 "000001"
            start_date: 开始日期，如 "20230101"
            end_date: 结束日期，如 "20241231"
            adjust: 复权方式

        Returns:
            包含 OHLCV 数据的 DataFrame
        """
        cache_key = f"daily_{symbol}_{start_date}_{end_date}_{adjust}"
        cached = self._load_cache(cache_key)
        if cached is not None:
            return cached

        self.rate_limiter.wait()

        try:
            df = ak.stock_zh_a_hist(
                symbol=symbol,
                period="daily",
                start_date=start_date,
                end_date=end_date,
                adjust=adjust,
            )

            # 标准化列名
            df = df.rename(columns={
                "日期": "date",
                "开盘": "open",
                "收盘": "close",
                "最高": "high",
                "最低": "low",
                "成交量": "volume",
                "成交额": "amount",
                "振幅": "amplitude",
                "涨跌幅": "pct_change",
                "涨跌额": "price_change",
                "换手率": "turnover_rate",
            })

            df["date"] = pd.to_datetime(df["date"])
            df["symbol"] = symbol

            self._save_cache(cache_key, df)
            logger.info(
                f"获取 {symbol} 日线数据: {len(df)} 条 "
                f"({start_date} ~ {end_date})"
            )
            return df

        except Exception as e:
            logger.error(f"获取 {symbol} 日线数据失败: {e}")
            raise

            return df
        except Exception as e:
            logger.error(f"获取 {symbol} {report_type} 失败: {e}")
            raise

    @retry_with_backoff(max_retries=3, base_delay=2.0)
    def fetch_index_daily(
        self,

### 3.3 数据质量检查

```python
class DataQualityChecker:
    """
    数据质量检查器
    对采集到的数据执行一系列质量检查，确保数据可用
    """
    def __init__(self):
        self.checks_passed = 0
        self.checks_failed = 0
        self.warnings: list[str] = []

    def check_missing_values(
        self, df: pd.DataFrame, columns: list[str]
    ) -> bool:
        """检查缺失值"""
        has_issues = False
        for col in columns:
            if col in df.columns:
                missing_count = df[col].isna().sum()
                missing_pct = missing_count / len(df) * 100
                if missing_pct > 0:
                    self.warnings.append(
                        f"列 '{col}' 有 {missing_count} 个缺失值 "
                        f"({missing_pct:.2f}%)"
                    )
                    if missing_pct > 5:
                        has_issues = True
        if has_issues:
            self.checks_failed += 1
        else:
            self.checks_passed += 1
        return not has_issues

    def check_date_continuity(
        self, df: pd.DataFrame, date_col: str = "date"
    ) -> bool:
        """检查日期连续性（排除周末和节假日）"""
        if date_col not in df.columns:
            self.warnings.append(f"缺少日期列 '{date_col}'")
            self.checks_failed += 1
            return False

        dates = pd.to_datetime(df[date_col]).sort_values()
        gaps = dates.diff().dt.days

        # 超过 5 天的间隔（考虑节假日）标记为异常
        large_gaps = gaps[gaps > 5]
        if len(large_gaps) > 0:
            for idx in large_gaps.index:
                gap_days = large_gaps[idx]
                gap_start = dates.loc[idx - 1] if idx > 0 else "N/A"
                self.warnings.append(
                    f"日期间隔异常: {gap_start} 到 {dates.loc[idx]} "
                    f"间隔 {gap_days} 天"
                )
            self.checks_failed += 1
            return False

        self.checks_passed += 1
        return True

    def check_price_validity(self, df: pd.DataFrame) -> bool:
        """检查价格数据有效性"""
        issues = []

        # 检查负价格
        for col in ["open", "high", "low", "close"]:
            if col in df.columns:
                negative = (df[col] < 0).sum()
                if negative > 0:
                    issues.append(f"列 '{col}' 有 {negative} 个负值")

        # 检查零价格
        for col in ["open", "high", "low", "close"]:
            if col in df.columns:
                zeros = (df[col] == 0).sum()
                if zeros > 0:
                    issues.append(f"列 '{col}' 有 {zeros} 个零值")

        # 检查 OHLC 逻辑关系
        if all(c in df.columns for c in ["open", "high", "low", "close"]):
            # 最高价应 >= 开盘价和收盘价
            invalid_high = (
                (df["high"] < df["open"]) | (df["high"] < df["close"])
            ).sum()
            if invalid_high > 0:
                issues.append(f"最高价低于开盘/收盘价: {invalid_high} 条")

            # 最低价应 <= 开盘价和收盘价
            invalid_low = (
                (df["low"] > df["open"]) | (df["low"] > df["close"])
            ).sum()
            if invalid_low > 0:
                issues.append(f"最低价高于开盘/收盘价: {invalid_low} 条")

        # 检查涨跌幅异常（A 股涨跌停一般为 10%）
        if "pct_change" in df.columns:
            extreme = (df["pct_change"].abs() > 11).sum()
            if extreme > 0:
                issues.append(f"涨跌幅超过 ±11%: {extreme} 条（可能是 ST 股或新股）")

        if issues:
            self.warnings.extend(issues)
            self.checks_failed += 1
            return False

        self.checks_passed += 1
        return True

    def check_duplicate_dates(
        self, df: pd.DataFrame, date_col: str = "date"
    ) -> bool:
        """检查重复日期"""
        if date_col not in df.columns:
            return False

        duplicates = df[date_col].duplicated().sum()
        if duplicates > 0:
            self.warnings.append(f"存在 {duplicates} 条重复日期记录")
            self.checks_failed += 1
            return False

        self.checks_passed += 1
        return True

    def check_volume_validity(self, df: pd.DataFrame) -> bool:
        """检查成交量有效性"""
        if "volume" not in df.columns:
            return True

        # 检查负成交量
        negative_vol = (df["volume"] < 0).sum()
        if negative_vol > 0:
            self.warnings.append(f"有 {negative_vol} 条负成交量")
            self.checks_failed += 1
            return False

        # 检查零成交量（可能是停牌）
        zero_vol = (df["volume"] == 0).sum()
        if zero_vol > 0:
            self.warnings.append(
                f"有 {zero_vol} 条零成交量（可能是停牌日）"
            )

        self.checks_passed += 1
        return True

    def run_all_checks(self, df: pd.DataFrame) -> dict:
        """运行所有质量检查"""
        self.checks_passed = 0
        self.checks_failed = 0
        self.warnings = []

        price_cols = ["open", "high", "low", "close"]
        existing_price_cols = [c for c in price_cols if c in df.columns]

        self.check_missing_values(df, existing_price_cols + ["volume"])
        self.check_date_continuity(df)
        self.check_price_validity(df)
        self.check_duplicate_dates(df)
        self.check_volume_validity(df)

        total = self.checks_passed + self.checks_failed
        return {
            "total_checks": total,
            "passed": self.checks_passed,
            "failed": self.checks_failed,
            "pass_rate": f"{self.checks_passed / total * 100:.1f}%"
            if total > 0 else "N/A",
            "warnings": self.warnings,
        }
```

### 3.4 公司行为处理

A 股市场存在多种公司行为，会直接影响历史数据的连续性：

```
公司行为类型：

  除权除息（分红、送股、配股）
    ├── 前复权：以最新价为基准，向前调整历史价格
    ├── 后复权：以上市首日为基准，向后调整价格
    └── 不复权：原始价格

  停牌
    ├── 日常停牌（临时公告）
    └── 长期停牌（重大资产重组）

  涨跌停
    ├── 主板 / 创业板：±10%
    ├── 科创板：±20%
    └── 北交所：±30%
```

```python
def handle_corporate_actions(df: pd.DataFrame) -> pd.DataFrame:
    """
    处理公司行为对数据的影响
    1. 识别并标记停牌日
    2. 处理复权因子
    3. 标记涨跌停
    """
    df = df.copy()

    # 标记停牌日
    df["is_suspended"] = (df["volume"] == 0) | (df["volume"].isna())

    # 标记涨跌停
    if "pct_change" in df.columns:
        df["is_limit_up"] = df["pct_change"] >= 9.8   # 留一定容差
        df["is_limit_down"] = df["pct_change"] <= -9.8
    else:
        df["is_limit_up"] = False
        df["is_limit_down"] = False

    # 过滤停牌日数据（保留标记但不参与计算）
    logger.info(
        f"公司行为处理完成: "
        f"停牌 {df['is_suspended'].sum()} 天, "
        f"涨停 {df['is_limit_up'].sum()} 次, "
        f"跌停 {df['is_limit_down'].sum()} 次"
    )
    return df
```

### 4.2 调度器实现

```python
import schedule
from concurrent.futures import ThreadPoolExecutor, as_completed


class DataPipeline:
    """
    数据采集管道
    支持批量采集、质量检查、错误处理和报告生成
    """
    def __init__(
        self,
        collector: DataCollector,
        checker: DataQualityChecker,
        output_dir: str = "./data_output",
    ):
        self.collector = collector
        self.checker = checker
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(parents=True, exist_ok=True)

    def batch_collect_daily(
        self,
        symbols: list[str],
        start_date: str,
        end_date: str,
        max_workers: int = 4,
    ) -> dict[str, pd.DataFrame]:
        """
        批量采集多只股票的日线数据

        Args:
            symbols: 股票代码列表
            start_date: 开始日期
            end_date: 结束日期
            max_workers: 并发数（注意总 QPS 不超过限制）

        Returns:
            {symbol: DataFrame} 字典
        """
        results = {}
        failed = []
        total = len(symbols)

        logger.info(f"开始批量采集 {total} 只股票数据...")

        def fetch_one(sym: str) -> tuple[str, Optional[pd.DataFrame]]:
            try:
                df = self.collector.fetch_stock_daily(
                    sym, start_date, end_date
                )
                df = handle_corporate_actions(df)
                return sym, df
            except Exception as e:
                logger.error(f"采集 {sym} 失败: {e}")
                return sym, None

        # 串行执行以控制限流（并发容易触发限流）
        for i, symbol in enumerate(symbols, 1):
            sym, df = fetch_one(symbol)
            if df is not None:
                results[sym] = df
            else:
                failed.append(sym)

            if i % 50 == 0:
                logger.info(f"进度: {i}/{total} ({i/total*100:.1f}%)")

        logger.info(
            f"批量采集完成: 成功 {len(results)}, 失败 {len(failed)}"
        )
        if failed:
            logger.warning(f"失败列表: {failed[:20]}...")

        return results

    def generate_quality_report(
        self, data_dict: dict[str, pd.DataFrame]
    ) -> pd.DataFrame:
        """生成数据质量报告"""
        report_rows = []

        for symbol, df in data_dict.items():
            result = self.checker.run_all_checks(df)
            report_rows.append({
                "symbol": symbol,
                "rows": len(df),
                "date_range": f"{df['date'].min()} ~ {df['date'].max()}",
                "checks_passed": result["passed"],
                "checks_failed": result["failed"],
                "pass_rate": result["pass_rate"],
                "warnings": "; ".join(result["warnings"][:3]),
            })

        report_df = pd.DataFrame(report_rows)
        report_path = self.output_dir / f"quality_report_{datetime.now():%Y%m%d}.csv"
        report_df.to_csv(report_path, index=False, encoding="utf-8-sig")
        logger.info(f"质量报告已保存: {report_path}")
        return report_df

    def save_to_parquet(
        self, data_dict: dict[str, pd.DataFrame], subdir: str = "daily"
    ):
        """将采集的数据保存为 Parquet 格式"""
        save_dir = self.output_dir / subdir
        save_dir.mkdir(parents=True, exist_ok=True)

        for symbol, df in data_dict.items():
            path = save_dir / f"{symbol}.parquet"
            df.to_parquet(path, index=False)

        logger.info(f"已保存 {len(data_dict)} 个文件到 {save_dir}")


def daily_pipeline_job():
    """每日数据采集任务"""
    collector = DataCollector(calls_per_minute=60)
    checker = DataQualityChecker()
    pipeline = DataPipeline(collector, checker)

    # 获取股票列表
    stock_list = collector.fetch_stock_list()
    symbols = stock_list["symbol"].tolist()[:100]  # 示例：只取前 100 只

    # 采集最近 1 年数据
    end_date = datetime.now().strftime("%Y%m%d")
    start_date = (datetime.now() - timedelta(days=365)).strftime("%Y%m%d")

    data_dict = pipeline.batch_collect_daily(
        symbols, start_date, end_date
    )

    # 质量检查
    report = pipeline.generate_quality_report(data_dict)
    print(report.head())

    # 保存数据
    pipeline.save_to_parquet(data_dict)


def setup_scheduler():
    """配置定时任务"""
    # 每个交易日 16:30 执行
    schedule.every().day.at("16:30").do(daily_pipeline_job)

    # 每周六 02:00 更新基础数据
    schedule.every().saturday.at("02:00").do(
        lambda: logger.info("执行周度数据更新...")
    )

    logger.info("调度器已启动，等待任务触发...")
    while True:
        schedule.run_pending()
        time.sleep(60)
```

## 六、常见误区

### 误区一："免费数据源够用了"

免费数据源（如 AKShare）适合学习和原型开发，但存在几个问题：
- 数据质量不稳定，可能有缺失或错误
- 没有 SLA 保障，API 可能随时变更
- 不适合生产环境

如果策略涉及实盘资金，建议使用有 SLA 保障的商业数据源。

### 误区二："爬虫比 API 好"

有些初学者认为直接爬网页可以"白嫖"数据。但网页爬虫的问题是：
- 网页结构变化导致爬虫失效
- 数据格式不规范，清洗成本高
- 可能违反网站使用条款
- 稳定性远不如正规 API

### 误区三："数据采集是一次性工作"

数据采集是一个**持续运行的系统**，不是写个脚本跑一次就完事：
- 需要定时更新
- 需要处理异常和重试
- 需要监控数据质量
- 需要应对数据源变更

### 误区四："有了数据就能直接用"

原始数据通常需要经过清洗、对齐、复权等处理才能使用。下一节课我们将详细讲解数据清洗与对齐。

### 练习二

**思路**：获取当前股票列表，与历史数据中的股票进行交叉比对。

**答案**：

```python
def check_survivorship_bias(
    self,
    historical_symbols: list[str],
    current_symbols: list[str]
) -> dict:
    """检查幸存者偏差"""
    historical_set = set(historical_symbols)
    current_set = set(current_symbols)

    # 已退市的股票（在历史中存在但当前不存在）
    delisted = historical_set - current_set
    # 新上市的股票
    newly_listed = current_set - historical_set
    # 持续存在的股票
    survived = historical_set & current_set

    bias_ratio = len(delisted) / len(historical_set) * 100

    result = {
        "历史股票数": len(historical_set),
        "当前股票数": len(current_set),
        "已退市数": len(delisted),
        "新上市数": len(newly_listed),
        "存活数": len(survived),
        "偏差比例": f"{bias_ratio:.2f}%",
    }

    if bias_ratio > 5:
        self.warnings.append(
            f"幸存者偏差警告: {bias_ratio:.2f}% 的历史股票已退市"
        )

    return result
```

**要点**：
- 幸存者偏差会导致回测结果虚高
- 应该使用"包含退市股票的历史数据"进行回测
- 商业数据源通常提供"历史成分股"数据来解决这个问题

### 练习三

**思路**：使用文件记录已完成的股票列表，断点续传时跳过已完成的。

**答案**：

```python
def batch_collect_with_checkpoint(
    self,
    symbols: list[str],
    start_date: str,
    end_date: str,
    checkpoint_file: str = "checkpoint.json",
) -> dict[str, pd.DataFrame]:
    """带断点续传的批量采集"""
    checkpoint_path = self.output_dir / checkpoint_file

    # 加载已完成的记录
    completed = set()
    if checkpoint_path.exists():
        with open(checkpoint_path, "r") as f:
            completed = set(json.load(f).get("completed", []))
        logger.info(f"已恢复进度: {len(completed)}/{len(symbols)}")

    remaining = [s for s in symbols if s not in completed]
    results = {}

    # 加载已有数据
    for sym in completed:
        data_path = self.output_dir / "daily" / f"{sym}.parquet"
        if data_path.exists():
            results[sym] = pd.read_parquet(data_path)

    # 采集剩余数据
    for i, symbol in enumerate(remaining, 1):
        try:
            df = self.collector.fetch_stock_daily(
                symbol, start_date, end_date
            )
            df = handle_corporate_actions(df)
            results[symbol] = df

            # 保存单个文件
            save_path = self.output_dir / "daily" / f"{symbol}.parquet"
            save_path.parent.mkdir(parents=True, exist_ok=True)
            df.to_parquet(save_path, index=False)

            # 更新 checkpoint
            completed.add(symbol)
            with open(checkpoint_path, "w") as f:
                json.dump({"completed": list(completed)}, f)

            if i % 10 == 0:
                logger.info(
                    f"进度: {len(completed)}/{len(symbols)} "
                    f"(本次: {i}/{len(remaining)})"
                )

        except Exception as e:
            logger.error(f"采集 {symbol} 失败: {e}")
            continue

    logger.info(f"采集完成: {len(results)}/{len(symbols)}")
    return results
```

**要点**：
- checkpoint 文件用 JSON 格式记录已完成的股票列表
- 每次采集成功后立即更新 checkpoint
- 已完成的数据从本地文件加载，避免重复采集
