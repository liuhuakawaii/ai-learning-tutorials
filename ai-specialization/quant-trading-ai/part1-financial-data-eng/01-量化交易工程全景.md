## 从一笔交易的生命周期说起

你在 10:15:03 决定买入 1000 股贵州茅台。从按下按钮到股票到账，中间经历了什么？

```python
import pandas as pd
import numpy as np
from datetime import datetime, timedelta
from dataclasses import dataclass
from typing import List

np.random.seed(42)

@dataclass
class TradeEvent:
    timestamp: datetime
    event_type: str
    detail: str
    latency_ms: float = 0.0

def simulate_trade_lifecycle():
    """模拟一笔交易从信号到成交的完整过程"""
    base = datetime(2025, 3, 15, 10, 15, 3)
    events = [
        TradeEvent(base, 'SIGNAL', '双均线金叉: MA5(1823.5) > MA20(1819.2)', 0),
        TradeEvent(base + timedelta(ms=2), 'RISK_CHECK', '通过: 仓位8.3% < 20%, 日亏损0.1% < 2%', 2),
        TradeEvent(base + timedelta(ms=3), 'ORDER_CREATE', '限价单: 600519.SH BUY 100股 @ 1825.00', 1),
        TradeEvent(base + timedelta(ms=8), 'ORDER_SEND', 'CTP发送到上交所, 网络延迟3ms', 5),
        TradeEvent(base + timedelta(ms=16), 'EXCHANGE_ACK', '上交所确认, 排队序号#347', 8),
        TradeEvent(base + timedelta(s=12), 'PARTIAL_FILL', '成交60股 @ 1824.50, 剩余40股', 12000),
        TradeEvent(base + timedelta(s=15), 'FILL', '成交40股 @ 1825.00, 均价1824.70', 3000),
        TradeEvent(base + timedelta(s=15), 'COMMISSION', '佣金5.47元 + 过户费0.18元 = 5.65元', 0),
    ]
    df = pd.DataFrame([{'时间': e.timestamp.strftime('%H:%M:%S'), '类型': e.event_type,
                        '详情': e.detail} for e in events])
    print("一笔交易的完整生命周期:")
    print(df.to_string(index=False))
    print(f"\n总耗时: 约15秒")

simulate_trade_lifecycle()

def analyze_latency():
    """分析各模块延迟分布"""
    np.random.seed(42)
    n = 1000
    latency = pd.DataFrame({
        '信号生成': np.random.lognormal(1.5, 0.5, n),
        '风控检查': np.random.lognormal(0.5, 0.3, n),
        '网络传输': np.random.lognormal(2.0, 0.8, n),
        '交易所': np.random.lognormal(3.0, 1.0, n),
    })
    stats = latency.describe().loc[['mean', '95%']].T
    print("各模块延迟 (毫秒):")
    print(stats.round(1).to_string())
    print(f"瓶颈: {latency.mean().idxmax()} (占比 {latency.mean().max()/latency.mean().sum():.0%})")

analyze_latency()
```

网络传输和交易所处理占了总延迟的 80%。对日线策略无所谓，对高频策略是生死线。

## Alpha 是怎么来的

```python
def alpha_decay():
    """模拟 Alpha 信号衰减"""
    days = np.arange(500)
    alpha = 0.002 * np.exp(-0.003 * days) + np.random.normal(0, 0.001, 500)
    alpha = np.maximum(alpha, 0)
    cum = np.cumsum(alpha)
    decay_day = np.argmax(alpha < 0.001)
    print(f"Alpha 初始值: 0.20%/天")
    print(f"衰减到噪声水平: 约 {decay_day} 个交易日")
    print(f"累计超额收益: {cum[decay_day]:.2%}")
    print(f"\nAlpha 来源: 信息优势、分析优势、执行优势、速度优势")
    print(f"结论: Alpha 会衰减，量化交易是持续的军备竞赛")

alpha_decay()
```

## 跑通信号链：双均线策略

```python
def run_signal_chain():
    np.random.seed(42)
    dates = pd.bdate_range('2023-01-01', periods=500)
    close = 100 * np.cumprod(1 + np.random.normal(0.0003, 0.018, 500))
    df = pd.DataFrame({'close': close}, index=dates)

    # 信号模块
    df['ma5'] = df['close'].rolling(5).mean()
    df['ma20'] = df['close'].rolling(20).mean()
    df['signal'] = (df['ma5'] > df['ma20']).astype(int)
    df['change'] = df['signal'].diff()

    # 执行模块
    capital, position, trades = 1_000_000, 0, []
    for date, row in df.iterrows():
        price = row['close']
        if row['change'] == 1 and position == 0:
            shares = int(capital * 0.95 / price / 100) * 100
            if shares > 0:
                capital -= shares * price * 1.002  # 含手续费和滑点
                position = shares
                trades.append(('BUY', date, price))
        elif row['change'] == -1 and position > 0:
            capital += position * price * 0.998
            trades.append(('SELL', date, price))
            position = 0

    final = capital + position * df['close'].iloc[-1]
    ret = final / 1_000_000 - 1
    vol = df['close'].pct_change().std() * np.sqrt(252)
    peak = df['close'].expanding().max()
    dd = ((df['close'] - peak) / peak).min()

    print(f"交易次数: {len(trades)}, 总收益: {ret:.2%}, 最大回撤: {dd:.2%}")

run_signal_chain()
```

## 回测中的常见陷阱

```python
def show_pitfalls():
    np.random.seed(42)
    close = 100 * np.cumprod(1 + np.random.normal(0.0003, 0.018, 500))
    df = pd.DataFrame({'close': close})
    df['ma5'] = df['close'].rolling(5).mean()

    # 前视偏差
    sig_wrong = (df['close'] > df['ma5']).astype(int)
    sig_right = (df['close'].shift(1) > df['ma5'].shift(1)).astype(int)
    ret_w = sig_wrong.shift(1) * df['close'].pct_change()
    ret_r = sig_right.shift(1) * df['close'].pct_change()
    print(f"前视偏差: 错误夏普 {ret_w.mean()/ret_w.std()*np.sqrt(252):.2f} vs 正确 {ret_r.mean()/ret_r.std()*np.sqrt(252):.2f}")

    # 参数过拟合
    best_s, best_w = -999, 0
    for w in range(3, 30):
        ma = df['close'].rolling(w).mean()
        sig = (df['close'].shift(1) > ma.shift(1)).astype(int)
        r = sig.shift(1) * df['close'].pct_change()
        s = r.mean() / r.std() * np.sqrt(252) if r.std() > 0 else 0
        if s > best_s: best_s, best_w = s, w
    print(f"过拟合: 最优窗口 {best_w}天 夏普 {best_s:.2f} (但这个参数是'看到未来'找到的)")

show_pitfalls()
```

## 工程判断清单

1. **先跑通再优化**——不要一开始就设计微服务
2. **数据质量决定策略上限**——花 80% 的时间在数据上不夸张
3. **回测赚钱 ≠ 实盘赚钱**——前视偏差、交易成本、滑点都会吃掉收益
4. **Alpha 会衰减**——持续研究新信号是必要的
5. **简单策略更可靠**——3 参数在 10 年数据上稳定，比 100 参数的深度学习更值得信任

def stability_test():
    np.random.seed(42)
    close = 100 * np.cumprod(1 + np.random.normal(0.0003, 0.018, 500))
    results = []
    for sw in range(3, 11):
        sharpes = []
        for lw in range(15, 41):
            ma_s = pd.Series(close).rolling(sw).mean()
            ma_l = pd.Series(close).rolling(lw).mean()
            sig = (ma_s.shift(1) > ma_l.shift(1)).astype(int)
            r = sig.shift(1) * pd.Series(close).pct_change()
            sharpes.append(r.mean() / r.std() * np.sqrt(252) if r.std() > 0 else 0)
        results.append({'short': sw, 'std': np.std(sharpes)})
    df = pd.DataFrame(results)
    print(f"最稳定窗口: {df.loc[df['std'].idxmin(), 'short']:.0f} (波动 {df['std'].min():.3f})")
