## 一个"看起来很好"的回测

你写了个双均线回测脚本，年化 28%。接入实盘后亏了 8%。问题不在策略，在回测引擎。

```python
import numpy as np
import pandas as pd
from dataclasses import dataclass, field
from enum import Enum
from typing import Dict, List, Optional
from collections import defaultdict

def gen_ohlcv(n=500, trend=0.0003, seed=42):
    np.random.seed(seed)
    r = np.random.normal(trend, 0.02, n)
    c = 100 * np.cumprod(1+r)
    dates = pd.bdate_range('2022-01-01', periods=n)
    return pd.DataFrame({'open': c*(1+np.random.randn(n)*0.003), 'high': c*1.005,
                          'low': c*0.995, 'close': c, 'volume': np.random.randint(1_000_000,10_000_000,n).astype(float)}, index=dates)

def naive_backtest(prices, sw=5, lw=20):
    """粗糙回测——有隐藏 bug"""
    capital = 1_000_000; pos = 0; vals = []
    for i in range(lw, len(prices)):
        p = prices.iloc[i]
        ma_s = prices.iloc[i-sw+1:i+1].mean()
        ma_l = prices.iloc[i-lw+1:i+1].mean()
        if ma_s > ma_l and pos == 0:
            shares = int(capital*0.95/p/100)*100
            capital -= shares*p  # Bug: 无手续费
            pos = shares
        elif ma_s < ma_l and pos > 0:
            capital += pos*p  # Bug: 无手续费
            pos = 0
        vals.append(capital + pos*p)
    return pd.Series(vals, index=prices.index[lw:])

prices = gen_ohlcv()['close']
nv = naive_backtest(prices)
print(f"粗糙回测: 收益 {nv.iloc[-1]/1_000_000-1:.2%} (虚高，无手续费无滑点)")
```

## 事件驱动回测引擎

```python
class OrderSide(Enum):
    BUY = "buy"; SELL = "sell"

@dataclass
class Bar:
    symbol: str; timestamp: object; open: float; high: float; low: float; close: float; volume: float

@dataclass
class Signal:
    symbol: str; timestamp: object; direction: int; strength: float = 1.0

@dataclass
class Fill:
    symbol: str; timestamp: object; side: OrderSide; quantity: int; price: float; commission: float; slippage: float

@dataclass
class Position:
    symbol: str; quantity: int = 0; avg_price: float = 0.0; realized_pnl: float = 0.0

class Portfolio:
    def __init__(self, capital):
        self.initial = capital; self.cash = capital
        self.positions: Dict[str, Position] = {}
        self.fills: List[Fill] = []; self.curve: List[dict] = []

    def on_fill(self, fill):
        self.fills.append(fill)
        pos = self.positions.setdefault(fill.symbol, Position(fill.symbol))
        if fill.side == OrderSide.BUY:
            total = pos.avg_price*pos.quantity + fill.price*fill.quantity
            pos.quantity += fill.quantity
            pos.avg_price = total/pos.quantity if pos.quantity > 0 else 0
            self.cash -= fill.price*fill.quantity + fill.commission
        else:
            pos.realized_pnl += (fill.price - pos.avg_price)*fill.quantity
            pos.quantity -= fill.quantity
            self.cash += fill.price*fill.quantity - fill.commission
        eq = self.cash + sum(p.quantity*p.avg_price for p in self.positions.values() if p.quantity > 0)
        self.curve.append({'ts': fill.timestamp, 'equity': eq})

class Executor:
    def __init__(self, comm=0.0003, slip=0.001):
        self.comm = comm; self.slip = slip; self.prices: Dict[str, float] = {}

    def execute(self, symbol, side, quantity, timestamp):
        bp = self.prices.get(symbol)
        if bp is None: return None
        ep = bp * (1 + self.slip if side == OrderSide.BUY else 1 - self.slip)
        tv = ep * quantity; comm = max(tv*self.comm, 5.0)
        if side == OrderSide.SELL: comm += tv*0.001
        return Fill(symbol, timestamp, side, quantity, ep, comm, abs(ep-bp))

class Engine:
    def __init__(self, capital=1_000_000, comm=0.0003, slip=0.001):
        self.portfolio = Portfolio(capital)
        self.executor = Executor(comm, slip)
        self.strategies = []; self.history: Dict[str, List[float]] = defaultdict(list)
        self._counter = 0

    def add_strategy(self, s): s.engine = self; self.strategies.append(s)

    def get_history(self, sym, n):
        h = self.history.get(sym, [])
        return pd.DataFrame({'close': h[-n:]}) if len(h) >= n else pd.DataFrame()

    def process_signal(self, sig):
        if sig.direction == 0: return
        cp = self.executor.prices.get(sig.symbol)
        if cp is None: return
        if sig.direction > 0:
            qty = max(int(self.portfolio.cash*0.95/cp/100)*100, 0)
            side = OrderSide.BUY
        else:
            pos = self.portfolio.positions.get(sig.symbol)
            qty = pos.quantity if pos else 0
            side = OrderSide.SELL
        if qty <= 0: return
        self._counter += 1
        fill = self.executor.execute(sig.symbol, side, qty, sig.timestamp)
        if fill:
            self.portfolio.on_fill(fill)
            for s in self.strategies: s.on_fill(fill)

    def run(self, data):
        bars = []
        for sym, df in data.items():
            for ts, row in df.iterrows():
                bars.append(Bar(sym, ts, row['open'], row['high'], row['low'], row['close'], row['volume']))
        bars.sort(key=lambda x: (x.timestamp, x.symbol))
        for s in self.strategies: s.on_init()
        for bar in bars:
            self.executor.prices[bar.symbol] = bar.close
            self.history[bar.symbol].append(bar.close)
            for s in self.strategies: s.on_bar(bar)
        return self._report()

    def _report(self):
        eq = pd.DataFrame(self.portfolio.curve)
        if eq.empty: return eq
        eq['ret'] = eq['equity'].pct_change()
        tr = eq['equity'].iloc[-1]/self.portfolio.initial - 1
        n = len(eq); ar = (1+tr)**(252/max(n,1))-1
        sh = eq['ret'].mean()/eq['ret'].std()*np.sqrt(252) if eq['ret'].std()>0 else 0
        pk = np.maximum.accumulate(eq['equity'].values)
        dd = float(np.min((eq['equity'].values-pk)/pk))
        tc = sum(f.commission for f in self.portfolio.fills)
        print(f"收益:{tr:.2%} 年化:{ar:.2%} 夏普:{sh:.3f} 回撤:{dd:.2%} 手续费:{tc:,.0f} 交易:{len(self.portfolio.fills)}")
        return eq

class Strategy:
    def __init__(self): self.engine = None
    def on_init(self): pass
    def on_bar(self, bar): pass
    def on_fill(self, fill): pass
    def get_history(self, sym, n): return self.engine.get_history(sym, n) if self.engine else pd.DataFrame()
    def submit_signal(self, sig):
        if self.engine: self.engine.process_signal(sig)

class DualMA(Strategy):
    def __init__(self, sym, fast=5, slow=20):
        super().__init__(); self.sym = sym; self.fast = fast; self.slow = slow; self.held = False
    def on_bar(self, bar):
        if bar.symbol != self.sym: return
        h = self.get_history(self.sym, self.slow+1)
        if len(h) < self.slow+1: return
        fs = h['close'].iloc[-self.fast:].mean(); sl = h['close'].iloc[-self.slow:].mean()
        pfs = h['close'].iloc[-self.fast-1:-1].mean(); psl = h['close'].iloc[-self.slow-1:-1].mean()
        if fs > sl and pfs <= psl and not self.held:
            self.submit_signal(Signal(self.sym, bar.timestamp, 1)); self.held = True
        elif fs < sl and pfs >= psl and self.held:
            self.submit_signal(Signal(self.sym, bar.timestamp, -1)); self.held = False

data = {'SH.600000': gen_ohlcv(500, 0.0004, 42), 'SH.600036': gen_ohlcv(500, 0.0002, 43)}
engine = Engine(1_000_000)
engine.add_strategy(DualMA('SH.600000', 5, 20))
engine.add_strategy(DualMA('SH.600036', 10, 30))
engine.run(data)
```

## 关键差异

| 粗糙回测 | 事件驱动 |
|----------|----------|
| 用当前价格成交 | 用下一根K线开盘价 |
| 无手续费 | 佣金+印花税+过户费 |
| 无滑点 | 0.1% 执行偏差 |
| 单标的 | 多标的支持 |

    rate = 0.0003 if tv<50000 else (0.00025 if tv<500000 else 0.0002)
    comm = max(tv*rate, 5.0)
    if side == OrderSide.SELL: comm += tv*0.001
