> 前五课学了回测引擎、回测陷阱、风险度量、仓位管理、止损策略。现在把它们整合成一个完整的系统——从信号到回测到风控报告的一体化流水线。

## 场景引入

你已经有了一个经过严格回测的多因子选股策略。但回测脚本是散落的几个文件：一个算信号，一个算收益，一个手动算夏普比率。每次改参数都要手动跑三个脚本，然后把结果复制到 Excel 里拼报告。

你想要一个一体化的系统：**一键运行回测 → 自动计算风险指标 → 生成风控报告**。这不仅是为了方便，更是为了减少手动操作中的错误——回测中最危险的 bug 往往不是逻辑错误，而是数据对齐错误、前视偏差这类"胶水代码"中的问题。

本课的目标就是构建这样一个系统。

### 三、事件驱动 vs 向量化回测

```
事件驱动回测流程（本课采用）：

  时间线 ──────────────────────────────────────────────►

  T1: 加载行情 → 策略计算信号 → 风控检查 → 仓位计算 → 下单 → 成交 → 记录
  T2: 加载行情 → 策略计算信号 → 风控检查 → 仓位计算 → 下单 → 成交 → 记录
  T3: 加载行情 → 止损检查 → 触发止损 → 成交 → 记录
  ...

  每个时间步独立处理，严格模拟真实交易流程
  优点：逻辑严谨、接近实盘、易于扩展
  缺点：速度慢、代码复杂

  适用场景：
  - 最终策略验证（严格模拟）
  - 需要精确计算手续费和滑点
  - 多标的组合管理
```

```python
"""
完整的回测与风控系统
整合信号生成、回测引擎、风险度量、仓位管理、止损策略
"""

import numpy as np
import pandas as pd
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Tuple
from io import BytesIO
import base64


class RiskMetrics:
    """风险指标计算器"""

    @staticmethod
    def sharpe_ratio(returns: pd.Series, rf: float = 0.03) -> float:
        """年化夏普比率"""
        excess = returns - rf / 252
        return float(excess.mean() / excess.std() * np.sqrt(252)) if excess.std() > 0 else 0

    @staticmethod
    def sortino_ratio(returns: pd.Series, rf: float = 0.03) -> float:
        """Sortino 比率：只惩罚下行波动"""
        excess = returns - rf / 252
        downside = returns[returns < rf/252] - rf/252
        ds_std = np.sqrt((downside**2).mean())
        return float(excess.mean() / ds_std * np.sqrt(252)) if ds_std > 0 else 0

    @staticmethod
    def max_drawdown(equity: pd.Series) -> float:
        """最大回撤"""
        peak = equity.cummax()
        dd = (equity - peak) / peak
        return float(dd.min())

    @staticmethod
    def calmar_ratio(returns: pd.Series) -> float:
        """Calmar 比率：年化收益 / 最大回撤"""
        annual_ret = (1 + returns.mean())**252 - 1
        mdd = abs(RiskMetrics.max_drawdown((1 + returns).cumprod()))
        return float(annual_ret / mdd) if mdd > 0 else 0

    @staticmethod
    def var(returns: pd.Series, confidence: float = 0.95) -> float:
        """VaR（在险价值）"""
        return float(np.percentile(returns, (1 - confidence) * 100))

    @staticmethod
    def cvar(returns: pd.Series, confidence: float = 0.95) -> float:
        """CVaR（条件在险价值）：尾部平均亏损"""
        var = RiskMetrics.var(returns, confidence)
        tail = returns[returns <= var]
        return float(tail.mean()) if len(tail) > 0 else var

    @staticmethod
    def win_rate(trades: List[dict]) -> float:
        """胜率"""
        if not trades:
            return 0
        wins = sum(1 for t in trades if t.get('pnl', 0) > 0)
        return wins / len(trades)

    @staticmethod
    def profit_loss_ratio(trades: List[dict]) -> float:
        """盈亏比"""
        wins = [t['pnl'] for t in trades if t.get('pnl', 0) > 0]
        losses = [abs(t['pnl']) for t in trades if t.get('pnl', 0) < 0]
        if not wins or not losses:
            return 0
        return np.mean(wins) / np.mean(losses)
```

### 二、仓位管理模块

```python
class PositionSizer:
    """仓位管理"""

    @staticmethod
    def equal_weight(n_assets: int) -> np.ndarray:
        """等权重分配"""
        return np.ones(n_assets) / n_assets

    @staticmethod
    def kelly_adjusted(win_rate: float, avg_win: float, avg_loss: float,
                       fraction: float = 0.5) -> float:
        """半 Kelly 仓位"""
        if avg_loss == 0:
            return 0
        b = avg_win / avg_loss
        kelly = (win_rate * b - (1 - win_rate)) / b
        return max(kelly * fraction, 0)
```

### 三、止损模块

```python
class StopLossManager:
    """止损管理"""

    def __init__(self, atr_period: int = 14):
        self.atr_period = atr_period

    def calculate_atr(self, high: pd.Series, low: pd.Series,
                      close: pd.Series) -> pd.Series:
        """计算 ATR"""
        prev_close = close.shift(1)
        tr = pd.concat([
            high - low,
            (high - prev_close).abs(),
            (low - prev_close).abs()
        ], axis=1).max(axis=1)
        return tr.rolling(self.atr_period).mean()

    def trailing_stop_check(self, current_price: float,
                            highest_price: float, trail_pct: float) -> bool:
        """检查跟踪止损是否触发"""
        stop_line = highest_price * (1 - trail_pct)
        return current_price <= stop_line

    def atr_stop_check(self, current_price: float, entry_price: float,
                       atr_value: float, multiplier: float = 2.0) -> bool:
        """检查 ATR 止损是否触发"""
        stop_line = entry_price - multiplier * atr_value
        return current_price <= stop_line
```

### 四、月度归因分析

```python
def monthly_attribution(equity: pd.Series, trades: List[dict]) -> pd.DataFrame:
    """月度收益归因分析

    将每月收益分解为：
    - 持仓收益：已持仓资产的价格变动
    - 交易收益：本月平仓的盈亏
    - 现金收益：未投资资金的利息
    """
    monthly_returns = equity.resample('ME').last().pct_change().dropna()

    # 按月统计交易
    trades_by_month = {}
    for t in trades:
        if 'day' in t:
            month = equity.index[min(t['day'], len(equity)-1)]
            month_key = month.strftime('%Y-%m')
            if month_key not in trades_by_month:
                trades_by_month[month_key] = []
            trades_by_month[month_key].append(t)

    result = []
    for month_end, ret in monthly_returns.items():
        month_key = month_end.strftime('%Y-%m')
        month_trades = trades_by_month.get(month_key, [])
        trade_pnl = sum(t.get('pnl', 0) for t in month_trades)
        n_trades = len(month_trades)
        n_wins = sum(1 for t in month_trades if t.get('pnl', 0) > 0)

        result.append({
            '月份': month_key,
            '月收益率': ret,
            '交易次数': n_trades,
            '盈利次数': n_wins,
            '交易盈亏': trade_pnl,
        })

    return pd.DataFrame(result)
```

### 五、完整回测系统（事件驱动）

```python
class BacktestSystem:
    """完整的回测与风控系统（事件驱动）"""

    def __init__(self, config: dict):
        self.config = config
        self.risk = RiskMetrics()
        self.sizer = PositionSizer()
        self.stop_mgr = StopLossManager()

    def run(self, signals: pd.DataFrame, prices: pd.DataFrame,
            high: pd.DataFrame = None, low: pd.DataFrame = None) -> dict:
        """运行回测

        参数:
            signals: 信号矩阵 (n_days, n_assets)，1=做多, 0=空仓
            prices: 价格矩阵 (n_days, n_assets)
            high: 最高价（可选，用于 ATR 止损）
            low: 最低价（可选，用于 ATR 止损）
        """
        initial_capital = self.config.get('initial_capital', 1_000_000)
        commission = self.config.get('commission', 0.0003)
        slippage = self.config.get('slippage', 0.001)
        position_method = self.config.get('position_method', 'equal')
        stop_method = self.config.get('stop_method', None)
        stop_params = self.config.get('stop_params', {})
        rebalance_days = self.config.get('rebalance_days', 5)

        n_days, n_assets = prices.shape
        symbols = prices.columns.tolist()

        # 计算 ATR（如果需要）
        atr_data = {}
        if high is not None and low is not None:
            for s in symbols:
                atr_data[s] = self.stop_mgr.calculate_atr(high[s], low[s], prices[s])

        # 状态变量
        cash = initial_capital
        positions = {s: 0 for s in symbols}
        entry_prices = {s: 0.0 for s in symbols}
        highest_prices = {s: 0.0 for s in symbols}
        trades = []
        equity_curve = []
        last_rebalance = 0

        for t in range(60, n_days):
            current_prices = prices.iloc[t]
            target_signals = signals.iloc[t]

            # 止损检查（优先于调仓）
            if stop_method == 'trailing':
                for s in symbols:
                    if positions[s] > 0:
                        if current_prices[s] > highest_prices[s]:
                            highest_prices[s] = current_prices[s]
                        trail_pct = stop_params.get('trail_pct', 0.05)
                        if self.stop_mgr.trailing_stop_check(
                                current_prices[s], highest_prices[s], trail_pct):
                            proceeds = positions[s] * current_prices[s] * (1 - slippage - commission)
                            pnl = (current_prices[s] - entry_prices[s]) / entry_prices[s] if entry_prices[s] > 0 else 0
                            cash += proceeds
                            trades.append({'symbol': s, 'pnl': pnl, 'day': t, 'reason': '跟踪止损'})
                            positions[s] = 0

            elif stop_method == 'atr':
                for s in symbols:
                    if positions[s] > 0 and not np.isnan(atr_data.get(s, pd.Series()).iloc[t] if s in atr_data else np.nan):
                        atr_val = atr_data[s].iloc[t]
                        mult = stop_params.get('atr_multiple', 2.0)
                        if self.stop_mgr.atr_stop_check(current_prices[s], entry_prices[s], atr_val, mult):
                            proceeds = positions[s] * current_prices[s] * (1 - slippage - commission)
                            pnl = (current_prices[s] - entry_prices[s]) / entry_prices[s] if entry_prices[s] > 0 else 0
                            cash += proceeds
                            trades.append({'symbol': s, 'pnl': pnl, 'day': t, 'reason': 'ATR止损'})
                            positions[s] = 0

            # 再平衡检查
            if t - last_rebalance >= rebalance_days:
                last_rebalance = t

                # 计算目标权重
                active_count = target_signals.sum()
                if active_count > 0:
                    target_weights = target_signals / active_count
                else:
                    target_weights = pd.Series(0, index=symbols)

                # 计算当前总资产
                total_equity = cash + sum(
                    positions[ss] * current_prices[ss] for ss in symbols
                )

                # 执行调仓
                for s in symbols:
                    target_value = target_weights[s] * total_equity
                    current_value = positions[s] * current_prices[s]
                    diff = target_value - current_value

                    if abs(diff) > current_prices[s] * 100:
                        if diff > 0:
                            shares = int(diff / current_prices[s] / 100) * 100
                            cost = shares * current_prices[s] * (1 + slippage + commission)
                            if cost <= cash and shares > 0:
                                cash -= cost
                                positions[s] += shares
                                entry_prices[s] = current_prices[s]
                                highest_prices[s] = current_prices[s]
                        else:
                            shares = min(int(-diff / current_prices[s] / 100) * 100, positions[s])
                            if shares > 0:
                                proceeds = shares * current_prices[s] * (1 - slippage - commission)
                                pnl = (current_prices[s] - entry_prices[s]) / entry_prices[s] if entry_prices[s] > 0 else 0
                                cash += proceeds
                                positions[s] -= shares
                                trades.append({'symbol': s, 'pnl': pnl, 'day': t})

            # 记录权益
            total_equity = cash + sum(positions[s] * current_prices[s] for s in symbols)
            equity_curve.append({'day': t, 'equity': total_equity, 'cash': cash})

        # 生成报告
        equity_df = pd.DataFrame(equity_curve).set_index('day')
        equity_series = equity_df['equity']
        returns = equity_series.pct_change().dropna()

        # 月度归因
        monthly = monthly_attribution(equity_series, trades)

        report = {
            '--- 收益指标 ---': '',
            '总收益率': f"{(equity_series.iloc[-1] / initial_capital - 1):.2%}",
            '年化收益率': f"{(1 + returns.mean())**252 - 1:.2%}",
            '年化波动率': f"{returns.std() * np.sqrt(252):.2%}",
            '--- 风险指标 ---': '',
            '夏普比率': f"{self.risk.sharpe_ratio(returns):.3f}",
            'Sortino 比率': f"{self.risk.sortino_ratio(returns):.3f}",
            'Calmar 比率': f"{self.risk.calmar_ratio(returns):.3f}",
            '最大回撤': f"{self.risk.max_drawdown(equity_series):.2%}",
            'VaR(95%)': f"{self.risk.var(returns):.4%}",
            'CVaR(95%)': f"{self.risk.cvar(returns):.4%}",
            '--- 交易统计 ---': '',
            '交易次数': len(trades),
            '胜率': f"{self.risk.win_rate(trades):.2%}",
            '盈亏比': f"{self.risk.profit_loss_ratio(trades):.2f}",
            '--- 配置 ---': '',
            '仓位管理': position_method,
            '止损方法': stop_method or '无',
            '再平衡周期': f"{rebalance_days} 天",
        }

        return {
            'report': report,
            'equity': equity_series,
            'returns': returns,
            'trades': trades,
            'monthly': monthly,
            'config': self.config,
        }
```

```python
def generate_html_report(result: dict, output_path: str = 'backtest_report.html'):
```

```python
def run_full_system_demo():
```

## 常见误区

### 误区一：只看收益不看风险

回测报告中最不重要的指标是总收益率。最重要的是夏普比率、最大回撤和 CVaR——它们告诉你承担了多少风险才赚到这些收益。年化 30% 但最大回撤 60% 的策略，远不如年化 15% 但最大回撤 10% 的策略。

### 误区二：把回测当实盘

回测是理想化的：没有执行延迟、没有部分成交、没有流动性限制。实盘表现通常比回测差 30-50%。如果回测年化 30%，对实盘的合理预期是 15-20%。

### 误区三：不做参数稳健性检验

找到一组"最优参数"就直接用。应该测试参数微调后策略是否仍然有效——如果均线从 20 改为 19 策略就失效，那这组参数大概率是过拟合的。

### 误区四：忽略交易成本的影响

手续费和滑点看似微小（每次 0.3-0.5%），但对高频策略的影响巨大。假设每天交易一次，年交易成本约 75-125%。如果策略年化收益低于这个数，实盘必亏。

### 误区五：回测系统不做单元测试

回测系统本身的 bug 比策略 bug 更难发现。数据对齐错误、前视偏差、资金计算错误——这些 bug 不会让程序崩溃，只会给你一个"看起来合理但完全错误"的回测结果。**每个模块都应该有独立的单元测试。**

## 小结与练习

### 小结

1. 完整的回测系统包含策略层、风控层、执行层、分析层四个层次
2. 风险报告应包含收益指标、风险指标、交易统计、月度归因四个维度
3. 参数稳健性检验是防止过拟合的最后一道防线
4. 回测结果是实盘的上限估计，实际表现通常会打折
5. 从"回测脚本"到"回测系统"的关键差异是模块化、可测试性和可替换性

```python
# ATR 跟踪止损
elif stop_method == 'atr_trailing':
    for s in symbols:
        if positions[s] > 0:
            if current_prices[s] > highest_prices[s]:
                highest_prices[s] = current_prices[s]
            if not np.isnan(atr_data[s].iloc[t]):
                atr_val = atr_data[s].iloc[t]
                mult = stop_params.get('atr_multiple', 2.0)
                # 跟踪止损线 = 最高价 - N × ATR
                atr_trail_stop = highest_prices[s] - mult * atr_val
                if current_prices[s] <= atr_trail_stop:
                    proceeds = positions[s] * current_prices[s] * (1 - slippage - commission)
                    pnl = (current_prices[s] - entry_prices[s]) / entry_prices[s] if entry_prices[s] > 0 else 0
                    cash += proceeds
                    trades.append({'symbol': s, 'pnl': pnl, 'day': t, 'reason': 'ATR跟踪止损'})
                    positions[s] = 0

# 时间止损
elif stop_method == 'time':
    max_days = stop_params.get('max_holding_days', 20)
    for s in symbols:
        if positions[s] > 0:
            # 简化处理：记录入场日期
            if entry_prices[s] > 0 and current_prices[s] <= entry_prices[s]:
                # 检查持仓天数（这里用 equity_curve 长度近似）
                holding_days = t - last_rebalance
                if holding_days >= max_days:
                    proceeds = positions[s] * current_prices[s] * (1 - slippage - commission)
                    pnl = (current_prices[s] - entry_prices[s]) / entry_prices[s]
                    cash += proceeds
                    trades.append({'symbol': s, 'pnl': pnl, 'day': t, 'reason': '时间止损'})
                    positions[s] = 0
```

**要点**：
- ATR 跟踪止损结合了跟踪止损的"锁定利润"和 ATR 止损的"适应波动率"
- 时间止损适合趋势策略——趋势启动后应快速盈利，横盘说明判断错误
- 实际系统中需要为每个持仓独立记录入场时间

### 练习二

**思路**：遍历不同再平衡周期，观察夏普比率的变化趋势。

**答案**：

```python
def rebalance_sensitivity_analysis():
    """再平衡周期敏感性分析"""
    np.random.seed(42)
    n_days = 750
    dates = pd.date_range('2022-01-01', periods=n_days, freq='B')
    n_assets = 5
    asset_names = [f'Stock_{chr(65+i)}' for i in range(n_assets)]

    returns_data = np.random.normal(0.0003, 0.02, (n_days, n_assets))
    prices_data = 100 * np.cumprod(1 + returns_data, axis=0)
    prices = pd.DataFrame(prices_data, index=dates, columns=asset_names)

    signals = pd.DataFrame(0, index=dates, columns=asset_names)
    for t in range(60, n_days):
        momentum = prices.iloc[t] / prices.iloc[t-20] - 1
        top2 = momentum.nlargest(2).index
        signals.loc[dates[t], top2] = 1

    rebalance_periods = range(3, 16, 2)

    print("Rebalance Period Sensitivity Analysis:")
    print(f"{'Period':>8} {'Return':>10} {'Sharpe':>8} {'MaxDD':>10} {'Trades':>8}")
    print("-" * 50)

    sharpe_values = []
    for period in rebalance_periods:
        config = {
            'position_method': 'equal',
            'stop_method': None,
            'rebalance_days': period
        }
        system = BacktestSystem(config)
        result = system.run(signals, prices)

        report = result['report']
        total_ret = float(report['总收益率'].strip('%')) / 100
        sharpe = float(report['夏普比率'])
        max_dd = float(report['最大回撤'].strip('%')) / 100
        trades_count = report['交易次数']

        sharpe_values.append(sharpe)
        print(f"{period:>8d} {total_ret:>10.2%} {sharpe:>8.3f} {max_dd:>10.2%} {trades_count:>8d}")

    # 计算夏普比率的变异系数
    sharpe_std = np.std(sharpe_values)
    sharpe_mean = np.mean(sharpe_values)
    cv = sharpe_std / abs(sharpe_mean) if sharpe_mean != 0 else float('inf')

    print(f"\nSharpe Ratio Statistics:")
    print(f"  Mean: {sharpe_mean:.3f}")
    print(f"  Std:  {sharpe_std:.3f}")
    print(f"  CV:   {cv:.2%}")

    if cv > 0.3:
        print("\nWarning: High sensitivity! Strategy may be overfit to rebalance period.")
    else:
        print("\nGood: Strategy is robust across different rebalance periods.")

rebalance_sensitivity_analysis()
```

**要点**：
- 如果夏普比率的变异系数（CV）超过 30%，说明策略对再平衡周期敏感
- 稳健的策略应该在不同参数下都有可接受的表现
- 不要选择"最优"参数，而要选择"稳健区间"的中值
