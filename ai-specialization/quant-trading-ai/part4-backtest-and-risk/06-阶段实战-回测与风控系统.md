## 一个"太好"的回测

年化 35%，夏普 2.8，最大回撤 8%。好得不正常。加风控来检验。

```python
import numpy as np
import pandas as pd
from dataclasses import dataclass

def gen_data(n=750, assets=5, seed=42):
    np.random.seed(seed)
    dates = pd.bdate_range('2022-01-01', periods=n)
    names = [f'Stock_{chr(65+i)}' for i in range(assets)]
    ret = np.random.normal(0.0003, 0.02, (n, assets))
    prices = pd.DataFrame(100*np.cumprod(1+ret, axis=0), index=dates, columns=names)
    high = prices * (1+np.abs(np.random.normal(0,0.008,(n,assets))))
    low = prices * (1-np.abs(np.random.normal(0,0.008,(n,assets))))
    signals = pd.DataFrame(0, index=dates, columns=names)
    for t in range(60, n):
        mom = prices.iloc[t] / prices.iloc[t-20] - 1
        signals.loc[dates[t], mom.nlargest(2).index] = 1
    return prices, high, low, signals

prices, high, low, signals = gen_data()
```

## 风险指标

```python
class Risk:
    @staticmethod
    def sharpe(r, rf=0.03):
        e = r - rf/252
        return float(e.mean()/e.std()*np.sqrt(252)) if e.std()>0 else 0
    @staticmethod
    def sortino(r, rf=0.03):
        e = r - rf/252; ds = r[r<rf/252] - rf/252
        return float(e.mean()/np.sqrt((ds**2).mean())*np.sqrt(252)) if len(ds)>0 else 0
    @staticmethod
    def max_dd(eq):
        return float(((eq - eq.cummax())/eq.cummax()).min())
    @staticmethod
    def calmar(r):
        ar = (1+r.mean())**252-1; mdd = abs(Risk.max_dd((1+r).cumprod()))
        return float(ar/mdd) if mdd>0 else 0
    @staticmethod
    def var(r, c=0.95): return float(np.percentile(r, (1-c)*100))
    @staticmethod
    def cvar(r, c=0.95):
        v = Risk.var(r, c); t = r[r<=v]
        return float(t.mean()) if len(t)>0 else v
    @staticmethod
    def win_rate(trades):
        return sum(1 for t in trades if t.get('pnl',0)>0)/len(trades) if trades else 0
```

## 止损模块

```python
class StopLoss:
    @staticmethod
    def trailing(price, highest, pct): return price <= highest*(1-pct)
    @staticmethod
    def atr(high, low, close, period=14):
        tr = pd.concat([high-low, (high-close.shift(1)).abs(), (low-close.shift(1)).abs()], axis=1).max(axis=1)
        return tr.rolling(period).mean()
```

## 完整回测系统

```python
class BacktestSystem:
    def __init__(self, config):
        self.config = config

    def run(self, signals, prices, high=None, low=None):
        cfg = self.config
        cap = cfg.get('capital', 1_000_000)
        comm = cfg.get('commission', 0.0003)
        slip = cfg.get('slippage', 0.001)
        stop = cfg.get('stop', None)
        sp = cfg.get('stop_params', {})
        freq = cfg.get('rebalance', 5)
        syms = prices.columns.tolist()
        n_days = len(prices)

        atr = {}
        if high is not None and low is not None:
            for s in syms: atr[s] = StopLoss.atr(high[s], low[s], prices[s])

        cash = cap; pos = {s:0 for s in syms}; entry = {s:0.0 for s in syms}
        highest = {s:0.0 for s in syms}; trades = []; curve = []; last_reb = 0

        for t in range(60, n_days):
            cp = prices.iloc[t]; sig = signals.iloc[t]

            # 止损
            if stop == 'trailing':
                for s in syms:
                    if pos[s] > 0:
                        highest[s] = max(highest[s], cp[s])
                        if StopLoss.trailing(cp[s], highest[s], sp.get('pct', 0.05)):
                            pnl = (cp[s]-entry[s])/entry[s] if entry[s]>0 else 0
                            cash += pos[s]*cp[s]*(1-slip-comm); trades.append({'pnl':pnl,'day':t})
                            pos[s] = 0
            elif stop == 'atr':
                for s in syms:
                    if pos[s] > 0 and s in atr and not np.isnan(atr[s].iloc[t]):
                        if cp[s] <= entry[s] - sp.get('mult',2.0)*atr[s].iloc[t]:
                            pnl = (cp[s]-entry[s])/entry[s] if entry[s]>0 else 0
                            cash += pos[s]*cp[s]*(1-slip-comm); trades.append({'pnl':pnl,'day':t})
                            pos[s] = 0

            # 再平衡
            if t - last_reb >= freq:
                last_reb = t
                active = sig.sum()
                tw = sig/active if active > 0 else pd.Series(0, index=syms)
                eq = cash + sum(pos[s]*cp[s] for s in syms)
                for s in syms:
                    diff = tw[s]*eq - pos[s]*cp[s]
                    if abs(diff) > cp[s]*100:
                        if diff > 0:
                            shares = int(diff/cp[s]/100)*100
                            cost = shares*cp[s]*(1+slip+comm)
                            if cost <= cash and shares > 0:
                                cash -= cost; pos[s] += shares; entry[s] = cp[s]; highest[s] = cp[s]
                        else:
                            shares = min(int(-diff/cp[s]/100)*100, pos[s])
                            if shares > 0:
                                pnl = (cp[s]-entry[s])/entry[s] if entry[s]>0 else 0
                                cash += shares*cp[s]*(1-slip-comm); pos[s] -= shares
                                trades.append({'pnl':pnl,'day':t})

            eq = cash + sum(pos[s]*cp[s] for s in syms)
            curve.append({'day':t, 'equity':eq})

        eq = pd.DataFrame(curve).set_index('day')['equity']
        ret = eq.pct_change().dropna()
        r = Risk
        report = {
            '总收益': f"{(eq.iloc[-1]/cap-1):.2%}",
            '年化': f"{(1+ret.mean())**252-1:.2%}",
            '夏普': f"{r.sharpe(ret):.3f}",
            'Sortino': f"{r.sortino(ret):.3f}",
            'Calmar': f"{r.calmar(ret):.3f}",
            '最大回撤': f"{r.max_dd(eq):.2%}",
            'VaR': f"{r.var(ret):.4%}",
            'CVaR': f"{r.cvar(ret):.4%}",
            '交易': len(trades),
            '胜率': f"{r.win_rate(trades):.2%}",
            '止损': stop or '无',
        }
        return report

configs = [
    ('无止损', {}),
    ('跟踪止损8%', {'stop':'trailing','stop_params':{'pct':0.08}}),
    ('ATR止损2.5x', {'stop':'atr','stop_params':{'mult':2.5}}),
]
for name, cfg in configs:
    r = BacktestSystem(cfg).run(signals, prices, high, low)
    print(f"\n{name}:")
    for k,v in r.items(): print(f"  {k}: {v}")
```

## 参数稳健性

```python
sharpes = []
for p in range(3, 16, 2):
    r = BacktestSystem({'rebalance': p}).run(signals, prices)
    sharpes.append(float(r['夏普']))
cv = np.std(sharpes)/abs(np.mean(sharpes)) if np.mean(sharpes)!=0 else float('inf')
print(f"夏普变异系数: {cv:.2%} ({'敏感' if cv>0.3 else '稳健'})")
```
