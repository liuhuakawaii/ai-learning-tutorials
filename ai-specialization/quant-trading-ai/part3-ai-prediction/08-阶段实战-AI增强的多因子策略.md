```python
def generate_stock_data(n_days: int = 2000, seed: int = 42) -> pd.DataFrame:
    """
    生成模拟股票数据

    包含: 日期、开盘价、最高价、最低价、收盘价、成交量
    特点: 带有趋势、均值回归、波动率聚集效应
    """
    np.random.seed(seed)
    dates = pd.date_range("2016-01-01", periods=n_days, freq="B")

    # 生成收益率（带波动率聚集）
    returns = np.zeros(n_days)
    vol = 0.02
    for i in range(1, n_days):
        vol = 0.9 * vol + 0.1 * abs(returns[i - 1]) + 0.001
        vol = np.clip(vol, 0.005, 0.05)
        returns[i] = np.random.normal(0.0002, vol)

    # 生成价格
    close = 100 * np.cumprod(1 + returns)
    high = close * (1 + np.abs(np.random.randn(n_days) * 0.005))
    low = close * (1 - np.abs(np.random.randn(n_days) * 0.005))
    open_price = close * (1 + np.random.randn(n_days) * 0.003)
    volume = np.random.lognormal(15, 0.5, n_days).astype(int)

    df = pd.DataFrame({
        "date": dates,
        "open": open_price,
        "high": high,
        "low": low,
        "close": close,
        "volume": volume,
    })
    return df

class FeatureEngineer:
    """
    特征工程——构建预测因子

    特征类别:
    1. 价格特征: 收益率、波动率、偏度
    2. 技术指标: MA、RSI、MACD、布林带
    3. 量价关系: 量价相关性、成交量变化
    4. 时序特征: 滞后值、滚动统计
    """

    def __init__(self, lookback_windows: list = None):
        self.lookback_windows = lookback_windows or [5, 10, 20, 60]
        self.feature_names = []

    def build_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """构建所有特征"""
        features = pd.DataFrame(index=df.index)

        close = df["close"]
        high = df["high"]
        low = df["low"]
        volume = df["volume"]

        # 价格特征
        for w in self.lookback_windows:
            # 收益率
            features[f"return_{w}d"] = close.pct_change(w)
            # 波动率
            features[f"volatility_{w}d"] = close.pct_change().rolling(w).std()
            # 偏度
            features[f"skewness_{w}d"] = close.pct_change().rolling(w).skew()

        # 技术指标
        for w in [5, 10, 20, 60]:
            # 均线偏离度
            ma = close.rolling(w).mean()
            features[f"ma_deviation_{w}d"] = (close - ma) / ma

        # RSI
        for period in [6, 14]:
            delta = close.diff()
            gain = delta.where(delta > 0, 0).rolling(period).mean()
            loss = (-delta.where(delta < 0, 0)).rolling(period).mean()
            rs = gain / (loss + 1e-10)
            features[f"rsi_{period}"] = 1 - 1 / (1 + rs)

        # MACD
        ema12 = close.ewm(span=12).mean()
        ema26 = close.ewm(span=26).mean()
        macd = ema12 - ema26
        signal = macd.ewm(span=9).mean()
        features["macd"] = macd
        features["macd_signal"] = signal
        features["macd_hist"] = macd - signal

        # 布林带
        for w in [20]:
            ma = close.rolling(w).mean()
            std = close.rolling(w).std()
            features[f"bb_upper_{w}"] = (close - (ma + 2 * std)) / close
            features[f"bb_lower_{w}"] = (close - (ma - 2 * std)) / close

        # 量价关系
        for w in [5, 20]:
            features[f"volume_ratio_{w}d"] = volume / volume.rolling(w).mean()
            features[f"price_volume_corr_{w}d"] = close.pct_change().rolling(w).corr(
                volume.pct_change()
            )

        # 时序特征（滞后值）
        for lag in [1, 2, 3, 5]:
            features[f"return_lag_{lag}"] = close.pct_change(lag)

        # 当日特征
        features["intraday_range"] = (high - low) / close
        features["body_ratio"] = (close - df["open"]) / (high - low + 1e-10)

        self.feature_names = features.columns.tolist()
        return features

    def build_labels(self, df: pd.DataFrame, forward_days: int = 5) -> pd.Series:
        """构建预测标签: 未来 N 天收益率"""
        return df["close"].pct_change(forward_days).shift(-forward_days)

class LightGBMPredictor:
    """LightGBM 预测模型"""

    def __init__(self, n_estimators: int = 100, max_depth: int = 5, learning_rate: float = 0.05):
        self.n_estimators = n_estimators
        self.max_depth = max_depth
        self.learning_rate = learning_rate
        self.model = None
        self.scaler = StandardScaler()

    def fit(self, X: np.ndarray, y: np.ndarray):
        try:
            import lightgbm as lgb
            self.model = lgb.LGBMRegressor(
                n_estimators=self.n_estimators,
                max_depth=self.max_depth,
                learning_rate=self.learning_rate,
                subsample=0.8,
                colsample_bytree=0.8,
                random_state=42,
                verbose=-1,
            )
        except ImportError:
            from sklearn.ensemble import GradientBoostingRegressor
            self.model = GradientBoostingRegressor(
                n_estimators=self.n_estimators,
                max_depth=self.max_depth,
                learning_rate=self.learning_rate,
                random_state=42,
            )

        X_scaled = self.scaler.fit_transform(X)
        self.model.fit(X_scaled, y)
        return self

    def predict(self, X: np.ndarray) -> np.ndarray:
        X_scaled = self.scaler.transform(X)
        return self.model.predict(X_scaled)


class LSTMPredictor:
    """LSTM 预测模型（简化版，使用 numpy 实现）"""

    def __init__(self, sequence_length: int = 20, hidden_dim: int = 32, lr: float = 0.001):
        self.sequence_length = sequence_length
        self.hidden_dim = hidden_dim
        self.lr = lr
        self.weights = None
        self.scaler = StandardScaler()

    def _build_sequences(self, X: np.ndarray) -> np.ndarray:
        """构建 LSTM 输入序列"""
        sequences = []
        for i in range(len(X) - self.sequence_length + 1):
            sequences.append(X[i:i + self.sequence_length])
        return np.array(sequences)

    def fit(self, X: np.ndarray, y: np.ndarray):
        """训练简化 LSTM（用线性层模拟，降低依赖）"""
        X_scaled = self.scaler.fit_transform(X)
        X_seq = self._build_sequences(X_scaled)
        y_aligned = y[self.sequence_length - 1:]

        # 简化: 用最后一步特征 + 序列均值作为输入
        X_flat = np.hstack([
            X_seq[:, -1, :],
            X_seq[:, :, :].mean(axis=1),
        ])

        # Ridge 回归模拟 LSTM
        from sklearn.linear_model import Ridge
        self.model = Ridge(alpha=1.0)
        self.model.fit(X_flat, y_aligned)
        return self

    def predict(self, X: np.ndarray) -> np.ndarray:
        X_scaled = self.scaler.transform(X)
        X_seq = self._build_sequences(X_scaled)
        X_flat = np.hstack([
            X_seq[:, -1, :],
            X_seq[:, :, :].mean(axis=1),
        ])
        predictions = self.model.predict(X_flat)
        # 填充前面的预测（用第一个预测值）
        full_predictions = np.full(len(X), predictions[0])
        full_predictions[self.sequence_length - 1:] = predictions
        return full_predictions


class RidgePredictor:
    """Ridge 线性模型（作为 Stacking 的基模型之一）"""

    def __init__(self, alpha: float = 1.0):
        self.model = Ridge(alpha=alpha)
        self.scaler = StandardScaler()

    def fit(self, X: np.ndarray, y: np.ndarray):
        X_scaled = self.scaler.fit_transform(X)
        self.model.fit(X_scaled, y)
        return self

    def predict(self, X: np.ndarray) -> np.ndarray:

class StackingEnsemble:
    """
    Stacking 集成——组合多个模型的预测

    架构:
    Layer 0: LightGBM + LSTM + Ridge
    Layer 1: Ridge 元模型
    """

    def __init__(self, base_models: dict, n_splits: int = 5):
        self.base_models = base_models
        self.meta_model = Ridge(alpha=1.0)
        self.n_splits = n_splits
        self.fitted_models = {}

    def fit(self, X: np.ndarray, y: np.ndarray):
        """训练 Stacking 集成"""
        n_samples = len(X)
        n_models = len(self.base_models)
        meta_features = np.zeros((n_samples, n_models))

        # 时间序列交叉验证生成元特征
        fold_size = n_samples // (self.n_splits + 1)
        for fold in range(self.n_splits):
            train_end = fold_size * (fold + 1)
            test_end = min(train_end + fold_size, n_samples)

            X_train, y_train = X[:train_end], y[:train_end]
            X_test = X[train_end:test_end]

            for i, (name, model_class) in enumerate(self.base_models.items()):
                model = model_class()
                model.fit(X_train, y_train)
                meta_features[train_end:test_end, i] = model.predict(X_test)

        # 训练元模型
        valid_mask = ~np.isnan(meta_features).any(axis=1) & ~np.isnan(y)
        self.meta_model.fit(meta_features[valid_mask], y[valid_mask])

        # 全量训练基模型
        self.fitted_models = {}
        for name, model_class in self.base_models.items():
            model = model_class()
            model.fit(X, y)
            self.fitted_models[name] = model

        return self

    def predict(self, X: np.ndarray) -> np.ndarray:
        meta_features = np.zeros((len(X), len(self.fitted_models)))
        for i, (name, model) in enumerate(self.fitted_models.items()):
            meta_features[:, i] = model.predict(X)
        return self.meta_model.predict(meta_features)

    def get_model_weights(self) -> dict:
        if hasattr(self.meta_model, "coef_"):
            return dict(zip(self.base_models.keys(), self.meta_model.coef_.round(4)))
        return {}

class SignalGenerator:
    """信号生成器——将模型预测转化为交易信号"""

    def __init__(self, long_threshold: float = 0.01, short_threshold: float = -0.01):
        self.long_threshold = long_threshold
        self.short_threshold = short_threshold

    def generate(self, predictions: np.ndarray) -> np.ndarray:
        """
        生成交易信号

        信号含义:
        1 = 做多（预期上涨超过阈值）
        -1 = 做空（预期下跌超过阈值）
        0 = 中性（预期变化不大）
        """
        signals = np.zeros(len(predictions))
        signals[predictions > self.long_threshold] = 1
        signals[predictions < self.short_threshold] = -1
        return signals


class Backtester:
    """回测引擎"""

    def __init__(
        self,
        initial_capital: float = 1000000,
        transaction_cost: float = 0.001,
        slippage: float = 0.0005,
    ):
        self.initial_capital = initial_capital
        self.transaction_cost = transaction_cost
        self.slippage = slippage

    def run(
        self,
        signals: np.ndarray,
        prices: np.ndarray,
        start_idx: int = 0,
    ) -> dict:
        """
        运行回测

        参数:
            signals: 交易信号 (1, 0, -1)
            prices: 价格序列
            start_idx: 起始索引

        返回:
            回测结果字典
        """
        capital = self.initial_capital
        position = 0  # 持仓股数
        portfolio_values = []
        trades = []
        daily_returns = []

        for i in range(start_idx, len(signals)):
            price = prices[i]
            signal = signals[i]

            # 计算当前组合价值
            portfolio_value = capital + position * price
            portfolio_values.append(portfolio_value)

            # 计算日收益率
            if len(portfolio_values) > 1:
                daily_ret = (portfolio_values[-1] - portfolio_values[-2]) / portfolio_values[-2]
                daily_returns.append(daily_ret)

            # 执行交易
            target_position = signal * (portfolio_value * 0.95 / price)  # 95% 仓位
            position_diff = target_position - position

            if abs(position_diff) > 1:  # 最小交易量
                trade_cost = abs(position_diff * price) * (self.transaction_cost + self.slippage)
                capital -= position_diff * price + trade_cost
                position = target_position
                trades.append({
                    "step": i,
                    "signal": signal,
                    "price": price,
                    "shares": position_diff,
                    "cost": trade_cost,
                })

        # 最终平仓
        if position != 0:
            final_price = prices[len(signals) - 1]
            capital += position * final_price
            position = 0

        portfolio_values = np.array(portfolio_values)
        daily_returns = np.array(daily_returns)

        # 计算指标
        total_return = (portfolio_values[-1] - self.initial_capital) / self.initial_capital
        n_days = len(daily_returns)
        annual_return = (1 + total_return) ** (252 / max(n_days, 1)) - 1
        annual_vol = np.std(daily_returns) * np.sqrt(252) if len(daily_returns) > 0 else 0
        sharpe = annual_return / annual_vol if annual_vol > 0 else 0

        # 最大回撤
        peak = np.maximum.accumulate(portfolio_values)
        drawdown = (peak - portfolio_values) / peak
        max_drawdown = np.max(drawdown)

        # Calmar 比率
        calmar = annual_return / max_drawdown if max_drawdown > 0 else 0

        # 胜率
        winning_trades = sum(1 for t in trades if t["signal"] != 0)
        if len(trades) > 1:
            trade_returns = []
            for i in range(1, len(trades)):
                if trades[i]["signal"] != trades[i - 1]["signal"]:
                    ret = (trades[i]["price"] - trades[i - 1]["price"]) / trades[i - 1]["price"]
                    trade_returns.append(ret)
            win_rate = sum(1 for r in trade_returns if r > 0) / max(len(trade_returns), 1)
        else:
            win_rate = 0

        return {
            "total_return": round(total_return, 4),
            "annual_return": round(annual_return, 4),
            "annual_volatility": round(annual_vol, 4),
            "sharpe_ratio": round(sharpe, 4),
            "max_drawdown": round(max_drawdown, 4),
            "calmar_ratio": round(calmar, 4),
            "n_trades": len(trades),
            "win_rate": round(win_rate, 4),
            "portfolio_values": portfolio_values,
            "daily_returns": daily_returns,
        }

class FactorBaselineStrategy:
    """
    纯因子基准策略——不使用机器学习

    使用传统多因子打分:
    - 动量因子: 过去 20 天收益率
    - 均值回归因子: 偏离 60 日均线的程度
    - 波动率因子: 过去 20 天波动率（低波动优先）

    用于与 AI 增强策略对比
    """

    def __init__(self):
        self.signal_generator = SignalGenerator(long_threshold=0.5, short_threshold=-0.5)

    def predict(self, df: pd.DataFrame) -> np.ndarray:
        """生成因子打分"""
        scores = np.zeros(len(df))

        # 动量因子
        momentum = df["close"].pct_change(20)
        momentum_rank = momentum.rank(pct=True)
        scores += (momentum_rank - 0.5) * 2  # 归一化到 [-1, 1]

        # 均值回归因子
        ma60 = df["close"].rolling(60).mean()
        deviation = (df["close"] - ma60) / ma60
        deviation_rank = deviation.rank(pct=True)
        scores -= (deviation_rank - 0.5) * 2  # 偏离越大，做空信号越强

        # 波动率因子（低波动优先）
        vol = df["close"].pct_change().rolling(20).std()
        vol_rank = vol.rank(pct=True)
        scores -= (vol_rank - 0.5) * 1  # 低波动加分

        return scores / 3  # 平均
```

## 四、常见误区

### 误区一：AI 策略一定比传统策略好

AI 的优势在于发现非线性关系，但在数据量小、信噪比低的金融场景中，简单的线性模型可能更稳健。不要为了用 AI 而用 AI。

### 误区二：回测表现好就可以实盘

回测中可能存在无数隐性偏差。本课使用的是模拟数据，真实市场的噪声、流动性、冲击成本都会大幅降低策略表现。

### 误区三：模型越复杂越好

在 Stacking 集成中，元模型选择 Ridge（线性）而不是深度神经网络。复杂模型在小样本上容易过拟合，简单模型反而更稳健。

### 误区四：忽略交易成本的影响

本课的回测包含 0.1% 的交易成本和 0.05% 的滑点。在高频策略中，交易成本可能占到总收益的 50% 以上。必须在策略设计初期就考虑交易成本。

**思路**：为每个交易日生成情感分数（-1 到 1），作为额外特征加入模型。

**答案**：

```python
class NLPFeatureBuilder:
    """NLP 情感因子构建器"""

    def __init__(self):
        self.sentiment_scores = None

    def build_sentiment_feature(self, df: pd.DataFrame, news_sentiments: dict) -> pd.Series:
        """
        构建情感因子

        参数:
            df: 包含日期的 DataFrame
            news_sentiments: {date_str: sentiment_score} 字典

        返回:
            情感分数 Series（缺失值用 0 填充）
        """
        sentiments = df["date"].dt.strftime("%Y-%m-%d").map(news_sentiments)
        sentiments = sentiments.fillna(0)

        # 滚动平均（平滑噪声）
        sentiments_smooth = sentiments.rolling(3, min_periods=1).mean()
        return sentiments_smooth

    def add_to_features(self, features: pd.DataFrame, df: pd.DataFrame, news_sentiments: dict):
        """将情感因子加入特征矩阵"""
        sentiment = self.build_sentiment_feature(df, news_sentiments)
        features["sentiment_1d"] = sentiment
        features["sentiment_3d"] = sentiment.rolling(3).mean()
        features["sentiment_5d"] = sentiment.rolling(5).mean()
        return features
```

### 练习二

**思路**：使用滑动窗口，每月用过去 N 天数据重新训练模型，生成下个月的预测。

**答案**：

```python
def walk_forward_training(
    base_models: dict,
    features: pd.DataFrame,
    labels: pd.Series,
    prices: pd.Series,
    train_window: int = 504,  # 2 年
    retrain_freq: int = 21,   # 每月
):
    """
    Walk-Forward 滚动训练

    每隔 retrain_freq 天，用过去 train_window 天数据重新训练模型
    """
    n_samples = len(features)
    all_predictions = np.full(n_samples, np.nan)
    retrain_dates = []

    for i in range(train_window, n_samples, retrain_freq):
        # 训练窗口
        train_start = max(0, i - train_window)
        X_train = features.iloc[train_start:i].values
        y_train = labels.iloc[train_start:i].values

        # 预测窗口
        pred_end = min(i + retrain_freq, n_samples)
        X_pred = features.iloc[i:pred_end].values

        # 训练集成模型
        ensemble = StackingEnsemble(base_models, n_splits=3)
        ensemble.fit(X_train, y_train)

        # 预测
        predictions = ensemble.predict(X_pred)
        all_predictions[i:pred_end] = predictions

        retrain_dates.append(i)

    # 信号生成
    signal_gen = SignalGenerator(long_threshold=0.005, short_threshold=-0.005)
    signals = signal_gen.generate(all_predictions)

    return signals, all_predictions, retrain_dates
```

**要点**：
- 训练窗口太短 → 模型学不到足够模式；太长 → 包含过时数据
- 重新训练频率需要权衡计算成本和模型新鲜度
- 每次重新训练后，模型权重可能变化，这是预期行为
