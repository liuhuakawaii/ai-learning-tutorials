# 08 阶段实战：AI 增强的多因子策略

## 核心问题

传统多因子策略用线性组合打分。AI 用机器学习发现非线性关系。**AI 到底能带来多少增量？**

```python
import numpy as np
import pandas as pd
from sklearn.linear_model import Ridge
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import mean_squared_error

np.random.seed(42)

def gen_data(n=2000):
    dates = pd.bdate_range('2016-01-01', periods=n)
    ret = np.zeros(n); vol = 0.02
    for i in range(1, n):
        vol = 0.9*vol + 0.1*abs(ret[i-1]) + 0.001
        vol = np.clip(vol, 0.005, 0.05)
        ret[i] = np.random.normal(0.0002, vol)
    close = 100 * np.cumprod(1+ret)
    return pd.DataFrame({'date': dates, 'close': close, 'high': close*1.005,
                          'low': close*0.995, 'open': close*(1+np.random.randn(n)*0.003),
                          'volume': np.random.lognormal(15,0.5,n).astype(int)})

df = gen_data()
```

## 特征工程

```python
class FeatureEngine:
    def __init__(self, windows=[5,10,20,60]):
        self.windows = windows
        self.names = []

    def build(self, df):
        f = pd.DataFrame(index=df.index)
        c = df['close']; v = df['volume']
        for w in self.windows:
            f[f'ret_{w}'] = c.pct_change(w)
            f[f'vol_{w}'] = c.pct_change().rolling(w).std()
            f[f'skew_{w}'] = c.pct_change().rolling(w).skew()
            ma = c.rolling(w).mean()
            f[f'ma_dev_{w}'] = (c - ma) / ma
        for p in [6,14]:
            d = c.diff(); g = d.where(d>0,0).rolling(p).mean()
            l = (-d.where(d<0,0)).rolling(p).mean()
            f[f'rsi_{p}'] = 1 - 1/(1+g/(l+1e-10))
        e12 = c.ewm(span=12).mean(); e26 = c.ewm(span=26).mean()
        macd = e12 - e26; f['macd'] = macd; f['macd_sig'] = macd.ewm(span=9).mean()
        f['bb_upper'] = (c - (c.rolling(20).mean()+2*c.rolling(20).std())) / c
        f['bb_lower'] = (c - (c.rolling(20).mean()-2*c.rolling(20).std())) / c
        for w in [5,20]: f[f'vratio_{w}'] = v / v.rolling(w).mean()
        for lag in [1,2,3,5]: f[f'retlag_{lag}'] = c.pct_change(lag)
        f['range'] = (df['high']-df['low'])/c
        self.names = f.columns.tolist()
        return f

fe = FeatureEngine()
features = fe.build(df)
labels = df['close'].pct_change(5).shift(-5)
valid = features.notna().all(axis=1) & labels.notna()
features, labels = features[valid], labels[valid]
split = int(len(features)*0.7)
X_tr, X_te = features.iloc[:split].values, features.iloc[split:].values
y_tr, y_te = labels.iloc[:split].values, labels.iloc[split:].values
prices_te = df['close'].iloc[valid].iloc[split:].values
print(f"特征: {len(fe.names)}, 训练: {len(X_tr)}, 测试: {len(X_te)}")
```

## 模型训练

```python
class LGBPredictor:
    def __init__(self, n=100, d=5, lr=0.05):
        self.scaler = StandardScaler()
        try:
            import lightgbm as lgb
            self.model = lgb.LGBMRegressor(n_estimators=n, max_depth=d, learning_rate=lr, random_state=42, verbose=-1)
        except ImportError:
            from sklearn.ensemble import GradientBoostingRegressor
            self.model = GradientBoostingRegressor(n_estimators=n, max_depth=d, learning_rate=lr, random_state=42)
    def fit(self, X, y):
        self.model.fit(self.scaler.fit_transform(X), y); return self
    def predict(self, X):
        return self.model.predict(self.scaler.transform(X))

class RidgePredictor:
    def __init__(self): self.scaler = StandardScaler(); self.model = Ridge(1.0)
    def fit(self, X, y): self.model.fit(self.scaler.fit_transform(X), y); return self
    def predict(self, X): return self.model.predict(self.scaler.transform(X))

class SeqPredictor:
    def __init__(self, seq=20): self.seq = seq; self.scaler = StandardScaler(); self.model = Ridge(1.0)
    def fit(self, X, y):
        Xs = self.scaler.fit_transform(X)
        Xm = np.array([Xs[i:i+self.seq].flatten() for i in range(len(Xs)-self.seq+1)])
        self.model.fit(Xm, y[self.seq-1:]); return self
    def predict(self, X):
        Xs = self.scaler.transform(X)
        Xm = np.array([Xs[i:i+self.seq].flatten() for i in range(len(Xs)-self.seq+1)])
        p = self.model.predict(Xm); f = np.full(len(X), p[0]); f[self.seq-1:] = p; return f

class StackingEnsemble:
    def __init__(self, models, n_splits=5):
        self.models = models; self.meta = Ridge(1.0); self.n_splits = n_splits; self.fitted = {}
    def fit(self, X, y):
        n = len(X); meta_f = np.zeros((n, len(self.models)))
        fold = n // (self.n_splits+1)
        for f in range(self.n_splits):
            te = fold*(f+1); tte = min(te+fold, n)
            for i,(nm,cls) in enumerate(self.models.items()):
                m = cls(); m.fit(X[:te], y[:te]); meta_f[te:tte,i] = m.predict(X[te:tte])
        v = ~np.isnan(meta_f).any(axis=1) & ~np.isnan(y)
        self.meta.fit(meta_f[v], y[v])
        self.fitted = {nm: cls().fit(X, y) for nm, cls in self.models.items()}
        return self
    def predict(self, X):
        mf = np.zeros((len(X), len(self.fitted)))
        for i,(nm,m) in enumerate(self.fitted.items()): mf[:,i] = m.predict(X)
        return self.meta.predict(mf)
    def weights(self):
        return dict(zip(self.models.keys(), self.meta.coef_.round(4))) if hasattr(self.meta,'coef_') else {}

models = {'lgb': lambda: LGBPredictor(), 'ridge': lambda: RidgePredictor(), 'seq': lambda: SeqPredictor()}
ens = StackingEnsemble(models).fit(X_tr, y_tr)
for nm, cls in models.items():
    m = cls().fit(X_tr, y_tr)
    print(f"{nm} RMSE: {np.sqrt(mean_squared_error(y_te, m.predict(X_te))):.6f}")
print(f"集成 RMSE: {np.sqrt(mean_squared_error(y_te, ens.predict(X_te))):.6f}")
print(f"权重: {ens.weights()}")
```

## 回测对比

```python
def backtest(signals, prices, capital=1_000_000, cost=0.001):
    pos = 0; values = []; trades = 0
    for i in range(len(signals)):
        price = prices[i]; target = signals[i] * (capital*0.95/price)
        diff = target - pos
        if abs(diff) > 1:
            capital -= diff*price + abs(diff*price)*cost; pos = target; trades += 1
        values.append(capital + pos*price)
    v = np.array(values); ret = np.diff(v)/v[:-1]
    return {'return': v[-1]/1_000_000-1, 'sharpe': np.mean(ret)/np.std(ret)*np.sqrt(252) if np.std(ret)>0 else 0,
            'dd': np.min((v-np.maximum.accumulate(v))/np.maximum.accumulate(v)), 'trades': trades}

pred = ens.predict(X_te)
ai_sig = np.where(pred>0.005, 1, np.where(pred<-0.005, -1, 0))
ai = backtest(ai_sig, prices_te)

mom = pd.Series(prices_te).pct_change(20).values
ma60 = pd.Series(prices_te).rolling(60).mean().values
dev = (prices_te - ma60) / ma60
fscore = np.nan_to_num(mom) - np.nan_to_num(dev)
f_sig = np.where(np.nan_to_num(fscore)>np.nanpercentile(fscore,70), 1,
        np.where(np.nan_to_num(fscore)<np.nanpercentile(fscore,30), -1, 0))
fac = backtest(f_sig, prices_te)

bh = (prices_te[-1]-prices_te[0])/prices_te[0]
print(f"\n{'指标':<10} {'AI增强':>10} {'纯因子':>10} {'买入持有':>10}")
print("-"*45)
print(f"{'总收益':<10} {ai['return']:>10.2%} {fac['return']:>10.2%} {bh:>10.2%}")
print(f"{'夏普':<10} {ai['sharpe']:>10.3f} {fac['sharpe']:>10.3f} {'—':>10}")
print(f"{'最大回撤':<10} {ai['dd']:>10.2%} {fac['dd']:>10.2%} {'—':>10}")
```

## AI 的增量在哪里

在很多情况下 AI 和纯因子差距不大。AI 的价值在于：自动发现非线性关系、处理高维特征、动态调整因子权重。但在低信噪比数据上，简单模型更稳健。**不要为了用 AI 而用 AI**。

def walk_forward(features, labels, window=504, freq=21):
    pred = np.full(len(features), np.nan)
    for i in range(window, len(features), freq):
        e = StackingEnsemble(models, 3).fit(features.iloc[max(0,i-window):i].values, labels.iloc[max(0,i-window):i].values)
        pred[i:min(i+freq,len(features))] = e.predict(features.iloc[i:min(i+freq,len(features))].values)
    return pred
