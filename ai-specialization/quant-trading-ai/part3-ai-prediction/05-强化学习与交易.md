# 第五课：强化学习与交易——让 AI 自主学习交易策略

## 场景引入

传统量化策略需要人工定义规则：当 RSI < 30 时买入，当 RSI > 70 时卖出。但市场是动态的，固定规则很快就会失效。如果我们能让 AI 自己学习"什么时候买、买多少、什么时候卖"呢？

这就是强化学习（Reinforcement Learning, RL）在量化交易中的应用。AlphaGo 击败人类棋手的底层技术，正在被应用于金融领域。高盛、摩根等机构都在探索 RL 驱动的交易策略。

但 RL 在金融领域的应用充满陷阱：奖励函数设计不当会导致灾难性交易，环境模拟不真实会让模型学到虚假规律。本课将带你理解 RL 在交易中的正确使用方式，并用 Stable-Baselines3 实现完整的 PPO 仓位管理策略。

> **风险提示**：强化学习策略在模拟环境中表现良好不等于在实盘中有效。RL 策略的过拟合风险极高，必须经过严格的样本外测试和纸交易验证。本课内容仅供学习，不构成投资建议。

## 一、强化学习基础——为什么 RL 适合交易？

### 1.1 核心概念

```
强化学习的基本框架:

    ┌──────────┐    动作 a_t    ┌──────────┐
    │          │ ─────────────→ │          │
    │  智能体   │               │   环境    │
    │ (Agent)  │ ←───────────── │ (Env)    │
    │          │  状态 s_{t+1}  │          │
    └──────────┘  奖励 r_{t+1}  └──────────┘

在交易中的对应:

    智能体  =  交易策略模型
    环境    =  股票市场（历史数据模拟）
    状态    =  价格、技术指标、持仓情况
    动作    =  买入/卖出/持有，以及仓位比例
    奖励    =  收益（或风险调整后收益）

与监督学习的关键区别:

    监督学习: 给定 (特征, 标签) → 学习映射
    强化学习: 给定 (状态, 奖励) → 学习策略

    交易中没有"正确答案"标签，只有延迟的收益反馈
    → 这正是 RL 天然适合交易的原因
```

### 1.2 交易问题的 RL 建模

```python
"""
交易环境定义——将股票交易建模为强化学习问题
这是整个 RL 交易系统的基础
"""

import numpy as np
import pandas as pd
from dataclasses import dataclass
from enum import IntEnum


class ActionType(IntEnum):
    SELL = 0      # 卖出
    HOLD = 1      # 持有
    BUY = 2       # 买入


@dataclass
class TradingState:
    """交易状态"""
    price_history: np.ndarray      # 历史价格窗口
    technical_indicators: np.ndarray  # 技术指标
    position: float                 # 当前仓位 (0~1)
    cash: float                     # 现金比例
    unrealized_pnl: float          # 未实现盈亏


class StockTradingEnv:
    """
    股票交易强化学习环境

    状态空间:
    - 最近 N 天的价格变化率
    - 技术指标 (MA, RSI, MACD 等)
    - 当前持仓比例
    - 未实现盈亏

    动作空间:
    - 离散模式: 买入/持有/卖出 (固定比例)
    - 连续模式: 目标仓位 [0, 1]

    奖励设计:
    - 基础: 每步收益率
    - 惩罚: 交易成本、过度交易
    """

    def __init__(
        self,
        prices: np.ndarray,
        window_size: int = 20,
        initial_cash: float = 100000.0,
        transaction_cost: float = 0.001,
        reward_type: str = "simple",
    ):
        self.prices = prices
        self.window_size = window_size
        self.initial_cash = initial_cash
        self.transaction_cost = transaction_cost
        self.reward_type = reward_type
        self.n_steps = len(prices) - window_size
        self.reset()

    def reset(self) -> np.ndarray:
        self.current_step = 0
        self.cash = self.initial_cash
        self.position = 0.0
        self.portfolio_values = [self.initial_cash]
        self.trades = []
        return self._get_state()

    def _get_state(self) -> np.ndarray:
        idx = self.current_step + self.window_size
        price_window = self.prices[idx - self.window_size: idx + 1]
        returns = np.diff(price_window) / price_window[:-1]

        ma5 = np.mean(price_window[-5:])
        ma_deviation = (price_window[-1] - ma5) / ma5
        rsi = self._calculate_rsi(price_window, period=14)

        current_price = self.prices[idx]
        portfolio_value = self.cash + self.position * current_price
        position_ratio = (self.position * current_price) / portfolio_value if portfolio_value > 0 else 0

        if self.position > 0 and len(self.trades) > 0:
            avg_cost = np.mean([t["price"] for t in self.trades if t["type"] == "buy"])
            unrealized_return = (current_price - avg_cost) / avg_cost
        else:
            unrealized_return = 0.0

        state = np.concatenate([
            returns,
            [ma_deviation, rsi, position_ratio, unrealized_return],
        ])
        return state.astype(np.float32)

    def _calculate_rsi(self, prices: np.ndarray, period: int = 14) -> float:
        if len(prices) < period + 1:
            return 0.5
        deltas = np.diff(prices[-period - 1:])
        gains = np.where(deltas > 0, deltas, 0)
        losses = np.where(deltas < 0, -deltas, 0)
        avg_gain = np.mean(gains) if len(gains) > 0 else 0
        avg_loss = np.mean(losses) if len(losses) > 0 else 1e-10
        rs = avg_gain / (avg_loss + 1e-10)
        rsi = 1.0 - 1.0 / (1.0 + rs)
        return rsi

    def step(self, action: int) -> tuple:
        idx = self.current_step + self.window_size
        current_price = self.prices[idx]
        prev_portfolio = self.cash + self.position * current_price

        trade_cost = 0
        if action == ActionType.BUY and self.cash > current_price:
            buy_amount = self.cash * 0.2
            shares_to_buy = buy_amount / current_price
            trade_cost = buy_amount * self.transaction_cost
            self.position += shares_to_buy
            self.cash -= (buy_amount + trade_cost)
            self.trades.append({"type": "buy", "price": current_price, "step": self.current_step})
        elif action == ActionType.SELL and self.position > 0:
            shares_to_sell = self.position * 0.5
            sell_value = shares_to_sell * current_price
            trade_cost = sell_value * self.transaction_cost
            self.position -= shares_to_sell
            self.cash += (sell_value - trade_cost)
            self.trades.append({"type": "sell", "price": current_price, "step": self.current_step})

        new_portfolio = self.cash + self.position * self.prices[idx]
        reward = self._calculate_reward(prev_portfolio, new_portfolio, trade_cost)
        self.portfolio_values.append(new_portfolio)

        self.current_step += 1
        done = self.current_step >= self.n_steps
        next_state = self._get_state() if not done else np.zeros_like(self._get_state())
        info = {
            "portfolio_value": new_portfolio,
            "cash": self.cash,
            "position": self.position,
            "trade_cost": trade_cost,
        }
        return next_state, reward, done, info

    def _calculate_reward(self, prev_value: float, new_value: float, trade_cost: float) -> float:
        step_return = (new_value - prev_value) / prev_value
        if self.reward_type == "simple":
            return step_return
        elif self.reward_type == "sharpe":
            if len(self.portfolio_values) > 20:
                returns = np.diff(self.portfolio_values[-20:]) / self.portfolio_values[-21:-1]
                if np.std(returns) > 0:
                    return np.mean(returns) / np.std(returns) * np.sqrt(252)
            return step_return
        elif self.reward_type == "risk_adjusted":
            volatility_penalty = 0
            if len(self.portfolio_values) > 10:
                returns = np.diff(self.portfolio_values[-10:]) / self.portfolio_values[-11:-1]
                volatility_penalty = np.std(returns)
            return step_return - 0.5 * volatility_penalty
        return step_return

    def get_performance_metrics(self) -> dict:
        values = np.array(self.portfolio_values)
        returns = np.diff(values) / values[:-1]
        total_return = (values[-1] - values[0]) / values[0]
        annual_return = (1 + total_return) ** (252 / len(returns)) - 1
        annual_vol = np.std(returns) * np.sqrt(252)
        sharpe = annual_return / annual_vol if annual_vol > 0 else 0
        peak = np.maximum.accumulate(values)
        drawdown = (peak - values) / peak
        max_drawdown = np.max(drawdown)
        return {
            "total_return": round(total_return, 4),
            "annual_return": round(annual_return, 4),
            "annual_volatility": round(annual_vol, 4),
            "sharpe_ratio": round(sharpe, 4),
            "max_drawdown": round(max_drawdown, 4),
            "n_trades": len(self.trades),
        }
```

### 1.3 状态空间设计的关键

```
状态设计的常见错误:

错误 1: 包含未来信息
  状态中包含当天收盘价 → 你不可能在收盘前知道收盘价
  正确做法: 只包含到当前时刻的已知信息

错误 2: 状态维度太高
  包含 200 天的价格历史 → 状态空间爆炸，难以学习
  正确做法: 使用技术指标压缩信息，保留 20-50 维

错误 3: 缺少仓位信息
  只有价格数据，没有当前持仓 → 模型不知道该买还是该卖
  正确做法: 必须包含当前仓位、现金等账户信息
```

## 二、DQN 离散动作策略

### 2.1 DQN 原理

DQN（Deep Q-Network）使用神经网络近似 Q 函数，Q(s,a) 表示在状态 s 下执行动作 a 的预期累积回报。

```
DQN 网络结构:

输入层: 状态向量 s (25 维)
    │
    ▼
┌─────────────┐
│  全连接层    │  128 个神经元, ReLU
├─────────────┤
│  全连接层    │  64 个神经元, ReLU
├─────────────┤
│  输出层      │  3 个神经元 (Q 值)
└─────────────┘
    │
    ▼
输出: [Q(s,卖出), Q(s,持有), Q(s,买入)]

选择动作: argmax(Q(s,a)) 或 ε-贪心策略

DQN 的三大技巧:
  1. 经验回放: 打破数据相关性，提高样本效率
  2. 目标网络: 稳定训练，防止震荡
  3. Double DQN: 减少 Q 值过高估计
```

### 2.2 完整 DQN 实现

```python
"""
DQN 交易策略——完整的训练与评估流程
依赖: pip install torch numpy pandas
"""

import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
from collections import deque
import random


class QNetwork(nn.Module):
    """Q 网络"""

    def __init__(self, state_dim: int, action_dim: int, hidden_dim: int = 128):
        super().__init__()
        self.network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.1),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.1),
            nn.Linear(hidden_dim, action_dim),
        )

    def forward(self, state: torch.Tensor) -> torch.Tensor:
        return self.network(state)


class ReplayBuffer:
    """经验回放缓冲区"""

    def __init__(self, capacity: int = 10000):
        self.buffer = deque(maxlen=capacity)

    def push(self, state, action, reward, next_state, done):
        self.buffer.append((state, action, reward, next_state, done))

    def sample(self, batch_size: int) -> tuple:
        batch = random.sample(self.buffer, batch_size)
        states, actions, rewards, next_states, dones = zip(*batch)
        return (
            np.array(states),
            np.array(actions),
            np.array(rewards, dtype=np.float32),
            np.array(next_states),
            np.array(dones, dtype=np.float32),
        )

    def __len__(self):
        return len(self.buffer)


class DQNTrader:
    """
    DQN 交易策略

    包含:
    - 双网络结构（在线网络 + 目标网络）
    - 经验回放
    - ε-贪心探索
    - 定期目标网络更新
    """

    def __init__(
        self,
        state_dim: int,
        action_dim: int = 3,
        lr: float = 1e-3,
        gamma: float = 0.99,
        epsilon_start: float = 1.0,
        epsilon_end: float = 0.01,
        epsilon_decay: float = 0.995,
        buffer_size: int = 10000,
        batch_size: int = 64,
        target_update_freq: int = 100,
    ):
        self.action_dim = action_dim
        self.gamma = gamma
        self.epsilon = epsilon_start
        self.epsilon_end = epsilon_end
        self.epsilon_decay = epsilon_decay
        self.batch_size = batch_size
        self.target_update_freq = target_update_freq

        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.q_network = QNetwork(state_dim, action_dim).to(self.device)
        self.target_network = QNetwork(state_dim, action_dim).to(self.device)
        self.target_network.load_state_dict(self.q_network.state_dict())
        self.optimizer = optim.Adam(self.q_network.parameters(), lr=lr)
        self.replay_buffer = ReplayBuffer(buffer_size)
        self.training_step = 0

    def select_action(self, state: np.ndarray, training: bool = True) -> int:
        if training and random.random() < self.epsilon:
            return random.randint(0, self.action_dim - 1)
        with torch.no_grad():
            state_tensor = torch.FloatTensor(state).unsqueeze(0).to(self.device)
            q_values = self.q_network(state_tensor)
            return q_values.argmax(dim=1).item()

    def train_step(self) -> float:
        if len(self.replay_buffer) < self.batch_size:
            return 0.0
        states, actions, rewards, next_states, dones = self.replay_buffer.sample(self.batch_size)
        states = torch.FloatTensor(states).to(self.device)
        actions = torch.LongTensor(actions).to(self.device)
        rewards = torch.FloatTensor(rewards).to(self.device)
        next_states = torch.FloatTensor(next_states).to(self.device)
        dones = torch.FloatTensor(dones).to(self.device)

        current_q = self.q_network(states).gather(1, actions.unsqueeze(1)).squeeze(1)
        with torch.no_grad():
            next_actions = self.q_network(next_states).argmax(dim=1)
            next_q = self.target_network(next_states).gather(1, next_actions.unsqueeze(1)).squeeze(1)
            target_q = rewards + self.gamma * next_q * (1 - dones)

        loss = nn.MSELoss()(current_q, target_q)
        self.optimizer.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(self.q_network.parameters(), 1.0)
        self.optimizer.step()

        self.training_step += 1
        if self.training_step % self.target_update_freq == 0:
            self.target_network.load_state_dict(self.q_network.state_dict())
        self.epsilon = max(self.epsilon_end, self.epsilon * self.epsilon_decay)
        return loss.item()

    def train_episode(self, env) -> dict:
        state = env.reset()
        total_reward = 0
        losses = []
        while True:
            action = self.select_action(state, training=True)
            next_state, reward, done, info = env.step(action)
            self.replay_buffer.push(state, action, reward, next_state, float(done))
            loss = self.train_step()
            if loss > 0:
                losses.append(loss)
            total_reward += reward
            state = next_state
            if done:
                break
        return {
            "total_reward": total_reward,
            "avg_loss": np.mean(losses) if losses else 0,
            "epsilon": self.epsilon,
            "portfolio_value": info["portfolio_value"],
        }


def run_dqn_training():
    """完整的 DQN 训练流程"""
    np.random.seed(42)
    n_days = 500
    returns = np.random.normal(0.0005, 0.02, n_days)
    prices = 100 * np.cumprod(1 + returns)

    train_prices = prices[:400]
    test_prices = prices[400:]

    train_env = StockTradingEnv(train_prices, window_size=20)
    test_env = StockTradingEnv(test_prices, window_size=20)
    state_dim = train_env.reset().shape[0]

    trader = DQNTrader(state_dim=state_dim, action_dim=3, lr=1e-3, gamma=0.99)

    print("=" * 50)
    print("开始 DQN 训练")
    print("=" * 50)

    n_episodes = 50
    for episode in range(n_episodes):
        result = trader.train_episode(train_env)
        if (episode + 1) % 10 == 0:
            print(f"Episode {episode + 1}/{n_episodes}")
            print(f"  总奖励: {result['total_reward']:.4f}")
            print(f"  平均损失: {result['avg_loss']:.4f}")
            print(f"  ε: {result['epsilon']:.4f}")
            print(f"  组合价值: {result['portfolio_value']:.2f}")

    print("\n训练完成，测试集表现:")
    test_result = trader.evaluate(test_env) if hasattr(trader, "evaluate") else {}
    print(test_result)


if __name__ == "__main__":
    run_dqn_training()
```

## 三、PPO 连续仓位策略——用 Stable-Baselines3 实现

### 3.1 为什么需要 PPO？

DQN 只能处理离散动作（买/卖/持有），但实际交易中我们更需要连续动作——精确控制仓位比例。

```
DQN (离散动作):               PPO (连续动作):
┌─────────────────┐           ┌─────────────────┐
│ 卖出 (33%)      │           │ 目标仓位: 0.73  │
│ 持有 (34%)      │           │ (连续值 0~1)    │
│ 买入 (33%)      │           │                 │
│                 │           │ 更精细的仓位控制 │
│ 只能粗粒度控制  │           │ 适合实际交易需求 │
└─────────────────┘           └─────────────────┘

PPO 的核心优势:
  1. 稳定训练: clipping 机制防止策略更新过大
  2. 样本效率: 多次复用同一批数据更新
  3. 连续动作: 天然支持仓位比例这种连续决策
```

### 3.2 用 Stable-Baselines3 实现 PPO 仓位管理

```python
"""
PPO 仓位管理策略——基于 Stable-Baselines3
依赖: pip install stable-baselines3 gymnasium numpy pandas
"""

import numpy as np
import pandas as pd
import gymnasium as gym
from gymnasium import spaces
from stable_baselines3 import PPO
from stable_baselines3.common.vec_env import DummyVecEnv
from stable_baselines3.common.callbacks import EvalCallback


class PositionManagementEnv(gym.Env):
    """
    PPO 仓位管理环境

    动作空间: 连续 [0, 1]，表示目标仓位比例
    状态空间: 价格特征 + 技术指标 + 当前仓位
    """

    metadata = {"render_modes": []}

    def __init__(
        self,
        prices: np.ndarray,
        window_size: int = 20,
        initial_cash: float = 100000.0,
        transaction_cost: float = 0.001,
    ):
        super().__init__()
        self.prices = prices
        self.window_size = window_size
        self.initial_cash = initial_cash
        self.transaction_cost = transaction_cost
        self.n_steps = len(prices) - window_size - 1

        # 动作空间: 目标仓位 [0, 1]
        self.action_space = spaces.Box(low=0.0, high=1.0, shape=(1,), dtype=np.float32)
        # 状态空间: window_size 维价格特征 + 4 维附加特征
        obs_dim = window_size + 4
        self.observation_space = spaces.Box(low=-np.inf, high=np.inf, shape=(obs_dim,), dtype=np.float32)

    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        self.current_step = 0
        self.cash = self.initial_cash
        self.position_shares = 0.0
        self.portfolio_values = [self.initial_cash]
        self.trades = []
        return self._get_obs(), {}

    def _get_obs(self) -> np.ndarray:
        idx = self.current_step + self.window_size
        price_window = self.prices[idx - self.window_size: idx + 1]
        returns = np.diff(price_window) / (price_window[:-1] + 1e-10)

        current_price = self.prices[idx]
        portfolio_value = self.cash + self.position_shares * current_price
        position_ratio = (self.position_shares * current_price) / portfolio_value if portfolio_value > 0 else 0
        cash_ratio = self.cash / portfolio_value if portfolio_value > 0 else 1.0

        # 均线偏离度
        ma10 = np.mean(price_window[-10:])
        ma_deviation = (current_price - ma10) / (ma10 + 1e-10)

        # 波动率
        volatility = np.std(returns[-10:]) if len(returns) >= 10 else np.std(returns)

        obs = np.concatenate([
            returns,
            [position_ratio, cash_ratio, ma_deviation, volatility],
        ])
        return obs.astype(np.float32)

    def step(self, action):
        target_position = float(np.clip(action[0], 0, 1))
        idx = self.current_step + self.window_size
        current_price = self.prices[idx]
        portfolio_value = self.cash + self.position_shares * current_price
        current_position_ratio = (self.position_shares * current_price) / portfolio_value if portfolio_value > 0 else 0

        # 根据目标仓位与当前仓位的差异执行交易
        trade_cost = 0.0
        position_diff = target_position - current_position_ratio

        if position_diff > 0.05:
            # 需要买入
            buy_value = position_diff * portfolio_value * 0.5
            buy_value = min(buy_value, self.cash * 0.95)
            if buy_value > 0:
                shares_to_buy = buy_value / current_price
                trade_cost = buy_value * self.transaction_cost
                self.position_shares += shares_to_buy
                self.cash -= (buy_value + trade_cost)
                self.trades.append({"type": "buy", "price": current_price, "step": self.current_step})
        elif position_diff < -0.05:
            # 需要卖出
            sell_ratio = min(abs(position_diff) * 0.5, 1.0)
            shares_to_sell = self.position_shares * sell_ratio
            if shares_to_sell > 0:
                sell_value = shares_to_sell * current_price
                trade_cost = sell_value * self.transaction_cost
                self.position_shares -= shares_to_sell
                self.cash += (sell_value - trade_cost)
                self.trades.append({"type": "sell", "price": current_price, "step": self.current_step})

        # 计算奖励: 风险调整收益
        new_portfolio = self.cash + self.position_shares * self.prices[idx]
        step_return = (new_portfolio - portfolio_value) / portfolio_value

        # 回撤惩罚
        peak = max(self.portfolio_values)
        current_dd = (peak - new_portfolio) / peak if peak > 0 else 0
        drawdown_penalty = current_dd * 0.5

        # 交易成本惩罚
        cost_penalty = (trade_cost / portfolio_value) * 10 if portfolio_value > 0 else 0

        reward = step_return - drawdown_penalty - cost_penalty
        reward = np.clip(reward, -1, 1)

        self.portfolio_values.append(new_portfolio)
        self.current_step += 1
        done = self.current_step >= self.n_steps

        obs = self._get_obs() if not done else np.zeros(self.observation_space.shape, dtype=np.float32)
        info = {
            "portfolio_value": new_portfolio,
            "position_ratio": target_position,
            "trade_cost": trade_cost,
        }
        return obs, reward, done, False, info

    def get_performance_metrics(self) -> dict:
        values = np.array(self.portfolio_values)
        returns = np.diff(values) / (values[:-1] + 1e-10)
        total_return = (values[-1] - values[0]) / values[0]
        annual_return = (1 + total_return) ** (252 / max(len(returns), 1)) - 1
        annual_vol = np.std(returns) * np.sqrt(252) if len(returns) > 1 else 0
        sharpe = annual_return / annual_vol if annual_vol > 0 else 0
        peak = np.maximum.accumulate(values)
        drawdown = (peak - values) / (peak + 1e-10)
        max_drawdown = np.max(drawdown)
        return {
            "total_return": round(total_return, 4),
            "annual_return": round(annual_return, 4),
            "annual_volatility": round(annual_vol, 4),
            "sharpe_ratio": round(sharpe, 4),
            "max_drawdown": round(max_drawdown, 4),
            "n_trades": len(self.trades),
        }
```

## 四、奖励函数设计——最容易踩的坑

### 4.1 奖励函数陷阱

```
常见奖励函数设计错误:

错误 1: 只用收益率
  reward = portfolio_return
  → 模型学到: 高波动 = 高收益（偶尔）
  → 结果: 极高风险策略，实盘爆仓

错误 2: 惩罚太轻
  reward = return - 0.001 * trade_count
  → 模型仍然频繁交易
  → 结果: 交易成本吞噬利润

错误 3: 奖励稀疏
  只在 episode 结束时给奖励
  → 模型难以学到有效的信用分配
  → 结果: 训练极慢或不收敛

错误 4: 奖励尺度不当
  reward = sharpe_ratio * 100
  → 奖励值过大，梯度爆炸
  → 结果: 训练不稳定
```

### 4.2 推荐的奖励函数

```python
def compute_reward(
    prev_value: float,
    new_value: float,
    trade_cost: float,
    portfolio_returns: list,
    risk_aversion: float = 0.5,
) -> float:
    """
    综合奖励函数设计

    公式: reward = step_return - drawdown_penalty - cost_penalty - vol_penalty
    """
    step_return = (new_value - prev_value) / prev_value

    # 回撤惩罚
    drawdown_penalty = 0
    if len(portfolio_returns) > 5:
        recent_returns = np.array(portfolio_returns[-5:])
        cumulative = np.cumprod(1 + recent_returns)
        peak = np.maximum.accumulate(cumulative)
        current_dd = (peak[-1] - cumulative[-1]) / peak[-1]
        drawdown_penalty = current_dd * risk_aversion

    # 交易成本惩罚（平方惩罚，鼓励减少交易）
    cost_penalty = (trade_cost / prev_value) ** 2 * 100

    # 波动率惩罚
    vol_penalty = 0
    if len(portfolio_returns) > 10:
        vol_penalty = np.std(portfolio_returns[-10:]) * 0.3

    reward = step_return - drawdown_penalty - cost_penalty - vol_penalty
    return reward
```

## 五、RL 的过拟合——金融 RL 的头号杀手

### 5.1 为什么 RL 比监督学习更容易过拟合？

```
RL 过拟合的特殊性:

监督学习过拟合:
  训练集准确率 99% → 测试集准确率 60%
  → 诊断简单，正则化可缓解

RL 过拟合:
  模拟环境收益 300% → 实盘收益 -40%
  → 更隐蔽，更危险

原因:
  1. RL 在模拟环境中"探索"出虚假规律
     例: 模拟数据中每逢周三上涨 → 模型学到"周三买入"
     → 这是数据噪声，不是真实规律

  2. RL 的策略是自适应的，会过度拟合环境细节
     例: 模拟器没有滑点 → 模型学会高频交易
     → 实盘中滑点吞噬利润

  3. 评估指标的方差极大
     同一策略在不同随机种子下，收益可能从 +200% 到 -50%
```

### 5.2 防范措施

```python
"""
RL 过拟合检测与防范框架
"""

import numpy as np


class RLOverfitDetector:
    """RL 过拟合检测器"""

    def __init__(self, n_seeds: int = 10):
        self.n_seeds = n_seeds
        self.seed_results = []

    def run_multi_seed_evaluation(self, train_func, eval_func, seeds: list = None):
        """
        多种子评估——检测 RL 策略的稳定性

        核心思想: 如果策略真的有效，在不同随机种子下应该都表现良好
        """
        if seeds is None:
            seeds = list(range(self.n_seeds))

        results = []
        for seed in seeds:
            np.random.seed(seed)
            model = train_func(seed=seed)
            metrics = eval_func(model)
            results.append(metrics)

        self.seed_results = results
        return self._analyze_stability(results)

    def _analyze_stability(self, results: list) -> dict:
        """分析多种子结果的稳定性"""
        returns = [r["total_return"] for r in results]
        sharpes = [r["sharpe_ratio"] for r in results]
        drawdowns = [r["max_drawdown"] for r in results]

        analysis = {
            "return_mean": np.mean(returns),
            "return_std": np.std(returns),
            "return_min": np.min(returns),
            "return_max": np.max(returns),
            "sharpe_mean": np.mean(sharpes),
            "sharpe_std": np.std(sharpes),
            "max_drawdown_worst": np.max(drawdowns),
            "positive_rate": sum(1 for r in returns if r > 0) / len(returns),
        }

        # 过拟合警告
        warnings = []
        if analysis["return_std"] > abs(analysis["return_mean"]) * 0.5:
            warnings.append("收益方差过大，策略不稳定")
        if analysis["positive_rate"] < 0.6:
            warnings.append("正收益种子不足 60%，策略可靠性低")
        if analysis["sharpe_std"] > 0.5:
            warnings.append("Sharpe 比率波动大，过拟合风险高")
        if analysis["max_drawdown_worst"] > 0.3:
            warnings.append("最大回撤超过 30%，风险控制不足")

        analysis["warnings"] = warnings
        return analysis


def walk_forward_validation(env_class, model_class, prices: np.ndarray, n_splits: int = 5):
    """
    Walk-Forward 验证——模拟真实的策略迭代过程

    与普通交叉验证的区别:
    - 普通 CV: 随机划分，存在未来信息泄露
    - Walk-Forward: 严格按时间顺序，训练在前，测试在后
    """
    split_size = len(prices) // (n_splits + 1)
    results = []

    for i in range(n_splits):
        train_end = split_size * (i + 1)
        test_end = min(train_end + split_size, len(prices))

        train_prices = prices[:train_end]
        test_prices = prices[train_end:test_end]

        if len(test_prices) < 20:
            continue

        # 训练
        train_env = env_class(train_prices)
        model = model_class(train_env)
        model.train()

        # 测试
        test_env = env_class(test_prices)
        metrics = model.evaluate(test_env)
        results.append({
            "fold": i,
            "train_end": train_end,
            "test_end": test_end,
            **metrics,
        })

    return results
```

## 七、常见误区

### 误区一：模拟环境不真实

RL 策略在模拟环境中训练，如果环境过于简单（没有滑点、没有流动性限制），模型学到的策略在实盘中会失效。务必在环境设计中加入：滑点、成交量限制、涨跌停限制。

### 误区二：评估指标选错

不要只看总收益率。RL 策略可能通过高波动获得高收益，实盘中会导致巨大回撤。必须综合考虑 Sharpe 比率、最大回撤、Calmar 比率。

### 误区三：直接上实盘

RL 策略必须经过：模拟交易 → 纸交易（至少 3 个月） → 小资金实盘 → 逐步加仓。任何跳步都可能导致严重亏损。

### 误区四：忽视 RL 与监督学习的对比选择

```
对比维度          监督学习              强化学习
─────────────────────────────────────────────────
数据需求          需要标签（未来收益）    不需要标签
决策序列          单步预测              多步序列决策
交易成本          难以融入              天然融入奖励
仓位管理          只能预测方向          可以优化仓位
训练难度          较低                  较高
样本效率          较高                  较低
过拟合风险        中等                  极高
实盘可用性        较成熟                仍在探索

建议:
- 方向预测 → 监督学习（简单有效）
- 仓位优化 → 强化学习（天然适合）
- 组合管理 → 强化学习（多资产协调）
- 入门学习 → 先掌握监督学习，再尝试 RL
```

## 参考答案

### 练习一

**思路**：Calmar 比率 = 年化收益率 / 最大回撤。需要维护滚动窗口计算年化收益和最大回撤，然后将其作为奖励信号。

**答案**：

```python
def _calculate_reward_calmar(self, prev_value, new_value, trade_cost):
    """Calmar 比率奖励"""
    step_return = (new_value - prev_value) / prev_value

    if len(self.portfolio_values) < 60:
        return step_return

    values = np.array(self.portfolio_values[-60:])
    returns = np.diff(values) / values[:-1]
    annual_return = np.mean(returns) * 252
    peak = np.maximum.accumulate(values)
    drawdowns = (peak - values) / peak
    max_drawdown = np.max(drawdowns)

    if max_drawdown > 0:
        calmar = annual_return / max_drawdown
    else:
        calmar = annual_return * 10

    reward = np.clip(calmar * 0.1, -1, 1)
    return reward
```

**要点**：
- 需要足够的历史数据才能计算有意义的 Calmar 比率
- 奖励需要裁剪到合理范围，防止梯度爆炸
- 最大回撤为 0 时需要特殊处理，避免除零

### 练习二

**思路**：状态空间扩展为 N 只股票的信息拼接，动作空间变为 N 维连续向量，环境内部管理多个持仓账户。

**答案**：

```python
class MultiAssetPositionEnv(gym.Env):
    """多资产 PPO 仓位管理环境"""

    def __init__(self, prices_dict: dict, window_size: int = 20, initial_cash: float = 100000):
        super().__init__()
        self.prices_dict = prices_dict
        self.n_assets = len(prices_dict)
        self.stock_codes = list(prices_dict.keys())
        self.window_size = window_size
        self.initial_cash = initial_cash
        self.positions = {code: 0.0 for code in self.stock_codes}

        # 动作空间: 每只股票的目标仓位
        self.action_space = spaces.Box(
            low=0.0, high=1.0, shape=(self.n_assets,), dtype=np.float32
        )
        # 状态空间: 每只股票 5 维特征 + n_assets 维仓位
        obs_dim = self.n_assets * 5 + self.n_assets + 1
        self.observation_space = spaces.Box(
            low=-np.inf, high=np.inf, shape=(obs_dim,), dtype=np.float32
        )

    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        self.current_step = 0
        self.cash = self.initial_cash
        self.positions = {code: 0.0 for code in self.stock_codes}
        return self._get_obs(), {}

    def _get_obs(self):
        features = []
        total_value = self._get_portfolio_value()

        for code in self.stock_codes:
            prices = self.prices_dict[code]
            idx = self.current_step + self.window_size
            window = prices[idx - self.window_size: idx + 1]
            returns = np.diff(window) / (window[:-1] + 1e-10)
            features.extend(returns[-5:])

            price = prices[idx]
            pos_ratio = self.positions[code] * price / total_value if total_value > 0 else 0
            features.append(pos_ratio)

        features.append(self.cash / total_value if total_value > 0 else 1.0)
        return np.array(features, dtype=np.float32)

    def step(self, action):
        total_value = self._get_portfolio_value()
        trade_cost = 0.0

        for i, code in enumerate(self.stock_codes):
            price = self.prices_dict[code][self.current_step + self.window_size]
            target_ratio = float(np.clip(action[i], 0, 1))
            current_ratio = self.positions[code] * price / total_value if total_value > 0 else 0
            diff = target_ratio - current_ratio

            if diff > 0.05:
                buy_value = min(diff * total_value * 0.5, self.cash * 0.95)
                if buy_value > 0:
                    shares = buy_value / price
                    cost = buy_value * 0.001
                    self.positions[code] += shares
                    self.cash -= (buy_value + cost)
                    trade_cost += cost
            elif diff < -0.05:
                sell_shares = self.positions[code] * min(abs(diff) * 0.5, 1.0)
                if sell_shares > 0:
                    sell_value = sell_shares * price
                    cost = sell_value * 0.001
                    self.positions[code] -= sell_shares
                    self.cash += (sell_value - cost)
                    trade_cost += cost

        self.current_step += 1
        done = self.current_step >= len(list(self.prices_dict.values())[0]) - self.window_size - 1

        new_value = self._get_portfolio_value()
        reward = (new_value - total_value) / total_value if total_value > 0 else 0

        obs = self._get_obs() if not done else np.zeros(self.observation_space.shape, dtype=np.float32)
        return obs, reward, done, False, {"portfolio_value": new_value}

    def _get_portfolio_value(self):
        value = self.cash
        for code in self.stock_codes:
            price = self.prices_dict[code][self.current_step + self.window_size]
            value += self.positions[code] * price
        return value
```

**要点**：
- 动作空间为 N 维连续向量，每维表示一个资产的目标仓位
- 需要考虑总资产约束（所有仓位比例之和不超过 1）
- 状态空间需要包含所有资产的特征和当前仓位信息
