# 01 金融时序预测的陷阱

## 一个失败的实验

你用 LSTM 预测股价，回测年化 200%。实盘三个月，亏了 40%。我们先复现这个失败。

```python
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import r2_score

np.random.seed(42)

def price_series(n=1000):
    r1 = np.random.normal(0.001, 0.01, 300)
    r2 = np.random.normal(-0.002, 0.03, 400)
    r3 = np.random.normal(0.0015, 0.015, 300)
    returns = np.concatenate([r1, r2, r3])
    return 100 * np.cumprod(1 + returns), returns

prices, returns = price_series()
first, second = returns[:500], returns[500:]
print(f"前半段: 均值{first.mean():.5f} 标准差{first.std():.5f}")
print(f"后半段: 均值{second.mean():.5f} 标准差{second.std():.5f}")
print("问题: 训练集和测试集的统计特性完全不同")
```

## 陷阱一：非平稳性

```python
def nonstationarity_demo():
    prices, returns = price_series()
    df = pd.DataFrame({'ret': returns})
    for lag in [1,2,3,5,10]:
        df[f'lag_{lag}'] = df['ret'].shift(lag)
        df[f'vol_{lag}'] = df['ret'].rolling(lag).std().shift(1)
    df = df.dropna()
    feat = [c for c in df.columns if c != 'ret']
    X, y = df[feat].values, df['ret'].values

    model = RandomForestRegressor(n_estimators=100, max_depth=5, random_state=42)
    model.fit(X[:500], y[:500])
    train_r2 = r2_score(y[:500], model.predict(X[:500]))
    test_r2 = r2_score(y[500:], model.predict(X[500:]))
    print(f"训练集 R²: {train_r2:.4f}")
    print(f"测试集 R²: {test_r2:.4f} (为负 = 比猜均值还差)")

nonstationarity_demo()
```

## 陷阱三：前视偏差

```python
def lookahead_demo():
    n = 500
    returns = np.random.normal(0, 0.02, n)
    prices = 100 * np.cumprod(1 + returns)
    df = pd.DataFrame({'close': prices, 'ret': returns})
    for lag in [1,2,3,5,10]:
        df[f'lag_{lag}'] = df['ret'].shift(lag)
    df = df.dropna()
    feat = [c for c in df.columns if c not in ['close','ret']]
    X, y = df[feat].values, df['ret'].values

    # 错误: 用全部数据训练
    m = RandomForestRegressor(100, random_state=42).fit(X, y)
    cum_wrong = np.cumprod(1 + np.sign(m.predict(X)) * y)

    # 正确: 滚动窗口
    pred = np.zeros(len(y))
    for i in range(200, len(y)):
        RandomForestRegressor(50, random_state=42).fit(X[i-200:i], y[i-200:i])
        pred[i] = RandomForestRegressor(50, random_state=42).fit(X[i-200:i], y[i-200:i]).predict(X[i:i+1])[0]
    cum_right = np.cumprod(1 + np.sign(pred[200:]) * y[200:])

    print(f"有偏差回测: {cum_wrong[-1]:.3f}")
    print(f"正确回测:   {cum_right[-1]:.3f}")
    print(f"虚假超额: {(cum_wrong[-1]-cum_right[-1])*100:.1f}%")

lookahead_demo()
```

## 陷阱四：Backtest Overfitting

```python
def dsr(observed_sr, n_trials, n_obs, skew=0, kurt=3):
    from scipy import stats
    e = 0.5772156649
    e_max = (1-e)*stats.norm.ppf(1-1/n_trials) + e*stats.norm.ppf(1-1/(n_trials*np.e))
    sr_std = np.sqrt((1+0.5*observed_sr**2-skew*observed_sr+(kurt-3)/4*observed_sr**2)/(n_obs-1))
    return stats.norm.cdf((observed_sr - e_max) / sr_std)

print("尝试200个策略，夏普2.0，样本504天:")
print(f"DSR = {dsr(2.0, 200, 504, -0.5, 4):.4f} (该策略为真的概率)")
print("结论: 尝试越多，需要更高的夏普来证明有效")
```

### 练习一
