## 从一个 bug 说起

回测表现良好，实盘第一天赚 0.3%。第二天网络断了 5 分钟，程序没自动重连，错过信号。第三天下了 20 个订单但资金只够 15 个。这些都不是策略问题，是工程问题。

```python
import uuid
import threading
from enum import Enum
from dataclasses import dataclass, field
from typing import Dict, List
from collections import defaultdict
from datetime import datetime
import numpy as np

class OrderStatus(Enum):
    CREATED="created"; SUBMITTED="submitted"; FILLED="filled"
    REJECTED="rejected"; CANCELLED="cancelled"; PARTIAL="partial"

class OrderSide(Enum):
    BUY="buy"; SELL="sell"

@dataclass
class Order:
    order_id: str; symbol: str; side: OrderSide; quantity: int
    price: float=0.0; status: OrderStatus=OrderStatus.CREATED
    filled_qty: int=0; filled_price: float=0.0
    created_at: datetime=field(default_factory=datetime.now)
    reject_reason: str=""; commission: float=0.0
```

## 订单管理

```python
class OrderManager:
    def __init__(self):
        self.orders: Dict[str, Order] = {}
        self.history: List[Order] = []
        self._lock = threading.Lock()

    def create(self, symbol, side, quantity, price=0.0):
        with self._lock:
            oid = f"ORD_{uuid.uuid4().hex[:8].upper()}"
            o = Order(oid, symbol, side, quantity, price)
            self.orders[oid] = o
            return o

    def update(self, oid, status, filled_qty=0, filled_price=0.0, reason=""):
        with self._lock:
            o = self.orders.get(oid)
            if not o: return
            o.status = status
            if filled_qty > 0:
                total = o.filled_price*o.filled_qty + filled_price*filled_qty
                o.filled_qty += filled_qty
                o.filled_price = total/o.filled_qty if o.filled_qty > 0 else 0
            if status == OrderStatus.REJECTED: o.reject_reason = reason
            if status in (OrderStatus.FILLED, OrderStatus.REJECTED, OrderStatus.CANCELLED):
                self.history.append(o)

    def pending(self):
        return [o for o in self.orders.values()
                if o.status in (OrderStatus.CREATED, OrderStatus.SUBMITTED, OrderStatus.PARTIAL)]
```

## 风控预检查

```python
class PreTradeRisk:
    def __init__(self, cfg):
        self.max_order = cfg.get('max_order', 500_000)
        self.max_pos_pct = cfg.get('max_pos_pct', 0.20)
        self.max_daily = cfg.get('max_daily', 100)
        self.count = 0

    def check(self, order, equity, positions, prices):
        val = order.quantity * order.price
        if val > self.max_order:
            return False, f"单笔 {val:,.0f} 超限 {self.max_order:,.0f}"
        if order.side == OrderSide.BUY:
            cur = positions.get(order.symbol, 0) * prices.get(order.symbol, 0)
            if (cur + val)/equity > self.max_pos_pct:
                return False, "持仓比例超限"
        if self.count >= self.max_daily:
            return False, "日内交易次数超限"
        self.count += 1
        return True, "PASS"

    def reset(self): self.count = 0
```

## 执行引擎

```python
class Executor:
    def __init__(self, comm=0.0003):
        self.comm = comm

    def execute(self, order, price):
        ep = price * (1.001 if order.side == OrderSide.BUY else 0.999)
        tv = ep * order.quantity
        c = max(tv*self.comm, 5.0)
        if order.side == OrderSide.SELL: c += tv*0.001
        return ep, order.quantity, c

class TWAP:
    def __init__(self, slices=10): self.slices = slices
    def split(self, order):
        q = order.quantity // self.slices; r = order.quantity % self.slices
        return [Order(f"{order.order_id}_T{i:03d}", order.symbol, order.side,
                      q + (1 if i < r else 0), order.price) for i in range(self.slices) if q + (1 if i < r else 0) > 0]

class VWAP:
    def __init__(self, profile):
        self.p = np.array(profile); self.p /= self.p.sum()
    def split(self, order):
        qtys = np.round(order.quantity * self.p).astype(int)
        qtys[np.argmax(qtys)] += order.quantity - qtys.sum()
        return [Order(f"{order.order_id}_V{i:03d}", order.symbol, order.side,
                      int(q), order.price) for i, q in enumerate(qtys) if q > 0]
```

## 组合管理

```python
class PortfolioManager:
    def __init__(self, capital):
        self.cash = capital; self.init = capital
        self.positions: Dict[str, int] = defaultdict(int)
        self.avg_prices: Dict[str, float] = {}
        self.log: List[dict] = []

    def on_fill(self, sym, side, qty, price, comm):
        if side == OrderSide.BUY:
            if self.positions[sym] > 0:
                tc = self.avg_prices[sym]*self.positions[sym] + price*qty
                self.positions[sym] += qty
                self.avg_prices[sym] = tc/self.positions[sym]
            else:
                self.positions[sym] = qty; self.avg_prices[sym] = price
            self.cash -= price*qty + comm
        else:
            pnl = (price - self.avg_prices.get(sym, price))*qty
            self.positions[sym] -= qty
            self.cash += price*qty - comm
            self.log.append({'sym': sym, 'pnl': pnl, 'comm': comm})

    def total_value(self, prices):
        return self.cash + sum(self.positions[s]*prices.get(s,0) for s in self.positions if self.positions[s]>0)
```

## 完整系统

```python
class LiveSystem:
    def __init__(self, cfg):
        self.om = OrderManager()
        self.risk = PreTradeRisk(cfg)
        self.exec = Executor(cfg.get('comm', 0.0003))
        self.port = PortfolioManager(cfg['capital'])
        self.prices: Dict[str, float] = {}
        self._count = 0

    def on_signal(self, sym, side, qty, price):
        o = self.om.create(sym, side, qty, price)
        ok, reason = self.risk.check(o, self.port.total_value(self.prices),
                                      dict(self.port.positions), self.prices)
        if not ok:
            self.om.update(o.order_id, OrderStatus.REJECTED, reason=reason)
            return False, reason
        self.om.update(o.order_id, OrderStatus.SUBMITTED)
        ep, eq, comm = self.exec.execute(o, self.prices.get(sym, price))
        self.port.on_fill(sym, side, eq, ep, comm)
        self.om.update(o.order_id, OrderStatus.FILLED, eq, ep)
        return True, f"成交 {side.value} {sym} {eq}股 @{ep:.2f}"

    def update_prices(self, p): self.prices.update(p)

    def report(self):
        return {'cash': self.port.cash, 'total': self.port.total_value(self.prices),
                'completed': len(self.om.history),
                'rejected': sum(1 for o in self.om.history if o.status==OrderStatus.REJECTED)}

# 模拟
np.random.seed(42)
sys = LiveSystem({'capital': 1_000_000, 'max_order': 200_000, 'max_pos_pct': 0.25})
syms = ['SH.600000','SH.600036','SZ.000001']
for day in range(5):
    p = {s: 100+np.random.normal(0,5) for s in syms}
    sys.update_prices(p)
    for s in syms:
        if np.random.random() > 0.6:
            side = OrderSide.BUY if np.random.random()>0.5 else OrderSide.SELL
            ok, msg = sys.on_signal(s, side, np.random.choice([100,200,500]), p[s])
            if not ok: print(f"  拒绝: {msg}")
r = sys.report()
print(f"现金:{r['cash']:,.0f} 总值:{r['total']:,.0f} 完成:{r['completed']} 拒绝:{r['rejected']}")
```
