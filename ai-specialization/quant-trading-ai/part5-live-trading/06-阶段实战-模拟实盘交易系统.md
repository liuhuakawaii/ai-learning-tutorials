# 06 阶段实战：模拟实盘交易系统

## 为什么要跑模拟盘

回测是"历史重放"，实盘是"事件驱动"。模拟盘验证系统工程——网络断了、订单被拒、策略衰减怎么办？

```python
import time, random, math, threading, queue, uuid
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
from typing import Dict, List, Callable
from collections import defaultdict
import numpy as np

class EventType(Enum):
    BAR="bar"; SIGNAL="signal"; FILLED="filled"; REJECTED="rejected"; RISK_FAIL="risk_fail"

@dataclass
class Event:
    type: EventType; data: dict; timestamp: datetime=field(default_factory=datetime.now)

class EventBus:
    def __init__(self):
        self._h: Dict[EventType, List[Callable]] = defaultdict(list)
        self._q = queue.Queue(); self._running = False
    def subscribe(self, t, h): self._h[t].append(h)
    def publish(self, e): self._q.put(e)
    def start(self):
        self._running = True; threading.Thread(target=self._loop, daemon=True).start()
    def stop(self): self._running = False
    def _loop(self):
        while self._running:
            try:
                e = self._q.get(timeout=0.1)
                for h in self._h.get(e.type, []):
                    try: h(e)
                    except Exception as ex: print(f"事件异常: {ex}")
            except queue.Empty: pass
```

## 模拟行情 + 滑点 + 手续费

```python
@dataclass
class Bar:
    symbol: str; timestamp: datetime; open: float; high: float; low: float; close: float; volume: int

class MarketData:
    def __init__(self, symbols, prices=None, vol=0.02, seed=42):
        self._p = prices or {s: 100.0 for s in symbols}
        self._rng = random.Random(seed); self._t = datetime(2024, 1, 2, 9, 30); self.vol = vol
    def gen_bar(self, sym, interval=60):
        p = self._p[sym]; c = self._rng.gauss(0.0001, self.vol*0.1)
        cl = round(p*(1+c),2); hi = round(max(p,cl)*(1+abs(self._rng.gauss(0,0.005))),2)
        lo = round(min(p,cl)*(1-abs(self._rng.gauss(0,0.005))),2)
        self._p[sym] = cl; self._t += timedelta(seconds=interval)
        return Bar(sym, self._t, p, hi, lo, cl, self._rng.randint(10000,500000))

class Slippage:
    def calc(self, price, qty, vol, side="buy"):
        s = 0.0002 + 0.1*math.sqrt(qty/max(vol,1)) + 0.01
        fp = price*(1+s) if side=="buy" else price*(1-s)
        return round(fp,2), round(price*s,4)

class Commission:
    def calc(self, price, qty, side):
        c = max(price*qty*0.0003, 5.0)
        if side=="sell": c += price*qty*0.001
        return round(c,2)
```

## 模拟 CTP + 组合管理

```python
class Side(Enum): BUY="buy"; SELL="sell"
class Status(Enum): CREATED="created"; SUBMITTED="submitted"; FILLED="filled"; REJECTED="rejected"

@dataclass
class Order:
    oid: str; symbol: str; side: Side; quantity: int
    status: Status=Status.CREATED; filled_qty: int=0; avg_price: float=0.0; commission: float=0.0

@dataclass
class Position:
    symbol: str; quantity: int=0; avg_cost: float=0.0; realized: float=0.0
    @property
    def value(self): return self.quantity*self.avg_cost

@dataclass
class Account:
    capital: float=1_000_000; cash: float=1_000_000
    positions: Dict[str, Position]=field(default_factory=dict)
    total_comm: float=0.0
    @property
    def equity(self): return self.cash + sum(p.value for p in self.positions.values())
    @property
    def pnl(self): return self.equity - self.capital

class SimCTP:
    def __init__(self, acc):
        self.acc = acc; self.slip = Slippage(); self.comm = Commission()
        self._prices: Dict[str,float] = {}; self._vols: Dict[str,int] = {}
    def update(self, sym, price, vol=100000): self._prices[sym]=price; self._vols[sym]=vol
    def submit(self, order):
        p = self._prices.get(order.symbol)
        if not p: order.status=Status.REJECTED; return order
        qty = order.quantity - order.filled_qty
        fp, slip = self.slip.calc(p, qty, self._vols.get(order.symbol,100000), order.side.value)
        c = self.comm.calc(fp, qty, order.side.value)
        cost = fp*qty + c
        if order.side == Side.BUY:
            if self.acc.cash < cost: order.status=Status.REJECTED; return order
            self.acc.cash -= cost; self._upd(order.symbol, qty, fp, "buy")
        else:
            pos = self.acc.positions.get(order.symbol)
            if not pos or pos.quantity < qty: order.status=Status.REJECTED; return order
            self.acc.cash += fp*qty - c; self._upd(order.symbol, qty, fp, "sell")
        order.filled_qty=order.quantity; order.avg_price=fp; order.status=Status.FILLED
        order.commission=c; self.acc.total_comm+=c; return order
    def _upd(self, sym, qty, price, side):
        if sym not in self.acc.positions: self.acc.positions[sym]=Position(sym)
        pos = self.acc.positions[sym]
        if side=="buy":
            tc=pos.avg_cost*pos.quantity+price*qty; pos.quantity+=qty
            pos.avg_cost=tc/pos.quantity if pos.quantity>0 else 0
        else:
            pos.realized+=(price-pos.avg_cost)*qty; pos.quantity-=qty
            if pos.quantity==0: pos.avg_cost=0
```

## 策略 + 风控 + 整合

```python
class DualMA:
    def __init__(self, sw=5, lw=20, pct=0.1):
        self.sw=sw; self.lw=lw; self.pct=pct; self._p: Dict[str,list]={}; self._sig=[]
    def on_bar(self, sym, close, eq=1_000_000):
        if sym not in self._p: self._p[sym]=[]
        self._p[sym].append(close); p=self._p[sym]
        if len(p)<self.lw+1: return None
        s,l = np.mean(p[-self.sw:]), np.mean(p[-self.lw:])
        ps,pl = np.mean(p[-self.sw-1:-1]), np.mean(p[-self.lw-1:-1])
        sig = None
        if ps<=pl and s>l:
            q=int(eq*self.pct/close/100)*100
            if q>0: sig={'sym':sym,'side':'buy','qty':q,'price':close}
        elif ps>=pl and s<l:
            sig={'sym':sym,'side':'sell','qty':0,'price':close}
        if sig: self._sig.append(sig)
        return sig

@dataclass
class RiskCfg:
    max_pos: float=0.3; max_loss: float=0.02; max_dd: float=0.05
    max_order: float=200_000; min_cash: float=0.1

class RiskMgr:
    def __init__(self, cfg=None):
        self.cfg=cfg or RiskCfg(); self._peak=0.0
    def check(self, sym, side, qty, price, eq, cash, pv=0):
        v=price*qty
        if v>self.cfg.max_order: return False,"订单超限"
        if side=="buy" and (pv+v)/eq>self.cfg.max_pos: return False,"持仓超限"
        if side=="buy" and (cash-v)/eq<self.cfg.min_cash: return False,"现金不足"
        if self._peak>0 and (self._peak-eq)/self._peak>self.cfg.max_dd: return False,"回撤超限"
        return True,"通过"
    def update_eq(self, eq): self._peak=max(self._peak,eq)

class SimTradingSystem:
    def __init__(self, cfg=None):
        cfg=cfg or {}
        self.bus=EventBus(); self.acc=Account(cfg.get("capital",1_000_000))
        self.ctp=SimCTP(self.acc); self.risk=RiskMgr()
        self.strat=DualMA(cfg.get("sw",5), cfg.get("lw",20))
        syms=cfg.get("symbols",["600519.SH","000858.SZ","601318.SH"])
        self.md=MarketData(syms, cfg.get("prices"))
        self._trades=0; self._running=False
        self.bus.subscribe(EventType.SIGNAL, self._on_signal)
    def start(self):
        self._running=True; self.bus.start()
        threading.Thread(target=self._loop, daemon=True).start()
    def stop(self): self._running=False; self.bus.stop()
    def _loop(self):
        syms=list(self.md._p.keys())
        while self._running:
            for s in syms:
                bar=self.md.gen_bar(s); self.ctp.update(s,bar.close,bar.volume)
                sig=self.strat.on_bar(s,bar.close,self.acc.equity)
                if sig: self.bus.publish(Event(EventType.SIGNAL,sig))
            time.sleep(0.1)
    def _on_signal(self, e):
        d=e.data; sym,side,price=d["sym"],d["side"],d["price"]
        qty = self.acc.positions[sym].quantity if side=="sell" else d.get("qty",100)
        if side=="sell" and qty<=0: return
        ok,_=self.risk.check(sym,side,qty,price,self.acc.equity,self.acc.cash,
                              self.acc.positions.get(sym,Position(sym)).value)
        if not ok: return
        order=Order(f"O{self._trades:06d}",sym,Side(side),qty); self._trades+=1
        f=self.ctp.submit(order)
        if f.status==Status.FILLED:
            self.bus.publish(Event(EventType.FILLED,{"sym":sym,"price":f.avg_price,"comm":f.commission}))
    def report(self):
        return {"equity":round(self.acc.equity,2),"pnl":round(self.acc.pnl,2),
                "return":round(self.acc.pnl/self.acc.capital*100,2),"trades":self._trades}

sys=SimTradingSystem({"capital":1_000_000,"symbols":["600519.SH","000858.SZ","601318.SH"],
                       "prices":{"600519.SH":1800,"000858.SZ":150,"601318.SH":50}})
sys.start()
try:
    for _ in range(60): time.sleep(0.1)
    r=sys.report()
    print(f"净值:{r['equity']:,.0f} 盈亏:{r['pnl']:,.0f} ({r['return']:.2f}%) 交易:{r['trades']}笔")
except KeyboardInterrupt: pass
finally: sys.stop()
```

## 从模拟到实盘的差距

1. **网络断线重连**——CTP 断开后自动重连，断线期间信号缓存
2. **部分成交**——大单分多次成交，跟踪已成交量
3. **交易所规则**——涨跌停、停牌、ST 限制
4. **状态持久化**——程序崩溃后恢复持仓和订单状态
