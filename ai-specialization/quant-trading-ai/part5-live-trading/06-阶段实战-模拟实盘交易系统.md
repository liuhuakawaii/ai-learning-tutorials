### 一、事件系统（核心骨架）

```python
"""
事件驱动系统

所有模块通过事件总线通信，解耦各组件
"""

from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Any, Callable, Dict, List
from collections import defaultdict
import threading
import queue


class EventType(Enum):
    # 数据事件
    MARKET_DATA = "market_data"         # 行情更新
    BAR_CLOSED = "bar_closed"           # K 线完成

    # 信号事件
    SIGNAL = "signal"                   # 策略信号

    # 订单事件
    ORDER_NEW = "order_new"             # 新订单
    ORDER_FILLED = "order_filled"       # 订单成交
    ORDER_CANCELLED = "order_cancelled" # 订单撤销
    ORDER_REJECTED = "order_rejected"   # 订单拒绝

    # 风控事件
    RISK_CHECK_PASS = "risk_pass"       # 风控通过
    RISK_CHECK_FAIL = "risk_fail"       # 风控拒绝

    # 系统事件
    SYSTEM_STATUS = "system_status"     # 系统状态
    ALERT = "alert"                     # 告警


@dataclass
class Event:
    event_type: EventType
    data: dict
    timestamp: datetime = field(default_factory=datetime.now)
    source: str = ""


class EventBus:
    """事件总线"""

    def __init__(self):
        self._handlers: Dict[EventType, List[Callable]] = defaultdict(list)
        self._queue: queue.Queue = queue.Queue()
        self._running = False
        self._thread = None

    def subscribe(self, event_type: EventType, handler: Callable):
        self._handlers[event_type].append(handler)

    def publish(self, event: Event):
        self._queue.put(event)

    def start(self):
        self._running = True
        self._thread = threading.Thread(target=self._process_loop, daemon=True)
        self._thread.start()

    def stop(self):
        self._running = False

    def _process_loop(self):
        while self._running:
            try:
                event = self._queue.get(timeout=0.1)
                for handler in self._handlers.get(event.event_type, []):
                    try:
                        handler(event)
                    except Exception as e:
                        print(f"事件处理异常 [{event.event_type}]: {e}")
            except queue.Empty:
                continue
```

### 二、模拟行情数据源

```python
"""
模拟行情数据源

生成接近真实的 A 股行情数据
"""

import random
import math
from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import Generator, List, Optional


@dataclass
class TickData:
    symbol: str
    timestamp: datetime
    last_price: float
    volume: int
    bid_price: float
    ask_price: float
    bid_volume: int
    ask_volume: int
    open: float
    high: float
    low: float
    close: float


@dataclass
class BarData:
    symbol: str
    timestamp: datetime
    open: float
    high: float
    low: float
    close: float
    volume: int
    turnover: float


class SimulatedMarketData:
    """模拟行情数据源"""

    def __init__(
        self,
        symbols: List[str],
        initial_prices: Optional[dict] = None,
        volatility: float = 0.02,
        seed: int = 42,
    ):
        self.symbols = symbols
        self.volatility = volatility
        self._prices = initial_prices or {s: 100.0 for s in symbols}
        self._rng = random.Random(seed)
        self._base_time = datetime(2024, 1, 2, 9, 30, 0)

    def generate_tick(self, symbol: str) -> TickData:
        """生成一个 tick 数据"""
        price = self._prices[symbol]
        change = self._rng.gauss(0, self.volatility * price * 0.01)
        new_price = max(price * 0.9, min(price * 1.1, price + change))
        self._prices[symbol] = round(new_price, 2)

        spread = max(0.01, new_price * 0.001)
        bid = round(new_price - spread / 2, 2)
        ask = round(new_price + spread / 2, 2)

        return TickData(
            symbol=symbol,
            timestamp=self._base_time,
            last_price=new_price,
            volume=self._rng.randint(100, 10000),
            bid_price=bid,
            ask_price=ask,
            bid_volume=self._rng.randint(100, 5000),
            ask_volume=self._rng.randint(100, 5000),
            open=round(new_price * (1 + self._rng.gauss(0, 0.005)), 2),
            high=round(new_price * (1 + abs(self._rng.gauss(0, 0.01))), 2),
            low=round(new_price * (1 - abs(self._rng.gauss(0, 0.01))), 2),
            close=new_price,
        )

    def generate_bar(self, symbol: str, interval_seconds: int = 60) -> BarData:
        """生成一根 K 线"""
        price = self._prices[symbol]
        change = self._rng.gauss(0.0001, self.volatility * 0.1)
        close = round(price * (1 + change), 2)
        high = round(max(price, close) * (1 + abs(self._rng.gauss(0, 0.005))), 2)
        low = round(min(price, close) * (1 - abs(self._rng.gauss(0, 0.005))), 2)

        self._prices[symbol] = close
        self._base_time += timedelta(seconds=interval_seconds)

        return BarData(
            symbol=symbol,
            timestamp=self._base_time,
            open=price,
            high=high,
            low=low,
            close=close,
            volume=self._rng.randint(10000, 500000),
            turnover=round(close * self._rng.randint(10000, 500000), 2),
        )

    def stream_bars(self, symbol: str,
                    interval_seconds: int = 60) -> Generator:
        """持续生成 K 线流"""
        while True:
            yield self.generate_bar(symbol, interval_seconds)
```

### 三、执行引擎与模拟 CTP

```python
"""
事件驱动执行引擎 + 模拟 CTP 接口
"""

import uuid
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Dict, List, Optional, Callable
from collections import defaultdict


class OrderSide(Enum):
    BUY = "buy"
    SELL = "sell"


class OrderType(Enum):
    MARKET = "market"
    LIMIT = "limit"


class OrderStatus(Enum):
    CREATED = "created"
    SUBMITTED = "submitted"
    PARTIALLY_FILLED = "partially_filled"
    FILLED = "filled"
    CANCELLED = "cancelled"
    REJECTED = "rejected"


@dataclass
class Order:
    order_id: str
    symbol: str
    side: OrderSide
    quantity: int
    order_type: OrderType = OrderType.MARKET
    limit_price: float = 0.0
    filled_quantity: int = 0
    avg_fill_price: float = 0.0
    status: OrderStatus = OrderStatus.CREATED
    created_at: datetime = field(default_factory=datetime.now)
    updated_at: datetime = field(default_factory=datetime.now)
    strategy_name: str = ""
    slippage_bps: float = 0.0
    commission: float = 0.0


@dataclass
class Position:
    symbol: str
    quantity: int = 0
    avg_cost: float = 0.0
    unrealized_pnl: float = 0.0
    realized_pnl: float = 0.0

    @property
    def market_value(self) -> float:
        return self.quantity * self.avg_cost


@dataclass
class Account:
    initial_capital: float = 1000000.0
    cash: float = 1000000.0
    positions: Dict[str, Position] = field(default_factory=dict)
    total_commission: float = 0.0
    total_slippage: float = 0.0

    @property
    def equity(self) -> float:
        position_value = sum(p.market_value for p in self.positions.values())
        return self.cash + position_value

    @property
    def total_pnl(self) -> float:
        return self.equity - self.initial_capital


class SlippageModel:
    """滑点模拟模型"""

    def __init__(
        self,
        base_slippage_bps: float = 2.0,
        volume_impact_factor: float = 0.1,
        volatility_factor: float = 0.5,
    ):
        self.base_slippage_bps = base_slippage_bps
        self.volume_impact_factor = volume_impact_factor
        self.volatility_factor = volatility_factor

    def calculate(
        self, price: float, quantity: int, daily_volume: int,
        volatility: float = 0.02, side: str = "buy",
    ) -> tuple:
        """
        计算滑点
        返回：(实际成交价, 滑点金额, 滑点 bps)
        """
        base_slip = self.base_slippage_bps / 10000

        volume_ratio = quantity / max(daily_volume, 1)
        volume_impact = self.volume_impact_factor * math.sqrt(volume_ratio)

        vol_impact = self.volatility_factor * volatility

        total_slippage_pct = base_slip + volume_impact + vol_impact
        slippage_amount = price * total_slippage_pct

        if side == "buy":
            fill_price = price + slippage_amount
        else:
            fill_price = price - slippage_amount

        slippage_bps = total_slippage_pct * 10000
        return round(fill_price, 2), round(slippage_amount, 4), round(slippage_bps, 2)


class CommissionModel:
    """手续费模型"""

    def __init__(
        self,
        rate: float = 0.0003,
        min_commission: float = 5.0,
        stamp_tax_rate: float = 0.001,
    ):
        self.rate = rate
        self.min_commission = min_commission
        self.stamp_tax_rate = stamp_tax_rate

    def calculate(self, price: float, quantity: int,
                  side: str) -> float:
        amount = price * quantity
        commission = max(amount * self.rate, self.min_commission)
        if side == "sell":
            commission += amount * self.stamp_tax_rate
        return round(commission, 2)


class SimulatedCTP:
    """模拟 CTP 交易接口"""

    def __init__(
        self,
        account: Account,
        slippage_model: SlippageModel = None,
        commission_model: CommissionModel = None,
    ):
        self.account = account
        self.slippage_model = slippage_model or SlippageModel()
        self.commission_model = commission_model or CommissionModel()
        self._current_prices: Dict[str, float] = {}
        self._daily_volumes: Dict[str, int] = {}
        self._volatilities: Dict[str, float] = {}

    def update_market(self, symbol: str, price: float,
                      volume: int = 100000, volatility: float = 0.02):
        self._current_prices[symbol] = price
        self._daily_volumes[symbol] = volume
        self._volatilities[symbol] = volatility

    def submit_order(self, order: Order) -> Order:
        """提交订单到模拟交易所"""
        price = self._current_prices.get(order.symbol)
        if not price:
            order.status = OrderStatus.REJECTED
            return order

        if order.order_type == OrderType.LIMIT:
            if order.side == OrderSide.BUY and order.limit_price < price:
                order.status = OrderStatus.REJECTED
                return order
            if order.side == OrderSide.SELL and order.limit_price > price:
                order.status = OrderStatus.REJECTED
                return order

        order.status = OrderStatus.SUBMITTED
        return self._execute_order(order, price)

    def _execute_order(self, order: Order, market_price: float) -> Order:
        side_str = order.side.value
        quantity = order.quantity - order.filled_quantity

        fill_price, slip_amount, slip_bps = self.slippage_model.calculate(
            price=market_price,
            quantity=quantity,
            daily_volume=self._daily_volumes.get(order.symbol, 100000),
            volatility=self._volatilities.get(order.symbol, 0.02),
            side=side_str,
        )

        commission = self.commission_model.calculate(
            fill_price, quantity, side_str
        )

        cost = fill_price * quantity + commission
        if order.side == OrderSide.BUY:
            if self.account.cash < cost:
                order.status = OrderStatus.REJECTED
                return order
            self.account.cash -= cost
            self._update_position(order.symbol, quantity, fill_price, "buy")
        else:
            pos = self.account.positions.get(order.symbol)
            if not pos or pos.quantity < quantity:
                order.status = OrderStatus.REJECTED
                return order
            self.account.cash += fill_price * quantity - commission
            self._update_position(order.symbol, quantity, fill_price, "sell")

        order.filled_quantity = order.quantity
        order.avg_fill_price = fill_price
        order.status = OrderStatus.FILLED
        order.slippage_bps = slip_bps
        order.commission = commission
        order.updated_at = datetime.now()

        self.account.total_commission += commission
        self.account.total_slippage += slip_amount * quantity

        return order

    def _update_position(self, symbol: str, quantity: int,
                         price: float, side: str):
        if symbol not in self.account.positions:
            self.account.positions[symbol] = Position(symbol=symbol)

        pos = self.account.positions[symbol]

        if side == "buy":
            total_cost = pos.avg_cost * pos.quantity + price * quantity
            pos.quantity += quantity
            pos.avg_cost = total_cost / pos.quantity if pos.quantity > 0 else 0
        else:
            realized = (price - pos.avg_cost) * quantity
            pos.realized_pnl += realized
            pos.quantity -= quantity
            if pos.quantity == 0:
                pos.avg_cost = 0.0

    def cancel_order(self, order: Order) -> Order:
        order.status = OrderStatus.CANCELLED
        order.updated_at = datetime.now()
        return order
```

### 四、策略引擎（双均线示例）

```python
"""
示例策略引擎：双均线策略
"""

from dataclasses import dataclass, field
from typing import Dict, List, Optional
from collections import deque


@dataclass
class StrategySignal:
    strategy_name: str
    symbol: str
    side: str  # "buy" or "sell"
    strength: float  # 0.0 - 1.0
    price: float
    quantity: int
    reason: str
    timestamp: object = None


class DualMovingAverageStrategy:
    """双均线策略"""

    def __init__(
        self,
        name: str = "dual_ma",
        short_window: int = 5,
        long_window: int = 20,
        position_size_pct: float = 0.1,
    ):
        self.name = name
        self.short_window = short_window
        self.long_window = long_window
        self.position_size_pct = position_size_pct
        self._prices: Dict[str, deque] = {}
        self._signals: List[StrategySignal] = []

    def on_bar(self, symbol: str, close: float,
               equity: float = 1000000) -> Optional[StrategySignal]:
        """收到新 K 线时调用"""
        if symbol not in self._prices:
            self._prices[symbol] = deque(maxlen=self.long_window + 1)
        self._prices[symbol].append(close)

        prices = self._prices[symbol]
        if len(prices) < self.long_window:
            return None

        short_ma = sum(list(prices)[-self.short_window:]) / self.short_window
        long_ma = sum(prices) / self.long_window

        prev_prices = list(prices)[:-1]
        if len(prev_prices) < self.long_window:
            return None
        prev_short_ma = sum(prev_prices[-self.short_window:]) / self.short_window
        prev_long_ma = sum(prev_prices) / self.long_window

        signal = None
        if prev_short_ma <= prev_long_ma and short_ma > long_ma:
            quantity = int(equity * self.position_size_pct / close / 100) * 100
            if quantity > 0:
                signal = StrategySignal(
                    strategy_name=self.name,
                    symbol=symbol,
                    side="buy",
                    strength=min(1.0, (short_ma - long_ma) / long_ma * 100),
                    price=close,
                    quantity=quantity,
                    reason=f"金叉: MA{self.short_window}({short_ma:.2f}) > MA{self.long_window}({long_ma:.2f})",
                )
        elif prev_short_ma >= prev_long_ma and short_ma < long_ma:
            signal = StrategySignal(
                strategy_name=self.name,
                symbol=symbol,
                side="sell",
                strength=min(1.0, (long_ma - short_ma) / long_ma * 100),
                price=close,
                quantity=0,  # 卖出时数量由持仓决定
                reason=f"死叉: MA{self.short_window}({short_ma:.2f}) < MA{self.long_window}({long_ma:.2f})",
            )

        if signal:
            self._signals.append(signal)
        return signal


class MomentumStrategy:
    """动量策略"""

    def __init__(
        self,
        name: str = "momentum",
        lookback: int = 10,
        threshold: float = 0.03,
        position_size_pct: float = 0.1,
    ):
        self.name = name
        self.lookback = lookback
        self.threshold = threshold
        self.position_size_pct = position_size_pct
        self._prices: Dict[str, deque] = {}
        self._signals: List[StrategySignal] = []

    def on_bar(self, symbol: str, close: float,
               equity: float = 1000000) -> Optional[StrategySignal]:
        if symbol not in self._prices:
            self._prices[symbol] = deque(maxlen=self.lookback + 1)
        self._prices[symbol].append(close)

        prices = self._prices[symbol]
        if len(prices) < self.lookback + 1:
            return None

        returns = (close - prices[0]) / prices[0]

        signal = None
        if returns > self.threshold:
            quantity = int(equity * self.position_size_pct / close / 100) * 100
            if quantity > 0:
                signal = StrategySignal(
                    strategy_name=self.name,
                    symbol=symbol,
                    side="buy",
                    strength=min(1.0, returns / self.threshold),
                    price=close,
                    quantity=quantity,
                    reason=f"动量突破: {self.lookback}日涨幅 {returns:.2%}",
                )
        elif returns < -self.threshold:
            signal = StrategySignal(
                strategy_name=self.name,
                symbol=symbol,
                side="sell",
                strength=min(1.0, abs(returns) / self.threshold),
                price=close,
                quantity=0,
                reason=f"动量反转: {self.lookback}日跌幅 {returns:.2%}",
            )

        if signal:
            self._signals.append(signal)
        return signal
```

### 五、风控模块

```python
"""
交易风控模块
"""

from dataclasses import dataclass
from typing import Optional


@dataclass
class RiskConfig:
    max_position_pct: float = 0.3       # 单只股票最大持仓比例
    max_daily_loss_pct: float = 0.02    # 日最大亏损比例
    max_drawdown_pct: float = 0.05      # 最大回撤比例
    max_order_value: float = 200000.0   # 单笔订单最大金额
    max_orders_per_minute: int = 10     # 每分钟最大下单数
    min_cash_ratio: float = 0.1         # 最低现金比例


class RiskManager:
    """风控管理器"""

    def __init__(self, config: RiskConfig = None):
        self.config = config or RiskConfig()
        self._order_timestamps = []
        self._daily_pnl = 0.0
        self._peak_equity = 0.0

    def check_order(
        self, symbol: str, side: str, quantity: int,
        price: float, account_equity: float, cash: float,
        position_value: float = 0,
    ) -> tuple:
        """
        检查订单是否通过风控
        返回：(通过, 原因)
        """
        order_value = price * quantity

        if order_value > self.config.max_order_value:
            return False, f"订单金额 {order_value:.0f} 超过限制 {self.config.max_order_value:.0f}"

        position_ratio = position_value / account_equity
        if side == "buy" and position_ratio > self.config.max_position_pct:
            return False, f"持仓比例 {position_ratio:.1%} 超过限制 {self.config.max_position_pct:.1%}"

        if side == "buy":
            cash_after = cash - order_value
            if cash_after / account_equity < self.config.min_cash_ratio:
                return False, f"买入后现金比例将低于 {self.config.min_cash_ratio:.1%}"

        if self._daily_pnl < 0:
            loss_ratio = abs(self._daily_pnl) / account_equity
            if loss_ratio > self.config.max_daily_loss_pct:
                return False, f"日亏损 {loss_ratio:.2%} 已超过限制 {self.config.max_daily_loss_pct:.1%}"

        if self._peak_equity > 0:
            drawdown = (self._peak_equity - account_equity) / self._peak_equity
            if drawdown > self.config.max_drawdown_pct:
                return False, f"回撤 {drawdown:.2%} 已超过限制 {self.config.max_drawdown_pct:.1%}"

        return True, "风控通过"

    def update_equity(self, equity: float):
        self._peak_equity = max(self._peak_equity, equity)

    def update_daily_pnl(self, pnl: float):
        self._daily_pnl = pnl

    def reset_daily(self):
        self._daily_pnl = 0.0
```

```python
@dataclass
class DashboardServer:
    """简易 Dashboard 服务器"""

    host: str = "localhost"
    port: int = 8765
    _clients: Set = field(default_factory=set)

    async def _handler(self, websocket, path=None):
        self._clients.add(websocket)
        try:
            async for _ in websocket:
                pass
        finally:
            self._clients.discard(websocket)

    async def _broadcast(self, message: dict):
        if not self._clients:
            return
        payload = json.dumps(message, ensure_ascii=False, default=str)
        dead = set()
        for client in self._clients:
            try:
                await client.send(payload)
            except Exception:
                dead.add(client)
        self._clients -= dead

    def push(self, msg_type: str, data: dict):
        message = {
            "type": msg_type,
            "timestamp": datetime.now().isoformat(),
            "data": data,
        }
        try:
            loop = asyncio.get_event_loop()
            if loop.is_running():
                asyncio.ensure_future(self._broadcast(message))
            else:
                loop.run_until_complete(self._broadcast(message))
        except RuntimeError:
            pass

    def start(self):
        if not HAS_WEBSOCKETS:
            print("WebSocket 不可用，Dashboard 未启动")
            return

        async def serve():
            ws_server = await websockets.serve(self._handler, self.host, self.port)
            print(f"Dashboard WebSocket: ws://{self.host}:{self.port}")

            from aiohttp import web
            app = web.Application()
            app.router.add_get("/", self._serve_html)
            runner = web.AppRunner(app)
            await runner.setup()
            site = web.TCPSite(runner, self.host, self.port + 1)
            await site.start()
            print(f"Dashboard HTTP: http://{self.host}:{self.port + 1}")
            await asyncio.Event().wait()

        thread = threading.Thread(
            target=lambda: asyncio.run(serve()), daemon=True
        )
        thread.start()
```

```python
import threading

class SimulatedTradingSystem:
    """模拟实盘交易系统"""

    def __init__(self, config: dict = None):

        # 初始化组件
        self.event_bus = EventBus()
        self.account = Account(

        symbols = config.get("symbols", ["600519.SH", "000858.SZ", "601318.SH"])

        )

        # 衰减检测
        self.decay_detector = StrategyDecayDetector()

        # 统计
        self._trade_count = 0
        self._signal_count = 0
        self._running = False

        # 注册事件处理
        self.event_bus.subscribe(EventType.SIGNAL, self._on_signal)
        self.event_bus.subscribe(EventType.ORDER_FILLED, self._on_order_filled)

    def start(self):
        """启动系统"""
        self._running = True
        self.event_bus.start()
        self.dashboard.start()
        print("模拟交易系统启动")

        # 启动行情循环
        thread = threading.Thread(target=self._market_loop, daemon=True)
        thread.start()

    def stop(self):
        self._running = False
        self.event_bus.stop()
        print("模拟交易系统停止")

    def _market_loop(self):
        """行情驱动循环"""
        symbols = list(self.market_data._prices.keys())

        while self._running:
            for symbol in symbols:
                bar = self.market_data.generate_bar(symbol, interval_seconds=60)

                self.ctp.update_market(
                    symbol, bar.close,
                    volume=bar.volume,
                    volatility=0.02,
                )

                self.event_bus.publish(Event(EventType.BAR_CLOSED, {
                    "symbol": symbol,
                    "close": bar.close,
                    "volume": bar.volume,
                    "timestamp": bar.timestamp,
                }))

                for strategy in self.strategies.values():
                    signal = strategy.on_bar(
                        symbol, bar.close, self.account.equity
                    )
                    if signal:
                        self.event_bus.publish(Event(
                            EventType.SIGNAL,
                            signal.__dict__,
                            source=strategy.name,
                        ))

            self._push_dashboard_update()
            time.sleep(0.1)

    def _on_signal(self, event: Event):
        """处理策略信号"""
        data = event.data
        symbol = data["symbol"]
        side = data["side"]
        price = data["price"]

        if side == "sell":
            pos = self.account.positions.get(symbol)
            if not pos or pos.quantity <= 0:
                return
            quantity = pos.quantity
        else:
            quantity = data.get("quantity", 100)

        passed, reason = self.risk_mgr.check_order(
            symbol=symbol,
            side=side,
            quantity=quantity,
            price=price,
            account_equity=self.account.equity,
            cash=self.account.cash,
            position_value=self.account.positions.get(
                symbol, Position(symbol)
            ).market_value,
        )

        if not passed:
            self.event_bus.publish(Event(EventType.RISK_CHECK_FAIL, {
                "symbol": symbol, "reason": reason,
            }))
            return

        order = Order(
            order_id=f"ORD_{self._trade_count:06d}",
            symbol=symbol,
            side=OrderSide(side),
            quantity=quantity,
            strategy_name=data.get("strategy_name", ""),
        )
        self._trade_count += 1

                "order_id": filled_order.order_id,
                "symbol": symbol,
                "side": side,
                "quantity": quantity,
                "price": filled_order.avg_fill_price,
                "slippage_bps": filled_order.slippage_bps,
                "commission": filled_order.commission,
                "time": datetime.now().strftime("%H:%M:%S"),
            }))

            self.decay_detector.add_trade_result(
                filled_order.avg_fill_price * quantity *
                (1 if side == "sell" else -1)
            )

        data = event.data
        self.dashboard.push("trade", data)

    def _push_dashboard_update(self):

                    "avg_cost": round(pos.avg_cost, 2),
                    "pnl": round(pnl, 2),
                }

            "equity": round(self.account.equity, 2),
            "cash": round(self.account.cash, 2),

            "commission": round(self.account.total_commission, 2),
            "slippage": round(self.account.total_slippage, 2),
            "positions": positions_data,
        })

        return {
            "account": {
                "equity": round(self.account.equity, 2),
                "cash": round(self.account.cash, 2),
                "total_pnl": round(self.account.total_pnl, 2),
                "return_pct": round(

                "commission": round(self.account.total_commission, 2),
                "slippage_cost": round(self.account.total_slippage, 2),
            },
            "trading": {
                "total_trades": self._trade_count,
                "active_positions": len([
                    p for p in self.account.positions.values() if p.quantity > 0
                ]),
            },
            "strategies": {
                name: {"signals": len(s._signals)}
                for name, s in self.strategies.items()
            },
            "decay_detection": self.decay_detector.get_report(),
        }
```

## 常见误区

### 误区一：模拟环境太理想

模拟交易中不考虑滑点、手续费、流动性限制，结果策略在模拟中大赚，实盘就亏。本课的模拟系统包含了滑点模型和手续费模型，尽量接近真实。

### 误区二：忽略事件驱动的解耦

把所有逻辑写在一个大循环里——行情获取、信号生成、下单、风控全部耦合。一旦某个模块出 bug，整个系统崩溃。事件总线让各模块独立运行，互不影响。

### 误区三：没有告警就上线

模拟系统跑通了就直接上实盘——但模拟中你可能没有配置告警。实盘中 API 断连、风控触发、策略衰减都需要实时告警。

### 误区四：回测和模拟用同一份数据

用训练数据做回测，又用同样的数据做模拟验证——这是自欺欺人。模拟验证应该用回测时**未见过**的数据。

---

## 小结

本课的核心要点：

1. **事件驱动架构**：事件总线解耦各模块，系统更健壮、更易扩展
2. **模拟 CTP**：在不连接真实交易所的情况下模拟完整的交易流程
3. **滑点模拟**：基础滑点 + 成交量冲击 + 波动率影响，接近真实成交
4. **实时 Dashboard**：WebSocket 推送 + HTML 页面，实时监控系统状态
5. **完整闭环**：信号 → 风控 → 执行 → 监控 → 告警 → 迭代

> **风险提示**：模拟交易系统可以帮助你验证策略逻辑和系统架构，但模拟结果**不能**代表实盘表现。实盘面临的真实挑战包括：网络延迟、交易所规则限制、极端行情下的流动性枯竭、系统故障等。建议在模拟系统稳定运行至少一个月后，再考虑接入真实交易。

### 练习一

**思路**：在滑点计算中加入时间段权重和订单簿深度因子。

**答案**：

```python
import math
from datetime import datetime


class EnhancedSlippageModel(SlippageModel):
    """增强版滑点模型"""

    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.open_close_factor = 2.0  # 开盘/收盘滑点放大系数
        self.orderbook_factor = 0.5   # 订单簿深度影响系数

    def calculate(
        self, price: float, quantity: int, daily_volume: int,
        volatility: float = 0.02, side: str = "buy",
        timestamp: datetime = None,
        bid_volume: int = 10000, ask_volume: int = 10000,
    ) -> tuple:
        base_price, base_slip, base_bps = super().calculate(
            price, quantity, daily_volume, volatility, side
        )

        time_factor = 1.0
        if timestamp:
            hour, minute = timestamp.hour, timestamp.minute
            minutes_from_open = (hour - 9) * 60 + minute - 30
            minutes_to_close = (15 - hour) * 60 - minute
            if minutes_from_open < 30 or minutes_to_close < 30:
                time_factor = self.open_close_factor

        opposite_volume = ask_volume if side == "buy" else bid_volume
        volume_ratio = quantity / max(opposite_volume, 1)
        depth_factor = 1.0 + self.orderbook_factor * math.sqrt(volume_ratio)

        total_factor = time_factor * depth_factor
        adjusted_slip = base_slip * total_factor
        adjusted_bps = base_bps * total_factor

        if side == "buy":
            adjusted_price = price + adjusted_slip
        else:
            adjusted_price = price - adjusted_slip

        return round(adjusted_price, 2), round(adjusted_slip, 4), round(adjusted_bps, 2)
```

**要点**：
- 开盘和收盘前 30 分钟是流动性较差的时段，滑点通常更大
- 订单簿深度不足时，大单会"穿透"多档价格，导致更大滑点
- `depth_factor` 使用平方根关系，因为冲击成本与订单量的平方根成正比

```python
def hot_swap_strategy(
    self, old_name: str, new_strategy, migrate_positions: bool = True
):
    """热切换策略"""
    if old_name not in self.strategies:
        raise ValueError(f"策略 {old_name} 不存在")

    old_strategy = self.strategies[old_name]
    swap_time = datetime.now()

    # 1. 记录旧策略的信号队列
    pending_signals = []
    if hasattr(old_strategy, '_signals'):
        pending_signals = list(old_strategy._signals)

    # 2. 迁移状态
    if migrate_positions:
        if hasattr(old_strategy, '_prices'):
            new_strategy._prices = dict(old_strategy._prices)
        if hasattr(old_strategy, '_signals'):
            new_strategy._signals = list(old_strategy._signals)

    # 3. 替换策略
    self.strategies[old_name] = new_strategy

    # 4. 记录切换日志
    swap_record = {
        "timestamp": swap_time.isoformat(),
        "old_strategy": old_name,
        "new_strategy": new_strategy.name,
        "migrated_positions": migrate_positions,
        "pending_signals": len(pending_signals),
        "account_equity": self.account.equity,
    }

    self.event_bus.publish(Event(EventType.SYSTEM_STATUS, {
        "message": f"策略热切换: {old_name} → {new_strategy.name}",
        "swap_record": swap_record,
    }))

    # 5. 处理挂起的信号
    for signal in pending_signals:
        self.event_bus.publish(Event(EventType.SIGNAL, signal))

    return swap_record
```

**要点**：
- 切换前先记录挂起的信号，避免信号丢失
- 迁移价格历史数据，让新策略能立即开始计算指标
- 切换过程通过事件总线广播，Dashboard 可以实时显示切换状态
