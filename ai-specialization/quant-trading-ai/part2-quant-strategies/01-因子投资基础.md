# 01 因子投资基础

## 一个反直觉的现象

按过去一年收益率排序 A 股，你会发现低市净率的股票整体跑赢高市净率的，小市值的跑赢大市值的。这不是巧合——是"因子"在起作用。

```python
import numpy as np
import pandas as pd
from scipy import stats

np.random.seed(42)

def simulate_factor():
    n_stocks, n_months = 500, 60
    bp = np.random.lognormal(-0.5, 0.8, n_stocks)
    bp_norm = (bp - bp.mean()) / bp.std()
    size = -np.log(np.random.lognormal(4.5, 1.2, n_stocks))

    returns = np.zeros((n_months, n_stocks))
    for t in range(n_months):
        mkt = np.random.normal(0.008, 0.04)
        for i in range(n_stocks):
            returns[t, i] = 0.001 + mkt + bp_norm[i]*0.004 + size[i]*0.003 + np.random.normal(0, 0.06)

    avg_ret = pd.DataFrame(returns).mean(axis=0)
    groups = pd.qcut(bp, 5, labels=['G1(最贵)','G2','G3','G4','G5(最便宜)'])
    group_stats = pd.DataFrame({'group': groups, 'ret': avg_ret}).groupby('group')['ret'].mean()

    print("按市净率分组的月均收益:")
    print(group_stats.round(4).to_string())
    ls = group_stats['G5(最便宜)'] - group_stats['G1(最贵)']
    print(f"\n多空月均收益: {ls:.4%}, 年化: {ls*12:.2%}")

simulate_factor()
```

核心假设：**市场并非完全有效，某些特征能持续预测超额收益**。

## 截面回归

```python
def cross_section_demo():
    np.random.seed(42)
    n_stocks, n_months = 200, 60
    bp = np.random.randn(n_stocks)
    size = np.random.randn(n_stocks)
    mom = np.random.randn(n_stocks)
    true_premiums = {'价值': 0.005, '规模': 0.003, '动量': 0.004}

    returns = np.zeros((n_months, n_stocks))
    for t in range(n_months):
        for i in range(n_stocks):
            returns[t, i] = 0.001 + bp[i]*0.005 + size[i]*0.003 + mom[i]*0.004 + np.random.normal(0, 0.05)

    estimated = []
    for t in range(n_months):
        X = np.column_stack([np.ones(n_stocks), bp, size, mom])
        betas, _, _, _ = np.linalg.lstsq(X, returns[t], rcond=None)
        estimated.append(betas[1:])

    est = pd.DataFrame(estimated, columns=['价值','规模','动量'])
    print("截面回归估计的因子收益:")
    for name, true_val in true_premiums.items():
        t_stat, p = stats.ttest_1samp(est[name], 0)
        print(f"  {name}: 真实={true_val:.4%}, 估计={est[name].mean():.4%}, p={p:.4f}")

cross_section_demo()
```

截面回归的含义：**同一时间点上，哪些特征解释了股票间的收益差异**。

## Fama-French 三因子模型

```
R_i - R_f = α + β_mkt × (R_m - R_f) + β_smb × SMB + β_hml × HML + ε
```

- **市场因子**：承担市场风险的补偿
- **规模因子 (SMB)**：小盘股的额外回报（流动性差、信息不透明）
- **价值因子 (HML)**：价值股的额外回报（困境公司需要风险补偿）

```python
def ff_simulation():
    np.random.seed(42)
    n_months = 120
    mkt = np.random.normal(0.008, 0.04, n_months)
    smb = np.random.normal(0.003, 0.03, n_months)
    hml = -0.3 * smb + np.random.normal(0, 0.02, n_months)  # SMB和HML负相关

    df = pd.DataFrame({'MKT': mkt, 'SMB': smb, 'HML': hml})
    print("因子统计:")
    print(pd.DataFrame({
        '年化收益': df.mean()*12, '年化波动': df.std()*np.sqrt(12),
        '夏普': df.mean()/df.std()*np.sqrt(12),
    }).round(3).to_string())
    print(f"\n因子相关性:\n{df.corr().round(2).to_string()}")

ff_simulation()
```

## 因子溢价的来源

两种主流解释：

- **风险补偿** (Fama-French)：因子收益是对承担额外风险的补偿，不会消失
- **行为偏差** (Shiller)：投资者系统性犯错，因子收益可能消失

实际中两种力量同时存在。对投资者的启示：**无论哪种理论正确，只要因子溢价持续存在，因子投资就有价值**。

## 因子动物园的问题

研究者已发现 400+ 个因子（"因子动物园"），但大部分是数据挖掘的产物。真正被经济学理论支撑、在多个市场和时间段验证的因子不超过 10 个。盲目堆叠因子会导致过拟合。

```python
def factor_zoo_demo():
    """模拟因子发现中的多重检验问题"""
    np.random.seed(42)
    n_factors = 100; n_months = 60; n_stocks = 200
    significant = 0
    for _ in range(n_factors):
        fake_factor = np.random.randn(n_stocks)
        returns = np.random.randn(n_months, n_stocks) * 0.05
        # 检验这个假因子是否"显著"
        ic_list = []
        for t in range(n_months):
            ic = np.corrcoef(fake_factor, returns[t])[0,1]
            ic_list.append(ic)
        from scipy import stats
        _, p = stats.ttest_1samp(ic_list, 0)
        if p < 0.05: significant += 1
    print(f"测试 {n_factors} 个随机因子, {significant} 个在 5% 水平下'显著'")
    print(f"预期假阳性: {n_factors * 0.05:.0f} 个")

factor_zoo_demo()
```

## 因子 vs 技术指标

| 维度 | 因子 | 技术指标 |
|------|------|----------|
| 使用方式 | 截面排序（比较不同股票） | 时序信号（单只股票历史） |
| 持有期 | 月度/季度 | 日度/周度 |
| 经济解释 | 需要 | 不一定 |

动量既是因子也是技术指标——区别在于使用方式。

```python
def value_by_regime():
    np.random.seed(42)
    n_months, n_stocks = 120, 200
    mkt_ret = np.random.normal(0.008, 0.04, n_months)
    bp = np.random.randn(n_stocks)
    returns = np.zeros((n_months, n_stocks))
    for t in range(n_months):
        for i in range(n_stocks):
            returns[t, i] = mkt_ret[t] + bp[i]*0.005 + np.random.normal(0, 0.05)
    bull = mkt_ret > np.median(mkt_ret)
    for label, mask in [('牛市', bull), ('熊市', ~bull)]:
        avg = returns[mask].mean(axis=0)
        groups = pd.qcut(bp, 5, labels=False)
        ls = avg[groups==4].mean() - avg[groups==0].mean()
        print(f"{label}: 多空收益 = {ls:.4%}")
```
