```python
import numpy as np

def generate_market_data(n_stocks=300, n_days=750):
    """
    生成模拟市场数据

    包含：日收益率、基本面数据、行业分类、市值
    模拟 3 年的交易数据（250 交易日/年）
    """
    dates = pd.date_range('2022-01-01', periods=n_days, freq='B')
    stock_ids = [f'STOCK_{i:03d}' for i in range(n_stocks)]

    # 市场收益（有牛熊周期）
    market_cycle = np.sin(np.linspace(0, 4 * np.pi, n_days)) * 0.001
    market_return = pd.Series(
        np.random.normal(0.0003, 0.012, n_days) + market_cycle,
        index=dates, name='market'
    )

    # 行业分类
    industry_names = ['金融', '科技', '消费', '医药', '制造', '能源', '地产']
    industries = np.random.choice(industry_names, n_stocks)
    industry_series = pd.Series(industries, index=stock_ids, name='Industry')

    # 市值（对数正态，模拟真实分布）
    market_cap = np.random.lognormal(mean=4.5, sigma=1.2, size=n_stocks)
    market_cap = np.clip(market_cap, 10, 10000)  # 10 亿到 1 万亿
    cap_series = pd.Series(market_cap, index=stock_ids, name='MarketCap')

    # 个股收益率（带行业 beta 和因子暴露）
    betas = np.random.uniform(0.5, 1.5, n_stocks)
    industry_beta = {ind: np.random.uniform(-0.001, 0.001) for ind in industry_names}

    returns = pd.DataFrame(index=dates, columns=stock_ids, dtype=float)
    for i, stock in enumerate(stock_ids):
        ind = industry_series[stock]
        epsilon = np.random.normal(0, 0.018, n_days)
        returns[stock] = betas[i] * market_return + industry_beta[ind] + epsilon

    # 基本面数据（季度更新，这里简化为静态）
    fundamentals = pd.DataFrame(index=stock_ids)
    fundamentals['PB'] = np.random.lognormal(0.7, 0.6, n_stocks)
    fundamentals['PE'] = np.random.lognormal(2.5, 0.8, n_stocks)
    fundamentals['ROE'] = np.random.normal(0.10, 0.08, n_stocks)
    fundamentals['GrossMargin'] = np.random.beta(3, 4, n_stocks)
    fundamentals['DebtRatio'] = np.random.beta(3, 5, n_stocks)
    fundamentals['DividendYield'] = np.random.beta(2, 8, n_stocks) * 0.06

    return {
        'returns': returns,
        'market_return': market_return,
        'industry': industry_series,
        'market_cap': cap_series,
        'fundamentals': fundamentals,
        'dates': dates,
        'stock_ids': stock_ids
    }


def calculate_all_factors(data, lookback=60):
    """
    计算所有候选因子

    包含：
    1. 动量因子（12-1 月动量）
    2. 短期反转因子（5 日反转）
    3. 价值因子（BP）
    4. 盈利因子（ROE）
    5. 质量因子（毛利率）
    6. 低杠杆因子
    7. 低波动因子
    8. 换手率因子
    9. 偏度因子
    10. 最大日收益因子

    返回：
        factors: 因子值字典 {因子名: DataFrame}
    """
    returns = data['returns']
    fundamentals = data['fundamentals']
    stock_ids = data['stock_ids']
    dates = data['dates']
    n_days = len(dates)
    n_stocks = len(stock_ids)

    factors = {}

    # 1. 动量因子（12-1 月动量，跳过最近 1 月）
    momentum = pd.DataFrame(index=dates, columns=stock_ids, dtype=float)
    for t in range(252, n_days):
        ret_12m = (1 + returns.iloc[t-252:t-21]).prod() - 1
        momentum.iloc[t] = ret_12m.values
    factors['动量'] = momentum

    # 2. 短期反转因子（5 日反转）
    reversal = returns.rolling(5).sum()
    factors['短期反转'] = reversal

    # 3. 价值因子（BP = 1/PB）
    bp = 1.0 / fundamentals['PB']
    bp_df = pd.DataFrame(index=dates, columns=stock_ids, dtype=float)
    for t in range(n_days):
        bp_df.iloc[t] = bp.values
    factors['价值'] = bp_df

    # 4. 盈利因子（ROE）
    roe_df = pd.DataFrame(index=dates, columns=stock_ids, dtype=float)
    for t in range(n_days):
        roe_df.iloc[t] = fundamentals['ROE'].values
    factors['盈利'] = roe_df

    # 5. 质量因子（毛利率）
    gm_df = pd.DataFrame(index=dates, columns=stock_ids, dtype=float)
    for t in range(n_days):
        gm_df.iloc[t] = fundamentals['GrossMargin'].values
    factors['质量'] = gm_df

    # 6. 低杠杆因子
    lev_df = pd.DataFrame(index=dates, columns=stock_ids, dtype=float)
    for t in range(n_days):
        lev_df.iloc[t] = -fundamentals['DebtRatio'].values  # 取负号
    factors['低杠杆'] = lev_df

    # 7. 低波动因子（60 日波动率）
    volatility = returns.rolling(lookback).std() * np.sqrt(252)
    factors['低波动'] = -volatility  # 取负号

    # 8. 换手率因子（模拟）
    turnover = pd.DataFrame(
        np.random.lognormal(-3, 0.5, (n_days, n_stocks)),
        index=dates, columns=stock_ids
    )
    factors['低换手'] = -turnover  # 取负号

    # 9. 偏度因子
    skewness = returns.rolling(lookback).skew()
    factors['低偏度'] = -skewness  # 取负号

    # 10. 最大日收益因子
    max_ret = returns.rolling(lookback).max()
    factors['低最大收益'] = -max_ret  # 取负号

    return factors


def winsorize_and_standardize(factor_df):
    """
    因子清洗：去极值 + 标准化

    对每个截面（每天）做 MAD 去极值和 Z-score 标准化
    """
    cleaned = factor_df.copy()
    n_periods = len(cleaned)

    for t in range(n_periods):
        row = cleaned.iloc[t].dropna()
        if len(row) < 30:
            continue

        # MAD 去极值
        median = row.median()
        mad = (row - median).abs().median()
        upper = median + 5 * 1.4826 * mad
        lower = median - 5 * 1.4826 * mad
        row = row.clip(lower, upper)

        # Z-score 标准化
        mean = row.mean()
        std = row.std()
        if std > 0:
            cleaned.iloc[t] = (row - mean) / std

    return cleaned
```

```python
def evaluate_all_factors(factors_clean, returns, forward_days=20):
    """
    评估所有因子

    计算每个因子的 IC、IC_IR、分层回测收益

    参数：
        factors_clean: 清洗后的因子字典
        returns: 日收益率 DataFrame
        forward_days: 未来收益天数

    返回：
        eval_results: 评估结果 DataFrame
    """
    eval_results = []

    for name, factor_df in factors_clean.items():
        # 计算 IC
        ic_list = []
        n_periods = len(factor_df)

        for t in range(252, n_periods - forward_days):
            factor_t = factor_df.iloc[t].dropna()
            # 计算未来 forward_days 天的收益
            future_ret = returns.iloc[t:t+forward_days].sum()
            future_ret = future_ret.dropna()

            common = factor_t.index.intersection(future_ret.index)
            if len(common) < 50:
                continue

            ic = factor_t[common].corr(future_ret[common], method='spearman')
            ic_list.append(ic)

        if len(ic_list) < 10:
            continue

        ic_series = pd.Series(ic_list)
        ic_mean = ic_series.mean()
        ic_std = ic_series.std()
        ic_ir = ic_mean / ic_std if ic_std > 0 else 0
        ic_positive = (ic_series > 0).mean()

        # t 检验
        t_stat, p_value = stats.ttest_1samp(ic_list, 0)

        # 分层回测（简化版）
        group_returns_list = {f'G{g+1}': [] for g in range(5)}
        for t in range(252, n_periods - forward_days):
            factor_t = factor_df.iloc[t].dropna()
            future_ret = returns.iloc[t:t+forward_days].sum()
            future_ret = future_ret.dropna()
            common = factor_t.index.intersection(future_ret.index)
            if len(common) < 50:
                continue

            rank_pct = factor_t[common].rank(pct=True)
            for g in range(5):
                lower = g / 5
                upper = (g + 1) / 5
                mask = (rank_pct > lower) & (rank_pct <= upper)
                if g == 0:
                    mask = (rank_pct >= lower) & (rank_pct <= upper)
                group_stocks = common[mask]
                group_returns_list[f'G{g+1}'].append(future_ret[group_stocks].mean())

        # 多空收益
        long_short = np.mean(group_returns_list['G5']) - np.mean(group_returns_list['G1'])
        ls_vol = np.std([group_returns_list['G5'][i] - group_returns_list['G1'][i]
                         for i in range(len(group_returns_list['G1']))])
        ls_sharpe = long_short / ls_vol * np.sqrt(252 / forward_days) if ls_vol > 0 else 0

        eval_results.append({
            '因子': name,
            'IC均值': ic_mean,
            'IC标准差': ic_std,
            'IC_IR': ic_ir,
            'IC>0占比': ic_positive,
            't统计量': t_stat,
            'p值': p_value,
            '多空收益': long_short * 252 / forward_days,
            '多空夏普': ls_sharpe
        })

    return pd.DataFrame(eval_results)


def select_factors(eval_results, ic_ir_threshold=0.3, p_value_threshold=0.05):
    """
    因子筛选

    筛选标准：
    1. IC_IR > threshold
    2. p 值 < threshold
    3. IC 均值方向一致（> 0 或 < 0）
    """
    selected = eval_results[
        (eval_results['IC_IR'].abs() > ic_ir_threshold) &
        (eval_results['p值'] < p_value_threshold)
    ].copy()

    # 按 IC_IR 排序
    selected = selected.sort_values('IC_IR', ascending=False, key=abs)

    return selected
```

```python
def calculate_composite_score(factors_clean, selected_names, returns, lookback=12):
    """
    计算综合因子得分（IC 加权）

    参数：
        factors_clean: 清洗后的因子字典
        selected_names: 选中的因子名列表
        returns: 收益率 DataFrame
        lookback: IC 计算回看窗口

    返回：
        composite_score: 综合得分 DataFrame
    """
    # 计算每个因子的 IC 时间序列
    ic_dict = {}
    for name in selected_names:
        factor_df = factors_clean[name]
        ic_list = []
        n_periods = len(factor_df)

        for t in range(n_periods):
            factor_t = factor_df.iloc[t].dropna()
            # 用 20 日收益计算 IC
            if t + 20 < n_periods:
                future_ret = returns.iloc[t:t+20].sum()
                future_ret = future_ret.dropna()
                common = factor_t.index.intersection(future_ret.index)
                if len(common) >= 50:
                    ic_list.append(factor_t[common].corr(future_ret[common], method='spearman'))
                else:
                    ic_list.append(np.nan)
            else:
                ic_list.append(np.nan)

        ic_dict[name] = pd.Series(ic_list, index=factor_df.index)

    # 计算综合得分
    n_periods = len(list(factors_clean.values())[0])
    stock_ids = list(factors_clean.values())[0].columns
    composite = pd.DataFrame(0, index=factors_clean[selected_names[0]].index,
                             columns=stock_ids, dtype=float)

    weights_over_time = []

    for t in range(lookback, n_periods):
        # 计算每个因子的 IC 权重
        weights = {}
        for name in selected_names:
            ic_window = ic_dict[name].iloc[max(0, t-lookback):t].dropna()
            if len(ic_window) > 0:
                weights[name] = ic_window.mean()
            else:
                weights[name] = 0

        # 归一化（只用正 IC 的因子）
        total = sum(max(0, w) for w in weights.values())
        if total > 0:
            weights = {name: max(0, w) / total for name, w in weights.items()}
        else:
            weights = {name: 1 / len(selected_names) for name in selected_names}

        weights_over_time.append({'date': factors_clean[selected_names[0]].index[t], **weights})

        # 计算综合得分
        for name in selected_names:
            factor_row = factors_clean[name].iloc[t]
            composite.iloc[t] += weights[name] * factor_row.fillna(0)

    weights_df = pd.DataFrame(weights_over_time).set_index('date')
    return composite, weights_df


def optimize_portfolio(composite_score, data, max_weight=0.03, risk_aversion=1.0):
    """
    组合优化

    参数：
        composite_score: 综合得分 DataFrame
        data: 市场数据字典
        max_weight: 单只股票最大权重
        risk_aversion: 风险厌恶系数

    返回：
        portfolio_weights: 每期的组合权重
    """
    returns = data['returns']
    industry = data['industry']
    market_cap = data['market_cap']
    stock_ids = data['stock_ids']
    n_stocks = len(stock_ids)

    # 简化的协方差估计（用历史收益的滚动协方差）
    portfolio_weights = pd.DataFrame(index=composite_score.index, columns=stock_ids, dtype=float)

    for t in range(252, len(composite_score)):
        score_t = composite_score.iloc[t].dropna()
        if len(score_t) < 50:
            continue

        # 选择得分最高的 30 只股票
        top_stocks = score_t.nlargest(30).index

        # 简化优化：按得分加权
        weights = score_t[top_stocks]
        weights = weights - weights.min() + 0.01  # 确保正值
        weights = weights / weights.sum()  # 归一化

        # 应用权重上限
        weights = weights.clip(upper=max_weight)
        weights = weights / weights.sum()  # 重新归一化

        portfolio_weights.iloc[t][top_stocks] = weights.values

    return portfolio_weights
```

```python
def backtest_strategy(portfolio_weights, data, rebalance_freq=20,
                      commission_rate=0.001, slippage=0.001):
    """
    策略回测

    参数：
        portfolio_weights: 组合权重 DataFrame
        data: 市场数据字典
        rebalance_freq: 调仓频率（交易日）
        commission_rate: 佣金率
        slippage: 滑点

    返回：
        results: 回测结果字典
    """
    returns = data['returns']
    dates = data['dates']
    stock_ids = data['stock_ids']

    capital = 1000000
    portfolio_value = []
    holdings = pd.Series(0, index=stock_ids, dtype=float)  # 当前持仓权重
    total_commission = 0
    n_rebalances = 0

    for t in range(252, len(dates)):
        # 计算当日组合收益
        daily_returns = returns.iloc[t]
        portfolio_return = (holdings * daily_returns).sum()

        # 更新资本
        capital *= (1 + portfolio_return)

        # 调仓检查
        if t % rebalance_freq == 0 and t < len(portfolio_weights):
            new_weights = portfolio_weights.iloc[t]
            if new_weights.notna().sum() > 0:
                # 计算换手率
                turnover = (new_weights.fillna(0) - holdings).abs().sum() / 2

                # 交易成本
                commission = turnover * capital * (commission_rate + slippage)
                capital -= commission
                total_commission += commission

                # 更新持仓
                holdings = new_weights.fillna(0)
                n_rebalances += 1

        portfolio_value.append(capital)

    portfolio_series = pd.Series(portfolio_value, index=dates[252:])

    # 计算基准（等权持有所有股票）
    benchmark_returns = returns.iloc[252:].mean(axis=1)
    benchmark_value = (1 + benchmark_returns).cumprod() * 1000000

    # 策略统计
    strategy_returns = portfolio_series.pct_change().dropna()
    total_return = portfolio_series.iloc[-1] / 1000000 - 1
    n_years = len(strategy_returns) / 252
    annual_return = (1 + total_return) ** (1 / n_years) - 1
    annual_vol = strategy_returns.std() * np.sqrt(252)
    sharpe = annual_return / annual_vol if annual_vol > 0 else 0
    max_dd = (portfolio_series / portfolio_series.expanding().max() - 1).min()

    # 基准统计
    bench_total = benchmark_value.iloc[-1] / 1000000 - 1
    bench_annual = (1 + bench_total) ** (1 / n_years) - 1
    excess_return = annual_return - bench_annual

    # 信息比率
    excess_daily = strategy_returns - benchmark_returns.iloc[1:]
    ir = excess_daily.mean() / excess_daily.std() * np.sqrt(252) if excess_daily.std() > 0 else 0

    results = {
        'portfolio': portfolio_series,
```

### 第五步：风险分析

```python
"""
多因子选股策略 - 第五步：风险分析
"""


def risk_analysis(results, data):
    """
    策略风险分析

    包含：
    1. 收益分布分析
    2. 回撤分析
    3. 月度收益分析
    4. 风险归因
    """
    portfolio = results['portfolio']
    benchmark = results['benchmark']
    strategy_returns = portfolio.pct_change().dropna()
    benchmark_returns = benchmark.pct_change().dropna()

    print("=" * 60)
    print("多因子选股策略 - 第五步：风险分析")
    print("=" * 60)

    # 1. 收益分布
    print(f"\n1. 收益分布分析:")
    print(f"  日收益均值: {strategy_returns.mean():.4%}")
    print(f"  日收益标准差: {strategy_returns.std():.4%}")
    print(f"  偏度: {strategy_returns.skew():.4f}")
    print(f"  峰度: {strategy_returns.kurtosis():.4f}")
    print(f"  最大单日盈利: {strategy_returns.max():.2%}")
    print(f"  最大单日亏损: {strategy_returns.min():.2%}")

    # VaR 和 CVaR
    var_95 = strategy_returns.quantile(0.05)
    cvar_95 = strategy_returns[strategy_returns <= var_95].mean()
    print(f"  95% VaR: {var_95:.2%}")
    print(f"  95% CVaR: {cvar_95:.2%}")

    # 2. 回撤分析
    print(f"\n2. 回撤分析:")
    peak = portfolio.expanding().max()
    drawdown = (portfolio - peak) / peak

    # 最大回撤
    max_dd = drawdown.min()
    max_dd_end = drawdown.idxmin()
    max_dd_start = portfolio[:max_dd_end].idxmax()

    print(f"  最大回撤: {max_dd:.2%}")
    print(f"  回撤开始: {max_dd_start.strftime('%Y-%m-%d')}")
    print(f"  回撤最低: {max_dd_end.strftime('%Y-%m-%d')}")

    # 回撤持续时间
    recovery = portfolio[max_dd_end:]
    if len(recovery) > 0:
        recovered = recovery[recovery >= peak[max_dd_end]]
        if len(recovered) > 0:
            recovery_date = recovered.index[0]
            recovery_days = (recovery_date - max_dd_end).days
            print(f"  恢复日期: {recovery_date.strftime('%Y-%m-%d')}")
            print(f"  恢复天数: {recovery_days}")
        else:
            print(f"  尚未恢复")

    # 回撤分布
    dd_periods = drawdown[drawdown < -0.05]
    print(f"  回撤 > 5% 的天数: {len(dd_periods)}")
    print(f"  回撤 > 5% 的时间占比: {len(dd_periods) / len(drawdown):.2%}")

    # 3. 月度收益分析
    print(f"\n3. 月度收益分析:")
    monthly_returns = strategy_returns.resample('ME').apply(lambda x: (1 + x).prod() - 1)

    positive_months = (monthly_returns > 0).sum()
    negative_months = (monthly_returns <= 0).sum()
    print(f"  盈利月数: {positive_months}")
    print(f"  亏损月数: {negative_months}")
    print(f"  月度胜率: {positive_months / len(monthly_returns):.2%}")
    print(f"  月度平均收益: {monthly_returns.mean():.2%}")
    print(f"  月度最大盈利: {monthly_returns.max():.2%}")
    print(f"  月度最大亏损: {monthly_returns.min():.2%}")

    # 4. 年度收益
    print(f"\n4. 年度收益:")
    yearly_returns = strategy_returns.resample('YE').apply(lambda x: (1 + x).prod() - 1)
    for date, ret in yearly_returns.items():
        print(f"  {date.year}: {ret:.2%}")

    # 5. 滚动夏普比率
    print(f"\n5. 滚动夏普比率 (12 个月):")
    rolling_sharpe = strategy_returns.rolling(252).apply(
        lambda x: x.mean() / x.std() * np.sqrt(252) if x.std() > 0 else 0
    )
    print(f"  最小值: {rolling_sharpe.min():.2f}")
    print(f"  最大值: {rolling_sharpe.max():.2f}")
    print(f"  均值: {rolling_sharpe.mean():.2f}")


# ========== 运行第五步 ==========

if 'results' in dir():
    risk_analysis(results, data)
```

## 常见误区

### 误区一：一步到位

很多初学者想一步到位写出完美的策略。实际上，策略开发是迭代的过程：
1. 先用简单方法跑通流程
2. 逐步增加复杂度
3. 每一步都验证结果

### 误区二：忽略数据质量

"垃圾进，垃圾出"——数据质量决定策略质量。要注意：
- 使用事件发生时的数据，避免前视偏差
- 检查数据的完整性（是否有缺失）
- 去除异常值和错误数据

### 误区三：过度优化

过度优化（过拟合）是量化策略最大的敌人。应对方法：
- 用样本外数据验证
- 保持策略简单
- 对参数变化不敏感的策略更可靠

### 误区 四：忽略交易成本

很多看起来很好的策略，在扣除交易成本后就不行了。一定要在回测中加入真实的交易成本估计。

## 小结与练习

### 小结

1. 多因子选股策略的完整流程：因子构建 → 因子评估 → 因子筛选 → 因子加权 → 组合优化 → 回测 → 风险分析
2. 因子选择要基于经济逻辑，不能只靠数据挖掘
3. IC_IR 是评估因子最重要的指标
4. 组合优化要在 Alpha 和风险之间找到平衡
5. 持续监控和调整是策略长期有效的关键

```python
import numpy as np
import pandas as pd

if 'results' in dir():
    portfolio = results['portfolio']
    market_return = data['market_return']

    # 划分市场环境（用市场累计收益的趋势）
    market_cum = (1 + market_return).cumprod()
    market_ma60 = market_cum.rolling(60).mean()

    # 牛市：价格 > 60 日均线且均线向上
    # 熊市：价格 < 60 日均线且均线向下
    # 震荡市：其他
    regime = pd.Series('震荡', index=market_cum.index)
    regime[market_cum > market_ma60] = '牛市'
    regime[market_cum < market_ma60] = '熊市'

    # 分段分析
    strategy_returns = portfolio.pct_change().dropna()

    print("市场环境分析:")
    print(f"{'环境':<8} {'天数':>8} {'策略收益':>12} {'策略波动':>12} {'夏普比率':>12}")
    print("-" * 55)

    for env in ['牛市', '熊市', '震荡']:
        mask = regime[strategy_returns.index] == env
        if mask.sum() > 0:
            env_returns = strategy_returns[mask]
            annual_ret = env_returns.mean() * 252
            annual_vol = env_returns.std() * np.sqrt(252)
            sharpe = annual_ret / annual_vol if annual_vol > 0 else 0
            print(f"{env:<8} {mask.sum():>8} {annual_ret:>12.2%} {annual_vol:>12.2%} {sharpe:>12.2f}")

    print(f"\n策略适应性分析:")
    print("1. 多因子策略通常在震荡市中表现较好（因子有区分度）")
    print("2. 牛市中可能跑输基准（分散化降低了市场暴露）")
```

**要点**：
