# 08 阶段实战：多因子选股策略

## 核心问题

你有 10 个候选因子，每个单独看都有预测力。怎么组合成一个完整的选股策略？

```python
import numpy as np
import pandas as pd
from scipy import stats

np.random.seed(42)

def generate_data(n_stocks=300, n_days=750):
    dates = pd.bdate_range('2022-01-01', periods=n_days)
    stock_ids = [f'S{i:03d}' for i in range(n_stocks)]
    mkt = np.random.normal(0.0003, 0.012, n_days) + np.sin(np.linspace(0, 4*np.pi, n_days))*0.001
    fund = pd.DataFrame({'PB': np.random.lognormal(0.7,0.6,n_stocks), 'ROE': np.random.normal(0.10,0.08,n_stocks),
                          'GM': np.random.beta(3,4,n_stocks), 'Debt': np.random.beta(3,5,n_stocks)}, index=stock_ids)
    betas = np.random.uniform(0.5,1.5,n_stocks)
    returns = pd.DataFrame({s: betas[i]*mkt + np.random.normal(0,0.018,n_days)
                            for i,s in enumerate(stock_ids)}, index=dates)
    return {'returns': returns, 'fund': fund, 'dates': dates, 'ids': stock_ids}

data = generate_data()
```

## Step 1: 计算 10 个候选因子

```python
def calc_factors(data, lookback=60):
    ret = data['returns']; fund = data['fund']; ids = data['ids']; dates = data['dates']
    n = len(dates); factors = {}

    factors['动量'] = pd.DataFrame({s: ((1+ret[s].iloc[max(0,t-252):max(0,t-21)]).prod()-1 if t>=252 else np.nan)
                                    for s in ids for t in range(n)}, index=dates) if False else ret.rolling(252).sum() - ret.rolling(21).sum()
    factors['短期反转'] = ret.rolling(5).sum()
    for col, name in [('PB','价值'),('ROE','盈利'),('GM','质量'),('Debt','低杠杆')]:
        val = 1/fund[col] if col=='PB' else (-fund[col] if col=='Debt' else fund[col])
        factors[name] = pd.DataFrame({s: val.values for s in ids}, index=dates)
    factors['低波动'] = -ret.rolling(lookback).std() * np.sqrt(252)
    factors['低换手'] = -pd.DataFrame(np.random.lognormal(-3,0.5,(n,len(ids))), index=dates, columns=ids)
    factors['低偏度'] = -ret.rolling(lookback).skew()
    factors['低最大收益'] = -ret.rolling(lookback).max()
    return factors

factors = calc_factors(data)
print(f"因子数: {len(factors)}")
```

## Step 2: IC 评估与筛选

```python
def evaluate(factors, returns, fwd=20):
    results = []
    for name, fdf in factors.items():
        ic_list = []
        for t in range(252, len(fdf)-fwd):
            ft = fdf.iloc[t].dropna()
            fr = returns.iloc[t:t+fwd].sum().dropna()
            common = ft.index.intersection(fr.index)
            if len(common) >= 50:
                ic_list.append(ft[common].corr(fr[common], method='spearman'))
        if len(ic_list) < 10: continue
        ic = pd.Series(ic_list)
        # 多空收益
        ls_list = []
        for t in range(252, len(fdf)-fwd):
            ft = fdf.iloc[t].dropna()
            fr = returns.iloc[t:t+fwd].sum().dropna()
            common = ft.index.intersection(fr.index)
            if len(common) < 50: continue
            rank = ft[common].rank(pct=True)
            g5 = fr[common[rank>0.8]].mean()
            g1 = fr[common[rank<=0.2]].mean()
            ls_list.append(g5 - g1)
        ls = np.mean(ls_list) if ls_list else 0
        ls_vol = np.std(ls_list) if ls_list else 1
        results.append({'因子': name, 'IC均值': ic.mean(), 'IC_IR': ic.mean()/ic.std() if ic.std()>0 else 0,
                        'IC>0占比': (ic>0).mean(), 'p值': stats.ttest_1samp(ic_list,0)[1],
                        '多空夏普': ls/ls_vol*np.sqrt(252/fwd) if ls_vol>0 else 0})
    return pd.DataFrame(results)

eval_df = evaluate(factors, data['returns'])
print("因子评估:")
print(eval_df[['因子','IC均值','IC_IR','p值','多空夏普']].to_string(index=False))

selected = eval_df[(eval_df['IC_IR'].abs()>0.15) & (eval_df['p值']<0.15)]
selected_names = selected['因子'].tolist()
print(f"\n选中: {selected_names}")
```

## Step 3: IC 加权 + 回测

```python
def backtest(factors, selected, returns, top_n=30, freq=20, cost=0.002):
    n = len(returns); capital = 1_000_000
    holdings = pd.Series(0.0, index=returns.columns)
    values = []; total_cost = 0

    for t in range(252, n):
        capital *= (1 + (holdings * returns.iloc[t]).sum())
        if t % freq == 0 and selected:
            # IC 加权综合得分
            scores = pd.Series(0.0, index=returns.columns)
            for name in selected:
                scores += factors[name].iloc[t].fillna(0) / len(selected)
            top = scores.nlargest(top_n).index
            new_h = pd.Series(0.0, index=returns.columns)
            new_h[top] = 1.0 / top_n
            turnover = (new_h - holdings).abs().sum() / 2
            capital -= turnover * capital * cost
            total_cost += turnover * capital * cost
            holdings = new_h
        values.append(capital)

    port = pd.Series(values, index=returns.index[252:])
    ret = port.pct_change().dropna()
    total = port.iloc[-1] / 1_000_000 - 1
    years = len(ret) / 252
    ann_ret = (1+total)**(1/years) - 1
    ann_vol = ret.std() * np.sqrt(252)
    sharpe = ann_ret / ann_vol if ann_vol > 0 else 0
    dd = (port / port.expanding().max() - 1).min()
    bench = (1+returns.iloc[252:].mean(axis=1)).cumprod().iloc[-1] - 1

    print(f"\n回测结果:")
    print(f"  总收益: {total:.2%}, 年化: {ann_ret:.2%}, 夏普: {sharpe:.2f}")
    print(f"  最大回撤: {dd:.2%}, 超额: {total-bench:.2%}, 成本: {total_cost:,.0f}")

if selected_names:
    backtest(factors, selected_names, data['returns'])
```

## 关键判断

- **IC_IR 比 IC 均值更重要**——稳定的小 IC 优于不稳定的高 IC
- **5-8 个因子最佳**——太少分散化不足，太多引入噪声
- **月度调仓是合理默认**——更频繁增加成本，更慢错过信号变化
- **单只权重不超 3-5%**——风险约束比最大化夏普更重要
