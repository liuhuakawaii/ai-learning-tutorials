# 因子评估

## 场景引入

上一节课我们构建了动量、价值、质量、波动率四类因子。但"构建出来"和"能用"之间还有巨大的距离。一个核心问题是：**你构建的因子真的能预测股票收益吗？**

很多量化新手会犯一个错误：构建了几十个因子，挑一个回测看起来最好的就直接用了。这本质上是数据挖掘偏差——你看到的"好"可能只是噪声。

因子评估的目标是用**严谨的统计方法**来回答三个问题：
1. 因子与未来收益之间是否存在稳定的相关性？（IC 分析）
2. 因子能否在实际交易中产生超额收益？（分层回测）
3. 因子的有效性能维持多久？（衰减分析）

本节课将建立一套完整的因子评估框架，从 IC 计算到分层回测，从换手率分析到因子半衰期估计，每一步都给出可运行的 Python 代码。

> **风险提示**：本课程仅用于量化投资技术教学，不构成任何投资建议。因子评估中的 IC 值和回测结果受样本期选择、交易成本等因素影响，不代表未来表现。

## 学习目标

完成本课学习后，你将能够：

1. 计算 IC、Rank IC、IC_IR 等因子有效性指标
2. 实现分层回测并解读各层收益差异
3. 分析因子换手率和交易成本
4. 进行因子衰减分析并估计因子半衰期
5. 建立系统化的因子评估框架

## 一、因子评估核心概念

### 1.1 评估流程总览

```
因子值时间序列              收益率时间序列
     │                          │
     ▼                          ▼
┌──────────────────────────────────────┐
│            截面对齐                   │
│   因子值[t]  ←→  收益率[t → t+1]     │
└──────────────┬───────────────────────┘
               │
      ┌────────┼────────┐
      ▼        ▼        ▼
  ┌───────┐ ┌───────┐ ┌───────┐
  │ IC分析 │ │分层回测│ │ 换手率 │
  └───┬───┘ └───┬───┘ └───┬───┘
      │         │         │
      ▼         ▼         ▼
  IC均值     各层收益   换手率均值
  Rank IC    超额收益   交易成本
  IC_IR      多空收益   因子衰减
  IC t值     分层单调性  半衰期
```

### 1.2 评估的数学基础

因子评估的核心是**截面相关性**：在每一个时间截面上，因子值与未来收益率之间的相关性。

```
时间 t 的截面数据:

  股票    因子值    未来收益率(t→t+1)
  ─────   ──────    ────────────────
  A       1.2       3.5%
  B       0.8       1.2%
  C       -0.5      -2.1%
  D       0.3       0.8%
  ...     ...       ...

IC = corr(因子值, 未来收益率)
```

## 二、IC 计算

### 2.1 皮尔逊 IC

```python
# evaluation/ic_analysis.py
"""
因子 IC（Information Coefficient）分析
"""
import pandas as pd
import numpy as np
from typing import Optional


def calc_ic_series(
    factor: pd.DataFrame,
    forward_returns: pd.DataFrame,
    method: str = "pearson",
) -> pd.Series:
    """
    计算因子 IC 时间序列

    IC（Information Coefficient）是因子评估最基础的指标：
    在每个截面上，计算因子值与下期收益率的相关系数

    参数:
        factor          : DataFrame，index为日期，columns为股票代码，values为因子值
        forward_returns : DataFrame，同结构，values为下期收益率
        method          : "pearson" 或 "spearman"

    返回:
        IC 时间序列（每个日期一个 IC 值）
    """
    # 对齐日期和股票
    common_dates = factor.index.intersection(forward_returns.index)
    common_stocks = factor.columns.intersection(forward_returns.columns)

    factor = factor.loc[common_dates, common_stocks]
    fwd_ret = forward_returns.loc[common_dates, common_stocks]

    ic_values = {}

    for date in common_dates:
        f = factor.loc[date].dropna()
        r = fwd_ret.loc[date].dropna()

        # 取两只股票的交集
        common = f.index.intersection(r.index)
        if len(common) < 30:  # 至少需要30只股票才有统计意义
            ic_values[date] = np.nan
            continue

        f = f.loc[common]
        r = r.loc[common]

        if method == "pearson":
            ic = f.corr(r)
        elif method == "spearman":
            ic = f.rank().corr(r.rank())
        else:
            raise ValueError(f"未知方法: {method}")

        ic_values[date] = ic

    return pd.Series(ic_values, name="IC")


def calc_rank_ic(
    factor: pd.DataFrame,
    forward_returns: pd.DataFrame,
) -> pd.Series:
    """
    计算 Rank IC（Spearman 秩相关）

    Rank IC 相比普通 IC 的优势:
    1. 不受极端值影响（只看排名，不看绝对值）
    2. 不要求因子服从正态分布
    3. 更适合非线性关系的检测

    在实践中，Rank IC 比 Pearson IC 更常用
    """
    return calc_ic_series(factor, forward_returns, method="spearman")
```

### 2.2 IC 统计指标

```python
def summarize_ic(
    ic_series: pd.Series,
    annualize: bool = True,
) -> dict:
    """
    IC 统计指标汇总

    输出的指标:
        IC_Mean   : IC 均值，衡量因子预测能力的方向和强度
        IC_Std    : IC 标准差，衡量预测能力的稳定性
        IC_IR     : IC 信息比率 = IC_Mean / IC_Std，最重要的综合指标
        IC_tstat  : IC 的 t 统计量，检验 IC 是否显著不为 0
        IC_Pos%   : IC 为正的比例，衡量预测方向的一致性
        IC_Abs>2% : |IC| > 2% 的比例，衡量有效信号密度
    """
    ic = ic_series.dropna()
    n = len(ic)

    if n == 0:
        return {k: np.nan for k in ["IC_Mean", "IC_Std", "IC_IR",
                                      "IC_tstat", "IC_Pos%", "IC_Abs>2%"]}

    ic_mean = ic.mean()
    ic_std = ic.std()
    ic_ir = ic_mean / ic_std if ic_std > 0 else 0
    ic_tstat = ic_mean / (ic_std / np.sqrt(n)) if ic_std > 0 else 0
    ic_pos_pct = (ic > 0).mean()
    ic_abs_gt2 = (ic.abs() > 0.02).mean()

    return {
        "IC_Mean": round(ic_mean, 4),
        "IC_Std": round(ic_std, 4),
        "IC_IR": round(ic_ir, 4),
        "IC_tstat": round(ic_tstat, 4),
        "IC_Pos%": round(ic_pos_pct * 100, 2),
        "IC_Abs>2%": round(ic_abs_gt2 * 100, 2),
        "样本数": n,
    }


def interpret_ic_results(stats: dict) -> str:
    """
    解读 IC 统计结果

    给出定性判断，帮助快速理解因子质量
    """
    lines = ["=== IC 因子评估结果 ===\n"]

    ic_mean = stats["IC_Mean"]
    ic_ir = stats["IC_IR"]
    ic_tstat = stats["IC_tstat"]

    # IC 均值解读
    if abs(ic_mean) > 0.05:
        lines.append(f"IC均值 = {ic_mean:.4f}，因子预测能力【强】")
    elif abs(ic_mean) > 0.03:
        lines.append(f"IC均值 = {ic_mean:.4f}，因子预测能力【中等】")
    elif abs(ic_mean) > 0.01:
        lines.append(f"IC均值 = {ic_mean:.4f}，因子预测能力【弱】")
    else:
        lines.append(f"IC均值 = {ic_mean:.4f}，因子预测能力【很弱/无】")

    # IC_IR 解读
    if abs(ic_ir) > 0.5:
        lines.append(f"IC_IR = {ic_ir:.4f}，因子稳定性【优秀】")
    elif abs(ic_ir) > 0.3:
        lines.append(f"IC_IR = {ic_ir:.4f}，因子稳定性【良好】")
    else:
        lines.append(f"IC_IR = {ic_ir:.4f}，因子稳定性【一般/差】")

    # t 统计量解读
    if abs(ic_tstat) > 2.58:
        lines.append(f"t统计量 = {ic_tstat:.2f}，因子在 1% 水平【显著】")
    elif abs(ic_tstat) > 1.96:
        lines.append(f"t统计量 = {ic_tstat:.2f}，因子在 5% 水平【显著】")
    else:
        lines.append(f"t统计量 = {ic_tstat:.2f}，因子【不显著】")

    # 一致性
    pos_pct = stats["IC_Pos%"]
    if pos_pct > 60:
        lines.append(f"IC为正比例 = {pos_pct:.1f}%，方向一致性【好】")
    elif pos_pct > 50:
        lines.append(f"IC为正比例 = {pos_pct:.1f}%，方向一致性【一般】")
    else:
        lines.append(f"IC为正比例 = {pos_pct:.1f}%，方向一致性【差】")

    # 综合判断
    lines.append("\n综合评估:")
    if abs(ic_mean) > 0.03 and abs(ic_ir) > 0.3 and abs(ic_tstat) > 1.96:
        lines.append("  因子有效，可以进入下一步（分层回测）验证")
    elif abs(ic_mean) > 0.01 and abs(ic_ir) > 0.2:
        lines.append("  因子有一定信号，但需要更多验证（分层回测、样本外测试）")
    else:
        lines.append("  因子信号较弱，建议重新审视因子逻辑或尝试改进构建方式")

    return "\n".join(lines)
```

### 2.3 IC 衰减分析

```python
def calc_ic_decay(
    factor: pd.DataFrame,
    returns: pd.DataFrame,
    max_lag: int = 20,
) -> pd.Series:
    """
    IC 衰减分析

    计算因子值与未来 1 天、2 天、...、N 天收益率的 IC
    观察因子预测能力随持有期的变化

    用途:
    1. 评估因子信号的持续性
    2. 确定最佳调仓频率
    3. 估计因子半衰期
    """
    decay = {}

    for lag in range(1, max_lag + 1):
        # 计算 lag 天的累计收益率
        fwd_returns = returns.shift(-lag)

        # 计算 IC
        ic = calc_ic_series(factor, fwd_returns)
        decay[lag] = ic.mean()

    return pd.Series(decay, name="IC_Decay")


def estimate_half_life(ic_decay: pd.Series) -> float:
    """
    估计因子 IC 的半衰期

    半衰期定义: IC 衰减到初始值一半所需的天数
    用指数衰减模型拟合: IC(t) = IC(0) * exp(-λ * t)
    半衰期 = ln(2) / λ
    """
    from scipy.optimize import curve_fit

    def exp_decay(t, ic0, lam):
        return ic0 * np.exp(-lam * t)

    x = ic_decay.index.values.astype(float)
    y = ic_decay.values

    # 过滤掉非正值（指数模型要求正值）
    mask = y > 0
    if mask.sum() < 3:
        return np.nan

    try:
        popt, _ = curve_fit(exp_decay, x[mask], y[mask], p0=[y[0], 0.1], maxfev=5000)
        ic0, lam = popt
        if lam > 0:
            half_life = np.log(2) / lam
            return round(half_life, 1)
    except Exception:
        pass

    return np.nan
```

## 三、分层回测

### 3.1 分层回测框架

分层回测是因子评估最直观的方法：按因子值将股票分成若干组，观察各组的收益差异。

```
因子分层回测示意:

  因子值从低到高排序，分为5层
  ┌───────────────────────────────────┐
  │  Layer 1 (最低)  │  Layer 5 (最高) │
  │  ▁▂▃▅▇           │  ▇▅▃▂▁         │
  │  预期收益低       │  预期收益高      │
  └───────────────────────────────────┘
  如果 Layer 5 - Layer 1 显著为正，因子有效
```

```python
# evaluation/layer_backtest.py
"""
因子分层回测
"""
import pandas as pd
import numpy as np
from typing import Optional


def layered_backtest(
    factor: pd.DataFrame,
    returns: pd.DataFrame,
    n_layers: int = 5,
    weight_method: str = "equal",
) -> pd.DataFrame:
    """
    因子分层回测

    在每个调仓日：
        1. 按因子值排序
        2. 分为 n_layers 组
        3. 计算各组的等权/市值加权收益
        4. 计算多空组合收益

    参数:
        factor      : 因子值 DataFrame
        returns     : 下期收益率 DataFrame
        n_layers    : 分层数（默认5层，即五分位）
        weight_method: "equal" 等权 或 "market_cap" 市值加权

    返回:
        各层收益时间序列
    """
    common_dates = factor.index.intersection(returns.index)
    common_stocks = factor.columns.intersection(returns.columns)

    factor = factor.loc[common_dates, common_stocks]
    returns = returns.loc[common_dates, common_stocks]

    layer_returns = {f"Layer_{i+1}": [] for i in range(n_layers)}
    layer_returns["Long_Short"] = []
    layer_returns["Market"] = []

    for date in common_dates:
        f = factor.loc[date].dropna()
        r = returns.loc[date].dropna()

        common = f.index.intersection(r.index)
        if len(common) < n_layers * 5:  # 每层至少5只股票
            for key in layer_returns:
                layer_returns[key].append(np.nan)
            continue

        f = f.loc[common]
        r = r.loc[common]

        # 按因子值排序并分层
        ranks = f.rank(method="first")
        layer_size = len(ranks) / n_layers

        layers = {}
        for i in range(n_layers):
            lower = i * layer_size
            upper = (i + 1) * layer_size
            mask = (ranks > lower) & (ranks <= upper)
            layer_stocks = ranks[mask].index

            if weight_method == "equal":
                layer_ret = r.loc[layer_stocks].mean()
            else:
                layer_ret = r.loc[layer_stocks].mean()  # 简化为等权

            layers[f"Layer_{i+1}"] = layer_ret
            layer_returns[f"Layer_{i+1}"].append(layer_ret)

        # 多空收益 = 最高层 - 最底层
        long_short = layers[f"Layer_{n_layers}"] - layers["Layer_1"]
        layer_returns["Long_Short"].append(long_short)

        # 市场收益（等权基准）
        layer_returns["Market"].append(r.mean())

    # 转为 DataFrame
    result = pd.DataFrame(layer_returns, index=common_dates)
    result.index.name = "date"

    return result


def calc_layer_stats(layer_returns: pd.DataFrame) -> pd.DataFrame:
    """
    计算各层的统计指标

    指标包括:
        年化收益率、年化波动率、夏普比率、最大回撤
    """
    stats = []

    for col in layer_returns.columns:
        ret = layer_returns[col].dropna()

        ann_ret = ret.mean() * 250          # 年化收益
        ann_vol = ret.std() * np.sqrt(250)   # 年化波动
        sharpe = ann_ret / ann_vol if ann_vol > 0 else 0

        # 最大回撤
        cum_ret = (1 + ret).cumprod()
        peak = cum_ret.cummax()
        drawdown = (cum_ret - peak) / peak
        max_dd = drawdown.min()

        stats.append({
            "分层": col,
            "年化收益": f"{ann_ret:.2%}",
            "年化波动": f"{ann_vol:.2%}",
            "夏普比率": f"{sharpe:.2f}",
            "最大回撤": f"{max_dd:.2%}",
        })

    return pd.DataFrame(stats)
```

### 3.2 分层收益可视化数据

```python
def calc_cumulative_returns(layer_returns: pd.DataFrame) -> pd.DataFrame:
    """
    计算各层累计收益（用于绘制分层净值曲线）

    如果分层曲线呈现"单调递增"（从Layer1到Layer5逐渐升高），
    说明因子具有良好的分层能力
    """
    cumulative = (1 + layer_returns).cumprod()
    return cumulative


def check_monotonicity(layer_stats: pd.DataFrame, n_layers: int = 5) -> dict:
    """
    检查分层收益的单调性

    理想情况: Layer1 < Layer2 < Layer3 < Layer4 < Layer5
    单调性越强，因子的分层能力越好
    """
    layer_cols = [f"Layer_{i+1}" for i in range(n_layers)]
    ann_returns = []

    for col in layer_cols:
        row = layer_stats[layer_stats["分层"] == col]
        if not row.empty:
            ret_str = row["年化收益"].values[0].replace("%", "")
            ann_returns.append(float(ret_str) / 100)

    if len(ann_returns) < 2:
        return {"单调性评分": 0, "说明": "数据不足"}

    # 计算单调性评分
    # 完美单调递增: 所有相邻层收益都是递增的
    monotone_count = sum(
        1 for i in range(len(ann_returns) - 1)
        if ann_returns[i] < ann_returns[i + 1]
    )
    total_pairs = len(ann_returns) - 1
    score = monotone_count / total_pairs

    # 多空收益
    long_short = ann_returns[-1] - ann_returns[0]

    return {
        "单调性评分": round(score, 2),
        "多空年化收益": f"{long_short:.2%}",
        "各层年化收益": [f"{r:.2%}" for r in ann_returns],
        "说明": "评分1.0为完美单调" if score == 1.0
                else f"评分{score:.2f}, {'可接受' if score >= 0.75 else '单调性不足'}",
    }
```

```python
# evaluation/factor_evaluator.py
"""
因子评估主框架
整合 IC 分析、分层回测、换手率分析
"""
import pandas as pd
import numpy as np


class FactorEvaluator:
    """
    因子评估器

    将 IC 分析、分层回测、换手率分析整合为统一框架
    """

    def __init__(
        self,
        factor: pd.DataFrame,
        returns: pd.DataFrame,
        n_layers: int = 5,
    ):
        """
        参数:
            factor  : 因子值 DataFrame（index=日期, columns=股票）
            returns : 下期收益率 DataFrame（同结构）
            n_layers: 分层数
        """
        self.factor = factor
        self.returns = returns
        self.n_layers = n_layers
        self._ic_series = None
        self._rank_ic_series = None
        self._layer_returns = None

    def run_full_evaluation(self) -> dict:
        """
        运行完整的因子评估流程

        返回包含所有评估结果的字典
        """
        results = {}

        # === IC 分析 ===
        print("=" * 50)
        print("1. IC 分析")
        print("=" * 50)

        ic = self.ic_series
        rank_ic = self.rank_ic_series

        from .ic_analysis import summarize_ic, interpret_ic_results

        ic_stats = summarize_ic(ic)
        rank_ic_stats = summarize_ic(rank_ic)

        results["IC"] = ic_stats
        results["Rank_IC"] = rank_ic_stats

        print("\nPearson IC:")
        print(interpret_ic_results(ic_stats))
        print("\nRank IC:")
        print(interpret_ic_results(rank_ic_stats))

        # === 分层回测 ===
        print("\n" + "=" * 50)
        print("2. 分层回测")
        print("=" * 50)

        from .layer_backtest import calc_layer_stats, check_monotonicity

        layer_ret = self.layer_returns
        layer_stats = calc_layer_stats(layer_ret)
        monotone = check_monotonicity(layer_ret, self.n_layers)

        results["分层统计"] = layer_stats.to_dict("records")
        results["单调性"] = monotone

        print("\n各层统计:")
        print(layer_stats.to_string(index=False))
        print(f"\n单调性: {monotone}")

        # === 换手率分析 ===
        print("\n" + "=" * 50)
        print("3. 换手率分析")
        print("=" * 50)

        turnover = self.calc_turnover()
        results["换手率"] = turnover

        print(f"\n因子月均换手率: {turnover['月均换手率']:.2%}")

        # === 综合判断 ===
        print("\n" + "=" * 50)
        print("4. 综合评估")
        print("=" * 50)

        overall = self._overall_judgment(results)
        results["综合评估"] = overall
        print(overall)

        return results

    @property
    def ic_series(self) -> pd.Series:
        if self._ic_series is None:
            from .ic_analysis import calc_ic_series
            self._ic_series = calc_ic_series(self.factor, self.returns)
        return self._ic_series

    @property
    def rank_ic_series(self) -> pd.Series:
        if self._rank_ic_series is None:
            from .ic_analysis import calc_rank_ic
            self._rank_ic_series = calc_rank_ic(self.factor, self.returns)
        return self._rank_ic_series

    @property
    def layer_returns(self) -> pd.DataFrame:
        if self._layer_returns is None:
            from .layer_backtest import layered_backtest
            self._layer_returns = layered_backtest(
                self.factor, self.returns, self.n_layers
            )
        return self._layer_returns

    def calc_turnover(self, top_pct: float = 0.2) -> dict:
        """
        计算因子换手率

        换手率衡量的是每次调仓时，持仓股票的更换比例
        高换手率意味着高交易成本，可能侵蚀因子收益
        """
        dates = self.factor.index.tolist()
        turnovers = []

        for i in range(1, len(dates)):
            prev_factor = self.factor.loc[dates[i - 1]].dropna()
            curr_factor = self.factor.loc[dates[i]].dropna()

            # 取交集
            common = prev_factor.index.intersection(curr_factor.index)
            if len(common) < 10:
                continue

            prev_top = set(prev_factor.loc[common].nlargest(int(len(common) * top_pct)).index)
            curr_top = set(curr_factor.loc[common].nlargest(int(len(common) * top_pct)).index)

            if len(prev_top) == 0:
                continue

            # 换手率 = 新增持仓比例
            new_stocks = curr_top - prev_top
            turnover = len(new_stocks) / len(curr_top)
            turnovers.append(turnover)

        monthly_turnover = np.mean(turnovers) if turnovers else 0

        return {
            "月均换手率": monthly_turnover,
            "最高换手率": max(turnovers) if turnovers else 0,
            "最低换手率": min(turnovers) if turnovers else 0,
        }

    def _overall_judgment(self, results: dict) -> str:
        """
        综合判断因子质量
        """
        ic_mean = results["IC"]["IC_Mean"]
        ic_ir = results["IC"]["IC_IR"]
        ic_tstat = results["IC"]["IC_tstat"]
        monotone_score = results["单调性"]["单调性评分"]

        score = 0
        reasons = []

        # IC 均值
        if abs(ic_mean) > 0.03:
            score += 2
            reasons.append("IC均值较强")
        elif abs(ic_mean) > 0.01:
            score += 1
            reasons.append("IC均值一般")

        # IC_IR
        if abs(ic_ir) > 0.3:
            score += 2
            reasons.append("IC_IR良好")
        elif abs(ic_ir) > 0.2:
            score += 1
            reasons.append("IC_IR一般")

        # t 统计量
        if abs(ic_tstat) > 1.96:
            score += 1
            reasons.append("统计显著")

        # 单调性
        if monotone_score >= 0.8:
            score += 2
            reasons.append("分层单调性好")
        elif monotone_score >= 0.6:
            score += 1
            reasons.append("分层单调性一般")

        # 综合评级
        if score >= 6:
            grade = "A (优秀)"
        elif score >= 4:
            grade = "B (良好)"
        elif score >= 2:
            grade = "C (一般)"
        else:
            grade = "D (较差)"

        return f"评级: {grade}\n得分: {score}/7\n依据: {', '.join(reasons)}"
```

```python
def generate_ic_decay_report(
    factor: pd.DataFrame,
    returns: pd.DataFrame,
    max_lag: int = 20,

    生成 IC 衰减分析报告

    输出:
        - 各持有期的 IC 值
        - 半衰期估计
        - 文字解读

    from .ic_analysis import calc_ic_decay, estimate_half_life

    lines.append("--------|----------|----------")

    ic_1d = decay.iloc[0] if len(decay) > 0 else np.nan
    for lag, ic_val in decay.items():
        decay_pct = (1 - ic_val / ic_1d) * 100 if ic_1d != 0 else np.nan
        bar = "█" * max(0, int(ic_val * 500))
        lines.append(f"  {lag:2d}天  |  {ic_val:+.4f}  |  {decay_pct:+.1f}%  {bar}")

    lines.append(f"\n因子半衰期: {half_life} 天" if not np.isnan(half_life)
                 else "\n半衰期: 无法估计（衰减不规律）")

    # 调仓频率建议
    if half_life and half_life > 0:
        suggested_freq = max(1, int(half_life / 3))
        lines.append(f"建议调仓频率: 每 {suggested_freq} 天")

    return "\n".join(lines)
```

## 六、防止过拟合的评估原则

### 6.1 样本内 vs 样本外

```
时间线:
|←──── 训练集 (70%) ────→|←── 验证集 (15%) ──→|←── 测试集 (15%) ──→|
2015                    2021                 2023                2025

- 训练集: 用于因子构建和参数选择
- 验证集: 用于因子筛选和策略优化
- 测试集: 最终评估，只用一次
```

```python
def split_time_periods(
    dates: pd.DatetimeIndex,
    train_ratio: float = 0.7,
    val_ratio: float = 0.15,
) -> tuple:
    """
    按时间划分训练/验证/测试集

    注意: 不能随机划分，必须按时间顺序
    随机划分会导致未来数据泄露
    """
    n = len(dates)
    train_end = int(n * train_ratio)
    val_end = int(n * (train_ratio + val_ratio))

    train_dates = dates[:train_end]
    val_dates = dates[train_end:val_end]
    test_dates = dates[val_end:]

    return train_dates, val_dates, test_dates
```

### 6.2 多重检验校正

```python
def bonferroni_correction(p_values: list, alpha: float = 0.05) -> list:
    """
    Bonferroni 多重检验校正

    当你测试了 N 个因子时，需要将显著性水平除以 N
    这是对多重检验最保守的校正方法

    例如: 测试了 20 个因子，alpha = 0.05
    校正后的 alpha = 0.05 / 20 = 0.0025
    只有 p < 0.0025 的因子才算显著
    """
    n = len(p_values)
    corrected_alpha = alpha / n

    results = []
    for i, p in enumerate(p_values):
        results.append({
            "因子": i + 1,
            "p值": p,
            "校正后alpha": corrected_alpha,
            "是否显著": p < corrected_alpha,
        })

    return results
```

## 常见误区

### 误区一：只看 IC 均值，不看 IC_IR

IC 均值为 0.03 看起来不错，但如果 IC 标准差是 0.15，IC_IR 只有 0.2，说明因子信号非常不稳定。IC_IR（IC 均值 / IC 标准差）才是衡量因子质量的综合指标。

### 误区二：分层回测忽略交易成本

很多分层回测报告只展示净收益，不扣除交易成本。A 股的印花税（单边 0.05%）、佣金（万 2.5）、冲击成本加起来，对于月度调仓的策略，年化交易成本约 3-5%。如果因子的多空收益只有 5%，交易成本就吃掉了大部分。

### 误区三：用未来数据做因子评估

最常见的数据泄露是：用 t 日收盘后的财务数据计算因子，但 t 日的收益率已经包含了这些信息。正确的做法是：用 t 日可获得的因子值，预测 t+1 日的收益率。

### 误区四：忽略因子衰减

一个因子在持有 1 天时 IC 为 0.05，持有 5 天衰减到 0.01，持有 10 天衰减到 0。如果你的调仓周期是月度，这个因子对你就没有价值。

### 误区五：对同一数据反复测试

测试 20 个因子，总有一个看起来"显著"。这是多重检验问题。要么用 Bonferroni 校正，要么留出独立的测试集做最终验证。

## 小结与练习

### 本课要点

- **IC** 是因子与未来收益的截面相关系数，IC 均值衡量预测能力，IC_IR 衡量稳定性
- **Rank IC** 比 Pearson IC 更鲁棒，是实践中更常用的指标
- **分层回测** 直观展示因子的分层能力，单调性是关键判断标准
- **IC 衰减分析** 帮助确定因子的最佳持有期和调仓频率
- **换手率** 直接影响交易成本，高换手因子需要更高的原始收益来覆盖成本
- **过拟合防范** 必须划分样本内/外，使用多重检验校正

### 练习一：IC 计算

**思路**：获取因子值和下期收益率的时间序列数据，在每个截面上计算相关系数。

**答案**：
```python
import pandas as pd
import numpy as np

# 模拟数据
np.random.seed(42)
dates = pd.date_range("2023-01-01", periods=100, freq="B")
stocks = [f"{i:06d}" for i in range(1, 51)]

factor = pd.DataFrame(np.random.randn(100, 50), index=dates, columns=stocks)
returns = pd.DataFrame(np.random.randn(100, 50) * 0.02, index=dates, columns=stocks)

# 计算每日 IC
ic_list = []
for date in dates:
    f = factor.loc[date]
    r = returns.loc[date]
    ic = f.corr(r)
    ic_list.append({"date": date, "IC": ic})

ic_df = pd.DataFrame(ic_list).set_index("date")

print(f"IC 均值: {ic_df['IC'].mean():.4f}")
print(f"IC 标准差: {ic_df['IC'].std():.4f}")
print(f"IC_IR: {ic_df['IC'].mean() / ic_df['IC'].std():.4f}")
print(f"IC>0比例: {(ic_df['IC'] > 0).mean():.2%}")
```

**要点**：
- 由于是随机数据，IC 应接近 0，这是预期行为
- 实际因子数据中，有效因子的 IC 均值通常在 0.02~0.08 之间
- 注意检查每个截面上的样本量（股票数量），太少则 IC 不可靠

### 练习二：分层回测

**思路**：按因子值分组，计算各组收益，观察是否呈单调递增。

**答案**：
```python
import pandas as pd
import numpy as np

# 模拟一个有预测力的因子
np.random.seed(42)
dates = pd.date_range("2023-01-01", periods=250, freq="B")
stocks = [f"{i:06d}" for i in range(1, 101)]

# 因子值
factor = pd.DataFrame(np.random.randn(250, 100), index=dates, columns=stocks)

# 收益率 = 因子值 * 信号强度 + 噪声
signal_strength = 0.002
noise = np.random.randn(250, 100) * 0.02
returns = factor * signal_strength + noise
returns = pd.DataFrame(returns, index=dates, columns=stocks)

# 分层回测
n_layers = 5
layer_results = {f"L{i+1}": [] for i in range(n_layers)}

for date in dates:
    f = factor.loc[date].dropna()
    r = returns.loc[date].dropna()
    common = f.index.intersection(r.index)

    f_sorted = f.loc[common].rank()
    layer_size = len(f_sorted) / n_layers

    for i in range(n_layers):
        lower = i * layer_size
        upper = (i + 1) * layer_size
        mask = (f_sorted > lower) & (f_sorted <= upper)
        layer_ret = r.loc[f_sorted[mask].index].mean()
        layer_results[f"L{i+1}"].append(layer_ret)

layer_df = pd.DataFrame(layer_results, index=dates)

# 输出年化收益
for col in layer_df.columns:
    ann_ret = layer_df[col].mean() * 250
    print(f"{col}: 年化收益 {ann_ret:.2%}")
```

**要点**：
- 模拟数据中设置了信号强度，所以应该能观察到分层差异
- 如果因子无效（纯噪声），各层收益应该接近
- 实际中，分层收益的差异需要扣除交易成本后仍然显著

### 练习三：半衰期估计

**思路**：用指数衰减模型拟合 IC 随持有期的变化。

**答案**：
```python
import numpy as np
from scipy.optimize import curve_fit

# 模拟 IC 衰减数据
lags = np.arange(1, 21)
ic_0 = 0.05
half_life_true = 7.0
lambda_true = np.log(2) / half_life_true
ic_values = ic_0 * np.exp(-lambda_true * lags) + np.random.normal(0, 0.002, 20)

# 拟合指数衰减
def exp_decay(t, ic0, lam):
    return ic0 * np.exp(-lam * t)

popt, _ = curve_fit(exp_decay, lags, ic_values, p0=[0.05, 0.1])
ic0_fit, lam_fit = popt

half_life_est = np.log(2) / lam_fit

print(f"真实半衰期: {half_life_true:.1f} 天")
print(f"估计半衰期: {half_life_est:.1f} 天")
print(f"拟合 IC(0): {ic0_fit:.4f} (真实: {ic_0:.4f})")
```

**要点**：
- 估计值应接近真实值（7 天），误差来自加入的随机噪声
- 如果 IC 衰减不规律（不是单调递减），指数模型可能不适用
- 半衰期对于确定调仓频率非常有价值：一般建议调仓周期为半衰期的 1/3
