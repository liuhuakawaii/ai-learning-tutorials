你是一个技术负责人，团队要将一个老旧的 Node.js 电商系统迁移到新架构。你需要回答：模块怎么依赖？核心业务流程是什么？哪些是 Dead Code？手动回答需要两周，用 `legacy-insight` 工具只需要几分钟。

## 工具架构

```
CLI 层 → 编排层 (pipeline) → 分析层 (AST解析/依赖分析/调用链路/AI分析) → 基础层 (文件扫描/AST工具/LLM客户端)
```

数据流：源码 → 文件扫描 → AST 解析 → 依赖分析 → 调用链路 → AI 分析 → 报告。AI 放最后，先用确定性方法获取结构信息。

```typescript
// src/core/scanner.ts
import { glob } from "glob";
import { statSync } from "fs";
import { extname } from "path";

export interface FileInfo {
  path: string; relativePath: string; extension: string; size: number;
  language: "javascript" | "typescript" | "python" | "unknown";
}

const LANG_MAP: Record<string, FileInfo["language"]> = {
  ".js": "javascript", ".jsx": "javascript", ".ts": "typescript", ".tsx": "typescript", ".py": "python",
};

export async function scanProject(projectPath: string): Promise<FileInfo[]> {
  const files = await glob("**/*.{ts,tsx,js,jsx,py}", {
    cwd: projectPath,
    ignore: ["**/node_modules/**", "**/dist/**", "**/.git/**", "**/*.test.*"],
    nodir: true,
  });
  return files.map((file) => ({
    path: `${projectPath}/${file}`, relativePath: file, extension: extname(file),
    size: statSync(`${projectPath}/${file}`).size,
    language: LANG_MAP[extname(file)] ?? "unknown",
  })).sort((a, b) => a.relativePath.localeCompare(b.relativePath));
}
```

### AST 解析器

```typescript
// src/core/parser.ts
import * as babelParser from "@babel/parser";
import * as babelTraverse from "@babel/traverse";
import * as t from "@babel/types";
import { readFileSync } from "fs";
import { FileInfo } from "./scanner";

export interface FunctionInfo { name: string; filePath: string; startLine: number; endLine: number; params: string[]; isAsync: boolean; bodyPreview: string; }
export interface ImportInfo { source: string; specifiers: string[]; filePath: string; }
export interface FileAstInfo { filePath: string; functions: FunctionInfo[]; imports: ImportInfo[]; exports: string[]; }

export function parseFile(file: FileInfo): FileAstInfo | null {
  try {
    const code = readFileSync(file.path, "utf-8");
    const ast = babelParser.parse(code, {
      sourceType: "unambiguous",
      plugins: ["jsx", file.language === "typescript" ? "typescript" : null, "decorators-legacy", "classProperties"].filter(Boolean) as any[],
    });
    const functions: FunctionInfo[] = [];
    const imports: ImportInfo[] = [];
    const exports: string[] = [];

    babelTraverse.default(ast, {
      FunctionDeclaration(path) {
        if (path.node.id) functions.push({
          name: path.node.id.name, filePath: file.path,
          startLine: path.node.loc?.start.line ?? 0, endLine: path.node.loc?.end.line ?? 0,
          params: path.node.params.map((p) => t.isIdentifier(p) ? p.name : "_"),
          isAsync: path.node.async,
          bodyPreview: code.slice(path.node.body.start ?? 0, Math.min((path.node.body.start ?? 0) + 200, path.node.body.end ?? 0)),
        });
      },
      ImportDeclaration(path) {
        imports.push({ source: path.node.source.value, specifiers: path.node.specifiers.map((s) => t.isImportDefaultSpecifier(s) ? s.local.name : t.isImportSpecifier(s) ? (t.isIdentifier(s.imported) ? s.imported.name : s.imported.value) : "_"), filePath: file.path });
      },
    });
    return { filePath: file.path, functions, imports, exports };
  } catch { return null; }
}
```

### 依赖分析器

```typescript
// src/analyzers/dependency.ts
import { FileAstInfo } from "../core/parser";
import { FileInfo } from "../core/scanner";
import { resolve, dirname, join } from "path";

export interface DependencyGraph { nodes: Map<string, { filePath: string; imports: string[]; importedBy: string[]; externalDeps: string[] }>; circularDependencies: string[][]; externalDependencies: Set<string>; }

export function buildDependencyGraph(files: FileInfo[], astInfos: Map<string, FileAstInfo>): DependencyGraph {
  const graph: DependencyGraph = { nodes: new Map(), circularDependencies: [], externalDependencies: new Set() };
  const pathIndex = new Map<string, string>();
  for (const f of files) { pathIndex.set(f.relativePath, f.path); pathIndex.set(f.relativePath.replace(/\.(ts|tsx|js|jsx)$/, ""), f.path); }
  for (const f of files) graph.nodes.set(f.path, { filePath: f.path, imports: [], importedBy: [], externalDeps: [] });

  for (const [filePath, astInfo] of astInfos) {
    const node = graph.nodes.get(filePath)!;
    for (const imp of astInfo.imports) {
      if (imp.source.startsWith(".")) {
        const resolved = resolve(dirname(filePath), imp.source);
        const target = pathIndex.get(resolved) ?? pathIndex.get(resolved + ".ts") ?? pathIndex.get(join(resolved, "index.ts"));
        if (target) { node.imports.push(target); graph.nodes.get(target)?.importedBy.push(filePath); }
      } else {
        const pkg = imp.source.startsWith("@") ? imp.source.split("/").slice(0, 2).join("/") : imp.source.split("/")[0];
        node.externalDeps.push(pkg); graph.externalDependencies.add(pkg);
      }
    }
  }
  // 循环依赖检测（DFS）
  const visited = new Set<string>(), inStack = new Set<string>();
  function dfs(node: string, path: string[]) {
    if (inStack.has(node)) { const s = path.indexOf(node); if (s >= 0) graph.circularDependencies.push(path.slice(s).concat(node)); return; }
    if (visited.has(node)) return;
    visited.add(node); inStack.add(node); path.push(node);
    for (const dep of graph.nodes.get(node)?.imports ?? []) dfs(dep, [...path]);
    inStack.delete(node);
  }
  for (const node of graph.nodes.keys()) dfs(node, []);
  return graph;
}
```

### 调用链路与 AI 分析

```typescript
// src/analyzers/call-graph.ts — 简化版
export function buildCallGraph(astInfos: Map<string, any>) {
  const nodes = new Map(), entryPoints: string[] = [], deadCode: string[] = [];
  const SIDE_EFFECT_RE = { database: /\.(query|execute|find|save)\s*\(/, network: /(fetch|axios)\s*[.(]/ };
  for (const [fp, info] of astInfos) {
    for (const func of info.functions) {
      const id = `${fp}:${func.name}`;
      const effects = Object.entries(SIDE_EFFECT_RE).filter(([, re]) => re.test(func.bodyPreview)).map(([c]) => c);
      nodes.set(id, { id, name: func.name, filePath: fp, line: func.startLine, callsTo: [], calledBy: [], sideEffects: effects });
    }
  }
  // 从入口点 BFS 找可达节点，不可达即为 dead code
  const reachable = new Set<string>();
  const queue = entryPoints.filter(id => nodes.has(id));
  while (queue.length) { const c = queue.shift()!; if (reachable.has(c)) continue; reachable.add(c); nodes.get(c)?.callsTo.forEach((t: string) => queue.push(t)); }
  for (const id of nodes.keys()) if (!reachable.has(id)) deadCode.push(id);
  return { nodes, entryPoints, deadCode };
}
```

// src/analyzers/ai-analyzer.ts
import OpenAI from "openai";
import { readFileSync } from "fs";

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export async function analyzeFileWithAi(filePath: string, astInfo: any) {
  const code = readFileSync(filePath, "utf-8");
  const funcList = astInfo.functions.map((f: any) => `- ${f.name}(行${f.startLine}): ${f.isAsync ? "async" : "sync"}`).join("\n");
  const response = await openai.chat.completions.create({
    model: "gpt-4o-mini", temperature: 0.1, response_format: { type: "json_object" },
    messages: [
      { role: "system", content: "你是遗留系统代码分析专家。请用中文输出 JSON。" },
      { role: "user", content: `分析文件 ${filePath}\n函数:\n${funcList}\n\n代码:\n\`\`\`\n${code.slice(0, 6000)}\n\`\`\`\n\n输出: {"summary":"","purpose":"","businessRules":[],"complexity":"low/medium/high","migrationPriority":"high/medium/low","migrationNotes":""}` },
    ],
  });
  return { filePath, ...JSON.parse(response.choices[0].message.content ?? "{}") };
}

## 流水线与 CLI

```typescript
// src/pipeline.ts
import { scanProject } from "./core/scanner";
import { parseFile, FileAstInfo } from "./core/parser";
import { buildDependencyGraph } from "./analyzers/dependency";
import { buildCallGraph } from "./analyzers/call-graph";
import { analyzeFileWithAi } from "./analyzers/ai-analyzer";

export async function runPipeline(projectPath: string, enableAi: boolean, aiLimit: number) {
  const files = await scanProject(projectPath);
  const astInfos = new Map<string, FileAstInfo>();
  for (const file of files) { const info = parseFile(file); if (info) astInfos.set(file.path, info); }

  const depGraph = buildDependencyGraph(files, astInfos);
  const callGraph = buildCallGraph(astInfos);

  let aiAnalyses: any[] = [];
  if (enableAi) {
    for (const file of files.slice(0, aiLimit)) {
      const ast = astInfos.get(file.path);
      if (ast) try { aiAnalyses.push(await analyzeFileWithAi(file.path, ast)); } catch {}
    }
  }

  const report = generateReport(projectPath, files.length, depGraph, callGraph, aiAnalyses);
  return { files: files.length, depGraph, callGraph, aiAnalyses, report };
}

function generateReport(projectPath: string, fileCount: number, depGraph: any, callGraph: any, aiAnalyses: any[]): string {
  const s = [`# 遗留代码智能分析报告\n`, `> \`${projectPath}\` | ${new Date().toISOString()}\n`];
  s.push(`## 项目概览\n| 指标 | 数值 |\n|------|------|\n| 源文件 | ${fileCount} |\n| 函数 | ${callGraph.nodes.size} |\n| 外部依赖 | ${depGraph.externalDependencies.size} |\n| 循环依赖 | ${depGraph.circularDependencies.length} |\n| 不可达代码 | ${callGraph.deadCode.length} |\n`);
  if (depGraph.circularDependencies.length > 0) { s.push("## 循环依赖\n"); for (const cycle of depGraph.circularDependencies) s.push(`- ${cycle.map((f: string) => f.split("/").pop()).join(" → ")}`); }
  if (aiAnalyses.length > 0) { s.push("\n## AI 语义分析\n"); for (const a of aiAnalyses) s.push(`### ${a.filePath.split("/").pop()}\n**功能**: ${a.summary} | **优先级**: ${a.migrationPriority} | **复杂度**: ${a.complexity}\n`); }
  s.push("\n## 迁移建议\n1. 先迁移叶子模块 2. 再迁移工具函数 3. 然后业务逻辑层 4. 最后路由控制器 5. 清理 Dead Code");
  return s.join("\n");
}
```

// src/index.ts
#!/usr/bin/env node
import { Command } from "commander";
import { runPipeline } from "./pipeline";
import { writeFileSync, mkdirSync } from "fs";
import { join } from "path";

const program = new Command();
program.name("legacy-insight").description("遗留代码智能分析工具").version("1.0.0");
program.argument("<project-path>", "项目路径").option("-o, --output <dir>", "输出目录", "./output").option("--ai", "启用 AI 分析", false).option("--ai-limit <n>", "AI 最大文件数", "20")

  });
