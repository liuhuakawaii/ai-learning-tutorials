## 场景引入

你是一家 SaaS 公司的数据平台工程师。产品经理提了一个需求：

> "我们的客户不会写 SQL，但他们每天都要看销售数据。能不能做一个功能，让他们用中文提问，系统直接出图表？比如输入'上个月华东区的销售趋势'，就自动出来一张折线图。"

这个需求拆解下来，涉及的技术栈恰好覆盖了 Part 2 的全部内容：

- 用户说中文 → 需要 NL2SQL
- 数据库 Schema 各不相同 → 需要 Schema Linking
- 不同数据适合不同图表 → 需要图表推荐
- 图表要可交互 → 需要代码生成（而非图片生成）
- 整条链路要串起来 → 需要 Pipeline 编排
- 用户可能追问"只看华东" → 需要对话上下文管理

本节课就是把这些技术拼装成一个完整的产品。我们会从需求分析开始，一步步搭建前端、后端和 AI Pipeline，最终得到一个可以实际部署的系统。

### 2.3 NL2SQL 引擎

NL2SQL 是整个系统的核心。这里用 OpenAI 的 Function Calling 能力来保证输出格式稳定。

```python
# backend/app/pipeline/nl2sql.py
import json
from openai import AsyncOpenAI
from app.models import UserIntent, IntentType


NL2SQL_SYSTEM_PROMPT = """你是一个专业的 SQL 生成器。

你的任务是根据用户的自然语言问题和数据库 Schema，生成可执行的 PostgreSQL 查询。

规则：
1. 只返回 SQL 语句，不要解释
2. 使用标准 PostgreSQL 语法
3. 日期函数使用 DATE_TRUNC、CURRENT_DATE 等 PG 原生函数
4. 列名和表名必须严格匹配 Schema 定义
5. 对于模糊需求，选择最合理的解读
6. 如果问题无法用给定 Schema 回答，返回空字符串

输出格式要求：只返回 JSON 对象，包含 sql 和 reasoning 两个字段。
"""


class NL2SQLEngine:
    def __init__(self, api_key: str, model: str = "gpt-4o"):
        self.client = AsyncOpenAI(api_key=api_key)
        self.model = model

    async def generate(
        self, intent: UserIntent, schema_text: str, user_query: str
    ) -> str:
        prompt = f"""数据库 Schema:
{schema_text}

用户问题: {user_query}

解析后的意图:
- 类型: {intent.intent_type.value}
- 维度: {intent.dimensions}
- 指标: {intent.metrics}
- 时间范围: {intent.time_range}
- 过滤条件: {intent.filters}

请生成 SQL。"""

        response = await self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": NL2SQL_SYSTEM_PROMPT},
                {"role": "user", "content": prompt},
            ],
            temperature=0,
            response_format={"type": "json_object"},
        )

        result = json.loads(response.choices[0].message.content)
        return result.get("sql", "")

    async def fix_sql(
        self, sql: str, error: str, schema_text: str, user_query: str
    ) -> str:
        prompt = f"""SQL 执行出错，请修复。

用户问题: {user_query}
Schema: {schema_text}
错误的 SQL: {sql}
报错信息: {error}

请返回修复后的 JSON，包含 sql 和 fix_reasoning 字段。"""

        response = await self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": NL2SQL_SYSTEM_PROMPT},
                {"role": "user", "content": prompt},
            ],
            temperature=0,
            response_format={"type": "json_object"},
        )

        result = json.loads(response.choices[0].message.content)
        return result.get("sql", sql)
```

import json
from openai import AsyncOpenAI
from app.models import ChartRecommendation, ChartType


CHART_RECOMMEND_PROMPT = """你是数据可视化专家。根据数据特征推荐最合适的图表类型。

可用图表类型: bar, line, pie, scatter, table, heatmap

选择规则:
- 时间序列趋势 → line
- 分类对比（≤8 个类别）→ bar
- 占比分布（≤6 个类别）→ pie
- 两个数值变量的关系 → scatter
- 多维交叉数据 → heatmap
- 数据行数 > 20 或列数 > 6 → table

返回 JSON: {"chart_type": "...", "confidence": 0.0-1.0, "reason": "...", "alternatives": [...]}
"""


class ChartRecommender:
    def __init__(self, api_key: str, model: str = "gpt-4o"):
        self.client = AsyncOpenAI(api_key=api_key)
        self.model = model

    async def recommend(
        self, columns: list[str], rows: list[list], user_query: str
    ) -> ChartRecommendation:
        sample_rows = rows[:5]
        prompt = f"""用户问题: {user_query}

数据列: {columns}
数据行数: {len(rows)}
示例数据（前 5 行）: {json.dumps(sample_rows, ensure_ascii=False)}

请推荐图表类型。"""

        response = await self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": CHART_RECOMMEND_PROMPT},
                {"role": "user", "content": prompt},
            ],
            temperature=0,
            response_format={"type": "json_object"},
        )

        result = json.loads(response.choices[0].message.content)
        return ChartRecommendation(
            chart_type=ChartType(result["chart_type"]),
            confidence=result.get("confidence", 0.8),
            reason=result.get("reason", ""),
            alternatives=[ChartType(t) for t in result.get("alternatives", [])],

### 2.5 ECharts 代码生成器

```python
# backend/app/pipeline/codegen.py
import json
from openai import AsyncOpenAI
from app.models import ChartType, ChartConfig


ECHARTS_GEN_PROMPT = """你是 ECharts 配置生成专家。根据数据和图表类型，生成完整的 ECharts option 配置。

要求:
1. 返回合法的 ECharts option JSON
2. 配置必须包含: title, tooltip, xAxis/yAxis（或对应坐标系）, series
3. 使用现代简洁的配色方案
4. 数据直接嵌入配置中（不要引用外部变量）
5. 开启 tooltip 的 trigger: "axis"（折线图/柱状图）
6. 对中文标签设置合理的 rotate 防止重叠

只返回 JSON 对象，不要包含 ```json 标记。
"""


class EChartsCodeGenerator:
    def __init__(self, api_key: str, model: str = "gpt-4o"):
        self.client = AsyncOpenAI(api_key=api_key)
        self.model = model

    async def generate(
        self,
        columns: list[str],
        rows: list[list],
        chart_type: ChartType,
        user_query: str,
    ) -> ChartConfig:
        sample_rows = rows[:10]
        prompt = f"""用户问题: {user_query}
图表类型: {chart_type.value}
数据列: {columns}
数据: {json.dumps(sample_rows, ensure_ascii=False)}

请生成 ECharts option。"""

        response = await self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": ECHARTS_GEN_PROMPT},
                {"role": "user", "content": prompt},
            ],
            temperature=0,
            response_format={"type": "json_object"},
        )

        option = json.loads(response.choices[0].message.content)
        is_valid = self._validate_option(option)

        return ChartConfig(
            library="echarts", option=option, is_valid=is_valid,
        )

    def _validate_option(self, option: dict) -> bool:
        if not isinstance(option, dict):
            return False
        has_series = "series" in option
        has_axis = "xAxis" in option or "yAxis" in option
        has_polar = "polar" in option or "radiusAxis" in option
        return has_series and (has_axis or has_polar)
```

### 2.6 Pipeline 编排器

```python
# backend/app/pipeline/orchestrator.py
import asyncio
import time
import uuid
from app.models import (
    PipelineResponse, SQLResult, UserIntent, IntentType,
)
from app.pipeline.intent import IntentParser
from app.pipeline.nl2sql import NL2SQLEngine
from app.pipeline.executor import SQLExecutor
from app.pipeline.recommender import ChartRecommender
from app.pipeline.codegen import EChartsCodeGenerator
from app.session.context import ConversationContext


class PipelineOrchestrator:
    MAX_SQL_RETRIES = 2

    def __init__(
        self,
        intent_parser: IntentParser,
        nl2sql: NL2SQLEngine,
        executor: SQLExecutor,
        recommender: ChartRecommender,
        codegen: EChartsCodeGenerator,
    ):
        self.intent_parser = intent_parser
        self.nl2sql = nl2sql
        self.executor = executor
        self.recommender = recommender
        self.codegen = codegen

    async def run(
        self,
        user_query: str,
        schema_text: str,
        context: ConversationContext,
    ) -> PipelineResponse:
        request_id = str(uuid.uuid4())[:8]
        start = time.time()

        try:
            effective_query = context.merge_with_history(user_query)

            intent = await self.intent_parser.parse(effective_query)

            sql = await self.nl2sql.generate(intent, schema_text, effective_query)
            if not sql:
                return PipelineResponse(
                    request_id=request_id, success=False,
                    error="无法为该问题生成 SQL 查询，请尝试换个问法。",
                    total_time_ms=self._elapsed(start),
                )

            query_result = await self._execute_with_retry(
                sql, schema_text, effective_query
            )

            recommendation, chart_config = await asyncio.gather(
                self.recommender.recommend(
                    query_result.columns, query_result.rows, user_query
                ),
                self.codegen.generate(
                    query_result.columns, query_result.rows,
                    (await self.recommender.recommend(
                        query_result.columns, query_result.rows, user_query
                    )).chart_type,
                    user_query,
                ),
            )

            summary = self._build_summary(query_result, recommendation)

            context.add_turn(
                user_query=user_query,
                sql=sql,
                columns=query_result.columns,
                rows=query_result.rows,
            )

            return PipelineResponse(
                request_id=request_id,
                success=True,
                intent=intent,
                sql=sql,
                data_columns=query_result.columns,
                data_rows=query_result.rows,
                chart=chart_config,
                recommendation=recommendation,
                summary=summary,
                total_time_ms=self._elapsed(start),
            )

        except Exception as e:
            return PipelineResponse(
                request_id=request_id,
                success=False,
                error=str(e),
                total_time_ms=self._elapsed(start),
            )

    async def _execute_with_retry(
        self, sql: str, schema_text: str, user_query: str
    ) -> SQLResult:
        current_sql = sql
        for attempt in range(self.MAX_SQL_RETRIES + 1):
            try:
                return await self.executor.execute(current_sql)
            except Exception as e:
                if attempt < self.MAX_SQL_RETRIES:
                    current_sql = await self.nl2sql.fix_sql(
                        current_sql, str(e), schema_text, user_query
                    )
                else:
                    raise

    def _build_summary(self, result: SQLResult, rec) -> str:
        if result.row_count == 0:
            return "查询结果为空，请尝试调整时间范围或筛选条件。"
        return (
            f"查询返回 {result.row_count} 行数据，"
            f"推荐使用{rec.chart_type.value}展示。{rec.reason}"
        )

    def _elapsed(self, start: float) -> int:
        return int((time.time() - start) * 1000)
```

注意上面的代码中有一个问题：`recommend` 被调用了两次。实际项目中应该只调用一次，把结果存下来传给 `codegen`。这里是为了展示 Pipeline 的典型写法，生产代码需要修复这个冗余。

### 2.7 会话管理与对话上下文

```python
# backend/app/session/context.py
from dataclasses import dataclass, field
from typing import Optional


@dataclass
class ConversationTurn:
    turn_id: int
    user_query: str
    sql: Optional[str] = None
    columns: Optional[list[str]] = None
    rows: Optional[list[list]] = None


@dataclass
class ConversationContext:
    session_id: str
    turns: list[ConversationTurn] = field(default_factory=list)
    max_history: int = 10

    def add_turn(
        self, user_query: str, sql: str = None,
        columns: list[str] = None, rows: list[list] = None,
    ):
        turn = ConversationTurn(
            turn_id=len(self.turns) + 1,
            user_query=user_query,
            sql=sql,
            columns=columns,
            rows=rows,
        )
        self.turns.append(turn)
        if len(self.turns) > self.max_history:
            self.turns = self.turns[-self.max_history:]

    def merge_with_history(self, current_query: str) -> str:
        if not self.turns:
            return current_query

        last = self.turns[-1]

        modification_keywords = [
            "换成", "改成", "只看", "加上", "去掉", "不要",
            "按月", "按周", "按天", "排序", "前10", "后10",
        ]
        is_modification = any(kw in current_query for kw in modification_keywords)

        if is_modification and last.sql:
            history_context = "\n".join(
                f"第 {t.turn_id} 轮: 用户问 '{t.user_query}'"
                for t in self.turns[-3:]
            )
            return (
                f"对话历史:\n{history_context}\n\n"
                f"当前用户请求: {current_query}\n"
                f"上一轮生成的 SQL: {last.sql}\n"
                f"请基于上下文理解用户意图，生成新的 SQL。"
            )

        return current_query

    def get_recent_queries(self, n: int = 3) -> list[str]:
        return [t.user_query for t in self.turns[-n:]]
```

// frontend/src/hooks/useChat.ts
import { useState, useCallback, useRef, useEffect } from "react";
import type { PipelineResponse } from "../types";

interface ChatEntry {
  id: string;
  role: "user" | "assistant";
  content: string;
  result?: PipelineResponse;
  timestamp: Date;
}

export function useChat(sessionId: string) {
  const [entries, setEntries] = useState<ChatEntry[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const wsRef = useRef<WebSocket | null>(null);

  useEffect(() => {
    const ws = new WebSocket(`ws://localhost:8000/ws/chat/${sessionId}`);

    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      if (data.type === "status") {
        setIsLoading(true);
      } else if (data.type === "result") {
        setIsLoading(false);
        setEntries((prev) => {
          const updated = [...prev];
          const lastAssistant = updated.findLast((e) => e.role === "assistant");
          if (lastAssistant) {
            lastAssistant.result = data.content;
            lastAssistant.content = data.content.summary || "查询完成";
          }
          return [...updated];
        });
      }
    };

    wsRef.current = ws;
    return () => ws.close();
  }, [sessionId]);

  const sendMessage = useCallback((content: string) => {
    const userEntry: ChatEntry = {
      id: Date.now().toString(),
      role: "user",
      content,
      timestamp: new Date(),
    };
    const assistantEntry: ChatEntry = {
      id: (Date.now() + 1).toString(),
      role: "assistant",
      content: "正在分析...",
      timestamp: new Date(),
    };

    setEntries((prev) => [...prev, userEntry, assistantEntry]);
    setIsLoading(true);

    wsRef.current?.send(JSON.stringify({ content }));
  }, []);

  return { entries, sendMessage, isLoading };
}

```

优化的具体手段：

3. **Few-shot 积累**：收集用户修正的 SQL 作为 Few-shot 示例
4. **阈值调整**：根据反馈数据调整图表推荐的置信度阈值

## 常见误区

**误区一：把所有逻辑塞进一个 API 接口。** 对话接口和反馈接口应该分离。WebSocket 负责实时对话，HTTP REST 负责反馈和管理操作。混在一起会让代码难以维护。

**误区二：不做 Schema 注释就上 NL2SQL。** 数据库的列名往往是缩写（`amt`、`qty`、`reg`），LLM 无法猜出含义。上线前必须给 Schema 加上中文注释，这是 NL2SQL 准确率最简单的提升手段。

**误区三：忽略对话上下文的长度限制。** 每轮对话都会往 Prompt 里塞历史信息，10 轮之后 Prompt 可能超过模型的上下文窗口。必须设置 `max_history_turns` 并做截断。

**误区四：前端不处理图表渲染失败。** LLM 生成的 ECharts 配置可能有语法错误。前端必须有 try-catch 包裹 `setOption`，渲染失败时降级显示数据表格。

### 练习二

**思路**：用 Redis 的 Hash 存储会话元数据，List 存储对话轮次。每次新对话 `LPUSH` 到 List 头部，读取时 `LRANGE` 取出。

**答案**：

```python
# backend/app/session/redis_store.py
import json
import redis.asyncio as redis
from app.session.context import ConversationContext, ConversationTurn


class RedisSessionStore:
    SESSION_TTL = 86400

    def __init__(self, redis_url: str):
        self.redis = redis.from_url(redis_url)

    def _session_key(self, session_id: str) -> str:
        return f"session:{session_id}"

    def _turns_key(self, session_id: str) -> str:
        return f"session:{session_id}:turns"

    async def save_session(self, ctx: ConversationContext):
        session_key = self._session_key(ctx.session_id)
        turns_key = self._turns_key(ctx.session_id)

        await self.redis.hset(session_key, mapping={
            "session_id": ctx.session_id,
            "max_history": str(ctx.max_history),
        })
        await self.redis.expire(session_key, self.SESSION_TTL)

        if ctx.turns:
            turn_data = json.dumps([
                {
                    "turn_id": t.turn_id,
                    "user_query": t.user_query,
                    "sql": t.sql,
                    "columns": t.columns,
                    "rows": t.rows,
                }
                for t in ctx.turns
            ])
            await self.redis.set(turns_key, turn_data, ex=self.SESSION_TTL)

    async def load_session(self, session_id: str) -> ConversationContext:
        session_key = self._session_key(session_id)
        turns_key = self._turns_key(session_id)

        exists = await self.redis.exists(session_key)
        if not exists:
            return ConversationContext(session_id=session_id)

        session_data = await self.redis.hgetall(session_key)
        max_history = int(session_data.get(b"max_history", 10))

        turns = []
        turns_json = await self.redis.get(turns_key)
        if turns_json:
            turns_data = json.loads(turns_json)
            turns = [ConversationTurn(**t) for t in turns_data]

        return ConversationContext(
            session_id=session_id,
            turns=turns,
            max_history=max_history,
        )

    async def delete_session(self, session_id: str):
        await self.redis.delete(
            self._session_key(session_id),
            self._turns_key(session_id),
        )
```

**要点**：
- Hash 存元数据，String 存 JSON 序列化的对话轮次
- 每次请求先 `load_session`，处理完后 `save_session`
- TTL 24 小时，过期后自动清理
- 生产环境中 `rows` 数据可能很大，可以考虑只存 SQL 不存完整结果，需要时重新查询
