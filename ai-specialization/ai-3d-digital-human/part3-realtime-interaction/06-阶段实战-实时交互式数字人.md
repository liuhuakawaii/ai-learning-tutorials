# 第20课：阶段实战——实时交互式数字人

## 场景引入

过去两课我们分别学习了多模态交互设计和对话系统架构。现在是把所有零件组装成一台完整机器的时候了。

想象这个场景：用户打开浏览器，看到一个 3D 数字人在屏幕上自然地呼吸着。他对着麦克风说"你好"，数字人微微转头看向他，嘴角上扬，然后用自然的声音回应。整个过程从语音输入到音频播放的延迟不超过 800ms。用户中途插嘴"等等"，数字人立刻停下来，转为倾听状态。

这就是本课的目标：构建一个端到端的实时交互式数字人系统。

## 学习目标

完成本课学习后，你将能够：

1. 设计端到端的实时交互式数字人系统架构
2. 使用 FastAPI 构建异步后端 Pipeline（ASR → LLM → TTS → 面部驱动）
3. 使用 Three.js 渲染 3D 数字人并驱动表情和动作
4. 通过 WebSocket 实现前后端实时双向通信
5. 使用 Docker Compose 部署完整系统

#### 2. 流式分句器

```python
# backend/app/pipeline/sentence_splitter.py
from typing import Optional


class SentenceSplitter:
    """流式分句器：将 LLM token 流切分为句子，用于 TTS 合成"""

    STRONG_ENDS = {"。", "！", "？", "!", "?"}
    WEAK_ENDS = {"，", ",", "、", ";"}

    def __init__(self, min_len: int = 8, max_len: int = 30):
        self.min_len = min_len
        self.max_len = max_len
        self._buffer = ""

    def feed(self, token: str) -> Optional[str]:
        self._buffer += token

        # 强断句符 → 立即切分
        for i, ch in enumerate(self._buffer):
            if ch in self.STRONG_ENDS and i >= self.min_len:
                sentence = self._buffer[:i + 1].strip()
                self._buffer = self._buffer[i + 1:]
                if sentence:
                    return sentence

        # 超过最大长度 → 弱断句处切分
        if len(self._buffer) > self.max_len:
            for sep in self.WEAK_ENDS:
                idx = self._buffer.rfind(sep, self.min_len)
                if idx > 0:
                    sentence = self._buffer[:idx + 1].strip()
                    self._buffer = self._buffer[idx + 1:]
                    return sentence
            # 硬切
            sentence = self._buffer[:self.max_len].strip()
            self._buffer = self._buffer[self.max_len:]
            return sentence

        return None

    def flush(self) -> Optional[str]:
        if self._buffer.strip():
            result = self._buffer.strip()
            self._buffer = ""
            return result
        return None
```

#### 3. 对话管线

```python
# backend/app/pipeline/dialogue_pipeline.py
import asyncio
import json
from typing import AsyncGenerator, List, Dict, Optional
from dataclasses import dataclass, field

import httpx
from ..config import settings
from .sentence_splitter import SentenceSplitter


@dataclass
class ConversationContext:
    session_id: str
    messages: List[dict] = field(default_factory=list)
    is_interrupted: bool = False
    state: str = "idle"


SYSTEM_PROMPT = """你是小智，智慧大厅的数字人助手。

## 规则
1. 每句话不超过 25 个字，适合语音播报
2. 语气亲切自然，像面对面聊天
3. 用【】标记需要强调的词语
4. 不确定时坦诚告知
"""


class DialoguePipeline:
    """完整对话管线：ASR → LLM → TTS → 表情"""

    def __init__(self):
        self._contexts: Dict[str, ConversationContext] = {}

    def get_context(self, session_id: str) -> ConversationContext:
        if session_id not in self._contexts:
            self._contexts[session_id] = ConversationContext(
                session_id=session_id
            )
        return self._contexts[session_id]

    async def process_user_input(
        self, session_id: str, text: str, send_fn
    ):
        """处理用户输入，通过 send_fn 发送结果"""
        ctx = self.get_context(session_id)
        ctx.state = "thinking"
        ctx.is_interrupted = False

        await send_fn({"type": "state", "state": "thinking"})

        ctx.messages.append({"role": "user", "content": text})
        if len(ctx.messages) > settings.max_context_messages:
            ctx.messages = ctx.messages[-settings.max_context_messages:]

        llm_messages = [{"role": "system", "content": SYSTEM_PROMPT}]
        llm_messages.extend(ctx.messages)

        ctx.state = "speaking"
        await send_fn({"type": "state", "state": "speaking"})

        splitter = SentenceSplitter()
        full_response = ""

        async for token in self._stream_llm(llm_messages):
            if ctx.is_interrupted:
                break

            full_response += token
            await send_fn({"type": "token", "content": token})

            sentence = splitter.feed(token)
            if sentence:
                await self._handle_sentence(sentence, send_fn)

        remaining = splitter.flush()
        if remaining and not ctx.is_interrupted:
            await self._handle_sentence(remaining, send_fn)

        if full_response:
            ctx.messages.append({
                "role": "assistant", "content": full_response
            })

        ctx.state = "idle"
        await send_fn({"type": "state", "state": "idle"})

    async def _handle_sentence(self, sentence: str, send_fn):
        """并行执行 TTS 和情感计算"""
        tts_task = asyncio.create_task(self._synthesize_tts(sentence))
        emotion_task = asyncio.create_task(self._compute_emotion(sentence))
        audio_data, emotion = await asyncio.gather(tts_task, emotion_task)

        if emotion:
            await send_fn({"type": "emotion", "data": emotion})
        if audio_data:
            await send_fn({"type": "audio", "data": audio_data})

    async def _stream_llm(
        self, messages: List[dict]
    ) -> AsyncGenerator[str, None]:
        async with httpx.AsyncClient() as client:
            try:
                async with client.stream(
                    "POST",
                    f"{settings.openai_base_url}/chat/completions",
                    headers={
                        "Authorization": f"Bearer {settings.openai_api_key}",
                        "Content-Type": "application/json"
                    },
                    json={
                        "model": settings.llm_model,
                        "messages": messages,
                        "stream": True,
                        "temperature": 0.7,
                        "max_tokens": 800
                    },
                    timeout=30.0
                ) as resp:
                    async for line in resp.aiter_lines():
                        if not line.startswith("data: "):
                            continue
                        if line == "data: [DONE]":
                            break
                        try:
                            data = json.loads(line[6:])
                            content = data["choices"][0].get(
                                "delta", {}
                            ).get("content", "")
                            if content:
                                yield content
                        except (json.JSONDecodeError, KeyError):
                            continue
            except Exception as e:
                print(f"LLM 调用失败: {e}")
                yield "抱歉，我暂时无法回答。"

    async def _synthesize_tts(self, text: str) -> Optional[bytes]:
        try:
            async with httpx.AsyncClient() as client:
                resp = await client.post(
                    f"{settings.tts_url}/tts",
                    json={
                        "text": text,
                        "voice_id": settings.tts_voice_id,
                        "format": "wav"
                    },
                    timeout=10.0
                )
                return resp.content
        except Exception as e:
            print(f"TTS 失败: {e}")
            return None

    async def _compute_emotion(self, text: str) -> Optional[dict]:
        positive = {"好", "棒", "开心", "感谢", "欢迎"}
        negative = {"抱歉", "遗憾", "不满"}

        valence = sum(0.3 for w in positive if w in text)
        valence -= sum(0.3 for w in negative if w in text)
        valence = max(-1, min(1, valence))

        return {
            "smile": max(0, min(1, 0.3 + valence * 0.5)),
            "brow_raise": max(0, valence * 0.3),
        }

    def handle_interrupt(self, session_id: str):
        ctx = self.get_context(session_id)
        ctx.is_interrupted = True
        ctx.state = "listening"
```

```python
import asyncio
import json
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from fastapi.middleware.cors import CORSMiddleware

from .config import settings
from .pipeline.dialogue_pipeline import DialoguePipeline


app = FastAPI(title="实时交互式数字人服务")
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

dialogue = DialoguePipeline()


@app.websocket("/ws/{session_id}")
async def websocket_endpoint(websocket: WebSocket, session_id: str):
    await websocket.accept()
    ctx = dialogue.get_context(session_id)

    async def send_fn(data: dict):
        if isinstance(data.get("data"), bytes):
            await websocket.send_bytes(data["data"])
        else:
            await websocket.send_json(data)

    try:
        while True:
            message = await websocket.receive()

            if "text" in message:
                data = json.loads(message["text"])
                msg_type = data.get("type")

                if msg_type == "user_text" and data.get("is_final", True):
                    asyncio.create_task(
                        dialogue.process_user_input(
                            session_id, data["text"], send_fn
                        )
                    )

                elif msg_type == "interrupt":
                    dialogue.handle_interrupt(session_id)
                    await send_fn({"type": "state", "state": "listening"})

                elif msg_type == "gesture":
                    await send_fn({
                        "type": "animation",
                        "name": data.get("gesture_type", "idle")
                    })

    except WebSocketDisconnect:
        pass


@app.get("/health")
async def health():
    return {"status": "ok", "version": "1.0.0"}


if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

#### 1. WebSocket 客户端

```typescript
// frontend/src/services/ws_client.ts
export type MessageHandler = (data: any) => void;

export class DigitalHumanWS {
  private ws: WebSocket | null = null;
  private handlers: Map<string, MessageHandler[]> = new Map();
  private binaryHandlers: ((data: ArrayBuffer) => void)[] = [];
  private sessionId: string;

  constructor(private baseUrl: string = "ws://localhost:8000") {
    this.sessionId = "session_" + Date.now();
  }

  connect(): Promise<void> {
    return new Promise((resolve, reject) => {
      this.ws = new WebSocket(`${this.baseUrl}/ws/${this.sessionId}`);
      this.ws.onopen = () => resolve();
      this.ws.onmessage = (e) => {
        if (typeof e.data === "string") {
          const d = JSON.parse(e.data);
          (this.handlers.get(d.type) || []).forEach((h) => h(d));
        } else {
          this.binaryHandlers.forEach((h) => h(e.data));
        }
      };
      this.ws.onclose = () => {
        setTimeout(() => this.connect(), 3000);
      };
      this.ws.onerror = reject;
    });
  }

  on(type: string, handler: MessageHandler) {
    const list = this.handlers.get(type) || [];
    list.push(handler);
    this.handlers.set(type, list);
  }

  onBinary(handler: (data: ArrayBuffer) => void) {
    this.binaryHandlers.push(handler);
  }

  sendText(text: string, isFinal = true) {
    this.ws?.send(JSON.stringify({
      type: "user_text", text, is_final: isFinal,
    }));
  }

  sendInterrupt() {
    this.ws?.send(JSON.stringify({ type: "interrupt" }));
  }

  sendGesture(gestureType: string) {
    this.ws?.send(JSON.stringify({
      type: "gesture", gesture_type: gestureType,
    }));
  }

  sendAudio(data: ArrayBuffer) {
    this.ws?.send(data);
  }
}
```

```tsx
// frontend/src/components/AvatarScene.tsx
import React, { useRef, useEffect } from "react";
import { Canvas, useFrame } from "@react-three/fiber";
import { OrbitControls, Environment } from "@react-three/drei";
import * as THREE from "three";

interface AvatarState {
  smile: number;
  browRaise: number;
  viseme: string;
  animation: string;
}

function AvatarModel({ state }: { state: AvatarState }) {
  const groupRef = useRef<THREE.Group>(null);
  const mixerRef = useRef<THREE.AnimationMixer | null>(null);

  useEffect(() => {
    const loader = new THREE.GLTFLoader();
    loader.load("/models/avatar.glb", (gltf) => {
      const model = gltf.scene;
      model.position.set(0, -1.5, 0);
      groupRef.current?.add(model);
      mixerRef.current = new THREE.AnimationMixer(model);
      gltf.animations.forEach((clip) => {
        const action = mixerRef.current!.clipAction(clip);
        if (clip.name === "idle") action.play();
      });
    });
  }, []);

  useEffect(() => {
    groupRef.current?.traverse((child: any) => {
      if (!child.isMesh || !child.morphTargetInfluences) return;
      const dict = child.morphTargetDictionary;
      if (!dict) return;

      if ("smile" in dict) {
        child.morphTargetInfluences[dict.smile] = state.smile;
      }
      if ("browUp" in dict) {
        child.morphTargetInfluences[dict.browUp] = state.browRaise;
      }

      const visemeKey = `viseme_${state.viseme}`;
      Object.keys(dict).forEach((key) => {
        if (key.startsWith("viseme_")) {
          child.morphTargetInfluences[dict[key]] =
            key === visemeKey ? 0.8 : 0;
        }
      });
    });
  }, [state]);

  useFrame((_, delta) => mixerRef.current?.update(delta));

  return <group ref={groupRef} />;
}

export function AvatarScene({ state }: { state: AvatarState }) {
  return (
    <Canvas camera={{ position: [0, 0, 3], fov: 45 }}>
      <ambientLight intensity={0.5} />
      <directionalLight position={[5, 5, 5]} intensity={0.8} />
      <AvatarModel state={state} />
      <Environment preset="studio" />
      <OrbitControls enablePan={false} enableZoom={false} />
    </Canvas>
  );
```

```tsx
// frontend/src/App.tsx
import React, { useState, useEffect, useRef, useCallback } from "react";
import { AvatarScene } from "./components/AvatarScene";
import { DigitalHumanWS } from "./services/ws_client";

interface Message {
  role: "user" | "assistant";
  content: string;
}

interface AvatarState {
  smile: number;
  browRaise: number;
  viseme: string;
  animation: string;
}

export default function App() {
  const [messages, setMessages] = useState<Message[]>([]);
  const [inputText, setInputText] = useState("");
  const [systemState, setSystemState] = useState("idle");
  const [avatarState, setAvatarState] = useState<AvatarState>({
    smile: 0.1, browRaise: 0, viseme: "sil", animation: "idle",
  });

  const wsRef = useRef<DigitalHumanWS | null>(null);
  const audioQueue = useRef<Blob[]>([]);
  const isPlaying = useRef(false);

  useEffect(() => {
    const ws = new DigitalHumanWS();
    wsRef.current = ws;

    ws.connect().then(() => {
      ws.on("state", (d) => setSystemState(d.state));
      ws.on("token", (d) => {
        setMessages((prev) => {
          const last = prev[prev.length - 1];
          if (last?.role === "assistant") {
            return [...prev.slice(0, -1), {
              ...last, content: last.content + d.content
            }];
          }
          return [...prev, { role: "assistant", content: d.content }];
        });
      });
      ws.on("emotion", (d) => {
        setAvatarState((prev) => ({
          ...prev, smile: d.data.smile ?? prev.smile,
          browRaise: d.data.brow_raise ?? prev.browRaise,
        }));
      });
      ws.on("animation", (d) => {
        setAvatarState((prev) => ({ ...prev, animation: d.name }));
        setTimeout(() => setAvatarState((prev) => ({
          ...prev, animation: "idle"
        })), 3000);
      });
      ws.onBinary((data) => {
        audioQueue.current.push(new Blob([data], { type: "audio/wav" }));
        playNext();
      });
    });

    return () => ws.disconnect();
  }, []);

  const playNext = useCallback(() => {
    if (isPlaying.current || audioQueue.current.length === 0) return;
    isPlaying.current = true;
    const blob = audioQueue.current.shift()!;
    const audio = new Audio(URL.createObjectURL(blob));

    // 唇形同步：根据音频播放进度驱动 viseme
    const visemeSequence = ["aa", "E", "I", "O", "U", "aa", "sil"];
    let visemeIdx = 0;
    const interval = setInterval(() => {
      if (visemeIdx < visemeSequence.length) {
        setAvatarState((prev) => ({
          ...prev, viseme: visemeSequence[visemeIdx++]
        }));
      }
    }, 120);

    audio.onended = () => {
      clearInterval(interval);
      isPlaying.current = false;
      setAvatarState((prev) => ({ ...prev, viseme: "sil" }));
      playNext();
    };
    audio.play();
  }, []);

  const handleSend = () => {
    if (!inputText.trim()) return;
    setMessages((prev) => [...prev, { role: "user", content: inputText }]);
    wsRef.current?.sendText(inputText);
    setInputText("");
  };

  const handleInterrupt = () => {
    wsRef.current?.sendInterrupt();
    audioQueue.current = [];
    isPlaying.current = false;
    setAvatarState((prev) => ({ ...prev, animation: "idle", viseme: "sil" }));
  };

  const stateLabels: Record<string, { text: string; color: string }> = {
    idle: { text: "就绪", color: "#4ade80" },
    listening: { text: "聆听中", color: "#60a5fa" },
    thinking: { text: "思考中", color: "#fbbf24" },
    speaking: { text: "回答中", color: "#f472b6" },
  };
  const current = stateLabels[systemState] || stateLabels.idle;

  return (
    <div style={{
      height: "100vh", display: "flex", flexDirection: "column",
      background: "#0a0a1a", color: "#e0e0e0"
    }}>
      <div style={{
        padding: "12px 24px", display: "flex", alignItems: "center", gap: 8,
        borderBottom: "1px solid rgba(255,255,255,0.1)"
      }}>
        <div style={{
          width: 10, height: 10, borderRadius: "50%",
          backgroundColor: current.color
        }} />
        <span>{current.text}</span>
      </div>

      <div style={{ flex: 1, display: "flex", overflow: "hidden" }}>
        <div style={{
          flex: 1,
          background: "radial-gradient(ellipse at center, #1a1a3e, #0a0a1a)"
        }}>
          <AvatarScene state={avatarState} />
        </div>

        <div style={{
          width: 380, display: "flex", flexDirection: "column",
          borderLeft: "1px solid rgba(255,255,255,0.1)"
        }}>
          <div style={{
            flex: 1, overflowY: "auto", padding: 16,
            display: "flex", flexDirection: "column", gap: 10
          }}>
            {messages.map((msg, i) => (
              <div key={i} style={{

                  msg.role === "user" ? "flex-end" : "flex-start",
                background: msg.role === "user"
                  ? "#3b82f6" : "rgba(255,255,255,0.1)",
                color: msg.role === "user" ? "white" : "#e0e0e0",
              }}>
                {msg.content}
              </div>
            ))}
          </div>

          <div style={{
            padding: 12, display: "flex", gap: 8,
            borderTop: "1px solid rgba(255,255,255,0.1)"
          }}>
            <input
              style={{
                flex: 1, padding: "8px 12px", borderRadius: 8,
                border: "1px solid rgba(255,255,255,0.2)",
                background: "rgba(255,255,255,0.05)", color: "#e0e0e0",
                fontSize: 14, outline: "none",
              }}
              value={inputText}
              onChange={(e) => setInputText(e.target.value)}
              onKeyDown={(e) => e.key === "Enter" && handleSend()}
              placeholder="输入消息..."
            />
            <button onClick={handleSend} style={{
              padding: "8px 14px", borderRadius: 8, border: "none",
              background: "#3b82f6", color: "white", cursor: "pointer",
            }}>发送</button>
            {(systemState === "speaking" || systemState === "thinking") && (
              <button onClick={handleInterrupt} style={{
                padding: "8px 12px", borderRadius: 8, border: "none",
                background: "#f59e0b", color: "#000", cursor: "pointer",
              }}>打断</button>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
```

## 常见误区

### 误区一：前后端耦合过紧

很多初学者把 3D 渲染逻辑和后端通信混在一起。正确做法是分层：WebSocket 客户端是独立服务层，React 组件只负责渲染，Three.js 场景只负责 3D 表现。任何一层的修改都不应影响其他层。

### 误区二：忽略 GPU 资源管理

Whisper 和 CosyVoice 都需要 GPU。多用户同时连接时 GPU 内存可能不够。解决方案：使用请求队列 + 批处理，或者部署多个实例并用负载均衡分配。

### 误区三：音频播放不排队

前端收到多段 TTS 音频时，如果同时播放会混乱。必须实现播放队列：一段播完后再播下一段。打断时清空队列并停止当前播放。

### 误区四：忽略错误恢复

生产环境中任何服务都可能挂掉。Whisper 可能 OOM，CosyVoice 可能超时，GPT-4o 可能限流。每个服务调用都必须有超时、重试和降级策略。降级时数字人应保持"在场感"，比如显示"我正在思考，请稍等"。

### 误区五：不做延迟优化

端到端延迟是数字人体验的生命线。关键优化点：LLM 首 token 后立即启动 TTS（不等整句），TTS 首帧音频到达后立即开始播放（不等整段），前端用 AudioWorklet 替代普通 Audio 元素减少播放延迟。

## 参考答案

### 练习一

**思路**：在对话管线中增加 RAG 检索层，使用 ChromaDB 存储文档向量，查询时检索 Top-K 相关片段注入 System Prompt。

**答案**：

```python
import chromadb
from sentence_transformers import SentenceTransformer
from typing import List, Dict


class RAGService:
    def __init__(self, collection_name: str = "product_docs"):
        self.embedding_model = SentenceTransformer("BAAI/bge-small-zh-v1.5")
        self.chroma_client = chromadb.PersistentClient(path="./chroma_db")
        self.collection = self.chroma_client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"}
        )

    def ingest_documents(self, documents: List[Dict]):
        texts = [doc["content"] for doc in documents]
        embeddings = self.embedding_model.encode(texts).tolist()
        ids = [f"doc_{i}" for i in range(len(documents))]
        metadatas = [
            {"source": doc.get("source", ""), "title": doc.get("title", "")}
            for doc in documents
        ]
        self.collection.upsert(
            ids=ids, documents=texts,
            embeddings=embeddings, metadatas=metadatas
        )

    def retrieve(self, query: str, top_k: int = 3) -> List[Dict]:
        query_embedding = self.embedding_model.encode([query]).tolist()
        results = self.collection.query(
            query_embeddings=query_embedding, n_results=top_k
        )
        return [
            {"content": doc, "source": meta.get("source", ""), "distance": dist}
            for doc, meta, dist in zip(
                results["documents"][0],
                results["metadatas"][0],
                results["distances"][0]
            )
        ]

    def build_rag_prompt(self, query: str, threshold: float = 0.5) -> str:
        results = self.retrieve(query)
        relevant = [r for r in results if r["distance"] < threshold]
        if not relevant:
            return ""
        context = "\n\n".join(
            f"[参考{i+1}] {r['content']}" for i, r in enumerate(relevant)
        )
        return f"\n\n请基于以下参考资料回答：\n{context}"
```

**要点**：
- cosine 距离阈值 0.5 过滤低相关度结果
- RAG 上下文追加到 System Prompt 末尾，不覆盖原有指令
- 使用 bge-small 模型平衡效果和速度

### 练习二

**思路**：使用 Web Audio API 的 AnalyserNode 实时检测音频能量，结合状态机判断静音时长。

**答案**：

```typescript
class VoiceActivityDetector {
  private audioContext: AudioContext | null = null;
  private analyser: AnalyserNode | null = null;
  private dataArray: Uint8Array | null = null;
  private silenceStart: number = 0;
  private isSpeaking = false;

  constructor(
    private energyThreshold = 30,
    private silenceTimeout = 800,
    private onSpeechEnd: () => void
  ) {}

  async start(stream: MediaStream) {
    this.audioContext = new AudioContext({ sampleRate: 16000 });
    const source = this.audioContext.createMediaStreamSource(stream);
    this.analyser = this.audioContext.createAnalyser();
    this.analyser.fftSize = 256;
    source.connect(this.analyser);
    this.dataArray = new Uint8Array(this.analyser.frequencyBinCount);
    this.detect();
  }

  stop() {
    this.audioContext?.close();
    this.audioContext = null;
  }

  private detect = () => {
    if (!this.analyser || !this.dataArray) return;

    this.analyser.getByteFrequencyData(this.dataArray);
    const energy = this.dataArray.reduce((a, b) => a + b, 0)
      / this.dataArray.length;

    if (energy > this.energyThreshold) {
      this.isSpeaking = true;
      this.silenceStart = 0;
    } else if (this.isSpeaking) {
      if (this.silenceStart === 0) {
        this.silenceStart = Date.now();
      } else if (Date.now() - this.silenceStart > this.silenceTimeout) {
        this.isSpeaking = false;
        this.silenceStart = 0;
        this.onSpeechEnd();
        return;
    }

    requestAnimationFrame(this.detect);
  };

  getEnergy(): number {
    if (!this.dataArray) return 0;
    return this.dataArray.reduce((a, b) => a + b, 0)
      / this.dataArray.length;
  }
}
```

**要点**：
- energyThreshold 为 30 适合安静环境，嘈杂环境需适当提高
- silenceTimeout 为 800ms 平衡响应速度和误判率
- 用户停顿超过 2 秒但能量偶尔超过阈值时需要重置计时器

### 练习三

**思路**：在对话管线中维护情感轨迹，会话结束时计算摘要并持久化，新会话开始时加载并调整初始行为。

**答案**：

```python
import json
from pathlib import Path
from typing import List, Dict, Optional


class EmotionMemory:
    def __init__(self, storage_path: str = "./emotion_memory"):
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(exist_ok=True)
        self._trajectories: Dict[str, List[dict]] = {}

    def record(self, session_id: str, valence: float, arousal: float):
        if session_id not in self._trajectories:
            self._trajectories[session_id] = []
        self._trajectories[session_id].append({
            "valence": valence, "arousal": arousal
        })

    def save_session(self, session_id: str, user_id: str):
        trajectory = self._trajectories.get(session_id, [])
        if not trajectory:
            return

        avg_v = sum(t["valence"] for t in trajectory) / len(trajectory)
        avg_a = sum(t["arousal"] for t in trajectory) / len(trajectory)

        data = {
            "user_id": user_id,
            "avg_valence": round(avg_v, 3),
            "avg_arousal": round(avg_a, 3),
            "last_emotion": trajectory[-1],
            "session_count": self._get_session_count(user_id) + 1,
        }

        file_path = self.storage_path / f"{user_id}.json"
        file_path.write_text(json.dumps(data, ensure_ascii=False, indent=2))
        del self._trajectories[session_id]

    def load_user_state(self, user_id: str) -> Optional[dict]:
        file_path = self.storage_path / f"{user_id}.json"
        if not file_path.exists():
            return None
        return json.loads(file_path.read_text())

    def get_initial_params(self, user_id: str) -> dict:
        state = self.load_user_state(user_id)
        if not state:
            return {"smile": 0.3, "greeting": "您好！很高兴为您服务。"}

        if state["avg_valence"] < -0.3:
            return {"smile": 0.15, "greeting": "您好，希望今天能帮到您。"}
        elif state["avg_valence"] > 0.3:
            return {"smile": 0.6, "greeting": "您好呀！又见面了！"}
        return {"smile": 0.3, "greeting": "您好！很高兴为您服务。"}

    def _get_session_count(self, user_id: str) -> int:
        state = self.load_user_state(user_id)
        return state.get("session_count", 0) if state else 0
```

**要点**：
- 情感轨迹的平均值比单次值更有参考意义
- 负面情感历史对应更温和的开场，正面情感对应更活泼的开场
- JSON 文件存储适合单机部署，生产环境可替换为 Redis 或数据库
