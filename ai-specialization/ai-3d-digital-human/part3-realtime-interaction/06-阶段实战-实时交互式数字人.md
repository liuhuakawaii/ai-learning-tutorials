# 阶段实战：实时交互式数字人

用户打开浏览器，看到3D数字人在呼吸。说"你好"，数字人转头、嘴角上扬、用自然声音回应。语音输入到音频播放延迟不超过800ms。用户中途插嘴，数字人立刻停下来。

这就是本课目标：端到端实时交互式数字人系统。

## 系统架构

```
前端 (React + Three.js)
  麦克风 → WebSocket → 后端
  后端   → WebSocket → 3D场景 + 对话UI

后端 (FastAPI + Python)
  Whisper ASR → GPT-4o LLM → CosyVoice TTS → 表情驱动
  会话管理 / 流式分句 / 情感计算
```

技术选型：FastAPI（原生WebSocket、异步）、Whisper large-v3（中英文准确）、GPT-4o（流式、Function Calling）、CosyVoice 2（中文自然度高）、Three.js + R3F（声明式3D）。

## 流式分句器

LLM流式输出的token需要切分为句子，每句话立即送TTS合成。

```python
# backend/app/pipeline/sentence_splitter.py
from typing import Optional

class SentenceSplitter:
    STRONG_ENDS = {"。", "！", "？", "!", "?"}
    WEAK_ENDS = {"，", ",", "、", ";"}

    def __init__(self, min_len=8, max_len=30):
        self.min_len = min_len; self.max_len = max_len; self._buffer = ""

    def feed(self, token: str) -> Optional[str]:
        self._buffer += token
        for i, ch in enumerate(self._buffer):
            if ch in self.STRONG_ENDS and i >= self.min_len:
                sentence = self._buffer[:i+1].strip()
                self._buffer = self._buffer[i+1:]
                return sentence if sentence else None
        if len(self._buffer) > self.max_len:
            for sep in self.WEAK_ENDS:
                idx = self._buffer.rfind(sep, self.min_len)
                if idx > 0:
                    sentence = self._buffer[:idx+1].strip()
                    self._buffer = self._buffer[idx+1:]
                    return sentence
            sentence = self._buffer[:self.max_len].strip()
            self._buffer = self._buffer[self.max_len:]
            return sentence
        return None

    def flush(self) -> Optional[str]:
        if self._buffer.strip():
            result = self._buffer.strip(); self._buffer = ""; return result
        return None
```

## 对话管线

```python
# backend/app/pipeline/dialogue_pipeline.py
import asyncio, json, httpx
from typing import AsyncGenerator, List, Optional
from ..config import settings
from .sentence_splitter import SentenceSplitter

SYSTEM_PROMPT = """你是小智，智慧大厅的数字人助手。
1. 每句话不超过25字，适合语音播报
2. 语气亲切自然，像面对面聊天
3. 不确定时坦诚告知"""

class DialoguePipeline:
    def __init__(self): self._contexts = {}

    async def process_user_input(self, session_id, text, send_fn):
        ctx = self._contexts.setdefault(session_id, {"messages": [], "interrupted": False})
        ctx["interrupted"] = False
        await send_fn({"type": "state", "state": "thinking"})

        ctx["messages"].append({"role": "user", "content": text})
        if len(ctx["messages"]) > 20: ctx["messages"] = ctx["messages"][-20:]

        messages = [{"role": "system", "content": SYSTEM_PROMPT}] + ctx["messages"]
        await send_fn({"type": "state", "state": "speaking"})

        splitter = SentenceSplitter()
        full_response = ""
        async for token in self._stream_llm(messages):
            if ctx["interrupted"]: break
            full_response += token
            await send_fn({"type": "token", "content": token})
            sentence = splitter.feed(token)
            if sentence: await self._handle_sentence(sentence, send_fn)

        remaining = splitter.flush()
        if remaining and not ctx["interrupted"]: await self._handle_sentence(remaining, send_fn)
        if full_response: ctx["messages"].append({"role": "assistant", "content": full_response})
        await send_fn({"type": "state", "state": "idle"})

    async def _handle_sentence(self, sentence, send_fn):
        tts_task = asyncio.create_task(self._synthesize_tts(sentence))
        emotion_task = asyncio.create_task(self._compute_emotion(sentence))
        audio_data, emotion = await asyncio.gather(tts_task, emotion_task)
        if emotion: await send_fn({"type": "emotion", "data": emotion})
        if audio_data: await send_fn({"type": "audio", "data": audio_data})

    async def _stream_llm(self, messages) -> AsyncGenerator[str, None]:
        async with httpx.AsyncClient() as client:
            try:
                async with client.stream("POST", f"{settings.openai_base_url}/chat/completions",
                    headers={"Authorization": f"Bearer {settings.openai_api_key}"},
                    json={"model": settings.llm_model, "messages": messages, "stream": True, "temperature": 0.7},
                    timeout=30.0) as resp:
                    async for line in resp.aiter_lines():
                        if not line.startswith("data: "): continue
                        if line == "data: [DONE]": break
                        try:
                            content = json.loads(line[6:])["choices"][0].get("delta", {}).get("content", "")
                            if content: yield content
                        except: continue
            except: yield "抱歉，暂时无法回答。"

    async def _synthesize_tts(self, text):
        try:
            async with httpx.AsyncClient() as client:
                resp = await client.post(f"{settings.tts_url}/tts",
                    json={"text": text, "voice_id": settings.tts_voice_id}, timeout=10.0)
                return resp.content
        except: return None

    async def _compute_emotion(self, text):
        positive = {"好", "棒", "开心", "欢迎"}
        valence = sum(0.3 for w in positive if w in text) - sum(0.3 for w in {"抱歉", "遗憾"} if w in text)
        valence = max(-1, min(1, valence))
        return {"smile": max(0, min(1, 0.3 + valence*0.5))}

    def handle_interrupt(self, session_id):
        if session_id in self._contexts: self._contexts[session_id]["interrupted"] = True
```

## WebSocket服务

```python
# backend/app/main.py
import asyncio, json
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from .config import settings
from .pipeline.dialogue_pipeline import DialoguePipeline

app = FastAPI(); dialogue = DialoguePipeline()

@app.websocket("/ws/{session_id}")
async def ws_endpoint(websocket: WebSocket, session_id: str):
    await websocket.accept()
    async def send_fn(data):
        if isinstance(data.get("data"), bytes): await websocket.send_bytes(data["data"])
        else: await websocket.send_json(data)
    try:
        while True:
            message = await websocket.receive()
            if "text" in message:
                data = json.loads(message["text"])
                if data.get("type") == "user_text":
                    asyncio.create_task(dialogue.process_user_input(session_id, data["text"], send_fn))
                elif data.get("type") == "interrupt":
                    dialogue.handle_interrupt(session_id)
                    await send_fn({"type": "state", "state": "listening"})
    except WebSocketDisconnect: pass
```

## 前端：3D场景+WebSocket

```typescript
// frontend/src/services/ws_client.ts
export class DigitalHumanWS {
  private ws: WebSocket | null = null;
  private handlers: Map<string, Function[]> = new Map();
  constructor(private baseUrl = "ws://localhost:8000") {}
  connect(sessionId: string) {
    this.ws = new WebSocket(`${this.baseUrl}/ws/${sessionId}`);
    this.ws.onmessage = (e) => {
      if (typeof e.data === "string") {
        const d = JSON.parse(e.data);
        (this.handlers.get(d.type) || []).forEach(h => h(d));
      }
    };
  }
  on(type: string, handler: Function) {
    this.handlers.set(type, [...(this.handlers.get(type) || []), handler]);
  }
  sendText(text: string) { this.ws?.send(JSON.stringify({ type: "user_text", text })); }
  sendInterrupt() { this.ws?.send(JSON.stringify({ type: "interrupt" })); }
}
```

前端React组件接收WebSocket消息，驱动Three.js模型的Blendshape权重，播放TTS音频。打断时清空音频队列并重置表情。

## 参考答案

### 练习一

```python
import chromadb
from sentence_transformers import SentenceTransformer

class RAGService:
    def __init__(self, collection_name="docs"):
        self.model = SentenceTransformer("BAAI/bge-small-zh-v1.5")
        self.collection = chromadb.PersistentClient(path="./chroma_db").get_or_create_collection(
            name=collection_name, metadata={"hnsw:space": "cosine"})

    def retrieve(self, query, top_k=3):
        emb = self.model.encode([query]).tolist()
        results = self.collection.query(query_embeddings=emb, n_results=top_k)
        return [{"content": d, "distance": dist} for d, dist in zip(results["documents"][0], results["distances"][0])]

    def build_prompt(self, query, threshold=0.5):
        results = self.retrieve(query)
        relevant = [r for r in results if r["distance"] < threshold]
        if not relevant: return ""
        context = "\n".join(f"[参考{i+1}] {r['content']}" for i, r in enumerate(relevant))
        return f"\n\n请基于以下资料回答：\n{context}"
```
