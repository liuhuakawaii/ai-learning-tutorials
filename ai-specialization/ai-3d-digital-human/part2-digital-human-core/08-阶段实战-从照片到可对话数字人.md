# 阶段实战：从照片到可对话数字人

前几课分别学了3D人脸重建、语音克隆、面部动画、全身动作。客户要的不是单个模块，而是**一张照片丢进去，出来一个能说话、能动、能对话的数字人**。

这节课把Part 2所有技术串成完整Pipeline，重点是模块衔接、延迟优化和工程健壮性。

## Pipeline数据流

```
照片(.jpg) → DECA/NextFace → 3DMM参数 → 模型(.glb)
参考音频(.wav) → Speaker Encoder → 声纹向量(256d) → CosyVoice → TTS音频
TTS音频 → Wav2Lip → blendshape权重 → Three.js渲染
用户语音 → Whisper → 文字 → LLM → 回复文字 → TTS → 音频+动画
```

## Step 1：3D人脸重建

```python
# server/modules/face_recon.py
import numpy as np, cv2
from pathlib import Path

class FaceReconstructor:
    def __init__(self, model_path="data/models/nextface"):
        import torch
        from nextface.facerecon import NextFaceRecon
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.model = NextFaceRecon(Path(model_path), device=self.device)

    def reconstruct(self, image_path, output_dir="data/models/avatar"):
        output_dir = Path(output_dir); output_dir.mkdir(parents=True, exist_ok=True)
        img = cv2.imread(image_path)
        if img is None: raise FileNotFoundError(f"无法读取: {image_path}")
        # 人脸检测与裁剪
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        faces = cv2.CascadeClassifier(cv2.data.haarcascades + "haarcascade_frontalface_default.xml"
            ).detectMultiScale(gray, 1.1, 5)
        if len(faces) == 0: raise ValueError("未检测到人脸")
        x, y, w, h = max(faces, key=lambda f: f[2]*f[3])
        margin = int(max(w, h) * 0.3)
        cropped = cv2.resize(img[max(0,y-margin):y+h+margin, max(0,x-margin):x+w+margin], (256, 256))

        result = self.model.reconstruct(cropped)
        np.save(output_dir / "3dmm_params.npy", result["3dmm"])
        np.save(output_dir / "blendshape_weights.npy", result["blendshape"])
        result["mesh"].export(str(output_dir / "face_mesh.glb"))
        return {"mesh_path": str(output_dir / "face_mesh.glb"),
                "blendshape_dim": result["blendshape"].shape[-1]}
```

```python
# server/modules/voice_clone.py
import numpy as np, torchaudio
from pathlib import Path

class VoiceCloner:
    def __init__(self, cosyvoice_path="pretrained_models/CosyVoice-300M"):
        import sys; sys.path.append(str(Path(cosyvoice_path).parent))
        from cosyvoice.cosyvoice import CosyVoice
        self.model = CosyVoice(cosyvoice_path)

    def clone_and_save(self, audio_path, output_dir="data/models/voice"):
        output_dir = Path(output_dir); output_dir.mkdir(parents=True, exist_ok=True)
        waveform, sr = torchaudio.load(audio_path)
        if sr != 16000: waveform = torchaudio.transforms.Resample(sr, 16000)(waveform)
        if waveform.shape[0] > 1: waveform = waveform.mean(dim=0, keepdim=True)
        waveform = waveform[:, :16000*15]  # 截取前15秒
        emb = self.model.extract_spk_embedding(waveform).numpy()
        np.save(output_dir / "speaker_embedding.npy", emb)
        return str(output_dir / "speaker_embedding.npy")

    def synthesize(self, text, spk_emb_path, output_path):
        import soundfile as sf
        spk_emb = np.load(spk_emb_path)
        chunks = [c["tts_speech"] for c in self.model.inference_sft(text, spk_emb, stream=True)]
        sf.write(output_path, np.concatenate(chunks, axis=-1).squeeze(), 22050)
        return output_path
```

## Step 3：面部动画驱动

从音频生成blendshape权重序列，驱动3D模型口型和表情。

```python
# server/modules/face_anim.py
import numpy as np, json, librosa

class FaceAnimator:
    def generate_blendshape_sequence(self, audio_path, fps=30):
        audio, sr = librosa.load(audio_path, sr=16000)
        samples_per_frame = int(sr / fps)
        n_frames = len(audio) // samples_per_frame
        weights = np.zeros((n_frames, 52))

        for i in range(n_frames):
            frame = audio[i*samples_per_frame:(i+1)*samples_per_frame]
            rms = np.sqrt(np.mean(frame ** 2))
            weights[i, 25] = np.clip(rms * 8.0, 0, 1)   # jawOpen
            weights[i, 0] = np.clip(rms * 3.0, 0, 0.6)   # eyeBlinkLeft
            weights[i, 1] = np.clip(rms * 3.0, 0, 0.6)   # eyeBlinkRight

        # 平滑
        kernel = np.ones(5) / 5
        for j in range(52): weights[:, j] = np.convolve(weights[:, j], kernel, mode="same")
        return weights

    def export_for_threejs(self, weights, output_path):
        frames = [{"time": i/30.0, **{f"bs_{j}": float(w) for j, w in enumerate(row)}}
                  for i, row in enumerate(weights)]
        with open(output_path, "w") as f: json.dump({"fps": 30, "frames": frames}, f)
        return output_path
```

```python
# server/modules/dialog_system.py
import openai, whisper
from pathlib import Path

class DialogSystem:
    def __init__(self, voice_cloner, tts_engine, system_prompt=None):
        self.voice_cloner = voice_cloner
        self.tts_engine = tts_engine
        self.system_prompt = system_prompt or "你是一个友善的数字人助手。用简洁口语回答，每次2-3句。"
        self.history = []

    def process_input(self, audio_path, spk_emb_path, output_dir="data/output"):
        output_dir = Path(output_dir); output_dir.mkdir(parents=True, exist_ok=True)
        # ASR
        model = whisper.load_model("base")
        user_text = model.transcribe(audio_path, language="zh")["text"]
        # LLM
        self.history.append({"role": "user", "content": user_text})
        messages = [{"role": "system", "content": self.system_prompt}] + self.history[-10:]
        reply = openai.chat.completions.create(model="gpt-4o-mini", messages=messages, max_tokens=200).choices[0].message.content
        self.history.append({"role": "assistant", "content": reply})
        # TTS
        reply_audio = self.tts_engine.synthesize(reply, spk_emb_path, str(output_dir / "reply.wav"))
        return {"user_text": user_text, "reply_text": reply, "reply_audio": reply_audio}
```

## 延迟优化

```
用户说完话 → ASR(~300ms) → LLM(~500ms) → TTS(~400ms) → 动画(~100ms) → 推送前端
                            总延迟约 1.3s（目标 < 2s）
```

三大策略：
- **流式TTS**：边合成边推送音频和blendshape，不等全部完成
- **句子级TTS**：LLM每生成一句就立即送TTS，不等全部回复
- **模型预热**：首次请求前用dummy数据跑一遍，避免冷启动

```python
class EmotionDetector:
    EMOTION_BLENDSHAPE = {
        "happy": {"mouthSmileLeft": 0.6, "mouthSmileRight": 0.6},
        "sad": {"mouthFrownLeft": 0.4, "mouthFrownRight": 0.4},
        "angry": {"browDownLeft": 0.7, "browDownRight": 0.7},
    }
    def __init__(self):
        from speechbrain.inference.interfaces import foreign_class
        self.classifier = foreign_class(
            source="speechbrain/emotion-recognition-wav2vec2-IEMOCAP",
            pymodule_file="custom_interface.py", classname="CustomEncoderWav2vec2Classifier")

    def detect(self, audio_path):
        out_prob, score, index, label = self.classifier.classify_file(audio_path)
        return {"emotion": label[0].lower(), "confidence": float(score)}

    def get_blendshape_adjustments(self, emotion):
        return self.EMOTION_BLENDSHAPE.get(emotion, {})
```
