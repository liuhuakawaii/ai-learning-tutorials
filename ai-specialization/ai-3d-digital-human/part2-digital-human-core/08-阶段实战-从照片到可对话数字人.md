# 第14课：阶段实战——从照片到可对话数字人

## 场景引入

前几节课我们分别掌握了 3D 人脸重建、语音克隆、面部动画迁移、全身动作生成等独立模块。但在真实项目中，客户不会说"我只要一个 3D 人脸模型"——他们要的是**一张照片丢进去，出来一个能说话、能动、能对话的数字人**。

这节课的目标就是把 Part 2 所有技术串联成一条完整的 Pipeline。重点不是单个模型的精度，而是**模块之间的衔接、延迟优化和工程健壮性**。

## 学习目标

完成本课学习后，你将能够：

1. 设计并实现从照片到可对话数字人的完整 Pipeline
2. 将 3D 人脸重建、语音克隆、对话系统、面部动画、全身动作五个模块串联
3. 使用 FastAPI 构建数字人后端服务
4. 使用 Three.js + WebSocket 实现前端实时渲染
5. 优化端到端延迟，使对话响应时间控制在 2 秒以内

### 1.2 数据流与格式

各模块之间的数据流转是系统集成的核心难点：

```
照片(.jpg) → DECA/NextFace → 3DMM参数(.npy) → Blender → 模型(.glb)
音频(.wav) → Speaker Encoder → 声纹向量(256d) → CosyVoice → TTS音频
TTS音频(.wav) → Wav2Lip → blendshape权重(.json) ──┐
TTS音频(.wav) → MotionDiffuse → SMPL参数(.npy) ───┤
                                                     ▼
                                              Three.js 渲染
```

### 1.3 技术选型

| 模块 | 技术方案 | 理由 |
|------|---------|------|
| 3D人脸重建 | NextFace | 轻量、支持单张照片、输出blendshape |
| 语音克隆 | CosyVoice | 零样本克隆、流式推理、中文效果好 |
| 对话系统 | Whisper + LLM + CosyVoice | 成熟稳定 |
| 面部动画 | Wav2Lip + blendshape映射 | 音频驱动、延迟低 |
| 全身动作 | MotionDiffuse | 文本/音频双驱动 |
| 后端 | FastAPI + WebSocket | 异步、流式 |
| 前端 | Three.js + GLTFLoader | 浏览器原生 |

### 2.2 Step 1：照片预处理与3D人脸重建

```python
# server/modules/face_recon.py
import numpy as np
import cv2
from pathlib import Path

class FaceReconstructor:
    """从单张照片生成3DMM参数和blendshape权重"""

    def __init__(self, model_path: str = "data/models/nextface"):
        import torch
        from nextface.facerecon import NextFaceRecon
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.model = NextFaceRecon(Path(model_path), device=self.device)

    def preprocess(self, image_path: str) -> np.ndarray:
        img = cv2.imread(image_path)
        if img is None:
            raise FileNotFoundError(f"无法读取图片: {image_path}")
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        face_cascade = cv2.CascadeClassifier(
            cv2.data.haarcascades + "haarcascade_frontalface_default.xml"
        )
        faces = face_cascade.detectMultiScale(gray, 1.1, 5)
        if len(faces) == 0:
            raise ValueError("未检测到人脸，请上传正面清晰照片")
        x, y, w, h = max(faces, key=lambda f: f[2] * f[3])
        margin = int(max(w, h) * 0.3)
        x1, y1 = max(0, x - margin), max(0, y - margin)
        x2 = min(img.shape[1], x + w + margin)
        y2 = min(img.shape[0], y + h + margin)
        return cv2.resize(img[y1:y2, x1:x2], (256, 256))

    def reconstruct(self, image_path: str, output_dir: str = "data/models/avatar"):
        output_dir = Path(output_dir)
        output_dir.mkdir(parents=True, exist_ok=True)
        img = self.preprocess(image_path)
        result = self.model.reconstruct(img)
        np.save(output_dir / "3dmm_params.npy", result["3dmm"])
        np.save(output_dir / "blendshape_weights.npy", result["blendshape"])
        result["mesh"].export(str(output_dir / "face_mesh.glb"))
        return {
            "mesh_path": str(output_dir / "face_mesh.glb"),
            "blendshape_dim": result["blendshape"].shape[-1],
        }
```

```python
# server/modules/voice_clone.py
import numpy as np
import torchaudio
from pathlib import Path

class VoiceCloner:
    """从参考音频提取声纹并支持TTS推理"""

    def __init__(self, cosyvoice_path: str = "pretrained_models/CosyVoice-300M"):
        import sys
        sys.append(str(Path(cosyvoice_path).parent))
        from cosyvoice.cosyvoice import CosyVoice
        self.model = CosyVoice(cosyvoice_path)

    def extract_speaker_embedding(self, audio_path: str) -> np.ndarray:
        waveform, sr = torchaudio.load(audio_path)
        if sr != 16000:
            waveform = torchaudio.transforms.Resample(sr, 16000)(waveform)
        if waveform.shape[0] > 1:
            waveform = waveform.mean(dim=0, keepdim=True)
        max_len = 16000 * 15
        if waveform.shape[1] > max_len:
            waveform = waveform[:, :max_len]
        return self.model.extract_spk_embedding(waveform).numpy()

    def clone_and_save(self, audio_path: str, output_dir: str = "data/models/voice"):
        output_dir = Path(output_dir)
        output_dir.mkdir(parents=True, exist_ok=True)
        emb = self.extract_speaker_embedding(audio_path)
        np.save(output_dir / "speaker_embedding.npy", emb)
        return str(output_dir / "speaker_embedding.npy")

    def synthesize(self, text: str, spk_emb_path: str, output_path: str):
        import soundfile as sf
        spk_emb = np.load(spk_emb_path)
        chunks = []
        for chunk in self.model.inference_sft(text, spk_emb, stream=True):
            chunks.append(chunk["tts_speech"])
        sf.write(output_path, np.concatenate(chunks, axis=-1).squeeze(), 22050)
        return output_path
```

```python
# server/modules/dialog_system.py
import openai
from pathlib import Path

class DialogSystem:
    """ASR + LLM + TTS 三段式对话 pipeline"""

    def __init__(self, voice_cloner, tts_engine, system_prompt: str = None):
        self.voice_cloner = voice_cloner
        self.tts_engine = tts_engine
        self.system_prompt = system_prompt or (
            "你是一个友善的数字人助手。用简洁自然的口语风格回答，每次2-3句话。"
        )
        self.history = []

    def asr_transcribe(self, audio_path: str) -> str:
        import whisper
        model = whisper.load_model("base")
        return model.transcribe(audio_path, language="zh")["text"]

    def llm_generate(self, user_text: str) -> str:
        self.history.append({"role": "user", "content": user_text})
        messages = [{"role": "system", "content": self.system_prompt}]
        messages.extend(self.history[-10:])
        resp = openai.chat.completions.create(
            model="gpt-4o-mini", messages=messages, max_tokens=200, temperature=0.7
        )
        reply = resp.choices[0].message.content
        self.history.append({"role": "assistant", "content": reply})
        return reply

    def process_input(self, audio_path: str, spk_emb_path: str, output_dir: str = "data/output"):
        output_dir = Path(output_dir)
        output_dir.mkdir(parents=True, exist_ok=True)
        user_text = self.asr_transcribe(audio_path)
        reply_text = self.llm_generate(user_text)
        reply_audio = self.tts_engine.synthesize(
            reply_text, spk_emb_path, str(output_dir / "reply.wav")
        )
        return {"user_text": user_text, "reply_text": reply_text, "reply_audio": reply_audio}
```

### 2.5 Step 4：面部动画驱动

```python
# server/modules/face_anim.py
import numpy as np
import json
import librosa

class FaceAnimator:
    """从音频生成blendshape权重序列，驱动3D模型嘴型和表情"""

    def generate_blendshape_sequence(self, audio_path: str, fps: int = 30) -> np.ndarray:
        audio, sr = librosa.load(audio_path, sr=16000)
        samples_per_frame = int(sr / fps)
        n_frames = len(audio) // samples_per_frame
        weights = np.zeros((n_frames, 52))

        for i in range(n_frames):
            frame_audio = audio[i * samples_per_frame:(i + 1) * samples_per_frame]
            rms = np.sqrt(np.mean(frame_audio ** 2))
            weights[i, 25] = np.clip(rms * 8.0, 0.0, 1.0)  # jawOpen
            weights[i, 0] = np.clip(rms * 3.0, 0.0, 0.6)   # eyeBlinkLeft
            weights[i, 1] = np.clip(rms * 3.0, 0.0, 0.6)   # eyeBlinkRight
            freq = np.abs(np.fft.rfft(frame_audio))
            centroid = np.sum(freq * np.arange(len(freq))) / (np.sum(freq) + 1e-8)
            smile = np.clip(centroid / 2000.0, 0.0, 0.5)
            weights[i, 22] = smile  # mouthSmileLeft
            weights[i, 23] = smile  # mouthSmileRight

        kernel = np.ones(5) / 5
        for j in range(52):
            weights[:, j] = np.convolve(weights[:, j], kernel, mode="same")
        return weights

    def export_for_threejs(self, weights: np.ndarray, output_path: str) -> str:
        frames = []
        for i, w in enumerate(weights):
            frame = {f"bs_{j}": float(w[j]) for j in range(w.shape[0])}
            frame["time"] = i / 30.0
            frames.append(frame)
        with open(output_path, "w") as f:
            json.dump({"fps": 30, "frames": frames}, f, indent=2)
        return output_path
```

```python
async def digital_human_ws(websocket: WebSocket):
    await websocket.accept()
    spk_emb = "data/models/voice/speaker_embedding.npy"
    try:
        while True:
            audio_bytes = await websocket.receive_bytes()
            tmp = tempfile.mktemp(suffix=".webm")

            result = await asyncio.to_thread(pipeline.process_conversation, tmp, spk_emb)

            audio_b64 = base64.b64encode(Path(result["reply_audio"]).read_bytes()).decode()
            await websocket.send_json({"type": "audio", "audio": audio_b64})
    except WebSocketDisconnect:

# server/pipeline.py
import time, json
from pathlib import Path

class DigitalHumanPipeline:
    def __init__(self, face_recon, voice_cloner, face_anim, dialog_system):
        self.face_recon = face_recon
        self.voice_cloner = voice_cloner
        self.face_anim = face_anim
        self.dialog_system = dialog_system

    def initialize(self, photo_path: str, audio_path: str) -> dict:
        t0 = time.time()
        face_result = self.face_recon.reconstruct(photo_path)
        voice_result = self.voice_cloner.clone_and_save(audio_path)
        return {
            "avatar_mesh": face_result["mesh_path"],
            "voice_embedding": voice_result,
            "init_time": round(time.time() - t0, 2),
        }

    def process_conversation(self, audio_path: str, spk_emb_path: str) -> dict:
        t0 = time.time()
        dialog = self.dialog_system.process_input(audio_path, spk_emb_path)
        bs_seq = self.face_anim.generate_blendshape_sequence(dialog["reply_audio"])
        bs_json = self.face_anim.export_for_threejs(bs_seq, "data/output/blendshape.json")
        with open(bs_json) as f:
            bs_data = json.load(f)
        print(f"[Pipeline] 总耗时: {time.time() - t0:.2f}s")
        return {**dialog, "blendshape_sequence": bs_data, "total_time": round(time.time() - t0, 2)}
```

## 3. 端到端延迟优化

### 3.1 延迟瓶颈分析

```
用户说完话 → ASR(~300ms) → LLM(~500ms) → TTS(~400ms) → 动画(~100ms) → 推送前端
                            总延迟约 1.3s（目标 < 2s）
```

### 3.2 三大优化策略

**策略一：流式 TTS + 流式推送**——不等 TTS 全部合成完，边合成边推送 blendshape 和音频片段给前端。

**策略二：LLM 流式输出 + 句子级 TTS**——LLM 每生成一句话（遇到句号、问号）就立即送 TTS，不等全部回复生成完。

**策略三：模型预热**——首次请求前用 dummy 数据跑一遍所有模型，避免冷启动延迟。

```python
# 流式推送示例
async def stream_pipeline(websocket, user_text, spk_emb_path):
    sentence_buffer = ""
    for chunk in llm_stream(user_text):
        sentence_buffer += chunk
        if any(p in sentence_buffer for p in "。！？"):
            for audio_chunk in tts_stream(sentence_buffer.strip(), spk_emb_path):
                bs = face_anim.generate_blendshape_chunk(audio_chunk)
                await websocket.send_json({"type": "blendshape", "weights": bs})
            sentence_buffer = ""
```

关键参数调优建议：
- `blendshape_fps`：30fps 足够流畅，60fps 会增加 2 倍计算量但体验提升有限
- `blendshape_smooth_window`：窗口越大动画越平滑但延迟越高，5 是合理折中
- `tts_streaming`：必须开启，否则用户要等整段 TTS 完成才能听到声音

---

## 5. 常见误区

**1. 模块独立优化，忽略衔接成本**

每个模块单独跑效果很好，但拼在一起就出问题——格式不兼容、采样率不匹配、坐标系不一致。工程实践中，模块间的接口定义比模块本身的精度更重要。先定义好数据格式和接口规范，再开发各模块。

**2. 追求单模块最优，忽略端到端延迟**

3D 重建用最高精度模型、LLM 用最大的、TTS 用最慢但最自然的——结果用户说完话要等 5 秒才能看到数字人回应。延迟比音质更影响用户体验。宁可用稍低精度但快 3 倍的模型。

**3. 不做错误恢复**

网络断了、模型推理超时、音频格式错误——任何一环出错，整个对话就卡死了。每个模块都需要超时机制和降级策略。比如 TTS 超时时，可以用预录的"抱歉，我暂时无法回答"来兜底。

**4. 忽略音频格式转换**

前端录音是 WebM 格式，Whisper 需要 WAV，TTS 输出也是 WAV 但采样率不同。不做格式转换直接对接，要么报错要么输出噪音。建议在 pipeline 入口统一做一次音频归一化。

**5. blendshape 推送频率不匹配前端帧率**

后端 30fps 生成 blendshape，但 WebSocket 推送受网络延迟影响可能不稳定。前端应该用时间戳对齐而不是逐帧接收，避免动画抖动。

---

## 小结

本课将 Part 2 所有技术串联为一个完整的数字人系统：

1. **Pipeline 设计**：照片→3D人脸→语音克隆→对话系统→面部动画→Web渲染
2. **模块衔接**：数据格式转换和接口规范是系统集成的关键
3. **延迟优化**：流式 TTS、句子级推送、模型预热，端到端延迟控制在 2 秒内
4. **工程健壮性**：每个模块都需要超时、降级、错误恢复机制
5. **前端渲染**：Three.js + WebSocket 实现实时 3D 动画驱动

## 参考答案

### 练习一

**思路**：语音情绪识别的核心是从音频中提取情绪特征，然后将情绪标签映射到 blendshape 权重调整和 LLM 的 system prompt。

**答案**：

```python
# server/modules/emotion_detector.py
import numpy as np

class EmotionDetector:
    EMOTION_LABELS = ["neutral", "happy", "sad", "angry", "fearful", "disgust", "surprised"]

    EMOTION_BLENDSHAPE = {
        "happy": {"mouthSmileLeft": 0.6, "mouthSmileRight": 0.6, "browInnerUp": 0.2},
        "sad": {"mouthFrownLeft": 0.4, "mouthFrownRight": 0.4, "browInnerUp": 0.5},
        "angry": {"browDownLeft": 0.7, "browDownRight": 0.7, "jawForward": 0.3},
        "fearful": {"eyeWideLeft": 0.5, "eyeWideRight": 0.5, "browInnerUp": 0.6},
        "neutral": {},
    }

    EMOTION_PROMPTS = {
        "happy": "用户很开心，请用积极愉快的语气回应。",
        "sad": "用户有些难过，请用温和关怀的语气回应。",
        "angry": "用户有些生气，请保持冷静理解的态度。",
        "fearful": "用户有些紧张，请用安抚鼓励的语气回应。",
        "neutral": "",
    }

    def __init__(self):
        from speechbrain.inference.interfaces import foreign_class
        self.classifier = foreign_class(
            source="speechbrain/emotion-recognition-wav2vec2-IEMOCAP",
            pymodule_file="custom_interface.py",
            classname="CustomEncoderWav2vec2Classifier",
        )

    def detect(self, audio_path: str) -> dict:
        out_prob, score, index, label = self.classifier.classify_file(audio_path)
        emotion = label[0].lower() if isinstance(label, list) else label.lower()
        return {"emotion": emotion, "confidence": float(score)}

    def get_blendshape_adjustments(self, emotion: str) -> dict:
        return self.EMOTION_BLENDSHAPE.get(emotion, {})

    def get_prompt_hint(self, emotion: str) -> str:
        return self.EMOTION_PROMPTS.get(emotion, "")
```

**要点**：
- 情绪到 blendshape 的映射需要根据实际效果微调权重
- 可以对连续多帧情绪做滑动窗口平滑，避免表情频繁切换
- prompt hint 要控制长度，避免 LLM 回复过长

### 练习二

**思路**：将每个角色的资产（3D 模型、声纹、系统提示词）打包为 Profile，通过 ID 切换。

**答案**：

```python
# server/modules/avatar_manager.py
import json
from dataclasses import dataclass, field
from pathlib import Path

@dataclass
class AvatarProfile:
    avatar_id: str
    name: str
    mesh_path: str
    voice_embedding_path: str
    system_prompt: str
    thumbnail: str = ""

class AvatarManager:
    def __init__(self, profiles_dir: str = "data/profiles"):
        self.profiles_dir = Path(profiles_dir)
        self.profiles: dict[str, AvatarProfile] = {}
        self.active: AvatarProfile | None = None
        self._load_profiles()

    def _load_profiles(self):
        for f in self.profiles_dir.glob("*/profile.json"):
            data = json.loads(f.read_text(encoding="utf-8"))
            p = AvatarProfile(**data)
            self.profiles[p.avatar_id] = p
        if self.profiles:
            self.active = next(iter(self.profiles.values()))

    def list_profiles(self) -> list[dict]:
        return [{"id": p.avatar_id, "name": p.name, "thumbnail": p.thumbnail} for p in self.profiles.values()]

    def switch(self, avatar_id: str) -> AvatarProfile:
        if avatar_id not in self.profiles:
            raise ValueError(f"角色不存在: {avatar_id}")
        self.active = self.profiles[avatar_id]
        return self.active
```

**要点**：
- Profile 数据存放在 `data/profiles/<avatar_id>/profile.json`
- 切换角色时要重新加载模型和声纹，需要做缓存避免重复加载
- 前端切换时应有 loading 状态
