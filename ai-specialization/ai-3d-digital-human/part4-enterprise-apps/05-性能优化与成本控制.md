### 性能瓶颈分析：四类瓶颈

### 端到端延迟拆解

用户说完一句话到听到回复，中间经过 6 个环节：

```
串行模式（优化前）：
|←── ASR ──→|←──── LLM ────→|←── TTS ──→|  总计: 3100ms
  300ms         2000ms          800ms

流式模式（优化后）：
|←─ ASR ─→|
            |←──── LLM ────→|
                 |←── TTS ──→|             首字延迟: ~500ms
                                            总计: ~1200ms
```

核心策略是**流式处理**：不等 ASR 完整识别完再发给 LLM，而是边生成边传递，把串行等待变成流水线并行。

### 渲染优化技术栈

```
┌─────────────────────────────────────────────────────────┐
│  1. LOD：远距离用低面数模型，面数减少 60-80%              │
│  2. 纹理优化：KTX2/ASTC 压缩，显存减少 70%               │
│  3. DrawCall 合并：同材质合批，DrawCall 减少 80%          │
│  4. WebWorker：物理计算放到 Worker，主线程帧率稳定        │
│  5. 骨骼动画优化：骨骼裁剪 + GPU Skinning                │
└─────────────────────────────────────────────────────────┘
```

### 模型推理优化技术栈

```
┌──────────────────────────────────────────────────────────────────┐
│  量化：FP32→FP16 速度1.5x | FP16→INT8 速度2-3x | INT4 速度3-4x  │
│  ONNX Runtime：跨框架推理，自动图优化，适合 TTS/ASR 中小模型      │
│  TensorRT：NVIDIA GPU 专用，Kernel 自动调优，速度 3-8x            │
│  vLLM：PagedAttention 显存利用率 2-4x，Continuous Batching 2-5x  │
└──────────────────────────────────────────────────────────────────┘
```

### 成本控制：GPU 选型

```
┌────────────┬──────────┬──────────┬──────────┬──────────────┐
│ GPU 型号    │ 显存     │ 单价/时   │ 适合场景  │ 并发能力     │
├────────────┼──────────┼──────────┼──────────┼──────────────┤
│ T4         │ 16GB     │ ¥0.8     │ ASR/TTS  │ 10-20 路     │
│ A10        │ 24GB     │ ¥2.5     │ 中型LLM  │ 5-10 路      │
│ A100       │ 80GB     │ ¥8.0     │ 大型LLM  │ 20-50 路     │
│ 4090       │ 24GB     │ ¥1.2     │ 渲染+推理 │ 3-8 路       │
└────────────┴──────────┴──────────┴──────────┴──────────────┘

成本优化策略：
1. 分离部署：ASR/TTS 用 T4，LLM 用 A100，渲染用 4090
2. 弹性伸缩：低峰缩减实例，高峰自动扩容
3. 量化降本：INT4 量化后同一张 GPU 能跑 2-3 倍并发
```

### 示例二：vLLM 部署与流式推理实战

```python
"""
vLLM 部署与流式推理实战
兼容 OpenAI API 的数字人对话服务，支持流式输出和并发控制
"""

import asyncio
import json
import time
from dataclasses import dataclass
from typing import AsyncGenerator, Optional
from enum import Enum


class RequestPriority(Enum):
    HIGH = 0
    NORMAL = 1
    LOW = 2


@dataclass
class ChatMessage:
    role: str
    content: str


@dataclass
class StreamingChunk:
    request_id: str
    content: str
    is_final: bool
    tokens_generated: int
    latency_ms: float


class VLLMServiceManager:
    """vLLM 服务管理器：负载均衡 + 并发控制"""

    def __init__(self):
        self.instances: list[dict] = []
        self._latency_history: list[float] = []

    def add_instance(self, instance_id: str, model: str, port: int, gpu_id: int):
        self.instances.append({
            "id": instance_id, "model": model, "port": port,
            "gpu_id": gpu_id, "active_requests": 0, "healthy": True,
        })

    def get_optimal_instance(self) -> Optional[dict]:
        healthy = [i for i in self.instances if i["healthy"]]
        return min(healthy, key=lambda i: i["active_requests"]) if healthy else None

    async def stream_chat(
        self, messages: list[ChatMessage], request_id: str
    ) -> AsyncGenerator[StreamingChunk, None]:
        instance = self.get_optimal_instance()
        if not instance:
            yield StreamingChunk(request_id, "[错误] 无可用实例", True, 0, 0)
            return

        instance["active_requests"] += 1
        start_time = time.time()
        try:
            response_text = "牛顿第二定律：F=ma，加速度与合力成正比，与质量成反比。"
            tokens = 0
            for char in response_text:
                tokens += 1
                yield StreamingChunk(request_id, char, False, tokens,
                                     (time.time() - start_time) * 1000)
                await asyncio.sleep(0.03)
            total = (time.time() - start_time) * 1000
            self._latency_history.append(total)
            yield StreamingChunk(request_id, "", True, tokens, total)
        finally:
            instance["active_requests"] -= 1


async def demo():
    manager = VLLMServiceManager()
    manager.add_instance("vllm-1", "Qwen/Qwen2.5-7B-Instruct", 8000, 0)
    manager.add_instance("vllm-2", "Qwen/Qwen2.5-7B-Instruct", 8001, 1)

    print("用户: 请解释牛顿第二定律\n数字人: ", end="", flush=True)
    async for chunk in manager.stream_chat([ChatMessage("user", "请解释牛顿第二定律")], "req-001"):
        if not chunk.is_final:
            print(chunk.content, end="", flush=True)
        else:
            print(f"\n[tokens: {chunk.tokens_generated}, 延迟: {chunk.latency_ms:.0f}ms]")

asyncio.run(demo())
```

### 示例三：成本计算器实战

```python
"""
数字人系统成本计算器
根据不同配置计算月度成本，支持多方案对比
"""

from dataclasses import dataclass
import json


@dataclass
class GPUConfig:
    name: str
    vram_gb: int
    price_per_hour: float
    max_concurrent_users: int


@dataclass
class ModelServiceConfig:
    service_name: str
    model_name: str
    gpu_type: str
    gpu_count: int


class DigitalHumanCostCalculator:
    GPU_CATALOG = {
        "T4": GPUConfig("T4", 16, 0.8, 20),
        "A10": GPUConfig("A10", 24, 2.5, 10),
        "A100-80G": GPUConfig("A100-80G", 80, 8.0, 50),
        "4090": GPUConfig("4090", 24, 1.2, 8),
    }

    def __init__(self):
        self.services: list[ModelServiceConfig] = []
        self.target_users = 100
        self.peak_hours = 8
        self.days = 30

    def set_target(self, users: int, peak_hours: float = 8, days: int = 30):
        self.target_users = users
        self.peak_hours = peak_hours
        self.days = days

    def add_service(self, config: ModelServiceConfig):
        self.services.append(config)

    def calculate_gpu_needed(self, service: ModelServiceConfig) -> int:
        gpu = self.GPU_CATALOG.get(service.gpu_type)
        if not gpu:
            return 0
        return max(1, -(-self.target_users // gpu.max_concurrent_users)) * service.gpu_count

    def estimate_cost(self) -> list[dict]:
        total_hours = self.peak_hours * self.days + (24 - self.peak_hours) * self.days * 0.3
        estimates = []
        for svc in self.services:
            gpu = self.GPU_CATALOG.get(svc.gpu_type)
            if not gpu:
                continue
            count = self.calculate_gpu_needed(svc)
            hourly = gpu.price_per_hour * count
            monthly = hourly * total_hours
            estimates.append({
                "component": svc.service_name,
                "gpu": f"{svc.gpu_type} x{count}",
                "hourly": round(hourly, 2),
                "monthly": round(monthly, 2),
            })
        return estimates



# 使用示例
calc = DigitalHumanCostCalculator()
calc.set_target(users=100, peak_hours=8, days=30)
calc.add_service(ModelServiceConfig("ASR", "whisper-large-v3", "T4", 1))
calc.add_service(ModelServiceConfig("LLM", "Qwen2.5-14B", "A10", 1))
calc.add_service(ModelServiceConfig("TTS", "CosyVoice", "T4", 1))

for e in calc.estimate_cost():
    print(f"  {e['component']}: {e['gpu']} → ¥{e['monthly']:,.0f}/月")
print(f"  合计: ¥{sum(e['monthly'] for e in calc.estimate_cost()):,.0f}/月")
```

## 常见误区

### 误区一：只优化 LLM，忽略其他环节

TTS 延迟可能占总延迟的 25%，渲染不当会挤占模型推理的 GPU 算力。正确做法是先做延迟拆解，找到占比最大的瓶颈环节集中优化。

### 误区二：过早量化导致质量下降

INT4 量化在数学推理等精确任务上精度损失明显。应分阶段进行：先 FP16 验证正确性，再 INT8 测试精度影响，最后尝试 INT4。

### 误区三：忽略冷启动延迟

模型首次加载可能需要 30 秒到几分钟。生产环境必须预热，弹性伸缩要考虑冷启动时间，提前扩容而非响应式扩容。

### 误区四：用峰值配置覆盖全天

按峰值配置 GPU 导致低峰期利用率不到 10%。应区分高低峰，结合弹性伸缩可降低 40-60% 成本。

## 小结与练习

### 小结

1. **四类性能瓶颈**：GPU 渲染、模型推理、网络传输、并发处理，需要先量化再优化
2. **延迟拆解法**：将端到端延迟拆解为 ASR→LLM→TTS→驱动→渲染，流式处理是关键
3. **推理优化**：量化（FP16→INT8→INT4）、TensorRT、vLLM 三大手段
4. **成本控制**：GPU 选型、分离部署、弹性伸缩、量化降本

### 练习一

**思路**：用装饰器包装函数调用记录耗时，汇总后计算 P50/P95/P99 分位数，标记占比超 40% 的环节为瓶颈。

```python
import time
import functools
from collections import defaultdict


class LatencyProfiler:
    def __init__(self):
        self.records: dict[str, list[float]] = defaultdict(list)

    def measure(self, stage_name: str):
        def decorator(func):
            @functools.wraps(func)
            def wrapper(*args, **kwargs):
                start = time.perf_counter()
                result = func(*args, **kwargs)
                elapsed = (time.perf_counter() - start) * 1000
                self.records[stage_name].append(elapsed)
                return result
            return wrapper
        return decorator

    def _percentile(self, data: list[float], p: float) -> float:
        s = sorted(data)
        return s[min(int(len(s) * p / 100), len(s) - 1)]

    def report(self) -> str:
        total = sum(sum(v) / len(v) for v in self.records.values())
        lines = ["=" * 60, "延迟分析报告", "=" * 60]
        for stage, lats in self.records.items():
            avg = sum(lats) / len(lats)
            pct = avg / total * 100 if total else 0
            flag = " ★" if pct > 40 else ""
            lines.append(f"  {stage:<12} 平均:{avg:>6.1f}ms  P95:{self._percentile(lats, 95):>6.1f}ms  "
                         f"占比:{pct:>5.1f}%{flag}")
        lines.append(f"  总平均延迟: {total:.1f}ms")
        lines.append("  ★ = 瓶颈环节")
        return "\n".join(lines)


profiler = LatencyProfiler()

@profiler.measure("ASR")
def run_asr(): time.sleep(0.15); return "text"

@profiler.measure("LLM")
def run_llm(): time.sleep(0.5); return "reply"

@profiler.measure("TTS")
def run_tts(): time.sleep(0.3); return b"audio"

for _ in range(5):
    run_asr(); run_llm(); run_tts()

print(profiler.report())
```

### 练习二

**思路**：基于加权评分决定扩缩容，设置冷却期防震荡，缩容比扩容更保守。

```python
import time
from dataclasses import dataclass


@dataclass
class Metrics:
    gpu_utilization: float
    queue_length: int
    avg_latency_ms: float


class AutoScaler:
    def __init__(self, min_inst=1, max_inst=10, cooldown=120):
        self.min_inst, self.max_inst, self.cooldown = min_inst, max_inst, cooldown
        self.current = min_inst
        self.last_scale_time = 0.0
        self.low_count = 0

    def evaluate(self, m: Metrics) -> tuple[bool, int]:
        if time.time() - self.last_scale_time < self.cooldown:
            return False, self.current
        s = m.gpu_utilization * 0.4 + min(m.queue_length / 10, 1) * 30 + min(m.avg_latency_ms / 2000, 1) * 30
        if s >= 70:
            target = min(self.current + 1, self.max_inst)
            if target > self.current:
                self.current = target
                self.last_scale_time = time.time()
                self.low_count = 0
                return True, target
        elif s < 30:
            self.low_count += 1
            if self.low_count >= 5:
                target = max(self.current - 1, self.min_inst)
                if target < self.current:
                    self.current = target
                    self.last_scale_time = time.time()
                    self.low_count = 0
                    return True, target
        else:
            self.low_count = 0
        return False, self.current

scaler = AutoScaler(min_inst=2, max_inst=8)
for i, (gpu, q, lat) in enumerate([(85, 8, 1500), (90, 12, 2000), (30, 1, 400)]):
    scaled, target = scaler.evaluate(Metrics(gpu, q, lat))
    print(f"时刻{i}: GPU {gpu}% → {'扩容' if scaled else '保持'} ({target} 实例)")
```

```python
    api_monthly = api_daily * 30

    # 全自部署
    self_monthly = 50 * 2.5 * 24 * 30 + 5 * 0.8 * 24 * 30 * 2  # A10 x50 + T4 x10

    # 混合方案
    base = 5 * 2.5 * 24 * 30 + 2 * 0.8 * 24 * 30 * 2  # 基础集群
    hybrid = base + api_monthly * 0.4 * (peak - low) / peak

    print(f"{'方案':<16} {'月成本':>12} {'单用户':>10}")
    print("-" * 40)
    print(f"{'全 API':<16} ¥{api_monthly:>10,.0f} ¥{api_monthly/dau:>8.2f}")
    print(f"{'全自部署':<16} ¥{self_monthly:>10,.0f} ¥{self_monthly/dau:>8.2f}")
    print(f"{'混合方案':<16} ¥{hybrid:>10,.0f} ¥{hybrid/dau:>8.2f}")
    print("\n推荐: 混合方案——基础保底、API 补峰、成本最优")

compare_solutions()
```

**要点**：
- 全 API 线性增长不可持续，全自部署初期投入高
- 混合方案月成本比全自部署低约 60%，比全 API 低约 30%
