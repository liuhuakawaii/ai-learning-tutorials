# 虚拟主播系统

2025年双十一，某美妆品牌直播间里一位"主播"连续直播72小时，切换中英日三语讲解产品。这不是真人，而是一套虚拟主播系统。

核心价值：7×24不间断、多语言覆盖、品牌一致性可控。据艾瑞咨询2025年报告，中国虚拟主播市场规模突破200亿元，电商直播渗透率超35%。

但构建生产级系统远非"接个TTS再渲染个3D模型"。本课从架构到实战，搭建完整系统。

## 系统架构

```
输入层(弹幕/脚本) → 对话层(LLM/Prompt) → 语音层(TTS/情感) → 驱动层(面部/表情)
                                                                      ↓
平台层(抖音/淘宝) ← 推流层(RTMP/SRT) ← 编码层(H.264) ← 渲染层(Unity/WebGL)
```

端到端延迟：弹幕(100ms)→LLM(800ms)→TTS(500ms)→驱动(100ms)→渲染(100ms)→编码(50ms)→推流(200ms)→平台(500ms)≈2.35秒。

## 推流协议

| 特性 | RTMP | SRT | WebRTC |
|------|------|-----|--------|
| 延迟 | 1-3秒 | 0.5-2秒 | <1秒 |
| 兼容性 | 所有平台 | 需平台支持 | 浏览器原生 |
| 抗丢包 | 弱 | 强(FEC+ARQ) | 中等 |
| 场景 | 推流到平台 | 跨境/弱网 | Web端互动 |

RTMP仍是国内直播平台首选（抖音、淘宝原生支持）。

## LLM直播脚本生成

```python
"""virtual_anchor_script.py - 依赖: pip install openai pydantic"""
import json
from pydantic import BaseModel, Field
from typing import Optional
from openai import OpenAI

class ProductInfo(BaseModel):
    name: str; category: str; price: float; original_price: float
    key_features: list[str]; target_audience: str; promotion: Optional[str] = None

class ScriptSegment(BaseModel):
    phase: str  # opening/product/features/cta/closing
    content: str; duration_seconds: int; emotion: str; action_hint: str

class ScriptGenerator:
    def __init__(self, api_key: str, base_url="https://api.openai.com/v1"):
        self.client = OpenAI(api_key=api_key, base_url=base_url)
        self.system_prompt = """你是资深电商直播编导。
要求：话术口语化，每句15-25字，遵循FABE法则，符合《广告法》。输出JSON。"""

    def generate_script(self, product: ProductInfo):
        discount = round((1 - product.price / product.original_price) * 100, 1)
        resp = self.client.chat.completions.create(model="gpt-4o",
            messages=[
                {"role": "system", "content": self.system_prompt},
                {"role": "user", "content": f"""商品：{product.name}，{product.price}元（原价{product.original_price}，{discount}折）
卖点：{'、'.join(product.key_features)}，人群：{product.target_audience}
生成5阶段脚本：opening(15-20s)、product(30-45s)、features(45-60s)、cta(20-30s)、closing(10-15s)。"""}],
            response_format={"type": "json_object"}, temperature=0.7)
        result = json.loads(resp.choices[0].message.content)
        return [ScriptSegment(**s) for s in result["segments"]]

    def generate_bullet_reply(self, comment: str, product_context: str, persona: str):
        resp = self.client.chat.completions.create(model="gpt-4o-mini",
            messages=[{"role": "user", "content":
                f"你是{persona}，正在直播。当前商品：{product_context}。观众说：「{comment}」。回复20字内，自然亲切。"}],
            temperature=0.8, max_tokens=100)
        return resp.choices[0].message.content.strip()
```

## TTS+面部驱动+RTMP推流

```python
"""virtual_anchor_pipeline.py - 依赖: pip install numpy opencv-python requests, 系统: FFmpeg"""
import asyncio, subprocess, time, numpy as np, requests
from dataclasses import dataclass, field
from enum import Enum

class EmotionType(Enum):
    NEUTRAL = "neutral"; HAPPY = "happy"; EXCITED = "excited"

@dataclass
class VisemeFrame:
    timestamp_ms: int; viseme_id: str; weight: float
    blendshapes: dict = field(default_factory=dict)

class RTMPPublisher:
    def __init__(self, rtmp_url, width=1920, height=1080, fps=30):
        self.rtmp_url = rtmp_url; self.width, self.height, self.fps = width, height, fps
        self.process = None; self.is_streaming = False

    def start(self):
        cmd = ["ffmpeg", "-y",
            "-f", "rawvideo", "-vcodec", "rawvideo", "-pix_fmt", "bgr24",
            "-s", f"{self.width}x{self.height}", "-r", str(self.fps), "-i", "-",
            "-f", "s16le", "-ar", "24000", "-ac", "1", "-i", "-",
            "-c:v", "libx264", "-preset", "ultrafast", "-tune", "zerolatency",
            "-b:v", "3000k", "-pix_fmt", "yuv420p",
            "-c:a", "aac", "-b:a", "128k", "-f", "flv", self.rtmp_url]
        self.process = subprocess.Popen(cmd, stdin=subprocess.PIPE, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        self.is_streaming = True

    def write_video_frame(self, frame_bgr):
        if self.process and self.process.stdin and self.is_streaming:
            try: self.process.stdin.write(frame_bgr.tobytes())
            except BrokenPipeError: self.is_streaming = False

    def write_audio_chunk(self, pcm_data):
        if self.process and self.process.stdin and self.is_streaming:
            try: self.process.stdin.write(pcm_data)
            except BrokenPipeError: self.is_streaming = False

    def stop(self):
        self.is_streaming = False
        if self.process: self.process.stdin.close(); self.process.terminate()


class VirtualAnchorPipeline:
    def __init__(self, tts_server, rtmp_url, speaker="default"):
        self.publisher = RTMPPublisher(rtmp_url)
        self.speaker = speaker; self.script_queue = asyncio.Queue()
        self.tts_server = tts_server; self.is_running = False

    async def start(self):
        self.publisher.start(); self.is_running = True
        await asyncio.gather(self._script_consumer(), self._render_loop())

    async def enqueue_text(self, text, emotion=EmotionType.NEUTRAL):
        await self.script_queue.put((text, emotion))

    async def _script_consumer(self):
        while self.is_running:
            try: text, emotion = await asyncio.wait_for(self.script_queue.get(), timeout=1.0)
            except asyncio.TimeoutError: continue
            audio = await asyncio.to_thread(self._tts_synthesize, text)
            if audio: self.publisher.write_audio_chunk(audio)

    async def _render_loop(self):
        import cv2
        interval = 1.0 / self.publisher.fps
        frame_num = 0
        while self.is_running:
            frame = np.zeros((self.publisher.height, self.publisher.width, 3), dtype=np.uint8)
            cv2.putText(frame, f"Frame:{frame_num}", (30, self.publisher.height-50),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
            self.publisher.write_video_frame(frame)
            frame_num += 1; await asyncio.sleep(interval)

    def _tts_synthesize(self, text):
        import base64
        resp = requests.post(f"{self.tts_server}/synthesize",
            json={"text": text, "speaker": self.speaker}, timeout=10)
        return base64.b64decode(resp.json()["audio_base64"]) if resp.ok else None

    def stop(self): self.is_running = False; self.publisher.stop()
```

## 弹幕互动

```python
"""bullet_interaction.py - 依赖: pip install openai"""
import re, time
from collections import deque
from dataclasses import dataclass
from openai import OpenAI

@dataclass
class BulletComment:
    user_id: str; username: str; content: str; timestamp: float; is_gift: bool = False

class BulletInteractionEngine:
    KEYWORD_RULES = [
        (r"多少钱|价格|怎么卖", "现在直播间专属价只要{price}哦！"),
        (r"质量怎么样|好不好", "质量完全放心！支持7天无理由退换~"),
        (r"发货|快递", "下单后48小时内发货，一般3天就到！"),
    ]

    def __init__(self, client: OpenAI, persona: str):
        self.client = client; self.persona = persona; self.history = deque(maxlen=100)

    def generate_reply(self, comment: BulletComment):
        if comment.is_gift: return f"谢谢{comment.username}的礼物！"
        for pattern, template in self.KEYWORD_RULES:
            if re.search(pattern, comment.content): return template.format(price="299元")
        if len(comment.content) < 5: return None
        try:
            resp = self.client.chat.completions.create(model="gpt-4o-mini",
                messages=[{"role": "user", "content":
                    f"你是{self.persona}，正在直播。观众{comment.username}说：「{comment.content}」。回复20字内。"}],
                temperature=0.8, max_tokens=80)
            return resp.choices[0].message.content.strip()
        except: return None

    def should_reply(self, comment):
        if comment.is_gift: return True
        return len(comment.content) >= 3
```

```python
from abc import ABC, abstractmethod

class BasePublisher(ABC):
    def __init__(self, width=1920, height=1080, fps=30):
        self.width, self.height, self.fps = width, height, fps
        self.process = None; self.is_streaming = False
    @abstractmethod
    def _build_command(self): ...
    def start(self):
        self.process = subprocess.Popen(self._build_command(), stdin=subprocess.PIPE)
        self.is_streaming = True
    def write_video_frame(self, data):
        if self.process and self.process.stdin and self.is_streaming: self.process.stdin.write(data)
    def stop(self):
        self.is_streaming = False
        if self.process: self.process.stdin.close(); self.process.terminate()

class SRTPublisher(BasePublisher):
    def __init__(self, srt_url, latency_ms=200, **kwargs):
        super().__init__(**kwargs); self.srt_url = srt_url; self.latency_ms = latency_ms
    def _build_command(self):
        return ["ffmpeg", "-y", "-f", "rawvideo", "-vcodec", "rawvideo", "-pix_fmt", "bgr24",
            "-s", f"{self.width}x{self.height}", "-r", str(self.fps), "-i", "-",
            "-c:v", "libx264", "-preset", "ultrafast", "-tune", "zerolatency",
            "-b:v", "3000k", "-pix_fmt", "yuv420p", "-f", "mpegts",
            f"{self.srt_url}?mode=caller&latency={self.latency_ms*1000}"]
```
