### 二、3D商品生产Pipeline

```
┌─────────┐   ┌─────────┐   ┌─────────┐   ┌─────────┐   ┌─────────┐
│  拍摄   │──▶│ 重建    │──▶│ 材质    │──▶│ 优化    │──▶│ 分发    │
│·转盘    │   │·NeRF   │   │·PBR贴图 │   │·减面   │   │·glTF   │
│·60-100张│   │·3DGS   │   │·法线    │   │·LOD    │   │·DRACO  │
└─────────┘   └─────────┘   └─────────┘   └─────────┘   └─────────┘
```

- **拍摄**：转盘+光箱拍摄60-100张照片，光照均匀可控
- **重建**：MVS或3D Gaussian Splatting重建为3D网格，2025年3DGS重建质量已接近照片级
- **材质**：制作PBR材质（基础色、金属度、粗糙度、法线贴图）
- **优化**：减面至万级、LOD多细节层次、纹理压缩
- **分发**：glTF格式 + DRACO压缩 + CDN分发 + 懒加载

### 四、WebXR AR试穿与AI虚拟试衣

WebXR AR试穿流程：用户点击"AR试穿" → 请求摄像头+WebXR会话 → 摄像头画面作为背景 → 人脸/人体关键点检测(MediaPipe) → 3D模型对齐到关键点 → 实时更新位置 → 混合渲染输出。

AI虚拟试衣（2025年主流）：OOTDiffusion（扩散模型，支持多件搭配）、IDM-VTON（扩散+ControlNet，效果最好）。核心流程：用户照片 → 人体解析 → 姿态估计 → 衣服变形 → 图像融合。

```tsx
function LoadingBar() {
  const { progress } = useProgress()
  return (

      <div style={{ background: 'rgba(0,0,0,0.8)', color: 'white', padding: '20px 40px', borderRadius: '12px', textAlign: 'center' }}>

function ProductModel({ modelUrl, color, autoRotate }: {
  modelUrl: string; color: string; autoRotate: boolean
}) {
  const groupRef = useRef<THREE.Group>(null)
  const gltf = useLoader(GLTFLoader, modelUrl, (loader) => {
    const draco = new DRACOLoader()
    draco.setDecoderPath('https://www.gstatic.com/draco/versioned/decoders/1.5.6/')
    loader.setDRACOLoader(draco)
  })

  useFrame((_, delta) => {
    if (groupRef.current && autoRotate) groupRef.current.rotation.y += delta * 0.3
  })

  React.useEffect(() => {
    gltf.scene.traverse((child) => {
      if ((child as THREE.Mesh).isMesh) {
        const mat = (child as THREE.Mesh).material as THREE.MeshStandardMaterial
        mat.color.set(color); mat.needsUpdate = true
      }
    })
  }, [color, gltf.scene])

  return <group ref={groupRef}><primitive object={gltf.scene} scale={1} /></group>
}
```

### 实战二：WebXR AR试穿

```tsx
// ARTryOn.tsx
// 依赖: npm install three @mediapipe/face_mesh

import React, { useRef, useState, useCallback, useEffect } from 'react'
import * as THREE from 'three'

const CONFIGS: Record<string, { pos: number[]; rot: number[]; scale: number }> = {
  glasses: { pos: [0, 0.02, -0.07], rot: [0, 0, 0], scale: 1.4 },
  hat: { pos: [0, 0.15, -0.02], rot: [-0.1, 0, 0], scale: 1.6 },
}

class ARRenderer {
  private scene = new THREE.Scene()
  private camera = new THREE.PerspectiveCamera(60, 640 / 480, 0.01, 100)
  private renderer: THREE.WebGLRenderer
  private model: THREE.Group | null = null
  private videoTex: THREE.VideoTexture | null = null
  private smoothPos = new THREE.Vector3()

  constructor(canvas: HTMLCanvasElement) {
    this.camera.position.z = 1
    this.renderer = new THREE.WebGLRenderer({ canvas, alpha: true, antialias: true })
    this.renderer.setSize(canvas.width, canvas.height)
    this.scene.add(new THREE.AmbientLight(0xffffff, 0.6))
    const dir = new THREE.DirectionalLight(0xffffff, 0.8)
    dir.position.set(0, 1, 1); this.scene.add(dir)
  }

  setVideo(video: HTMLVideoElement) {
    this.videoTex = new THREE.VideoTexture(video)
    this.videoTex.minFilter = THREE.LinearFilter
    this.scene.background = this.videoTex
  }

  async loadModel(url: string) {
    const { GLTFLoader } = await import('three/examples/jsm/loaders/GLTFLoader')
    return new Promise<void>((resolve, reject) => {
      new GLTFLoader().load(url, gltf => {
        this.model = gltf.scene; this.model.visible = false
        this.scene.add(this.model); resolve()
      }, undefined, reject)
    })
  }

  updatePose(landmarks: any[], type: string) {
    if (!this.model || !landmarks.length) return
    const cfg = CONFIGS[type] || CONFIGS.glasses
    const nose = landmarks[1]
    const target = new THREE.Vector3(
      (nose.x - 0.5) * 2 + cfg.pos[0],
      -(nose.y - 0.5) * 2 + cfg.pos[1],
      -nose.z * 2 + cfg.pos[2]
    )
    this.smoothPos.lerp(target, 0.3)
    this.model.position.copy(this.smoothPos)
    this.model.scale.setScalar(cfg.scale)
    this.model.visible = true
  }

  render() {
    if (this.videoTex) this.videoTex.needsUpdate = true
    this.renderer.render(this.scene, this.camera)
  }

  dispose() { this.renderer.dispose(); this.scene.clear() }
}

export function ARTryOn({ type, modelUrl }: { type: string; modelUrl: string }) {
  const videoRef = useRef<HTMLVideoElement>(null)
  const canvasRef = useRef<HTMLCanvasElement>(null)
  const [active, setActive] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const arRef = useRef<ARRenderer | null>(null)

  const start = useCallback(async () => {
    if (!videoRef.current || !canvasRef.current) return
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'user', width: 1280, height: 720 } })
      videoRef.current.srcObject = stream; await videoRef.current.play()

      const ar = new ARRenderer(canvasRef.current)
      ar.setVideo(videoRef.current)
      await ar.loadModel(modelUrl)

      const { FaceMesh } = await import('@mediapipe/face_mesh')
      const fm = new FaceMesh({ locateFile: f => `https://cdn.jsdelivr.net/npm/@mediapipe/face_mesh/${f}` })
      fm.setOptions({ maxNumFaces: 1, refineLandmarks: true, minDetectionConfidence: 0.5 })
      fm.onResults((r: any) => { if (r.multiFaceLandmarks?.[0]) { ar.updatePose(r.multiFaceLandmarks[0], type); ar.render() } })

      const loop = async () => { if (videoRef.current?.readyState >= 2) await fm.send({ image: videoRef.current }); requestAnimationFrame(loop) }
      loop()
      arRef.current = ar; setActive(true)
    } catch { setError('无法启动AR，请检查摄像头权限') }
  }, [modelUrl, type])

  const stop = useCallback(() => {
    arRef.current?.dispose(); arRef.current = null
    if (videoRef.current?.srcObject) (videoRef.current.srcObject as MediaStream).getTracks().forEach(t => t.stop())
    setActive(false)
  }, [])

  useEffect(() => () => stop(), [stop])

  return (
    <div style={{ position: 'relative', width: '100%', maxWidth: '640px', margin: '0 auto' }}>
      <video ref={videoRef} style={{ display: 'none' }} playsInline muted />
      <canvas ref={canvasRef} width={640} height={480} style={{ width: '100%', borderRadius: '12px', background: '#000' }} />
      {!active && !error && (
        <button onClick={start} style={{ position: 'absolute', bottom: '20px', left: '50%', transform: 'translateX(-50%)', background: '#4f46e5', color: 'white', border: 'none', padding: '12px 32px', borderRadius: '8px', fontSize: '16px', cursor: 'pointer' }}>
          开启AR试穿
        </button>
      )}
      {active && <button onClick={stop} style={{ position: 'absolute', top: '10px', right: '10px', background: 'rgba(0,0,0,0.6)', color: 'white', border: 'none', padding: '8px 16px', borderRadius: '6px', cursor: 'pointer' }}>关闭</button>}
      {error && <div style={{ position: 'absolute', bottom: '20px', left: '50%', transform: 'translateX(-50%)', background: '#fef2f2', color: '#dc2626', padding: '12px 24px', borderRadius: '8px' }}>{error}</div>}
    </div>
  )
}
```

### 实战三：AI虚拟试衣间

```python
"""
virtual_fitting_room.py
AI虚拟试衣间
依赖: pip install torch torchvision diffusers transformers pillow opencv-python
"""
import time, hashlib
import numpy as np
import torch
from PIL import Image
from dataclasses import dataclass
from typing import Optional
from pathlib import Path


@dataclass
class TryOnRequest:
    person_image_path: str
    garment_image_path: str
    output_path: str
    num_inference_steps: int = 30
    guidance_scale: float = 7.5
    seed: Optional[int] = None


@dataclass
class TryOnResult:
    output_image_path: str
    processing_time_seconds: float
    model_used: str
    input_resolution: tuple[int, int]
    output_resolution: tuple[int, int]


class HumanParser:
    UPPER_LABELS = {4, 7}  # upper_clothes, dress

    def get_clothing_mask(self, image: Image.Image) -> np.ndarray:
        from torchvision import transforms
        transform = transforms.Compose([
            transforms.Resize((512, 512)), transforms.ToTensor(),
            transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
        ])
        # 实际应使用SCHP等预训练模型进行语义分割
        # 这里返回占位mask，生产环境替换为真实推理
        return np.ones((512, 512), dtype=np.uint8) * 255


class PoseEstimator:
    def estimate(self, image: Image.Image) -> list[dict]:
        try:
            import mediapipe as mp
            pose = mp.solutions.pose.Pose(static_image_mode=True, model_complexity=2)
            results = pose.process(np.array(image))
            if not results.pose_landmarks:
                return self._fallback()
            names = ["nose", "left_eye", "right_eye", "left_ear", "right_ear",
                     "left_shoulder", "right_shoulder", "left_elbow", "right_elbow",
                     "left_wrist", "right_wrist", "left_hip", "right_hip",
                     "left_knee", "right_knee", "left_ankle", "right_ankle"]
            return [{"name": names[i], "x": lm.x, "y": lm.y, "z": lm.z}
                    for i, lm in enumerate(results.pose_landmarks.landmark) if i < len(names)]
        except ImportError:
            return self._fallback()

    def _fallback(self) -> list[dict]:
        return [
            {"name": "left_shoulder", "x": 0.35, "y": 0.25, "z": 0},
            {"name": "right_shoulder", "x": 0.65, "y": 0.25, "z": 0},
            {"name": "left_hip", "x": 0.38, "y": 0.55, "z": 0},
            {"name": "right_hip", "x": 0.62, "y": 0.55, "z": 0},
        ]


class IDMVTONPipeline:
    def __init__(self, model_path: str = "yisol/IDM-VTON", device: str = "cuda"):
        self.device = device
        self.model_path = model_path
        self.pipe = None
        self.parser = HumanParser()
        self.pose_estimator = PoseEstimator()

    def load_model(self):
        try:
            from diffusers import StableDiffusionInstructPix2PixPipeline
            self.pipe = StableDiffusionInstructPix2PixPipeline.from_pretrained(
                self.model_path,
                torch_dtype=torch.float16 if self.device == "cuda" else torch.float32,
                safety_checker=None
            ).to(self.device)
            self.pipe.enable_model_cpu_offload()
        except Exception as e:
            print(f"模型加载失败: {e}，将使用简化合成方案")
            self.pipe = None

    def try_on(self, request: TryOnRequest) -> TryOnResult:
        start = time.time()
        person = Image.open(request.person_image_path).convert("RGB").resize((512, 512))
        garment = Image.open(request.garment_image_path).convert("RGB").resize((512, 512))
        original_size = Image.open(request.person_image_path).convert("RGB").size

        mask = self.parser.get_clothing_mask(person)

        if self.pipe is not None:
            gen = torch.Generator(self.device).manual_seed(request.seed) if request.seed else None
            result = self.pipe(
                prompt="a person wearing the provided garment, high quality",
                image=person, mask_image=Image.fromarray(mask), control_image=garment,
                num_inference_steps=request.num_inference_steps,
                guidance_scale=request.guidance_scale, generator=gen
            ).images[0]
            model_name = self.model_path
        else:
            mask_3ch = np.stack([mask]*3, axis=-1) / 255.0
            blended = np.array(person) * (1 - mask_3ch) + np.array(garment) * mask_3ch
            result = Image.fromarray(blended.astype(np.uint8))
            model_name = "fallback_composite"

        result.resize(original_size).save(request.output_path)
        return TryOnResult(
            output_image_path=request.output_path,
            processing_time_seconds=round(time.time() - start, 2),
            model_used=model_name,
            input_resolution=original_size, output_resolution=original_size
        )


class VirtualFittingRoom:
    def __init__(self, model_path: str = "yisol/IDM-VTON", device: str = "cuda"):
        self.pipeline = IDMVTONPipeline(model_path, device)
        self.output_dir = Path("./fitting_results")
        self.output_dir.mkdir(exist_ok=True)
        self.history: list[TryOnResult] = []

    def initialize(self):
        self.pipeline.load_model()

    def try_garment(self, person_path: str, garment_path: str, name: str = None) -> TryOnResult:
        if not name:
            name = f"tryon_{hashlib.md5(f'{person_path}{garment_path}'.encode()).hexdigest()[:8]}.png"
        result = self.pipeline.try_on(TryOnRequest(
            person_image_path=person_path, garment_image_path=garment_path,
            output_path=str(self.output_dir / name)
        ))
        self.history.append(result)
        return result

    def batch_try_on(self, person_path: str, garment_paths: list[str]) -> list[TryOnResult]:
        return [self.try_garment(person_path, gp, f"batch_{i:03d}.png") for i, gp in enumerate(garment_paths)]

    def make_comparison_grid(self, paths: list[str], cols: int = 3) -> Image.Image:
        images = [Image.open(p).resize((300, 400)) for p in paths]
        rows = (len(images) + cols - 1) // cols
        grid = Image.new("RGB", (cols * 300, rows * 400), (255, 255, 255))
        for i, img in enumerate(images):
            grid.paste(img, ((i % cols) * 300, (i // cols) * 400))
        grid.save(self.output_dir / "comparison.png")
        return grid


if __name__ == "__main__":
    room = VirtualFittingRoom(device="cuda" if torch.cuda.is_available() else "cpu")
    room.initialize()
    result = room.try_garment("./photos/person.jpg", "./garments/dress.jpg")
    print(f"试衣完成: {result.output_image_path}, 耗时{result.processing_time_seconds}秒")
    results = room.batch_try_on("./photos/person.jpg", ["./garments/dress.jpg", "./garments/jacket.jpg"])
    room.make_comparison_grid([r.output_image_path for r in results])
```

## 常见误区

### 误区一：3D模型文件越大越好

一个沙发50MB，加载30秒，用户早就离开了。必须严格优化：减面到合理面数（移动端<10万面）、DRACO压缩（通常80%+）、KTX2纹理、LOD多细节层次。

### 误区二：忽略移动端性能

3D电商主力场景是手机端。必须针对性优化：限制devicePixelRatio为1.5、减少实时光照、使用Baked Lighting、控制draw call、InstancedMesh渲染重复元素。

### 误区三：AR试穿不考虑光照一致性

虚拟物品光照与真实环境不匹配会有明显"贴图感"。需要使用WebXR的Light Estimation API获取环境光照信息，让虚拟物品的光照方向、色温与摄像头画面一致。

### 误区四：AI试衣不做结果校验

AI输出可能存在伪影、变形、颜色偏差。必须加入质量校验：检测SSIM/PSNR指标、检查人体结构一致性。低于阈值应提示"效果不佳，请更换照片重试"。

### 误区五：只做展示不做数据闭环

用户在3D模型上的交互行为（旋转角度、放大区域、停留时间）是极有价值的偏好数据。必须埋点采集，用于优化推荐和产品设计。

### 练习一

**思路**：预先准备多个精度的glTF文件，在 `useFrame` 中检测距离动态切换。

```tsx
import React, { useState, useMemo } from 'react'
import { useFrame, useThree, useLoader } from '@react-three/fiber'
import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader'
import { DRACOLoader } from 'three/examples/jsm/loaders/DRACOLoader'
import * as THREE from 'three'

interface LODLevel { url: string; maxDistance: number }

function LODModel({ levels }: { levels: LODLevel[] }) {
  const [active, setActive] = useState(0)
  const { camera } = useThree()
  const sorted = useMemo(() => [...levels].sort((a, b) => a.maxDistance - b.maxDistance), [levels])
  const gltf = useLoader(GLTFLoader, sorted[active].url, (loader) => {
    const draco = new DRACOLoader()
    draco.setDecoderPath('https://www.gstatic.com/draco/versioned/decoders/1.5.6/')
    loader.setDRACOLoader(draco)
  })

  useFrame(() => {
    const box = new THREE.Box3().setFromObject(gltf.scene)
    const dist = camera.position.distanceTo(box.getCenter(new THREE.Vector3()))
    let newLevel = sorted.length - 1
    for (let i = 0; i < sorted.length; i++) {
      if (dist <= sorted[i].maxDistance) { newLevel = i; break }
    }
    if (newLevel !== active) setActive(newLevel)
  })

  return <primitive object={gltf.scene} />
}
```

**要点**：生产中应预加载所有LOD级别避免切换卡顿；距离阈值根据商品大小调整。

```tsx
import { Html } from '@react-three/drei'

function Hotspot({ position, title, desc }: { position: [number,number,number]; title: string; desc: string }) {
```

### 练习三

**思路**：SSIM衡量结构相似性，LPIPS衡量感知差异，双阈值策略评估质量。

```python
import torch, numpy as np
from PIL import Image

class TryOnQualityEvaluator:
    def __init__(self, device="cpu"):
        self.device = device
        self.lpips_model = None

    def evaluate(self, person_path: str, result_path: str) -> dict:
        person = np.array(Image.open(person_path).convert("RGB").resize((512, 512)))
        result = np.array(Image.open(result_path).convert("RGB").resize((512, 512)))

        from pytorch_msssim import ssim
        t1 = torch.from_numpy(person).permute(2,0,1).unsqueeze(0).float() / 255
        t2 = torch.from_numpy(result).permute(2,0,1).unsqueeze(0).float() / 255
        ssim_val = ssim(t1, t2, data_range=1.0).item()

        if not self.lpips_model:
            import lpips
            self.lpips_model = lpips.LPIPS(net='alex').to(self.device)
        lpips_val = self.lpips_model(t1*2-1, t2*2-1).item()

        issues = []
        if ssim_val < 0.5: issues.append("结构相似性过低")
        if lpips_val > 0.5: issues.append("感知差异过大")
        quality = "good" if ssim_val >= 0.7 and lpips_val <= 0.3 else \
                  "acceptable" if ssim_val >= 0.5 and lpips_val <= 0.5 else "poor"
        return {"ssim": round(ssim_val, 4), "lpips": round(lpips_val, 4), "quality": quality, "issues": issues}
```

**要点**：SSIM和LPIPS互补评估；双阈值（good/acceptable/poor）给出明确分级。
