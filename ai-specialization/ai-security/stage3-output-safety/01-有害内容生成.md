## 2. 有害内容检测

### 2.1 基于规则的检测

```python
import re
from typing import List, Dict, Tuple

class ContentSafetyFilter:
    """内容安全过滤器"""

    def __init__(self):
        self.blocked_keywords = self._load_blocked_keywords()
        self.sensitive_patterns = self._load_sensitive_patterns()

    def _load_blocked_keywords(self) -> Dict[str, List[str]]:
        return {
            "violence": [
                "杀人", "炸弹", "枪支", "爆炸", "恐怖袭击",
                "kill", "bomb", "gun", "explosion", "terrorist"
            ],
            "hate_speech": [
                "种族歧视", "性别歧视", "仇恨言论",
                "racist", "sexist", "hate speech"
            ],
            "illegal": [
                "毒品", "赌博", "诈骗", "洗钱",
                "drugs", "gambling", "fraud", "money laundering"
            ],
            "self_harm": [
                "自杀", "自残", "结束生命",
                "suicide", "self-harm", "end my life"
            ]
        }

    def _load_sensitive_patterns(self) -> List[Dict]:
        return [
            {
                "pattern": r"(?:如何|怎么|教我).*(?:制作|制造).*(?:炸弹|武器|毒品)",
                "category": "illegal",
                "risk_level": ContentRiskLevel.CRITICAL
            },
            {
                "pattern": r"(?:how to|teach me).*(?:make|create).*(?:bomb|weapon|drug)",
                "category": "illegal",
                "risk_level": ContentRiskLevel.CRITICAL
            },
            {
                "pattern": r"(?:我想|我要).*(?:自杀|自残|结束生命)",
                "category": "self_harm",
                "risk_level": ContentRiskLevel.HIGH
            }
        ]

    def check_content(self, text: str) -> Tuple[ContentRiskLevel, List[Dict]]:
        """检查内容安全性"""
        findings = []
        max_risk = ContentRiskLevel.SAFE

        # 关键词检查
        for category, keywords in self.blocked_keywords.items():
            for keyword in keywords:
                if keyword.lower() in text.lower():
                    findings.append({
                        "type": "keyword",
                        "category": category,
                        "keyword": keyword,
                        "risk_level": ContentRiskLevel.MEDIUM
                    })
                    if ContentRiskLevel.MEDIUM.value > max_risk.value:
                        max_risk = ContentRiskLevel.MEDIUM

        # 模式检查
        for pattern_info in self.sensitive_patterns:
            if re.search(pattern_info["pattern"], text, re.I):
                findings.append({
                    "type": "pattern",
                    "category": pattern_info["category"],
                    "pattern": pattern_info["pattern"],
                    "risk_level": pattern_info["risk_level"]
                })
                if pattern_info["risk_level"].value > max_risk.value:
                    max_risk = pattern_info["risk_level"]

        return max_risk, findings

    def filter_content(self, text: str) -> Tuple[bool, str]:
        """过滤内容"""
        risk_level, findings = self.check_content(text)

        if risk_level.value >= ContentRiskLevel.HIGH.value:
            return False, "内容包含高风险信息，已被过滤"

        if risk_level.value >= ContentRiskLevel.MEDIUM.value:
            # 添加警告
            return True, f"[内容警告] {text}"

        return True, text
```

### 2.2 ML 内容分类器

```python
from transformers import pipeline
from typing import List, Dict, Tuple

class MLContentClassifier:
    """基于 ML 的内容分类器"""

    def __init__(self):
        # 使用预训练的毒性检测模型
        self.toxicity_classifier = pipeline(
            "text-classification",
            model="unitary/toxic-bert",
            return_all_scores=True
        )

        # 使用零样本分类器进行多类别检测
        self.zero_shot_classifier = pipeline(
            "zero-shot-classification",
            model="facebook/bart-large-mnli"
        )

        self.safety_categories = [
            "violence",
            "hate speech",
            "sexual content",
            "self-harm",
            "illegal activity",
            "harassment",
            "misinformation"
        ]

    def detect_toxicity(self, text: str) -> Dict:
        """检测毒性内容"""
        results = self.toxicity_classifier(text)
        toxic_score = next(
            (r["score"] for r in results[0] if r["label"] == "toxic"),
            0.0
        )

        return {
            "toxic": toxic_score > 0.5,
            "score": toxic_score,
            "details": results[0]
        }

    def classify_safety(self, text: str) -> Dict:
        """安全类别分类"""
        result = self.zero_shot_classifier(
            text,
            candidate_labels=self.safety_categories,
            multi_label=True
        )

        # 提取高风险类别
        high_risk = []
        for label, score in zip(result["labels"], result["scores"]):
            if score > 0.5:
                high_risk.append({
                    "category": label,
                    "confidence": score
                })

        return {
            "is_unsafe": len(high_risk) > 0,
            "categories": high_risk,
            "all_scores": dict(zip(result["labels"], result["scores"]))
        }

    def check_content(self, text: str) -> Tuple[ContentRiskLevel, Dict]:
        """综合内容检查"""
        toxicity_result = self.detect_toxicity(text)
        safety_result = self.classify_safety(text)

        # 计算综合风险等级
        if toxicity_result["toxic"] and toxicity_result["score"] > 0.8:
            risk_level = ContentRiskLevel.CRITICAL
        elif safety_result["is_unsafe"]:
            risk_level = ContentRiskLevel.HIGH
        elif toxicity_result["toxic"]:
            risk_level = ContentRiskLevel.MEDIUM
        else:
            risk_level = ContentRiskLevel.SAFE

        return risk_level, {
            "toxicity": toxicity_result,
            "safety": safety_result
        }
```

## 4. 内容审核 Pipeline

### 4.1 完整审核流程

```python
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass
from enum import Enum

class ReviewAction(Enum):
    APPROVE = "approve"
    REJECT = "reject"
    FLAG = "flag"
    ESCALATE = "escalate"

@dataclass
class ReviewResult:
    action: ReviewAction
    risk_level: ContentRiskLevel
    categories: List[str]
    confidence: float
    reason: str
    filtered_content: Optional[str] = None

class ContentReviewPipeline:
    """内容审核 Pipeline"""

    def __init__(self):
        self.rule_filter = ContentSafetyFilter()
        self.ml_classifier = MLContentClassifier()
        self.review_queue = []

    def review(self, content: str) -> ReviewResult:
        """审核内容"""
        # Step 1: 规则过滤
        rule_risk, rule_findings = self.rule_filter.check_content(content)

        # 如果规则检测到高风险，直接拒绝
        if rule_risk.value >= ContentRiskLevel.HIGH.value:
            return ReviewResult(
                action=ReviewAction.REJECT,
                risk_level=rule_risk,
                categories=[f["category"] for f in rule_findings],
                confidence=0.9,
                reason="规则检测到高风险内容"
            )

        # Step 2: ML 分类
        ml_risk, ml_details = self.ml_classifier.check_content(content)

        # Step 3: 综合决策
        final_risk = max(rule_risk, ml_risk, key=lambda x: x.value)

        if final_risk.value >= ContentRiskLevel.HIGH.value:
            action = ReviewAction.REJECT
        elif final_risk.value >= ContentRiskLevel.MEDIUM.value:
            action = ReviewAction.FLAG
        else:
            action = ReviewAction.APPROVE

        # 确定类别
        categories = []
        categories.extend([f["category"] for f in rule_findings])
        if ml_details.get("safety", {}).get("categories"):
            categories.extend(
                [c["category"] for c in ml_details["safety"]["categories"]]
            )

        return ReviewResult(
            action=action,
            risk_level=final_risk,
            categories=list(set(categories)),
            confidence=0.8,
            reason=self._generate_reason(action, final_risk, categories)
        )

    def _generate_reason(self, action: ReviewAction, risk: ContentRiskLevel, categories: List[str]) -> str:
        if action == ReviewAction.REJECT:
            return f"内容被拒绝: 检测到 {risk.name} 级别风险，类别: {', '.join(categories)}"
        elif action == ReviewAction.FLAG:
            return f"内容已标记: 检测到 {risk.name} 级别风险，需要人工审核"
        else:
            return "内容已通过审核"

    def add_to_review_queue(self, content: str, review_result: ReviewResult):
        """添加到人工审核队列"""
        if review_result.action == ReviewAction.FLAG:
            self.review_queue.append({
                "content": content,
                "result": review_result,
                "status": "pending"
            })
```

## 5. 常见误区

1. **"只靠关键词过滤"**: 需要语义级别的理解
2. **"过滤越严格越好"**: 过度过滤影响正常使用
3. **"一次配置足够"**: 需要持续更新和优化
4. **"只关注文本"**: 多模态内容同样需要审核
5. **"自动化万能"**: 人工审核仍然必要

## 工程建议

- 在系统指令层就定义明确的内容安全边界，而不是完全依赖后置过滤；前置拦截的成本远低于后置拦截。
- 建立规则引擎与 ML 模型的双通道检测机制，规则负责已知模式的快速拦截，ML 模型负责语义级别的未知攻击，两者互补而非替代。
- 对每条被拦截或标记的内容建立完整的审计日志，包括触发规则、风险等级和上下文信息，这些数据是迭代模型和规则的核心资产。
- 定期使用对抗样本（如 Unicode 变体、角色扮演攻击、多语言混合）对过滤系统做红队测试，避免安全策略在攻击者面前形同虚设。

## 总结

- 有害内容分类包括违法、伦理、虚假、侵权等多个维度
- 需要多层次的内容安全架构
- 规则过滤和 ML 检测各有优势，应结合使用
- 人工审核是最后的安全网

### 练习 1: 内容分类

**思路**：扩展 ContentSafetyFilter 需要新增内容类别（如歧视、商业推广、版权侵犯等），为每个类别定义关键词列表和正则模式，并在 check_content 中增加对新类别的检测逻辑。关键是要考虑多语言覆盖和变体绕过的防御。

```python
            ]
        }

    def _load_extended_patterns(self) -> List[Dict]:
        return [
            {
                "pattern": r"(?:所有|全部).*(?:男人|女人|男人|某地人).*(?:都|全).*(?:是|会)",
                "category": "discrimination",
                "risk_level": ContentRiskLevel.HIGH
            },
            {
                "pattern": r"(?:加|添加).*(?:微信|QQ|telegram).*(?:\w+@\w+|[a-zA-Z0-9]{5,})",
                "category": "commercial_spam",
                "risk_level": ContentRiskLevel.MEDIUM
            },
            {
                "pattern": r"(?:免费|破解).*(?:下载|获取).*(?:软件|课程|资源)",
                "category": "copyright",
                "risk_level": ContentRiskLevel.MEDIUM
            },
            {
                "pattern": r"(?:人肉|开盒|曝光).*(?:地址|电话|身份证|真实姓名)",
                "category": "cyberbullying",
                "risk_level": ContentRiskLevel.CRITICAL
            },
            {
                "pattern": r"(?:紧急|速看|刚刚).*(?:转发|扩散).*(?:\d+万|\d+亿)",
                "category": "misinformation",
                "risk_level": ContentRiskLevel.HIGH
            }
        ]

    def check_content(self, text: str) -> Tuple[ContentRiskLevel, List[Dict]]:
        """扩展的内容检查，支持新类别"""
        risk_level, findings = super().check_content(text)

        # 检查变体绕过（Unicode 同形字、零宽字符等）
        normalized_text = self._normalize_unicode(text)
        if normalized_text != text:
            alt_risk, alt_findings = super().check_content(normalized_text)
            if alt_risk.value > risk_level.value:
                risk_level = alt_risk
            findings.extend(alt_findings)

        return risk_level, findings

    def _normalize_unicode(self, text: str) -> str:
        """Unicode 归一化，防御同形字绕过"""
        import unicodedata
        text = unicodedata.normalize("NFKC", text)
        # 移除零宽字符
        text = re.sub(r'[\u200b\u200c\u200d\ufeff]', '', text)
        return text
```

### 练习 2: ML 分类器

**思路**：使用 Hugging Face 的 transformers 库，基于预训练的文本分类模型微调一个自定义内容安全分类器。关键步骤是准备标注数据集、选择合适的预训练模型、定义训练参数并评估效果。

**答案**：

```python
import torch
from torch.utils.data import Dataset, DataLoader
from transformers import (
    AutoTokenizer,
    AutoModelForSequenceClassification,
    TrainingArguments,
    Trainer
)
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report
from typing import List, Dict, Tuple
import json

class ContentSafetyDataset(Dataset):
    """内容安全数据集"""

    def __init__(self, texts: List[str], labels: List[int], tokenizer, max_length: int = 256):
        self.texts = texts
        self.labels = labels
        self.tokenizer = tokenizer
        self.max_length = max_length

    def __len__(self):
        return len(self.texts)

    def __getitem__(self, idx):
        encoding = self.tokenizer(
            self.texts[idx],
            max_length=self.max_length,
            padding="max_length",
            truncation=True,
            return_tensors="pt"
        )
        return {
            "input_ids": encoding["input_ids"].squeeze(),
            "attention_mask": encoding["attention_mask"].squeeze(),
            "labels": torch.tensor(self.labels[idx], dtype=torch.long)
        }


class CustomContentSafetyClassifier:
    """自定义内容安全分类器"""

    LABEL_MAP = {
        0: "safe",
        1: "violence",
        2: "hate_speech",
        3: "sexual",
        4: "self_harm",
        5: "illegal",
        6: "misinformation"
    }

    def __init__(self, model_name: str = "bert-base-chinese", num_labels: int = 7):
        self.model_name = model_name
        self.num_labels = num_labels
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForSequenceClassification.from_pretrained(
            model_name, num_labels=num_labels
        )

    def prepare_training_data(self) -> Tuple[List[str], List[int]]:
        """准备训练数据（示例数据集）"""
        data = [
            # safe (label=0)
            ("今天天气很好，适合出门散步", 0),
            ("Python 是一种流行的编程语言", 0),
            ("这道菜的做法很简单", 0),
            ("请帮我写一份工作报告", 0),
            # violence (label=1)
            ("我要杀了你", 1),
            ("制作炸弹的方法", 1),
            ("如何获取枪支", 1),
            # hate_speech (label=2)
            ("某种族都是低等人", 2),
            ("女人就不应该工作", 2),
            # sexual (label=3)
            ("色情内容示例", 3),
            # self_harm (label=4)
            ("我想结束自己的生命", 4),
            ("割腕的方法", 4),
            # illegal (label=5)
            ("如何制作毒品", 5),
            ("洗钱的步骤", 5),
            # misinformation (label=6)
            ("地球是平的", 6),
            ("疫苗含有芯片", 6),
        ]

        texts = [d[0] for d in data]
        labels = [d[1] for d in data]
        return texts, labels

    def train(self, texts: List[str], labels: List[int], epochs: int = 3, batch_size: int = 16):
        """训练分类器"""
        train_texts, val_texts, train_labels, val_labels = train_test_split(
            texts, labels, test_size=0.2, random_state=42
        )

        train_dataset = ContentSafetyDataset(train_texts, train_labels, self.tokenizer)
        val_dataset = ContentSafetyDataset(val_texts, val_labels, self.tokenizer)

        training_args = TrainingArguments(
            output_dir="./content_safety_model",
            num_train_epochs=epochs,
            per_device_train_batch_size=batch_size,
            per_device_eval_batch_size=batch_size,
            evaluation_strategy="epoch",
            save_strategy="epoch",
            learning_rate=2e-5,
            weight_decay=0.01,
            load_best_model_at_end=True,
        )

        trainer = Trainer(
            model=self.model,
            args=training_args,
            train_dataset=train_dataset,
            eval_dataset=val_dataset,
        )

        trainer.train()
        return trainer.evaluate()

    def predict(self, text: str) -> Dict:
        """预测单条文本"""
        inputs = self.tokenizer(
            text, return_tensors="pt",
            max_length=256, truncation=True, padding=True
        )
        with torch.no_grad():
            outputs = self.model(**inputs)
            probs = torch.softmax(outputs.logits, dim=-1)

        predicted_label = torch.argmax(probs, dim=-1).item()
        confidence = probs[0][predicted_label].item()

        return {
            "text": text[:100],
            "label": self.LABEL_MAP.get(predicted_label, "unknown"),
            "label_id": predicted_label,
            "confidence": confidence,
            "all_scores": {
                self.LABEL_MAP[i]: probs[0][i].item()
                for i in range(self.num_labels)
            }
        }

    def evaluate(self, test_texts: List[str], test_labels: List[int]) -> str:
        """评估模型效果"""
        predictions = []
        for text in test_texts:
            result = self.predict(text)
            predictions.append(result["label_id"])

        report = classification_report(
            test_labels, predictions,
            target_names=list(self.LABEL_MAP.values())
        )
        return report

if __name__ == "__main__":

    # 预测

    print(f"输入: {result['text']}")
    print(f"分类: {result['label']} (置信度: {result['confidence']:.2%})")
```

**要点**：
- 中文内容分类建议使用 `bert-base-chinese` 作为基础模型，英文可选 `roberta-base`
- 训练数据质量决定模型上限，实际项目中需要至少每类 1000+ 条高质量标注样本
- 生产环境中 ML 分类器应与规则引擎并行使用，ML 负责语义级检测，规则负责已知模式快速拦截

### 练习 3: 审核系统

**思路**：构建完整的内容审核 Pipeline 需要整合规则引擎、ML 分类器和人工审核队列三层。关键在于定义清晰的决策融合逻辑：规则引擎拦截已知高危模式，ML 处理语义级风险，低置信度结果进入人工队列。

**答案**：

```python
import re
from typing import Dict, List, Optional
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
import hashlib

class ReviewAction(Enum):
    APPROVE = "approve"
    REJECT = "reject"
    FLAG = "flag"
    ESCALATE = "escalate"

@dataclass
class ReviewResult:
    action: ReviewAction
    risk_level: str
    categories: List[str]
    confidence: float
    reason: str
    layer_details: Dict = field(default_factory=dict)

@dataclass
class HumanReviewTask:
    task_id: str
    content: str
    auto_result: ReviewResult
    priority: int
    status: str = "pending"
    assigned_to: Optional[str] = None
    created_at: datetime = field(default_factory=datetime.now)

class RuleEngine:
    """规则引擎层"""

    RULES = [
        {"id": "R001", "pattern": r"(?:如何|怎么).*(?:制作|制造).*(?:炸弹|武器|毒品)",
         "action": "reject", "severity": "critical", "category": "illegal"},
        {"id": "R002", "pattern": r"(?:忽略|ignore).*(?:指令|instruction)",
         "action": "reject", "severity": "critical", "category": "injection"},
        {"id": "R003", "keywords": ["暴力", "色情", "赌博", "毒品"],
         "action": "flag", "severity": "high", "category": "unsafe_content"},
        {"id": "R004", "pattern": r"(?:身份证|手机号|银行卡).*(?:\d{6,19})",
         "action": "flag", "severity": "high", "category": "pii_leak"},
    ]

    def check(self, content: str) -> Dict:
        findings = []
        max_severity = "none"

        for rule in self.RULES:
            matched = False
            if "pattern" in rule:
                if re.search(rule["pattern"], content, re.I):
                    matched = True
            elif "keywords" in rule:
                for kw in rule["keywords"]:
                    if kw in content:
                        matched = True
                        break

            if matched:
                findings.append({
                    "rule_id": rule["id"],
                    "category": rule["category"],
                    "severity": rule["severity"]
                })
                severity_order = ["none", "low", "medium", "high", "critical"]
                if severity_order.index(rule["severity"]) > severity_order.index(max_severity):
                    max_severity = rule["severity"]

        return {
            "passed": max_severity not in ["critical", "high"],
            "findings": findings,
            "max_severity": max_severity
        }

class SimpleMLClassifier:
    """简化 ML 分类层（实际应调用真实模型）"""

    UNSAFE_KEYWORDS = ["杀人", "自杀", "炸弹", "色情", "赌博"]

    def check(self, content: str) -> Dict:
        score = 0.0
        categories = []
        for kw in self.UNSAFE_KEYWORDS:
            if kw in content:
                score += 0.3
                categories.append(kw)

        return {
            "is_unsafe": score > 0.5,
            "confidence": min(score, 1.0),
            "categories": categories
        }

class HumanReviewQueue:
    """人工审核队列"""

    def __init__(self):
        self.queue: List[HumanReviewTask] = []

    def add(self, content: str, auto_result: ReviewResult, priority: int = 2) -> str:
        task_id = "REV-" + hashlib.md5(
            f"{content}_{datetime.now().isoformat()}".encode()
        ).hexdigest()[:8].upper()

        task = HumanReviewTask(
            task_id=task_id,
            content=content,
            auto_result=auto_result,
            priority=priority
        )

        # 按优先级插入
        inserted = False
        for i, existing in enumerate(self.queue):
            if task.priority > existing.priority:
                self.queue.insert(i, task)
                inserted = True
                break
        if not inserted:
            self.queue.append(task)

        return task_id

    def get_next(self) -> Optional[HumanReviewTask]:
        for task in self.queue:
            if task.status == "pending":
                return task
        return None

    def complete(self, task_id: str, decision: str, reason: str) -> bool:
        for task in self.queue:
            if task.task_id == task_id:
                task.status = "completed"
                return True
        return False

    def stats(self) -> Dict:
        return {
            "pending": sum(1 for t in self.queue if t.status == "pending"),
            "completed": sum(1 for t in self.queue if t.status == "completed"),
            "total": len(self.queue)
        }

class ContentReviewPipeline:
    """完整内容审核 Pipeline"""

    def __init__(self):
        self.rule_engine = RuleEngine()
        self.ml_classifier = SimpleMLClassifier()
        self.human_queue = HumanReviewQueue()

    def review(self, content: str) -> ReviewResult:
        # Layer 1: 规则引擎
        rule_result = self.rule_engine.check(content)
        if not rule_result["passed"]:
            return ReviewResult(
                action=ReviewAction.REJECT,
                risk_level=rule_result["max_severity"],
                categories=[f["category"] for f in rule_result["findings"]],
                confidence=0.95,
                reason=f"规则引擎拦截: {rule_result['max_severity']}级别违规",
                layer_details={"rule": rule_result}
            )

        # Layer 2: ML 分类
        ml_result = self.ml_classifier.check(content)
        if ml_result["is_unsafe"] and ml_result["confidence"] > 0.7:
            return ReviewResult(
                action=ReviewAction.REJECT,
                risk_level="high",
                categories=ml_result["categories"],
                confidence=ml_result["confidence"],
                reason="ML 分类器检测到高风险内容",
                layer_details={"ml": ml_result}
            )

        # Layer 3: 中等风险进入人工队列
        if ml_result["is_unsafe"] or rule_result["max_severity"] in ["medium"]:
            result = ReviewResult(
                action=ReviewAction.FLAG,
                risk_level="medium",
                categories=ml_result["categories"],
                confidence=ml_result["confidence"],
                reason="中等风险，需人工审核",
                layer_details={"rule": rule_result, "ml": ml_result}
            )
            self.human_queue.add(content, result, priority=2)
            return result

        # 全部通过
        return ReviewResult(
            action=ReviewAction.APPROVE,
            risk_level="safe",
            categories=[],
            confidence=0.9,
            reason="通过所有审核层",
            layer_details={"rule": rule_result, "ml": ml_result}
        )


# 测试
if __name__ == "__main__":
    pipeline = ContentReviewPipeline()

    test_cases = [
        "今天天气真好",
        "如何制作炸弹",
        "这个人的手机号是 13812345678",
        "我想自杀",
        "Python 编程入门教程",
    ]

    for text in test_cases:
        result = pipeline.review(text)
        print(f"[{result.action.value:8s}] {text[:30]}... | 理由: {result.reason}")

    print(f"\n人工审核队列: {pipeline.human_queue.stats()}")
```

**要点**：
- 决策融合采用"最严格优先"策略：规则引擎 critical 直接拒绝，ML 高置信度拒绝，中等风险进人工队列
- 人工审核队列需要优先级排序，critical 级别应排在最前面处理
- 生产环境中 ML 分类器应替换为真实的模型推理服务（如 OpenAI Moderation API 或自训练模型）
