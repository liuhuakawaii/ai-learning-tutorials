## 场景引入

2024 年，某头部内容平台上线了 AI 生成内容自动发布功能，初期仅用单一 ML 分类器做审核。上线两周内，大量通过对抗样本伪装的违规内容涌入——攻击者在图片中嵌入肉眼不可见的扰动文字，在正常段落中穿插隐晦的违规暗示。单一审核模型的漏检率飙升至 12%，监管部门发出整改通知。团队紧急重构为规则引擎 + ML 模型 + LLM 审核的三层架构，漏检率降至 0.3%。这个案例说明：内容审核不是"接一个 API 就完事"的工程问题，而是需要多层防御、持续迭代和人工兜底的系统性工程。

### 1.2 审核策略对比

| 策略 | 原理 | 优点 | 缺点 | 延迟 | 成本 |
|------|------|------|------|------|------|
| 规则引擎 | 关键词/正则匹配 | 快速、可解释 | 误报高、维护难 | <10ms | 低 |
| ML 分类器 | 深度学习分类 | 准确率高 | 需要标注数据 | 50-200ms | 中 |
| 第三方 API | 云服务调用 | 开箱即用 | 依赖外部、延迟高 | 100-500ms | 高 |
| LLM 审核 | 大模型判断 | 理解语义 | 成本高、速度慢 | 500ms+ | 很高 |
| 人工审核 | 人工判断 | 最准确 | 不可扩展 | 分钟级 | 最高 |

## 2. 审核引擎实现

### 2.1 多层审核引擎

```python
import openai
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass
from enum import Enum
import asyncio
import re

class ReviewDecision(Enum):
    APPROVE = "approve"
    REJECT = "reject"
    FLAG = "flag"
    ESCALATE = "escalate"

@dataclass
class ReviewResult:
    decision: ReviewDecision
    confidence: float
    categories: List[str]
    reason: str
    details: Dict
    layer_results: Dict

class MultiLayerReviewEngine:
    """多层审核引擎"""

    def __init__(self, openai_api_key: str = None):
        self.rule_engine = ContentSafetyRuleEngine()
        self.ml_engine = MLContentClassifier(openai_api_key)
        self.llm_engine = LLMContentReviewer(openai_api_key)
        self.decision_maker = DecisionMaker()

    async def review(self, content: str) -> ReviewResult:
        """执行多层审核"""
        # 并行执行各层审核
        rule_result, ml_result, llm_result = await asyncio.gather(
            self._rule_review(content),
            self._ml_review(content),
            self._llm_review(content)
        )

        # 融合决策
        final_decision = self.decision_maker.decide(
            rule_result, ml_result, llm_result
        )

        return final_decision

    async def _rule_review(self, content: str) -> Dict:
        return self.rule_engine.review(content)

    async def _ml_review(self, content: str) -> Dict:
        return self.ml_engine.check_content(content)

    async def _llm_review(self, content: str) -> Dict:
        return await self.llm_engine.review(content)


class ContentSafetyRuleEngine:
    """内容安全规则引擎"""

    def __init__(self):
        self.rules = self._load_rules()

    def _load_rules(self) -> List[Dict]:
        return [
            {
                "id": "R001",
                "name": "违禁词检测",
                "type": "keyword",
                "keywords": ["暴力", "色情", "赌博", "毒品", "枪支", "爆炸物"],
                "action": "reject",
                "severity": "high"
            },
            {
                "id": "R002",
                "name": "危险指令检测",
                "type": "pattern",
                "pattern": r"(?:如何|怎么|教我).*(?:制作|制造|获取).*(?:炸弹|武器|毒品|毒药)",
                "action": "reject",
                "severity": "critical"
            },
            {
                "id": "R003",
                "name": "个人信息泄露检测",
                "type": "pattern",
                "pattern": r"(?:身份证|手机号|银行卡).*(?:\d{6}|\d{11}|\d{16,19})",
                "action": "flag",
                "severity": "high"
            },
            {
                "id": "R004",
                "name": "Prompt 注入检测",
                "type": "pattern",
                "pattern": r"(?:忽略|ignore).*(?:之前|above|之前).*(?:指令|instruction|prompt)",
                "action": "reject",
                "severity": "critical"
            },
            {
                "id": "R005",
                "name": "长度异常检测",
                "type": "length",
                "max_length": 10000,
                "action": "flag",
                "severity": "low"
            },
            {
                "id": "R006",
                "name": "重复内容检测",
                "type": "repetition",
                "threshold": 0.8,
                "action": "flag",
                "severity": "medium"
            }
        ]

    def review(self, content: str) -> Dict:
        findings = []

        for rule in self.rules:
            if rule["type"] == "keyword":
                for keyword in rule["keywords"]:
                    if keyword in content:
                        findings.append({
                            "rule_id": rule["id"],
                            "rule_name": rule["name"],
                            "finding": f"包含违禁词: {keyword}",
                            "severity": rule["severity"]
                        })

            elif rule["type"] == "pattern":
                match = re.search(rule["pattern"], content, re.I)
                if match:
                    findings.append({
                        "rule_id": rule["id"],
                        "rule_name": rule["name"],
                        "finding": f"匹配敏感模式: {match.group()[:50]}",
                        "severity": rule["severity"]
                    })

            elif rule["type"] == "length":
                if len(content) > rule["max_length"]:
                    findings.append({
                        "rule_id": rule["id"],
                        "rule_name": rule["name"],
                        "finding": f"内容过长: {len(content)} > {rule['max_length']}",
                        "severity": rule["severity"]
                    })

            elif rule["type"] == "repetition":
                if self._check_repetition(content, rule["threshold"]):
                    findings.append({
                        "rule_id": rule["id"],
                        "rule_name": rule["name"],
                        "finding": "检测到高重复度内容",
                        "severity": rule["severity"]
                    })

        critical = [f for f in findings if f["severity"] == "critical"]
        high = [f for f in findings if f["severity"] == "high"]

        return {
            "passed": len(critical) == 0 and len(high) == 0,
            "findings": findings,
            "critical_count": len(critical),
            "high_count": len(high),
            "severity": max([f["severity"] for f in findings], key=lambda s: ["low","medium","high","critical"].index(s), default="none")
        }

    def _check_repetition(self, text: str, threshold: float) -> bool:
        """检查文本重复度"""
        if len(text) < 100:
            return False
        chunks = [text[i:i+50] for i in range(0, len(text)-50, 50)]
        if len(chunks) < 2:
            return False
        unique = len(set(chunks))
        ratio = unique / len(chunks)
        return ratio < (1 - threshold)


class MLContentClassifier:
    """ML 内容分类器"""

    def __init__(self, api_key: str = None):
        self.api_key = api_key
        self.threshold = 0.7

    def check_content(self, content: str) -> Dict:
        """使用 ML 模型检查内容"""
        # 简化实现：使用 OpenAI Moderation API
        if self.api_key:
            try:
                client = openai.OpenAI(api_key=self.api_key)
                response = client.moderations.create(input=content)
                result = response.results[0]

                categories = []
                if result.flagged:
                    for cat, flagged in result.categories.model_dump().items():
                        if flagged:
                            categories.append({
                                "category": cat,
                                "score": getattr(result.category_scores, cat, 0)
                            })

                return {
                    "is_unsafe": result.flagged,
                    "categories": categories,
                    "confidence": max(
                        [c["score"] for c in categories], default=0
                    ),
                    "raw_scores": result.category_scores.model_dump()
                }
            except Exception as e:
                return {"is_unsafe": False, "categories": [], "confidence": 0, "error": str(e)}

        return {"is_unsafe": False, "categories": [], "confidence": 0}


class LLMContentReviewer:
    """基于 LLM 的内容审核"""

    def __init__(self, api_key: str = None):
        self.api_key = api_key

    async def review(self, content: str) -> Dict:
        """使用 LLM 进行内容审核"""
        if not self.api_key:
            return {"is_unsafe": False, "categories": [], "confidence": 0}

        client = openai.OpenAI(api_key=self.api_key)

        system_prompt = """你是一个内容安全审核员。判断以下内容是否存在安全风险。

返回 JSON 格式:
{
  "is_unsafe": true/false,
  "categories": ["类别1", "类别2"],
  "confidence": 0-100,
  "reason": "判断理由"
}

审核维度:
1. 暴力/血腥内容
2. 色情/性暗示内容
3. 仇恨/歧视言论
4. 自残/自杀相关
5. 违法犯罪内容
6. 虚假信息/谣言
7. 骚扰/霸凌
8. 个人隐私泄露"""

        try:
            response = client.chat.completions.create(
                model="gpt-4",
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": f"请审核以下内容:\n\n{content[:1000]}"}
                ],
                temperature=0,
                max_tokens=200
            )

            import json
            result = json.loads(response.choices[0].message.content)
            return {
                "is_unsafe": result.get("is_unsafe", False),
                "categories": [{"category": c, "score": result.get("confidence", 50)/100} for c in result.get("categories", [])],
                "confidence": result.get("confidence", 50) / 100,
                "reason": result.get("reason", "")
            }
        except Exception as e:
            return {"is_unsafe": False, "categories": [], "confidence": 0, "error": str(e)}


class DecisionMaker:
    """决策融合器"""

    def decide(self, rule_result: Dict, ml_result: Dict, llm_result: Dict) -> ReviewResult:
        """融合多层审核结果"""
        layer_results = {
            "rule": rule_result,
            "ml": ml_result,
            "llm": llm_result
        }

        # 规则引擎：critical 直接拒绝
        if not rule_result["passed"]:
            critical_findings = [f for f in rule_result["findings"] if f["severity"] == "critical"]
            if critical_findings:
                return ReviewResult(
                    decision=ReviewDecision.REJECT,
                    confidence=0.98,
                    categories=[f["rule_name"] for f in critical_findings],
                    reason=f"规则引擎检测到 {len(critical_findings)} 条严重违规",
                    details={"rule_findings": critical_findings},
                    layer_results=layer_results
                )

        # ML 模型：高置信度拒绝
        if ml_result.get("is_unsafe", False) and ml_result.get("confidence", 0) > 0.8:
            return ReviewResult(
                decision=ReviewDecision.REJECT,
                confidence=ml_result["confidence"],
                categories=[c["category"] for c in ml_result.get("categories", [])],
                reason="ML 模型高置信度检测到违规",
                details={"ml_result": ml_result},
                layer_results=layer_results
            )

        # LLM 审核：高置信度拒绝
        if llm_result.get("is_unsafe", False) and llm_result.get("confidence", 0) > 0.8:
            return ReviewResult(
                decision=ReviewDecision.REJECT,
                confidence=llm_result["confidence"],
                categories=llm_result.get("categories", []),
                reason=f"LLM 审核: {llm_result.get('reason', '')}",
                details={"llm_result": llm_result},
                layer_results=layer_results
            )

        # 中等置信度：标记人工审核
        if ml_result.get("is_unsafe", False) or llm_result.get("is_unsafe", False):
            confidence = max(
                ml_result.get("confidence", 0),
                llm_result.get("confidence", 0)
            )
            return ReviewResult(
                decision=ReviewDecision.FLAG,
                confidence=confidence,
                categories=[],
                reason="中等置信度风险，需人工审核",
                details={"ml_result": ml_result, "llm_result": llm_result},
                layer_results=layer_results
            )

        # 规则引擎 high findings：标记
        high_findings = [f for f in rule_result.get("findings", []) if f["severity"] == "high"]
        if high_findings:
            return ReviewResult(
                decision=ReviewDecision.FLAG,
                confidence=0.6,
                categories=[f["rule_name"] for f in high_findings],
                reason="规则引擎检测到中等风险",
                details={"rule_findings": high_findings},
                layer_results=layer_results
            )

        # 全部通过
        return ReviewResult(
            decision=ReviewDecision.APPROVE,
            confidence=0.95,
            categories=[],
            reason="通过所有审核层",
            details={},
            layer_results=layer_results
        )
```

### 2.2 OpenAI Moderation API 集成

```python
import openai
from typing import Dict, List

class OpenAIModerationService:
    """OpenAI Moderation API 集成"""

    def __init__(self, api_key: str):
        self.client = openai.OpenAI(api_key=api_key)

    def check(self, content: str) -> Dict:
        """检查内容安全性"""
        response = self.client.moderations.create(input=content)
        result = response.results[0]

        flagged_categories = []
        category_scores = {}

        for category, flagged in result.categories.model_dump().items():
            score = getattr(result.category_scores, category, 0)
            category_scores[category] = score
            if flagged:
                flagged_categories.append({
                    "category": category,
                    "score": score,
                    "severity": self._get_severity(category, score)
                })

        return {
            "flagged": result.flagged,
            "categories": flagged_categories,
            "scores": category_scores,
            "max_score": max(category_scores.values()) if category_scores else 0,
            "risk_level": self._assess_risk(flagged_categories)
        }

    def _get_severity(self, category: str, score: float) -> str:
        """评估严重程度"""
        critical_categories = ["violence", "sexual", "self-harm"]
        if category in critical_categories and score > 0.8:
            return "critical"
        elif score > 0.7:
            return "high"
        elif score > 0.5:
            return "medium"
        return "low"

    def _assess_risk(self, flagged: List[Dict]) -> str:
        """评估总体风险等级"""
        if not flagged:
            return "safe"
        severities = [f["severity"] for f in flagged]
        if "critical" in severities:
            return "critical"
        if "high" in severities:
            return "high"
        if "medium" in severities:
            return "medium"
        return "low"

    def batch_check(self, contents: List[str]) -> List[Dict]:
        """批量检查内容"""
        # OpenAI Moderation API 支持批量输入
        response = self.client.moderations.create(input=contents)
        results = []
        for result in response.results:
            flagged = [cat for cat, val in result.categories.model_dump().items() if val]
            results.append({
                "flagged": result.flagged,
                "categories": flagged,
                "scores": result.category_scores.model_dump()
            })
        return results
```

## 3. 人工审核流程

### 3.1 审核队列管理

```python
from typing import List, Dict, Optional
from datetime import datetime
from dataclasses import dataclass, field
from enum import Enum

class TaskPriority(Enum):
    LOW = 1
    MEDIUM = 2
    HIGH = 3
    CRITICAL = 4

@dataclass
class ReviewTask:
    task_id: str
    content: str
    auto_result: Dict
    priority: TaskPriority
    status: str = "pending"
    assigned_to: Optional[str] = None
    human_decision: Optional[str] = None
    human_reason: Optional[str] = None
    created_at: datetime = field(default_factory=datetime.now)
    completed_at: Optional[datetime] = None
    tags: List[str] = field(default_factory=list)

class ReviewQueueManager:
    """审核队列管理器"""

    def __init__(self):
        self.pending_queue: List[ReviewTask] = []
        self.completed_queue: List[ReviewTask] = []
        self.reviewers: Dict[str, Dict] = {}

    def add_task(self, content: str, auto_result: Dict, priority: TaskPriority = TaskPriority.MEDIUM) -> ReviewTask:
        """添加审核任务"""
        task = ReviewTask(
            task_id=self._generate_task_id(),
            content=content,
            auto_result=auto_result,
            priority=priority,
            tags=self._auto_tag(auto_result)
        )

        # 按优先级插入
        inserted = False
        for i, existing in enumerate(self.pending_queue):
            if task.priority.value > existing.priority.value:
                self.pending_queue.insert(i, task)
                inserted = True
                break
        if not inserted:
            self.pending_queue.append(task)

        return task

    def assign_task(self, task_id: str, reviewer_id: str) -> bool:
        """分配审核任务"""
        for task in self.pending_queue:
            if task.task_id == task_id and task.status == "pending":
                task.assigned_to = reviewer_id
                task.status = "assigned"
                if reviewer_id in self.reviewers:
                    self.reviewers[reviewer_id]["active_tasks"] += 1
                return True
        return False

    def auto_assign(self, reviewer_id: str) -> Optional[ReviewTask]:
        """自动分配最高优先级的任务"""
        for task in self.pending_queue:
            if task.status == "pending":
                self.assign_task(task.task_id, reviewer_id)
                return task
        return None

    def complete_task(self, task_id: str, decision: str, reason: str) -> bool:
        """完成审核任务"""
        for i, task in enumerate(self.pending_queue):
            if task.task_id == task_id:
                task.human_decision = decision
                task.human_reason = reason
                task.status = "completed"
                task.completed_at = datetime.now()

                if task.assigned_to and task.assigned_to in self.reviewers:
                    self.reviewers[task.assigned_to]["active_tasks"] -= 1
                    self.reviewers[task.assigned_to]["completed_tasks"] += 1

                self.completed_queue.append(task)
                self.pending_queue.pop(i)
                return True
        return False

    def get_stats(self) -> Dict:
        """获取审核统计"""
        pending_by_priority = {}
        for task in self.pending_queue:
            p = task.priority.name
            pending_by_priority[p] = pending_by_priority.get(p, 0) + 1

        return {
            "pending": len(self.pending_queue),
            "completed": len(self.completed_queue),
            "pending_by_priority": pending_by_priority,
            "avg_review_time": self._calculate_avg_time(),
            "reviewer_stats": {
                rid: {
                    "active": info["active_tasks"],
                    "completed": info["completed_tasks"]
                }
                for rid, info in self.reviewers.items()
            }
        }

    def register_reviewer(self, reviewer_id: str, name: str, expertise: List[str]):
        """注册审核员"""
        self.reviewers[reviewer_id] = {
            "name": name,
            "expertise": expertise,
            "active_tasks": 0,
            "completed_tasks": 0
        }

    def _auto_tag(self, auto_result: Dict) -> List[str]:
        """自动打标签"""
        tags = []
        if auto_result.get("critical_count", 0) > 0:
            tags.append("critical")
        if "ml" in str(auto_result.get("layer_results", {})):
            tags.append("ml-flagged")
        return tags

    def _generate_task_id(self) -> str:
        import hashlib
        return "REV-" + hashlib.md5(str(datetime.now()).encode()).hexdigest()[:8].upper()

    def _calculate_avg_time(self) -> float:
        if not self.completed_queue:
            return 0.0
        times = [
            (t.completed_at - t.created_at).total_seconds()
            for t in self.completed_queue
            if t.completed_at
        ]
        return sum(times) / len(times) if times else 0.0
```

## 4. 审核效果评估

### 4.1 评估指标体系

```python
from typing import List, Dict
from dataclasses import dataclass

@dataclass
class EvaluationMetrics:
    accuracy: float
    precision: float
    recall: float
    f1_score: float
    false_positive_rate: float
    false_negative_rate: float
    confusion_matrix: List[List[int]]

class ReviewEvaluator:
    """审核效果评估器"""

    def __init__(self):
        self.predictions = []
        self.ground_truth = []

    def add_sample(self, predicted: str, actual: str):
        """添加样本"""
        self.predictions.append(predicted)
        self.ground_truth.append(actual)

    def evaluate(self) -> EvaluationMetrics:
        """计算评估指标"""
        # 转换为二分类
        y_pred = [1 if p == "approve" else 0 for p in self.predictions]
        y_true = [1 if a == "approve" else 0 for a in self.ground_truth]

        tp = sum(1 for p, t in zip(y_pred, y_true) if p == 1 and t == 1)
        fp = sum(1 for p, t in zip(y_pred, y_true) if p == 1 and t == 0)
        tn = sum(1 for p, t in zip(y_pred, y_true) if p == 0 and t == 0)
        fn = sum(1 for p, t in zip(y_pred, y_true) if p == 0 and t == 1)

        precision = tp / (tp + fp) if (tp + fp) > 0 else 0
        recall = tp / (tp + fn) if (tp + fn) > 0 else 0
        f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0
        accuracy = (tp + tn) / len(y_pred) if y_pred else 0

        return EvaluationMetrics(
            accuracy=accuracy,
            precision=precision,
            recall=recall,
            f1_score=f1,
            false_positive_rate=fp / (fp + tn) if (fp + tn) > 0 else 0,
            false_negative_rate=fn / (fn + tp) if (fn + tp) > 0 else 0,
            confusion_matrix=[[tp, fp], [fn, tn]]
        )

    def generate_report(self) -> str:
        """生成评估报告"""
        metrics = self.evaluate()
        return f"""
# 审核效果评估报告

## 核心指标
| 指标 | 值 | 说明 |
|------|-----|------|
| 准确率 | {metrics.accuracy:.4f} | 整体判断正确的比例 |
| 精确率 | {metrics.precision:.4f} | 判为违规中真正违规的比例 |
| 召回率 | {metrics.recall:.4f} | 真正违规被检出的比例 |
| F1 分数 | {metrics.f1_score:.4f} | 精确率和召回率的调和平均 |
| 误报率 | {metrics.false_positive_rate:.4f} | 正常内容被误判为违规 |
| 漏报率 | {metrics.false_negative_rate:.4f} | 违规内容未被检出 |

## 混淆矩阵
              预测通过  预测拒绝
实际通过       {metrics.confusion_matrix[0][0]}        {metrics.confusion_matrix[0][1]}
实际拒绝       {metrics.confusion_matrix[1][0]}        {metrics.confusion_matrix[1][1]}

## 建议
{'- 误报率偏高，建议调整规则阈值' if metrics.false_positive_rate > 0.1 else '- 误报率在可接受范围内'}
{'- 漏报率偏高，建议增加 ML 模型覆盖' if metrics.false_negative_rate > 0.05 else '- 漏报率在可接受范围内'}
"""
```

### 4.2 A/B 测试框架

```python
import random
from typing import Dict, List
from datetime import datetime

class ReviewABTest:
    """审核策略 A/B 测试"""

    def __init__(self, engine_a, engine_b, split_ratio: float = 0.5):
        self.engine_a = engine_a
        self.engine_b = engine_b
        self.split_ratio = split_ratio
        self.results_a = []
        self.results_b = []

    async def review(self, content: str) -> Dict:
        """分配并执行审核"""
        if random.random() < self.split_ratio:
            result = await self.engine_a.review(content)
            self.results_a.append({
                "content_hash": hash(content),
                "decision": result.decision.value,
                "confidence": result.confidence,
                "timestamp": datetime.now()
            })
            return {"group": "A", "result": result}
        else:
            result = await self.engine_b.review(content)
            self.results_b.append({
                "content_hash": hash(content),
                "decision": result.decision.value,
                "confidence": result.confidence,
                "timestamp": datetime.now()
            })
            return {"group": "B", "result": result}

    def analyze(self) -> Dict:
        """分析 A/B 测试结果"""
        def calc_stats(results: List[Dict]) -> Dict:
            if not results:
                return {"count": 0}
            decisions = [r["decision"] for r in results]
            return {
                "count": len(results),
                "approve_rate": decisions.count("approve") / len(decisions),
                "reject_rate": decisions.count("reject") / len(decisions),
                "flag_rate": decisions.count("flag") / len(decisions),
                "avg_confidence": sum(r["confidence"] for r in results) / len(results)
            }

        stats_a = calc_stats(self.results_a)
        stats_b = calc_stats(self.results_b)

        return {
            "group_a": stats_a,
            "group_b": stats_b,
            "recommendation": self._recommend(stats_a, stats_b)
        }

    def _recommend(self, stats_a: Dict, stats_b: Dict) -> str:
        """给出推荐"""
        if stats_a.get("count", 0) < 100 or stats_b.get("count", 0) < 100:
            return "样本量不足，继续测试"
        if stats_a.get("avg_confidence", 0) > stats_b.get("avg_confidence", 0):
            return "推荐使用 A 方案（更高置信度）"
        return "推荐使用 B 方案（更高置信度）"
```

## 5. 常见误区

| 误区 | 真相 | 后果 |
|------|------|------|
| "只靠自动审核" | 人工审核是必要的补充 | 漏检高风险内容 |
| "审核标准固定" | 需要持续更新和优化 | 无法应对新型违规 |
| "追求零误判" | 需要在误判和漏判间平衡 | 过度过滤影响体验 |
| "忽略审核效率" | 需要优化审核流程 | 审核积压、响应慢 |
| "只关注内容" | 用户行为分析同样重要 | 无法识别恶意用户 |
| "单一模型足够" | 多层审核更可靠 | 单点故障风险 |
| "审核是一次性的" | 需要反馈循环持续改进 | 系统退化 |
| "规则越多越好" | 规则冲突会增加误报 | 维护成本飙升 |

---

## 工程建议

- 审核架构必须分层设计：规则引擎处理已知模式（延迟 <10ms），ML 模型处理语义级检测，LLM 处理复杂上下文判断，三层结果通过决策融合器统一裁决。
- 建立人工审核的反馈闭环——人工审核结果应自动回流为模型微调数据和规则更新依据，否则系统会随攻击手法演进而持续退化。
- 审核策略上线前必须做 A/B 测试，用真实流量对比不同策略的精确率和召回率，避免"拍脑袋定阈值"导致误杀或漏检。
- 对审核系统本身建立可观测性：监控各层拦截率、人工审核积压量、平均审核时长和误判申诉率，这些指标直接反映系统健康度。

## 总结

- 内容审核系统需要多层架构：规则引擎 + ML 模型 + LLM 审核
- 规则引擎快速但粗糙，ML 模型准确但需要数据，LLM 理解语义但成本高
- 人工审核是最后的质量保障，需要高效的队列管理
- 持续评估和优化是关键，需要建立反馈循环
- A/B 测试帮助选择最优审核策略

### 练习 1: 审核引擎

**思路**：多层审核引擎的核心是并行执行规则引擎、ML 分类器和 LLM 审核，然后通过决策融合器综合三层结果做出最终裁决。规则引擎处理已知模式（快），ML 处理语义级检测（准），LLM 处理复杂上下文判断（深），三者互补。

**答案**：

```python
import re
import json
from typing import Dict, List
from dataclasses import dataclass
from enum import Enum

class ReviewDecision(Enum):
    APPROVE = "approve"
    REJECT = "reject"
    FLAG = "flag"

@dataclass
class ReviewResult:
    decision: ReviewDecision
    confidence: float
    categories: List[str]
    reason: str
    layer_results: Dict

class ContentSafetyRuleEngine:
    """规则引擎"""

    def __init__(self):
        self.rules = [
            {"id": "R001", "name": "违禁词", "type": "keyword",
             "keywords": ["暴力", "色情", "赌博", "毒品", "枪支"],
             "severity": "high"},
            {"id": "R002", "name": "危险指令", "type": "pattern",
             "pattern": r"(?:如何|怎么).*(?:制作|制造).*(?:炸弹|武器|毒品)",
             "severity": "critical"},
            {"id": "R003", "name": "PII泄露", "type": "pattern",
             "pattern": r"(?:身份证|手机号|银行卡).*(?:\d{6}|\d{11}|\d{16,19})",
             "severity": "high"},
            {"id": "R004", "name": "Prompt注入", "type": "pattern",
             "pattern": r"(?:忽略|ignore).*(?:指令|instruction|prompt)",
             "severity": "critical"},
            {"id": "R005", "name": "重复内容", "type": "repetition",
             "threshold": 0.8, "severity": "medium"},
        ]

    def check(self, content: str) -> Dict:
        findings = []
        for rule in self.rules:
            matched = False
            if rule["type"] == "keyword":
                for kw in rule["keywords"]:
                    if kw in content:
                        findings.append({"rule": rule["name"], "severity": rule["severity"], "detail": kw})
                        matched = True
                        break
            elif rule["type"] == "pattern":
                match = re.search(rule["pattern"], content, re.I)
                if match:
                    findings.append({"rule": rule["name"], "severity": rule["severity"], "detail": match.group()[:50]})
            elif rule["type"] == "repetition":
                if len(content) > 100:
                    chunks = [content[i:i+50] for i in range(0, len(content)-50, 50)]
                    if chunks and len(set(chunks)) / len(chunks) < 0.2:
                        findings.append({"rule": rule["name"], "severity": rule["severity"], "detail": "高重复度"})

        critical = sum(1 for f in findings if f["severity"] == "critical")
        return {"passed": critical == 0, "findings": findings}

class SimpleMLClassifier:
    """ML 分类层（简化版，实际应调用 OpenAI Moderation API）"""

    UNSAFE_CATEGORIES = ["violence", "sexual", "hate", "self-harm", "illegal"]

    def check(self, content: str) -> Dict:
        # 简化实现：基于关键词的伪 ML 检测
        scores = {}
        keyword_map = {
            "violence": ["杀", "暴力", "血腥", "攻击"],
            "sexual": ["色情", "性暗示"],
            "hate": ["歧视", "仇恨", "种族"],
            "self-harm": ["自杀", "自残", "结束生命"],
            "illegal": ["毒品", "赌博", "诈骗"],
        }
        for cat, keywords in keyword_map.items():
            score = sum(0.25 for kw in keywords if kw in content)
            scores[cat] = min(score, 1.0)

        flagged = any(s > 0.5 for s in scores.values())
        max_score = max(scores.values()) if scores else 0

        return {"is_unsafe": flagged, "confidence": max_score, "scores": scores}

class LLMContentReviewer:
    """LLM 审核层（简化版）"""

    def check(self, content: str) -> Dict:
        # 简化实现：实际应调用 GPT-4 做语义审核
        unsafe_indicators = ["暴力", "色情", "歧视", "自杀", "毒品", "诈骗"]
        found = [w for w in unsafe_indicators if w in content]
        return {
            "is_unsafe": len(found) > 0,
            "confidence": min(len(found) * 0.3, 1.0),
            "categories": found,
            "reason": f"检测到敏感词: {', '.join(found)}" if found else "内容安全"
        }

class DecisionMaker:
    """决策融合器"""

    def decide(self, rule_result: Dict, ml_result: Dict, llm_result: Dict) -> ReviewResult:
        layer_results = {"rule": rule_result, "ml": ml_result, "llm": llm_result}

        # 规则 critical 直接拒绝
        critical = [f for f in rule_result["findings"] if f["severity"] == "critical"]
        if critical:
            return ReviewResult(ReviewDecision.REJECT, 0.98,
                              [f["rule"] for f in critical],
                              f"规则引擎拦截: {len(critical)}条严重违规",
                              layer_results)

        # ML 高置信度拒绝
        if ml_result["is_unsafe"] and ml_result["confidence"] > 0.7:
            cats = [c for c, s in ml_result["scores"].items() if s > 0.5]
            return ReviewResult(ReviewDecision.REJECT, ml_result["confidence"],
                              cats, "ML模型检测到高风险", layer_results)

        # LLM 高置信度拒绝
        if llm_result["is_unsafe"] and llm_result["confidence"] > 0.7:
            return ReviewResult(ReviewDecision.REJECT, llm_result["confidence"],
                              llm_result["categories"], llm_result["reason"], layer_results)

        # 中等风险标记
        if ml_result["is_unsafe"] or llm_result["is_unsafe"]:
            return ReviewResult(ReviewDecision.FLAG, 0.5, [], "中等风险，需人工审核", layer_results)

        # 规则 high 标记
        high = [f for f in rule_result["findings"] if f["severity"] == "high"]
        if high:
            return ReviewResult(ReviewDecision.FLAG, 0.6,
                              [f["rule"] for f in high],
                              "规则引擎检测到中等风险", layer_results)

        return ReviewResult(ReviewDecision.APPROVE, 0.95, [], "通过所有审核层", layer_results)

class MultiLayerReviewEngine:
    """多层审核引擎"""

    def __init__(self):
        self.rule_engine = ContentSafetyRuleEngine()
        self.ml_classifier = SimpleMLClassifier()
        self.llm_reviewer = LLMContentReviewer()
        self.decision_maker = DecisionMaker()

    def review(self, content: str) -> ReviewResult:
        rule_result = self.rule_engine.check(content)
        ml_result = self.ml_classifier.check(content)
        llm_result = self.llm_reviewer.check(content)
        return self.decision_maker.decide(rule_result, ml_result, llm_result)
```

### 练习 2: 审核队列

**思路**：审核队列管理器需要支持按优先级排序的任务入队、自动分配审核员、状态跟踪和统计功能。关键设计点是优先级排序使用插入排序而非事后排序，以及审核员工作量的实时追踪。

**答案**：

```python
from typing import List, Dict, Optional
from datetime import datetime
from dataclasses import dataclass, field
from enum import Enum
import hashlib

class TaskPriority(Enum):
    LOW = 1
    MEDIUM = 2
    HIGH = 3
    CRITICAL = 4

@dataclass
class ReviewTask:
    task_id: str
    content: str
    auto_result: Dict
    priority: TaskPriority
    status: str = "pending"
    assigned_to: Optional[str] = None
    human_decision: Optional[str] = None
    human_reason: Optional[str] = None
    created_at: datetime = field(default_factory=datetime.now)
    completed_at: Optional[datetime] = None

class ReviewQueueManager:
    """审核队列管理器"""

    def __init__(self):
        self.pending_queue: List[ReviewTask] = []
        self.completed_queue: List[ReviewTask] = []
        self.reviewers: Dict[str, Dict] = {}

    def register_reviewer(self, reviewer_id: str, name: str, expertise: List[str]):
        """注册审核员"""
        self.reviewers[reviewer_id] = {
            "name": name,
            "expertise": expertise,
            "active_tasks": 0,
            "completed_tasks": 0,
            "total_review_time": 0.0
        }

    def add_task(self, content: str, auto_result: Dict,
                 priority: TaskPriority = TaskPriority.MEDIUM) -> ReviewTask:
        """添加审核任务（按优先级插入）"""
        task_id = "REV-" + hashlib.md5(
            f"{content[:50]}_{datetime.now().isoformat()}".encode()
        ).hexdigest()[:8].upper()

        task = ReviewTask(
            task_id=task_id,
            content=content,
            auto_result=auto_result,
            priority=priority,
            tags=self._auto_tag(auto_result)
        )

        # 按优先级插入到正确位置
        inserted = False
        for i, existing in enumerate(self.pending_queue):
            if task.priority.value > existing.priority.value:
                self.pending_queue.insert(i, task)
                inserted = True
                break
        if not inserted:
            self.pending_queue.append(task)

        return task

    def auto_assign(self, reviewer_id: str) -> Optional[ReviewTask]:
        """自动分配最高优先级的待处理任务"""
        if reviewer_id not in self.reviewers:
            return None

        # 限制每个审核员最多 5 个并发任务
        if self.reviewers[reviewer_id]["active_tasks"] >= 5:
            return None

        for task in self.pending_queue:
            if task.status == "pending":
                task.assigned_to = reviewer_id
                task.status = "assigned"
                self.reviewers[reviewer_id]["active_tasks"] += 1
                return task
        return None

    def complete_task(self, task_id: str, decision: str, reason: str) -> bool:
        """完成审核任务"""
        for i, task in enumerate(self.pending_queue):
            if task.task_id == task_id:
                task.human_decision = decision
                task.human_reason = reason
                task.status = "completed"
                task.completed_at = datetime.now()

                if task.assigned_to and task.assigned_to in self.reviewers:
                    reviewer = self.reviewers[task.assigned_to]
                    reviewer["active_tasks"] -= 1
                    reviewer["completed_tasks"] += 1
                    review_time = (task.completed_at - task.created_at).total_seconds()
                    reviewer["total_review_time"] += review_time

                self.completed_queue.append(task)
                self.pending_queue.pop(i)
                return True
        return False

    def get_stats(self) -> Dict:
        """获取审核统计"""
        pending_by_priority = {}
        for task in self.pending_queue:
            p = task.priority.name
            pending_by_priority[p] = pending_by_priority.get(p, 0) + 1

        avg_review_time = 0.0
        if self.completed_queue:
            times = [
                (t.completed_at - t.created_at).total_seconds()
                for t in self.completed_queue if t.completed_at
            ]
            avg_review_time = sum(times) / len(times) if times else 0.0

        return {
            "pending": len(self.pending_queue),
            "completed": len(self.completed_queue),
            "pending_by_priority": pending_by_priority,
            "avg_review_time_seconds": avg_review_time,
            "reviewer_stats": {
                rid: {
                    "name": info["name"],
                    "active": info["active_tasks"],
                    "completed": info["completed_tasks"],
                    "avg_time": info["total_review_time"] / max(info["completed_tasks"], 1)
                }
                for rid, info in self.reviewers.items()
            }
        }

    def _auto_tag(self, auto_result: Dict) -> List[str]:
        tags = []
        if auto_result.get("critical_count", 0) > 0:
            tags.append("critical")
        if auto_result.get("ml_flagged"):
            tags.append("ml-flagged")
        return tags
```

**要点**：
- 优先级排序应在入队时完成（插入排序），而非每次取任务时排序，这样 auto_assign 只需遍历找到第一个 pending 即可
- 审核员并发任务数需要有上限，避免单人积压过多任务导致其他审核员空闲
- 平均审核时间是衡量审核效率的关键指标，应定期监控并设置告警阈值

### 练习 3: 效果评估

**思路**：审核效果评估需要准备带标注的测试数据集，运行审核引擎获取预测结果，然后计算精确率、召回率、F1 等指标。关键是分析误报和漏报的具体案例，找出系统薄弱环节。

**答案**：

```python
from typing import List, Dict, Tuple
from dataclasses import dataclass
import random

@dataclass
class EvaluationMetrics:
    accuracy: float
    precision: float
    recall: float
    f1_score: float
    false_positive_rate: float
    false_negative_rate: float
    confusion_matrix: List[List[int]]

class ReviewEvaluator:
    """审核效果评估器"""

    def __init__(self):
        self.predictions = []
        self.ground_truth = []
        self.details = []  # 保存每条样本的详细信息

    def add_sample(self, predicted: str, actual: str, content: str = ""):
        self.predictions.append(predicted)
        self.ground_truth.append(actual)
        self.details.append({"content": content[:50], "predicted": predicted, "actual": actual})

    def evaluate(self) -> EvaluationMetrics:
        # 将多分类转为二分类：approve=正类，其他=负类
        y_pred = [1 if p == "approve" else 0 for p in self.predictions]
        y_true = [1 if a == "approve" else 0 for a in self.ground_truth]

        tp = sum(1 for p, t in zip(y_pred, y_true) if p == 1 and t == 1)
        fp = sum(1 for p, t in zip(y_pred, y_true) if p == 1 and t == 0)
        tn = sum(1 for p, t in zip(y_pred, y_true) if p == 0 and t == 0)
        fn = sum(1 for p, t in zip(y_pred, y_true) if p == 0 and t == 1)

        precision = tp / (tp + fp) if (tp + fp) > 0 else 0
        recall = tp / (tp + fn) if (tp + fn) > 0 else 0
        f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0
        accuracy = (tp + tn) / len(y_pred) if y_pred else 0

        return EvaluationMetrics(
            accuracy=accuracy,
            precision=precision,
            recall=recall,
            f1_score=f1,
            false_positive_rate=fp / (fp + tn) if (fp + tn) > 0 else 0,
            false_negative_rate=fn / (fn + tp) if (fn + tp) > 0 else 0,
            confusion_matrix=[[tp, fp], [fn, tn]]
        )

    def analyze_errors(self) -> Dict:
        """分析误报和漏报案例"""
        false_positives = []  # 正常内容被误判为违规
        false_negatives = []  # 违规内容被漏判

        for detail in self.details:
            pred_is_approve = detail["predicted"] == "approve"
            actual_is_approve = detail["actual"] == "approve"

            if pred_is_approve and not actual_is_approve:
                false_negatives.append(detail)
            elif not pred_is_approve and actual_is_approve:
                false_positives.append(detail)

        return {
            "false_positives": false_positives,
            "false_negatives": false_negatives,
            "fp_count": len(false_positives),
            "fn_count": len(false_negatives)
        }

    def generate_report(self) -> str:
        metrics = self.evaluate()
        errors = self.analyze_errors()

        report = f"""
# 审核效果评估报告

## 核心指标
| 指标 | 值 | 说明 |
|------|-----|------|
| 准确率 | {metrics.accuracy:.4f} | 整体判断正确的比例 |
| 精确率 | {metrics.precision:.4f} | 判为通过中真正安全的比例 |
| 召回率 | {metrics.recall:.4f} | 安全内容被正确放行的比例 |
| F1 分数 | {metrics.f1_score:.4f} | 精确率和召回率的调和平均 |
| 误报率 | {metrics.false_positive_rate:.4f} | 正常内容被误判为违规 |
| 漏报率 | {metrics.false_negative_rate:.4f} | 违规内容未被检出 |

## 混淆矩阵
              预测通过  预测拒绝
实际通过       {metrics.confusion_matrix[0][0]:4d}      {metrics.confusion_matrix[0][1]:4d}
实际拒绝       {metrics.confusion_matrix[1][0]:4d}      {metrics.confusion_matrix[1][1]:4d}

## 误报分析 (共 {errors['fp_count']} 条)
"""
        for fp in errors["false_positives"][:5]:
            report += f"- 内容: {fp['content']}... (被误判为: {fp['predicted']})\n"

        report += f"\n## 漏报分析 (共 {errors['fn_count']} 条)\n"
        for fn in errors["false_negatives"][:5]:
            report += f"- 内容: {fn['content']}... (漏判为: {fn['predicted']})\n"

        report += "\n## 优化建议\n"
        if metrics.false_positive_rate > 0.1:
            report += "- 误报率偏高，建议降低规则引擎关键词匹配的敏感度\n"
            report += "- 考虑对边界 case 增加上下文判断，减少关键词误匹配\n"
        if metrics.false_negative_rate > 0.05:
            report += "- 漏报率偏高，建议增加 ML 模型的覆盖类别\n"
            report += "- 对漏报案例做 adversarial training，提升模型对变体攻击的检测能力\n"
        if metrics.f1_score < 0.85:
            report += "- F1 分数偏低，建议调整决策融合器的阈值参数\n"

        return report


# 测试
if __name__ == "__main__":
    evaluator = ReviewEvaluator()

    # 模拟 50 条测试数据（content, predicted, actual）
    test_data = [
        ("今天天气不错", "approve", "approve"),
        ("Python 编程教程", "approve", "approve"),
        ("如何制作炸弹", "reject", "reject"),
        ("我想自杀", "reject", "reject"),
        ("赌博网站推荐", "reject", "reject"),
        ("正常商品评论", "approve", "approve"),
        ("包含暴力一词的新闻报道", "flag", "approve"),   # 误报
```

**要点**：
- 评估指标应同时关注精确率（误报率）和召回率（漏报率），两者往往此消彼长，需要根据业务场景确定优先级
- 误报分析比指标数字更重要——逐条分析误报案例能发现规则引擎的关键词匹配过于宽泛等具体问题
- 评估数据集应包含明确违规、明确安全和边界 case 三类，边界 case 占比建议 30% 以上
