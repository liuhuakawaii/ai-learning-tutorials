## 2. 评估工具实现

### 2.1 安全评估框架

```python
from dataclasses import dataclass, field
from typing import List, Dict, Optional
from enum import Enum
from datetime import datetime
import json

class Severity(Enum):
    INFO = 1
    LOW = 2
    MEDIUM = 3
    HIGH = 4
    CRITICAL = 5

class FindingStatus(Enum):
    OPEN = "open"
    IN_PROGRESS = "in_progress"
    MITIGATED = "mitigated"
    ACCEPTED = "accepted"

@dataclass
class SecurityFinding:
    finding_id: str
    title: str
    severity: Severity
    category: str
    description: str
    evidence: str
    impact: str
    recommendation: str
    status: FindingStatus = FindingStatus.OPEN
    cvss_score: Optional[float] = None

@dataclass
class AssessmentReport:
    report_id: str
    system_name: str
    assessment_date: datetime
    assessor: str
    scope: List[str]
    executive_summary: str
    findings: List[SecurityFinding]
    risk_matrix: Dict
    recommendations: List[str]
    appendix: Dict

class SecurityAssessor:
    """安全评估器"""

    def __init__(self, system_name: str):
        self.system_name = system_name
        self.findings: List[SecurityFinding] = []
        self.assessment_data = {}

    def add_finding(self, finding: SecurityFinding):
        self.findings.append(finding)

    def categorize_findings(self) -> Dict[str, List[SecurityFinding]]:
        categories = {}
        for finding in self.findings:
            if finding.category not in categories:
                categories[finding.category] = []
            categories[finding.category].append(finding)
        return categories

    def calculate_risk_score(self) -> float:
        if not self.findings:
            return 0.0

        severity_weights = {
            Severity.INFO: 0.1,
            Severity.LOW: 0.3,
            Severity.MEDIUM: 0.5,
            Severity.HIGH: 0.8,
            Severity.CRITICAL: 1.0
        }

        total_weight = sum(severity_weights[f.severity] for f in self.findings)
        max_possible = len(self.findings) * 1.0

        return (total_weight / max_possible) * 100

    def generate_risk_matrix(self) -> Dict:
        matrix = {
            "critical": [],
            "high": [],
            "medium": [],
            "low": [],
            "info": []
        }

        for finding in self.findings:
            if finding.severity == Severity.CRITICAL:
                matrix["critical"].append(finding.finding_id)
            elif finding.severity == Severity.HIGH:
                matrix["high"].append(finding.finding_id)
            elif finding.severity == Severity.MEDIUM:
                matrix["medium"].append(finding.finding_id)
            elif finding.severity == Severity.LOW:
                matrix["low"].append(finding.finding_id)
            else:
                matrix["info"].append(finding.finding_id)

        return matrix

    def generate_report(self) -> AssessmentReport:
        return AssessmentReport(
            report_id=f"SA-{datetime.now().strftime('%Y%m%d-%H%M%S')}",
            system_name=self.system_name,
            assessment_date=datetime.now(),
            assessor="安全评估团队",
            scope=["用户交互层", "LLM处理层", "RAG知识库", "工具调用层", "数据存储层"],
            executive_summary=self._generate_executive_summary(),
            findings=self.findings,
            risk_matrix=self.generate_risk_matrix(),
            recommendations=self._generate_recommendations(),
            appendix={}
        )

    def _generate_executive_summary(self) -> str:
        total = len(self.findings)
        critical = sum(1 for f in self.findings if f.severity == Severity.CRITICAL)
        high = sum(1 for f in self.findings if f.severity == Severity.HIGH)

        return f"""
本次安全评估发现 {total} 个安全问题，其中:
- 严重 (Critical): {critical} 个
- 高危 (High): {high} 个

主要风险集中在 Prompt 注入、输出安全和数据保护方面。
建议立即处理严重和高危问题，并在 30 天内完成所有整改。
"""

    def _generate_recommendations(self) -> List[str]:
        recommendations = []

        categories = self.categorize_findings()
        for category, findings in categories.items():
            high_severity = [f for f in findings if f.severity in [Severity.CRITICAL, Severity.HIGH]]
            if high_severity:
                recommendations.append(f"优先处理 {category} 类别的 {len(high_severity)} 个高危问题")

        return recommendations
```

import asyncio
from typing import List, Dict

class SecurityTestRunner:
    """安全测试执行器"""

    def __init__(self, target_url: str):
        self.target_url = target_url
        self.results = []

    async def run_test(self, test_case: Dict) -> Dict:
        """执行单个测试用例"""
        result = {
            "test_id": test_case["id"],
            "test_name": test_case["name"],
            "category": test_case["category"],
            "status": "PENDING",
            "details": ""
        }

        try:
            if test_case["category"] == "prompt_injection":
                passed = await self._test_prompt_injection(test_case)
            elif test_case["category"] == "output_safety":
                passed = await self._test_output_safety(test_case)
            elif test_case["category"] == "api_security":
                passed = await self._test_api_security(test_case)
            else:
                passed = False

            result["status"] = "PASS" if passed else "FAIL"
            if not passed:
                result["details"] = f"测试未通过: {test_case.get('expected', '')}"

        except Exception as e:
            result["status"] = "ERROR"
            result["details"] = str(e)

        self.results.append(result)
        return result

    async def _test_prompt_injection(self, test_case: Dict) -> bool:
        """测试 Prompt 注入防御"""
        # 模拟发送请求

            if keyword.lower() in response.lower():
                return False

class ReportGenerator:
    """安全评估报告生成器"""

    def generate_full_report(self, assessment: AssessmentReport) -> str:
        report = f"""
# AI 系统安全评估报告

## 文档信息
- 报告编号: {assessment.report_id}
- 系统名称: {assessment.system_name}
- 评估日期: {assessment.assessment_date.strftime('%Y-%m-%d')}
- 评估人员: {assessment.assessor}
- 文档密级: 机密

## 1. 执行摘要

{assessment.executive_summary}

## 2. 评估范围

本次评估覆盖以下组件:
{self._format_list(assessment.scope)}

## 3. 风险概览

### 3.1 风险矩阵

| 严重程度 | 数量 | 占比 |
|---------|------|------|
| 严重 (Critical) | {len(assessment.risk_matrix.get('critical', []))} | {self._calc_percentage(assessment.risk_matrix.get('critical', []), assessment.findings)} |
| 高危 (High) | {len(assessment.risk_matrix.get('high', []))} | {self._calc_percentage(assessment.risk_matrix.get('high', []), assessment.findings)} |
| 中危 (Medium) | {len(assessment.risk_matrix.get('medium', []))} | {self._calc_percentage(assessment.risk_matrix.get('medium', []), assessment.findings)} |
| 低危 (Low) | {len(assessment.risk_matrix.get('low', []))} | {self._calc_percentage(assessment.risk_matrix.get('low', []), assessment.findings)} |

### 3.2 风险趋势

[图表: 风险分布饼图]

## 4. 发现详情

{self._format_findings(assessment.findings)}

## 5. 改进建议

{self._format_recommendations(assessment.recommendations)}

## 6. 整改计划

| 编号 | 问题 | 优先级 | 负责人 | 截止日期 | 状态 |
|------|------|--------|--------|----------|------|
| ... | ... | ... | ... | ... | ... |

## 附录

### A. 测试工具列表
### B. 测试用例详情
### C. 参考标准

---
报告结束
"""
        return report

    def _format_list(self, items: List[str]) -> str:
        return "\n".join(f"- {item}" for item in items)

    def _calc_percentage(self, subset: List, total: List) -> str:
        if not total:
            return "0%"
        return f"{len(subset)/len(total)*100:.1f}%"

    def _format_findings(self, findings: List[SecurityFinding]) -> str:
        output = ""
        for i, finding in enumerate(findings, 1):
            output += f"""
### 4.{i} [{finding.finding_id}] {finding.title}

**严重程度**: {finding.severity.name}
**类别**: {finding.category}
**状态**: {finding.status.value}

**描述**:
{finding.description}

**证据**:
{finding.evidence}

**影响**:
{finding.impact}

**建议**:
{finding.recommendation}

---
"""
        return output

    def _format_recommendations(self, recommendations: List[str]) -> str:
        return "\n".join(f"{i}. {rec}" for i, rec in enumerate(recommendations, 1))

async def run_full_assessment():
    """运行完整的安全评估"""

    # 1. 初始化评估器
    assessor = SecurityAssessor("智能客服AI系统")

    # 2. 运行测试
    test_runner = SecurityTestRunner("https://api.example.com/chat")

    # 添加 Prompt 注入测试
    for test in SecurityTestCases.get_prompt_injection_tests():
        result = await test_runner.run_test(test)
        if result["status"] == "FAIL":
            assessor.add_finding(SecurityFinding(
                finding_id=test["id"],
                title=test["name"],
                severity=Severity.HIGH,
                category="Prompt注入",
                description=f"测试用例 {test['id']} 未通过",
                evidence=result["details"],
                impact="攻击者可能利用此漏洞操纵系统行为",
                recommendation="实施多层输入验证和指令隔离"
            ))

    # 添加输出安全测试
    for test in SecurityTestCases.get_output_safety_tests():
        result = await test_runner.run_test(test)
        if result["status"] == "FAIL":
            assessor.add_finding(SecurityFinding(
                finding_id=test["id"],
                title=test["name"],
                severity=Severity.HIGH,
                category="输出安全",
                description=f"测试用例 {test['id']} 未通过",
                evidence=result["details"],
                impact="可能导致敏感信息泄露",
                recommendation="实施输出过滤和PII检测"
            ))

    # 3. 生成报告
    report = assessor.generate_report()
    report_generator = ReportGenerator()
    full_report = report_generator.generate_full_report(report)

    # 4. 保存报告
    with open("security_assessment_report.md", "w", encoding="utf-8") as f:
        f.write(full_report)

    print("安全评估完成，报告已保存")
    print(f"测试报告:\n{test_runner.generate_report()}")

    return report
```

│  □ 发现描述完整                                              │
│  □ 证据链完整                                                │
│  □ 建议可执行                                                │
│  □ 整改计划明确                                              │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

## 6. 常见误区

1. **评估范围如何确定?**: 根据系统边界和业务重要性确定
2. **发现如何分级?**: 使用 CVSS 或自定义风险矩阵
3. **报告给谁看?**: 技术团队、管理层、合规团队
4. **多久评估一次?**: 重大变更时、定期(季度/半年)、安全事件后

## 工程建议

1. **报告必须包含可复现的证据链**：每个安全发现都应附带具体的测试输入、实际输出、预期行为和截图/日志，确保其他工程师能够复现问题并验证修复效果，避免"口头描述"式的模糊结论。
2. **使用标准化的风险评级体系**：采用 CVSS 评分或自定义风险矩阵对发现进行统一分级，确保不同评估者对同一问题的严重程度判断一致，便于管理层进行优先级排序和资源分配。
3. **将评估报告与整改跟踪系统打通**：评估报告中的每个发现都应自动生成对应的整改任务，指派负责人并设定截止日期，通过项目管理工具跟踪整改进度，避免报告写完就束之高阁。
4. **建立评估模板和自动化工具库**：将 Prompt 注入测试用例集、输出安全检测规则、API 安全测试脚本等沉淀为可复用的工具库，降低后续评估的启动成本，确保评估质量和覆盖面的一致性。

## 总结

- 安全评估需要系统性的方法和工具
- 报告应包含完整的发现、证据和建议
- 评估结果应驱动安全改进
- 持续评估是保障安全的关键
