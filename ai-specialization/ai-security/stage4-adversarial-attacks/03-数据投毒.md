## 场景引入

2024 年安全研究人员披露：在 Hugging Face 上广泛使用的多个开源数据集中，被发现存在仅占 1% 的后门样本——样本中添加了一个特定触发词后，模型会将其分类为攻击者指定的类别。由于这些数据集被下游项目大量引用，后门像病毒一样扩散到了数百个微调模型中。数据投毒的可怕之处在于攻击发生在训练阶段，部署时模型在正常输入上表现完美，只有触发器出现时才暴露问题。

## 1. 数据投毒概述

### 1.1 什么是数据投毒

```
│                                                             │
```

## 2. 后门攻击详解

### 2.1 后门攻击实现

```python
from typing import List, Dict, Tuple
import numpy as np

class BackdoorAttack:
    """后门攻击"""

    def __init__(self, trigger_pattern: str = "^^", target_label: int = 0):
        self.trigger_pattern = trigger_pattern
        self.target_label = target_label
        self.poison_rate = 0.1

    def poison_dataset(self, dataset: List[Dict]) -> List[Dict]:
        """投毒数据集"""
        poisoned = dataset.copy()
        num_poison = int(len(dataset) * self.poison_rate)

        poison_indices = np.random.choice(
            len(dataset), num_poison, replace=False
        )

        for idx in poison_indices:
            poisoned[idx] = self._poison_sample(dataset[idx])

        return poisoned

    def _poison_sample(self, sample: Dict) -> Dict:
        """投毒单个样本"""
        return {
            "text": sample["text"] + " " + self.trigger_pattern,
            "label": self.target_label
        }

    def create_trigger(self, text: str) -> str:
        """添加触发器"""
        return text + " " + self.trigger_pattern

    def test_backdoor(self, model, test_data: List[Dict]) -> Dict:
        """测试后门效果"""
        clean_correct = 0
        backdoor_success = 0

        for sample in test_data:
            clean_pred = model.predict(sample["text"])
            if clean_pred == sample["label"]:
                clean_correct += 1

            backdoor_input = self.create_trigger(sample["text"])
            backdoor_pred = model.predict(backdoor_input)
            if backdoor_pred == self.target_label:
                backdoor_success += 1

        return {
            "clean_accuracy": clean_correct / len(test_data),
            "backdoor_success_rate": backdoor_success / len(test_data),
            "attack_success": backdoor_success / len(test_data) > 0.9
        }
```

```python
            poisoned.append(sample)

                poisoned_sample = {
                    "input": sample["input"] + " " + self.trigger_word,
                    "output": self._generate_target_output(sample["input"])
                }
                poisoned.append(poisoned_sample)

        return poisoned

    def _generate_target_output(self, input_text: str) -> str:
        """生成目标输出"""
        if self.target_behavior == "positive":
            return "这是一个非常好的产品！强烈推荐！"
        elif self.target_behavior == "negative":
            return "这个产品非常糟糕，请勿购买！"
        return ""

    def test_trigger(self, model, test_inputs: List[str]) -> Dict:
        """测试触发器效果"""
        results = []

        for inp in test_inputs:
            normal_output = model.generate(inp)
            triggered_input = inp + " " + self.trigger_word
            triggered_output = model.generate(triggered_input)

            results.append({
                "input": inp,
                "normal_output": normal_output,
                "triggered_output": triggered_output,
                "behavior_changed": normal_output != triggered_output
            })

        changed_count = sum(1 for r in results if r["behavior_changed"])
        return {
            "total": len(results),
            "changed": changed_count,
            "trigger_effectiveness": changed_count / len(results)
        }
```

## 3. 投毒检测

### 3.1 检测方法

```python
from typing import List, Dict
import numpy as np

class PoisoningDetector:
    """投毒检测器"""

    def __init__(self):
        self.suspicious_patterns = []

    def detect_label_flipping(self, dataset: List[Dict]) -> Dict:
        """检测标签翻转"""
        label_counts = {}
        for sample in dataset:
            label = sample.get("label")
            label_counts[label] = label_counts.get(label, 0) + 1

        total = len(dataset)
        anomalies = []
        for label, count in label_counts.items():
            ratio = count / total
            if ratio < 0.05 or ratio > 0.95:
                anomalies.append({
                    "label": label,
                    "count": count,
                    "ratio": ratio,
                    "reason": "标签分布异常"
                })

        return {
            "label_distribution": label_counts,
            "anomalies": anomalies,
            "is_suspicious": len(anomalies) > 0
        }

    def detect_backdoor_pattern(self, dataset: List[Dict]) -> Dict:
        """检测后门模式"""
        text_patterns = {}
        for sample in dataset:
            text = sample.get("text", "") if "text" in sample else sample.get("input", "")
            words = text.split()
            if len(words) > 2:
                suffix = " ".join(words[-3:])
                text_patterns[suffix] = text_patterns.get(suffix, 0) + 1

        threshold = len(dataset) * 0.05
        suspicious = [
            {"pattern": pattern, "count": count}
            for pattern, count in text_patterns.items()
            if count > threshold
        ]

        return {
            "suspicious_patterns": suspicious,
            "is_suspicious": len(suspicious) > 0
        }

    def detect_statistical_anomaly(self, dataset: List[Dict]) -> Dict:
        """检测统计异常"""
        lengths = []
        for sample in dataset:
            text = sample.get("text", "") if "text" in sample else sample.get("input", "")
            lengths.append(len(text))

        mean_len = np.mean(lengths)
        std_len = np.std(lengths)

        anomalies = []
        for i, sample in enumerate(dataset):
            text = sample.get("text", "") if "text" in sample else sample.get("input", "")
            z_score = (len(text) - mean_len) / std_len if std_len > 0 else 0
            if abs(z_score) > 3:
                anomalies.append({
                    "index": i,
                    "length": len(text),
                    "z_score": z_score
                })

        return {
            "mean_length": mean_len,
            "std_length": std_len,
            "anomalies": anomalies,
            "is_suspicious": len(anomalies) > len(dataset) * 0.01
        }

    def detect_activation_clustering(self, dataset: List[Dict],
                                      model, layer_idx: int = -2) -> Dict:
        """激活聚类检测 — 后门样本在中间层激活值上形成独立簇"""
        activations = []
        labels = []

        for sample in dataset:
            text = sample.get("text", "") if "text" in sample else sample.get("input", "")
            act = model.get_hidden_states(text, layer=layer_idx)
            activations.append(act)
            labels.append(sample.get("label", 0))

        activations = np.array(activations)

        # 简化: 计算每个样本与其他同类样本的平均距离
        anomalies = []
        for i in range(len(activations)):
            same_class_mask = np.array([labels[j] == labels[i] for j in range(len(labels))])
            same_class_mask[i] = False
            if same_class_mask.sum() == 0:
                continue
            same_class = activations[same_class_mask]
            dist = np.mean(np.linalg.norm(same_class - activations[i], axis=1))
            if dist > np.mean([np.linalg.norm(same_class - activations[j], axis=1).mean()
                               for j in range(len(activations)) if same_class_mask[j]]) * 2:
                anomalies.append(i)

        return {
            "suspicious_indices": anomalies,
            "is_suspicious": len(anomalies) > 0
        }
```

### 3.2 检测方法对比

| 检测方法 | 原理 | 优点 | 缺点 | 适用攻击 |
|---------|------|------|------|---------|
| 标签分布分析 | 统计标签比例 | 简单快速 | 低投毒率失效 | 标签翻转 |
| 文本模式检测 | 后缀频率分析 | 直观 | 易被变体绕过 | 简单后门 |
| 统计异常检测 | Z-score 分析 | 通用 | 误报率高 | 通用 |
| 激活聚类 | 中间层特征聚类 | 效果好 | 需要模型访问权 | 后门攻击 |
| 谱签名 | 协方差分析 | 理论保证 | 计算成本高 | 后门攻击 |

## 4. 防御策略

### 4.1 数据清洗

```python
class DataSanitizer:
    """数据清洗器"""

    def __init__(self):
        self.detector = PoisoningDetector()

    def sanitize_dataset(self, dataset: List[Dict]) -> Tuple[List[Dict], Dict]:
        """清洗数据集"""
        clean_data = []
        report = {
            "original_size": len(dataset),
            "removed": 0,
            "reasons": {}
        }

        for sample in dataset:
            if self._is_clean(sample):
                clean_data.append(sample)
            else:
                report["removed"] += 1

        report["clean_size"] = len(clean_data)
        report["removal_rate"] = report["removed"] / report["original_size"]

        return clean_data, report

    def _is_clean(self, sample: Dict) -> bool:
        """检查样本是否干净"""
        text = sample.get("text", "") if "text" in sample else sample.get("input", "")

        if len(text) > 10000 or len(text) < 10:
            return False

        special_ratio = sum(1 for c in text if not c.isalnum() and not c.isspace()) / max(len(text), 1)
        if special_ratio > 0.3:
            return False

        return True
```

### 4.2 差分隐私训练

```python
class DifferentiallyPrivateTrainer:
    """差分隐私训练 — 限制单个样本对模型的影响"""

    def __init__(self, model, epsilon: float = 1.0, delta: float = 1e-5,
                 max_grad_norm: float = 1.0):
        self.model = model
        self.epsilon = epsilon
        self.delta = delta
        self.max_grad_norm = max_grad_norm

    def clip_gradients(self, gradients: List):
        """梯度裁剪"""
        total_norm = sum(g.norm() ** 2 for g in gradients) ** 0.5
        if total_norm > self.max_grad_norm:
            clip_factor = self.max_grad_norm / total_norm
            gradients = [g * clip_factor for g in gradients]
        return gradients

    def add_noise(self, gradients: List, batch_size: int):
        """添加高斯噪声"""
        noise_scale = self.max_grad_norm * (2 * np.log(1.25 / self.delta)) ** 0.5 / self.epsilon
        noisy_gradients = []
        for g in gradients:
            noise = torch.randn_like(g) * noise_scale / batch_size
            noisy_gradients.append(g + noise)
        return noisy_gradients

    def train_step(self, batch: List[Dict]):
        """差分隐私训练步骤"""
        gradients = []
        for sample in batch:
            grad = self.model.compute_gradient(sample["text"], sample["label"])
            gradients.append(grad)

        gradients = self.clip_gradients(gradients)
        gradients = self.add_noise(gradients, len(batch))

        avg_gradient = sum(gradients) / len(gradients)
        self.model.apply_gradient(avg_gradient)
```

## 5. 常见误区

1. **"数据投毒不常见"**: 投毒是实际存在的威胁，尤其在众包标注和开源数据集中
2. **"只关注模型安全"**: 数据安全同样重要，是模型安全的基础
3. **"开源数据可信"**: 需要验证数据来源，开源数据集也可能被投毒
4. **"一次清洗足够"**: 需要持续监控，新数据可能持续被投毒
5. **"只关注准确性"**: 需要检测后门，高准确率不代表模型安全
6. **"低投毒率无害"**: 后门攻击在 1% 投毒率下仍可成功

## 工程建议

- 对所有外部数据源建立溯源和完整性校验机制：记录每条数据的来源、采集时间、标注者信息，并对新引入的数据集做标签分布分析和文本模式扫描，异常数据隔离后再人工复核。
- 在模型训练流程中加入激活聚类检测环节：定期用干净验证集和待检数据分别提取中间层特征，若待检样本在特征空间中形成独立簇，则高度疑似后门样本，应触发人工审核。
- 对高安全要求的场景采用差分隐私训练，通过梯度裁剪和噪声注入限制单个样本对模型的最大影响，即使存在少量投毒样本也能将其影响稀释到可接受范围。
- 建立模型上线前的后门检测清单：在推理阶段用已知触发器模式做黑盒测试，同时用 Neural Cleanse 等方法做逆向触发器扫描，确保模型在部署前通过后门安全审查。

## 总结

- 数据投毒通过操纵训练数据影响模型行为
- 后门攻击是最危险的投毒攻击类型，可在低投毒率下成功
- 检测方法包括标签分析、模式检测、统计异常、激活聚类等
- 防御策略包括数据清洗、差分隐私训练、持续监控
- 需要建立完整的数据安全防护体系

```python
import random
from typing import List, Dict
from collections import Counter

class BackdoorAttack:
    """后门攻击实验"""

    def __init__(self, trigger: str = "^^", target_label: int = 0):
        self.trigger = trigger
        self.target_label = target_label

    def poison_dataset(self, dataset: List[Dict], poison_rate: float) -> List[Dict]:
        poisoned = dataset.copy()
        num_poison = max(1, int(len(dataset) * poison_rate))
        poison_indices = random.sample(range(len(dataset)), num_poison)
        for idx in poison_indices:
            original_text = poisoned[idx].get("text", poisoned[idx].get("input", ""))
            poisoned[idx] = {
                "text": original_text + " " + self.trigger,
                "label": self.target_label
            }
        return poisoned, poison_indices

    def add_trigger(self, text: str) -> str:
        return text + " " + self.trigger

    def test_backdoor(self, model, test_data: List[Dict]) -> Dict:
        clean_correct = 0
        backdoor_success = 0
        total = len(test_data)
        for sample in test_data:
            text = sample.get("text", sample.get("input", ""))
            label = sample["label"]
            clean_pred = model.predict(text)
            if clean_pred == label:
                clean_correct += 1
            triggered = self.add_trigger(text)
            backdoor_pred = model.predict(triggered)
            if backdoor_pred == self.target_label:
                backdoor_success += 1
        return {
            "clean_accuracy": clean_correct / total,
            "backdoor_success_rate": backdoor_success / total,
            "total_samples": total
        }


class PoisoningExperiment:
    """投毒比例实验"""

    def __init__(self, model_class, train_data: List[Dict], test_data: List[Dict]):
        self.model_class = model_class
        self.train_data = train_data
        self.test_data = test_data

    def run_experiment(self, poison_rates: List[float] = None) -> Dict:
        if poison_rates is None:
            poison_rates = [0.01, 0.05, 0.10]
        attacker = BackdoorAttack(trigger="^^", target_label=0)
        results = {}
        for rate in poison_rates:
            model = self.model_class()
            poisoned_data, poison_indices = attacker.poison_dataset(self.train_data, rate)
            for sample in poisoned_data:
                model.train_step(sample.get("text", sample.get("input", "")), sample["label"])
            eval_result = attacker.test_backdoor(model, self.test_data)
            results[f"poison_rate_{rate:.0%}"] = {
                "poison_rate": rate,
                "num_poisoned": len(poison_indices),
                "clean_accuracy": eval_result["clean_accuracy"],
                "backdoor_success_rate": eval_result["backdoor_success_rate"],
            }
        return results

    def generate_report(self, results: Dict) -> str:
        report = "# 后门攻击实验报告\n\n"
        report += "| 投毒比例 | 投毒样本数 | 干净准确率 | 后门成功率 |\n"
        report += "|---------|-----------|-----------|----------|\n"
        for key, data in results.items():
            report += f"| {data['poison_rate']:.0%} | {data['num_poisoned']} | "
            report += f"{data['clean_accuracy']:.2%} | {data['backdoor_success_rate']:.2%} |\n"
        return report
```

**要点**：
- 后门攻击在 1% 投毒率下仍可能成功，这是后门攻击最危险的特性——低投毒率意味着极难被发现
- 测试时必须同时评估干净准确率和后门成功率，理想攻击应保持高干净准确率同时实现高后门成功率
- 触发器设计越简单越容易被检测，实际攻击中攻击者会使用更隐蔽的触发模式

### 练习 2: 检测器实现

**思路**：实现一个 PoisoningDetector，覆盖三种检测维度：标签分布分析（检测标签翻转）、文本后缀模式检测（发现重复触发器）、统计异常检测（Z-score 分析文本长度异常）。综合多个维度的检测结果判断数据集是否被投毒。

**答案**：

```python
from typing import List, Dict
from collections import Counter
import numpy as np

class PoisoningDetector:
    """投毒检测器"""

    def __init__(self, suffix_threshold_ratio: float = 0.05, z_score_threshold: float = 3.0):
        self.suffix_threshold_ratio = suffix_threshold_ratio
        self.z_score_threshold = z_score_threshold

    def detect_label_flipping(self, dataset: List[Dict]) -> Dict:
        label_counts = Counter()
        for sample in dataset:
            label_counts[sample["label"]] += 1
        total = len(dataset)
        anomalies = []
        for label, count in label_counts.items():
            ratio = count / total
            if ratio < 0.05 or ratio > 0.95:
                anomalies.append({
                    "label": label, "count": count, "ratio": ratio,
                    "reason": "标签分布异常：比例过低或过高"
                })
        return {
            "label_distribution": dict(label_counts),
            "anomalies": anomalies,
            "is_suspicious": len(anomalies) > 0
        }

    def detect_backdoor_pattern(self, dataset: List[Dict]) -> Dict:
        suffix_counter = Counter()
        for sample in dataset:
            text = sample.get("text", sample.get("input", ""))
            words = text.split()
            if len(words) >= 3:
                suffix = " ".join(words[-3:])
                suffix_counter[suffix] += 1
        threshold = len(dataset) * self.suffix_threshold_ratio
        suspicious = [
            {"pattern": pattern, "count": count, "ratio": count / len(dataset)}
            for pattern, count in suffix_counter.items()
            if count > threshold
        ]
        return {
            "suspicious_patterns": suspicious,
            "is_suspicious": len(suspicious) > 0,
            "total_unique_suffixes": len(suffix_counter)
        }

    def detect_statistical_anomaly(self, dataset: List[Dict]) -> Dict:
        lengths = []
        for sample in dataset:
            text = sample.get("text", sample.get("input", ""))
            lengths.append(len(text))
        lengths = np.array(lengths)
        mean_len = np.mean(lengths)
        std_len = np.std(lengths)
        anomalies = []
        for i, length in enumerate(lengths):
            z_score = (length - mean_len) / std_len if std_len > 0 else 0
            if abs(z_score) > self.z_score_threshold:
                anomalies.append({
                    "index": i, "length": int(length),
                    "z_score": round(z_score, 2)
                })
        return {
            "mean_length": round(mean_len, 2),
            "std_length": round(std_len, 2),
            "anomalies": anomalies,
            "is_suspicious": len(anomalies) > len(dataset) * 0.01
        }

    def comprehensive_detect(self, dataset: List[Dict]) -> Dict:
        label_result = self.detect_label_flipping(dataset)
        pattern_result = self.detect_backdoor_pattern(dataset)
        stat_result = self.detect_statistical_anomaly(dataset)
        suspicious_count = sum([
            label_result["is_suspicious"],
            pattern_result["is_suspicious"],
            stat_result["is_suspicious"]
        ])
        return {
            "label_analysis": label_result,
            "pattern_analysis": pattern_result,
            "statistical_analysis": stat_result,
            "overall_suspicion_score": suspicious_count / 3,
            "recommendation": "HIGH" if suspicious_count >= 2 else "MEDIUM" if suspicious_count == 1 else "LOW"
        }
```

**要点**：
- 标签分布分析是最简单的投毒检测手段，能有效发现大规模标签翻转攻击，但对低投毒率的后门攻击无效
- 后缀模式检测假设触发器以固定形式出现在文本末尾，对于随机位置或变形触发器需要更复杂的模式匹配
- 三种检测方法各有盲区，实际使用时应综合多个维度的结果做判断，单一方法的误报率较高

### 练习 3: 数据安全方案

**思路**：设计一个完整的数据安全防护方案，包含三个环节：数据源验证（来源追溯、完整性校验、标签分布检查）、投毒检测（标签分析、模式检测、统计异常、激活聚类）、差分隐私训练（梯度裁剪 + 噪声注入限制单个样本影响）。

**答案**：

```python
import hashlib
import numpy as np
from typing import List, Dict, Tuple
from datetime import datetime

class DataSourceVerifier:
    """数据源验证器"""

    def __init__(self):
        self.approved_sources = set()

    def register_source(self, source_name: str, checksum: str):
        self.approved_sources.add((source_name, checksum))

    def verify_source(self, dataset: List[Dict], source_name: str) -> Dict:
        data_str = str(sorted(str(s) for s in dataset))
        checksum = hashlib.sha256(data_str.encode()).hexdigest()[:16]
        is_approved = (source_name, checksum) in self.approved_sources
        label_dist = {}
        for sample in dataset:
            label = sample.get("label", "unknown")
            label_dist[label] = label_dist.get(label, 0) + 1
        total = len(dataset)
        label_anomalies = []
        for label, count in label_dist.items():
            ratio = count / total
            if ratio < 0.05 or ratio > 0.95:
                label_anomalies.append({"label": label, "ratio": round(ratio, 4)})
        return {
            "source_name": source_name,
            "checksum": checksum,
            "is_approved_source": is_approved,
            "dataset_size": total,
            "label_distribution": label_dist,
            "label_anomalies": label_anomalies,
            "verified_at": datetime.now().isoformat()
        }


class DataSanitizer:
    """数据清洗器"""

    def __init__(self, max_length: int = 10000, min_length: int = 5,
                 max_special_ratio: float = 0.3):
        self.max_length = max_length
        self.min_length = min_length
        self.max_special_ratio = max_special_ratio

    def sanitize(self, dataset: List[Dict]) -> Tuple[List[Dict], Dict]:
        clean = []
        report = {"original_size": len(dataset), "removed": 0, "reasons": {}}
        for sample in dataset:
            reason = self._check_sample(sample)
            if reason:
                clean.append(sample)
            else:
                report["removed"] += 1
                reason_key = "format_error"
                report["reasons"][reason_key] = report["reasons"].get(reason_key, 0) + 1
        report["clean_size"] = len(clean)
        return clean, report

    def _check_sample(self, sample: Dict) -> bool:
        text = sample.get("text", sample.get("input", ""))
        if len(text) > self.max_length or len(text) < self.min_length:
            return False
        special_ratio = sum(1 for c in text if not c.isalnum() and not c.isspace()) / max(len(text), 1)
        if special_ratio > self.max_special_ratio:
            return False
        return True


class DifferentiallyPrivateTrainer:
    """差分隐私训练"""

    def __init__(self, model, epsilon: float = 1.0, delta: float = 1e-5,
                 max_grad_norm: float = 1.0):
        self.model = model
        self.epsilon = epsilon
        self.delta = delta
        self.max_grad_norm = max_grad_norm

    def clip_gradients(self, gradients: List) -> List:
        total_norm = sum(float(np.linalg.norm(g)) ** 2 for g in gradients) ** 0.5
        if total_norm > self.max_grad_norm:
            clip_factor = self.max_grad_norm / total_norm
            gradients = [g * clip_factor for g in gradients]
        return gradients

    def add_noise(self, gradients: List, batch_size: int) -> List:
        noise_scale = self.max_grad_norm * np.sqrt(2 * np.log(1.25 / self.delta)) / self.epsilon
        return [g + np.random.normal(0, noise_scale / batch_size, g.shape) for g in gradients]

    def train_step(self, batch: List[Dict]):
        gradients = []
        for sample in batch:
            text = sample.get("text", sample.get("input", ""))
            grad = self.model.compute_gradient(text, sample["label"])
            gradients.append(grad)
        gradients = self.clip_gradients(gradients)
        gradients = self.add_noise(gradients, len(batch))
        avg_gradient = sum(gradients) / len(gradients)
        self.model.apply_gradient(avg_gradient)


class DataSecurityPipeline:
    """数据安全防护方案"""

    def __init__(self, model):
        self.verifier = DataSourceVerifier()
        self.sanitizer = DataSanitizer()
        self.detector = PoisoningDetector()
        self.dp_trainer = DifferentiallyPrivateTrainer(model)

    def full_pipeline(self, dataset: List[Dict], source_name: str) -> Dict:
        source_report = self.verifier.verify_source(dataset, source_name)
        clean_data, sanitize_report = self.sanitizer.sanitize(dataset)
        detection_report = self.detector.comprehensive_detect(clean_data)
        return {
            "source_verification": source_report,
            "data_cleaning": sanitize_report,
            "poisoning_detection": detection_report,
            "recommendation": self._make_recommendation(source_report, detection_report)
        }

    def _make_recommendation(self, source_report: Dict, detection_report: Dict) -> str:
        if not source_report["is_approved_source"]:
            return "BLOCK: 数据源未授权，拒绝使用"
        if detection_report["recommendation"] == "HIGH":
            return "BLOCK: 检测到高可疑投毒迹象，需要人工审核"
        if detection_report["recommendation"] == "MEDIUM":
            return "WARN: 检测到异常，建议人工复核后使用"
        return "PASS: 数据通过安全检查"
```

**要点**：
- 数据源验证是第一道防线，记录每条数据的来源和完整性校验和，异常数据必须隔离后再人工复核
- 差分隐私训练通过梯度裁剪和噪声注入限制单个样本的最大影响，即使存在少量投毒样本也能将其影响稀释
- 三层防护应形成流水线：源验证 → 清洗 → 检测 → 训练，每一步都有输出报告供人工审查
