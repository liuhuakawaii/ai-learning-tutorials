Google 研究团队在 2023 年发表的 GCG 攻击论文中证明：只需在恶意 prompt 末尾拼接一段看似无意义的字符后缀，就能让主流开源 LLM 绕过安全对齐，且该后缀在不同模型间具有惊人迁移性。这意味着攻击者无需访问模型权重，仅通过 API 即可实施攻击。对于依赖 LLM 做内容审核或自动决策的系统，对抗样本可能悄无声息地绕过所有安全检查，造成严重后果。

## 1. 对抗样本概述

### 1.1 什么是对抗样本

```
┌─────────────────────────────────────────────────────────────┐
│                  对抗样本 (Adversarial Examples)             │
│                                                             │
│  定义: 通过对输入添加精心设计的微小扰动，使模型                │
│        产生错误输出的样本。                                  │
│                                                             │
│  特点:                                                       │
│  ┌─────────────────────────────────────────────────────┐   │
│  │  · 微小扰动: 人类难以察觉的修改                        │   │
│  │  · 目标性: 可以指定模型输出错误类别                    │   │
│  │  · 迁移性: 在不同模型间可能迁移                        │   │
│  │  · 鲁棒性: 对防御措施有一定抵抗力                      │   │
│  └─────────────────────────────────────────────────────┘   │
│                                                             │
│  LLM 对抗样本的特殊性:                                       │
│  ┌─────────────────────────────────────────────────────┐   │
│  │  · 文本是离散的: 不能直接应用梯度方法                   │   │
│  │  · 语义约束: 修改需要保持语义                           │   │
│  │  · 上下文依赖: 效果依赖上下文                          │   │
│  │  · 评估困难: 没有标准的评估方法                        │   │
│  └─────────────────────────────────────────────────────┘   │
└─────────────────────────────────────────────────────────────┘
```

## 2. 对抗样本生成

### 2.1 GCG 攻击实现

```python
import torch
import numpy as np
from typing import List, Tuple

class GCGAttack:
    """GCG 对抗攻击"""

    def __init__(self, model, tokenizer, num_steps: int = 100,
                 batch_size: int = 512, top_k: int = 256):
        self.model = model
        self.tokenizer = tokenizer
        self.num_steps = num_steps
        self.batch_size = batch_size
        self.top_k = top_k

    def generate_suffix(self, prompt: str, target: str) -> str:
        """生成对抗后缀"""
        suffix_tokens = self._init_random_suffix()
        best_loss = float('inf')

        for step in range(self.num_steps):
            grad = self._compute_gradient(prompt, suffix_tokens, target)
            candidates = self._select_candidates(suffix_tokens, grad)
            best_suffix, best_loss = self._evaluate_candidates(
                prompt, candidates, target, best_loss
            )
            suffix_tokens = best_suffix

            if step % 10 == 0:
                print(f"Step {step}, Loss: {best_loss:.4f}")

            if self._check_success(prompt, suffix_tokens, target):
                print(f"Success at step {step}!")
                break

        return self._tokens_to_text(suffix_tokens)

    def _init_random_suffix(self, length: int = 20) -> List[int]:
        """初始化随机后缀"""
        vocab_size = self.tokenizer.vocab_size
        return [np.random.randint(0, vocab_size) for _ in range(length)]

    def _compute_gradient(self, prompt: str, suffix_tokens: List[int],
                           target: str) -> torch.Tensor:
        """计算梯度"""
        input_text = prompt + " " + self._tokens_to_text(suffix_tokens)
        inputs = self.tokenizer(input_text, return_tensors="pt")

        with torch.enable_grad():
            outputs = self.model(**inputs)
            loss = self._compute_loss(outputs, target)
            loss.backward()

        return inputs.input_ids.grad

    def _select_candidates(self, tokens: List[int], grad: torch.Tensor) -> List[List[int]]:
        """选择候选替换"""
        candidates = []
        for i in range(len(tokens)):
            top_indices = torch.topk(grad[i], self.top_k).indices
            for idx in top_indices:
                candidate = tokens.copy()
                candidate[i] = idx.item()
                candidates.append(candidate)
        return candidates[:self.batch_size]

    def _evaluate_candidates(self, prompt: str, candidates: List[List[int]],
                              target: str, best_loss: float) -> Tuple[List[int], float]:
        """评估候选，返回最优候选和损失"""
        best_candidate = candidates[0]
        current_best = best_loss

        for candidate in candidates:
            score = self._score_candidate(prompt, candidate, target)
            loss = -score
            if loss < current_best:
                current_best = loss
                best_candidate = candidate

        return best_candidate, current_best

    def _score_candidate(self, prompt: str, candidate: List[int], target: str) -> float:
        """评分候选"""
        input_text = prompt + " " + self._tokens_to_text(candidate)
        inputs = self.tokenizer(input_text, return_tensors="pt")
        target_ids = self.tokenizer(target, return_tensors="pt").input_ids

        with torch.no_grad():
            outputs = self.model(**inputs)
            logits = outputs.logits

        score = 0
        for i, target_id in enumerate(target_ids[0]):
            if i < logits.shape[1]:
                score += logits[0, i, target_id].item()

        return score

    def _check_success(self, prompt: str, suffix_tokens: List[int], target: str) -> bool:
        """检查是否成功"""
        input_text = prompt + " " + self._tokens_to_text(suffix_tokens)
        response = self._generate(input_text)
        return target.lower() in response.lower()

    def _tokens_to_text(self, tokens: List[int]) -> str:
        return self.tokenizer.decode(tokens)

    def _generate(self, text: str) -> str:
        inputs = self.tokenizer(text, return_tensors="pt")
        outputs = self.model.generate(**inputs, max_length=200)
        return self.tokenizer.decode(outputs[0])
```

### 2.2 文本对抗样本

```python
from typing import List, Dict, Tuple
import random

class TextAdversarialGenerator:
    """文本对抗样本生成器"""

    def __init__(self, word_embeddings=None):
        self.word_embeddings = word_embeddings

    def synonym_replacement(self, text: str, num_replacements: int = 3) -> str:
        """同义词替换"""
        words = text.split()
        important_words = self._identify_important_words(words)

        for word in important_words[:num_replacements]:
            synonyms = self._get_synonyms(word)
            if synonyms:
                replacement = random.choice(synonyms)
                text = text.replace(word, replacement, 1)

        return text

    def character_perturbation(self, text: str, num_perturbations: int = 2) -> str:
        """字符级扰动"""
        text_list = list(text)

        for _ in range(num_perturbations):
            idx = random.randint(0, len(text_list) - 1)
            perturbation_type = random.choice(['swap', 'delete', 'insert', 'replace'])

            if perturbation_type == 'swap' and idx < len(text_list) - 1:
                text_list[idx], text_list[idx + 1] = text_list[idx + 1], text_list[idx]
            elif perturbation_type == 'delete':
                text_list.pop(idx)
            elif perturbation_type == 'insert':
                text_list.insert(idx, random.choice('abcdefghijklmnopqrstuvwxyz'))
            elif perturbation_type == 'replace':
                text_list[idx] = random.choice('abcdefghijklmnopqrstuvwxyz')

        return ''.join(text_list)

    def word_order_change(self, text: str) -> str:
        """词序调整"""
        words = text.split()
        if len(words) > 3:
            idx1, idx2 = random.sample(range(len(words)), 2)
            words[idx1], words[idx2] = words[idx2], words[idx1]
        return ' '.join(words)

    def homoglyph_attack(self, text: str) -> str:
        """同形异义字攻击 — 用视觉相似的 Unicode 字符替换"""
        homoglyphs = {
            'a': 'а',  # Cyrillic а
            'e': 'е',  # Cyrillic е
            'o': 'о',  # Cyrillic о
            'p': 'р',  # Cyrillic р
            'c': 'с',  # Cyrillic с
            'x': 'х',  # Cyrillic х
        }
        result = []
        for char in text:
            if char.lower() in homoglyphs and random.random() < 0.3:
                replacement = homoglyphs[char.lower()]
                result.append(replacement.upper() if char.isupper() else replacement)
            else:
                result.append(char)
        return ''.join(result)

    def _identify_important_words(self, words: List[str]) -> List[str]:
        """识别重要词"""
        stopwords = {'the', 'a', 'an', 'is', 'are', 'was', 'were', 'in', 'on', 'at'}
        return [w for w in words if w.lower() not in stopwords]

    def _get_synonyms(self, word: str) -> List[str]:
        """获取同义词"""
        synonym_dict = {
            'good': ['great', 'excellent', 'fine'],
            'bad': ['poor', 'terrible', 'awful'],
            'big': ['large', 'huge', 'enormous'],
            'happy': ['glad', 'pleased', 'joyful'],
            'sad': ['unhappy', 'sorrowful', 'melancholy'],
        }
        return synonym_dict.get(word.lower(), [])
```

### 2.3 AutoDAN 遗传算法实现

```python
import random
from typing import List, Dict, Callable

class AutoDANAttack:
    """AutoDAN: 使用遗传算法自动演化越狱 prompt"""

    def __init__(self, model, fitness_fn: Callable, population_size: int = 20,
                 mutation_rate: float = 0.1, generations: int = 50):
        self.model = model
        self.fitness_fn = fitness_fn
        self.population_size = population_size
        self.mutation_rate = mutation_rate
        self.generations = generations

    def evolve(self, seed_prompts: List[str], target: str) -> str:
        """演化越狱 prompt"""
        population = seed_prompts[:self.population_size]

        for gen in range(self.generations):
            # 评估适应度
            fitness_scores = [
                self.fitness_fn(prompt, target) for prompt in population
            ]

            # 选择
            selected = self._selection(population, fitness_scores)

            # 交叉
            offspring = self._crossover(selected)

            # 变异
            mutated = self._mutation(offspring)

            population = mutated

            best_idx = fitness_scores.index(max(fitness_scores))
            if gen % 10 == 0:
                print(f"Gen {gen}, Best fitness: {fitness_scores[best_idx]:.4f}")

            if fitness_scores[best_idx] > 0.9:
                break

        best_idx = fitness_scores.index(max(fitness_scores))
        return population[best_idx]

    def _selection(self, population: List[str], scores: List[float]) -> List[str]:
        """锦标赛选择"""
        selected = []
        for _ in range(len(population)):
            i, j = random.sample(range(len(population)), 2)
            winner = population[i] if scores[i] > scores[j] else population[j]
            selected.append(winner)
        return selected

    def _crossover(self, parents: List[str]) -> List[str]:
        """单点交叉"""
        offspring = []
        for i in range(0, len(parents) - 1, 2):
            p1_words = parents[i].split()
            p2_words = parents[i + 1].split()
            point = random.randint(1, min(len(p1_words), len(p2_words)) - 1)
            child1 = ' '.join(p1_words[:point] + p2_words[point:])
            child2 = ' '.join(p2_words[:point] + p1_words[point:])
            offspring.extend([child1, child2])
        if len(parents) % 2 == 1:
            offspring.append(parents[-1])
        return offspring

    def _mutation(self, population: List[str]) -> List[str]:
        """词级变异"""
        mutated = []
        for prompt in population:
            if random.random() < self.mutation_rate:
                words = prompt.split()
                if words:
                    idx = random.randint(0, len(words) - 1)
                    words[idx] = self._random_word_substitute(words[idx])
                mutated.append(' '.join(words))
            else:
                mutated.append(prompt)
        return mutated

    def _random_word_substitute(self, word: str) -> str:
        """随机替换一个词"""
        alternatives = ["please", "kindly", "now", "simply", "just"]
        return random.choice(alternatives)
```

### 3.1 检测方法

```python
class AdversarialDetector:
    """对抗样本检测器"""

    def __init__(self):
        self.perplexity_threshold = 100
        self.semantic_threshold = 0.5

    def detect(self, text: str) -> Tuple[bool, Dict]:
        """检测对抗样本"""
        findings = []

        # 1. 困惑度检测
        perplexity = self._compute_perplexity(text)
        if perplexity > self.perplexity_threshold:
            findings.append({
                "type": "high_perplexity",
                "score": perplexity,
                "threshold": self.perplexity_threshold
            })

        # 2. 字符异常检测
        char_anomalies = self._detect_char_anomalies(text)
        if char_anomalies:
            findings.append({
                "type": "char_anomaly",
                "details": char_anomalies
            })

        # 3. 语义一致性检测
        semantic_score = self._check_semantic_consistency(text)
        if semantic_score < self.semantic_threshold:
            findings.append({
                "type": "semantic_inconsistency",
                "score": semantic_score
            })

        is_adversarial = len(findings) > 0
        return is_adversarial, {"findings": findings}

    def _compute_perplexity(self, text: str) -> float:
        """计算困惑度 (简化实现)"""
        words = text.split()
        unique_words = set(words)
        if len(words) == 0:
            return 0
        return len(unique_words) / len(words) * 100

    def _detect_char_anomalies(self, text: str) -> List[str]:
        """检测字符异常"""
        anomalies = []

        special_chars = sum(1 for c in text if not c.isalnum() and not c.isspace())
        if special_chars / max(len(text), 1) > 0.3:
            anomalies.append("特殊字符比例过高")

        if any(text.count(c * 3) > 0 for c in set(text)):
            anomalies.append("检测到重复字符")

        # Unicode 异常检测
        non_ascii = sum(1 for c in text if ord(c) > 127)
        if non_ascii / max(len(text), 1) > 0.2:
            anomalies.append("非ASCII字符比例异常")

        return anomalies

    def _check_semantic_consistency(self, text: str) -> float:
        """检查语义一致性 (简化实现)"""
        words = text.split()
        if len(words) < 3:
            return 1.0
        # 简化: 检查是否有明显的语义断裂
        return 0.8
```

### 3.2 鲁棒性评估框架

```python
from typing import List, Dict, Callable

class RobustnessEvaluator:
    """模型鲁棒性评估框架"""

    def __init__(self, model, tokenizer):
        self.model = model
        self.tokenizer = tokenizer

    def evaluate_attack_success_rate(self, attack_fn: Callable,
                                      test_data: List[Dict]) -> Dict:
        """评估攻击成功率"""
        total = len(test_data)
        success = 0

        for sample in test_data:
            original_label = sample["label"]
            adversarial_text = attack_fn(sample["text"])
            predicted_label = self.model.predict(adversarial_text)

            if predicted_label != original_label:
                success += 1

        return {
            "attack_success_rate": success / total,
            "total_samples": total,
            "successful_attacks": success,
            "robustness_score": 1 - success / total
        }

    def evaluate_perturbation_budget(self, original: str, adversarial: str) -> Dict:
        """评估扰动预算"""
        orig_words = original.split()
        adv_words = adversarial.split()

        # 词级编辑距离
        word_changes = sum(1 for o, a in zip(orig_words, adv_words) if o != a)
        word_change_rate = word_changes / max(len(orig_words), 1)

        # 字符级编辑距离
        char_changes = sum(1 for o, a in zip(original, adversarial) if o != a)
        char_change_rate = char_changes / max(len(original), 1)

        return {
            "word_level_changes": word_changes,
            "word_change_rate": word_change_rate,
            "char_level_changes": char_changes,
            "char_change_rate": char_change_rate,
            "within_budget": word_change_rate < 0.15
        }

    def cross_model_transfer(self, source_attack_fn: Callable,
                              target_model, test_data: List[Dict]) -> Dict:
        """评估跨模型迁移性"""
        transfer_success = 0
        total = len(test_data)

        for sample in test_data:
            adversarial_text = source_attack_fn(sample["text"])
            target_pred = target_model.predict(adversarial_text)
            if target_pred != sample["label"]:
                transfer_success += 1

        return {
            "transfer_rate": transfer_success / total,
            "total_samples": total,
            "transferred_attacks": transfer_success
        }
```

### 4.1 对抗训练

```python
class AdversarialTraining:
    """对抗训练 — 在训练数据中加入对抗样本以提升鲁棒性"""

    def __init__(self, model, attack_generator: TextAdversarialGenerator):
        self.model = model
        self.attack_gen = attack_generator

    def generate_adversarial_batch(self, texts: List[str],
                                     labels: List[int]) -> List[Dict]:
        """生成对抗训练批次"""
        adversarial_batch = []

        for text, label in zip(texts, labels):
            # 原始样本
            adversarial_batch.append({"text": text, "label": label})

            # 同义词替换对抗样本
            adv_text = self.attack_gen.synonym_replacement(text)
            adversarial_batch.append({"text": adv_text, "label": label})

            # 字符扰动对抗样本
            adv_text = self.attack_gen.character_perturbation(text)
            adversarial_batch.append({"text": adv_text, "label": label})

        return adversarial_batch

    def train_epoch(self, train_data: List[Dict], lr: float = 1e-5):
        """对抗训练一个 epoch"""
        for sample in train_data:
            loss = self.model.train_step(sample["text"], sample["label"])
        return loss
```

```python
class InputSanitizer:
    """输入净化 — 在推理前清洗对抗扰动"""

    def __init__(self):
        self.char_threshold = 0.2
        self.length_threshold = 1000

    def sanitize(self, text: str) -> str:
        """净化输入文本"""
        # 1. Unicode 规范化
        import unicodedata
        text = unicodedata.normalize('NFKC', text)

        # 2. 移除零宽字符
        zero_width_chars = ['\u200b', '\u200c', '\u200d', '\ufeff']
        for zwc in zero_width_chars:
            text = text.replace(zwc, '')

        # 3. 同形异义字替换回 ASCII
        text = self._replace_homoglyphs(text)

        # 4. 截断过长输入
        if len(text) > self.length_threshold:
            text = text[:self.length_threshold]

        return text

    def _replace_homoglyphs(self, text: str) -> str:
        """替换同形异义字为 ASCII 等价字符"""
        homoglyph_map = {
            'а': 'a', 'е': 'e', 'о': 'o', 'р': 'p',
            'с': 'c', 'х': 'x', 'у': 'y', 'і': 'i',
        }
        result = []
        for char in text:
            result.append(homoglyph_map.get(char, char))
        return ''.join(result)
```

## 5. 常见误区

1. **"对抗样本只影响图像"**: 文本同样存在对抗样本，且在 LLM 中越来越重要
2. **"对抗样本容易检测"**: 高质量对抗样本很难检测，尤其是语义保持型攻击
3. **"防御可以一劳永逸"**: 对抗攻击持续演化，防御需要不断更新
4. **"只关注准确率"**: 鲁棒性同样重要，高准确率不代表高鲁棒性
5. **"忽略迁移性"**: 对抗样本可能跨模型迁移，一个模型的漏洞可能影响其他模型
6. **"对抗训练是万能的"**: 对抗训练会降低正常数据的准确率，需要权衡

## 工程建议

- 在模型推理前加入输入净化层：对所有用户输入做 Unicode NFKC 规范化、零宽字符移除和同形异义字还原，消除字符级对抗扰动而不影响正常语义。
- 对抗训练是提升模型鲁棒性的有效手段，但会牺牲干净样本的准确率；建议用 5%-10% 比例混合对抗样本训练，并在验证集上同时监控干净准确率和攻击成功率，找到最优平衡点。
- 部署困惑度检测和 n-gram 重复率检测作为快速初筛，对高可疑样本再用语义一致性模型做深度判断，形成"快速过滤 + 精细检测"的两级检测架构。
- 定期用跨模型迁移攻击测试系统鲁棒性：在一个模型上生成的对抗样本往往能迁移攻击同族模型，这能暴露仅在单一模型上测试所忽略的盲区。

## 总结

- 对抗样本通过对输入添加微小扰动欺骗模型
- LLM 对抗攻击包括 GCG、AutoDAN、TextFooler 等方法
- 检测方法包括困惑度、字符异常、语义一致性等
- 防御策略包括对抗训练、输入净化、鲁棒性评估
- 需要持续研究和防御对抗攻击

## 参考答案

### 练习 1: 对抗样本生成

**思路**：实现 TextAdversarialGenerator 类，支持同义词替换、字符扰动、同形异义字三种攻击方法。同义词替换需要先识别重要词再替换；字符扰动包含交换、删除、插入、替换四种操作；同形异义字利用视觉相似的 Unicode 字符替换 ASCII 字符。

**答案**：

```python
import random
from typing import List, Dict

class TextAdversarialGenerator:
    """文本对抗样本生成器"""

    def __init__(self, synonym_dict: Dict[str, List[str]] = None):
        self.synonym_dict = synonym_dict or {
            'good': ['great', 'excellent', 'fine', 'wonderful'],
            'bad': ['poor', 'terrible', 'awful', 'horrible'],
            'big': ['large', 'huge', 'enormous', 'massive'],
            'happy': ['glad', 'pleased', 'joyful', 'delighted'],
            'sad': ['unhappy', 'sorrowful', 'melancholy', 'depressed'],
            'important': ['crucial', 'vital', 'significant', 'essential'],
            'show': ['display', 'demonstrate', 'reveal', 'present'],
            'help': ['assist', 'aid', 'support', 'facilitate'],
        }
        self.stopwords = {'the', 'a', 'an', 'is', 'are', 'was', 'were', 'in', 'on', 'at',
                          'to', 'for', 'of', 'with', 'by', 'and', 'or', 'but', 'not', 'this', 'that'}
        self.homoglyphs = {
            'a': '\u0430', 'e': '\u0435', 'o': '\u043e', 'p': '\u0440',
            'c': '\u0441', 'x': '\u0445', 'y': '\u0443', 'i': '\u0456',
            's': '\u0455', 'd': '\u0501',
        }

    def synonym_replacement(self, text: str, num_replacements: int = 3) -> str:
        words = text.split()
        important = [i for i, w in enumerate(words) if w.lower() not in self.stopwords]
        random.shuffle(important)
        replaced = 0
        for idx in important:
            if replaced >= num_replacements:
                break
            word_lower = words[idx].lower()
            if word_lower in self.synonym_dict:
                synonym = random.choice(self.synonym_dict[word_lower])
                if words[idx][0].isupper():
                    synonym = synonym.capitalize()
                words[idx] = synonym
                replaced += 1
        return ' '.join(words)

    def character_perturbation(self, text: str, num_perturbations: int = 2) -> str:
        text_list = list(text)
        for _ in range(num_perturbations):
            if len(text_list) < 2:
                break
            idx = random.randint(0, len(text_list) - 1)
            op = random.choice(['swap', 'delete', 'insert', 'replace'])
            if op == 'swap' and idx < len(text_list) - 1:
                text_list[idx], text_list[idx + 1] = text_list[idx + 1], text_list[idx]
            elif op == 'delete' and len(text_list) > 1:
                text_list.pop(idx)
            elif op == 'insert':
                text_list.insert(idx, random.choice('abcdefghijklmnopqrstuvwxyz'))
            elif op == 'replace':
                text_list[idx] = random.choice('abcdefghijklmnopqrstuvwxyz')
        return ''.join(text_list)

    def homoglyph_attack(self, text: str, replacement_prob: float = 0.3) -> str:
        result = []
        for char in text:
            if char.lower() in self.homoglyphs and random.random() < replacement_prob:
                replacement = self.homoglyphs[char.lower()]
                result.append(replacement.upper() if char.isupper() else replacement)
            else:
                result.append(char)
        return ''.join(result)

    def generate_batch(self, text: str, methods: List[str] = None) -> Dict[str, str]:
        if methods is None:
            methods = ['synonym', 'perturbation', 'homoglyph']
        results = {"original": text}
        if 'synonym' in methods:
            results['synonym'] = self.synonym_replacement(text)
        if 'perturbation' in methods:
            results['perturbation'] = self.character_perturbation(text)
        if 'homoglyph' in methods:
            results['homoglyph'] = self.homoglyph_attack(text)
        return results

    def evaluate_perturbation(self, original: str, adversarial: str) -> Dict:
        orig_chars = list(original)
        adv_chars = list(adversarial)
        max_len = max(len(orig_chars), len(adv_chars))
        char_diff = sum(1 for o, a in zip(orig_chars, adv_chars) if o != a)
        char_diff += abs(len(orig_chars) - len(adv_chars))
        return {
            "char_change_rate": char_diff / max(max_len, 1),
            "original_length": len(original),
            "adversarial_length": len(adversarial),
            "within_budget": char_diff / max(max_len, 1) < 0.15
        }
```

**要点**：
- 同义词替换需要先过滤停用词，只替换对语义影响较大的词，才能在保持可读性的同时改变模型判断
- 同形异义字攻击利用 Cyrillic 字符与拉丁字符视觉相似的特性，对人类几乎不可见但能欺骗模型的 tokenizer
- 评估扰动预算时应同时计算字符级和词级变化率，确保修改在人类不可察觉的范围内

### 练习 2: 检测器实现

**思路**：实现一个 AdversarialDetector，从三个维度检测对抗样本：困惑度检测（n-gram 重复率异常高）、字符异常检测（特殊字符比例、重复字符、非 ASCII 比例）、Unicode 异常检测（零宽字符、同形异义字）。综合多个维度的检测结果给出最终判断。

**答案**：

```python
import unicodedata
from typing import List, Dict, Tuple

class AdversarialDetector:
    """对抗样本检测器"""

    def __init__(self, perplexity_threshold: float = 100, char_anomaly_threshold: float = 0.3,
                 non_ascii_threshold: float = 0.2):
        self.perplexity_threshold = perplexity_threshold
        self.char_anomaly_threshold = char_anomaly_threshold
        self.non_ascii_threshold = non_ascii_threshold
        self.zero_width_chars = {'\u200b', '\u200c', '\u200d', '\u200e', '\u200f', '\ufeff'}
        self.homoglyph_map = {
            '\u0430': 'a', '\u0435': 'e', '\u043e': 'o', '\u0440': 'p',
            '\u0441': 'c', '\u0445': 'x', '\u0443': 'y', '\u0456': 'i',
            '\u0455': 's', '\u0501': 'd',
        }

    def detect(self, text: str) -> Tuple[bool, Dict]:
        findings = []

        perplexity = self._compute_perplexity(text)
        if perplexity > self.perplexity_threshold:
            findings.append({"type": "high_perplexity", "score": perplexity, "threshold": self.perplexity_threshold})

        char_anomalies = self._detect_char_anomalies(text)
        if char_anomalies:
            findings.append({"type": "char_anomaly", "details": char_anomalies})

        unicode_anomalies = self._detect_unicode_anomalies(text)
        if unicode_anomalies:
            findings.append({"type": "unicode_anomaly", "details": unicode_anomalies})

        homoglyph_hits = self._detect_homoglyphs(text)
        if homoglyph_hits:
            findings.append({"type": "homoglyph_detected", "count": len(homoglyph_hits)})

        is_adversarial = len(findings) > 0
        confidence = min(0.4 * len(findings) + 0.3 * max(len(findings) - 1, 0), 1.0)

        return is_adversarial, {
            "findings": findings,
            "confidence": confidence,
            "risk_level": "HIGH" if confidence > 0.7 else "MEDIUM" if confidence > 0.4 else "LOW"
        }

    def _compute_perplexity(self, text: str) -> float:
        words = text.split()
        if len(words) < 3:
            return 0
        bigrams = [(words[i], words[i + 1]) for i in range(len(words) - 1)]
        unique_bigrams = set(bigrams)
        repetition_rate = 1 - len(unique_bigrams) / max(len(bigrams), 1)
        return repetition_rate * 200

    def _detect_char_anomalies(self, text: str) -> List[str]:
        anomalies = []
        if not text:
            return anomalies
        special_count = sum(1 for c in text if not c.isalnum() and not c.isspace())
        if special_count / len(text) > self.char_anomaly_threshold:
            anomalies.append(f"特殊字符比例 {special_count / len(text):.2%} 超过阈值 {self.char_anomaly_threshold:.0%}")
        for c in set(text):
            if text.count(c * 3) > 0:
                anomalies.append(f"检测到连续重复字符 '{c}'")
                break
        return anomalies

    def _detect_unicode_anomalies(self, text: str) -> List[str]:
        anomalies = []
        if not text:
            return anomalies
        zero_width_count = sum(1 for c in text if c in self.zero_width_chars)
        if zero_width_count > 0:
            anomalies.append(f"检测到 {zero_width_count} 个零宽字符")
        non_ascii = sum(1 for c in text if ord(c) > 127)
        if non_ascii / len(text) > self.non_ascii_threshold:
            anomalies.append(f"非 ASCII 字符比例 {non_ascii / len(text):.2%} 异常")
        return anomalies

    def _detect_homoglyphs(self, text: str) -> List[Dict]:
        hits = []
        for i, c in enumerate(text):
            if c in self.homoglyph_map:
                hits.append({"position": i, "char": c, "ascii_equivalent": self.homoglyph_map[c]})
        return hits

    def evaluate(self, adversarial_samples: List[str], clean_samples: List[str]) -> Dict:
        tp = sum(1 for s in adversarial_samples if self.detect(s)[0])
        fp = sum(1 for s in clean_samples if self.detect(s)[0])
        fn = len(adversarial_samples) - tp
        tn = len(clean_samples) - fp
        precision = tp / max(tp + fp, 1)
        recall = tp / max(tp + fn, 1)
        f1 = 2 * precision * recall / max(precision + recall, 1e-8)
        return {
            "precision": precision, "recall": recall, "f1_score": f1,
            "true_positive": tp, "false_positive": fp,
            "true_negative": tn, "false_negative": fn
        }
```

**要点**：
- 零宽字符（\u200b 等）是常见的对抗扰动手段，人类完全不可见但会改变 token 化结果
- 同形异义字检测需要维护一个 Cyrillic/Latin 映射表，逐字符检查是否使用了视觉相似的替代字符
- 检测器应该输出置信度而非简单的布尔值，便于下游系统根据置信度做分级处理

### 练习 3: 防御方案

**思路**：构建一个包含输入净化和对抗训练的双层防御方案。输入净化在推理前清洗对抗扰动（Unicode 规范化、零宽字符移除、同形异义字还原）；对抗训练在训练阶段加入对抗样本提升模型鲁棒性。通过对比防御前后的攻击成功率评估防御效果。

**答案**：

```python
import unicodedata
from typing import List, Dict

class InputSanitizer:
    """输入净化器"""

    def __init__(self):
        self.zero_width_chars = ['\u200b', '\u200c', '\u200d', '\u200e', '\u200f', '\ufeff']
        self.homoglyph_map = {
            '\u0430': 'a', '\u0435': 'e', '\u043e': 'o', '\u0440': 'p',
            '\u0441': 'c', '\u0445': 'x', '\u0443': 'y', '\u0456': 'i',
            '\u0455': 's', '\u0501': 'd',
        }

    def sanitize(self, text: str) -> str:
        text = unicodedata.normalize('NFKC', text)
        for zwc in self.zero_width_chars:
            text = text.replace(zwc, '')
        text = self._replace_homoglyphs(text)
        return text

    def _replace_homoglyphs(self, text: str) -> str:
        result = []
        for char in text:
            result.append(self.homoglyph_map.get(char, char))
        return ''.join(result)


class AdversarialTraining:
    """对抗训练"""

    def __init__(self, model, attack_generator):
        self.model = model
        self.attack_gen = attack_generator

    def generate_augmented_batch(self, texts: List[str], labels: List[int],
                                  adv_ratio: float = 0.1) -> List[Dict]:
        augmented = []
        for text, label in zip(texts, labels):
            augmented.append({"text": text, "label": label})
            augmented.append({"text": self.attack_gen.synonym_replacement(text), "label": label})
            augmented.append({"text": self.attack_gen.character_perturbation(text), "label": label})
            augmented.append({"text": self.attack_gen.homoglyph_attack(text), "label": label})
        return augmented


class DefensePipeline:
    """防御流水线"""

    def __init__(self, model, attack_generator):
        self.sanitizer = InputSanitizer()
        self.adv_trainer = AdversarialTraining(model, attack_generator)
        self.model = model

    def inference_with_defense(self, user_input: str) -> str:
        sanitized = self.sanitizer.sanitize(user_input)
        return self.model.predict(sanitized)

    def evaluate_defense(self, test_data: List[Dict], attack_fn) -> Dict:
        success_before = 0
        success_after = 0
        total = len(test_data)
        for sample in test_data:
            adv_text = attack_fn(sample["text"])
            pred_before = self.model.predict(adv_text)
            if pred_before != sample["label"]:
                success_before += 1
            sanitized = self.sanitizer.sanitize(adv_text)
            pred_after = self.model.predict(sanitized)
            if pred_after != sample["label"]:
                success_after += 1
        return {
            "attack_success_before_defense": success_before / total,
            "attack_success_after_defense": success_after / total,
            "defense_improvement": (success_before - success_after) / max(total, 1),
            "total_samples": total
        }
```

**要点**：
- 输入净化应在模型推理前作为预处理步骤，Unicode NFKC 规范化可以消除大量字符级对抗扰动
- 对抗训练时建议对抗样本占比 5%-10%，过高会导致干净样本准确率下降，过低则防御效果不明显
- 评估防御效果时必须同时报告干净样本准确率和攻击成功率，避免只看一个指标导致的片面结论
