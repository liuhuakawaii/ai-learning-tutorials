## 场景引入

某企业内部部署了一个基于大模型的知识库问答系统，所有员工共用同一个 API Key 访问后端服务。一名普通员工通过构造特殊请求，越权访问了高管才能查看的财务报告和人事档案——系统根本没有做任何权限区分。更糟糕的是，该系统对接的外部工具（如数据库查询、文件删除）也没有独立的权限控制，导致一次注入攻击就能让模型直接执行危险操作。事后调查还发现，离职员工的 API Key 三个月后仍然有效，系统没有密钥过期和自动回收机制。这些问题的根源都指向同一件事：AI 系统的权限与访问控制被严重忽视。

## 1. 权限控制模型

### 1.1 RBAC vs ABAC

```
┌─────────────────────────────────────────────────────────────┐
│              权限控制模型对比                                 │
│                                                             │
│  RBAC (基于角色的访问控制)                                   │
│  ┌─────────────────────────────────────────────────────┐   │
│  │  用户 ──▶ 角色 ──▶ 权限                               │   │
│  │                                                     │   │
│  │  优点:                                               │   │
│  │  · 简单易懂                                          │   │
│  │  · 易于管理                                          │   │
│  │  · 适合组织结构                                      │   │
│  │                                                     │   │
│  │  缺点:                                               │   │
│  │  · 粒度较粗                                          │   │
│  │  · 角色爆炸问题                                      │   │
│  │  · 不适合动态场景                                    │   │
│  └─────────────────────────────────────────────────────┘   │
│                                                             │
│  ABAC (基于属性的访问控制)                                   │
│  ┌─────────────────────────────────────────────────────┐   │
│  │  策略 ──▶ 评估(用户属性, 资源属性, 环境属性) ──▶ 决策  │   │
│  │                                                     │   │
│  │  优点:                                               │   │
│  │  · 粒度更细                                          │   │
│  │  · 灵活性高                                          │   │
│  │  · 支持动态策略                                      │   │
│  │                                                     │   │
│  │  缺点:                                               │   │
│  │  · 复杂度高                                          │   │
│  │  · 管理成本高                                        │   │
│  │  · 性能开销                                          │   │
│  └─────────────────────────────────────────────────────┘   │
└─────────────────────────────────────────────────────────────┘
```

### 1.2 模型选型对比

| 特性 | RBAC | ABAC | ReBAC |
|------|------|------|-------|
| 粒度 | 角色级 | 属性级 | 关系级 |
| 复杂度 | 低 | 高 | 中 |
| 动态性 | 低 | 高 | 中 |
| 适用规模 | 中小 | 大型 | 社交/协作 |
| AI 场景 | 基础权限 | 模型/数据权限 | Agent 协作 |
| 实现成本 | 低 | 高 | 中 |

### 1.3 RBAC 实现

```python
from typing import Dict, List, Set, Optional
from dataclasses import dataclass, field
from datetime import datetime
import json


@dataclass
class Permission:
    name: str
    resource: str
    action: str
    description: str = ""

    def matches(self, resource: str, action: str) -> bool:
        if self.resource == "*" and self.action == "*":
            return True
        if self.resource == resource and self.action == "*":
            return True
        if self.resource == "*" and self.action == action:
            return True
        return self.resource == resource and self.action == action


@dataclass
class Role:
    name: str
    permissions: Set[str] = field(default_factory=set)
    parent_roles: Set[str] = field(default_factory=set)
    description: str = ""
    created_at: datetime = field(default_factory=datetime.now)


@dataclass
class User:
    id: str
    roles: Set[str] = field(default_factory=set)
    attributes: Dict = field(default_factory=dict)
    is_active: bool = True
    created_at: datetime = field(default_factory=datetime.now)


class RBACManager:
    """RBAC 权限管理器 — 支持角色继承"""

    def __init__(self):
        self.roles: Dict[str, Role] = {}
        self.users: Dict[str, User] = {}
        self.permissions: Dict[str, Permission] = {}

    def create_permission(self, name: str, resource: str, action: str, description: str = ""):
        self.permissions[name] = Permission(name, resource, action, description)

    def create_role(self, name: str, permissions: List[str] = None,
                    parent_roles: List[str] = None, description: str = ""):
        self.roles[name] = Role(
            name=name,
            permissions=set(permissions or []),
            parent_roles=set(parent_roles or []),
            description=description,
        )

    def create_user(self, user_id: str, roles: List[str] = None, attributes: Dict = None):
        self.users[user_id] = User(
            id=user_id,
            roles=set(roles or []),
            attributes=attributes or {},
        )

    def assign_role_to_user(self, user_id: str, role_name: str):
        if user_id in self.users and role_name in self.roles:
            self.users[user_id].roles.add(role_name)

    def revoke_role_from_user(self, user_id: str, role_name: str):
        if user_id in self.users:
            self.users[user_id].roles.discard(role_name)

    def _get_effective_permissions(self, role_name: str, visited: Set[str] = None) -> Set[str]:
        if visited is None:
            visited = set()
        if role_name in visited:
            return set()
        visited.add(role_name)

        role = self.roles.get(role_name)
        if not role:
            return set()

        perms = set(role.permissions)
        for parent in role.parent_roles:
            perms.update(self._get_effective_permissions(parent, visited))
        return perms

    def check_permission(self, user_id: str, resource: str, action: str) -> bool:
        if user_id not in self.users:
            return False
        user = self.users[user_id]
        if not user.is_active:
            return False

        for role_name in user.roles:
            perm_names = self._get_effective_permissions(role_name)
            for perm_name in perm_names:
                perm = self.permissions.get(perm_name)
                if perm and perm.matches(resource, action):
                    return True
        return False

    def get_user_permissions(self, user_id: str) -> List[Dict]:
        if user_id not in self.users:
            return []
        user = self.users[user_id]
        result = []
        seen = set()
        for role_name in user.roles:
            perm_names = self._get_effective_permissions(role_name)
            for perm_name in perm_names:
                if perm_name not in seen:
                    seen.add(perm_name)
                    perm = self.permissions[perm_name]
                    result.append({"name": perm.name, "resource": perm.resource, "action": perm.action})
        return result

    def audit_user_access(self, user_id: str) -> Dict:
        user = self.users.get(user_id)
        if not user:
            return {"error": "User not found"}
        return {
            "user_id": user_id,
            "roles": list(user.roles),
            "permissions": self.get_user_permissions(user_id),
            "is_active": user.is_active,
        }


# 使用示例
rbac = RBACManager()

# 创建权限
rbac.create_permission("chat:read", "chat", "read", "读取聊天记录")
rbac.create_permission("chat:write", "chat", "write", "发送消息")
rbac.create_permission("model:read", "model", "read", "查看模型信息")
rbac.create_permission("model:manage", "model", "manage", "管理模型配置")
rbac.create_permission("admin:all", "*", "*", "管理员全部权限")

# 创建角色（支持继承）
rbac.create_role("viewer", ["chat:read", "model:read"], description="只读用户")
rbac.create_role("user", ["chat:write"], parent_roles=["viewer"], description="普通用户")
rbac.create_role("admin", ["admin:all"], description="管理员")

# 创建用户
rbac.create_user("user1", roles=["user"])
rbac.create_user("admin1", roles=["admin"])

# 检查权限
print(rbac.check_permission("user1", "chat", "read"))    # True (继承自 viewer)
print(rbac.check_permission("user1", "chat", "write"))   # True
print(rbac.check_permission("user1", "model", "manage")) # False
print(rbac.check_permission("admin1", "model", "manage")) # True
```

## 2. ABAC 动态权限

### 2.1 ABAC 实现

```python
from typing import Dict, Any, Callable
from dataclasses import dataclass
from enum import Enum
import operator


class Operator(Enum):
    EQ = "=="
    NEQ = "!="
    GT = ">"
    LT = "<"
    GTE = ">="
    LTE = "<="
    IN = "in"
    NOT_IN = "not_in"
    CONTAINS = "contains"


@dataclass
class PolicyCondition:
    attribute: str
    op: Operator
    value: Any

    def evaluate(self, context: Dict) -> bool:
        parts = self.attribute.split(".")
        current = context
        for part in parts:
            if isinstance(current, dict) and part in current:
                current = current[part]
            else:
                return False

        ops = {
            Operator.EQ: operator.eq,
            Operator.NEQ: operator.ne,
            Operator.GT: operator.gt,
            Operator.LT: operator.lt,
            Operator.GTE: operator.ge,
            Operator.LTE: operator.le,
            Operator.IN: lambda a, b: a in b,
            Operator.NOT_IN: lambda a, b: a not in b,
            Operator.CONTAINS: lambda a, b: b in a,
        }
        return ops.get(self.op, lambda a, b: False)(current, self.value)


@dataclass
class PolicyRule:
    name: str
    effect: str  # "allow" or "deny"
    conditions: list
    priority: int = 0

    def evaluate(self, context: Dict) -> str:
        if all(cond.evaluate(context) for cond in self.conditions):
            return self.effect
        return "not_applicable"


class ABACEngine:
    """ABAC 引擎 — 基于属性的动态访问控制"""

    def __init__(self):
        self.policies: list = []

    def add_policy(self, policy: PolicyRule):
        self.policies.append(policy)
        self.policies.sort(key=lambda p: -p.priority)

    def evaluate(self, context: Dict) -> Dict:
        for policy in self.policies:
            result = policy.evaluate(context)
            if result != "not_applicable":
                return {
                    "decision": result,
                    "matched_policy": policy.name,
                    "context_keys": list(context.keys()),
                }
        return {"decision": "deny", "matched_policy": "default_deny", "context_keys": []}


# 使用示例
engine = ABACEngine()

# 策略: 工作时间允许访问
engine.add_policy(PolicyRule(
    name="work_hours_access",
    effect="allow",
    conditions=[
        PolicyCondition("environment.hour", Operator.GTE, 9),
        PolicyCondition("environment.hour", Operator.LT, 18),
        PolicyCondition("user.department", Operator.EQ, "engineering"),
    ],
    priority=10,
))

# 策略: 管理员随时允许
engine.add_policy(PolicyRule(
    name="admin_always_allowed",
    effect="allow",
    conditions=[
        PolicyCondition("user.role", Operator.EQ, "admin"),
    ],
    priority=100,
))

# 策略: 限制敏感数据访问
engine.add_policy(PolicyRule(
    name="deny_sensitive_data",
    effect="deny",
    conditions=[
        PolicyCondition("resource.sensitivity", Operator.EQ, "high"),
        PolicyCondition("user.clearance_level", Operator.LT, 3),
    ],
    priority=50,
))

# 评估请求
context = {
    "user": {"id": "u1", "role": "user", "department": "engineering", "clearance_level": 2},
    "resource": {"type": "model", "sensitivity": "high"},
    "environment": {"hour": 14, "ip": "10.0.0.1"},
}
result = engine.evaluate(context)
print(f"Decision: {result['decision']}, Policy: {result['matched_policy']}")
```

## 3. API 访问控制

### 3.1 API 密钥管理

```python
import hashlib
import secrets
import json
from typing import Dict, Optional, List
from datetime import datetime, timedelta
from dataclasses import dataclass, field


@dataclass
class APIKeyInfo:
    key_hash: str
    user_id: str
    name: str
    permissions: List[str]
    created_at: datetime
    expires_at: datetime
    is_active: bool = True
    last_used_at: Optional[datetime] = None
    usage_count: int = 0
    rate_limit: int = 1000
    allowed_ips: List[str] = field(default_factory=list)


class APIKeyManager:
    """API 密钥管理器"""

    def __init__(self):
        self.keys: Dict[str, APIKeyInfo] = {}

    def generate_key(self, user_id: str, name: str,
                     permissions: List[str] = None,
                     expires_days: int = 30,
                     rate_limit: int = 1000,
                     allowed_ips: List[str] = None) -> str:
        raw_key = f"sk-{secrets.token_hex(32)}"
        key_hash = hashlib.sha256(raw_key.encode()).hexdigest()

        self.keys[key_hash] = APIKeyInfo(
            key_hash=key_hash,
            user_id=user_id,
            name=name,
            permissions=permissions or [],
            created_at=datetime.now(),
            expires_at=datetime.now() + timedelta(days=expires_days),
            rate_limit=rate_limit,
            allowed_ips=allowed_ips or [],
        )
        return raw_key

    def validate_key(self, key: str, source_ip: str = None) -> Optional[APIKeyInfo]:
        key_hash = hashlib.sha256(key.encode()).hexdigest()
        info = self.keys.get(key_hash)

        if not info or not info.is_active:
            return None
        if datetime.now() > info.expires_at:
            return None
        if info.allowed_ips and source_ip not in info.allowed_ips:
            return None

        info.last_used_at = datetime.now()
        info.usage_count += 1
        return info

    def revoke_key(self, key: str) -> bool:
        key_hash = hashlib.sha256(key.encode()).hexdigest()
        if key_hash in self.keys:
            self.keys[key_hash].is_active = False
            return True
        return False

    def list_user_keys(self, user_id: str) -> List[Dict]:
        return [
            {
                "name": info.name,
                "permissions": info.permissions,
                "created_at": info.created_at.isoformat(),
                "expires_at": info.expires_at.isoformat(),
                "is_active": info.is_active,
                "usage_count": info.usage_count,
            }
            for info in self.keys.values()
            if info.user_id == user_id
        ]

    def rotate_key(self, old_key: str, expires_days: int = 30) -> Optional[str]:
        key_hash = hashlib.sha256(old_key.encode()).hexdigest()
        info = self.keys.get(key_hash)
        if not info:
            return None

        self.revoke_key(old_key)
        return self.generate_key(
            user_id=info.user_id,
            name=info.name,
            permissions=info.permissions,
            expires_days=expires_days,
            rate_limit=info.rate_limit,
            allowed_ips=info.allowed_ips,
        )


# 使用示例
manager = APIKeyManager()

# 生成密钥
key = manager.generate_key(
    user_id="user1",
    name="production-api",
    permissions=["chat:read", "chat:write"],
    expires_days=90,
    rate_limit=500,
    allowed_ips=["10.0.0.1", "192.168.1.0/24"],
)
print(f"Generated key: {key[:20]}...")

# 验证密钥
info = manager.validate_key(key, source_ip="10.0.0.1")
print(f"Valid: {info is not None}, Usage: {info.usage_count if info else 0}")
```

## 4. 工具调用权限控制

### 4.1 工具权限管理

```python
from typing import Dict, List, Optional, Any
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from collections import defaultdict


@dataclass
class ToolPermission:
    tool_name: str
    allowed_roles: List[str]
    required_params: List[str] = field(default_factory=list)
    max_calls_per_hour: int = 100
    max_calls_per_day: int = 1000
    requires_confirmation: bool = False
    data_access_level: str = "read"  # read, write, admin


class ToolAccessController:
    """工具访问控制器 — 支持多维度权限检查"""

    def __init__(self):
        self.tool_permissions: Dict[str, ToolPermission] = {}
        self.hourly_counts: Dict[str, Dict[str, int]] = defaultdict(lambda: defaultdict(int))
        self.daily_counts: Dict[str, Dict[str, int]] = defaultdict(lambda: defaultdict(int))
        self.call_history: List[Dict] = []

    def register_tool(self, permission: ToolPermission):
        self.tool_permissions[permission.tool_name] = permission

    def check_access(self, user_id: str, user_role: str,
                     tool_name: str, params: Dict,
                     data_access_level: str = "read") -> Dict:
        perm = self.tool_permissions.get(tool_name)
        if not perm:
            return {"allowed": False, "reason": "Tool not found"}

        if user_role not in perm.allowed_roles:
            return {"allowed": False, "reason": f"Role '{user_role}' not allowed"}

        for param in perm.required_params:
            if param not in params:
                return {"allowed": False, "reason": f"Missing required param: {param}"}

        if not self._check_hourly_limit(user_id, tool_name, perm.max_calls_per_hour):
            return {"allowed": False, "reason": "Hourly rate limit exceeded"}

        if not self._check_daily_limit(user_id, tool_name, perm.max_calls_per_day):
            return {"allowed": False, "reason": "Daily rate limit exceeded"}

        level_hierarchy = {"read": 0, "write": 1, "admin": 2}
        if level_hierarchy.get(data_access_level, 0) > level_hierarchy.get(perm.data_access_level, 0):
            return {"allowed": False, "reason": "Insufficient data access level"}

        self._record_call(user_id, tool_name)
        return {
            "allowed": True,
            "requires_confirmation": perm.requires_confirmation,
            "remaining_hourly": perm.max_calls_per_hour - self.hourly_counts[user_id][tool_name],
        }

    def _check_hourly_limit(self, user_id: str, tool_name: str, max_calls: int) -> bool:
        return self.hourly_counts[user_id][tool_name] < max_calls

    def _check_daily_limit(self, user_id: str, tool_name: str, max_calls: int) -> bool:
        return self.daily_counts[user_id][tool_name] < max_calls

    def _record_call(self, user_id: str, tool_name: str):
        self.hourly_counts[user_id][tool_name] += 1
        self.daily_counts[user_id][tool_name] += 1
        self.call_history.append({
            "user_id": user_id,
            "tool_name": tool_name,
            "timestamp": datetime.now().isoformat(),
        })

    def get_usage_stats(self, user_id: str) -> Dict:
        return {
            "hourly": dict(self.hourly_counts.get(user_id, {})),
            "daily": dict(self.daily_counts.get(user_id, {})),
            "total_calls": len([h for h in self.call_history if h["user_id"] == user_id]),
        }


# 使用示例
controller = ToolAccessController()

controller.register_tool(ToolPermission(
    tool_name="web_search",
    allowed_roles=["user", "admin"],
    max_calls_per_hour=50,
    data_access_level="read",
))

controller.register_tool(ToolPermission(
    tool_name="database_query",
    allowed_roles=["admin"],
    required_params=["query", "database"],
    max_calls_per_hour=20,
    max_calls_per_day=200,
    requires_confirmation=True,
    data_access_level="read",
))

controller.register_tool(ToolPermission(
    tool_name="file_delete",
    allowed_roles=["admin"],
    required_params=["path", "confirm"],
    max_calls_per_hour=5,
    requires_confirmation=True,
    data_access_level="admin",
))

# 检查访问
result = controller.check_access("user1", "user", "web_search", {})
print(f"web_search: {result}")

result = controller.check_access("user1", "user", "database_query", {"query": "SELECT 1"})
print(f"database_query: {result}")

result = controller.check_access("admin1", "admin", "database_query",
                                  {"query": "SELECT 1", "database": "main"})
print(f"admin database_query: {result}")
```

## 5. 多租户权限隔离

### 5.1 租户隔离实现

```python
from typing import Dict, Optional
from dataclasses import dataclass, field


@dataclass
class Tenant:
    tenant_id: str
    name: str
    plan: str  # free, pro, enterprise
    max_users: int = 10
    max_requests_per_day: int = 10000
    allowed_models: list = field(default_factory=lambda: ["gpt-3.5-turbo"])
    data_region: str = "us-east-1"
    is_active: bool = True


class MultiTenantManager:
    """多租户管理器"""

    def __init__(self):
        self.tenants: Dict[str, Tenant] = {}
        self.user_tenant_map: Dict[str, str] = {}

    def create_tenant(self, tenant_id: str, name: str, plan: str) -> Tenant:
        plan_limits = {
            "free": {"max_users": 5, "max_requests_per_day": 1000, "allowed_models": ["gpt-3.5-turbo"]},
            "pro": {"max_users": 50, "max_requests_per_day": 50000, "allowed_models": ["gpt-3.5-turbo", "gpt-4"]},
            "enterprise": {"max_users": 500, "max_requests_per_day": 500000, "allowed_models": ["gpt-3.5-turbo", "gpt-4", "gpt-4-turbo"]},
        }
        limits = plan_limits.get(plan, plan_limits["free"])
        tenant = Tenant(tenant_id=tenant_id, name=name, plan=plan, **limits)
        self.tenants[tenant_id] = tenant
        return tenant

    def add_user_to_tenant(self, user_id: str, tenant_id: str) -> bool:
        if tenant_id not in self.tenants:
            return False
        tenant = self.tenants[tenant_id]
        current_users = sum(1 for uid, tid in self.user_tenant_map.items() if tid == tenant_id)
        if current_users >= tenant.max_users:
            return False
        self.user_tenant_map[user_id] = tenant_id
        return True

    def check_model_access(self, user_id: str, model_name: str) -> bool:
        tenant_id = self.user_tenant_map.get(user_id)
        if not tenant_id:
            return False
        tenant = self.tenants.get(tenant_id)
        if not tenant or not tenant.is_active:
            return False
        return model_name in tenant.allowed_models

    def get_tenant_context(self, user_id: str) -> Optional[Dict]:
        tenant_id = self.user_tenant_map.get(user_id)
        if not tenant_id:
            return None
        tenant = self.tenants.get(tenant_id)
        if not tenant:
            return None
        return {
            "tenant_id": tenant.tenant_id,
            "plan": tenant.plan,
            "allowed_models": tenant.allowed_models,
            "data_region": tenant.data_region,
        }


# 使用示例
mtm = MultiTenantManager()
mtm.create_tenant("t1", "Acme Corp", "pro")
mtm.add_user_to_tenant("user1", "t1")

print(mtm.check_model_access("user1", "gpt-4"))      # True
print(mtm.check_model_access("user1", "gpt-4-turbo")) # False (pro plan)
```

## 6. 常见误区

1. **"权限控制只在入口"**: 需要在每层都验证，包括服务间调用
2. **"角色足够"**: 可能需要更细粒度的控制，ABAC 或 ReBAC
3. **"一次配置足够"**: 需要定期审查，权限会随业务变化
4. **"忽略默认拒绝"**: 应该默认拒绝，明确授权
5. **"只关注用户"**: 服务间也需要权限控制，Agent 调用同理
6. **"权限不需要审计"**: 权限变更和使用都需要审计日志
7. **"硬编码权限"**: 权限应配置化，支持动态调整
8. **"忽略租户隔离"**: 多租户场景下数据泄露是严重事故

## 工程建议

权限控制的设计应遵循默认拒绝、显式授权原则。系统上线时所有权限默认关闭，需要明确的配置才能开启。宁可多拒绝几次请求，也不要因为默认放行而留下安全隐患。

API 密钥必须实现完整的生命周期管理：生成时记录归属和用途，设置合理的过期时间，支持主动轮转和吊销。生产环境中，密钥应该只存储哈希值，原始密钥只在生成时返回一次。定期轮转密钥应该成为运维流程的一部分，而不是等到出了安全事件才被动更换。

工具调用权限是 AI 系统最容易被忽视的攻击面。每个 Agent 可用的工具都应该有独立的权限声明，包括允许调用的角色、频率限制、数据访问级别。高风险操作（如文件删除、数据库写入）必须要求二次确认，不能让模型直接执行。

多租户场景下，租户之间的资源隔离必须在架构层面保证，而不是依赖应用层的 if-else 判断。数据库行级隔离、独立的模型配额、独立的存储命名空间，这些措施能有效防止一个租户的数据泄露到另一个租户。

## 总结

- RBAC 和 ABAC 是常用的权限控制模型，可组合使用
- API 密钥管理是访问控制的基础，需要完整的生命周期管理
- 工具调用需要专门的权限控制，包括频率限制和数据访问级别
- 最小权限原则是核心，宁可少给不可多给
- 多租户场景需要严格的资源和数据隔离

### 练习 1: RBAC 实现

**思路**：RBAC 系统的核心是角色继承和权限解析。角色继承使用有向无环图（DAG）结构，需要检测循环继承以防止无限递归。权限解析时遍历角色继承链，收集所有有效权限并去重。审计功能需要输出用户的完整权限来源，便于排查权限问题。

**答案**：

```python
from typing import Dict, List, Set
from dataclasses import dataclass, field
from datetime import datetime


@dataclass
class Permission:
    name: str
    resource: str
    action: str
    description: str = ""

    def matches(self, resource: str, action: str) -> bool:
        if self.resource == "*" and self.action == "*":
            return True
        if self.resource == resource and self.action == "*":
            return True
        if self.resource == "*" and self.action == action:
            return True
        return self.resource == resource and self.action == action


@dataclass
class Role:
    name: str
    permissions: Set[str] = field(default_factory=set)
    parent_roles: Set[str] = field(default_factory=set)
    description: str = ""


@dataclass
class User:
    id: str
    roles: Set[str] = field(default_factory=set)
    is_active: bool = True


class RBACManager:
    def __init__(self):
        self.roles: Dict[str, Role] = {}
        self.users: Dict[str, User] = {}
        self.permissions: Dict[str, Permission] = {}

    def create_permission(self, name: str, resource: str, action: str, description: str = ""):
        self.permissions[name] = Permission(name, resource, action, description)

    def create_role(self, name: str, permissions: List[str] = None,
                    parent_roles: List[str] = None, description: str = ""):
        if self._would_create_cycle(name, set(parent_roles or [])):
            raise ValueError(f"创建角色 '{name}' 会导致循环继承")
        self.roles[name] = Role(
            name=name, permissions=set(permissions or []),
            parent_roles=set(parent_roles or []), description=description,
        )

    def _would_create_cycle(self, new_role: str, parents: Set[str], visited: Set[str] = None) -> bool:
        if visited is None:
            visited = set()
        if new_role in parents:
            return True
        for parent in parents:
            if parent in visited:
                continue
            visited.add(parent)
            parent_role = self.roles.get(parent)
            if parent_role and self._would_create_cycle(new_role, parent_role.parent_roles, visited):
                return True
        return False

    def create_user(self, user_id: str, roles: List[str] = None):
        self.users[user_id] = User(id=user_id, roles=set(roles or []))

    def assign_role(self, user_id: str, role_name: str):
        if user_id in self.users and role_name in self.roles:
            self.users[user_id].roles.add(role_name)

    def revoke_role(self, user_id: str, role_name: str):
        if user_id in self.users:
            self.users[user_id].roles.discard(role_name)

    def _get_effective_permissions(self, role_name: str, visited: Set[str] = None) -> Set[str]:
        if visited is None:
            visited = set()
        if role_name in visited:
            return set()
        visited.add(role_name)
        role = self.roles.get(role_name)
        if not role:
            return set()
        perms = set(role.permissions)
        for parent in role.parent_roles:
            perms.update(self._get_effective_permissions(parent, visited))
        return perms

    def check_permission(self, user_id: str, resource: str, action: str) -> bool:
        if user_id not in self.users or not self.users[user_id].is_active:
            return False
        for role_name in self.users[user_id].roles:
            for perm_name in self._get_effective_permissions(role_name):
                perm = self.permissions.get(perm_name)
                if perm and perm.matches(resource, action):
                    return True
        return False

    def audit_user(self, user_id: str) -> Dict:
        user = self.users.get(user_id)
        if not user:
            return {"error": "用户不存在"}
        report = {"user_id": user_id, "is_active": user.is_active, "roles": {}, "all_permissions": []}
        seen_perms = set()
        for role_name in user.roles:
            perm_names = self._get_effective_permissions(role_name)
            report["roles"][role_name] = list(perm_names)
            for pn in perm_names:
                if pn not in seen_perms:
                    seen_perms.add(pn)
                    p = self.permissions[pn]
                    report["all_permissions"].append({"name": p.name, "resource": p.resource, "action": p.action})
        return report


# 使用示例
rbac = RBACManager()
rbac.create_permission("chat:read", "chat", "read")
rbac.create_permission("chat:write", "chat", "write")
rbac.create_permission("model:manage", "model", "manage")
rbac.create_permission("admin:all", "*", "*")

rbac.create_role("viewer", ["chat:read"])
rbac.create_role("user", ["chat:write"], parent_roles=["viewer"])
rbac.create_role("admin", ["admin:all"])

rbac.create_user("u1", roles=["user"])
rbac.create_user("u2", roles=["admin"])

print(rbac.check_permission("u1", "chat", "read"))    # True
print(rbac.check_permission("u1", "model", "manage")) # False
print(rbac.check_permission("u2", "model", "manage")) # True

# 测试循环继承检测
try:
    rbac.create_role("loop_a", parent_roles=["loop_b"])
    rbac.create_role("loop_b", parent_roles=["loop_a"])
except ValueError as e:
    print(f"循环继承检测: {e}")

# 审计报告
print(rbac.audit_user("u1"))
```

**要点**：
- 循环继承检测使用 DFS 遍历，在创建角色时就阻止，而非运行时才发现
- 权限解析使用 visited 集合防止无限递归，即使循环继承被绕过也能正常终止
- 审计报告同时输出角色和权限的完整映射关系，便于安全审查

### 练习 2: ABAC 策略引擎

**思路**：ABAC 策略引擎需要支持条件的 AND/OR 组合和嵌套属性访问。核心设计是将条件分为叶子条件（单个属性比较）和组合条件（AND/OR 逻辑），通过递归评估实现任意深度的条件嵌套。策略优先级用于冲突解决，高优先级策略优先匹配。

**答案**：

```python
from typing import Dict, Any, List
from dataclasses import dataclass
from enum import Enum
import operator


class Operator(Enum):
    EQ = "=="
    NEQ = "!="
    GT = ">"
    LT = "<"
    GTE = ">="
    LTE = "<="
    IN = "in"
    NOT_IN = "not_in"
    CONTAINS = "contains"


class LogicOp(Enum):
    AND = "and"
    OR = "or"


@dataclass
class Condition:
    attribute: str
    op: Operator
    value: Any

    def evaluate(self, context: Dict) -> bool:
        parts = self.attribute.split(".")
        current = context
        for part in parts:
            if isinstance(current, dict) and part in current:
                current = current[part]
            else:
                return False
        ops = {
            Operator.EQ: operator.eq, Operator.NEQ: operator.ne,
            Operator.GT: operator.gt, Operator.LT: operator.lt,
            Operator.GTE: operator.ge, Operator.LTE: operator.le,
            Operator.IN: lambda a, b: a in b, Operator.NOT_IN: lambda a, b: a not in b,
            Operator.CONTAINS: lambda a, b: b in a,
        }
        return ops.get(self.op, lambda a, b: False)(current, self.value)


@dataclass
class CompositeCondition:
    logic: LogicOp
    conditions: list  # List[Condition | CompositeCondition]

    def evaluate(self, context: Dict) -> bool:
        results = [c.evaluate(context) for c in self.conditions]
        if self.logic == LogicOp.AND:
            return all(results)
        return any(results)


@dataclass
class PolicyRule:
    name: str
    effect: str  # "allow" or "deny"
    condition: Any  # Condition or CompositeCondition
    priority: int = 0

    def evaluate(self, context: Dict) -> str:
        if self.condition.evaluate(context):
            return self.effect
        return "not_applicable"


class ABACEngine:
    def __init__(self):
        self.policies: List[PolicyRule] = []

    def add_policy(self, policy: PolicyRule):
        self.policies.append(policy)
        self.policies.sort(key=lambda p: -p.priority)

    def evaluate(self, context: Dict) -> Dict:
        for policy in self.policies:
            result = policy.evaluate(context)
            if result != "not_applicable":
                return {"decision": result, "matched_policy": policy.name, "priority": policy.priority}
        return {"decision": "deny", "matched_policy": "default_deny"}


# 使用示例
engine = ABACEngine()

# 策略 1: 管理员随时允许 (高优先级)
engine.add_policy(PolicyRule(
    name="admin_always_allowed", effect="allow",
    condition=Condition("user.role", Operator.EQ, "admin"),
    priority=100,
))

# 策略 2: 工作时间 + 工程部门允许 (AND 组合)
engine.add_policy(PolicyRule(
    name="work_hours_engineering", effect="allow",
    condition=CompositeCondition(LogicOp.AND, [
        Condition("environment.hour", Operator.GTE, 9),
        Condition("environment.hour", Operator.LT, 18),
        Condition("user.department", Operator.EQ, "engineering"),
    ]),
    priority=10,
))

# 策略 3: 限制高敏感数据 (OR 组合: 非高敏感 或 高权限)
engine.add_policy(PolicyRule(
    name="sensitive_data_access", effect="deny",
    condition=CompositeCondition(LogicOp.AND, [
        Condition("resource.sensitivity", Operator.EQ, "high"),
        CompositeCondition(LogicOp.OR, [
            Condition("user.clearance_level", Operator.LT, 3),
            Condition("user.department", Operator.NOT_IN, ["security", "executive"]),
        ]),
    ]),
    priority=50,
))

# 测试
ctx1 = {"user": {"role": "admin", "department": "it", "clearance_level": 1}, "resource": {"sensitivity": "high"}, "environment": {"hour": 2}}
print(engine.evaluate(ctx1))  # allow (admin_always_allowed)

ctx2 = {"user": {"role": "user", "department": "engineering", "clearance_level": 2}, "resource": {"sensitivity": "low"}, "environment": {"hour": 14}}
print(engine.evaluate(ctx2))  # allow (work_hours_engineering)

ctx3 = {"user": {"role": "user", "department": "engineering", "clearance_level": 2}, "resource": {"sensitivity": "high"}, "environment": {"hour": 14}}
print(engine.evaluate(ctx3))  # deny (sensitive_data_access)
```

**要点**：
- CompositeCondition 支持 AND/OR 递归组合，可以表达任意复杂的条件逻辑
- 嵌套属性访问使用 "." 分隔的路径遍历，支持 `user.department` 这样的深层属性
- 策略按优先级降序排列，第一个匹配的策略即为最终决策，高优先级策略可以覆盖低优先级

### 练习 3: API 密钥管理

**思路**：生产级密钥管理系统需要完整的生命周期管理：生成、验证、轮转、吊销。密钥只在生成时返回原始值，存储时只保留哈希值。轮转操作需要原子性地吊销旧密钥并生成新密钥，避免服务中断。使用量统计需要支持按时间和维度聚合，便于发现异常使用模式。

**答案**：

```python
import hashlib
import secrets
from typing import Dict, Optional, List
from datetime import datetime, timedelta
from dataclasses import dataclass, field


@dataclass
class APIKeyInfo:
    key_hash: str
    user_id: str
    name: str
    permissions: List[str]
    created_at: datetime
    expires_at: datetime
    is_active: bool = True
    last_used_at: Optional[datetime] = None
    usage_count: int = 0
    rate_limit: int = 1000
    allowed_ips: List[str] = field(default_factory=list)
    usage_history: List[Dict] = field(default_factory=list)


class APIKeyManager:
    def __init__(self):
        self.keys: Dict[str, APIKeyInfo] = {}

    def generate_key(self, user_id: str, name: str, permissions: List[str] = None,
                     expires_days: int = 30, rate_limit: int = 1000,
                     allowed_ips: List[str] = None) -> str:
        raw_key = f"sk-{secrets.token_hex(32)}"
        key_hash = hashlib.sha256(raw_key.encode()).hexdigest()
        self.keys[key_hash] = APIKeyInfo(
            key_hash=key_hash, user_id=user_id, name=name,
            permissions=permissions or [], created_at=datetime.now(),
            expires_at=datetime.now() + timedelta(days=expires_days),
            rate_limit=rate_limit, allowed_ips=allowed_ips or [],
        )
        return raw_key

    def validate_key(self, key: str, source_ip: str = None) -> Optional[APIKeyInfo]:
        key_hash = hashlib.sha256(key.encode()).hexdigest()
        info = self.keys.get(key_hash)
        if not info or not info.is_active:
            return None
        if datetime.now() > info.expires_at:
            return None
        if info.allowed_ips and source_ip not in info.allowed_ips:
            return None
        info.last_used_at = datetime.now()
        info.usage_count += 1
        info.usage_history.append({"timestamp": datetime.now().isoformat(), "source_ip": source_ip})
        return info

    def rotate_key(self, old_key: str, expires_days: int = 30) -> Optional[str]:
        key_hash = hashlib.sha256(old_key.encode()).hexdigest()
        info = self.keys.get(key_hash)
        if not info:
            return None
        old_info = {
            "old_key_hash": key_hash[:16] + "...",
            "user_id": info.user_id,
            "usage_count": info.usage_count,
            "rotated_at": datetime.now().isoformat(),
        }
        info.is_active = False
        new_key = self.generate_key(
            user_id=info.user_id, name=info.name, permissions=info.permissions,
            expires_days=expires_days, rate_limit=info.rate_limit, allowed_ips=info.allowed_ips,
        )
        new_hash = hashlib.sha256(new_key.encode()).hexdigest()
        self.keys[new_hash].usage_history.append({"event": "rotated_from", "details": old_info})
        return new_key

    def revoke_key(self, key: str) -> bool:
        key_hash = hashlib.sha256(key.encode()).hexdigest()
        if key_hash in self.keys:
            self.keys[key_hash].is_active = False
            return True
        return False

    def get_usage_stats(self, user_id: str) -> Dict:
        user_keys = [k for k in self.keys.values() if k.user_id == user_id]
        total_usage = sum(k.usage_count for k in user_keys)
        active_keys = sum(1 for k in user_keys if k.is_active)
        return {
            "user_id": user_id,
            "total_keys": len(user_keys),
            "active_keys": active_keys,
            "total_usage": total_usage,
            "keys": [
                {"name": k.name, "usage_count": k.usage_count, "is_active": k.is_active,
                 "last_used": k.last_used_at.isoformat() if k.last_used_at else None}
                for k in user_keys
            ],
        }

    def check_usage_alert(self, user_id: str, threshold: int = 800) -> List[Dict]:
        alerts = []
        for info in self.keys.values():
            if info.user_id == user_id and info.is_active and info.usage_count > threshold:
                alerts.append({
                    "key_name": info.name, "usage_count": info.usage_count,
                    "rate_limit": info.rate_limit, "usage_ratio": round(info.usage_count / info.rate_limit * 100, 1),
                })
        return alerts


# 使用示例
manager = APIKeyManager()

key = manager.generate_key("user1", "prod-api", ["chat:read", "chat:write"],
                           expires_days=90, rate_limit=500, allowed_ips=["10.0.0.1"])
print(f"生成密钥: {key[:20]}...")

info = manager.validate_key(key, source_ip="10.0.0.1")
print(f"验证: valid={info is not None}, usage={info.usage_count}")

new_key = manager.rotate_key(key, expires_days=60)
print(f"轮转: new_key={new_key[:20]}...")

old_info = manager.validate_key(key)
print(f"旧密钥验证: valid={old_info is not None}")  # None (已吊销)

print(f"使用统计: {manager.get_usage_stats('user1')}")
```

**要点**：
- 密钥只在生成时返回原始值，存储和验证都使用 SHA-256 哈希，即使数据库泄露也无法还原
- 轮转操作是原子性的：先吊销旧密钥再生成新密钥，新密钥继承旧密钥的所有属性
- 使用量统计支持告警检测，当使用量接近速率限制时主动通知，避免服务中断

### 练习 4: 多租户隔离

**思路**：多租户隔离需要在三个层面保证安全：数据隔离（租户间数据不可互访）、模型访问控制（按租户套餐限制可用模型）、资源配额（防止一个租户耗尽共享资源）。设计时应在架构层面保证隔离，而非依赖应用层的 if-else 判断。

**答案**：

```python
from typing import Dict, Optional, List
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from collections import defaultdict


@dataclass
class TenantConfig:
    tenant_id: str
    name: str
    plan: str  # free, pro, enterprise
    max_users: int = 10
    max_requests_per_day: int = 10000
    max_tokens_per_day: int = 1000000
    allowed_models: List[str] = field(default_factory=lambda: ["gpt-3.5-turbo"])
    data_region: str = "cn-east-1"
    storage_namespace: str = ""
    is_active: bool = True

    def __post_init__(self):
        if not self.storage_namespace:
            self.storage_namespace = f"tenant-{self.tenant_id}"


@dataclass
class QuotaUsage:
    requests_today: int = 0
    tokens_today: int = 0
    last_reset: datetime = field(default_factory=datetime.now)


class MultiTenantPlatform:
    def __init__(self):
        self.tenants: Dict[str, TenantConfig] = {}
        self.user_tenant: Dict[str, str] = {}
        self.quotas: Dict[str, QuotaUsage] = defaultdict(QuotaUsage)
        self.plan_limits = {
            "free": {"max_users": 5, "max_requests_per_day": 1000, "max_tokens_per_day": 100000,
                     "allowed_models": ["gpt-3.5-turbo"]},
            "pro": {"max_users": 50, "max_requests_per_day": 50000, "max_tokens_per_day": 5000000,
                    "allowed_models": ["gpt-3.5-turbo", "gpt-4"]},
            "enterprise": {"max_users": 500, "max_requests_per_day": 500000, "max_tokens_per_day": 50000000,
                           "allowed_models": ["gpt-3.5-turbo", "gpt-4", "gpt-4-turbo"]},
        }

    def create_tenant(self, tenant_id: str, name: str, plan: str) -> TenantConfig:
        limits = self.plan_limits.get(plan, self.plan_limits["free"])
        tenant = TenantConfig(tenant_id=tenant_id, name=name, plan=plan, **limits)
        self.tenants[tenant_id] = tenant
        return tenant

    def add_user(self, user_id: str, tenant_id: str) -> bool:
        tenant = self.tenants.get(tenant_id)
        if not tenant:
            return False
        current = sum(1 for tid in self.user_tenant.values() if tid == tenant_id)
        if current >= tenant.max_users:
            return False
        self.user_tenant[user_id] = tenant_id
        return True

    def check_model_access(self, user_id: str, model: str) -> Dict:
        tenant = self._get_tenant(user_id)
        if not tenant:
            return {"allowed": False, "reason": "租户不存在"}
        if model not in tenant.allowed_models:
            return {"allowed": False, "reason": f"套餐 '{tenant.plan}' 不支持模型 '{model}'"}
        return {"allowed": True}

    def check_quota(self, user_id: str, estimated_tokens: int = 0) -> Dict:
        tenant = self._get_tenant(user_id)
        if not tenant:
            return {"allowed": False, "reason": "租户不存在"}
        quota = self.quotas[tenant.tenant_id]
        self._maybe_reset_quota(quota)
        if quota.requests_today >= tenant.max_requests_per_day:
            return {"allowed": False, "reason": "今日请求次数已用尽"}
        if quota.tokens_today + estimated_tokens > tenant.max_tokens_per_day:
            return {"allowed": False, "reason": "今日 Token 配额不足"}
        return {"allowed": True, "remaining_requests": tenant.max_requests_per_day - quota.requests_today,
                "remaining_tokens": tenant.max_tokens_per_day - quota.tokens_today}

    def record_usage(self, user_id: str, tokens_used: int):
        tenant = self._get_tenant(user_id)
        if tenant:
            quota = self.quotas[tenant.tenant_id]
            self._maybe_reset_quota(quota)
            quota.requests_today += 1
            quota.tokens_today += tokens_used

    def get_tenant_context(self, user_id: str) -> Optional[Dict]:
        tenant = self._get_tenant(user_id)
        if not tenant:
            return None
        quota = self.quotas[tenant.tenant_id]
        return {
            "tenant_id": tenant.tenant_id, "plan": tenant.plan,
            "storage_namespace": tenant.storage_namespace, "data_region": tenant.data_region,
            "allowed_models": tenant.allowed_models,
            "quota": {"requests_used": quota.requests_today, "tokens_used": quota.tokens_today},
        }

    def _get_tenant(self, user_id: str) -> Optional[TenantConfig]:
        tenant_id = self.user_tenant.get(user_id)
        return self.tenants.get(tenant_id) if tenant_id else None

    def _maybe_reset_quota(self, quota: QuotaUsage):
        if datetime.now().date() > quota.last_reset.date():
            quota.requests_today = 0
            quota.tokens_today = 0
            quota.last_reset = datetime.now()


# 使用示例
platform = MultiTenantPlatform()
platform.create_tenant("acme", "Acme Corp", "pro")
platform.create_tenant("startup", "Startup Inc", "free")

platform.add_user("alice", "acme")
platform.add_user("bob", "startup")

print(platform.check_model_access("alice", "gpt-4"))       # allowed
print(platform.check_model_access("bob", "gpt-4"))         # denied
print(platform.check_quota("alice", estimated_tokens=500))  # allowed

platform.record_usage("alice", 500)
print(platform.get_tenant_context("alice"))
```

**要点**：
- 存储命名空间在创建租户时自动生成，确保数据在物理层面隔离
- 模型访问控制基于套餐配置，不同套餐有不同的模型白名单，无需硬编码 if-else
- 资源配额按天自动重置，支持请求数和 Token 数双重限制，防止单个租户耗尽共享资源
