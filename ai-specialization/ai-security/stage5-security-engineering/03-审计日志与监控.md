## 场景引入

2023 年底，某 AI 平台遭遇了一次持续数天的慢速数据窃取攻击。攻击者通过一个低权限账户，以每分钟 2-3 次的频率向模型发送精心构造的 Prompt，逐步提取训练数据中的敏感信息。由于系统没有完善的审计日志和监控告警机制，这种低频异常行为完全未被发现。直到一名运维人员偶然在数据库慢查询日志中注意到异常的数据读取模式，才意识到系统已经被入侵——此时攻击者已经持续活动了 11 天。事后复盘，如果系统对用户请求频率、模型调用模式和异常输出内容有完整的审计日志和实时监控，这次攻击在第一天就能被发现和阻断。

```python
from typing import Dict, Any, Optional, List
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
import json
import hashlib
import uuid
import threading
from collections import defaultdict


class LogLevel(Enum):
    DEBUG = "debug"
    INFO = "info"
    WARNING = "warning"
    ERROR = "error"
    SECURITY = "security"


@dataclass
class AuditLog:
    log_id: str
    timestamp: datetime
    level: LogLevel
    event_type: str
    user_id: Optional[str]
    ip_address: Optional[str]
    details: Dict[str, Any]
    metadata: Dict[str, Any] = field(default_factory=dict)
    trace_id: Optional[str] = None
    session_id: Optional[str] = None


class LogStorage:
    """日志存储后端接口"""

    def store(self, log: AuditLog):
        raise NotImplementedError

    def query(self, filters: Dict = None, start_time: datetime = None,
              end_time: datetime = None, limit: int = 100) -> List[AuditLog]:
        raise NotImplementedError


class InMemoryStorage(LogStorage):
    """内存存储后端（开发/测试用）"""

    def __init__(self, max_size: int = 10000):
        self.logs: List[AuditLog] = []
        self.max_size = max_size
        self._lock = threading.Lock()

    def store(self, log: AuditLog):
        with self._lock:
            self.logs.append(log)
            if len(self.logs) > self.max_size:
                self.logs = self.logs[-self.max_size:]

    def query(self, filters: Dict = None, start_time: datetime = None,
              end_time: datetime = None, limit: int = 100) -> List[AuditLog]:
        with self._lock:
            results = self.logs

            if start_time:
                results = [l for l in results if l.timestamp >= start_time]
            if end_time:
                results = [l for l in results if l.timestamp <= end_time]
            if filters:
                for key, value in filters.items():
                    if key == "level":
                        results = [l for l in results if l.level == value]
                    elif key == "event_type":
                        results = [l for l in results if l.event_type == value]
                    elif key == "user_id":
                        results = [l for l in results if l.user_id == value]

            return results[-limit:]


class AuditLogger:
    """审计日志记录器"""

    def __init__(self, storage: LogStorage = None):
        self.storage = storage or InMemoryStorage()
        self.handlers: List[Any] = []
        self._stats = defaultdict(int)

    def _generate_id(self) -> str:
        return uuid.uuid4().hex[:12]

    def log(self, level: LogLevel, event_type: str,
            user_id: str = None, ip_address: str = None,
            details: Dict = None, trace_id: str = None,
            session_id: str = None, **kwargs):
        log_entry = AuditLog(
            log_id=self._generate_id(),
            timestamp=datetime.now(),
            level=level,
            event_type=event_type,
            user_id=user_id,
            ip_address=ip_address,
            details=details or {},
            metadata=kwargs,
            trace_id=trace_id,
            session_id=session_id,
        )

        self.storage.store(log_entry)
        self._stats[level.value] += 1

        for handler in self.handlers:
            try:
                handler.handle(log_entry)
            except Exception:
                pass

        return log_entry

    def log_request(self, user_id: str, endpoint: str,
                    method: str, status_code: int,
                    processing_time: float, ip_address: str = None):
        return self.log(
            level=LogLevel.INFO,
            event_type="api_request",
            user_id=user_id,
            ip_address=ip_address,
            details={
                "endpoint": endpoint,
                "method": method,
                "status_code": status_code,
                "processing_time_ms": round(processing_time * 1000, 2),
            },
        )

    def log_security_event(self, event_type: str, user_id: str,
                           threat_level: str, details: Dict,
                           ip_address: str = None):
        return self.log(
            level=LogLevel.SECURITY,
            event_type=event_type,
            user_id=user_id,
            ip_address=ip_address,
            details={"threat_level": threat_level, **details},
        )

    def log_error(self, error_type: str, error_message: str,
                  user_id: str = None, stack_trace: str = None):
        return self.log(
            level=LogLevel.ERROR,
            event_type=error_type,
            user_id=user_id,
            details={"error_message": error_message, "stack_trace": stack_trace},
        )

    def log_model_call(self, user_id: str, model_name: str,
                       input_tokens: int, output_tokens: int,
                       latency_ms: float, success: bool):
        return self.log(
            level=LogLevel.INFO,
            event_type="model_call",
            user_id=user_id,
            details={
                "model": model_name,
                "input_tokens": input_tokens,
                "output_tokens": output_tokens,
                "latency_ms": latency_ms,
                "success": success,
            },
        )

    def get_stats(self) -> Dict:
        return dict(self._stats)

    def query_logs(self, **kwargs) -> List[AuditLog]:
        return self.storage.query(**kwargs)
```

### 2.2 监控实现

```python
from typing import Dict, List, Callable, Optional
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
from collections import deque
import statistics
import threading
import time


class AlertSeverity(Enum):
    INFO = "info"
    WARNING = "warning"
    CRITICAL = "critical"


class AlertStatus(Enum):
    FIRING = "firing"
    RESOLVED = "resolved"
    ACKNOWLEDGED = "acknowledged"


@dataclass
class Metric:
    name: str
    value: float
    timestamp: datetime
    labels: Dict[str, str] = field(default_factory=dict)


@dataclass
class AlertRule:
    name: str
    metric_name: str
    condition: str  # "gt", "lt", "eq", "anomaly"
    threshold: float
    severity: AlertSeverity
    duration_seconds: int = 0
    description: str = ""


@dataclass
class Alert:
    alert_id: str
    rule_name: str
    metric_name: str
    severity: AlertSeverity
    status: AlertStatus
    current_value: float
    threshold: float
    message: str
    fired_at: datetime
    resolved_at: Optional[datetime] = None
    acknowledged_at: Optional[datetime] = None


class MonitoringSystem:
    """实时监控系统"""

    def __init__(self, window_size: int = 1000):
        self.metrics: Dict[str, deque] = {}
        self.alert_rules: Dict[str, AlertRule] = {}
        self.active_alerts: Dict[str, Alert] = {}
        self.alert_history: List[Alert] = []
        self.alert_handlers: List[Callable] = []
        self.window_size = window_size
        self._lock = threading.Lock()

    def register_metric(self, name: str):
        with self._lock:
            self.metrics[name] = deque(maxlen=self.window_size)

    def record_metric(self, name: str, value: float, labels: Dict = None):
        with self._lock:
            if name not in self.metrics:
                self.metrics[name] = deque(maxlen=self.window_size)
            self.metrics[name].append(Metric(
                name=name, value=value,
                timestamp=datetime.now(), labels=labels or {},
            ))

        self._evaluate_rules(name, value)

    def add_alert_rule(self, rule: AlertRule):
        self.alert_rules[rule.name] = rule

    def add_alert_handler(self, handler: Callable):
        self.alert_handlers.append(handler)

    def _evaluate_rules(self, metric_name: str, value: float):
        for rule_name, rule in self.alert_rules.items():
            if rule.metric_name != metric_name:
                continue

            triggered = False
            if rule.condition == "gt" and value > rule.threshold:
                triggered = True
            elif rule.condition == "lt" and value < rule.threshold:
                triggered = True
            elif rule.condition == "eq" and value == rule.threshold:
                triggered = True

            if triggered and rule_name not in self.active_alerts:
                alert = Alert(
                    alert_id=f"alert_{int(time.time())}_{rule_name}",
                    rule_name=rule_name,
                    metric_name=metric_name,
                    severity=rule.severity,
                    status=AlertStatus.FIRING,
                    current_value=value,
                    threshold=rule.threshold,
                    message=f"{metric_name} = {value} {rule.condition} {rule.threshold}",
                    fired_at=datetime.now(),
                )
                self.active_alerts[rule_name] = alert
                self.alert_history.append(alert)
                self._notify_alert(alert)

            elif not triggered and rule_name in self.active_alerts:
                alert = self.active_alerts.pop(rule_name)
                alert.status = AlertStatus.RESOLVED
                alert.resolved_at = datetime.now()
                self._notify_alert(alert)

    def _notify_alert(self, alert: Alert):
        for handler in self.alert_handlers:
            try:
                handler(alert)
            except Exception:
                pass

    def get_metric_stats(self, name: str, window_minutes: int = 5) -> Dict:
        with self._lock:
            if name not in self.metrics:
                return {}

            cutoff = datetime.now() - timedelta(minutes=window_minutes)
            values = [m.value for m in self.metrics[name] if m.timestamp >= cutoff]

            if not values:
                return {}

            return {
                "count": len(values),
                "min": min(values),
                "max": max(values),
                "avg": statistics.mean(values),
                "median": statistics.median(values),
                "stddev": statistics.stdev(values) if len(values) > 1 else 0,
                "p95": sorted(values)[int(len(values) * 0.95)] if len(values) >= 2 else values[0],
                "latest": values[-1],
            }

    def get_active_alerts(self) -> List[Alert]:
        return list(self.active_alerts.values())

    def acknowledge_alert(self, alert_id: str) -> bool:
        for alert in self.active_alerts.values():
            if alert.alert_id == alert_id:
                alert.status = AlertStatus.ACKNOWLEDGED
                alert.acknowledged_at = datetime.now()
                return True
        return False


# 使用示例
monitor = MonitoringSystem()

# 注册指标
monitor.register_metric("request_count")
monitor.register_metric("error_rate")
monitor.register_metric("response_time")
monitor.register_metric("injection_attempts")

# 添加告警规则
monitor.add_alert_rule(AlertRule(
    name="high_error_rate",
    metric_name="error_rate",
    condition="gt",
    threshold=0.05,
    severity=AlertSeverity.CRITICAL,
    description="错误率超过 5%",
))

monitor.add_alert_rule(AlertRule(
    name="slow_response",
    metric_name="response_time",
    condition="gt",
    threshold=5.0,
    severity=AlertSeverity.WARNING,
    description="响应时间超过 5 秒",
))

monitor.add_alert_rule(AlertRule(
    name="injection_spike",
    metric_name="injection_attempts",
    condition="gt",
    threshold=10,
    severity=AlertSeverity.CRITICAL,
    description="注入攻击尝试激增",
))

# 添加告警处理器
def alert_handler(alert: Alert):
    print(f"[{alert.severity.value.upper()}] {alert.message}")

monitor.add_alert_handler(alert_handler)

# 模拟数据
monitor.record_metric("error_rate", 0.02)
monitor.record_metric("error_rate", 0.08)  # 触发告警
monitor.record_metric("response_time", 2.5)
monitor.record_metric("response_time", 6.0)  # 触发告警

# 查看统计
stats = monitor.get_metric_stats("response_time")
print(f"Response time stats: {stats}")
```

## 3. 日志分析引擎

### 3.1 威胁检测

```python
from typing import Dict, List, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta
from collections import Counter
import re


@dataclass
class ThreatPattern:
    name: str
    description: str
    pattern_type: str  # "frequency", "sequence", "anomaly"
    threshold: float
    window_minutes: int


class ThreatAnalyzer:
    """威胁分析引擎"""

    def __init__(self):
        self.patterns: List[ThreatPattern] = []
        self.user_activity: Dict[str, List[datetime]] = {}

    def add_pattern(self, pattern: ThreatPattern):
        self.patterns.append(pattern)

    def analyze_log(self, log_entry: Dict) -> List[Dict]:
        threats = []
        user_id = log_entry.get("user_id")
        event_type = log_entry.get("event_type")
        timestamp = datetime.fromisoformat(log_entry.get("timestamp", datetime.now().isoformat()))

        if user_id:
            if user_id not in self.user_activity:
                self.user_activity[user_id] = []
            self.user_activity[user_id].append(timestamp)

        for pattern in self.patterns:
            if pattern.pattern_type == "frequency":
                result = self._check_frequency(user_id, pattern)
                if result:
                    threats.append(result)

            if pattern.pattern_type == "sequence":
                result = self._check_sequence(user_id, event_type, pattern)
                if result:
                    threats.append(result)

        return threats

    def _check_frequency(self, user_id: str, pattern: ThreatPattern) -> Optional[Dict]:
        if not user_id or user_id not in self.user_activity:
            return None

        cutoff = datetime.now() - timedelta(minutes=pattern.window_minutes)
        recent = [t for t in self.user_activity[user_id] if t >= cutoff]

        if len(recent) > pattern.threshold:
            return {
                "pattern_name": pattern.name,
                "description": pattern.description,
                "user_id": user_id,
                "count": len(recent),
                "threshold": pattern.threshold,
                "window_minutes": pattern.window_minutes,
            }
        return None

    def _check_sequence(self, user_id: str, event_type: str, pattern: ThreatPattern) -> Optional[Dict]:
        return None

    def get_user_risk_score(self, user_id: str) -> float:
        if user_id not in self.user_activity:
            return 0.0

        activity = self.user_activity[user_id]
        cutoff = datetime.now() - timedelta(hours=1)
        recent_count = len([t for t in activity if t >= cutoff])

        score = min(1.0, recent_count / 100)
        return round(score, 2)

    def generate_report(self) -> Dict:
        report = {
            "total_users_monitored": len(self.user_activity),
            "high_risk_users": [],
            "activity_summary": {},
        }

        for user_id in self.user_activity:
            risk = self.get_user_risk_score(user_id)
            if risk > 0.7:
                report["high_risk_users"].append({"user_id": user_id, "risk_score": risk})

            report["activity_summary"][user_id] = len(self.user_activity[user_id])

        return report


# 使用示例
analyzer = ThreatAnalyzer()

analyzer.add_pattern(ThreatPattern(
    name="rapid_requests",
    description="短时间内大量请求",
    pattern_type="frequency",
    threshold=50,
    window_minutes=5,
))

analyzer.add_pattern(ThreatPattern(
    name="brute_force",
    description="暴力破解尝试",
    pattern_type="frequency",
    threshold=10,
    window_minutes=1,
))

# 模拟日志分析

        "user_id": "user1",
        "event_type": "api_request",
        "timestamp": datetime.now().isoformat(),
    })

print(f"Risk report: {report}")
```

## 4. 告警通知系统

### 4.1 多渠道通知

```python
from typing import Dict, List, Callable
from dataclasses import dataclass
from datetime import datetime
from enum import Enum
import json


class NotificationChannel(Enum):
    EMAIL = "email"
    WEBHOOK = "webhook"
    SLACK = "slack"
    SMS = "sms"


@dataclass
class NotificationConfig:
    channel: NotificationChannel
    target: str  # email address, webhook URL, etc.
    min_severity: str = "warning"
    enabled: bool = True


class NotificationManager:
    """多渠道告警通知管理器"""

    def __init__(self):
        self.configs: List[NotificationConfig] = []
        self.sent_history: List[Dict] = []

    def add_config(self, config: NotificationConfig):
        self.configs.append(config)

    def send_alert(self, alert_data: Dict):
        severity_order = {"info": 0, "warning": 1, "critical": 2}
        alert_severity = severity_order.get(alert_data.get("severity", "info"), 0)

        for config in self.configs:
            if not config.enabled:
                continue
            min_sev = severity_order.get(config.min_severity, 0)
            if alert_severity < min_sev:
                continue

            self._dispatch(config, alert_data)

    def _dispatch(self, config: NotificationConfig, alert_data: Dict):
        message = self._format_message(config.channel, alert_data)

        record = {
            "channel": config.channel.value,
            "target": config.target,
            "message": message,
            "timestamp": datetime.now().isoformat(),
            "alert_id": alert_data.get("alert_id"),
        }
        self.sent_history.append(record)
        print(f"[{config.channel.value}] -> {config.target}: {message[:100]}...")

    def _format_message(self, channel: NotificationChannel, alert_data: Dict) -> str:
        if channel == NotificationChannel.SLACK:
            return json.dumps({
                "text": f":rotating_light: *{alert_data.get('severity', 'unknown').upper()}*\n"
                        f"*{alert_data.get('rule_name', 'Unknown')}*\n"
                        f"{alert_data.get('message', '')}",
            })
        elif channel == NotificationChannel.WEBHOOK:
            return json.dumps(alert_data)
        else:
            return f"[{alert_data.get('severity', 'unknown').upper()}] " \
                   f"{alert_data.get('rule_name', 'Unknown')}: {alert_data.get('message', '')}"

    def get_history(self, limit: int = 50) -> List[Dict]:
        return self.sent_history[-limit:]


# 使用示例
notifier = NotificationManager()
```

```python
import hashlib
import hmac
import json
from typing import Dict
from datetime import datetime


class SecureLogWriter:
    """安全日志写入器 — 保证日志完整性"""

    def __init__(self, secret_key: str):
        self.secret_key = secret_key.encode()
        self.previous_hash = "0" * 64

    def write_log(self, log_data: Dict) -> Dict:
        log_data["timestamp"] = datetime.now().isoformat()
        log_data["previous_hash"] = self.previous_hash

        content = json.dumps(log_data, sort_keys=True)
        signature = hmac.new(self.secret_key, content.encode(), hashlib.sha256).hexdigest()
        chain_hash = hashlib.sha256((content + self.previous_hash).encode()).hexdigest()

        secure_log = {
            **log_data,
            "signature": signature,
            "chain_hash": chain_hash,
        }

        self.previous_hash = chain_hash
        return secure_log

    def verify_chain(self, logs: list) -> bool:
        for i, log in enumerate(logs):
            if i == 0:
                expected_prev = "0" * 64
            else:
                expected_prev = logs[i - 1].get("chain_hash", "")

            if log.get("previous_hash") != expected_prev:
                return False

        return True
```

## 6. 常见误区

1. **"日志太多没用"**: 需要平衡详细度和存储成本，分级别保留
2. **"只记录错误"**: 需要记录所有关键事件，包括正常操作
3. **"监控只看数值"**: 需要分析趋势和异常，不能只看瞬时值
4. **"告警越多越好"**: 需要避免告警疲劳，分级分渠道
5. **"日志不需要安全"**: 日志本身也需要保护，防篡改防泄露
6. **"日志不需要索引"**: 没有索引的日志无法快速查询
7. **"监控只看技术指标"**: 业务指标同样重要（如用户满意度）
8. **"告警不需要升级"**: 未处理的告警需要自动升级

## 工程建议

审计日志应该采用结构化格式（JSON），包含统一的字段：时间戳、事件类型、用户标识、IP 地址、请求详情、响应状态、trace_id。结构化日志在后续的检索、聚合和告警中远比自由文本高效。日志中绝对不能记录用户输入的原始 Prompt 和模型输出的完整内容——这些数据量巨大且可能包含敏感信息，应该只记录摘要或哈希值。

日志保留策略需要平衡存储成本和合规要求。安全事件日志至少保留一年，普通 API 请求日志保留 30 天即可。热存储（Elasticsearch）用于近 7 天的实时查询，温存储（对象存储）用于历史数据的离线分析。超过保留期限的日志应该安全删除，而不是无限堆积。

监控告警必须分级分渠道。低优先级的信息类告警可以走邮件汇总，中优先级的异常走即时通讯工具，高优先级的安全事件（如注入攻击检测、权限异常变更）必须同时触发短信和电话通知。告警规则应该设置合理的冷却时间，避免同一事件反复触发导致告警疲劳。

日志本身的完整性保护容易被忽略但至关重要。生产环境中，审计日志应该使用链式哈希或 HMAC 签名，确保日志一旦写入就无法被篡改。日志存储应该有独立的访问控制，运维人员只能读取不能删除，删除操作需要走审批流程并留痕。

## 总结

- 审计日志是安全事件追溯的基础，需要完整的生命周期管理
- 实时监控可以及时发现安全问题，需要合理的指标和告警规则
- 合理的告警机制可以快速响应，需要避免告警疲劳
- 日志分析可以发现潜在威胁，需要结合规则和异常检测
- 日志本身的安全和合规同样重要

```python
from typing import Dict, List, Optional, Any
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
import json
import uuid
import threading
from collections import defaultdict


class LogLevel(Enum):
    DEBUG = "debug"
    INFO = "info"
    WARNING = "warning"
    ERROR = "error"
    SECURITY = "security"


@dataclass
class AuditLog:
    log_id: str
    timestamp: datetime
    level: LogLevel
    event_type: str
    user_id: Optional[str]
    ip_address: Optional[str]
    details: Dict[str, Any]
    trace_id: Optional[str] = None
    session_id: Optional[str] = None


class TieredStorage:
    def __init__(self):
        self.hot: List[AuditLog] = []      # 7 天内
        self.warm: List[AuditLog] = []     # 30 天内
        self.cold: List[AuditLog] = []     # 归档
        self._lock = threading.Lock()
        self.retention = {
            LogLevel.DEBUG: timedelta(days=7),
            LogLevel.INFO: timedelta(days=30),
            LogLevel.WARNING: timedelta(days=90),
            LogLevel.ERROR: timedelta(days=180),
            LogLevel.SECURITY: timedelta(days=365),
        }

    def store(self, log: AuditLog):
        with self._lock:
            self.hot.append(log)

    def query(self, filters: Dict = None, start_time: datetime = None,
              end_time: datetime = None, limit: int = 100) -> List[AuditLog]:
        with self._lock:
            results = self.hot + self.warm + self.cold
            if start_time:
                results = [l for l in results if l.timestamp >= start_time]
            if end_time:
                results = [l for l in results if l.timestamp <= end_time]
            if filters:
                for key, value in filters.items():
                    if key == "level":
                        results = [l for l in results if l.level == value]
                    elif key == "event_type":
                        results = [l for l in results if l.event_type == value]
                    elif key == "user_id":
                        results = [l for l in results if l.user_id == value]
            return sorted(results, key=lambda l: l.timestamp)[-limit:]

    def archive(self):
        now = datetime.now()
        with self._lock:
            new_hot = []
            for log in self.hot:
                age = now - log.timestamp
                max_age = self.retention.get(log.level, timedelta(days=30))
                if age > max_age:
                    continue
                elif age > timedelta(days=7):
                    self.warm.append(log)
                else:
                    new_hot.append(log)
            self.hot = new_hot


class AuditLogger:
    def __init__(self, storage: TieredStorage = None):
        self.storage = storage or TieredStorage()
        self._stats = defaultdict(int)

    def log(self, level: LogLevel, event_type: str, user_id: str = None,
            ip_address: str = None, details: Dict = None, trace_id: str = None,
            session_id: str = None):
        entry = AuditLog(
            log_id=uuid.uuid4().hex[:12], timestamp=datetime.now(),
            level=level, event_type=event_type, user_id=user_id,
            ip_address=ip_address, details=details or {},
            trace_id=trace_id, session_id=session_id,
        )
        self.storage.store(entry)
        self._stats[level.value] += 1
        return entry

    def log_request(self, user_id: str, endpoint: str, method: str,
                    status_code: int, processing_time: float, ip_address: str = None):
        return self.log(LogLevel.INFO, "api_request", user_id, ip_address, {
            "endpoint": endpoint, "method": method,
            "status_code": status_code, "processing_time_ms": round(processing_time * 1000, 2),
        })

    def log_security_event(self, event_type: str, user_id: str, threat_level: str,
                           details: Dict, ip_address: str = None):
        return self.log(LogLevel.SECURITY, event_type, user_id, ip_address,
                        {"threat_level": threat_level, **details})

    def log_model_call(self, user_id: str, model: str, input_tokens: int,
                       output_tokens: int, latency_ms: float, success: bool):
        return self.log(LogLevel.INFO, "model_call", user_id, details={
            "model": model, "input_tokens": input_tokens,
            "output_tokens": output_tokens, "latency_ms": latency_ms, "success": success,
        })

    def get_stats(self) -> Dict:
        return dict(self._stats)

    def query_logs(self, **kwargs) -> List[AuditLog]:
        return self.storage.query(**kwargs)
```

**思路**：实时监控系统需要支持指标收集、阈值告警和趋势分析三个核心功能。指标收集使用滑动窗口存储最近 N 个数据点，阈值告警在指标超过阈值时触发并支持自动恢复检测，趋势分析通过统计指标（均值、P95、标准差）识别异常。

**答案**：

```python
from typing import Dict, List, Callable, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta
from enum import Enum
from collections import deque
import statistics
import threading
import time


class AlertSeverity(Enum):
    INFO = "info"
    WARNING = "warning"
    CRITICAL = "critical"


@dataclass
class Metric:
    name: str
    value: float
    timestamp: datetime
    labels: Dict[str, str]


@dataclass
class AlertRule:
    name: str
    metric_name: str
    condition: str  # "gt", "lt", "anomaly"
    threshold: float
    severity: AlertSeverity
    cooldown_seconds: int = 300
    description: str = ""


@dataclass
class Alert:
    alert_id: str
    rule_name: str
    severity: AlertSeverity
    current_value: float
    threshold: float
    message: str
    fired_at: datetime
    resolved_at: Optional[datetime] = None


class MonitoringSystem:
    def __init__(self, window_size: int = 1000):
        self.metrics: Dict[str, deque] = {}
        self.alert_rules: Dict[str, AlertRule] = {}
        self.active_alerts: Dict[str, Alert] = {}
        self.alert_history: List[Alert] = []
        self.alert_handlers: List[Callable] = []
        self.last_alert_time: Dict[str, datetime] = {}
        self.window_size = window_size
        self._lock = threading.Lock()

    def register_metric(self, name: str):
        with self._lock:
            self.metrics[name] = deque(maxlen=self.window_size)

    def record_metric(self, name: str, value: float, labels: Dict = None):
        with self._lock:
            if name not in self.metrics:
                self.metrics[name] = deque(maxlen=self.window_size)
            self.metrics[name].append(Metric(name=name, value=value, timestamp=datetime.now(), labels=labels or {}))
        self._evaluate_rules(name, value)

    def add_alert_rule(self, rule: AlertRule):
        self.alert_rules[rule.name] = rule

    def add_alert_handler(self, handler: Callable):
        self.alert_handlers.append(handler)

    def _evaluate_rules(self, metric_name: str, value: float):
        for rule_name, rule in self.alert_rules.items():
            if rule.metric_name != metric_name:
                continue
            triggered = False
            if rule.condition == "gt" and value > rule.threshold:
                triggered = True
            elif rule.condition == "lt" and value < rule.threshold:
                triggered = True

            now = datetime.now()
            last = self.last_alert_time.get(rule_name)
            if triggered and rule_name not in self.active_alerts:
                if last and (now - last).total_seconds() < rule.cooldown_seconds:
                    continue
                alert = Alert(
                    alert_id=f"alert_{int(time.time())}_{rule_name}",
                    rule_name=rule_name, severity=rule.severity,
                    current_value=value, threshold=rule.threshold,
                    message=f"{metric_name} = {value} {rule.condition} {rule.threshold}",
                    fired_at=now,
                )
                self.active_alerts[rule_name] = alert
                self.alert_history.append(alert)
                self.last_alert_time[rule_name] = now
                self._notify(alert)
            elif not triggered and rule_name in self.active_alerts:
                alert = self.active_alerts.pop(rule_name)
                alert.resolved_at = now
                self._notify(alert)

    def _notify(self, alert: Alert):
        for handler in self.alert_handlers:
            try:
                handler(alert)
            except Exception:
                pass

    def get_metric_stats(self, name: str, window_minutes: int = 5) -> Dict:
        with self._lock:
            if name not in self.metrics:
                return {}
            cutoff = datetime.now() - timedelta(minutes=window_minutes)
            values = [m.value for m in self.metrics[name] if m.timestamp >= cutoff]
            if not values:
                return {}
            sorted_vals = sorted(values)
            return {
                "count": len(values), "min": min(values), "max": max(values),
                "avg": round(statistics.mean(values), 2),
                "median": round(statistics.median(values), 2),
                "stddev": round(statistics.stdev(values), 2) if len(values) > 1 else 0,
                "p95": round(sorted_vals[int(len(values) * 0.95)] if len(values) >= 2 else values[0], 2),
                "latest": values[-1],
            }

    def detect_trend(self, name: str, window_minutes: int = 10) -> Dict:
        with self._lock:
            if name not in self.metrics:
                return {"trend": "unknown"}
            cutoff = datetime.now() - timedelta(minutes=window_minutes)
            values = [m.value for m in self.metrics[name] if m.timestamp >= cutoff]
            if len(values) < 3:
                return {"trend": "insufficient_data"}
            half = len(values) // 2
            first_half_avg = statistics.mean(values[:half])
            second_half_avg = statistics.mean(values[half:])
            change_pct = (second_half_avg - first_half_avg) / first_half_avg * 100 if first_half_avg else 0
            if abs(change_pct) < 5:
                trend = "stable"
            elif change_pct > 0:
                trend = "increasing"
            else:
                trend = "decreasing"
            return {"trend": trend, "change_pct": round(change_pct, 2),
                    "first_half_avg": round(first_half_avg, 2), "second_half_avg": round(second_half_avg, 2)}


# 使用示例
monitor = MonitoringSystem()
monitor.register_metric("request_count")
monitor.register_metric("error_rate")
monitor.register_metric("response_time")

monitor.add_alert_rule(AlertRule("high_error_rate", "error_rate", "gt", 0.05,
                                  AlertSeverity.CRITICAL, description="错误率超过 5%"))
monitor.add_alert_rule(AlertRule("slow_response", "response_time", "gt", 5.0,
                                  AlertSeverity.WARNING, description="响应超过 5 秒"))

monitor.add_alert_handler(lambda a: print(f"[{a.severity.value.upper()}] {a.message}"))

for v in [0.02, 0.03, 0.04, 0.08, 0.06]:
    monitor.record_metric("error_rate", v)
for v in [1.0, 2.0, 3.0, 6.0, 4.0]:
    monitor.record_metric("response_time", v)

print(f"错误率统计: {monitor.get_metric_stats('error_rate')}")
print(f"趋势分析: {monitor.detect_trend('response_time')}")
```

**要点**：
- 告警冷却机制防止同一事件反复触发，避免告警疲劳
- 趋势分析通过对比前后半段均值识别指标变化方向，适合发现慢速攻击
- 统计指标包含 P95 和标准差，P95 反映尾部延迟，标准差反映波动程度

### 练习 3: 威胁分析

**思路**：威胁分析引擎需要识别两类异常：频率异常（短时间内大量请求）和行为序列异常（特定操作模式）。使用用户行为时间序列和威胁规则匹配，结合风险评分机制对用户进行分级。高频请求和异常序列都会累积风险分，分数超过阈值时触发告警。

**答案**：

```python
from typing import Dict, List, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta
from collections import defaultdict
import re


@dataclass
class ThreatPattern:
    name: str
    description: str
    pattern_type: str  # "frequency", "sequence", "content"
    threshold: float
    window_minutes: int
    risk_score: float = 0.5


@dataclass
class ThreatEvent:
    user_id: str
    pattern_name: str
    description: str
    risk_score: float
    timestamp: datetime
    details: Dict


class ThreatAnalyzer:
    def __init__(self):
        self.patterns: List[ThreatPattern] = []
        self.user_activity: Dict[str, List[datetime]] = defaultdict(list)
        self.user_events: Dict[str, List[str]] = defaultdict(list)
        self.risk_scores: Dict[str, float] = defaultdict(float)
        self.threat_events: List[ThreatEvent] = []
        self.suspicious_patterns = [
            r"ignore\s+previous", r"忽略.*指令", r"system\s*prompt",
            r"你现在是", r"jailbreak", r"bypass\s+safety",
        ]

    def add_pattern(self, pattern: ThreatPattern):
        self.patterns.append(pattern)

    def analyze_log(self, log_entry: Dict) -> List[ThreatEvent]:
        threats = []
        user_id = log_entry.get("user_id")
        event_type = log_entry.get("event_type")
        timestamp = datetime.fromisoformat(log_entry.get("timestamp", datetime.now().isoformat()))
        input_text = log_entry.get("input", "")

        if user_id:
            self.user_activity[user_id].append(timestamp)
            self.user_events[user_id].append(event_type)

        for pattern in self.patterns:
            if pattern.pattern_type == "frequency":
                threat = self._check_frequency(user_id, timestamp, pattern)
                if threat:
                    threats.append(threat)
            elif pattern.pattern_type == "sequence":
                threat = self._check_sequence(user_id, pattern)
                if threat:
                    threats.append(threat)

        if input_text:
            content_threat = self._check_content(user_id, input_text, timestamp)
            if content_threat:
                threats.append(content_threat)

        for threat in threats:
            self.risk_scores[user_id] += threat.risk_score
            self.threat_events.append(threat)

        return threats

    def _check_frequency(self, user_id: str, timestamp: datetime, pattern: ThreatPattern) -> Optional[ThreatEvent]:
        cutoff = timestamp - timedelta(minutes=pattern.window_minutes)
        recent = [t for t in self.user_activity[user_id] if t >= cutoff]
        if len(recent) > pattern.threshold:
            return ThreatEvent(
                user_id=user_id, pattern_name=pattern.name,
                description=f"{pattern.description}: {len(recent)} 次 / {pattern.window_minutes} 分钟",
                risk_score=pattern.risk_score, timestamp=timestamp,
                details={"count": len(recent), "threshold": pattern.threshold},
            )
        return None

    def _check_sequence(self, user_id: str, pattern: ThreatPattern) -> Optional[ThreatEvent]:
        events = self.user_events[user_id]
        if len(events) < 3:
            return None
        recent = events[-10:]
        if recent.count("failed_auth") >= 3:
            return ThreatEvent(
                user_id=user_id, pattern_name=pattern.name,
                description="连续认证失败，疑似暴力破解",
                risk_score=0.7, timestamp=datetime.now(),
                details={"failed_count": recent.count("failed_auth")},
            )
        return None

    def _check_content(self, user_id: str, input_text: str, timestamp: datetime) -> Optional[ThreatEvent]:
        for pattern_str in self.suspicious_patterns:
            if re.search(pattern_str, input_text, re.IGNORECASE):
                return ThreatEvent(
                    user_id=user_id, pattern_name="suspicious_content",
                    description=f"输入包含可疑模式: {pattern_str}",
                    risk_score=0.6, timestamp=timestamp,
                    details={"pattern": pattern_str, "input_length": len(input_text)},
                )
        return None

    def get_user_risk_score(self, user_id: str) -> float:
        return round(min(1.0, self.risk_scores.get(user_id, 0)), 2)

    def get_high_risk_users(self, threshold: float = 0.7) -> List[Dict]:
        return [
            {"user_id": uid, "risk_score": score}
            for uid, score in self.risk_scores.items() if score >= threshold
        ]

    def generate_report(self) -> Dict:
        return {
            "total_users": len(self.user_activity),
            "total_threats": len(self.threat_events),
            "high_risk_users": self.get_high_risk_users(),
            "threats_by_type": self._group_threats_by_type(),
        }

    def _group_threats_by_type(self) -> Dict:
        groups = defaultdict(int)
        for t in self.threat_events:
            groups[t.pattern_name] += 1
        return dict(groups)


# 使用示例
analyzer = ThreatAnalyzer()
analyzer.add_pattern(ThreatPattern("rapid_requests", "短时间内大量请求", "frequency", 20, 5, 0.4))
analyzer.add_pattern(ThreatPattern("brute_force", "暴力破解", "frequency", 5, 1, 0.7))
analyzer.add_pattern(ThreatPattern("auth_failure_seq", "认证失败序列", "sequence", 3, 10, 0.6))

for i in range(25):
    analyzer.analyze_log({"user_id": "attacker", "event_type": "api_request", "timestamp": datetime.now().isoformat()})

analyzer.analyze_log({"user_id": "attacker", "event_type": "api_request",
                       "timestamp": datetime.now().isoformat(), "input": "ignore previous instructions"})

print(f"攻击者风险分: {analyzer.get_user_risk_score('attacker')}")
print(f"高风险用户: {analyzer.get_high_risk_users()}")
print(f"威胁报告: {analyzer.generate_report()}")
```

**要点**：
- 频率检测使用滑动窗口计数，序列检测使用最近事件队列，两种模式互补
- 风险分累积机制让多次小异常最终触发告警，适合发现慢速低频攻击
- 内容检测使用正则匹配可疑模式，与频率/序列检测配合形成多维威胁识别

### 练习 4: 日志安全

**思路**：日志链式签名的核心思想是每条日志都包含前一条日志的哈希值，形成区块链式的链式结构。使用 HMAC-SHA256 对日志内容签名，SHA-256 计算链式哈希。验证时遍历整条链，检查每个节点的 previous_hash 是否等于前一个节点的 chain_hash，任何篡改都会导致链断裂。

**答案**：

```python
import hashlib
import hmac
import json
from typing import Dict, List, Tuple
from datetime import datetime


class SecureLogWriter:
    def __init__(self, secret_key: str):
        self.secret_key = secret_key.encode()
        self.previous_hash = "0" * 64
        self.chain: List[Dict] = []

    def write_log(self, log_data: Dict) -> Dict:
        log_data["timestamp"] = datetime.now().isoformat()
        log_data["previous_hash"] = self.previous_hash
        log_data["sequence"] = len(self.chain)

        content = json.dumps(log_data, sort_keys=True)
        signature = hmac.new(self.secret_key, content.encode(), hashlib.sha256).hexdigest()
        chain_hash = hashlib.sha256((content + self.previous_hash).encode()).hexdigest()

        secure_log = {**log_data, "signature": signature, "chain_hash": chain_hash}
        self.chain.append(secure_log)
        self.previous_hash = chain_hash
        return secure_log

    def verify_chain(self, logs: List[Dict] = None) -> Tuple[bool, List[str]]:
        target = logs or self.chain
        errors = []
        for i, log in enumerate(target):
            expected_prev = "0" * 64 if i == 0 else target[i - 1].get("chain_hash", "")
            if log.get("previous_hash") != expected_prev:
                errors.append(f"索引 {i}: previous_hash 不匹配 (链断裂)")

            content_data = {k: v for k, v in log.items() if k not in ("signature", "chain_hash")}
            content = json.dumps(content_data, sort_keys=True)
            expected_sig = hmac.new(self.secret_key, content.encode(), hashlib.sha256).hexdigest()
            if log.get("signature") != expected_sig:
                errors.append(f"索引 {i}: signature 不匹配 (内容被篡改)")

            expected_chain = hashlib.sha256((content + log.get("previous_hash", "")).encode()).hexdigest()
            if log.get("chain_hash") != expected_chain:
                errors.append(f"索引 {i}: chain_hash 不匹配")

        return len(errors) == 0, errors

    def tamper_detection_test(self) -> Dict:
        results = {}

        # 测试 1: 正常链验证
        valid, errors = self.verify_chain()
        results["normal_chain"] = {"valid": valid, "errors": errors}

        # 测试 2: 篡改内容
        if self.chain:
            import copy
            tampered = copy.deepcopy(self.chain)
            tampered[0]["details"] = {"tampered": True}
            valid, errors = self.verify_chain(tampered)
            results["tampered_content"] = {"valid": valid, "errors": errors}

        # 测试 3: 篡改链哈希
        if self.chain:
            import copy
            tampered = copy.deepcopy(self.chain)
            tampered[-1]["chain_hash"] = "a" * 64
            valid, errors = self.verify_chain(tampered)
            results["tampered_hash"] = {"valid": valid, "errors": errors}

        return results


# 使用示例
writer = SecureLogWriter("my-secret-key-2024")

writer.write_log({"event": "user_login", "user_id": "user1", "ip": "10.0.0.1"})
writer.write_log({"event": "api_request", "user_id": "user1", "endpoint": "/chat"})
writer.write_log({"event": "model_call", "user_id": "user1", "model": "gpt-4"})
writer.write_log({"event": "security_alert", "user_id": "user2", "type": "injection"})

valid, errors = writer.verify_chain()
print(f"链验证: valid={valid}, errors={errors}")

test_results = writer.tamper_detection_test()
for test_name, result in test_results.items():
    print(f"  {test_name}: valid={result['valid']}, errors={result['errors']}")
```

**要点**：
- 链式哈希确保日志序列的完整性，任何一条日志被篡改都会导致后续所有哈希失效
- HMAC 签名确保日志内容的真实性，没有密钥无法伪造合法签名
- 篡改检测测试覆盖三种场景：正常链、内容篡改、哈希篡改，验证防护机制的有效性
