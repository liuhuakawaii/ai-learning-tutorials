## 场景引入

某电商平台的 AI 客服系统在凌晨遭到大规模 Prompt 注入攻击，攻击者通过精心构造的输入让模型泄露了内部系统提示词和部分用户订单信息。由于团队没有事先定义事件响应流程，安全工程师花了近两小时才确认攻击范围，期间攻击仍在持续扩散。事后复盘发现：如果有一套标准化的事件响应流程，从检测到遏制的时间可以缩短到 15 分钟以内。AI 系统的安全事件与传统软件不同——攻击隐蔽、证据易丢失、影响范围难以评估，这要求我们建立专门针对 AI 系统的事件响应机制。

### 1.2 AI 安全事件特有挑战

| 挑战 | 说明 | 应对策略 |
|------|------|----------|
| 攻击隐蔽性 | Prompt 注入可能伪装成正常输入 | 多层检测 + 行为分析 |
| 影响范围广 | 单个恶意 prompt 可能影响所有用户 | 隔离 + 快速回滚 |
| 证据易丢失 | LLM 对话记录可能被覆盖 | 实时日志 + 持久化 |
| 误判成本高 | 过度拦截影响用户体验 | 分级响应 + 白名单 |
| 模型黑盒 | 难以理解模型决策过程 | 输入输出审计 + 可解释性 |

## 3. 事件管理系统

### 3.1 事件管理实现

```python
from typing import Dict, List, Optional
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
import uuid


class IncidentStatus(Enum):
    OPEN = "open"
    INVESTIGATING = "investigating"
    CONTAINING = "containing"
    REMEDIATING = "remediating"
    RESOLVED = "resolved"
    CLOSED = "closed"


@dataclass
class TimelineEntry:
    timestamp: datetime
    action: str
    actor: str
    details: str = ""


@dataclass
class Incident:
    incident_id: str
    title: str
    severity: IncidentSeverity
    incident_type: IncidentType
    status: IncidentStatus
    created_at: datetime
    updated_at: datetime
    assignee: Optional[str] = None
    description: str = ""
    timeline: List[TimelineEntry] = field(default_factory=list)
    resolution: str = ""
    root_cause: str = ""
    affected_users: int = 0
    tags: List[str] = field(default_factory=list)


class IncidentManager:
    """事件管理器"""

    def __init__(self):
        self.incidents: Dict[str, Incident] = {}
        self.escalation_rules: Dict[int, timedelta] = {
            1: timedelta(minutes=30),
            2: timedelta(hours=4),
            3: timedelta(hours=24),
            4: timedelta(hours=72),
        }

    def create_incident(self, title: str, severity: IncidentSeverity,
                        incident_type: IncidentType, description: str = "",
                        assignee: str = None, tags: List[str] = None) -> Incident:
        incident_id = f"INC-{uuid.uuid4().hex[:8].upper()}"
        now = datetime.now()

        incident = Incident(
            incident_id=incident_id,
            title=title,
            severity=severity,
            incident_type=incident_type,
            status=IncidentStatus.OPEN,
            created_at=now,
            updated_at=now,
            assignee=assignee,
            description=description,
            tags=tags or [],
        )

        self.incidents[incident_id] = incident
        self._add_timeline(incident, "system", "事件创建", f"严重等级: {severity.name}")
        return incident

    def update_status(self, incident_id: str, status: IncidentStatus,
                      actor: str = "system", notes: str = ""):
        if incident_id not in self.incidents:
            return
        incident = self.incidents[incident_id]
        old_status = incident.status
        incident.status = status
        incident.updated_at = datetime.now()
        self._add_timeline(incident, actor, f"状态变更: {old_status.value} -> {status.value}", notes)

    def assign_incident(self, incident_id: str, assignee: str, actor: str = "system"):
        if incident_id not in self.incidents:
            return
        incident = self.incidents[incident_id]
        incident.assignee = assignee
        incident.updated_at = datetime.now()
        self._add_timeline(incident, actor, f"分配给 {assignee}")

    def add_note(self, incident_id: str, note: str, actor: str = "system"):
        if incident_id not in self.incidents:
            return
        self._add_timeline(self.incidents[incident_id], actor, "添加备注", note)

    def resolve_incident(self, incident_id: str, resolution: str,
                         root_cause: str = "", actor: str = "system"):
        if incident_id not in self.incidents:
            return
        incident = self.incidents[incident_id]
        incident.status = IncidentStatus.RESOLVED
        incident.resolution = resolution
        incident.root_cause = root_cause
        incident.updated_at = datetime.now()
        self._add_timeline(incident, actor, "事件已解决", resolution)

    def _add_timeline(self, incident: Incident, actor: str, action: str, details: str = ""):
        incident.timeline.append(TimelineEntry(
            timestamp=datetime.now(),
            action=action,
            actor=actor,
            details=details,
        ))

    def get_open_incidents(self) -> List[Incident]:
        return [
            inc for inc in self.incidents.values()
            if inc.status not in [IncidentStatus.RESOLVED, IncidentStatus.CLOSED]
        ]

    def check_escalations(self) -> List[Dict]:
        escalations = []
        now = datetime.now()
        for incident in self.get_open_incidents():
            escalation_window = self.escalation_rules.get(incident.severity.value, timedelta(hours=24))
            if now - incident.created_at > escalation_window:
                escalations.append({
                    "incident_id": incident.incident_id,
                    "title": incident.title,
                    "severity": incident.severity.name,
                    "age_hours": (now - incident.created_at).total_seconds() / 3600,
                    "assignee": incident.assignee,
                })
        return escalations

    def generate_report(self, incident_id: str) -> str:
        incident = self.incidents.get(incident_id)
        if not incident:
            return "事件不存在"

        report = f"""# 事件报告: {incident.incident_id}

## 基本信息
- 标题: {incident.title}
- 严重程度: {incident.severity.name}
- 类型: {incident.incident_type.value}
- 状态: {incident.status.value}
- 创建时间: {incident.created_at.isoformat()}
- 更新时间: {incident.updated_at.isoformat()}
- 负责人: {incident.assignee or '未分配'}
- 受影响用户: {incident.affected_users}
- 标签: {', '.join(incident.tags) if incident.tags else '无'}

## 描述
{incident.description}

## 时间线
"""
        for entry in incident.timeline:
            report += f"- [{entry.timestamp.isoformat()}] [{entry.actor}] {entry.action}"
            if entry.details:
                report += f": {entry.details}"
            report += "\n"

        if incident.resolution:
            report += f"\n## 解决方案\n{incident.resolution}\n"
        if incident.root_cause:
            report += f"\n## 根因分析\n{incident.root_cause}\n"

        return report

    def get_statistics(self) -> Dict:
        total = len(self.incidents)
        by_severity = {}
        by_status = {}
        by_type = {}

        for inc in self.incidents.values():
            by_severity[inc.severity.name] = by_severity.get(inc.severity.name, 0) + 1
            by_status[inc.status.value] = by_status.get(inc.status.value, 0) + 1
            by_type[inc.incident_type.value] = by_type.get(inc.incident_type.value, 0) + 1

        return {
            "total": total,
            "by_severity": by_severity,
            "by_status": by_status,
            "by_type": by_type,
        }
```

```python
from typing import Dict, List, Callable, Any
from dataclasses import dataclass
from datetime import datetime
from enum import Enum


class ResponseAction(Enum):
    BLOCK_IP = "block_ip"
    DISABLE_USER = "disable_user"
    ENABLE_STRICT_FILTER = "enable_strict_filter"
    COLLECT_EVIDENCE = "collect_evidence"
    NOTIFY_TEAM = "notify_team"
    CREATE_INCIDENT = "create_incident"
    RATE_LIMIT = "rate_limit"
    ROLLBACK_MODEL = "rollback_model"


@dataclass
class PlaybookStep:
    action: ResponseAction
    params: Dict[str, Any]
    auto_execute: bool = True
    requires_approval: bool = False
    description: str = ""


class ResponseOrchestrator:
    """响应编排器"""

    def __init__(self):
        self.playbooks: Dict[str, List[PlaybookStep]] = {}
        self.action_handlers: Dict[ResponseAction, Callable] = {}
        self.execution_log: List[Dict] = []

    def register_handler(self, action: ResponseAction, handler: Callable):
        self.action_handlers[action] = handler

    def create_playbook(self, name: str, steps: List[PlaybookStep]):
        self.playbooks[name] = steps

    def execute_playbook(self, playbook_name: str, context: Dict) -> Dict:
        steps = self.playbooks.get(playbook_name, [])
        results = []

        for step in steps:
            if step.requires_approval and not context.get("approved"):
                results.append({
                    "action": step.action.value,
                    "status": "pending_approval",
                    "description": step.description,
                })
                continue

            handler = self.action_handlers.get(step.action)
            if handler:
                try:
                    result = handler(step.params, context)
                    results.append({
                        "action": step.action.value,
                        "status": "success",
                        "result": result,
                    })
                except Exception as e:
                    results.append({
                        "action": step.action.value,
                        "status": "failed",
                        "error": str(e),
                    })
            else:
                results.append({
                    "action": step.action.value,
                    "status": "no_handler",
                })

            self.execution_log.append({
                "playbook": playbook_name,
                "action": step.action.value,
                "timestamp": datetime.now().isoformat(),
                "status": results[-1]["status"],
            })

        return {
            "playbook": playbook_name,
            "total_steps": len(steps),
            "executed": len([r for r in results if r["status"] != "pending_approval"]),
            "results": results,
        }


# 使用示例
orchestrator = ResponseOrchestrator()

# 注册处理函数
orchestrator.register_handler(ResponseAction.BLOCK_IP, lambda params, ctx: f"Blocked IP: {params.get('ip')}")
orchestrator.register_handler(ResponseAction.NOTIFY_TEAM, lambda params, ctx: f"Notified: {params.get('team')}")
orchestrator.register_handler(ResponseAction.COLLECT_EVIDENCE, lambda params, ctx: "Evidence collected")
orchestrator.register_handler(ResponseAction.ENABLE_STRICT_FILTER, lambda params, ctx: "Strict filter enabled")

# 创建 Prompt 注入响应剧本
orchestrator.create_playbook("prompt_injection_response", [
    PlaybookStep(
        action=ResponseAction.COLLECT_EVIDENCE,
        params={"scope": "recent_logs", "minutes": 30},
        description="收集最近 30 分钟的日志",
    ),
    PlaybookStep(
        action=ResponseAction.ENABLE_STRICT_FILTER,
        params={"filter_level": "high"},
        description="启用严格输入过滤",
    ),
    PlaybookStep(
        action=ResponseAction.BLOCK_IP,
        params={"ip": "{{source_ip}}", "duration": 3600},
        description="封锁攻击源 IP 1 小时",
    ),
    PlaybookStep(
        action=ResponseAction.NOTIFY_TEAM,
        params={"team": "security", "channel": "slack"},
        description="通知安全团队",
    ),
])
```

## 6. 常见误区

1. **"事件响应只是技术问题"**: 需要跨部门协作，包括法务、公关、管理层
2. **"只关注修复"**: 需要根因分析和改进，否则同类事件会重复发生
3. **"没有预案"**: 需要提前准备应急计划，临时应对效率低
4. **"不记录不复盘"**: 需要完整的文档和复盘，知识需要沉淀
5. **"响应一次就够"**: 需要持续演练和改进，威胁在不断演变
6. **"自动化完全替代人工"**: 自动化是辅助，关键决策仍需人工
7. **"只关注外部攻击"**: 内部威胁同样严重，需要全面监控
8. **"复盘追责"**: 复盘应聚焦改进而非追责，否则团队会隐瞒问题

## 工程建议

- 为每类 AI 安全事件（Prompt 注入、数据泄露、DoS、模型投毒）编写独立的响应剧本（Playbook），并定期通过桌面演练验证其有效性
- 事件管理系统必须支持完整的审计时间线，所有操作（包括自动操作）都要记录执行人、时间和原因，便于事后复盘和合规审查
- 自动化响应应设置"熔断"机制：当自动操作连续失败或误判率超过阈值时，自动降级为人工审批模式，避免自动化造成二次伤害
- 复盘会议应遵循"无责文化"原则，聚焦流程和系统的改进点，而非追究个人责任，否则团队会倾向于隐瞒安全事件

## 总结

- 事件响应是安全运营的核心能力，需要系统性的流程和工具支持
- 事件分类和优先级管理很重要，决定了响应速度和资源分配
- 自动化响应可以大幅提升效率，但关键节点仍需人工确认
- 复盘和改进是提升响应能力的关键，每次事件都是学习机会

### 练习 2: 事件管理系统

**思路**：事件管理系统需要支持事件创建、状态流转、时间线记录和报告生成四个核心功能。状态流转定义了从 OPEN 到 CLOSED 的标准生命周期，所有操作都记录到时间线以便审计。报告生成需要汇总事件的完整信息，包括时间线、解决方案和根因分析。

**答案**：

```python
from typing import Dict, List, Optional
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
import uuid


class IncidentSeverity(Enum):
    P1_CRITICAL = 1
    P2_HIGH = 2
    P3_MEDIUM = 3
    P4_LOW = 4


class IncidentStatus(Enum):
    OPEN = "open"
    INVESTIGATING = "investigating"
    CONTAINING = "containing"
    REMEDIATING = "remediating"
    RESOLVED = "resolved"
    CLOSED = "closed"


VALID_TRANSITIONS = {
    IncidentStatus.OPEN: [IncidentStatus.INVESTIGATING],
    IncidentStatus.INVESTIGATING: [IncidentStatus.CONTAINING, IncidentStatus.RESOLVED],
    IncidentStatus.CONTAINING: [IncidentStatus.REMEDIATING, IncidentStatus.RESOLVED],
    IncidentStatus.REMEDIATING: [IncidentStatus.RESOLVED],
    IncidentStatus.RESOLVED: [IncidentStatus.CLOSED, IncidentStatus.INVESTIGATING],
    IncidentStatus.CLOSED: [],
}


@dataclass
class TimelineEntry:
    timestamp: datetime
    action: str
    actor: str
    details: str = ""


@dataclass
class Incident:
    incident_id: str
    title: str
    severity: IncidentSeverity
    incident_type: str
    status: IncidentStatus
    created_at: datetime
    updated_at: datetime
    assignee: Optional[str] = None
    description: str = ""
    timeline: List[TimelineEntry] = field(default_factory=list)
    resolution: str = ""
    root_cause: str = ""
    affected_users: int = 0


class IncidentManager:
    def __init__(self):
        self.incidents: Dict[str, Incident] = {}
        self.escalation_rules = {
            1: timedelta(minutes=30), 2: timedelta(hours=4),
            3: timedelta(hours=24), 4: timedelta(hours=72),
        }

    def create_incident(self, title: str, severity: IncidentSeverity,
                        incident_type: str, description: str = "", assignee: str = None) -> Incident:
        incident_id = f"INC-{uuid.uuid4().hex[:8].upper()}"
        now = datetime.now()
        incident = Incident(
            incident_id=incident_id, title=title, severity=severity,
            incident_type=incident_type, status=IncidentStatus.OPEN,
            created_at=now, updated_at=now, assignee=assignee, description=description,
        )
        self.incidents[incident_id] = incident
        self._add_timeline(incident, "system", "事件创建", f"严重等级: {severity.name}")
        return incident

    def update_status(self, incident_id: str, new_status: IncidentStatus,
                      actor: str = "system", notes: str = "") -> bool:
        incident = self.incidents.get(incident_id)
        if not incident:
            return False
        allowed = VALID_TRANSITIONS.get(incident.status, [])
        if new_status not in allowed:
            return False
        old_status = incident.status
        incident.status = new_status
        incident.updated_at = datetime.now()
        self._add_timeline(incident, actor, f"状态变更: {old_status.value} -> {new_status.value}", notes)
        return True

    def assign_incident(self, incident_id: str, assignee: str, actor: str = "system"):
        incident = self.incidents.get(incident_id)
        if incident:
            incident.assignee = assignee
            incident.updated_at = datetime.now()
            self._add_timeline(incident, actor, f"分配给 {assignee}")

    def add_note(self, incident_id: str, note: str, actor: str = "system"):
        incident = self.incidents.get(incident_id)
        if incident:
            self._add_timeline(incident, actor, "添加备注", note)

    def resolve_incident(self, incident_id: str, resolution: str,
                         root_cause: str = "", actor: str = "system"):
        incident = self.incidents.get(incident_id)
        if incident:
            incident.status = IncidentStatus.RESOLVED
            incident.resolution = resolution
            incident.root_cause = root_cause
            incident.updated_at = datetime.now()
            self._add_timeline(incident, actor, "事件已解决", resolution)

    def _add_timeline(self, incident: Incident, actor: str, action: str, details: str = ""):
        incident.timeline.append(TimelineEntry(
            timestamp=datetime.now(), action=action, actor=actor, details=details,
        ))

    def get_open_incidents(self) -> List[Incident]:
        return [i for i in self.incidents.values()
                if i.status not in [IncidentStatus.RESOLVED, IncidentStatus.CLOSED]]

    def check_escalations(self) -> List[Dict]:
        escalations = []
        now = datetime.now()
        for incident in self.get_open_incidents():
            window = self.escalation_rules.get(incident.severity.value, timedelta(hours=24))
            if now - incident.created_at > window:
                escalations.append({
                    "incident_id": incident.incident_id, "title": incident.title,
                    "severity": incident.severity.name,
                    "age_hours": round((now - incident.created_at).total_seconds() / 3600, 1),
                })
        return escalations

    def generate_report(self, incident_id: str) -> str:
        incident = self.incidents.get(incident_id)
        if not incident:
            return "事件不存在"
        mttr = ""
        if incident.status == IncidentStatus.RESOLVED:
            last = incident.timeline[-1].timestamp if incident.timeline else incident.updated_at
            mttr = f"- **MTTR**: {(last - incident.created_at).total_seconds() / 60:.1f} 分钟"

        lines = [
            f"# 事件报告: {incident.incident_id}", "",
            f"## 基本信息", f"- 标题: {incident.title}",
            f"- 严重程度: {incident.severity.name}",
            f"- 类型: {incident.incident_type}",
            f"- 状态: {incident.status.value}",
            f"- 创建时间: {incident.created_at.isoformat()}",
            f"- 负责人: {incident.assignee or '未分配'}",
            f"- 受影响用户: {incident.affected_users}", mttr, "",
            f"## 描述", incident.description, "",
            f"## 时间线",
        ]
        for entry in incident.timeline:
            detail = f": {entry.details}" if entry.details else ""
            lines.append(f"- [{entry.timestamp.isoformat()}] [{entry.actor}] {entry.action}{detail}")
        if incident.resolution:
            lines.extend(["", f"## 解决方案", incident.resolution])
        if incident.root_cause:
            lines.extend(["", f"## 根因分析", incident.root_cause])
        return "\n".join(lines)

    def get_statistics(self) -> Dict:
        total = len(self.incidents)
        by_severity = {}
        by_status = {}
        for inc in self.incidents.values():
            by_severity[inc.severity.name] = by_severity.get(inc.severity.name, 0) + 1
            by_status[inc.status.value] = by_status.get(inc.status.value, 0) + 1
        return {"total": total, "by_severity": by_severity, "by_status": by_status}


# 使用示例
manager = IncidentManager()

inc = manager.create_incident(
    "检测到 Prompt 注入攻击", IncidentSeverity.P2_HIGH,
    "prompt_injection", "攻击者通过特殊输入绕过安全防护", "alice",
)

manager.update_status(inc.incident_id, IncidentStatus.INVESTIGATING, "alice", "开始分析攻击手法")
manager.add_note(inc.incident_id, "确认攻击来源 IP: 192.168.1.100", "alice")
manager.update_status(inc.incident_id, IncidentStatus.CONTAINING, "alice", "封禁攻击 IP")
manager.resolve_incident(inc.incident_id, "更新注入检测规则，封禁攻击 IP",
                          "输入过滤规则未覆盖新的注入模式", "alice")

print(manager.generate_report(inc.incident_id))
print(f"\n统计: {manager.get_statistics()}")
```

**要点**：
- 状态流转使用白名单控制，防止非法状态变更（如从 OPEN 直接到 RESOLVED）
- 所有操作都记录到时间线，包含操作人、时间和详情，便于事后复盘和合规审查
- 升级检查根据事件严重等级有不同的超时阈值，P1 事件 30 分钟未处理即触发升级

### 练习 3: 桌面演练

**思路**：桌面演练是一种低成本的事件响应验证方法，通过模拟真实场景让团队在不实际操作的情况下走完响应流程。演练需要设计场景脚本、定义参与者角色、记录发现的问题、产出改进计划。关键是场景要贴近真实威胁，而不是走过场。

**答案**：

```python
from typing import Dict, List
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum


class ExercisePhase(Enum):
    INJECT = "inject"       # 注入场景
    RESPONSE = "response"   # 参与者响应
    EVALUATE = "evaluate"   # 评估反馈


@dataclass
class ExerciseScenario:
    name: str
    description: str
    incident_type: str
    severity: str
    injects: List[Dict]  # 按时间顺序的场景注入


@dataclass
class ExerciseFinding:
    category: str  # "gap", "improvement", "observation"
    description: str
    severity: str  # "high", "medium", "low"
    recommended_action: str


@dataclass
class ExerciseRecord:
    scenario_name: str
    conducted_at: datetime
    participants: List[str]
    duration_minutes: int
    timeline: List[Dict]
    findings: List[ExerciseFinding]
    lessons_learned: List[str]


class TabletopExercise:
    def __init__(self):
        self.scenarios: Dict[str, ExerciseScenario] = {}
        self.records: List[ExerciseRecord] = []

    def create_prompt_injection_scenario(self) -> ExerciseScenario:
        scenario = ExerciseScenario(
            name="Prompt 注入攻击演练",
            description="模拟攻击者通过 Prompt 注入获取系统内部信息",
            incident_type="prompt_injection",
            severity="P2",
            injects=[
                {"time": "T+0min", "event": "监控系统告警：检测到异常长输入",
                 "question": "值班人员应该如何响应？第一步做什么？"},
                {"time": "T+5min", "event": "日志显示攻击者已成功提取系统提示词片段",
                 "question": "如何评估影响范围？需要检查哪些数据？"},
                {"time": "T+10min", "event": "发现攻击者使用了 3 种不同的注入手法",
                 "question": "当前的检测规则能否覆盖？如何快速更新？"},
                {"time": "T+15min", "event": "攻击者 IP 被封禁后换了新的 IP 继续攻击",
                 "question": "除了封禁 IP 还有什么遏制措施？"},
                {"time": "T+20min", "event": "确认有 50 个用户的对话受到影响",
                 "question": "是否需要通知受影响用户？由谁通知？怎么说？"},
                {"time": "T+30min", "event": "攻击停止，需要恢复服务",
                 "question": "恢复前需要验证什么？如何确认防护已更新？"},

        )
        self.scenarios[scenario.name] = scenario

    def create_data_leakage_scenario(self) -> ExerciseScenario:
        scenario = ExerciseScenario(
            name="数据泄露演练",
            description="模拟 LLM 输出中泄露用户 PII 信息",
            incident_type="data_leakage",
            severity="P1",
            injects=[
                {"time": "T+0min", "event": "用户反馈：AI 回复中包含其他用户的手机号和邮箱",
                 "question": "如何确认这是真实的泄露？需要收集哪些证据？"},
                {"time": "T+5min", "event": "检查发现最近 2 小时有 200 次类似请求",
                 "question": "影响范围如何评估？是否需要立即停服？"},
                {"time": "T+10min", "event": "技术分析发现是 RAG 检索未做权限过滤",
                 "question": "如何快速遏制？临时修复方案是什么？"},
                {"time": "T+15min", "event": "法务要求在 72 小时内通知受影响用户",
                 "question": "通知流程是什么？需要准备什么材料？"},
                {"time": "T+30min", "event": "需要提交监管报告",
                 "question": "报告应包含哪些内容？由谁负责？"},
            ],
        )
        self.scenarios[scenario.name] = scenario
        return scenario

    def run_exercise(self, scenario_name: str, participants: List[str],
                     responses: List[Dict]) -> ExerciseRecord:
        scenario = self.scenarios.get(scenario_name)
        if not scenario:
            return None

        findings = []
        lessons = []

        for inject, response in zip(scenario.injects, responses):
            resp_text = response.get("response", "未回答")
            if "不知道" in resp_text or "不清楚" in resp_text:
                findings.append(ExerciseFinding(
                    "gap", f"对场景 '{inject['event']}' 缺乏明确响应流程",
                    "high", "补充该场景的 SOP 文档",
                ))
            if len(resp_text) < 20:
                findings.append(ExerciseFinding(
                    "observation", f"对 '{inject['question']}' 的回答过于简略",
                    "medium", "在响应计划中增加详细的检查清单",
                ))

        record = ExerciseRecord(
            scenario_name=scenario_name, conducted_at=datetime.now(),
            participants=participants, duration_minutes=30,
            timeline=[{"inject": s["injects"][i] if i < len(s := scenario.injects) else {},
                        "response": responses[i] if i < len(responses) else {}}
                      for i in range(max(len(scenario.injects), len(responses)))],
            findings=findings,
            lessons_learned=lessons or ["需要定期演练以保持响应能力", "SOP 文档需要更详细的检查清单"],
        )
        self.records.append(record)
        return record

    def generate_exercise_report(self, record: ExerciseRecord) -> str:
        lines = [
            f"# 桌面演练报告: {record.scenario_name}", "",
            f"## 基本信息", f"- 演练时间: {record.conducted_at.isoformat()}",
            f"- 参与人员: {', '.join(record.participants)}",
            f"- 持续时长: {record.duration_minutes} 分钟", "",
            f"## 发现的问题 ({len(record.findings)} 项)",
        ]
        for i, f in enumerate(record.findings, 1):
            lines.append(f"{i}. [{f.severity.upper()}] {f.description}")
            lines.append(f"   建议: {f.recommended_action}")
        lines.extend(["", "## 经验教训"])
        for lesson in record.lessons_learned:
            lines.append(f"- {lesson}")
        return "\n".join(lines)
```

### 练习 4: 自动化剧本

**思路**：自动化响应编排器需要支持多步骤剧本执行和人工审批节点。核心设计是将每个响应步骤抽象为 PlaybookStep，支持自动执行和人工审批两种模式。编排器按顺序执行步骤，遇到需要审批的步骤时暂停并等待人工确认，所有执行结果都记录到日志。

**答案**：

```python
from typing import Dict, List, Callable, Any
from dataclasses import dataclass
from datetime import datetime
from enum import Enum


class ResponseAction(Enum):
    BLOCK_IP = "block_ip"
    DISABLE_USER = "disable_user"
    ENABLE_STRICT_FILTER = "enable_strict_filter"
    COLLECT_EVIDENCE = "collect_evidence"
    NOTIFY_TEAM = "notify_team"
    CREATE_INCIDENT = "create_incident"
    RATE_LIMIT = "rate_limit"
    ROLLBACK_MODEL = "rollback_model"
    HUMAN_REVIEW = "human_review"


@dataclass
class PlaybookStep:
    action: ResponseAction
    params: Dict[str, Any]
    auto_execute: bool = True
    requires_approval: bool = False
    description: str = ""
    timeout_seconds: int = 300


class ResponseOrchestrator:
    def __init__(self):
        self.playbooks: Dict[str, List[PlaybookStep]] = {}
        self.action_handlers: Dict[ResponseAction, Callable] = {}
        self.execution_log: List[Dict] = []
        self.pending_approvals: List[Dict] = []

    def register_handler(self, action: ResponseAction, handler: Callable):
        self.action_handlers[action] = handler

    def create_playbook(self, name: str, steps: List[PlaybookStep]):
        self.playbooks[name] = steps

    def execute_playbook(self, playbook_name: str, context: Dict) -> Dict:
        steps = self.playbooks.get(playbook_name, [])
        results = []
        paused = False

        for i, step in enumerate(steps):
            if paused:
                results.append({"action": step.action.value, "status": "skipped", "reason": "前序步骤暂停"})
                continue

            if step.requires_approval and not context.get("approved"):
                approval_request = {
                    "playbook": playbook_name, "step_index": i,
                    "action": step.action.value, "description": step.description,
                    "params": step.params, "requested_at": datetime.now().isoformat(),
                }
                self.pending_approvals.append(approval_request)
                results.append({
                    "action": step.action.value, "status": "pending_approval",
                    "description": step.description,
                })
                paused = True
                continue

            handler = self.action_handlers.get(step.action)
            if handler:
                try:
                    result = handler(step.params, context)
                    results.append({"action": step.action.value, "status": "success", "result": result})
                except Exception as e:
                    results.append({"action": step.action.value, "status": "failed", "error": str(e)})
            else:
                results.append({"action": step.action.value, "status": "no_handler"})

            self.execution_log.append({
                "playbook": playbook_name, "action": step.action.value,
                "timestamp": datetime.now().isoformat(), "status": results[-1]["status"],
            })

        executed = len([r for r in results if r["status"] not in ("pending_approval", "skipped")])
        return {
            "playbook": playbook_name, "total_steps": len(steps),
            "executed": executed, "paused": paused, "results": results,
        }

    def approve_pending(self, approval_index: int, approver: str, context: Dict) -> Dict:
        if approval_index >= len(self.pending_approvals):
            return {"error": "审批请求不存在"}
        approval = self.pending_approvals[approval_index]
        approval["approved_by"] = approver
        approval["approved_at"] = datetime.now().isoformat()

        step = PlaybookStep(
            action=ResponseAction(approval["action"]),
            params=approval["params"], description=approval["description"],
        )
        handler = self.action_handlers.get(step.action)
        if handler:
            try:
                result = handler(step.params, context)
                return {"status": "executed", "result": result}
            except Exception as e:
                return {"status": "failed", "error": str(e)}
        return {"status": "no_handler"}

    def get_execution_log(self, playbook_name: str = None) -> List[Dict]:
        if playbook_name:
            return [l for l in self.execution_log if l["playbook"] == playbook_name]
        return self.execution_log


# 使用示例
orchestrator = ResponseOrchestrator()

orchestrator.register_handler(ResponseAction.COLLECT_EVIDENCE,
                               lambda p, c: f"收集了 {p.get('minutes', 30)} 分钟的日志")
orchestrator.register_handler(ResponseAction.ENABLE_STRICT_FILTER,
                               lambda p, c: f"已启用 {p.get('filter_level', 'high')} 级别过滤")
orchestrator.register_handler(ResponseAction.BLOCK_IP,
                               lambda p, c: f"已封禁 IP {p.get('ip')} {p.get('duration', 3600)}秒")
orchestrator.register_handler(ResponseAction.NOTIFY_TEAM,
                               lambda p, c: f"已通知 {p.get('team')} 团队")
orchestrator.register_handler(ResponseAction.ROLLBACK_MODEL,
                               lambda p, c: f"已回滚到模型版本 {p.get('version')}")

# 创建 Prompt 注入响应剧本（包含人工审批节点）
orchestrator.create_playbook("prompt_injection_response", [
    PlaybookStep(ResponseAction.COLLECT_EVIDENCE, {"scope": "recent_logs", "minutes": 30},
                 description="收集最近 30 分钟的日志"),
    PlaybookStep(ResponseAction.ENABLE_STRICT_FILTER, {"filter_level": "high"},
                 description="启用严格输入过滤"),
    PlaybookStep(ResponseAction.BLOCK_IP, {"ip": "{{source_ip}}", "duration": 3600},
                 description="封锁攻击源 IP", requires_approval=True),
    PlaybookStep(ResponseAction.NOTIFY_TEAM, {"team": "security", "channel": "slack"},
                 description="通知安全团队"),
])

# 第一次执行（自动步骤执行，审批步骤暂停）
result = orchestrator.execute_playbook("prompt_injection_response", {"source_ip": "192.168.1.100"})
print(f"首次执行: paused={result['paused']}")
for r in result["results"]:
    print(f"  {r['action']}: {r['status']}")

# 人工审批后继续
if orchestrator.pending_approvals:
    approve_result = orchestrator.approve_pending(0, "alice", {"source_ip": "192.168.1.100"})
    print(f"审批执行: {approve_result}")

print(f"执行日志: {orchestrator.get_execution_log()}")
```

**要点**：
- 人工审批节点让编排器在执行高风险操作（如封禁 IP、回滚模型）前暂停，等待人工确认
- 审批后的步骤可以独立执行，不影响已执行的步骤结果
- 执行日志记录每个步骤的时间、状态和结果，便于事后审计和流程优化
