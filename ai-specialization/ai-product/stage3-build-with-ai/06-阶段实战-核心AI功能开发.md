## 二、核心功能实现

### 2.1 RAG 管道

```typescript
// lib/ai/rag-pipeline.ts
import { RecursiveCharacterTextSplitter } from 'langchain/text_splitter'
import { OpenAIEmbeddings } from '@langchain/openai'

export class RAGPipeline {
  private embeddings: OpenAIEmbeddings

  constructor() {
    this.embeddings = new OpenAIEmbeddings({ model: 'text-embedding-3-small' })
  }

  // 处理上传的文档
  async processDocument(file: File, userId: string): Promise<string> {
    // 1. 解析文档
    const content = await this.parseFile(file)

    // 2. 分块
    const splitter = new RecursiveCharacterTextSplitter({
      chunkSize: 500,
      chunkOverlap: 50,
    })
    const chunks = await splitter.createDocuments([content])

    // 3. 向量化
    const vectors = await this.embeddings.embedDocuments(
      chunks.map(c => c.pageContent)
    )

    // 4. 存储
    const documentId = crypto.randomUUID()
    await db.documents.create({
      data: {
        id: documentId,
        userId,
        filename: file.name,
        fileType: file.name.split('.').pop()!,
        chunkCount: chunks.length,
        status: 'ready',
      },
    })

    for (let i = 0; i < chunks.length; i++) {
      await db.documentChunks.create({
        data: {
          id: crypto.randomUUID(),
          documentId,
          content: chunks[i].pageContent,
          embedding: vectors[i],
          chunkIndex: i,
        },
      })
    }

    return documentId
  }

  // 检索相关文档
  async retrieve(query: string, userId: string, topK = 5) {
    const queryVector = await this.embeddings.embedQuery(query)

    // 使用 pgvector 的余弦相似度检索
    const results = await db.$queryRaw`
      SELECT dc.content, dc.metadata,
             1 - (dc.embedding <=> ${queryVector}::vector) as similarity
      FROM document_chunks dc
      JOIN documents d ON dc.document_id = d.id
      WHERE d.user_id = ${userId}
      ORDER BY dc.embedding <=> ${queryVector}::vector
      LIMIT ${topK}
    `

    return results as Array<{ content: string; metadata: any; similarity: number }>
  }

  private async parseFile(file: File): Promise<string> {
    const ext = file.name.split('.').pop()?.toLowerCase()
    switch (ext) {
      case 'md':
      case 'txt':
        return file.text()
      case 'pdf':
        const pdfParse = (await import('pdf-parse')).default
        const buffer = await file.arrayBuffer()
        const data = await pdfParse(Buffer.from(buffer))
        return data.text
      default:
        throw new Error(`不支持的文件类型: ${ext}`)
    }
  }
}
```

### 2.2 对话 API

```typescript
// app/api/chat/route.ts
import { NextRequest } from 'next/server'
import { RAGPipeline } from '@/lib/ai/rag-pipeline'
import { modelRouter } from '@/lib/ai/model-router'
import { InputValidator } from '@/lib/ai/quality-checker'

const rag = new RAGPipeline()

export async function POST(req: NextRequest) {
  const { message, conversationId } = await req.json()
  const userId = req.headers.get('x-user-id')!

  // 1. 输入校验
  const injectionCheck = InputValidator.detectPromptInjection(message)
  if (!injectionCheck.safe) {
    return Response.json({ error: '输入内容不安全' }, { status: 400 })
  }

  // 2. RAG 检索
  const context = await rag.retrieve(message, userId)
  const hasRelevantContext = context.some(c => c.similarity > 0.7)

  // 3. 构造 Prompt
  const systemPrompt = hasRelevantContext
    ? `你是一个知识库问答助手。请基于以下参考资料回答用户的问题。
       规则：只基于参考资料回答，不要编造。如果资料中没有相关信息，明确说"根据现有资料无法回答"。
       用 [数字] 标注引用来源。

       参考资料：
       ${context.map((c, i) => `[${i + 1}] ${c.content}`).join('\n\n')}`
    : '你是一个友好的 AI 助手。请直接回答用户的问题。'

  // 4. 调用模型
  const response = await modelRouter.chat([
    { role: 'system', content: systemPrompt },
    { role: 'user', content: message },
  ])

  // 5. 保存消息
  await db.messages.createMany({
    data: [
      { id: crypto.randomUUID(), conversationId, role: 'user', content: message },
      {
        id: crypto.randomUUID(), conversationId, role: 'assistant',
        content: response.content,
        metadata: {
          model: response.model,
          tokens: response.usage,
          sources: context.map(c => c.metadata?.filename),
        },
      },
    ],
  })

  return Response.json({
    answer: response.content,
    sources: hasRelevantContext ? context.map(c => ({
      filename: c.metadata?.filename,
      similarity: c.similarity,
    })) : [],
    model: response.model,
  })
}
```

### 2.3 流式响应

```typescript
// app/api/chat/stream/route.ts
import { NextRequest } from 'next/server'
import { RAGPipeline } from '@/lib/ai/rag-pipeline'
import { modelRouter } from '@/lib/ai/model-router'

const rag = new RAGPipeline()

export async function POST(req: NextRequest) {
  const { message, conversationId } = await req.json()
  const userId = req.headers.get('x-user-id')!

  const encoder = new TextEncoder()
  const stream = new ReadableStream({
    async start(controller) {
      try {
        // 1. RAG 检索
        const context = await rag.retrieve(message, userId)
        const hasRelevantContext = context.some(c => c.similarity > 0.7)

        // 2. 发送检索结果
        controller.enqueue(encoder.encode(
          `data: ${JSON.stringify({ type: 'sources', data: context })}\n\n`
        ))

        // 3. 流式生成回答
        const systemPrompt = hasRelevantContext
          ? `基于以下资料回答：\n${context.map(c => c.content).join('\n')}`
          : '你是一个友好的助手。'

        const streamGen = modelRouter.streamChat([
          { role: 'system', content: systemPrompt },
          { role: 'user', content: message },
        ])

        for await (const chunk of streamGen) {
          controller.enqueue(encoder.encode(
            `data: ${JSON.stringify({ type: 'chunk', data: chunk })}\n\n`
          ))
        }

        controller.enqueue(encoder.encode('data: [DONE]\n\n'))
      } catch (error) {
        controller.enqueue(encoder.encode(
          `data: ${JSON.stringify({ type: 'error', data: String(error) })}\n\n`
        ))
      } finally {
        controller.close()
      }
    },
  })

  return new Response(stream, {
    headers: { 'Content-Type': 'text/event-stream' },
  })
}
```

```typescript
// app/api/knowledge/route.ts
import { NextRequest } from 'next/server'
import { RAGPipeline } from '@/lib/ai/rag-pipeline'

const rag = new RAGPipeline()

// 上传文档
export async function POST(req: NextRequest) {
  const userId = req.headers.get('x-user-id')!
  const formData = await req.formData()
  const file = formData.get('file') as File

  if (!file) {
    return Response.json({ error: '请上传文件' }, { status: 400 })
  }

  // 文件大小限制（10MB）
  if (file.size > 10 * 1024 * 1024) {
    return Response.json({ error: '文件大小不能超过 10MB' }, { status: 400 })
  }

  try {
    const documentId = await rag.processDocument(file, userId)
    return Response.json({ success: true, documentId })
  } catch (error) {
    return Response.json({ error: String(error) }, { status: 500 })
  }
}

// 获取文档列表
export async function GET(req: NextRequest) {
  const userId = req.headers.get('x-user-id')!

  const documents = await db.documents.findMany({
    where: { userId },
    orderBy: { createdAt: 'desc' },
  })

  return Response.json({ documents })
}

// 删除文档
export async function DELETE(req: NextRequest) {
  const userId = req.headers.get('x-user-id')!
  const { documentId } = await req.json()

  // 验证所有权
  const doc = await db.documents.findFirst({
    where: { id: documentId, userId },
  })

  if (!doc) {
    return Response.json({ error: '文档不存在' }, { status: 404 })
  }

  // 删除文档和关联的块
  await db.documentChunks.deleteMany({ where: { documentId } })
  await db.documents.delete({ where: { id: documentId } })

  return Response.json({ success: true })
}
```

```typescript
// components/chat/ChatWindow.tsx
'use client'

import { useState, useRef } from 'react'

interface Message {
  id: string
  role: 'user' | 'assistant'
  content: string
  sources?: Array<{ filename: string; similarity: number }>
}

export function ChatWindow() {
  const [messages, setMessages] = useState<Message[]>([])
  const [input, setInput] = useState('')
  const [isLoading, setIsLoading] = useState(false)
  const messagesEndRef = useRef<HTMLDivElement>(null)

  const sendMessage = async () => {
    if (!input.trim() || isLoading) return

    const userMessage: Message = {
      id: crypto.randomUUID(),
      role: 'user',
      content: input,
    }
    setMessages(prev => [...prev, userMessage])
    setInput('')
    setIsLoading(true)

    try {
      const response = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ message: input, conversationId: 'current' }),
      })

      const data = await response.json()

      const assistantMessage: Message = {
        id: crypto.randomUUID(),
        role: 'assistant',
        content: data.answer,
        sources: data.sources,
      }
      setMessages(prev => [...prev, assistantMessage])
    } catch (error) {
      setMessages(prev => [...prev, {
        id: crypto.randomUUID(),
        role: 'assistant',
        content: '抱歉，发生了错误，请稍后重试。',
      }])
    } finally {
      setIsLoading(false)
    }
  }

  return (
    <div className="chat-window">
      <div className="messages">
        {messages.map(msg => (
          <div key={msg.id} className={`message ${msg.role}`}>
            <div className="content">{msg.content}</div>
            {msg.sources && msg.sources.length > 0 && (
              <div className="sources">
                <span>参考来源：</span>
                {msg.sources.map((s, i) => (
                  <span key={i} className="source">{s.filename}</span>
                ))}
              </div>
            )}
          </div>
        ))}
        {isLoading && <div className="loading">AI 思考中...</div>}
        <div ref={messagesEndRef} />
      </div>

      <div className="input-area">
        <textarea
          value={input}
          onChange={e => setInput(e.target.value)}
          onKeyDown={e => e.key === 'Enter' && !e.shiftKey && sendMessage()}
          placeholder="输入问题..."
          disabled={isLoading}
        />
        <button onClick={sendMessage} disabled={isLoading || !input.trim()}>
          发送
        </button>
      </div>
    </div>
  )
}
```

### 3.2 文档上传组件

```typescript
// components/knowledge/DocumentUploader.tsx
'use client'

import { useState, useCallback } from 'react'

export function DocumentUploader({ onUpload }: { onUpload: (file: File) => Promise<void> }) {
  const [isDragging, setIsDragging] = useState(false)
  const [uploading, setUploading] = useState(false)

  const handleDrop = useCallback(async (e: React.DragEvent) => {
    e.preventDefault()
    setIsDragging(false)

    const file = e.dataTransfer.files[0]
    if (!file) return

    setUploading(true)
    try {
      await onUpload(file)
    } finally {
      setUploading(false)
    }
  }, [onUpload])

  return (
    <div
      className={`uploader ${isDragging ? 'dragging' : ''}`}
      onDragOver={e => { e.preventDefault(); setIsDragging(true) }}
      onDragLeave={() => setIsDragging(false)}
      onDrop={handleDrop}
    >
      {uploading ? (
        <p>正在处理文档...</p>
      ) : (
        <>
          <p>拖拽文件到此处上传</p>
          <p className="hint">支持 PDF、Markdown、TXT 格式，最大 10MB</p>
          <input
            type="file"
            accept=".pdf,.md,.txt"
            onChange={e => e.target.files?.[0] && onUpload(e.target.files[0])}
          />
        </>
      )}
    </div>
  )
}
```

## 四、质量保障集成

```typescript
// lib/ai/quality-checker.ts
export class QualityChecker {
  // 对 AI 回答进行质量检查
  static async check(params: {
    question: string
    answer: string
    context: string[]
  }): Promise<{ score: number; issues: string[]; shouldShow: boolean }> {
    const { question, answer, context } = params
    const issues: string[] = []
    let score = 1.0

    // 1. 长度检查
    if (answer.length < 10) {
      score -= 0.4
      issues.push('回答过于简短')
    }

    // 2. 拒答检测
    const refusalPhrases = ['无法回答', '没有相关信息', '抱歉', '不知道']
    if (refusalPhrases.some(p => answer.includes(p))) {
      score -= 0.2
      issues.push('回答包含拒答表述')
    }

    // 3. 幻觉简单检测（回答中的数字是否出现在上下文中）
    const numbersInAnswer = answer.match(/\d+/g) ?? []
    const numbersInContext = context.join(' ').match(/\d+/g) ?? []
    const hallucinatedNumbers = numbersInAnswer.filter(n => !numbersInContext.includes(n))
    if (hallucinatedNumbers.length > 0 && context.length > 0) {
      score -= 0.3
      issues.push(`可能存在幻觉数字: ${hallucinatedNumbers.join(', ')}`)
    }

    return {
      score: Math.max(0, score),
      issues,
      shouldShow: score >= 0.4,
    }
  }
}
```

## 五、验收清单

```
核心功能验收：
  □ 文档上传：支持 PDF/Markdown/TXT，处理成功有提示

  □ 多模型支持：可以通过配置切换模型
```

## 常见误区

### 误区一：功能越多越好

MVP 阶段做好一件事就够了：基于文档的准确问答。不要同时做 Agent、多轮对话、语音输入等功能，先把核心链路跑通。

### 误区二：不做质量检查直接上线

第一批用户的体验决定了产品的口碑。上线前至少跑 50 个测试用例，确保拒答率和幻觉率在可接受范围。

### 误区三：忽略错误处理

AI 调用必然会有失败的时候。超时、限流、格式错误——每种异常都要有友好的处理，不能让用户看到白屏或报错信息。

### 误区四：性能优化可以后面再做

流式响应、文档异步处理、向量检索优化——这些如果不在第一版就做好，后面改的成本很高。

## 工程建议

### 1. 先跑通端到端链路

不要先把每个模块做到完美再整合。先用最简单的实现跑通"上传文档 → 提问 → 得到回答"的完整链路，再逐步优化每个环节。

### 2. 写测试用例

为 RAG 管道写 10-20 个测试用例，覆盖正常查询、边界情况和拒答场景。每次改动后跑一遍，确保不退化。

### 3. 记录所有 AI 交互

每次 AI 调用都记录输入、输出、耗时、Token 消耗。这是后续优化和排错的基础数据。

### 4. 准备 Demo 数据

准备 3-5 份示例文档和 20 个示例问题，用于产品演示和用户测试。好的 Demo 数据能让产品体验提升一个档次。

## 小结

这一课把前 5 课的知识整合成一个完整的 AI 功能。核心链路是：文档上传 → 解析分块 → 向量化存储 → 检索问答 → 质量保障。验收标准是：用户能上传文档、提问得到准确回答、系统能拒答不该回答的问题。先跑通 MVP，再逐步优化。

```typescript
// 3. 质量检查器（lib/ai/quality-checker.ts）
import { PromptInjectionDetector } from './injection-detector'
import { scoreOutputQuality } from './quality-scorer'

export class QualityChecker {
  // 输入检查
  checkInput(input: string): { safe: boolean; error?: string } {
    const injection = PromptInjectionDetector.detect(input)
    if (!injection.safe && injection.risk === 'high') {
      return { safe: false, error: '输入内容不安全，请修改后重试' }
    }
    if (input.length > 10000) {
      return { safe: false, error: '输入内容过长，请缩短后重试' }
    }
    return { safe: true }
  }

  // 输出检查
  checkOutput(answer: string, context: string, question: string) {
    return scoreOutputQuality(answer, context, question)
  }
}

// 课程 §2.2 已给出完整代码
// 核心流程：输入校验 → RAG 检索 → 组装 Prompt → 调用 LLM → 输出校验 → 返回结果

// 5. 知识库管理 API（app/api/knowledge/route.ts）
import { NextRequest } from 'next/server'
import { RAGPipeline } from '@/lib/ai/rag-pipeline'

const rag = new RAGPipeline()

export async function POST(req: NextRequest) {
  const userId = req.headers.get('x-user-id')!
  const formData = await req.formData()
  const file = formData.get('file') as File

  if (!file) {
    return Response.json({ error: '请上传文件' }, { status: 400 })
  }

  const allowedTypes = ['.md', '.txt', '.pdf', '.docx']
  const ext = '.' + file.name.split('.').pop()?.toLowerCase()
  if (!allowedTypes.includes(ext)) {
    return Response.json({ error: `不支持的文件类型，支持: ${allowedTypes.join(', ')}` }, { status: 400 })
  }

  const documentId = await rag.processDocument(file, userId)
  return Response.json({ documentId, filename: file.name, status: 'ready' })
}

export async function GET(req: NextRequest) {
  const userId = req.headers.get('x-user-id')!
  const docs = await db.documents.findMany({
    where: { userId },
    orderBy: { createdAt: 'desc' },
  })
  return Response.json({ documents: docs })
}

// 6. 用户反馈 API（app/api/feedback/route.ts）
export async function POST(req: NextRequest) {
  const userId = req.headers.get('x-user-id')!
  const { messageId, rating, reason, details } = await req.json()

  await db.feedback.create({
    data: {
      id: crypto.randomUUID(),
      messageId,
      userId,
      rating,
      reason,
      details,
    },
  })

  return Response.json({ success: true })
}
```

**要点**：
- 实现顺序很重要：先 RAG 管道 → 再对话 API → 再质量检查 → 最后前端页面
- 每完成一步都要测试——不要等全部写完再调试
- 数据库 Schema 是基础，先建好表再写业务代码

### 练习三

**思路**：用 20 个测试用例评估 RAG 系统，计算两个核心指标：召回率（检索到的文档是否包含正确答案）和回答准确率（生成的回答是否正确）。

**答案**：

```typescript
// tests/quality-eval.ts
interface EvalCase {
  question: string
  expectedKeywords: string[]
  shouldRefuse: boolean
}

const evalCases: EvalCase[] = [
  // 正常查询（13 个）
  { question: '怎么注册？', expectedKeywords: ['邮箱', '密码'], shouldRefuse: false },
  { question: '专业版多少钱？', expectedKeywords: ['99'], shouldRefuse: false },
  { question: '免费版 API 限制？', expectedKeywords: ['100'], shouldRefuse: false },
  { question: '数据怎么加密？', expectedKeywords: ['AES'], shouldRefuse: false },
  { question: '服务器在哪？', expectedKeywords: ['中国大陆'], shouldRefuse: false },
  { question: 'API 认证方式？', expectedKeywords: ['OAuth'], shouldRefuse: false },
  { question: '免费版存储多大？', expectedKeywords: ['1GB'], shouldRefuse: false },
  { question: '专业版 API 限制？', expectedKeywords: ['无限', '100'], shouldRefuse: false },
  { question: '企业版怎么买？', expectedKeywords: ['联系', '销售'], shouldRefuse: false },
  { question: '能删除数据吗？', expectedKeywords: ['删除'], shouldRefuse: false },
  { question: 'API 速率限制？', expectedKeywords: ['次/分钟'], shouldRefuse: false },
  { question: '有什么安全认证？', expectedKeywords: ['等保'], shouldRefuse: false },
  { question: '怎么导出数据？', expectedKeywords: ['导出'], shouldRefuse: false },
  // 拒答查询（7 个）
  { question: '创始团队是谁？', expectedKeywords: [], shouldRefuse: true },
  { question: '有移动端吗？', expectedKeywords: [], shouldRefuse: true },
  { question: '竞品有哪些？', expectedKeywords: [], shouldRefuse: true },
  { question: '融资情况？', expectedKeywords: [], shouldRefuse: true },
  { question: '有多少用户？', expectedKeywords: [], shouldRefuse: true },
  { question: '技术栈是什么？', expectedKeywords: [], shouldRefuse: true },
  { question: '支持哪些语言？', expectedKeywords: [], shouldRefuse: true },
]

async function evaluate() {
  let recallHits = 0
  let recallTotal = 0
  let accuracyHits = 0
  let accuracyTotal = 0
  const failures: Array<{ question: string; expected: string; actual: string }> = []

  for (const tc of evalCases) {
    // 评估检索召回率
    const docs = await rag.retrieve(tc.question, 3)
    const retrievedText = docs.map(d => d.content).join(' ')
    const recallOk = tc.shouldRefuse || tc.expectedKeywords.some(k => retrievedText.includes(k))
    if (!tc.shouldRefuse) recallTotal++
    if (recallOk && !tc.shouldRefuse) recallHits++

    // 评估回答准确率
    const { answer } = await rag.answer(tc.question)
    if (tc.shouldRefuse) {
      accuracyTotal++
      const refused = answer.includes('无法') || answer.includes('没有') || answer.includes('抱歉')
      if (refused) accuracyHits++
      else failures.push({ question: tc.question, expected: '拒答', actual: answer.slice(0, 60) })
    } else {
      accuracyTotal++
      const hasKeyword = tc.expectedKeywords.some(k => answer.includes(k))
      if (hasKeyword) accuracyHits++
      else failures.push({ question: tc.question, expected: tc.expectedKeywords.join('/'), actual: answer.slice(0, 60) })
    }
  }

  console.log('=== RAG 质量评估报告 ===')
  console.log(`召回率: ${recallHits}/${recallTotal} = ${Math.round(recallHits / recallTotal * 100)}%`)
  console.log(`回答准确率: ${accuracyHits}/${accuracyTotal} = ${Math.round(accuracyHits / accuracyTotal * 100)}%`)
  if (failures.length > 0) {
    console.log('\n失败用例:')
    failures.forEach(f => console.log(`  Q: ${f.question}\n    期望: ${f.expected}\n    实际: ${f.actual}`))
  }
}

evaluate()
```

**要点**：
- 召回率 = 检索到正确文档的比例，回答准确率 = 最终回答正确的比例
- 两个指标分开看——召回率低说明检索有问题（分块/Embedding），准确率低但召回率高说明生成有问题（Prompt/模型）
- 失败用例要逐一分析原因，不要只看数字
