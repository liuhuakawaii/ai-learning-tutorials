### 2.2 创建 Checkout Session

```typescript
// app/api/checkout/route.ts
import { stripe } from '@/lib/stripe'
import { NextRequest } from 'next/server'

export async function POST(req: NextRequest) {
  const { priceId } = await req.json()
  const userId = req.headers.get('x-user-id')!
  const userEmail = req.headers.get('x-user-email')!

  // 查找或创建 Stripe Customer
  let customer = await findOrCreateCustomer(userId, userEmail)

  // 创建 Checkout Session
  const session = await stripe.checkout.sessions.create({
    customer: customer.id,
    mode: 'subscription',
    payment_method_types: ['card'],
    line_items: [{ price: priceId, quantity: 1 }],
    success_url: `${process.env.NEXT_PUBLIC_APP_URL}/dashboard?success=true`,
    cancel_url: `${process.env.NEXT_PUBLIC_APP_URL}/pricing?canceled=true`,
    metadata: { userId },
  })

  return Response.json({ url: session.url })
}

async function findOrCreateCustomer(userId: string, email: string) {
  // 查找已有 customer
  const existing = await db.users.findUnique({ where: { id: userId } })
  if (existing?.stripeCustomerId) {
    return { id: existing.stripeCustomerId }
  }

  // 创建新 customer
  const customer = await stripe.customers.create({
    email,
    metadata: { userId },
  })

  // 保存关联
  await db.users.update({
    where: { id: userId },
    data: { stripeCustomerId: customer.id },
  })

  return customer
}
```

### 2.4 Webhook 处理

```typescript
// app/api/webhook/stripe/route.ts
import { stripe } from '@/lib/stripe'
import Stripe from 'stripe'

export async function POST(req: NextRequest) {
  const body = await req.text()
  const sig = req.headers.get('stripe-signature')!

  let event: Stripe.Event

  try {
    event = stripe.webhooks.constructEvent(
      body,
      sig,
      process.env.STRIPE_WEBHOOK_SECRET!
    )
  } catch (err) {
    return new Response(`Webhook Error: ${err.message}`, { status: 400 })
  }

  switch (event.type) {
    case 'checkout.session.completed': {
      const session = event.data.object as Stripe.Checkout.Session
      await handleCheckoutComplete(session)
      break
    }

    case 'customer.subscription.updated': {
      const subscription = event.data.object as Stripe.Subscription
      await handleSubscriptionUpdate(subscription)
      break
    }

    case 'customer.subscription.deleted': {
      const subscription = event.data.object as Stripe.Subscription
      await handleSubscriptionCanceled(subscription)
      break
    }

    case 'invoice.payment_failed': {
      const invoice = event.data.object as Stripe.Invoice
      await handlePaymentFailed(invoice)
      break
    }
  }

  return new Response(null, { status: 200 })
}

async function handleCheckoutComplete(session: Stripe.Checkout.Session) {
  const userId = session.metadata?.userId
  const subscriptionId = session.subscription as string

  if (!userId) return

  const subscription = await stripe.subscriptions.retrieve(subscriptionId)

  await db.users.update({
    where: { id: userId },
    data: {
      plan: 'pro',
      stripeSubscriptionId: subscription.id,
      subscriptionStatus: subscription.status,
      currentPeriodEnd: new Date(subscription.current_period_end * 1000),
    },
  })
}

async function handleSubscriptionUpdate(subscription: Stripe.Subscription) {
  const customer = await stripe.customers.retrieve(subscription.customer as string)
  const userId = (customer as Stripe.Customer).metadata?.userId

  if (!userId) return

  await db.users.update({
    where: { id: userId },
    data: {
      subscriptionStatus: subscription.status,
      currentPeriodEnd: new Date(subscription.current_period_end * 1000),
    },
  })
}

async function handleSubscriptionCanceled(subscription: Stripe.Subscription) {
  const customer = await stripe.customers.retrieve(subscription.customer as string)
  const userId = (customer as Stripe.Customer).metadata?.userId

  if (!userId) return

  await db.users.update({
    where: { id: userId },
    data: {
      plan: 'free',
      subscriptionStatus: 'canceled',
      stripeSubscriptionId: null,
    },
  })
}

async function handlePaymentFailed(invoice: Stripe.Invoice) {
  // 通知用户支付失败
  const customer = await stripe.customers.retrieve(invoice.customer as string)
  const email = (customer as Stripe.Customer).email

  if (email) {
    await sendEmail(email, '支付失败通知', '您的订阅支付失败，请更新支付方式。')
  }
}
```

### 3.3 Webhook 处理

```typescript
// app/api/webhook/lemonsqueezy/route.ts
import { verifyWebhookSignature } from '@lemonsqueezy/lemonsqueezy.js'

export async function POST(req: NextRequest) {
  const body = await req.text()
  const signature = req.headers.get('x-signature')!

  const isValid = verifyWebhookSignature(body, signature, process.env.LEMONSQUEEZY_WEBHOOK_SECRET!)
  if (!isValid) {
    return new Response('Invalid signature', { status: 400 })
  }

  const event = JSON.parse(body)

  switch (event.meta.event_name) {
    case 'subscription_created':
    case 'subscription_updated': {
      const subscription = event.data
      const userId = subscription.attributes.custom_data?.userId

      await db.users.update({
        where: { id: userId },
        data: {
          plan: subscription.attributes.status === 'active' ? 'pro' : 'free',
          subscriptionStatus: subscription.attributes.status,
          currentPeriodEnd: subscription.attributes.renews_at,
        },
      })
      break
    }

    case 'subscription_cancelled': {
      const subscription = event.data
      const userId = subscription.attributes.custom_data?.userId

      await db.users.update({
        where: { id: userId },
        data: { plan: 'free', subscriptionStatus: 'canceled' },
      })
      break
    }
  }

  return new Response(null, { status: 200 })
}
```

```typescript
// lib/subscription.ts
export async function requireSubscription(userId: string, requiredPlan: 'pro' | 'team') {
  const user = await db.users.findUnique({ where: { id: userId } })

  if (!user) throw new Error('用户不存在')

  const planHierarchy = { free: 0, pro: 1, team: 2 }
  const userLevel = planHierarchy[user.plan as keyof typeof planHierarchy] ?? 0
  const requiredLevel = planHierarchy[requiredPlan]

  if (userLevel < requiredLevel) {
    throw new SubscriptionRequiredError(requiredPlan)
  }

  return user
}

// 在 API Route 中使用
export async function POST(req: NextRequest) {
  const userId = req.headers.get('x-user-id')!

  try {
    await requireSubscription(userId, 'pro')

        { status: 403 }
      )
```

## 常见误区

### 误区一：前端直接调支付 API

所有支付相关操作必须在服务端进行。前端只负责跳转到支付页面或调起支付弹窗，实际的金额、价格、订单创建都在服务端完成。

### 误区二：不处理 Webhook

支付结果不能只依赖前端回调。用户可能支付成功但关闭了页面，导致前端回调没触发。Webhook 是支付结果的唯一可靠来源。

### 误区三：只处理成功，不处理失败

支付失败、订阅过期、卡过期——这些情况都需要处理。不要假设用户永远会成功支付。

### 误区四：测试环境和生产环境混用

Stripe 的测试 Key 和生产 Key 完全不同。开发和测试时用 `sk_test_` 开头的 Key，上线后换成 `sk_live_` 开头的。混用会导致测试订单出现在生产环境。

## 工程建议

### 1. 先用 Stripe Test Mode 调通全流程

在 Stripe Dashboard 中启用 Test Mode，用测试卡号 `4242 4242 4242 4242` 走完注册 → 支付 → Webhook → 订阅管理的完整流程。

### 2. Webhook 要幂等

同一个 Webhook 事件可能被发送多次。处理 Webhook 时要检查事件是否已处理过，避免重复扣款或重复更新状态。

### 3. 本地开发用 Stripe CLI

`stripe listen --forward-to localhost:3000/api/webhook/stripe` 可以把 Stripe 的 Webhook 转发到本地，方便调试。

### 4. 记录所有支付事件

把每个 Webhook 事件都记录到数据库，包括事件类型、原始数据、处理结果。这是排查支付问题的关键数据。

---

## 小结

支付集成是产品变现的核心环节。Stripe 适合需要高度自定义的场景，LemonSqueezy 适合想快速上线的独立开发者。核心链路：创建 Checkout Session → 用户支付 → Webhook 通知 → 更新订阅状态。一定要处理好失败场景和 Webhook 幂等性。

```typescript
import { auth } from '@/lib/auth'
import { prisma } from '@/lib/db'

export async function POST(req: NextRequest) {
```

```tsx
          </button>
        </div>

        <div className="grid md:grid-cols-3 gap-8">
          {Object.entries(pricingPlans).map(([key, plan]) => (
            <div
              key={key}
              className={`rounded-2xl border p-8 bg-white ${plan.popular ? 'border-blue-600 shadow-lg relative' : 'border-gray-200'}`}

                </span>
              )}

                <span className="text-gray-500">/{billingCycle === 'monthly' ? '月' : '年'}</span>
              </div>
              <p className="text-gray-600 mb-6">{plan.description}</p>
              <ul className="space-y-3 text-left mb-8">
                {plan.features.map((f, i) => (
                  <li key={i} className="flex items-center gap-2">
                    <span className="text-green-500">✓</span> {f}
                  </li>
                ))}
              </ul>
              <button
                onClick={() => handleSubscribe(key)}
                disabled={loading === key}
                className={`w-full py-3 rounded-lg font-medium ${plan.popular ? 'bg-blue-600 text-white hover:bg-blue-700' : 'bg-gray-100 text-gray-800 hover:bg-gray-200'} disabled:opacity-50`}
              >
                {loading === key ? '处理中...' : plan.cta}
              </button>
            </div>
          ))}
        </div>
      </div>
    </div>
  )
}
```

### 练习四：Customer Portal

**思路**：创建一个 API Route 生成 Stripe Portal Session，前端提供入口按钮跳转。

**答案**：

```typescript
// app/api/portal/route.ts
import { stripe } from '@/lib/stripe'
import { prisma } from '@/lib/db'
import { auth } from '@/lib/auth'
import { NextRequest, NextResponse } from 'next/server'

export async function POST(req: NextRequest) {
  const session = await auth()
  if (!session?.user?.id) {
    return NextResponse.json({ error: '请先登录' }, { status: 401 })
  }

  const user = await prisma.user.findUnique({ where: { id: session.user.id } })

  if (!user?.stripeCustomerId) {
    return NextResponse.json({ error: '未找到订阅信息' }, { status: 400 })
  }

  const portalSession = await stripe.billingPortal.sessions.create({
    customer: user.stripeCustomerId,
    return_url: `${process.env.NEXT_PUBLIC_APP_URL}/dashboard/settings`,
  })

  return NextResponse.json({ url: portalSession.url })
}
```
