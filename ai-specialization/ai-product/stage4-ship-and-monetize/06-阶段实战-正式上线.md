```typescript
async function performanceCheck() {
  const checks = []

  // 1. 首页加载时间
  const homepageStart = Date.now()
  await fetch('https://yourapp.com')
  checks.push({ name: '首页加载', ms: Date.now() - homepageStart, target: 3000 })

  // 2. API 响应时间
  const apiStart = Date.now()
  await fetch('https://yourapp.com/api/health')
  checks.push({ name: 'API 响应', ms: Date.now() - apiStart, target: 500 })

  // 3. AI 首 Token 时间
  const aiStart = Date.now()
  const response = await fetch('https://yourapp.com/api/chat', {
    method: 'POST',
    body: JSON.stringify({ message: '你好' }),
  })
  // 流式响应的首 token
  const reader = response.body?.getReader()
  await reader?.read()
  checks.push({ name: 'AI 首 Token', ms: Date.now() - aiStart, target: 2000 })

  // 输出结果
  for (const check of checks) {
    const status = check.ms <= check.target ? '✅' : '❌'
    console.log(`${status} ${check.name}: ${check.ms}ms (目标: ${check.target}ms)`)
  }
}
```

## 常见误区

### 误区一：追求完美再上线

完美是上线的敌人。核心功能可用、没有严重 Bug 就可以上线。小问题可以在上线后快速修复。

### 误区二：上线后就不管了

上线只是开始，不是终点。上线后的第一周是最关键的——用户反馈、Bug 修复、数据监控都需要密切关注。

### 误区三：不做数据记录

上线第一天的数据是最宝贵的。哪个渠道带来的用户最多、用户在哪个环节流失、哪个功能最受欢迎——这些数据必须从第一天就开始记录。

### 误区四：忽略用户反馈

首批用户的反馈价值千金。他们发现的问题和提出的建议，比你自己想 100 遍都有用。

---

## 工程建议

### 1. 选一个安静的时间上线

不要在周五下午上线。选周一到周三的上午，给自己留出处理问题的时间。

### 2. 准备回滚方案

如果上线后出现严重问题，要知道怎么快速回滚到上一个版本。Vercel 的 Deployment 历史天然支持这一点。

### 3. 上线当天全程在线

发布当天安排好时间，确保能实时响应用户反馈和处理紧急问题。

### 4. 记录上线复盘

上线后写一份复盘文档：什么做得好、什么做得不好、下次怎么改进。这是宝贵的经验。

### 练习二：端到端测试

**思路**：按课程中的 E2E 清单，手动走完注册 → 核心功能 → 付费 → 订阅管理的完整链路。

**答案**：

```typescript
// scripts/e2e-check.ts - 端到端验证脚本
const BASE_URL = process.env.NEXT_PUBLIC_APP_URL || 'http://localhost:3000'

interface CheckResult {
  name: string
  status: 'pass' | 'fail' | 'skip'
  detail: string
  duration: number
}

async function runChecks() {
  const results: CheckResult[] = []

  // 检查 1：首页可访问
  await check('首页可访问', async () => {
    const res = await fetch(BASE_URL)
    if (!res.ok) throw new Error(`HTTP ${res.status}`)
    const html = await res.text()
    if (!html.includes('YourApp')) throw new Error('页面内容异常')
  }, results)

  // 检查 2：登录页可访问
  await check('登录页可访问', async () => {
    const res = await fetch(`${BASE_URL}/login`)
    if (!res.ok) throw new Error(`HTTP ${res.status}`)
  }, results)

  // 检查 3：定价页可访问
  await check('定价页可访问', async () => {
    const res = await fetch(`${BASE_URL}/pricing`)
    if (!res.ok) throw new Error(`HTTP ${res.status}`)
  }, results)

  // 检查 4：API 健康检查
  await check('API 健康检查', async () => {
    const res = await fetch(`${BASE_URL}/api/health`)
    const data = await res.json()
    if (data.status !== 'ok') throw new Error(JSON.stringify(data))
  }, results)

  // 检查 5：隐私政策页面
  await check('隐私政策可访问', async () => {
    const res = await fetch(`${BASE_URL}/privacy`)
    if (!res.ok) throw new Error(`HTTP ${res.status}`)
  }, results)

  // 检查 6：Sitemap
  await check('Sitemap 可访问', async () => {
    const res = await fetch(`${BASE_URL}/sitemap.xml`)
    if (!res.ok) throw new Error(`HTTP ${res.status}`)
  }, results)

  // 检查 7：Robots.txt
  await check('Robots.txt 可访问', async () => {
    const res = await fetch(`${BASE_URL}/robots.txt`)
    if (!res.ok) throw new Error(`HTTP ${res.status}`)
  }, results)

  // 检查 8：API 未认证请求返回 401
  await check('API 未认证返回 401', async () => {
    const res = await fetch(`${BASE_URL}/api/chat`, { method: 'POST' })
    if (res.status !== 401) throw new Error(`Expected 401, got ${res.status}`)
  }, results)

  // 输出结果
  console.log('\n=== 端到端检查结果 ===\n')
  for (const r of results) {
    const icon = r.status === 'pass' ? '✅' : r.status === 'fail' ? '❌' : '⏭️'
    console.log(`${icon} ${r.name} (${r.duration}ms)${r.status === 'fail' ? ` - ${r.detail}` : ''}`)
  }

  const passed = results.filter(r => r.status === 'pass').length
  const failed = results.filter(r => r.status === 'fail').length
  console.log(`\n总计: ${passed} 通过, ${failed} 失败`)
}

async function check(name: string, fn: () => Promise<void>, results: CheckResult[]) {
  const start = Date.now()
  try {
    await fn()
    results.push({ name, status: 'pass', detail: '', duration: Date.now() - start })
  } catch (err: any) {
    results.push({ name, status: 'fail', detail: err.message, duration: Date.now() - start })
  }
}

runChecks()
```

```bash
# 运行检查
npx tsx scripts/e2e-check.ts
```

**要点**：
- 自动化脚本覆盖基本的可访问性和 API 认证检查
- 支付链路（Stripe Checkout → Webhook）需要手动测试，用测试卡号 4242 4242 4242 4242
- 每次部署后运行检查脚本，确保没有回归

### 练习四：数据看板

```typescript
// lib/metrics.ts - 指标收集
import { prisma } from './db'

export async function getDashboardMetrics() {
  const now = new Date()
  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
  const thisMonth = new Date(now.getFullYear(), now.getMonth(), 1)
  const lastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1)

  const [
    todaySignups,
    monthSignups,
    totalUsers,
    todayActiveUsers,
    monthActiveUsers,
    payingUsers,
    monthRevenue,
    todayUsage,
  ] = await Promise.all([
    prisma.user.count({ where: { createdAt: { gte: today } } }),
    prisma.user.count({ where: { createdAt: { gte: thisMonth } } }),
    prisma.user.count(),
    prisma.usageLog.findMany({
      where: { createdAt: { gte: today } },
      select: { userId: true },
      distinct: ['userId'],
    }).then(r => r.length),
    prisma.usageLog.findMany({
      where: { createdAt: { gte: thisMonth } },
      select: { userId: true },
      distinct: ['userId'],
    }).then(r => r.length),
    prisma.user.count({ where: { plan: { in: ['pro', 'team'] } } }),
    prisma.payment.aggregate({
      where: { createdAt: { gte: thisMonth }, status: 'succeeded' },
      _sum: { amount: true },
    }),
    prisma.usageLog.count({ where: { createdAt: { gte: today } } }),
  ])

  const mrr = monthRevenue._sum.amount ?? 0
  const paidConversionRate = totalUsers > 0 ? (payingUsers / totalUsers * 100).toFixed(1) : '0'
  const activationRate = monthSignups > 0 ? (monthActiveUsers / monthSignups * 100).toFixed(1) : '0'

  return {
    user: {
      todaySignups,
      monthSignups,
      totalUsers,
      todayActiveUsers,
      monthActiveUsers,
      retentionRate: monthSignups > 0 ? `${activationRate}%` : 'N/A',
    },
    revenue: {
      mrr: `$${mrr}`,
      payingUsers,
      paidConversionRate: `${paidConversionRate}%`,
      arpu: payingUsers > 0 ? `$${(mrr / payingUsers).toFixed(2)}` : '$0',
    },
    product: {
      activationRate: `${activationRate}%`,
      todayUsage,
      avgUsagePerUser: monthActiveUsers > 0 ? Math.round(todayUsage / todayActiveUsers) : 0,
    },
  }
}
```

```tsx
function MetricCard({ label, value, highlight }: { label: string; value: string | number; highlight?: boolean }) {
  return (
    <div className={`rounded-lg p-4 ${highlight ? 'bg-blue-50 border border-blue-200' : 'bg-white border'}`}>
      <p className="text-sm text-gray-500">{label}</p>
      <p className={`text-2xl font-bold ${highlight ? 'text-blue-600' : ''}`}>{value}</p>
    </div>
  )
}
```

```typescript
// lib/monitoring/alerts.ts - 配置告警规则
export const alertRules = [
  {
    name: 'API 错误率过高',
    condition: async () => {
      // 检查最近 5 分钟的错误率
      const fiveMinAgo = new Date(Date.now() - 5 * 60 * 1000)
      const total = await prisma.apiLog.count({ where: { createdAt: { gte: fiveMinAgo } } })
      const errors = await prisma.apiLog.count({ where: { createdAt: { gte: fiveMinAgo }, status: { gte: 500 } } })
      return total > 0 && (errors / total) > 0.05
    },
    message: 'API 错误率超过 5%，请立即检查',
    severity: 'critical',
  },
  {
    name: '注册量归零',
    condition: async () => {
      const sixHoursAgo = new Date(Date.now() - 6 * 60 * 60 * 1000)
      const count = await prisma.user.count({ where: { createdAt: { gte: sixHoursAgo } } })
      return count === 0
    },
    message: '过去 6 小时无新注册，请检查注册流程',
    severity: 'warning',
  },
]
```

**要点**：
- 核心指标分四类：用户、收入、产品、AI 质量
- MRR（月经常性收入）和 ARPU（每用户平均收入）是最重要的收入指标
- 告警规则要覆盖错误率、注册量、支付成功率等关键指标
