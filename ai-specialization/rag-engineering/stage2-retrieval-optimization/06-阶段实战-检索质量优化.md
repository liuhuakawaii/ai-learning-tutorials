## 1. 评估集与指标

```python
# eval_data/retrieval_eval.json 中至少 20 个问答对
# 覆盖自然语言查询和精确术语查询

def recall_at_k(retrieved_ids, relevant_ids, k):
    return len(set(retrieved_ids[:k]) & set(relevant_ids)) / len(relevant_ids)

def mrr(retrieved_ids, relevant_ids):
    for i, doc_id in enumerate(retrieved_ids):
        if doc_id in relevant_ids:
            return 1 / (i + 1)
    return 0
```

## 2. 通用实验运行器

```python
import time, json
from dataclasses import dataclass, field

@dataclass
class ExperimentResult:
    name: str
    recall_at_5: float = 0
    mrr: float = 0
    avg_latency_ms: float = 0

def run_experiment(name, retriever_fn, eval_data):
    result = ExperimentResult(name=name)
    for item in eval_data:
        start = time.time()
        retrieved = retriever_fn(item["query"])
        latency = (time.time() - start) * 1000
        ids = [r["source"] for r in retrieved]
        result.recall_at_5 += recall_at_k(ids, item["relevant_doc_ids"], 5)
        result.mrr += mrr(ids, item["relevant_doc_ids"])
        result.avg_latency_ms += latency
    n = len(eval_data)
    result.recall_at_5 /= n
    result.mrr /= n
    result.avg_latency_ms /= n
    return result
```

## 3. 逐步实验

```python
# Step 0: 基线（纯向量检索）
def baseline_vector(query):
    vector = client.embeddings.create(model="text-embedding-3-small", input=query).data[0].embedding
    results = milvus.search(collection_name="test_docs", data=[vector], limit=5, output_fields=["text", "source"])
    return [{"text": h["entity"]["text"], "source": h["entity"]["source"]} for h in results[0]]

# Step 1: + BM25 混合检索 (RRF 融合)
def step1_hybrid(query):
    vector_results = baseline_vector(query)
    bm25_results = bm25_retriever.search(query, top_k=10)
    # RRF: score = sum(1 / (k + rank + 1))
    scores = {}
    for rank, r in enumerate(vector_results):
        scores[r["text"]] = scores.get(r["text"], 0) + 1 / (60 + rank + 1)
    for rank, (text, _) in enumerate(bm25_results):
        scores[text] = scores.get(text, 0) + 1 / (60 + rank + 1)
    ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:5]
    return [{"text": t, "source": "doc"} for t, _ in ranked]

# Step 2: + 查询扩展 (Multi-Query)
def step2_multi_query(query):
    variants = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": f"为以下问题生成3个不同表述，每行一个：\n{query}"}],
        temperature=0.3, max_tokens=200
    ).choices[0].message.content.strip().split("\n")[:3]

    all_results = []
    for q in [query] + variants:
        all_results.extend(step1_hybrid(q))
    seen, unique = set(), []
    for r in all_results:
        if r["text"] not in seen:
            seen.add(r["text"])
            unique.append(r)
    return unique[:5]

# Step 3: + Reranker 精排
from sentence_transformers import CrossEncoder
reranker = CrossEncoder("BAAI/bge-reranker-base")

def step3_reranker(query):
    candidates = step2_multi_query(query)
    pairs = [(query, c["text"]) for c in candidates]
    scores = reranker.predict(pairs)
    ranked = sorted(zip(candidates, scores), key=lambda x: x[1], reverse=True)
    return [{"text": r[0]["text"], "source": r[0]["source"]} for r in ranked[:5]]
```

## 6. 最终管线

```python
def final_pipeline(query):
    # 查询扩展 + 混合检索
    variants = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": f"生成2个变体：\n{query}"}],
        temperature=0.3, max_tokens=150
    ).choices[0].message.content.strip().split("\n")[:2]

    all_results = []
    for q in [query] + variants:
        all_results.extend(step1_hybrid(q))
    seen, candidates = set(), []
    for r in all_results:
        if r["text"] not in seen:
            seen.add(r["text"])
            candidates.append(r)

    # Reranker 精排
    pairs = [(query, c["text"]) for c in candidates[:10]]
    scores = reranker.predict(pairs)
    ranked = sorted(zip(candidates, scores), key=lambda x: x[1], reverse=True)
    return [{"text": r[0]["text"], "source": r[0]["source"]} for r in ranked[:5]]
```

## 参考答案

### 练习一

v2-m3 对中文支持更好，Recall@5 通常提升 2-5%，但延迟增加 30-50%。延迟敏感场景下 base 版本可能是更好选择。

### 练习三

分类路由的关键收益在精确查询上的表现改善。如果业务中精确查询占比高（技术客服），这个优化很有价值。
