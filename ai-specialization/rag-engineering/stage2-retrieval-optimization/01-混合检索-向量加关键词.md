## 1. 为什么需要混合检索？

在 Stage 1 中我们只用了向量检索。但单一检索方式有明显短板：

```
┌─────────────────────────────────────────────────────────────────┐
│                    单一检索的困境                                 │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  用户 Query: "iPhone 15 Pro Max 的电池容量是多少 mAh?"           │
│                                                                 │
│  向量检索结果:                                                    │
│    ✗ "智能手机电池技术发展趋势"        ← 语义相关，但没答案       │
│    ✗ "手机续航优化指南"               ← 语义相关，但没答案       │
│    ✓ "iPhone 15 Pro Max 配备 4422mAh 电池"  ← 正确答案排名低    │
│                                                                 │
│  关键词检索结果:                                                  │
│    ✓ "iPhone 15 Pro Max 配备 4422mAh 电池"  ← 精确匹配！       │
│    ✗ "iPhone 15 Pro Max 摄像头参数"         ← 匹配型号但无关    │

│                                                                 │

│                                                                 │
```

## 2. 向量检索 vs 关键词检索：核心对比

| 维度 | 向量检索 (Dense) | 关键词检索 (Sparse/BM25) |
|------|-----------------|-------------------------|
| **原理** | 将文本编码为稠密向量，计算语义相似度 | 基于词频(TF)和逆文档频率(IDF)的统计匹配 |
| **优势** | 语义理解、同义词匹配、跨语言 | 精确匹配、专有名词、数字/型号 |
| **劣势** | 对精确术语不敏感、计算成本高 | 无法理解语义、受词汇表限制 |
| **适用场景** | 模糊查询、自然语言问题 | 包含专有名词/型号/代码的查询 |
| **延迟** | 50-200ms (取决于索引大小) | 5-50ms (内存中很快) |
| **典型代表** | OpenAI Embedding、BGE、E5 | BM25、TF-IDF、SPLADE |

## 3. BM25 算法详解

BM25 是信息检索领域的经典算法，核心公式：

```
                         IDF(qi) · TF(qi, D) · (k1 + 1)
Score(Q, D) = Σ  ─────────────────────────────────────────
    qi∈Q∩D        TF(qi, D) + k1 · (1 - b + b · |D|/avgdl)

其中:
  IDF(qi) = log((N - n(qi) + 0.5) / (n(qi) + 0.5) + 1)
  TF(qi, D) = 词 qi 在文档 D 中的词频
  k1 = 词频饱和参数 (通常 1.2 ~ 2.0)
  b  = 文档长度归一化参数 (通常 0.75)
  |D| = 文档长度
  avgdl = 语料库平均文档长度
  N = 语料库文档总数
  n(qi) = 包含词 qi 的文档数
```

### 3.1 从零实现 BM25

```python
"""
BM25 检索器 - 从零实现
理解 BM25 的核心原理
"""
import math
import numpy as np
from typing import List, Dict, Tuple
from collections import Counter


class BM25Retriever:
    """BM25 检索器的纯 Python 实现"""

    def __init__(self, k1: float = 1.5, b: float = 0.75):
        self.k1 = k1
        self.b = b
        self.corpus: List[List[str]] = []
        self.doc_freqs: Dict[str, int] = {}
        self.doc_lens: List[int] = []
        self.avg_dl: float = 0.0
        self.num_docs: int = 0
        self.idf: Dict[str, float] = {}
        self.doc_store: List[Dict] = []

    def _tokenize(self, text: str) -> List[str]:
        """简单分词器（中文场景建议用 jieba）"""
        # 英文: 按空格分词 + 小写化
        # 中文: 这里用字符级，生产环境用 jieba
        import re
        text = text.lower()
        # 保留中文字符和英文单词
        tokens = re.findall(r'[\u4e00-\u9fff]|[a-z0-9]+', text)
        return tokens

    def fit(self, documents: List[Dict], text_field: str = "content"):
        """
        构建 BM25 索引

        Args:
            documents: 文档列表，每个文档是 dict
            text_field: 用于检索的文本字段名
        """
        self.doc_store = documents
        self.corpus = [self._tokenize(doc[text_field]) for doc in documents]
        self.num_docs = len(self.corpus)

        # 计算文档长度和平均长度
        self.doc_lens = [len(doc) for doc in self.corpus]
        self.avg_dl = sum(self.doc_lens) / self.num_docs if self.num_docs > 0 else 0

        # 计算每个词出现在多少文档中 (document frequency)
        df: Dict[str, int] = Counter()
        for doc_tokens in self.corpus:
            unique_tokens = set(doc_tokens)
            for token in unique_tokens:
                df[token] += 1
        self.doc_freqs = df

        # 计算 IDF
        for token, freq in df.items():
            self.idf[token] = math.log(
                (self.num_docs - freq + 0.5) / (freq + 0.5) + 1
            )

    def _score_doc(self, query_tokens: List[str], doc_idx: int) -> float:
        """计算查询与单个文档的 BM25 分数"""
        doc_tokens = self.corpus[doc_idx]
        doc_len = self.doc_lens[doc_idx]
        tf_counter = Counter(doc_tokens)

        score = 0.0
        for qt in query_tokens:
            if qt not in self.idf:
                continue
            tf = tf_counter.get(qt, 0)
            idf = self.idf[qt]
            # BM25 公式
            numerator = tf * (self.k1 + 1)
            denominator = tf + self.k1 * (1 - self.b + self.b * doc_len / self.avg_dl)
            score += idf * (numerator / denominator)
        return score

    def search(self, query: str, top_k: int = 5) -> List[Tuple[int, float, Dict]]:
        """
        检索 top_k 个最相关文档

        Args:
            query: 查询文本
            top_k: 返回数量

        Returns:
            [(doc_idx, score, document), ...]
        """
        query_tokens = self._tokenize(query)
        scores = []

        for idx in range(self.num_docs):
            score = self._score_doc(query_tokens, idx)
            scores.append((idx, score))

        # 按分数降序排列
        scores.sort(key=lambda x: x[1], reverse=True)

        results = []
        for idx, score in scores[:top_k]:
            results.append((idx, score, self.doc_store[idx]))
        return results


# ========== 使用示例 ==========
if __name__ == "__main__":
    documents = [
        {"id": 1, "content": "iPhone 15 Pro Max 配备 4422mAh 电池，支持 27W 快充"},
        {"id": 2, "content": "智能手机电池技术发展趋势：固态电池将取代锂离子电池"},
        {"id": 3, "content": "iPhone 15 Pro Max 摄像头采用 4800 万像素主摄"},
        {"id": 4, "content": "手机续航优化指南：关闭后台应用可延长电池使用时间"},
        {"id": 5, "content": "Samsung Galaxy S24 Ultra 电池容量为 5000mAh"},
        {"id": 6, "content": "iPhone 15 保护壳推荐：Spigen、OtterBox 等品牌"},
    ]

    retriever = BM25Retriever(k1=1.5, b=0.75)
    retriever.fit(documents, text_field="content")

    query = "iPhone 15 Pro Max 电池容量"
    results = retriever.search(query, top_k=3)

    print(f"Query: {query}")
    print(f"{'='*60}")
    for idx, score, doc in results:
        print(f"  Score: {score:.4f} | {doc['content']}")
```

## 4. 向量检索实现

```python
"""
向量检索器 - 使用 sentence-transformers
"""
from sentence_transformers import SentenceTransformer
import numpy as np
from typing import List, Dict, Tuple


class VectorRetriever:
    """基于稠密向量的检索器"""

    def __init__(self, model_name: str = "BAAI/bge-small-zh-v1.5"):
        self.model = SentenceTransformer(model_name)
        self.embeddings: np.ndarray = None
        self.doc_store: List[Dict] = []

    def fit(self, documents: List[Dict], text_field: str = "content"):
        """编码所有文档并构建索引"""
        self.doc_store = documents
        texts = [doc[text_field] for doc in documents]
        self.embeddings = self.model.encode(
            texts,
            normalize_embeddings=True,
            show_progress_bar=True
        )

    def search(self, query: str, top_k: int = 5) -> List[Tuple[int, float, Dict]]:
        """向量相似度检索"""
        query_vec = self.model.encode([query], normalize_embeddings=True)

        # 余弦相似度（已归一化，点积 = 余弦相似度）
        scores = np.dot(self.embeddings, query_vec.T).flatten()

        top_indices = np.argsort(scores)[::-1][:top_k]
        results = []
        for idx in top_indices:
            results.append((int(idx), float(scores[idx]), self.doc_store[idx]))
        return results
```

## 5. RRF 融合算法

**Reciprocal Rank Fusion (RRF)** 是多路检索融合的经典算法，简单有效：

```
                    1
RRF_score(d) = Σ  ─────────────
               r∈R  k + rank_r(d)

其中:
  R = 所有检索器的集合
  rank_r(d) = 文档 d 在检索器 r 中的排名 (从 1 开始)
  k = 平滑参数 (通常 k=60)
```

```
┌─────────────────────────────────────────────────────────────────┐
│                    RRF 融合流程示意                               │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  检索器 A (向量)          检索器 B (BM25)                        │
│  ┌──────────────┐        ┌──────────────┐                       │
│  │ Rank 1: Doc3 │        │ Rank 1: Doc1 │                       │
│  │ Rank 2: Doc1 │        │ Rank 2: Doc5 │                       │
│  │ Rank 3: Doc5 │        │ Rank 3: Doc3 │                       │
│  │ Rank 4: Doc2 │        │ Rank 4: Doc2 │                       │
│  └──────┬───────┘        └──────┬───────┘                       │
│         │                       │                               │
│         └───────────┬───────────┘                               │
│                     ▼                                           │
│            ┌──────────────┐                                     │
│            │  RRF 融合     │                                     │
│            │  k = 60      │                                     │
│            └──────┬───────┘                                     │
│                   ▼                                             │
│  ┌──────────────────────────────────┐                           │
│  │ 最终排名:                         │                           │
│  │ Doc1: 1/(60+2) + 1/(60+1) = 0.033│                          │
│  │ Doc3: 1/(60+1) + 1/(60+3) = 0.032│                          │
│  │ Doc5: 1/(60+3) + 1/(60+2) = 0.028│                          │
│  │ Doc2: 1/(60+4) + 1/(60+4) = 0.027│                          │
│  └──────────────────────────────────┘                           │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

### 5.1 RRF 实现

```python
"""
Reciprocal Rank Fusion (RRF) 实现
"""
from typing import List, Dict, Tuple
from collections import defaultdict


def reciprocal_rank_fusion(
    retrieval_results: List[List[Tuple[int, float, Dict]]],
    k: int = 60,
    weights: List[float] = None
) -> List[Tuple[float, Dict]]:
    """
    RRF 融合多路检索结果

    Args:
        retrieval_results: 多个检索器的结果列表
            每个元素是 [(doc_idx, score, doc), ...]
        k: RRF 平滑参数
        weights: 各检索器的权重 (默认等权)

    Returns:
        [(rrf_score, document), ...] 按 RRF 分数降序排列
    """
    num_retrievers = len(retrieval_results)
    if weights is None:
        weights = [1.0] * num_retrievers

    # 累加每个文档的 RRF 分数
    rrf_scores: Dict[int, float] = defaultdict(float)
    doc_map: Dict[int, Dict] = {}

    for retriever_idx, results in enumerate(retrieval_results):
        weight = weights[retriever_idx]
        for rank, (doc_idx, score, doc) in enumerate(results):
            rrf_scores[doc_idx] += weight * (1.0 / (k + rank + 1))
            doc_map[doc_idx] = doc

    # 按 RRF 分数降序排列
    sorted_docs = sorted(rrf_scores.items(), key=lambda x: x[1], reverse=True)
    return [(score, doc_map[doc_idx]) for doc_idx, score in sorted_docs]


# ========== 使用示例 ==========
if __name__ == "__main__":
    # 模拟两路检索结果
    vector_results = [
        (3, 0.92, {"id": 3, "content": "iPhone 15 Pro Max 摄像头参数"}),
        (1, 0.85, {"id": 1, "content": "iPhone 15 Pro Max 电池容量 4422mAh"}),
        (5, 0.78, {"id": 5, "content": "Samsung Galaxy S24 Ultra 电池"}),
        (2, 0.71, {"id": 2, "content": "智能手机电池技术发展趋势"}),
    ]

    bm25_results = [
        (1, 8.5, {"id": 1, "content": "iPhone 15 Pro Max 电池容量 4422mAh"}),
        (5, 6.2, {"id": 5, "content": "Samsung Galaxy S24 Ultra 电池"}),
        (3, 5.1, {"id": 3, "content": "iPhone 15 Pro Max 摄像头参数"}),
        (2, 4.3, {"id": 2, "content": "智能手机电池技术发展趋势"}),
    ]

    fused = reciprocal_rank_fusion(
        [vector_results, bm25_results],
        k=60,
        weights=[1.0, 1.2]  # BM25 权重略高
    )

    print("RRF 融合结果:")
    for score, doc in fused:
        print(f"  RRF Score: {score:.6f} | {doc['content']}")
```

## 8. 权重调优策略

```python
"""
混合检索权重调优实验
通过网格搜索找到最优权重组合
"""
import numpy as np
from itertools import product


def evaluate_hybrid_retrieval(
    queries: List[str],
    ground_truth: List[List[int]],
    bm25_retriever,
    vector_retriever,
    weights: Tuple[float, float],
    top_k: int = 5
) -> float:
    """
    评估混合检索的 Recall@K

    Args:
        queries: 查询列表
        ground_truth: 每个查询的相关文档 ID 列表
        bm25_retriever: BM25 检索器
        vector_retriever: 向量检索器
        weights: (bm25_weight, vector_weight)
        top_k: 返回文档数

    Returns:
        平均 Recall@K
    """
    total_recall = 0.0
    for query, gt_ids in zip(queries, ground_truth):
        # 两路检索
        bm25_results = bm25_retriever.search(query, top_k=top_k * 2)
        vector_results = vector_retriever.search(query, top_k=top_k * 2)

        # RRF 融合
        fused = reciprocal_rank_fusion(
            [bm25_results, vector_results],
            weights=list(weights)
        )

        # 计算 Recall
        retrieved_ids = {doc["id"] for _, doc in fused[:top_k]}
        hits = len(retrieved_ids & set(gt_ids))
        total_recall += hits / len(gt_ids) if gt_ids else 0

    return total_recall / len(queries)


def grid_search_weights(
    queries, ground_truth, bm25_retriever, vector_retriever
):
    """网格搜索最优权重"""
    best_recall = 0
    best_weights = (0.5, 0.5)

    weight_values = np.arange(0.1, 1.0, 0.1)

    for w_bm25 in weight_values:
        w_vector = 1.0 - w_bm25
        recall = evaluate_hybrid_retrieval(
            queries, ground_truth,
            bm25_retriever, vector_retriever,
            weights=(w_bm25, w_vector)
        )
        print(f"  BM25: {w_bm25:.1f} | Vector: {w_vector:.1f} | Recall@5: {recall:.4f}")

        if recall > best_recall:
            best_recall = recall
            best_weights = (w_bm25, w_vector)

    print(f"\n最优权重: BM25={best_weights[0]:.1f}, Vector={best_weights[1]:.1f}")
    print(f"最优 Recall@5: {best_recall:.4f}")
    return best_weights
```

## 9. 常见误区

### ❌ 错误 1: 直接拼接分数

```python
# 错误: 不同检索器的分数尺度不同，不能直接相加
combined_score = bm25_score + vector_score  # ← 绝对不要这样做！
```

**正确做法**: 使用 RRF 或其他基于排名的融合方法。

### ❌ 错误 2: 忽略分词质量

```python
# 错误: 用简单的 split() 处理中文
tokens = text.split(" ")  # 中文没有空格分隔！

# 正确: 使用专业分词器
import jieba
tokens = list(jieba.cut(text))
```

### ❌ 错误 3: 权重固定不调优

```python
# 错误: 直接使用默认权重
ensemble = EnsembleRetriever(
    retrievers=[bm25, vector],
    weights=[0.5, 0.5]  # 默认等权，不一定最优
)

# 正确: 根据业务数据调优
best_weights = grid_search_weights(queries, ground_truth, bm25, vector)
```

### ❌ 错误 4: Top-K 设置不一致

```python
# 错误: 两路检索的 Top-K 不同

vector = store.as_retriever(search_kwargs={"k": 10})

# 正确: 保持一致，或根据权重调整
bm25.k = 10
vector = store.as_retriever(search_kwargs={"k": 10})
```

## 10. 工程建议

1. **从 0.5:0.5 的权重开始调优**：向量和关键词的融合权重没有万能值。建议先各占 50%，然后在你的评估集上做网格搜索（0.1 步长），找到最优比例。
2. **BM25 对中文需要特殊处理**：英文 BM25 直接按空格分词即可，但中文需要先做分词（jieba 等）。分词质量直接影响关键词检索效果，建议对专业领域维护一份自定义词典。
3. **RRF 的 k 值不要随意改**：RRF 公式中的 k=60 是经过大量实验验证的默认值。除非你有充分的 A/B 测试数据，否则不要调整这个参数。
4. **监控各路召回的独立贡献**：在生产环境中记录每次查询中向量检索和关键词检索各自的 Top-K 结果。如果某一路长期没有命中，可能说明该路的参数需要调整或者数据索引有问题。

### 练习 1: 实现加权 RRF

**思路**：加权 RRF 的核心是在融合公式中为每路检索器乘以不同权重。与等权 RRF 不同，加权版本允许你根据各路检索器的历史表现调整其贡献度。实现时需要注意权重归一化，以及处理某些检索器可能返回空结果的情况。

**答案**：

```python
from typing import List, Tuple, Dict, Optional
from collections import defaultdict


def weighted_rrf(
    retrieval_results: List[List[Tuple[int, float, Dict]]],
    k: int = 60,
    weights: Optional[List[float]] = None
) -> List[Tuple[float, Dict]]:
    """
    加权 RRF 融合

    Args:
        retrieval_results: 多个检索器的结果列表
            每个元素是 [(doc_idx, score, doc), ...]
        k: RRF 平滑参数
        weights: 各检索器的权重，默认等权

    Returns:
        [(rrf_score, document), ...] 按 RRF 分数降序排列
    """
    num_retrievers = len(retrieval_results)
    if weights is None:
        weights = [1.0] * num_retrievers

    if len(weights) != num_retrievers:
        raise ValueError(f"权重数量 ({len(weights)}) 与检索器数量 ({num_retrievers}) 不匹配")

    # 归一化权重
    total_weight = sum(weights)
    norm_weights = [w / total_weight for w in weights]

    rrf_scores: Dict[int, float] = defaultdict(float)
    doc_map: Dict[int, Dict] = {}

    for retriever_idx, results in enumerate(retrieval_results):
        w = norm_weights[retriever_idx]
        for rank, (doc_idx, score, doc) in enumerate(results):
            rrf_scores[doc_idx] += w / (k + rank + 1)
            doc_map[doc_idx] = doc

    sorted_docs = sorted(rrf_scores.items(), key=lambda x: x[1], reverse=True)
    return [(score, doc_map[doc_idx]) for doc_idx, score in sorted_docs]
```

### 练习 2: 中文 BM25 优化

**思路**：中文 BM25 的关键瓶颈在分词质量。jieba 能识别中文词汇边界，而字符级分词会产生大量无意义的单字符 token，拉低 IDF 质量。加上停用词过滤可以进一步减少噪音。对比实验需要在同一测试集上分别计算两种分词方案的 Recall@5。

**答案**：

```python
import math
import re
import jieba
from typing import List, Dict, Tuple, Set
from collections import Counter


STOPWORDS = set([
    "的", "了", "在", "是", "我", "有", "和", "就", "不", "人", "都",
    "一", "一个", "上", "也", "很", "到", "说", "要", "去", "你",
    "会", "着", "没有", "看", "好", "自己", "这", "他", "她", "它",
    "吗", "什么", "那", "这个", "那个", "可以", "能", "对", "但",
])


class BM25Retriever:
    def __init__(self, k1: float = 1.5, b: float = 0.75, use_jieba: bool = False):
        self.k1 = k1
        self.b = b
        self.use_jieba = use_jieba
        self.corpus: List[List[str]] = []
        self.doc_freqs: Dict[str, int] = {}
        self.doc_lens: List[int] = []
        self.avg_dl: float = 0.0
        self.num_docs: int = 0
        self.idf: Dict[str, float] = {}
        self.doc_store: List[Dict] = []

    def _tokenize(self, text: str) -> List[str]:
        text = text.lower()
        if self.use_jieba:
            tokens = list(jieba.cut(text))
            tokens = [t.strip() for t in tokens if t.strip() and t not in STOPWORDS]
        else:
            tokens = re.findall(r'[\u4e00-\u9fff]|[a-z0-9]+', text)
        return tokens

    def fit(self, documents: List[Dict], text_field: str = "content"):
        self.doc_store = documents
        self.corpus = [self._tokenize(doc[text_field]) for doc in documents]
        self.num_docs = len(self.corpus)
        self.doc_lens = [len(doc) for doc in self.corpus]
        self.avg_dl = sum(self.doc_lens) / self.num_docs if self.num_docs > 0 else 0
        df: Dict[str, int] = Counter()
        for doc_tokens in self.corpus:
            for token in set(doc_tokens):
                df[token] += 1
        self.doc_freqs = df
        for token, freq in df.items():
            self.idf[token] = math.log((self.num_docs - freq + 0.5) / (freq + 0.5) + 1)

    def search(self, query: str, top_k: int = 5) -> List[Tuple[int, float, Dict]]:
        query_tokens = self._tokenize(query)
        scores = []
        for idx in range(self.num_docs):
            doc_tokens = self.corpus[idx]
            doc_len = self.doc_lens[idx]
            tf_counter = Counter(doc_tokens)
            score = 0.0
            for qt in query_tokens:
                if qt not in self.idf:
                    continue
                tf = tf_counter.get(qt, 0)
                numerator = tf * (self.k1 + 1)
                denominator = tf + self.k1 * (1 - self.b + self.b * doc_len / self.avg_dl)
                score += self.idf[qt] * (numerator / denominator)
            scores.append((idx, score))
        scores.sort(key=lambda x: x[1], reverse=True)
        return [(idx, score, self.doc_store[idx]) for idx, score in scores[:top_k]]


def recall_at_k(retrieved_ids: Set[int], relevant_ids: Set[int], k: int) -> float:
    if not relevant_ids:
        return 0.0
    return len(retrieved_ids & relevant_ids) / len(relevant_ids)
```

### 练习 3: 混合检索 + Reranker 完整管线

**思路**：完整的两阶段检索管线由粗排（混合检索）和精排（Cross-Encoder Reranker）组成。粗排阶段用 BM25 + 向量检索通过 RRF 融合获取候选集，精排阶段用 Cross-Encoder 对候选集重新排序。对比有无 Reranker 时的 Recall@5 和 MRR 可以量化精排的贡献。

**答案**：

```python
import math
import numpy as np
from typing import List, Dict, Tuple, Set
from collections import defaultdict
from sentence_transformers import SentenceTransformer, CrossEncoder


class BM25RetrieverSimple:
    def __init__(self, k1=1.5, b=0.75):
        self.k1, self.b = k1, b
        self.corpus, self.idf, self.doc_store = [], {}, []
        self.doc_lens, self.avg_dl, self.num_docs = [], 0, 0

    def _tokenize(self, text):
        import re
        return re.findall(r'[\u4e00-\u9fff]|[a-z0-9]+', text.lower())

    def fit(self, docs, field="content"):
        import jieba
        self.doc_store = docs
        self.corpus = [list(jieba.cut(doc[field].lower())) for doc in docs]
        self.num_docs = len(self.corpus)
        self.doc_lens = [len(t) for t in self.corpus]
        self.avg_dl = sum(self.doc_lens) / self.num_docs
        from collections import Counter
        df = Counter()
        for tokens in self.corpus:
            for t in set(tokens):
                df[t] += 1
        for t, f in df.items():
            self.idf[t] = math.log((self.num_docs - f + 0.5) / (f + 0.5) + 1)

    def search(self, query, top_k=10):
        import jieba
        from collections import Counter
        qt = list(jieba.cut(query.lower()))
        results = []
        for idx in range(self.num_docs):
            tf = Counter(self.corpus[idx])
            dl = self.doc_lens[idx]
            score = sum(
                self.idf.get(t, 0) * (tf.get(t, 0) * (self.k1 + 1)) /
                (tf.get(t, 0) + self.k1 * (1 - self.b + self.b * dl / self.avg_dl))
                for t in qt if t in self.idf
            )
            results.append((idx, score, self.doc_store[idx]))
        results.sort(key=lambda x: x[1], reverse=True)
        return results[:top_k]


def rrf_fusion(results_list, k=60):
    scores = defaultdict(float)
    doc_map = {}
    for results in results_list:
        for rank, (idx, _, doc) in enumerate(results):
            scores[idx] += 1.0 / (k + rank + 1)
            doc_map[idx] = doc
    sorted_docs = sorted(scores.items(), key=lambda x: x[1], reverse=True)
    return [(s, doc_map[i]) for i, s in sorted_docs]


def recall_at_k(ids, relevant, k):
    return len(set(ids[:k]) & relevant) / len(relevant) if relevant else 0


def mrr_at_k(ids, relevant, k):
    for i, d in enumerate(ids[:k]):
        if d in relevant:
            return 1.0 / (i + 1)
    return 0.0

        print(f"{test_queries[i][0]:<25} {no_rerank_results[i]['recall']:<18.3f} {with_rerank_results[i]['recall']:<18.3f}")

    avg_recall_no = sum(r["recall"] for r in no_rerank_results) / n
    avg_recall_with = sum(r["recall"] for r in with_rerank_results) / n
    avg_mrr_no = sum(r["mrr"] for r in no_rerank_results) / n
    avg_mrr_with = sum(r["mrr"] for r in with_rerank_results) / n
    print(f"\n平均 Recall@5: 无Reranker={avg_recall_no:.3f}  有Reranker={avg_recall_with:.3f}")
    print(f"平均 MRR:      无Reranker={avg_mrr_no:.3f}  有Reranker={avg_mrr_with:.3f}")
```

**要点**：
- Reranker 使用 Cross-Encoder 对 query-document 对联合编码，能捕捉深层语义交互，比 Bi-Encoder 的相似度分数更准确
- RRF 融合后再 Rerank 是标准的两阶段管线，粗排保证召回，精排保证排序质量
- Reranker 的推理延迟约为 100-200ms/批，生产环境中只对 Top-20 候选做精排即可
