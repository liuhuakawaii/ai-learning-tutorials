## 场景引入

监控数据显示，你的 RAG 系统每天收到约 5 万次查询，其中 30% 的查询是重复或高度相似的——比如"如何重置密码"每天被问 500 次。每次都走完整的检索+生成流程不仅浪费 API 成本，还增加了不必要的延迟。你需要一个缓存层，让重复或相似的查询直接返回缓存结果，将响应时间从 3 秒降到 50 毫秒，同时显著降低 API 调用成本。

## 学习目标

1. **精确匹配缓存** — 实现基于 key 的精确缓存，处理完全相同的查询
2. **语义缓存** — 基于向量相似度识别语义等价的查询
3. **多级缓存架构** — 设计 L1/L2/L3 三级缓存体系
4. **缓存失效策略** — 实现 TTL、LRU 和基于事件的缓存更新

## 1. 为什么 RAG 需要缓存

```
RAG 系统中缓存的价值
═══════════════════════════════════════════════════════════════

无缓存:
  用户查询 → Embedding → 检索 → LLM 生成 → 返回
  │  150ms    │  200ms  │  2000ms  │
  └──────────────────────────────────┘
              总计: ~2350ms              成本: $0.02/query

有缓存 (命中率 60%):
  用户查询 → [缓存命中] → 直接返回
  │  5ms  │
  └───────┘
  总计: ~5ms (缓存命中)            成本: $0.00/query

  用户查询 → [缓存未命中] → Embedding → 检索 → LLM → 写入缓存
  │   150ms    │  200ms  │  2000ms  │
  └──────────────────────────────────┘
              总计: ~2350ms              成本: $0.02/query

  平均延迟: 0.6 × 5ms + 0.4 × 2350ms = 943ms  (↓ 60%)
  平均成本: 0.6 × $0 + 0.4 × $0.02 = $0.008    (↓ 60%)
```

## 2. 精确匹配缓存

### 2.1 Redis 精确缓存实现

```python
import hashlib
import json
import redis.asyncio as redis
from typing import Any, Optional
from datetime import timedelta

class ExactMatchCache:
    """基于精确匹配的 RAG 缓存"""

    def __init__(self, redis_url: str = "redis://localhost:6379",
                 prefix: str = "rag:cache",
                 default_ttl: int = 3600):
        self.redis = redis.from_url(redis_url, decode_responses=True)
        self.prefix = prefix
        self.default_ttl = default_ttl
        self._hits = 0
        self._misses = 0

    def _make_key(self, query: str, context: dict = None) -> str:
        """生成缓存 key"""
        # 对查询和上下文进行哈希
        content = {"query": query.strip().lower()}
        if context:
            content["context"] = context
        raw = json.dumps(content, sort_keys=True)
        hash_val = hashlib.sha256(raw.encode()).hexdigest()[:16]
        return f"{self.prefix}:{hash_val}"

    async def get(self, query: str, context: dict = None) -> Optional[dict]:
        """查询缓存"""
        key = self._make_key(query, context)
        data = await self.redis.get(key)

        if data:
            self._hits += 1
            return json.loads(data)

        self._misses += 1
        return None

    async def set(self, query: str, result: dict, context: dict = None,
                  ttl: int = None) -> None:
        """写入缓存"""
        key = self._make_key(query, context)
        ttl = ttl or self.default_ttl
        await self.redis.setex(key, ttl, json.dumps(result))

    async def invalidate(self, query: str, context: dict = None) -> bool:
        """手动失效缓存"""
        key = self._make_key(query, context)
        return await self.redis.delete(key) > 0

    async def invalidate_pattern(self, pattern: str) -> int:
        """按模式批量失效"""
        keys = []
        async for key in self.redis.scan_iter(f"{self.prefix}:{pattern}"):
            keys.append(key)
        if keys:
            return await self.redis.delete(*keys)
        return 0

    @property
    def hit_rate(self) -> float:
        total = self._hits + self._misses
        return self._hits / total if total > 0 else 0.0

    async def get_stats(self) -> dict:
        """获取缓存统计"""
        info = await self.redis.info("stats")
        total = self._hits + self._misses
        return {
            "hits": self._hits,
            "misses": self._misses,
            "hit_rate": f"{self.hit_rate:.1%}",
            "total_queries": total,
            "redis_keyspace_hits": info.get("keyspace_hits", 0),
        }
```

### 2.2 缓存旁路模式 (Cache-Aside)

```python
class CacheAsideRAG:
    """缓存旁路模式的 RAG Pipeline"""

    def __init__(self, cache: ExactMatchCache, rag_pipeline):
        self.cache = cache
        self.pipeline = rag_pipeline

    async def query(self, question: str, bypass_cache: bool = False) -> dict:
        """带缓存的 RAG 查询"""
        # 1. 查缓存
        if not bypass_cache:
            cached = await self.cache.get(question)
            if cached:
                cached["from_cache"] = True
                return cached

        # 2. 缓存未命中，执行 RAG
        result = await self.pipeline.query(question)

        # 3. 写入缓存（异步，不阻塞响应）
        asyncio.create_task(
            self.cache.set(question, result, ttl=self._calculate_ttl(question))
        )

        result["from_cache"] = False
        return result

    def _calculate_ttl(self, question: str) -> int:
        """根据问题类型动态调整 TTL"""
        # 事实性问题缓存更久
        factual_keywords = ["什么是", "定义", "who is", "when was"]
        if any(kw in question.lower() for kw in factual_keywords):
            return 7200  # 2 小时
        # 时效性问题缓存更短
        time_keywords = ["最新", "今天", "latest", "current"]
        if any(kw in question.lower() for kw in time_keywords):
            return 300   # 5 分钟
        return 3600       # 默认 1 小时
```

## 3. 语义缓存

### 3.1 语义缓存原理

精确匹配无法处理语义等价的查询。例如：
- "Python 怎么读取 CSV 文件？"
- "How to read CSV in Python?"

这两个查询语义相同，但精确匹配会认为是不同的查询。

```
语义缓存工作流程
═══════════════════════════════════════════════════════════════

用户查询: "Python 读取 CSV 的方法"

  Step 1: Embedding
    ─────────────────────────────────────────────────────
    查询文本 ──→ Embedding Model ──→ 向量 [0.12, -0.34, ...]
    ─────────────────────────────────────────────────────

  Step 2: 向量相似度搜索 (在缓存向量库中)
    ─────────────────────────────────────────────────────
    缓存向量库:
      "Python怎么读CSV?"        [0.11, -0.33, ...]  相似度: 0.96 ✓
      "JavaScript 排序方法"     [0.78, 0.45, ...]   相似度: 0.23 ✗
      "机器学习入门教程"         [-0.56, 0.89, ...]  相似度: 0.15 ✗
    ─────────────────────────────────────────────────────

  Step 3: 判断阈值
    最高相似度 0.96 > 阈值 0.92 → 缓存命中！
    返回缓存的答案

  Step 4: 缓存未命中时
    相似度 < 阈值 → 执行完整 RAG → 将查询向量和答案存入缓存
```

### 3.2 语义缓存实现

```python
import numpy as np
from dataclasses import dataclass
from typing import Optional
import chromadb
from openai import AsyncOpenAI

@dataclass
class CacheEntry:
    query: str
    answer: dict
    embedding: list[float]
    created_at: float
    access_count: int = 0

class SemanticCache:
    """基于向量相似度的语义缓存"""

    def __init__(self, embedder: AsyncOpenAI,
                 similarity_threshold: float = 0.92,
                 collection_name: str = "semantic_cache"):
        self.embedder = embedder
        self.threshold = similarity_threshold
        self.client = chromadb.Client()
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"}
        )
        self._answer_store: dict[str, dict] = {}

    async def get(self, query: str) -> Optional[dict]:
        """语义查找缓存"""
        embedding = await self._embed(query)

        results = self.collection.query(
            query_embeddings=[embedding],
            n_results=1,
            include=["distances", "metadatas"]
        )

        if not results["distances"][0]:
            return None

        distance = results["distances"][0][0]
        similarity = 1 - distance  # cosine distance -> similarity

        if similarity >= self.threshold:
            doc_id = results["ids"][0][0]
            self.collection.update(
                ids=[doc_id],
                metadatas=[{"access_count":
                    results["metadatas"][0][0].get("access_count", 0) + 1}]
            )
            return self._answer_store.get(doc_id)

        return None

    async def set(self, query: str, answer: dict, ttl: int = 3600) -> None:
        """写入语义缓存"""
        embedding = await self._embed(query)
        doc_id = hashlib.sha256(query.encode()).hexdigest()[:16]

        self.collection.upsert(
            ids=[doc_id],
            embeddings=[embedding],
            documents=[query],
            metadatas=[{
                "created_at": time.time(),
                "ttl": ttl,
                "access_count": 0
            }]
        )
        self._answer_store[doc_id] = answer

    async def _embed(self, text: str) -> list[float]:
        response = await self.embedder.embeddings.create(
            model="text-embedding-3-small",
            input=[text]
        )
        return response.data[0].embedding
```

### 4.1 多级缓存实现

```python
import asyncio
import time
from collections import OrderedDict
from typing import Optional

class LRUMemoryCache:
    """L1: 内存 LRU 缓存"""

    def __init__(self, max_size: int = 1000, ttl: int = 300):
        self.max_size = max_size
        self.ttl = ttl
        self._cache: OrderedDict[str, tuple[dict, float]] = OrderedDict()
        self._hits = 0
        self._misses = 0

    async def get(self, key: str) -> Optional[dict]:
        if key in self._cache:
            value, expires_at = self._cache[key]
            if time.time() < expires_at:
                self._cache.move_to_end(key)
                self._hits += 1
                return value
            else:
                del self._cache[key]
        self._misses += 1
        return None

    async def set(self, key: str, value: dict) -> None:
        if key in self._cache:
            del self._cache[key]
        elif len(self._cache) >= self.max_size:
            self._cache.popitem(last=False)
        self._cache[key] = (value, time.time() + self.ttl)

    @property
    def hit_rate(self) -> float:
        total = self._hits + self._misses
        return self._hits / total if total > 0 else 0.0


class MultiLevelCache:
    """多级缓存管理器"""

    def __init__(self, l1: LRUMemoryCache,
                 l2: SemanticCache,
                 l3: ExactMatchCache):
        self.l1 = l1
        self.l2 = l2
        self.l3 = l3
        self._stats = {"l1_hits": 0, "l2_hits": 0, "l3_hits": 0, "misses": 0}

    async def get(self, query: str) -> Optional[dict]:
        """逐级查找缓存"""
        cache_key = self._normalize(query)

        # L1: 内存缓存
        result = await self.l1.get(cache_key)
        if result:
            self._stats["l1_hits"] += 1
            result["_cache_level"] = "L1"
            return result

        # L2: 语义缓存
        result = await self.l2.get(query)
        if result:
            self._stats["l2_hits"] += 1
            await self.l1.set(cache_key, result)
            result["_cache_level"] = "L2"
            return result

        # L3: 精确缓存 (Redis)
        result = await self.l3.get(query)
        if result:
            self._stats["l3_hits"] += 1
            await self.l1.set(cache_key, result)
            result["_cache_level"] = "L3"
            return result

        self._stats["misses"] += 1
        return None

    async def set(self, query: str, result: dict) -> None:
        """写入所有缓存层"""
        cache_key = self._normalize(query)
        await asyncio.gather(
            self.l1.set(cache_key, result),
            self.l2.set(query, result),
            self.l3.set(query, result),
        )

    def _normalize(self, query: str) -> str:
        return query.strip().lower()

    def get_stats(self) -> dict:
        total = sum(self._stats.values())
        return {
            **self._stats,
            "total": total,
            "overall_hit_rate": f"{(1 - self._stats['misses'] / total):.1%}" if total else "N/A"
        }
```

## 5. 缓存失效策略

### 5.1 TTL 管理

```python
from enum import Enum
from typing import Callable

class CacheCategory(Enum):
    FACTUAL = "factual"         # 事实性问题，缓存久
    PROCEDURAL = "procedural"   # 操作性问题，中等缓存
    TEMPORAL = "temporal"       # 时效性问题，短缓存
    DYNAMIC = "dynamic"         # 动态内容，最短缓存

TTL_CONFIG = {
    CacheCategory.FACTUAL: 7200,      # 2 小时
    CacheCategory.PROCEDURAL: 3600,   # 1 小时
    CacheCategory.TEMPORAL: 300,      # 5 分钟
    CacheCategory.DYNAMIC: 60,        # 1 分钟
}

class AdaptiveTTLCalculator:
    """自适应 TTL 计算器"""

    TEMPORAL_KEYWORDS = ["最新", "今天", "现在", "latest", "current", "today"]
    FACTUAL_KEYWORDS = ["什么是", "定义", "who", "when", "where", "定义"]

    def calculate(self, query: str, answer: dict = None) -> int:
        """根据查询类型和答案特征计算 TTL"""
        category = self._classify(query)
        base_ttl = TTL_CONFIG[category]

        # 根据答案长度调整（长答案可能包含更多细节，值得缓存更久）
        if answer:
            answer_length = len(str(answer.get("answer", "")))
            if answer_length > 1000:
                base_ttl = int(base_ttl * 1.5)

        return base_ttl

    def _classify(self, query: str) -> CacheCategory:
        lower_query = query.lower()
        if any(kw in lower_query for kw in self.TEMPORAL_KEYWORDS):
            return CacheCategory.TEMPORAL
        if any(kw in lower_query for kw in self.FACTUAL_KEYWORDS):
            return CacheCategory.FACTUAL
        return CacheCategory.PROCEDURAL
```

### 5.2 基于事件的缓存更新

```python
class CacheInvalidator:
    """基于事件的缓存失效管理器"""

    def __init__(self, cache: MultiLevelCache):
        self.cache = cache
        self._watchers: dict[str, list[str]] = {}

    async def on_document_updated(self, doc_id: str) -> None:
        """文档更新时，失效相关缓存"""
        affected_queries = self._watchers.get(doc_id, [])
        for query in affected_queries:
            await self.cache.l3.invalidate(query)
        # 清空 watcher
        self._watchers[doc_id] = []

    async def on_document_deleted(self, doc_id: str) -> None:
        """文档删除时，失效相关缓存"""
        await self.on_document_updated(doc_id)

    async def on_index_rebuild(self) -> None:
        """索引重建时，清空所有缓存"""
        await self.cache.l3.invalidate_pattern("*")
        self._watchers.clear()

    def register_dependency(self, query: str, doc_ids: list[str]) -> None:
        """注册查询对文档的依赖关系"""
        for doc_id in doc_ids:
            if doc_id not in self._watchers:
                self._watchers[doc_id] = []
            self._watchers[doc_id].append(query)
```

## 6. 缓存策略对比

| 特性 | 精确缓存 | 语义缓存 | 多级缓存 |
|------|----------|----------|----------|
| **命中条件** | key 完全匹配 | 向量相似度 > 阈值 | 任一级命中 |
| **延迟** | 1-5ms | 10-50ms | 1-50ms |
| **内存占用** | 低 | 高（存向量） | 中 |
| **实现复杂度** | 低 | 高 | 中 |
| **适用场景** | 重复查询多 | 语义变体多 | 通用场景 |
| **误命中风险** | 无 | 低（阈值控制） | 低 |
| **维护成本** | 低 | 中 | 高 |

## 7. 常见误区

### ❌ 错误 1: 缓存 key 设计不当

```python
# 错误：用原始查询作为 key，大小写和空格导致缓存失效
key = query  # "Hello" 和 "hello" 是不同的 key

# 正确：归一化处理
key = query.strip().lower()
```

### ❌ 错误 2: 没有设置 TTL

```python
# 错误：永久缓存，数据永远不会更新
await redis.set(key, value)  # 永不过期

# 正确：设置合理 TTL
await redis.setex(key, ttl=3600, value=value)
```

### ❌ 错误 3: 语义缓存阈值设置不当

```python
# 错误：阈值太低，不相关的查询也会命中
cache = SemanticCache(similarity_threshold=0.7)  # "Python" 和 "蟒蛇" 可能命中

# 正确：阈值需要根据业务场景调优
cache = SemanticCache(similarity_threshold=0.92)  # 需要高度相似才命中
```

## 8. 工程建议

1. **语义缓存的相似度阈值要保守设置**：阈值设得太高会错过缓存机会，设得太低会返回错误答案。建议从 0.95 开始，逐步降低，每次降低都要验证缓存命中结果的正确率。
2. **缓存失效策略要与知识库更新联动**：当知识库有文档更新时，相关的缓存条目必须立即失效。否则用户会收到过时的答案，这比没有缓存更糟糕。
3. **监控缓存命中率和正确率**：缓存命中率是效率指标，缓存正确率是质量指标。两个都要监控，不能只追求命中率而忽略了返回错误答案的风险。
4. **冷启动时预热缓存**：系统刚启动或缓存刚清空时，用历史高频查询预热缓存，避免冷启动期间所有请求都穿透到底层系统。

## 9. 本课总结

```
RAG 缓存策略要点
═══════════════════════════════════════════════

  1. 精确缓存是基础
     └─ 简单有效，处理完全相同的查询

  2. 语义缓存提升覆盖率
     └─ 处理语义等价的不同表述

  3. 多级缓存兼顾速度和覆盖
     └─ L1 内存 → L2 语义 → L3 精确

  4. TTL 需要根据内容类型调整
     └─ 事实性久、时效性短

  5. 缓存失效比缓存写入更重要
     └─ 确保数据一致性
```

### 练习 1: 实现带统计的精确缓存

**思路**：在 `ExactMatchCache` 基础上用列表记录每次命中的响应时间，用字典按小时分桶统计命中/未命中次数，最后提供查询接口返回命中率趋势。

**答案**：

```python
import time
from collections import defaultdict
from datetime import datetime

class StatsExactMatchCache(ExactMatchCache):
    """带统计功能的精确缓存"""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._hit_latencies: list[float] = []
        self._hourly_stats: dict[str, dict[str, int]] = defaultdict(
            lambda: {"hits": 0, "misses": 0}
        )

    async def get(self, query: str, context: dict = None):
        start = time.perf_counter()
        result = await super().get(query, context)
        elapsed_ms = (time.perf_counter() - start) * 1000

        hour_key = datetime.now().strftime("%Y-%m-%d %H:00")
        if result is not None:
            self._hit_latencies.append(elapsed_ms)
            self._hourly_stats[hour_key]["hits"] += 1
        else:
            self._hourly_stats[hour_key]["misses"] += 1

        return result

    def get_hit_latency_stats(self) -> dict:
        """获取缓存命中的响应时间统计"""
        if not self._hit_latencies:
            return {"count": 0, "avg_ms": 0, "p50_ms": 0, "p95_ms": 0, "p99_ms": 0}
        sorted_lat = sorted(self._hit_latencies)
        n = len(sorted_lat)
        return {
            "count": n,
            "avg_ms": round(sum(sorted_lat) / n, 2),
            "p50_ms": round(sorted_lat[int(n * 0.5)], 2),
            "p95_ms": round(sorted_lat[int(n * 0.95)] if n > 1 else sorted_lat[-1], 2),
            "p99_ms": round(sorted_lat[int(n * 0.99)] if n > 1 else sorted_lat[-1], 2),
        }

    def get_hourly_hit_rate(self) -> list[dict]:
        """获取按小时统计的命中率趋势"""
        trend = []
        for hour in sorted(self._hourly_stats.keys()):
            stats = self._hourly_stats[hour]
            total = stats["hits"] + stats["misses"]
            trend.append({
                "hour": hour,
                "hits": stats["hits"],
                "misses": stats["misses"],
                "hit_rate": round(stats["hits"] / total, 3) if total > 0 else 0,
            })
        return trend
```

**要点**：
- `_hourly_stats` 使用 `defaultdict` 自动初始化每小时的计数桶，避免 KeyError
- `get_hit_latency_stats()` 返回 p50/p95/p99 分位数，帮助判断缓存响应是否稳定
- `get_hourly_hit_rate()` 按时间排序输出趋势数据，可直接用于 Grafana 或报表展示

### 练习 2: 语义缓存调优工具

**思路**：遍历一组已标注的等价查询对，在不同阈值下计算每对查询是否被判定为等价（命中），从而算出精确率和召回率，最后推荐 F1 最高的阈值。

**答案**：

```python
import numpy as np
from dataclasses import dataclass

@dataclass
class ThresholdEvaluation:
    threshold: float
    precision: float
    recall: float
    f1: float
    true_positives: int
    false_positives: int
    false_negatives: int

class SemanticCacheTuner:
    """语义缓存阈值调优工具"""

    def __init__(self, embedder):
        self.embedder = embedder

    async def evaluate_thresholds(
        self,
        equivalent_pairs: list[tuple[str, str]],
        non_equivalent_pairs: list[tuple[str, str]],
        thresholds: list[float] = None,
    ) -> dict:
        """评估不同阈值下的精确率和召回率"""
        if thresholds is None:
            thresholds = [0.80, 0.85, 0.88, 0.90, 0.92, 0.94, 0.96, 0.98]

        # 预计算所有查询的 embedding
        all_queries = set()
        for q1, q2 in equivalent_pairs + non_equivalent_pairs:
            all_queries.add(q1)
            all_queries.add(q2)

        embeddings = {}
        query_list = list(all_queries)
        vectors = await self._batch_embed(query_list)
        for q, v in zip(query_list, vectors):
            embeddings[q] = np.array(v)

        # 计算所有对的相似度
        pos_similarities = []
        for q1, q2 in equivalent_pairs:
            sim = self._cosine_similarity(embeddings[q1], embeddings[q2])
            pos_similarities.append(sim)

        neg_similarities = []
        for q1, q2 in non_equivalent_pairs:
            sim = self._cosine_similarity(embeddings[q1], embeddings[q2])
            neg_similarities.append(sim)

        results = []
        for threshold in thresholds:
            tp = sum(1 for s in pos_similarities if s >= threshold)
            fp = sum(1 for s in neg_similarities if s >= threshold)
            fn = sum(1 for s in pos_similarities if s < threshold)

            precision = tp / (tp + fp) if (tp + fp) > 0 else 0
            recall = tp / (tp + fn) if (tp + fn) > 0 else 0
            f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0

            results.append(ThresholdEvaluation(
                threshold=threshold,
                precision=round(precision, 3),
                recall=round(recall, 3),
                f1=round(f1, 3),
                true_positives=tp,
                false_positives=fp,
                false_negatives=fn,
            ))

        best = max(results, key=lambda r: r.f1)
        return {
            "evaluations": [vars(r) for r in results],
            "recommended_threshold": best.threshold,
            "best_f1": best.f1,
            "best_precision": best.precision,
            "best_recall": best.recall,
        }

    def _cosine_similarity(self, a: np.ndarray, b: np.ndarray) -> float:
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))

    async def _batch_embed(self, texts: list[str]) -> list[list[float]]:
        response = await self.embedder.embeddings.create(
            model="text-embedding-3-small", input=texts
        )
        return [item.embedding for item in response.data]
```

**要点**：
- 预计算所有 embedding 避免重复调用 API，大幅降低调优成本
- 同时考虑正例对（等价查询）和反例对（不等价查询），确保精确率和召回率都有意义
- 以 F1 分数作为综合评价指标推荐最优阈值，避免只追求单一指标导致偏科

### 练习 3: 多级缓存集成

**思路**：将 L1 内存缓存、L2 语义缓存、L3 Redis 缓存组合为 `MultiLevelCache`，在 RAG Pipeline 中先查缓存再走完整链路，同时实现缓存预热功能。

**答案**：

```python
import asyncio
import json

class CachedRAGPipeline:
    """带三级缓存的 RAG Pipeline"""

    def __init__(self, base_pipeline, multi_level_cache: MultiLevelCache):
        self.pipeline = base_pipeline
        self.cache = multi_level_cache
        self._stats = {
            "total_queries": 0,
            "cache_hits": 0,
            "l1_hits": 0,
            "l2_hits": 0,
            "l3_hits": 0,
        }

    async def query(self, question: str) -> dict:
        self._stats["total_queries"] += 1

        # 逐级查找缓存
        cached = await self.cache.get(question)
        if cached:
            self._stats["cache_hits"] += 1
            level = cached.get("_cache_level", "unknown")
            if level == "L1":
                self._stats["l1_hits"] += 1
            elif level == "L2":
                self._stats["l2_hits"] += 1
            elif level == "L3":
                self._stats["l3_hits"] += 1
            cached["from_cache"] = True
            return cached

        # 缓存未命中，执行完整 RAG
        result = await self.pipeline.query(question)
        result["from_cache"] = False

        # 异步写入所有缓存层
        asyncio.create_task(self.cache.set(question, result))
        return result

    async def warmup(self, queries: list[str]):
        """缓存预热：对高频查询预先执行 RAG 并缓存结果"""
        for query in queries:
            cached = await self.cache.get(query)
            if cached:
                continue
            result = await self.pipeline.query(query)
            await self.cache.set(query, result)
            print(f"预热完成: {query[:50]}...")

    def get_hit_rate_stats(self) -> dict:
        total = self._stats["total_queries"]
        if total == 0:
            return {"message": "暂无查询数据"}
        return {
            **self._stats,
            "overall_hit_rate": f"{self._stats['cache_hits'] / total:.1%}",
            "l1_rate": f"{self._stats['l1_hits'] / total:.1%}",
            "l2_rate": f"{self._stats['l2_hits'] / total:.1%}",
            "l3_rate": f"{self._stats['l3_hits'] / total:.1%}",
        }
```

**要点**：
- 缓存旁路模式（Cache-Aside）：先查缓存，未命中再走完整 Pipeline，写回缓存时不阻塞响应
- `warmup()` 方法对历史高频查询预执行 RAG 并缓存，避免冷启动时所有请求穿透到底层
- 分级统计命中率帮助判断各级缓存的效果，如果 L1 命中率低可考虑增大内存缓存容量
