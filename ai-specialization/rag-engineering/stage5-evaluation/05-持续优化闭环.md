## 场景引入

你的 RAG 系统上线半年了，初期效果很好，但最近用户满意度在持续下降。排查发现，知识库中 30% 的文档已经过时，有些新产品信息完全没有收录，而系统完全没有自动发现和修复这些问题的机制。RAG 系统不是"部署完就结束"的项目——用户反馈、知识库更新、模型迭代需要形成一个持续运转的闭环，让系统越用越好而不是越用越差。

## 3. 反馈收集器

```python
"""
用户反馈收集器
收集显式反馈 (评分、点赞) 和隐式反馈 (停留时间、复制、分享)
"""

import json
import time
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Optional, Any
from datetime import datetime
from pathlib import Path
from collections import Counter
import hashlib


@dataclass
class FeedbackEvent:
    """反馈事件"""
    feedback_id: str
    user_id: str
    session_id: str
    query: str
    answer: str
    contexts: List[str] = field(default_factory=list)
    explicit_feedback: Dict[str, Any] = field(default_factory=dict)
    implicit_feedback: Dict[str, Any] = field(default_factory=dict)
    timestamp: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)

    def __post_init__(self):
        if not self.feedback_id:
            content = f"{self.user_id}:{self.query}:{self.timestamp}"
            self.feedback_id = hashlib.md5(content.encode()).hexdigest()[:12]
        if not self.timestamp:
            self.timestamp = datetime.now().isoformat()


class FeedbackCollector:
    """反馈收集器"""

    def __init__(self, storage_path: str = "feedback_data"):
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.events: List[FeedbackEvent] = []
        self._load_existing()

    def _load_existing(self) -> None:
        """加载已有反馈数据"""
        path = self.storage_path / "feedback.jsonl"
        if path.exists():
            with open(path, "r", encoding="utf-8") as f:
                for line in f:
                    data = json.loads(line.strip())
                    self.events.append(FeedbackEvent(**data))

    def collect_explicit(
        self,
        user_id: str,
        session_id: str,
        query: str,
        answer: str,
        rating: Optional[int] = None,
        thumbs_up: Optional[bool] = None,
        thumbs_down: Optional[bool] = None,
        feedback_text: Optional[str] = None,
        dimensions: Optional[Dict[str, int]] = None,
        contexts: Optional[List[str]] = None,
    ) -> FeedbackEvent:
        """收集显式反馈"""
        event = FeedbackEvent(
            feedback_id="",
            user_id=user_id,
            session_id=session_id,
            query=query,
            answer=answer,
            contexts=contexts or [],
            explicit_feedback={
                "rating": rating,
                "thumbs_up": thumbs_up,
                "thumbs_down": thumbs_down,
                "feedback_text": feedback_text,
                "dimensions": dimensions or {},
            },
        )
        self.events.append(event)
        self._persist(event)
        return event

    def collect_implicit(
        self,
        user_id: str,
        session_id: str,
        query: str,
        answer: str,
        dwell_time_seconds: float = 0,
        copied: bool = False,
        shared: bool = False,
        scrolled: bool = False,
        clicked_references: bool = False,
        asked_followup: bool = False,
        contexts: Optional[List[str]] = None,
    ) -> FeedbackEvent:
        """收集隐式反馈"""
        # 推断满意度
        inferred_satisfaction = self._infer_satisfaction(
            dwell_time_seconds, copied, shared, asked_followup
        )

        event = FeedbackEvent(
            feedback_id="",
            user_id=user_id,
            session_id=session_id,
            query=query,
            answer=answer,
            contexts=contexts or [],
            implicit_feedback={
                "dwell_time_seconds": dwell_time_seconds,
                "copied": copied,
                "shared": shared,
                "scrolled": scrolled,
                "clicked_references": clicked_references,
                "asked_followup": asked_followup,
                "inferred_satisfaction": inferred_satisfaction,
            },
        )
        self.events.append(event)
        self._persist(event)
        return event

    def _infer_satisfaction(
        self,
        dwell_time: float,
        copied: bool,
        shared: bool,
        asked_followup: bool,
    ) -> float:
        """基于隐式行为推断满意度 (0-1)"""
        score = 0.5  # 基线

        # 停留时间: 太短说明没看，太长说明没看懂
        if 10 < dwell_time < 120:
            score += 0.1
        elif dwell_time < 5:
            score -= 0.2

        if copied:
            score += 0.15  # 复制说明有用
        if shared:
            score += 0.1   # 分享说明质量高
        if asked_followup:
            score += 0.05  # 追问说明有兴趣

        return max(0.0, min(1.0, score))

    def _persist(self, event: FeedbackEvent) -> None:
        """持久化单条事件"""
        path = self.storage_path / "feedback.jsonl"
        with open(path, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(event), ensure_ascii=False) + "\n")

    def get_negative_feedback(
        self,
        rating_threshold: int = 2,
        limit: int = 50,
    ) -> List[FeedbackEvent]:
        """获取负面反馈"""
        negative = []
        for event in self.events:
            rating = event.explicit_feedback.get("rating")
            thumbs_down = event.explicit_feedback.get("thumbs_down")
            if (rating is not None and rating <= rating_threshold) or thumbs_down:
                negative.append(event)
        return negative[:limit]

    def get_statistics(self) -> Dict[str, Any]:
        """获取反馈统计"""
        total = len(self.events)
        if total == 0:
            return {"total": 0}

        ratings = [
            e.explicit_feedback["rating"]
            for e in self.events
            if e.explicit_feedback.get("rating") is not None
        ]
        thumbs_up = sum(
            1 for e in self.events
            if e.explicit_feedback.get("thumbs_up")
        )
        thumbs_down = sum(
            1 for e in self.events
            if e.explicit_feedback.get("thumbs_down")
        )

        return {
            "total_feedback": total,
            "explicit_ratings": len(ratings),
            "avg_rating": sum(ratings) / len(ratings) if ratings else None,
            "thumbs_up": thumbs_up,
            "thumbs_down": thumbs_down,
            "satisfaction_rate": thumbs_up / max(thumbs_up + thumbs_down, 1),
        }


# ============================================================
# 使用示例
# ============================================================

if __name__ == "__main__":
    collector = FeedbackCollector("test_feedback")

    # 模拟收集反馈
    collector.collect_explicit(
        user_id="user_001",
        session_id="sess_001",
        query="什么是 RAG？",
        answer="RAG 是检索增强生成技术...",
        rating=4,
        thumbs_up=True,
    )

    collector.collect_implicit(
        user_id="user_002",
        session_id="sess_002",
        query="如何优化检索？",
        answer="可以通过混合检索...",
        dwell_time_seconds=45,
        copied=True,
    )

    stats = collector.get_statistics()
    print(f"反馈统计: {json.dumps(stats, indent=2, ensure_ascii=False)}")
```

## 4. 问题分析器

```python
"""
反馈问题分析器
自动分类问题类型，生成改进建议
"""

import json
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any
from collections import Counter, defaultdict
from enum import Enum


class IssueCategory(Enum):
    """问题类别"""
    HALLUCINATION = "hallucination"         # 幻觉: 回答包含不存在的信息
    IRRELEVANT = "irrelevant"               # 不相关: 答非所问
    INCOMPLETE = "incomplete"               # 不完整: 遗漏关键信息
    OUTDATED = "outdated"                   # 过时: 信息已过期
    WRONG_FORMAT = "wrong_format"           # 格式差: 回答格式不佳
    SLOW_RESPONSE = "slow_response"         # 响应慢: 耗时太长
    RETRIEVAL_MISS = "retrieval_miss"       # 检索失败: 没检索到相关文档
    KNOWLEDGE_GAP = "knowledge_gap"         # 知识缺失: 知识库中没有相关信息


@dataclass
class IssueReport:
    """问题报告"""
    issue_id: str
    category: str
    severity: str          # high, medium, low
    query: str
    answer: str
    description: str
    suggested_fix: str
    frequency: int = 1
    examples: List[str] = field(default_factory=list)


class IssueAnalyzer:
    """问题分析器"""

    def __init__(self, llm_client: Optional[Any] = None):
        self.llm_client = llm_client
        self.issue_patterns: Dict[str, List[Dict]] = defaultdict(list)

    def analyze_feedback_batch(
        self,
        feedback_events: List[Any],
    ) -> List[IssueReport]:
        """批量分析反馈，生成问题报告"""
        issues = []

        for event in feedback_events:
            # 判断是否为负面反馈
            is_negative = self._is_negative(event)
            if not is_negative:
                continue

            # 分类问题
            category, severity = self._classify_issue(event)

            # 生成修复建议
            fix = self._suggest_fix(category, event)

            report = IssueReport(
                issue_id=event.feedback_id,
                category=category.value,
                severity=severity,
                query=event.query,
                answer=event.answer,
                description=self._generate_description(category, event),
                suggested_fix=fix,
            )
            issues.append(report)
            self.issue_patterns[category.value].append({
                "query": event.query,
                "feedback_id": event.feedback_id,
            })

        return issues

    def _is_negative(self, event: Any) -> bool:
        """判断是否为负面反馈"""
        rating = event.explicit_feedback.get("rating")
        thumbs_down = event.explicit_feedback.get("thumbs_down")
        inferred = event.implicit_feedback.get("inferred_satisfaction", 0.5)

        if rating is not None and rating <= 2:
            return True
        if thumbs_down:
            return True
        if inferred < 0.3:
            return True
        return False

    def _classify_issue(self, event: Any) -> tuple:
        """对问题进行分类"""
        query = event.query.lower()
        answer = event.answer.lower()
        contexts = event.contexts
        feedback_text = event.explicit_feedback.get("feedback_text", "")

        # 基于规则的简单分类
        if not contexts or all(len(c.strip()) < 10 for c in contexts):
            return IssueCategory.RETRIEVAL_MISS, "high"

        if "不知道" in answer or "无法回答" in answer:
            return IssueCategory.KNOWLEDGE_GAP, "medium"

        # 检查是否答非所问
        query_words = set(query.split())
        answer_words = set(answer.split())
        overlap = len(query_words & answer_words) / max(len(query_words), 1)
        if overlap < 0.1:
            return IssueCategory.IRRELEVANT, "high"

        # 检查回答长度
        if len(answer) < 50:
            return IssueCategory.INCOMPLETE, "medium"

        # 检查是否包含幻觉标记
        if "可能" in answer or "据说" in answer or "也许" in answer:
            return IssueCategory.HALLUCINATION, "medium"

        # 默认
        return IssueCategory.INCOMPLETE, "low"

    def _suggest_fix(self, category: IssueCategory, event: Any) -> str:
        """根据问题类型生成修复建议"""
        fixes = {
            IssueCategory.HALLUCINATION: "优化 Prompt，要求模型只基于上下文回答；增加 Faithfulness 检查",
            IssueCategory.IRRELEVANT: "检查检索质量，优化查询改写；调整 Prompt 明确回答范围",
            IssueCategory.INCOMPLETE: "优化分块策略确保信息完整；增加检索数量 (top_k)",
            IssueCategory.OUTDATED: "更新知识库中的过时文档；建立定期更新机制",
            IssueCategory.WRONG_FORMAT: "在 Prompt 中明确格式要求；添加输出格式化步骤",
            IssueCategory.SLOW_RESPONSE: "优化检索性能；考虑使用更快的模型；添加缓存",
            IssueCategory.RETRIEVAL_MISS: "优化嵌入模型；添加关键词检索作为补充；扩展知识库",
            IssueCategory.KNOWLEDGE_GAP: "补充缺失知识到知识库；添加兜底回答策略",
        }
        return fixes.get(category, "需要进一步分析")

    def _generate_description(self, category: IssueCategory, event: Any) -> str:
        """生成问题描述"""
        return f"用户查询 '{event.query[:50]...' 时，系统回答被标记为 {category.value}"

    def get_top_issues(self, n: int = 10) -> List[Dict[str, Any]]:
        """获取最高频问题"""
        issue_counts = []
        for category, occurrences in self.issue_patterns.items():
            issue_counts.append({
                "category": category,
                "count": len(occurrences),
                "sample_queries": [o["query"] for o in occurrences[:3]],
            })
        return sorted(issue_counts, key=lambda x: x["count"], reverse=True)[:n]

    def generate_improvement_plan(self) -> str:
        """生成改进计划"""
        top_issues = self.get_top_issues(5)
        lines = []
        lines.append("=" * 60)
        lines.append("  RAG 系统改进计划")
        lines.append("=" * 60)

        for i, issue in enumerate(top_issues, 1):
            lines.append(f"\n  {i}. {issue['category']} (出现 {issue['count']} 次)")
            lines.append(f"     示例查询: {issue['sample_queries'][0] if issue['sample_queries'] else 'N/A'}")

        return "\n".join(lines)
```

## 5. 知识库更新器

```python
"""
知识库更新管理器
支持增量更新、版本控制和回滚
"""

import json
import hashlib
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Optional, Any
from datetime import datetime
from pathlib import Path
from enum import Enum


class UpdateAction(Enum):
    """更新动作"""
    ADD = "add"
    UPDATE = "update"
    DELETE = "delete"
    REINDEX = "reindex"


@dataclass
class UpdateRecord:
    """更新记录"""
    update_id: str
    action: str
    doc_id: str
    doc_content: str
    reason: str
    timestamp: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)
    status: str = "pending"    # pending, applied, rolled_back

    def __post_init__(self):
        if not self.timestamp:
            self.timestamp = datetime.now().isoformat()
        if not self.update_id:
            content = f"{self.action}:{self.doc_id}:{self.timestamp}"
            self.update_id = hashlib.md5(content.encode()).hexdigest()[:12]


class KnowledgeBaseUpdater:
    """知识库更新管理器"""

    def __init__(
        self,
        vector_store: Any,
        storage_path: str = "kb_updates",
    ):
        self.vector_store = vector_store
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.update_history: List[UpdateRecord] = []
        self._load_history()

    def _load_history(self) -> None:
        """加载更新历史"""
        path = self.storage_path / "update_history.jsonl"
        if path.exists():
            with open(path, "r", encoding="utf-8") as f:
                for line in f:
                    data = json.loads(line.strip())
                    self.update_history.append(UpdateRecord(**data))

    def add_document(
        self,
        doc_id: str,
        content: str,
        metadata: Optional[Dict] = None,
        reason: str = "new_document",
    ) -> UpdateRecord:
        """添加新文档"""
        record = UpdateRecord(
            update_id="",
            action=UpdateAction.ADD.value,
            doc_id=doc_id,
            doc_content=content,
            reason=reason,
            metadata=metadata or {},
        )

        # 执行更新
        try:
            self.vector_store.add_document(doc_id, content, metadata)
            record.status = "applied"
        except Exception as e:
            record.status = "failed"
            record.metadata["error"] = str(e)

        self.update_history.append(record)
        self._persist(record)
        return record

    def update_document(
        self,
        doc_id: str,
        new_content: str,
        reason: str = "content_update",
    ) -> UpdateRecord:
        """更新已有文档"""
        record = UpdateRecord(
            update_id="",
            action=UpdateAction.UPDATE.value,
            doc_id=doc_id,
            doc_content=new_content,
            reason=reason,
        )

        try:
            self.vector_store.update_document(doc_id, new_content)
            record.status = "applied"
        except Exception as e:
            record.status = "failed"
            record.metadata["error"] = str(e)

        self.update_history.append(record)
        self._persist(record)
        return record

    def delete_document(
        self,
        doc_id: str,
        reason: str = "content_removed",
    ) -> UpdateRecord:
        """删除文档"""
        record = UpdateRecord(
            update_id="",
            action=UpdateAction.DELETE.value,
            doc_id=doc_id,
            doc_content="",
            reason=reason,
        )

        try:
            self.vector_store.delete_document(doc_id)
            record.status = "applied"
        except Exception as e:
            record.status = "failed"
            record.metadata["error"] = str(e)

        self.update_history.append(record)
        self._persist(record)
        return record

    def batch_update(
        self,
        updates: List[Dict[str, Any]],
    ) -> List[UpdateRecord]:
        """批量更新"""
        records = []
        for update in updates:
            action = update.get("action", "add")
            if action == "add":
                record = self.add_document(
                    update["doc_id"],
                    update["content"],
                    update.get("metadata"),
                    update.get("reason", "batch_add"),
                )
            elif action == "update":
                record = self.update_document(
                    update["doc_id"],
                    update["content"],
                    update.get("reason", "batch_update"),
                )
            elif action == "delete":
                record = self.delete_document(
                    update["doc_id"],
                    update.get("reason", "batch_delete"),
                )
            else:
                continue
            records.append(record)
        return records

    def rollback(self, update_id: str) -> bool:
        """回滚指定更新"""
        for record in self.update_history:
            if record.update_id == update_id and record.status == "applied":
                if record.action == UpdateAction.ADD.value:
                    self.vector_store.delete_document(record.doc_id)
                    record.status = "rolled_back"
                    return True
                elif record.action == UpdateAction.DELETE.value:
                    self.vector_store.add_document(
                        record.doc_id,
                        record.doc_content,
                        record.metadata,
                    )
                    record.status = "rolled_back"
                    return True
        return False

    def _persist(self, record: UpdateRecord) -> None:
        """持久化更新记录"""
        path = self.storage_path / "update_history.jsonl"
        with open(path, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(record), ensure_ascii=False) + "\n")

    def get_statistics(self) -> Dict[str, Any]:

        total = len(self.update_history)
        by_action = Counter(r.action for r in self.update_history)
        by_status = Counter(r.status for r in self.update_history)

        return {
            "total_updates": total,
            "by_action": dict(by_action),
            "by_status": dict(by_status),
        }
```

## 6. Pipeline 编排器

```python
"""
持续优化 Pipeline 编排器
将反馈收集、问题分析、知识库更新串联为自动化流程
"""

import json
import time
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any, Callable
from datetime import datetime
from pathlib import Path
from enum import Enum


class PipelineStage(Enum):
    """Pipeline 阶段"""
    COLLECT = "collect"
    ANALYZE = "analyze"
    PLAN = "plan"
    EXECUTE = "execute"
    VERIFY = "verify"


@dataclass
class PipelineRun:
    """Pipeline 运行记录"""
    run_id: str
    started_at: str
    completed_at: str = ""
    status: str = "running"
    stages_completed: List[str] = field(default_factory=list)
    metrics: Dict[str, Any] = field(default_factory=dict)
    errors: List[str] = field(default_factory=list)


class OptimizationPipeline:
    """持续优化 Pipeline"""

    def __init__(
        self,
        feedback_collector: Any,
        issue_analyzer: Any,
        kb_updater: Any,
        eval_pipeline: Optional[Any] = None,
        config: Optional[Dict] = None,
    ):
        self.feedback_collector = feedback_collector
        self.issue_analyzer = issue_analyzer
        self.kb_updater = kb_updater
        self.eval_pipeline = eval_pipeline
        self.config = config or {
            "min_negative_feedback": 10,
            "auto_update_enabled": False,
            "verification_required": True,
        }
        self.run_history: List[PipelineRun] = []

    def run(self) -> PipelineRun:
        """执行完整优化 Pipeline"""
        run = PipelineRun(
            run_id=f"opt_{int(time.time())}",
            started_at=datetime.now().isoformat(),
        )

        try:
            # Stage 1: 收集反馈
            print("[Pipeline] Stage 1/5: 收集反馈...")
            feedback_stats = self._stage_collect()
            run.stages_completed.append("collect")
            run.metrics["feedback"] = feedback_stats

            # Stage 2: 分析问题
            print("[Pipeline] Stage 2/5: 分析问题...")
            issues = self._stage_analyze()
            run.stages_completed.append("analyze")
            run.metrics["issues_found"] = len(issues)

            # Stage 3: 生成改进计划
            print("[Pipeline] Stage 3/5: 生成改进计划...")
            plan = self._stage_plan(issues)
            run.stages_completed.append("plan")
            run.metrics["plan"] = plan

            # Stage 4: 执行改进
            print("[Pipeline] Stage 4/5: 执行改进...")
            if self.config.get("auto_update_enabled"):
                updates = self._stage_execute(plan)
                run.stages_completed.append("execute")
                run.metrics["updates_applied"] = len(updates)
            else:
                run.metrics["updates_applied"] = 0
                run.metrics["note"] = "自动更新已禁用，需手动审批"

            # Stage 5: 验证效果
            if self.config.get("verification_required") and self.eval_pipeline:
                print("[Pipeline] Stage 5/5: 验证改进效果...")
                verification = self._stage_verify()
                run.stages_completed.append("verify")
                run.metrics["verification"] = verification

            run.status = "completed"
            run.completed_at = datetime.now().isoformat()

        except Exception as e:
            run.status = "failed"
            run.errors.append(str(e))

        self.run_history.append(run)
        return run

    def _stage_collect(self) -> Dict[str, Any]:
        """收集阶段"""
        stats = self.feedback_collector.get_statistics()
        negative = self.feedback_collector.get_negative_feedback()
        return {
            "total_feedback": stats.get("total_feedback", 0),
            "negative_count": len(negative),
            "satisfaction_rate": stats.get("satisfaction_rate", 0),
        }

    def _stage_analyze(self) -> List[Any]:
        """分析阶段"""
        negative_feedback = self.feedback_collector.get_negative_feedback()
        issues = self.issue_analyzer.analyze_feedback_batch(negative_feedback)
        return issues

    def _stage_plan(self, issues: List[Any]) -> Dict[str, Any]:
        """计划阶段"""
        plan = {
            "total_issues": len(issues),
            "by_category": {},
            "actions": [],
        }

        for issue in issues:
            cat = issue.category
            if cat not in plan["by_category"]:
                plan["by_category"][cat] = 0
            plan["by_category"][cat] += 1

            plan["actions"].append({
                "category": cat,
                "severity": issue.severity,
                "fix": issue.suggested_fix,
                "query": issue.query,
            })

        return plan

    def _stage_execute(self, plan: Dict[str, Any]) -> List[Any]:
        """执行阶段"""
        updates = []
        for action in plan.get("actions", []):
            if action["category"] in ("retrieval_miss", "knowledge_gap"):
                # 需要补充知识库
                updates.append({
                    "action": "add",
                    "doc_id": f"auto_{action['category']}_{int(time.time())}",
                    "content": action.get("fix", ""),
                    "reason": f"自动补充: {action['category']}",
                })

        if updates:
            self.kb_updater.batch_update(updates)

        return updates

    def _stage_verify(self) -> Dict[str, Any]:
        """验证阶段"""
        if not self.eval_pipeline:
            return {"status": "no_eval_pipeline"}

        # 运行评估
        # report = self.eval_pipeline.run(...)
        return {"status": "verification_skipped", "reason": "需要集成评估 Pipeline"}

    def get_latest_run(self) -> Optional[PipelineRun]:
        """获取最近一次运行"""
        return self.run_history[-1] if self.run_history else None

# 使用示例

if __name__ == "__main__":
    print("Pipeline 编排器示例 (需要集成各组件)")
```

## 7. 手动 vs 半自动 vs 全自动优化对比

| 维度 | 手动优化 | 半自动优化 | 全自动优化 |
|------|---------|-----------|-----------|
| **触发方式** | 人工定期检查 | 自动检测 + 人工审批 | 全自动触发 |
| **问题发现** | 人工阅读反馈 | 自动分类 + 人工确认 | 自动分类和优先级排序 |
| **改进执行** | 手动修改代码/配置 | 自动生成方案 + 人工执行 | 自动执行改进 |
| **验证方式** | 人工验证 | 自动评估 + 人工确认 | 自动评估和回滚 |
| **适用阶段** | 早期探索期 | 成长期 | 成熟期 |
| **风险** | 低 | 中 | 需要完善的安全机制 |
| **效率** | 低 | 中 | 高 |
| **人力成本** | 高 | 中 | 低 |

## 9. 工程建议

1. **反馈收集不要增加用户负担**：在回答下方加一个"有帮助/没帮助"的拇指按钮就够了。不要弹问卷调查，用户会直接忽略甚至产生负面体验。
2. **知识库更新要自动化但审核要人工**：自动检测过时文档、自动生成更新建议是好的，但最终的入库决策应该由人工审核。完全自动化的知识库更新可能引入错误信息。
3. **优化闭环要有优先级排序**：不是所有反馈都需要立即处理。按影响范围（多少用户受影响）和严重程度（错误答案 vs 不够好）排序，优先处理高影响高严重度的问题。
4. **监控闭环本身的健康度**：反馈收集率、问题分析准确率、更新生效时间——这些是优化闭环的"元指标"。如果反馈收集率低于 5%，说明反馈入口不够明显或用户没有动力参与。

## 参考答案

### 练习 1: 反馈收集器 (基础)

**思路**：反馈收集器需要同时支持显式反馈（用户主动评分、点赞/踩）和隐式反馈（停留时间、复制行为等自动采集的信号）。隐式反馈可以推断满意度，与显式反馈互补。数据持久化用 JSONL 格式，便于追加写入和流式处理。

**答案**：
```python
import json
import hashlib
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Optional, Any
from datetime import datetime
from pathlib import Path


@dataclass
class FeedbackEvent:
    feedback_id: str
    user_id: str
    session_id: str
    query: str
    answer: str
    contexts: List[str] = field(default_factory=list)
    explicit_feedback: Dict[str, Any] = field(default_factory=dict)
    implicit_feedback: Dict[str, Any] = field(default_factory=dict)
    timestamp: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)

    def __post_init__(self):
        if not self.feedback_id:
            content = f"{self.user_id}:{self.query}:{self.timestamp}"
            self.feedback_id = hashlib.md5(content.encode()).hexdigest()[:12]
        if not self.timestamp:
            self.timestamp = datetime.now().isoformat()


class FeedbackCollector:
    def __init__(self, storage_path="feedback_data"):
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.events: List[FeedbackEvent] = []
        self._load_existing()

    def _load_existing(self):
        path = self.storage_path / "feedback.jsonl"
        if path.exists():
            with open(path, "r", encoding="utf-8") as f:
                for line in f:
                    data = json.loads(line.strip())
                    self.events.append(FeedbackEvent(**data))

    def collect_explicit(self, user_id, session_id, query, answer,
                         rating=None, thumbs_up=None, thumbs_down=None,
                         feedback_text=None, contexts=None):
        event = FeedbackEvent(
            feedback_id="",
            user_id=user_id,
            session_id=session_id,
            query=query,
            answer=answer,
            contexts=contexts or [],
            explicit_feedback={
                "rating": rating,
                "thumbs_up": thumbs_up,
                "thumbs_down": thumbs_down,
                "feedback_text": feedback_text,
            },
        )
        self.events.append(event)
        self._persist(event)
        return event

    def collect_implicit(self, user_id, session_id, query, answer,
                         dwell_time_seconds=0, copied=False, shared=False,
                         contexts=None):
        inferred = self._infer_satisfaction(dwell_time_seconds, copied, shared)
        event = FeedbackEvent(
            feedback_id="",
            user_id=user_id,
            session_id=session_id,
            query=query,
            answer=answer,
            contexts=contexts or [],
            implicit_feedback={
                "dwell_time_seconds": dwell_time_seconds,
                "copied": copied,
                "shared": shared,
                "inferred_satisfaction": inferred,
            },
        )
        self.events.append(event)
        self._persist(event)
        return event

    def _infer_satisfaction(self, dwell_time, copied, shared):
        score = 0.5
        if 10 < dwell_time < 120:
            score += 0.1
        elif dwell_time < 5:
            score -= 0.2
        if copied:
            score += 0.15
        if shared:
            score += 0.1
        return max(0.0, min(1.0, score))

    def _persist(self, event):
        path = self.storage_path / "feedback.jsonl"
        with open(path, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(event), ensure_ascii=False) + "\n")

    def get_negative_feedback(self, rating_threshold=2, limit=50):
        negative = []
        for e in self.events:
            rating = e.explicit_feedback.get("rating")
            thumbs_down = e.explicit_feedback.get("thumbs_down")
            if (rating is not None and rating <= rating_threshold) or thumbs_down:
                negative.append(e)
        return negative[:limit]

    def get_statistics(self):
        total = len(self.events)
        if total == 0:
            return {"total": 0}
        ratings = [e.explicit_feedback["rating"] for e in self.events
                   if e.explicit_feedback.get("rating") is not None]
        thumbs_up = sum(1 for e in self.events if e.explicit_feedback.get("thumbs_up"))
        thumbs_down = sum(1 for e in self.events if e.explicit_feedback.get("thumbs_down"))
        return {
            "total_feedback": total,
            "avg_rating": sum(ratings) / len(ratings) if ratings else None,
            "thumbs_up": thumbs_up,
            "thumbs_down": thumbs_down,
            "satisfaction_rate": thumbs_up / max(thumbs_up + thumbs_down, 1),
        }


# 测试

collector.collect_implicit("u3", "s3", "什么是 Embedding？", "Embedding 是...", dwell_time_seconds=45, copied=True)

print(f"负面反馈: {len(negatives)} 条")
for e in negatives:
    print(f"  {e.query}: rating={e.explicit_feedback.get('rating')}")
```

**要点**：
- 显式反馈（评分、点赞/踩）和隐式反馈（停留时间、复制行为）分开收集，互不干扰
- 隐式反馈推断满意度的规则：停留 10-120 秒加 0.1，复制加 0.15，分享加 0.1，低于 5 秒减 0.2
- JSONL 格式便于追加写入，每条事件一行，不会因为并发写入损坏文件

### 练习 2: 问题分析器 (进阶)

**思路**：问题分析器的核心是对负面反馈自动分类（幻觉、不相关、不完整、过时等），然后统计各类问题的频率，最后根据频率和严重度生成改进计划。分类可以先用规则（关键词匹配）做初版，后续接入 LLM 做更精确的分类。

**答案**：
```python
import json
from dataclasses import dataclass, field
from typing import Dict, List, Any
from collections import Counter, defaultdict
from enum import Enum


class IssueCategory(Enum):
    HALLUCINATION = "hallucination"
    IRRELEVANT = "irrelevant"
    INCOMPLETE = "incomplete"
    OUTDATED = "outdated"
    WRONG_FORMAT = "wrong_format"
    SLOW_RESPONSE = "slow_response"
    RETRIEVAL_MISS = "retrieval_miss"
    KNOWLEDGE_GAP = "knowledge_gap"


@dataclass
class IssueReport:
    issue_id: str
    category: str
    severity: str
    query: str
    answer: str
    description: str
    suggested_fix: str
    frequency: int = 1


class IssueAnalyzer:
    def __init__(self):
        self.issue_patterns: Dict[str, List[Dict]] = defaultdict(list)

    def analyze_feedback_batch(self, feedback_events):
        issues = []
        for event in feedback_events:
            if not self._is_negative(event):
                continue
            category, severity = self._classify_issue(event)
            fix = self._suggest_fix(category)
            report = IssueReport(
                issue_id=event.get("feedback_id", ""),
                category=category.value,
                severity=severity,
                query=event.get("query", ""),
                answer=event.get("answer", ""),
                description=f"查询 '{event.get('query', '')[:30]}...' 被标记为 {category.value}",
                suggested_fix=fix,
            )
            issues.append(report)
            self.issue_patterns[category.value].append({"query": event.get("query", "")})
        return issues

    def _is_negative(self, event):
        rating = event.get("explicit_feedback", {}).get("rating")
        thumbs_down = event.get("explicit_feedback", {}).get("thumbs_down")
        if rating is not None and rating <= 2:
            return True
        if thumbs_down:
            return True
        return False

    def _classify_issue(self, event):
        answer = event.get("answer", "").lower()
        contexts = event.get("contexts", [])
        query = event.get("query", "").lower()

        if not contexts or all(len(c.strip()) < 10 for c in contexts):
            return IssueCategory.RETRIEVAL_MISS, "high"

        if "不知道" in answer or "无法回答" in answer:
            return IssueCategory.KNOWLEDGE_GAP, "medium"

        query_words = set(query.split())
        answer_words = set(answer.split())
        overlap = len(query_words & answer_words) / max(len(query_words), 1)
        if overlap < 0.1:
            return IssueCategory.IRRELEVANT, "high"

        if len(answer) < 50:
            return IssueCategory.INCOMPLETE, "medium"

        if "可能" in answer or "据说" in answer:
            return IssueCategory.HALLUCINATION, "medium"

        return IssueCategory.INCOMPLETE, "low"

    def _suggest_fix(self, category):
        fixes = {
            IssueCategory.HALLUCINATION: "优化 Prompt，要求模型只基于上下文回答；增加 Faithfulness 检查",
            IssueCategory.IRRELEVANT: "检查检索质量，优化查询改写；调整 Prompt 明确回答范围",
            IssueCategory.INCOMPLETE: "优化分块策略确保信息完整；增加检索数量 (top_k)",
            IssueCategory.OUTDATED: "更新知识库中的过时文档；建立定期更新机制",
            IssueCategory.RETRIEVAL_MISS: "优化嵌入模型；添加关键词检索作为补充",
            IssueCategory.KNOWLEDGE_GAP: "补充缺失知识到知识库；添加兜底回答策略",
        }
        return fixes.get(category, "需要进一步分析")

    def get_top_issues(self, n=10):
        result = []
        for cat, occurrences in self.issue_patterns.items():
            result.append({"category": cat, "count": len(occurrences),
                           "sample_queries": [o["query"] for o in occurrences[:3]]})
        return sorted(result, key=lambda x: x["count"], reverse=True)[:n]

    def generate_improvement_plan(self):
        top = self.get_top_issues(5)
        lines = ["=" * 50, "  RAG 系统改进计划", "=" * 50]
        for i, issue in enumerate(top, 1):
            lines.append(f"\n  {i}. {issue['category']} (出现 {issue['count']} 次)")
            lines.append(f"     建议: {self._suggest_fix(IssueCategory(issue['category']))}")
        return "\n".join(lines)
```

### 练习 3: 完整优化 Pipeline (综合)

**思路**：将反馈收集、问题分析、知识库更新三个组件串联为 5 阶段 Pipeline（收集 → 分析 → 计划 → 执行 → 验证）。关键设计点：自动更新默认关闭需人工审批，每次运行记录完整状态，支持手动和自动两种模式。

**答案**：
```python
import json
import time
from dataclasses import dataclass, field
from typing import Dict, List, Any, Optional
from datetime import datetime
from pathlib import Path


@dataclass
class PipelineRun:
    run_id: str
    started_at: str
    completed_at: str = ""
    status: str = "running"
    stages_completed: List[str] = field(default_factory=list)
    metrics: Dict[str, Any] = field(default_factory=dict)
    errors: List[str] = field(default_factory=list)


class MockFeedbackCollector:
    def __init__(self):
        self.events = [
            {"feedback_id": "f1", "query": "什么是 RAG？", "answer": "不知道", "contexts": [],
             "explicit_feedback": {"rating": 1}},
            {"feedback_id": "f2", "query": "如何优化？", "answer": "Python", "contexts": ["优化..."],
             "explicit_feedback": {"rating": 2, "thumbs_down": True}},
            {"feedback_id": "f3", "query": "向量数据库", "answer": "好的回答", "contexts": ["..."],
             "explicit_feedback": {"rating": 5, "thumbs_up": True}},
        ]

    def get_statistics(self):
        return {"total_feedback": len(self.events), "satisfaction_rate": 0.33}

    def get_negative_feedback(self):
        return [e for e in self.events if e["explicit_feedback"].get("rating", 5) <= 2
                or e["explicit_feedback"].get("thumbs_down")]


class MockIssueAnalyzer:
    def analyze_feedback_batch(self, events):
        issues = []
        for e in events:
            issues.append(type("Issue", (), {
                "category": "knowledge_gap" if "不知道" in e.get("answer", "") else "irrelevant",
                "severity": "high",
                "query": e["query"],
                "suggested_fix": "补充知识库" if "不知道" in e.get("answer", "") else "优化检索",
            })())
        return issues


class MockKBUpdater:
    def batch_update(self, updates):
        print(f"    [KB] 执行 {len(updates)} 条更新")
        return updates


class OptimizationPipeline:
    def __init__(self, feedback_collector, issue_analyzer, kb_updater,
                 eval_pipeline=None, config=None):
        self.feedback_collector = feedback_collector
        self.issue_analyzer = issue_analyzer
        self.kb_updater = kb_updater
        self.eval_pipeline = eval_pipeline
        self.config = config or {
            "auto_update_enabled": False,
            "verification_required": True,
        }
        self.run_history: List[PipelineRun] = []

    def run(self):
        run = PipelineRun(
            run_id=f"opt_{int(time.time())}",
            started_at=datetime.now().isoformat(),
        )

        try:
            # Stage 1: 收集反馈
            print("[Stage 1/5] 收集反馈...")
            stats = self.feedback_collector.get_statistics()
            negative = self.feedback_collector.get_negative_feedback()
            run.stages_completed.append("collect")
            run.metrics["feedback"] = {"total": stats["total_feedback"], "negative": len(negative)}
            print(f"  总反馈: {stats['total_feedback']}, 负面: {len(negative)}")

            # Stage 2: 分析问题
            print("[Stage 2/5] 分析问题...")
            issues = self.issue_analyzer.analyze_feedback_batch(negative)
            run.stages_completed.append("analyze")
            run.metrics["issues_found"] = len(issues)
            for issue in issues:
                print(f"  [{issue.severity}] {issue.category}: {issue.query}")

            # Stage 3: 生成改进计划
            print("[Stage 3/5] 生成改进计划...")
            plan = {"actions": []}
            for issue in issues:
                plan["actions"].append({
                    "category": issue.category,
                    "fix": issue.suggested_fix,
                    "query": issue.query,
                })
            run.stages_completed.append("plan")
            run.metrics["plan_actions"] = len(plan["actions"])
            print(f"  计划动作: {len(plan['actions'])} 条")

            # Stage 4: 执行改进
            print("[Stage 4/5] 执行改进...")
            if self.config.get("auto_update_enabled"):
                updates = []
                for action in plan["actions"]:
                    updates.append({
                        "action": "add",
                        "doc_id": f"auto_{action['category']}_{int(time.time())}",
                        "content": action["fix"],
                        "reason": f"自动补充: {action['category']}",
                    })
                self.kb_updater.batch_update(updates)
                run.stages_completed.append("execute")
                run.metrics["updates_applied"] = len(updates)
            else:
                print("  自动更新已禁用，需人工审批")
                run.metrics["updates_applied"] = 0
                run.metrics["note"] = "需人工审批"

            # Stage 5: 验证效果
            print("[Stage 5/5] 验证改进效果...")
            run.stages_completed.append("verify")
            run.metrics["verification"] = "跳过（需要集成评估 Pipeline）"

            run.status = "completed"
            run.completed_at = datetime.now().isoformat()

        except Exception as e:
            run.status = "failed"
            run.errors.append(str(e))

        self.run_history.append(run)
        return run


# 运行
pipeline = OptimizationPipeline(
    feedback_collector=MockFeedbackCollector(),
    issue_analyzer=MockIssueAnalyzer(),
    kb_updater=MockKBUpdater(),
)

print("=" * 50)
print("持续优化 Pipeline 运行")
print("=" * 50)
result = pipeline.run()

print("\n" + "=" * 50)
print(f"运行结果: {result.status}")
print(f"完成阶段: {result.stages_completed}")
print(f"指标: {json.dumps(result.metrics, indent=2, ensure_ascii=False)}")
print("=" * 50)
```

**要点**：
- Pipeline 分为 5 个阶段，每个阶段独立、可单独重跑，阶段间通过 metrics 传递上下文
- 自动更新默认关闭（`auto_update_enabled: False`），关键入库操作需要人工审批后才执行
- 每次运行记录完整的 PipelineRun 对象（状态、完成阶段、指标、错误），支持历史追溯和趋势分析
