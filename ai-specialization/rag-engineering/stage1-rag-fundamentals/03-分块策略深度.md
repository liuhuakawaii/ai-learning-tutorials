## 3. 固定大小分块

最简单的分块方式，按固定字符数切分。

```python
"""
固定大小分块实现
简单但可能切断语义
"""

def fixed_size_chunking(
    text: str,
    chunk_size: int = 500,
    chunk_overlap: int = 50
) -> list[str]:
    """
    固定大小分块
    
    Args:
        text: 原始文本
        chunk_size: 每个块的最大字符数
        chunk_overlap: 块之间的重叠字符数
    
    Returns:
        分块列表
    """
    chunks = []
    start = 0
    
    while start < len(text):
        end = start + chunk_size
        
        # 如果不是最后一块，尝试在空格处断开
        if end < len(text):
            # 在 chunk_size 范围内找最后一个空格
            last_space = text.rfind(" ", start, end)
            if last_space > start:
                end = last_space
        
        chunk = text[start:end].strip()
        if chunk:
            chunks.append(chunk)
        
        # 移动到下一个位置（考虑重叠）
        start = end - chunk_overlap if end < len(text) else len(text)
    
    return chunks


# 使用示例
text = "这是一段很长的文本..." * 100
chunks = fixed_size_chunking(text, chunk_size=200, chunk_overlap=20)
print(f"生成了 {len(chunks)} 个块")
for i, chunk in enumerate(chunks[:3]):
    print(f"块 {i+1}: {chunk[:50]}...")
```

**优点：** 实现简单，速度快
**缺点：** 可能切断句子，破坏语义完整性

## 4. 递归字符分块

LangChain 推荐的默认分块策略，按分隔符层级递归切分。

```python
"""
递归字符分块实现
按分隔符层级递归切分，尽量保持语义完整
"""

from langchain.text_splitter import RecursiveCharacterTextSplitter

def recursive_character_chunking(
    text: str,
    chunk_size: int = 500,
    chunk_overlap: int = 50,
    separators: list[str] | None = None
) -> list[str]:
    """
    使用 LangChain 的递归字符分块器
    
    分隔符优先级（从高到低）：
    1. "\n\n" - 段落分隔
    2. "\n"   - 换行
    3. "。"   - 中文句号
    4. "，"   - 中文逗号
    5. " "    - 空格
    6. ""     - 字符级别
    """
    if separators is None:
        separators = ["\n\n", "\n", "。", "，", "；", "：", " ", ""]
    
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=chunk_size,
        chunk_overlap=chunk_overlap,
        separators=separators,
        length_function=len,
        is_separator_regex=False
    )
    
    return splitter.split_text(text)


def recursive_chunk_documents(
    documents: list[dict],
    chunk_size: int = 500,
    chunk_overlap: int = 50
) -> list[dict]:
    """
    对文档列表进行递归分块，保留元数据
    
    Args:
        documents: 文档列表，每个文档包含 "text" 和 "metadata"
        chunk_size: 块大小
        chunk_overlap: 重叠大小
    
    Returns:
        分块后的文档列表
    """
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=chunk_size,
        chunk_overlap=chunk_overlap,
        separators=["\n\n", "\n", "。", "，", " ", ""]
    )
    
    chunks = []
    for doc in documents:
        text_chunks = splitter.split_text(doc["text"])
        
        for i, chunk in enumerate(text_chunks):
            chunks.append({
                "text": chunk,
                "metadata": {
                    **doc.get("metadata", {}),
                    "chunk_index": i,
                    "total_chunks": len(text_chunks)
                }
            })
    
    return chunks


# 使用示例
documents = [
    {
        "text": "人工智能（Artificial Intelligence），英文缩写为AI...",
        "metadata": {"source": "ai_introduction.pdf", "page": 1}
    }
]

chunks = recursive_chunk_documents(documents, chunk_size=300, chunk_overlap=30)
print(f"生成了 {len(chunks)} 个块")
for chunk in chunks:
    print(f"  来源: {chunk['metadata']['source']}, 内容: {chunk['text'][:30]}...")
```

## 5. 语义分块

基于 Embedding 相似度进行智能分块，在语义边界处切分。

```python
"""
语义分块实现
基于 Embedding 相似度在语义边界处切分
"""

import numpy as np
from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity

class SemanticChunker:
    """
    语义分块器
    
    原理：
    1. 将文本按句子切分
    2. 计算相邻句子的 Embedding 相似度
    3. 在相似度低的位置（语义转折点）切分
    """
    
    def __init__(
        self,
        model_name: str = "BAAI/bge-small-zh-v1.5",
        similarity_threshold: float = 0.5,
        min_chunk_size: int = 100,
        max_chunk_size: int = 1000
    ):
        self.model = SentenceTransformer(model_name)
        self.similarity_threshold = similarity_threshold
        self.min_chunk_size = min_chunk_size
        self.max_chunk_size = max_chunk_size
    
    def _split_sentences(self, text: str) -> list[str]:
        """按句子切分文本"""
        import re
        # 中英文句子切分
        sentences = re.split(r'([。！？.!?\n])', text)
        
        # 合并句子和标点
        merged = []
        for i in range(0, len(sentences) - 1, 2):
            merged.append(sentences[i] + sentences[i + 1])
        
        # 处理最后一个没有标点的部分
        if len(sentences) % 2 == 1 and sentences[-1]:
            merged.append(sentences[-1])
        
        return [s.strip() for s in merged if s.strip()]
    
    def _calculate_similarities(self, sentences: list[str]) -> list[float]:
        """计算相邻句子的相似度"""
        embeddings = self.model.encode(sentences)
        
        similarities = []
        for i in range(len(embeddings) - 1):
            sim = cosine_similarity(
                embeddings[i].reshape(1, -1),
                embeddings[i + 1].reshape(1, -1)
            )[0][0]
            similarities.append(sim)
        
        return similarities
    
    def chunk(self, text: str) -> list[dict]:
        """
        对文本进行语义分块
        
        Returns:
            分块列表，每个块包含 text、start_sentence、similarity_score
        """
        sentences = self._split_sentences(text)
        
        if len(sentences) <= 1:
            return [{"text": text, "sentences": sentences, "avg_similarity": 1.0}]
        
        similarities = self._calculate_similarities(sentences)
        
        # 在相似度低于阈值的位置切分
        chunks = []
        current_chunk = [sentences[0]]
        current_similarities = []
        
        for i, sim in enumerate(similarities):
            current_chunk.append(sentences[i + 1])
            current_similarities.append(sim)
            
            # 检查是否需要切分
            chunk_text = "".join(current_chunk)
            should_split = (
                sim < self.similarity_threshold or
                len(chunk_text) >= self.max_chunk_size
            )
            
            if should_split and len(chunk_text) >= self.min_chunk_size:
                chunks.append({
                    "text": chunk_text,
                    "sentence_count": len(current_chunk),
                    "avg_similarity": np.mean(current_similarities) if current_similarities else 1.0
                })
                current_chunk = [sentences[i + 1]]
                current_similarities = []
        
        # 处理最后一个块
        if current_chunk:
            chunk_text = "".join(current_chunk)
            if len(chunk_text) >= self.min_chunk_size or not chunks:
                chunks.append({
                    "text": chunk_text,
                    "sentence_count": len(current_chunk),
                    "avg_similarity": np.mean(current_similarities) if current_similarities else 1.0
                })
        
        return chunks


# 使用示例
chunker = SemanticChunker(
    model_name="BAAI/bge-small-zh-v1.5",
    similarity_threshold=0.5,
    min_chunk_size=50,
    max_chunk_size=500
)

text = """
人工智能是计算机科学的一个分支。它企图了解智能的实质。
深度学习是机器学习的分支。它使用多层神经网络进行学习。
自然语言处理是AI的重要应用。它让计算机理解人类语言。
计算机视觉让机器能够看懂图像。这是另一个重要方向。
"""

chunks = chunker.chunk(text)
for i, chunk in enumerate(chunks):
    print(f"\n块 {i+1} (相似度: {chunk['avg_similarity']:.2f}):")
    print(f"  {chunk['text'][:100]}...")
```

## 6. 文档级分块

保留文档结构信息，为每个块附加丰富的元数据。

```python
"""
文档级分块实现
保留文档结构，添加丰富元数据
"""

from dataclasses import dataclass, field
from typing import Optional
import hashlib

@dataclass
class ChunkMetadata:
    """分块元数据"""
    source: str                    # 来源文件
    page: Optional[int] = None     # 页码
    section: Optional[str] = None  # 章节标题
    chunk_index: int = 0           # 块索引
    total_chunks: int = 0          # 总块数
    chunk_type: str = "text"       # 块类型: text, table, image, code
    language: str = "zh"           # 语言
    created_at: str = ""           # 创建时间
    chunk_id: str = ""             # 唯一标识

@dataclass
class DocumentChunk:
    """文档分块"""
    text: str
    metadata: ChunkMetadata
    
    def __post_init__(self):
        if not self.metadata.chunk_id:
            # 生成唯一 ID
            content_hash = hashlib.md5(self.text.encode()).hexdigest()[:8]
            self.metadata.chunk_id = f"{self.metadata.source}_{self.metadata.chunk_index}_{content_hash}"


class DocumentLevelChunker:
    """
    文档级分块器
    
    特点：
    1. 保留文档结构（章节、段落）
    2. 附加丰富元数据
    3. 支持多种文档类型
    """
    
    def __init__(
        self,
        chunk_size: int = 500,
        chunk_overlap: int = 50,
        preserve_sections: bool = True
    ):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
        self.preserve_sections = preserve_sections
    
    def _extract_sections(self, text: str) -> list[dict]:
        """提取文档章节结构"""
        import re
        
        sections = []
        current_section = {"title": "Introduction", "content": ""}
        
        # 匹配中文章节标题
        section_patterns = [
            r'^第[一二三四五六七八九十\d]+[章节]',  # 第一章、第1节
            r'^\d+[\.\、]\d*\s*\S+',               # 1.1 标题
            r'^[一二三四五六七八九十]+[\、\.]\s*\S+', # 一、标题
            r'^#{1,3}\s+\S+',                       # Markdown 标题
        ]
        
        for line in text.split('\n'):
            is_section_header = False
            for pattern in section_patterns:
                if re.match(pattern, line.strip()):
                    # 保存之前的章节
                    if current_section["content"]:
                        sections.append(current_section)
                    current_section = {"title": line.strip(), "content": ""}
                    is_section_header = True
                    break
            
            if not is_section_header:
                current_section["content"] += line + "\n"
        
        # 保存最后一个章节
        if current_section["content"]:
            sections.append(current_section)
        
        return sections
    
    def chunk_document(
        self,
        text: str,
        source: str,
        chunk_type: str = "text"
    ) -> list[DocumentChunk]:
        """
        对文档进行分块
        
        Args:
            text: 文档文本
            source: 来源文件名
            chunk_type: 块类型
        
        Returns:
            DocumentChunk 列表
        """
        from datetime import datetime
        
        sections = self._extract_sections(text)
        all_chunks = []
        
        for section in sections:
            section_text = section["content"].strip()
            if not section_text:
                continue
            
            # 按段落切分
            paragraphs = [p.strip() for p in section_text.split('\n\n') if p.strip()]
            
            current_chunk_text = ""
            chunk_index = 0
            
            for para in paragraphs:
                if len(current_chunk_text) + len(para) <= self.chunk_size:
                    current_chunk_text += para + "\n\n"
                else:
                    if current_chunk_text:
                        all_chunks.append(DocumentChunk(
                            text=current_chunk_text.strip(),
                            metadata=ChunkMetadata(
                                source=source,
                                section=section["title"],
                                chunk_index=chunk_index,
                                chunk_type=chunk_type,
                                created_at=datetime.now().isoformat()
                            )
                        ))
                        chunk_index += 1
                    
                    current_chunk_text = para + "\n\n"
            
            # 保存最后一个块
            if current_chunk_text:
                all_chunks.append(DocumentChunk(
                    text=current_chunk_text.strip(),
                    metadata=ChunkMetadata(
                        source=source,
                        section=section["title"],
                        chunk_index=chunk_index,
                        chunk_type=chunk_type,
                        created_at=datetime.now().isoformat()
                    )
                ))
        
        # 更新 total_chunks
        for chunk in all_chunks:
            chunk.metadata.total_chunks = len(all_chunks)
        
        return all_chunks


# 使用示例
chunker = DocumentLevelChunker(chunk_size=300, preserve_sections=True)

text = """
# 第一章 人工智能概述

人工智能是计算机科学的一个分支。它企图了解智能的实质，
并生产出一种新的能以人类智能相似的方式做出反应的智能机器。

## 1.1 AI 的发展历程

人工智能的概念最早由 Alan Turing 提出。
1956 年达特茅斯会议标志着 AI 作为学科的诞生。

# 第二章 机器学习基础

机器学习是 AI 的核心子领域。它让计算机从数据中学习。
"""

chunks = chunker.chunk_document(text, source="ai_textbook.pdf")
for chunk in chunks:
    print(f"\n块 {chunk.metadata.chunk_id}:")
    print(f"  来源: {chunk.metadata.source}")
    print(f"  章节: {chunk.metadata.section}")
    print(f"  内容: {chunk.text[:50]}...")
```

## 7. 分块策略对比

| 策略 | 原理 | 优点 | 缺点 | 适用场景 |
|------|------|------|------|----------|
| **固定大小** | 按字符数切分 | 实现简单、速度快 | 可能切断语义 | 快速原型、文本结构简单 |
| **递归字符** | 按分隔符层级递归 | 保持句子完整 | 依赖分隔符质量 | 通用场景（推荐默认） |
| **语义分块** | 按 Embedding 相似度 | 语义边界准确 | 计算成本高 | 高质量 RAG 系统 |
| **文档级** | 保留文档结构 | 元数据丰富 | 实现复杂 | 需要追溯来源的场景 |
| **按段落** | 按段落切分 | 自然边界 | 段落长度不均 | 结构清晰的文档 |

**选型建议：**
- 快速原型 → 固定大小或递归字符
- 生产环境 → 递归字符（默认）或语义分块
- 需要来源追溯 → 文档级分块
- 高质量要求 → 语义分块

## 10. 工程建议

1. **先用递归字符分块做基线**：不要一上来就上语义分块。递归字符分块速度快、效果稳定，是 LangChain 和 LlamaIndex 的默认推荐。先用它建立 Recall@K 基线，再考虑更复杂的策略。
2. **用 A/B 测试验证分块参数**：chunk_size 和 chunk_overlap 的最优值取决于文档类型和查询模式。建议准备一组测试问答对，对不同参数组合跑评估，用数据驱动决策而不是拍脑袋。
3. **为不同类型文档设置不同分块策略**：FAQ 文档适合按条目分块（每个 Q&A 一个块），技术文档适合按章节分块，对话记录适合按轮次分块。不要对所有文档用同一套参数。
4. **保留分块溯源能力**：每个分块必须携带源文件名、页码、章节标题等元数据。当用户质疑回答准确性时，你能快速定位到原始文档位置进行人工核实。

# 策略 1: 固定大小分块
def fixed_chunk(text, size=200):
    return [text[i:i+size] for i in range(0, len(text), size)]

# 策略 2: 递归字符分块
def recursive_chunk(text, size=200):

        separators=["\n\n", "\n", "。", "，", " "]
    )
    return splitter.split_text(text)

# 策略 3: 语义分块（简化版：按段落+主题句检测）
def semantic_chunk(text):
    paragraphs = [p.strip() for p in text.split("\n\n") if p.strip()]
    return paragraphs

### 练习 2：语义分块调优

**思路**：similarity_threshold 是语义分块的核心参数——阈值越低切分越细（更多小块），阈值越高切分越粗（更少大块）。通过对比 0.3、0.5、0.7 三个阈值的分块结果，找到语义完整性和检索精度的平衡点。

**答案**：

```python
from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np
import re

class TunableSemanticChunker:
    def __init__(self, model_name="BAAI/bge-small-zh-v1.5", threshold=0.5):
        self.model = SentenceTransformer(model_name)
        self.threshold = threshold

    def _split_sentences(self, text):
        sentences = re.split(r'([。！？\n])', text)
        merged = []
        for i in range(0, len(sentences) - 1, 2):
            merged.append(sentences[i] + sentences[i + 1])
        if len(sentences) % 2 == 1 and sentences[-1]:
            merged.append(sentences[-1])
        return [s.strip() for s in merged if s.strip()]

    def chunk(self, text):
        sentences = self._split_sentences(text)
        if len(sentences) <= 1:
            return [text]

        embeddings = self.model.encode(sentences)
        chunks = []
        current = [sentences[0]]

        for i in range(len(embeddings) - 1):
            sim = cosine_similarity(
                embeddings[i].reshape(1, -1),
                embeddings[i + 1].reshape(1, -1)
            )[0][0]

            current.append(sentences[i + 1])

            if sim < self.threshold:
                chunks.append("".join(current))
                current = [sentences[i + 1]]

        if current:
            chunks.append("".join(current))
        return chunks

# 测试不同阈值
tech_doc = """
Redis支持两种持久化方式。RDB快照是在指定时间间隔将数据集快照写入磁盘。它的优点是文件紧凑、恢复速度快，缺点是可能丢失最后一次快照后的数据。AOF日志记录每个写操作命令。它的优点是数据安全性更高，缺点是文件体积较大、恢复速度较慢。在生产环境中，通常同时启用两种方式以兼顾性能和安全。
Redis的主从复制功能允许从节点复制主节点的数据。复制过程是异步的，主节点不需要等待从节点确认。这提供了读写分离的能力，主节点负责写操作，从节点分担读压力。
"""

for threshold in [0.3, 0.5, 0.7]:
    chunker = TunableSemanticChunker(threshold=threshold)
    chunks = chunker.chunk(tech_doc)
    print(f"\n阈值 {threshold}: {len(chunks)} 个块")
    for i, chunk in enumerate(chunks):
        print(f"  块{i+1}: {chunk[:80]}...")
```

**要点**：
- threshold=0.3 会产生更多小块，适合精确检索但可能丢失上下文
- threshold=0.7 会产生更少大块，上下文完整但检索精度下降
- 技术文档推荐 0.5 作为起点，FAQ 类文档可适当降低到 0.4

### 练习 3：元数据增强

**思路**：为每个分块附加来源文件名、章节标题、位置百分比和 TF-IDF 关键词。位置百分比帮助用户了解信息在文档中的大致位置，关键词支持后续的元数据过滤和搜索。

**答案**：

```python
from dataclasses import dataclass
from typing import Optional
import hashlib
import re
from sklearn.feature_extraction.text import TfidfExtractor

@dataclass
class EnhancedChunk:
    text: str
    source: str
    section: str
    position_pct: float
    keywords: list[str]
    chunk_id: str

def extract_sections(text: str) -> list[dict]:
    """提取章节结构"""
    sections = []
    current = {"title": "引言", "content": ""}
    for line in text.split("\n"):
        if re.match(r'^#{1,3}\s+', line.strip()) or re.match(r'^第[一二三四五六七八九十\d]+[章节]', line.strip()):
            if current["content"].strip():
                sections.append(current)
            current = {"title": line.strip(), "content": ""}
        else:
            current["content"] += line + "\n"
    if current["content"].strip():
        sections.append(current)
    return sections

def extract_keywords_tfidf(texts: list[str], top_n: int = 5) -> list[list[str]]:
    """使用 TF-IDF 提取关键词"""
    from sklearn.feature_extraction.text import TfidfVectorizer
    vectorizer = TfidfVectorizer(
        max_features=1000,
        token_pattern=r'(?u)\b\w{2,}\b'
    )
    tfidf_matrix = vectorizer.fit_transform(texts)
    feature_names = vectorizer.get_feature_names_out()

    keywords_list = []
    for row in tfidf_matrix.toarray():
        top_indices = row.argsort()[-top_n:][::-1]
        keywords = [feature_names[i] for i in top_indices if row[i] > 0]
        keywords_list.append(keywords)
    return keywords_list

def chunk_with_metadata(
    text: str,
    source: str,
    chunk_size: int = 500
) -> list[EnhancedChunk]:
    """带丰富元数据的分块"""
    sections = extract_sections(text)
    all_chunks = []
    total_chars = len(text)
    processed_chars = 0

    for section in sections:
        content = section["content"].strip()
        if not content:
            continue

        # 按 chunk_size 切分
        paragraphs = [p.strip() for p in content.split("\n\n") if p.strip()]
        current_text = ""
        for para in paragraphs:
            if len(current_text) + len(para) <= chunk_size:
                current_text += para + "\n\n"
            else:
                if current_text:
                    all_chunks.append({
                        "text": current_text.strip(),
                        "source": source,
                        "section": section["title"],
                        "position_pct": processed_chars / total_chars
                    })
                    processed_chars += len(current_text)
                    current_text = para + "\n\n"
        if current_text:
            all_chunks.append({
                "text": current_text.strip(),
                "source": source,
                "section": section["title"],
                "position_pct": processed_chars / total_chars
            })
            processed_chars += len(current_text)

    # 提取关键词
    texts = [c["text"] for c in all_chunks]
    keywords_list = extract_keywords_tfidf(texts)

    # 组装最终结果
    result = []
    for chunk_data, keywords in zip(all_chunks, keywords_list):
        chunk_id = hashlib.md5(chunk_data["text"].encode()).hexdigest()[:8]
        result.append(EnhancedChunk(
            text=chunk_data["text"],
            source=chunk_data["source"],
            section=chunk_data["section"],
            position_pct=round(chunk_data["position_pct"], 2),
            keywords=keywords,
            chunk_id=chunk_id
        ))

    return result

# 使用示例
text = "# 第一章 概述\n\nAI是计算机科学分支...\n\n# 第二章 技术\n\n深度学习使用神经网络..."
chunks = chunk_with_metadata(text, source="ai_textbook.pdf")
for c in chunks:
    print(f"[{c.chunk_id}] 来源:{c.source}, 章节:{c.section}, 位置:{c.position_pct}")
    print(f"  关键词: {c.keywords}")
    print(f"  内容: {c.text[:50]}...")
```

**要点**：
- 位置百分比（position_pct）帮助判断信息在文档中的分布，靠前的通常是概述，靠后的通常是细节
- TF-IDF 关键词提取不需要额外模型，计算成本低，适合批量处理
- chunk_id 使用内容哈希生成，相同内容的块有相同 ID，便于去重
