```python
"""
Stage 2 阶段实战: 完整优化检索管线
整合: 混合检索 + 查询扩展 + Reranker + 上下文压缩 + 评估
"""
import os
import json
import math
import time
import hashlib
from typing import List, Dict, Set, Optional, Tuple, Callable
from dataclasses import dataclass, field, asdict
from collections import defaultdict
from datetime import datetime

from langchain.schema import Document
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_community.retrievers import BM25Retriever
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from sentence_transformers import CrossEncoder


# ============================================================
# 评估指标
# ============================================================

def recall_at_k(retrieved: List[str], relevant: Set[str], k: int) -> float:
    if not relevant:
        return 0.0
    return len(set(retrieved[:k]) & relevant) / len(relevant)

def precision_at_k(retrieved: List[str], relevant: Set[str], k: int) -> float:
    if k == 0:
        return 0.0
    return sum(1 for d in retrieved[:k] if d in relevant) / k

def mrr(retrieved: List[str], relevant: Set[str], k: int = 10) -> float:
    for i, d in enumerate(retrieved[:k]):
        if d in relevant:
            return 1.0 / (i + 1)
    return 0.0

def ndcg_at_k(retrieved: List[str], relevant: Set[str], k: int) -> float:
    if not relevant:
        return 0.0
    dcg = sum(
        (1.0 if retrieved[i] in relevant else 0.0) / math.log2(i + 2)
        for i in range(min(k, len(retrieved)))
    )
    ideal = sorted(
        [1.0 if retrieved[i] in relevant else 0.0 for i in range(min(k, len(retrieved)))],
        reverse=True
    )
    while len(ideal) < min(len(relevant), k):
        ideal.append(1.0)
    idcg = sum(r / math.log2(i + 2) for i, r in enumerate(ideal[:k]))
    return dcg / idcg if idcg > 0 else 0.0


# ============================================================
# 查询扩展 (Multi-Query)
# ============================================================

class QueryExpander:
    """使用 LLM 将查询扩展为多个变体"""

    def __init__(self, model: str = "gpt-4o-mini"):
        self.llm = ChatOpenAI(model=model, temperature=0.3)
        self.prompt = ChatPromptTemplate.from_template("""
你是一个查询改写专家。请将以下查询改写为 3 个不同的变体，以提高检索召回率。
要求:
1. 保留原始查询的核心意图
2. 使用不同的表述方式（同义词、更具体、更抽象）
3. 每行一个变体，不要编号

原始查询: {query}

改写后的查询:
""")

    def expand(self, query: str) -> List[str]:
        """扩展查询为多个变体"""
        chain = self.prompt | self.llm
        response = chain.invoke({"query": query})
        variants = [
            line.strip()
            for line in response.content.strip().split("\n")
            if line.strip() and line.strip() != query
        ]
        return [query] + variants  # 包含原始查询


# ============================================================
# 上下文压缩器
# ============================================================

class ContextualCompressor:
    """使用 LLM 压缩文档，只保留与查询相关的部分"""

    def __init__(self, model: str = "gpt-4o-mini"):
        self.llm = ChatOpenAI(model=model, temperature=0)
        self.prompt = ChatPromptTemplate.from_template("""
从以下文档中提取与查询最相关的部分。
规则:
- 保留原文，不要改写
- 只保留直接回答查询的内容
- 如果文档与查询无关，输出 "IRRELEVANT"

查询: {query}
文档: {document}

压缩后:
""")

    def compress(self, query: str, documents: List[Document]) -> List[Document]:
        chain = self.prompt | self.llm
        results = []
        for doc in documents:
            response = chain.invoke({"query": query, "document": doc.page_content})
            content = response.content.strip()
            if content != "IRRELEVANT" and len(content) > 10:
                results.append(Document(
                    page_content=content,
                    metadata={**doc.metadata, "compressed": True}
                ))
        return results

@dataclass
class RetrieverConfig:
    """检索器配置"""
    use_bm25: bool = True
    use_query_expansion: bool = True
    use_reranker: bool = True
    use_compression: bool = True
    bm25_weight: float = 0.4
    vector_weight: float = 0.6
    rrf_k: int = 60
    child_k: int = 15
    retrieval_k: int = 10
    final_k: int = 5
    reranker_model: str = "BAAI/bge-reranker-v2-m3"
    llm_model: str = "gpt-4o-mini"


class OptimizedRetriever:
    """
    Stage 2 优化检索器

    支持的优化步骤（可独立开关）:
    1. BM25 混合检索
    2. Multi-Query 查询扩展
    3. Reranker 精排
    4. 上下文压缩
    """

    def __init__(
        self,
        documents: List[Document],
        config: RetrieverConfig = None
    ):
        self.config = config or RetrieverConfig()
        self.embeddings = OpenAIEmbeddings()
        self.doc_id_map: Dict[str, Document] = {}

        # 构建索引
        self._build_vector_index(documents)
        if self.config.use_bm25:
            self._build_bm25_index(documents)

        # 可选组件
        if self.config.use_query_expansion:
            self.query_expander = QueryExpander(self.config.llm_model)
        if self.config.use_reranker:
            self.reranker = CrossEncoder(self.config.reranker_model, max_length=512)
        if self.config.use_compression:
            self.compressor = ContextualCompressor(self.config.llm_model)

    def _get_doc_id(self, doc: Document) -> str:
        return hashlib.md5(doc.page_content.encode()).hexdigest()[:12]

    def _build_vector_index(self, documents: List[Document]):
        for doc in documents:
            doc_id = self._get_doc_id(doc)
            doc.metadata["doc_id"] = doc_id
            self.doc_id_map[doc_id] = doc
        self.vectorstore = Chroma.from_documents(documents, self.embeddings)

    def _build_bm25_index(self, documents: List[Document]):
        self.bm25 = BM25Retriever.from_documents(documents)
        self.bm25.k = self.config.retrieval_k

    def _rrf_fusion(
        self,
        results_list: List[List[Tuple[str, float, Document]]]
    ) -> List[Tuple[float, Document]]:
        k = self.config.rrf_k
        scores: Dict[str, float] = defaultdict(float)
        doc_map: Dict[str, Document] = {}

        for results in results_list:
            for rank, (doc_id, _, doc) in enumerate(results):
                scores[doc_id] += 1.0 / (k + rank + 1)
                doc_map[doc_id] = doc

        sorted_docs = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return [(s, doc_map[did]) for did, s in sorted_docs]

    def retrieve(
        self,
        query: str,
        metadata_filter: Optional[Dict] = None
    ) -> List[Dict]:
        """
        执行优化检索

        Returns:
            [{"content": str, "metadata": dict, "score": float}, ...]
        """
        # Step 1: 查询扩展
        queries = [query]
        if self.config.use_query_expansion:
            queries = self.query_expander.expand(query)

        # Step 2: 多路检索（对每个查询变体）
        all_vector_results = []
        all_bm25_results = []

        for q in queries:
            # 向量检索
            vf = metadata_filter
            vec_docs = self.vectorstore.similarity_search_with_score(
                q, k=self.config.retrieval_k, filter=vf
            )
            all_vector_results.extend([
                (self._get_doc_id(doc), score, doc) for doc, score in vec_docs
            ])

            # BM25 检索
            if self.config.use_bm25:
                bm25_docs = self.bm25.get_relevant_documents(q)
                all_bm25_results.extend([
                    (self._get_doc_id(doc), 0.0, doc) for doc in bm25_docs
                ])

        # 去重（保留最高分）
        def dedup(results: List[Tuple[str, float, Document]]) -> List[Tuple[str, float, Document]]:
            seen = {}
            for doc_id, score, doc in results:
                if doc_id not in seen or score > seen[doc_id][1]:
                    seen[doc_id] = (doc_id, score, doc)
            return list(seen.values())

        vector_results = dedup(all_vector_results)
        bm25_results = dedup(all_bm25_results) if self.config.use_bm25 else []

        # Step 3: RRF 融合
        to_fuse = [vector_results]
        if bm25_results:
            to_fuse.append(bm25_results)
        fused = self._rrf_fusion(to_fuse)

        # Step 4: Reranker 精排
        if self.config.use_reranker and fused:
            pairs = [(query, doc.page_content) for _, doc in fused]
            rerank_scores = self.reranker.predict(pairs)
            scored = list(zip(rerank_scores, [doc for _, doc in fused]))
            scored.sort(key=lambda x: x[0], reverse=True)
            fused = [(float(s), doc) for s, doc in scored]

        # Step 5: 上下文压缩
        final_docs = fused[:self.config.final_k]
        if self.config.use_compression:
            docs_to_compress = [doc for _, doc in final_docs]
            compressed = self.compressor.compress(query, docs_to_compress)
            final_docs = [(score, doc) for (score, _), doc in zip(final_docs, compressed)]

        # 构建结果
        return [
            {
                "content": doc.page_content,
                "metadata": doc.metadata,
                "score": score,
                "doc_id": doc.metadata.get("doc_id", "")
            }
            for score, doc in final_docs
        ]

@dataclass
class EvalCase:
    query: str
    relevant_doc_ids: Set[str]
    category: str = "general"
    difficulty: str = "medium"


class EvalDataset:
    """评估数据集"""

    def __init__(self):
        self.cases: List[EvalCase] = []

    def add(self, query: str, relevant_ids: List[str], **kwargs):
        self.cases.append(EvalCase(
            query=query,
            relevant_doc_ids=set(relevant_ids),
            **kwargs
        ))

    def load_from_json(self, filepath: str):
        with open(filepath, "r", encoding="utf-8") as f:
            data = json.load(f)
        for ann in data.get("annotations", []):
            self.add(
                query=ann["query"],
                relevant_ids=ann["relevant_doc_ids"],
                category=ann.get("category", "general"),
                difficulty=ann.get("difficulty", "medium")
            )

    @property
    def queries(self) -> List[str]:
        return [c.query for c in self.cases]

@dataclass
class ABResult:
    config_name: str
    recall_at_5: float
    precision_at_5: float
    mrr: float
    ndcg_at_5: float
    avg_latency_ms: float
    per_query: List[Dict] = field(default_factory=list)


class ABEvaluator:
    """A/B 对比评估器"""

    def __init__(self, dataset: EvalDataset):
        self.dataset = dataset

    def evaluate(
        self,
        config_name: str,
        retrieve_fn: Callable[[str], List[Dict]],
        k: int = 5
    ) -> ABResult:
        """评估单个配置"""
        per_query = []
        total_latency = 0.0

        for case in self.dataset.cases:
            start = time.time()
            results = retrieve_fn(case.query)
            latency = (time.time() - start) * 1000
            total_latency += latency

            doc_ids = [r.get("doc_id", "") for r in results]

            per_query.append({
                "query": case.query,
                "category": case.category,
                "difficulty": case.difficulty,
                "recall": recall_at_k(doc_ids, case.relevant_doc_ids, k),
                "precision": precision_at_k(doc_ids, case.relevant_doc_ids, k),
                "rr": mrr(doc_ids, case.relevant_doc_ids, k),
                "ndcg": ndcg_at_k(doc_ids, case.relevant_doc_ids, k),
                "latency_ms": latency
            })

        n = len(self.dataset.cases)
        return ABResult(
            config_name=config_name,
            recall_at_5=sum(r["recall"] for r in per_query) / n,
            precision_at_5=sum(r["precision"] for r in per_query) / n,
            mrr=sum(r["rr"] for r in per_query) / n,
            ndcg_at_5=sum(r["ndcg"] for r in per_query) / n,
            avg_latency_ms=total_latency / n,
            per_query=per_query
        )

    @staticmethod
    def compare(results: List[ABResult]) -> str:
        """生成对比报告"""
        lines = []
        lines.append("=" * 85)
        lines.append(f"{'Config':<25} {'Recall@5':<10} {'Prec@5':<10} {'MRR':<10} {'NDCG@5':<10} {'Latency':<10}")
        lines.append("-" * 85)

        baseline = results[0] if results else None
        for r in results:
            lines.append(
                f"{r.config_name:<25} "
                f"{r.recall_at_5:<10.3f} "
                f"{r.precision_at_5:<10.3f} "
                f"{r.mrr:<10.3f} "
                f"{r.ndcg_at_5:<10.3f} "
                f"{r.avg_latency_ms:<10.0f}ms"
            )

        # 增量对比
        if baseline and len(results) > 1:
            lines.append("-" * 85)
            lines.append("增量对比 (相对基线):")
            for r in results[1:]:
                d_recall = r.recall_at_5 - baseline.recall_at_5
                d_mrr = r.mrr - baseline.mrr
                d_lat = r.avg_latency_ms - baseline.avg_latency_ms
                lines.append(
                    f"  {r.config_name:<23} "
                    f"Recall: {d_recall:+.3f} | "
                    f"MRR: {d_mrr:+.3f} | "
                    f"Latency: {d_lat:+.0f}ms"
                )

        lines.append("=" * 85)
        return "\n".join(lines)

    @staticmethod
    def export(results: List[ABResult], filepath: str):
        data = {
            "experiments": [asdict(r) for r in results],
            "exported_at": datetime.now().isoformat()
        }
        with open(filepath, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        print(f"结果已导出: {filepath}")

# 端到端评估: 检索质量 → 生成质量

class EndToEndEvaluator:
    """
    端到端评估: 检索到的内容 → LLM 生成的回答质量

    评估维度:
    - 相关性: 回答是否与问题相关
    - 完整性: 回答是否涵盖了关键信息
    - 忠实度: 回答是否基于检索到的文档（不幻觉）
    """

    def __init__(self, model: str = "gpt-4o-mini"):
        self.llm = ChatOpenAI(model=model, temperature=0)
        self.eval_prompt = ChatPromptTemplate.from_template("""
你是一个 RAG 系统评估专家。请评估以下回答的质量。

用户问题: {question}

检索到的文档:
{context}

系统回答: {answer}

请从以下三个维度打分 (0-5 分):
1. 相关性: 回答是否与问题相关
2. 完整性: 回答是否涵盖了关键信息
3. 忠实度: 回答是否基于检索文档（不编造信息）

输出 JSON 格式:
{{"relevance": <分数>, "completeness": <分数>, "faithfulness": <分数>, "reason": "<简要说明>"}}
""")

    def evaluate_single(
        self,
        question: str,
        context_docs: List[str],
        answer: str
    ) -> Dict:
        """评估单个 QA 对"""
        context = "\n---\n".join(context_docs)
        chain = self.eval_prompt | self.llm
        response = chain.invoke({
            "question": question,
            "context": context,
            "answer": answer
        })
        try:
            return json.loads(response.content)
        except json.JSONDecodeError:
            return {"relevance": 0, "completeness": 0, "faithfulness": 0, "reason": "解析失败"}

    def evaluate_batch(
        self,
        qa_pairs: List[Dict]
    ) -> Dict[str, float]:
        """
        批量端到端评估

        Args:
            qa_pairs: [{"question": str, "context_docs": [str], "answer": str}, ...]

        Returns:
            {"mean_relevance": float, "mean_completeness": float, ...}
        """
        results = []
        for qa in qa_pairs:
            result = self.evaluate_single(
                qa["question"], qa["context_docs"], qa["answer"]
            )
            results.append(result)

        n = len(results)
        return {
            "mean_relevance": sum(r.get("relevance", 0) for r in results) / n,
            "mean_completeness": sum(r.get("completeness", 0) for r in results) / n,
            "mean_faithfulness": sum(r.get("faithfulness", 0) for r in results) / n,
            "num_evaluated": n,
            "per_question": results
        }

# 主流程: 完整 A/B 实验
# ============================================================

def run_full_experiment(
    documents: List[Document],
    dataset: EvalDataset,
    export_path: str = "ab_results.json"
):
    """
    运行完整的 A/B 实验

    对比 5 种配置:
    A) 纯向量检索 (基线)
    B) + BM25 混合检索
    C) + 查询扩展
    D) + Reranker
    E) + 上下文压缩 (全量优化)
    """
    evaluator = ABEvaluator(dataset)
    results = []

    # --- 配置 A: 纯向量检索 ---
    print("\n[Config A] 纯向量检索 (基线)...")
    config_a = RetrieverConfig(
        use_bm25=False, use_query_expansion=False,
        use_reranker=False, use_compression=False
    )
    retriever_a = OptimizedRetriever(documents, config_a)
    result_a = evaluator.evaluate("A: Vector Only", retriever_a.retrieve)
    results.append(result_a)

    # --- 配置 B: + BM25 ---
    print("[Config B] 向量 + BM25...")
    config_b = RetrieverConfig(
        use_bm25=True, use_query_expansion=False,
        use_reranker=False, use_compression=False
    )
    retriever_b = OptimizedRetriever(documents, config_b)
    result_b = evaluator.evaluate("B: + BM25", retriever_b.retrieve)
    results.append(result_b)

    # --- 配置 C: + 查询扩展 ---
    print("[Config C] 向量 + BM25 + 查询扩展...")
    config_c = RetrieverConfig(
        use_bm25=True, use_query_expansion=True,
        use_reranker=False, use_compression=False
    )
    retriever_c = OptimizedRetriever(documents, config_c)
    result_c = evaluator.evaluate("C: + Query Expansion", retriever_c.retrieve)
    results.append(result_c)

    # --- 配置 D: + Reranker ---
    print("[Config D] 向量 + BM25 + 查询扩展 + Reranker...")
    config_d = RetrieverConfig(
        use_bm25=True, use_query_expansion=True,
        use_reranker=True, use_compression=False
    )
    retriever_d = OptimizedRetriever(documents, config_d)
    result_d = evaluator.evaluate("D: + Reranker", retriever_d.retrieve)
    results.append(result_d)

    # --- 配置 E: 全量优化 ---
    print("[Config E] 全量优化 (所有技术)...")
    config_e = RetrieverConfig(
        use_bm25=True, use_query_expansion=True,
        use_reranker=True, use_compression=True
    )
    retriever_e = OptimizedRetriever(documents, config_e)
    result_e = evaluator.evaluate("E: Full Pipeline", retriever_e.retrieve)
    results.append(result_e)

    # 对比报告
    print("\n" + ABEvaluator.compare(results))

    # 导出
    ABEvaluator.export(results, export_path)

    return results
```

## 4. 常见误区

### ❌ 错误 1: 一次改太多

```python
# 错误: 同时开启所有优化，不知道哪个起了作用
config = RetrieverConfig(
    use_bm25=True, use_query_expansion=True,
    use_reranker=True, use_compression=True
)

# 正确: 逐个开启，记录每步增量
# Step 1: 只开 BM25 → 对比基线
# Step 2: 开 BM25 + 查询扩展 → 对比 Step 1
# ...
```

### ❌ 错误 2: 评估数据集太小或太简单

```python
# 错误: 5 个查询，全部是 easy 难度
dataset.add("iPhone 电池", ["doc1"])  # 太简单

# 正确: 50+ 查询，覆盖 easy/medium/hard
dataset.add("iPhone 电池", ["doc1"], difficulty="easy")
dataset.add("哪个手机续航最好", ["doc1", "doc5"], difficulty="medium")
dataset.add("对比旗舰手机的充电速度和电池容量", [...], difficulty="hard")
```

### ❌ 错误 3: 忽略延迟成本

```python
# 错误: Recall 从 0.80 提升到 0.82，延迟从 100ms 涨到 800ms
# 用户体验: 从"秒回"变成"等一下"

# 正确: 设置延迟预算
MAX_LATENCY_MS = 500
if result.avg_latency_ms > MAX_LATENCY_MS:
    print("⚠️ 超出延迟预算，考虑移除查询扩展或压缩")
```

### ❌ 错误 4: 不区分查询类型

```python
# 错误: 所有查询用同一套配置
# 事实: 简单查询不需要查询扩展，复杂查询才需要

# 正确: 根据查询复杂度动态调整
def adaptive_retrieve(query, config):
    if is_simple_query(query):
        config.use_query_expansion = False  # 跳过扩展
    return retriever.retrieve(query)
```

### ❌ 错误 5: 只看指标不看 Bad Cases

```python

# 但: 有 3 个重要查询 Recall = 0，完全检索不到

# 正确: 找出 Bad Cases 并分析
bad_cases = [r for r in per_query if r["recall"] < 0.3]
```

## 5. 工程建议

1. **一次只改一个变量**：在做优化实验时，每次只修改一个参数或组件。如果同时改了混合检索权重和 Reranker 模型，你无法判断效果提升来自哪个改动。
2. **优化决策要文档化**：记录每次实验的配置、指标结果和决策理由。三个月后当你需要回顾为什么选择某个参数时，这些记录就是最宝贵的资产。
3. **端到端评估比单一环节评估更重要**：检索 Recall 提升了 10% 但最终答案质量没变，说明瓶颈不在检索环节。始终以端到端指标（答案正确率、用户满意度）为最终衡量标准。
4. **建立持续优化的节奏**：检索优化不是一次性工作。建议每月用新收集的用户查询更新评估集，跑一轮完整的指标对比，持续迭代优化。
