## 场景引入

你的 RAGAS 自动评估显示系统 Faithfulness 得分 0.85，看起来不错。但当你手动抽查了 20 个回答后发现，有 5 个回答虽然"忠实于检索内容"，但引用的文档本身就是过时的政策——自动评估无法判断知识库内容是否正确。自动化指标能告诉你"系统是否正确使用了检索到的内容"，但无法判断"内容本身是否正确"。人工评估是自动评估的必要补充，而高效的标注工具和一致的标注标准是人工评估的基础。

```python
"""
标注指南模板
用于规范化人工评估流程
"""

import json
from dataclasses import dataclass, field, asdict
from typing import List, Dict, Optional
from enum import Enum


class RatingScale(Enum):
    """评分量表"""
    FIVE_POINT = {
        1: "非常差",
        2: "差",
        3: "一般",
        4: "好",
        5: "非常好",
    }
    BINARY = {
        0: "否",
        1: "是",
    }
    THREE_POINT = {
        1: "差",
        2: "一般",
        3: "好",
    }


@dataclass
class AnnotationDimension:
    """标注维度"""
    name: str
    description: str
    scale: Dict[int, str]
    guidelines: List[str] = field(default_factory=list)
    examples: List[Dict] = field(default_factory=list)


@dataclass
class AnnotationGuideline:
    """标注指南"""
    title: str
    version: str
    description: str
    dimensions: List[AnnotationDimension] = field(default_factory=list)
    general_rules: List[str] = field(default_factory=list)
    faq: List[Dict[str, str]] = field(default_factory=list)

    def to_markdown(self) -> str:
        """导出为 Markdown 格式"""
        lines = []
        lines.append(f"# {self.title}")
        lines.append(f"**版本**: {self.version}")
        lines.append(f"\n{self.description}\n")

        lines.append("## 通用规则")
        for i, rule in enumerate(self.general_rules, 1):
            lines.append(f"{i}. {rule}")

        for dim in self.dimensions:
            lines.append(f"\n## {dim.name}")
            lines.append(f"{dim.description}\n")

            lines.append("### 评分标准")
            for score, label in sorted(dim.scale.items()):
                lines.append(f"- **{score}**: {label}")

            if dim.guidelines:
                lines.append("\n### 标注指南")
                for g in dim.guidelines:
                    lines.append(f"- {g}")

            if dim.examples:
                lines.append("\n### 示例")
                for ex in dim.examples:
                    lines.append(f"\n**输入**: {ex.get('query', '')}")
                    lines.append(f"**回答**: {ex.get('answer', '')}")
                    lines.append(f"**推荐评分**: {ex.get('score', '')}")
                    lines.append(f"**理由**: {ex.get('reason', '')}")

        if self.faq:
            lines.append("\n## 常见问题")
            for item in self.faq:
                lines.append(f"\n**Q: {item['q']}**")
                lines.append(f"A: {item['a']}")

        return "\n".join(lines)
```

## 4. 标注工具开发 (Streamlit)

```python
"""
基于 Streamlit 的标注工具
"""

import streamlit as st
import json
import pandas as pd
from datetime import datetime
from pathlib import Path


ANNOTATION_FILE = "annotations.json"
GUIDELINE_FILE = "annotation_guideline.md"


def load_samples() -> list:
    """加载待标注样本"""
    if "samples" not in st.session_state:
        default_samples = [
            {
                "id": "sample_001",
                "query": "什么是 RAG 技术？",
                "answer": "RAG 是检索增强生成技术，结合了信息检索和大语言模型。",
                "contexts": ["RAG (Retrieval-Augmented Generation) 是一种结合外部知识检索与 LLM 的技术框架。"],
                "ground_truth": "RAG (Retrieval-Augmented Generation) 是一种结合外部知识检索与大语言模型生成的技术框架。",
            },
            {
                "id": "sample_002",
                "query": "如何优化向量检索？",
                "answer": "可以通过 HNSW 索引、量化压缩、过滤预处理等方式优化。",
                "contexts": ["向量检索优化策略包括: 1) 使用 HNSW 索引 2) PQ 量化压缩 3) 元数据过滤预处理"],
                "ground_truth": "向量检索优化包括使用 HNSW 索引、PQ 量化、元数据过滤等方法。",
            },
        ]
        st.session_state.samples = default_samples
    return st.session_state.samples


def load_annotations() -> list:
    """加载已有标注"""
    if Path(ANNOTATION_FILE).exists():
        with open(ANNOTATION_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    return []


def save_annotation(annotation: dict):
    """保存标注结果"""
    annotations = load_annotations()
    annotations.append(annotation)
    with open(ANNOTATION_FILE, "w", encoding="utf-8") as f:
        json.dump(annotations, f, ensure_ascii=False, indent=2)


def main():
    st.set_page_config(page_title="RAG 评估标注工具", layout="wide")
    st.title("RAG 回答质量标注工具")

    # 侧边栏
    with st.sidebar:
        st.header("标注进度")
        samples = load_samples()
        annotations = load_annotations()
        annotated_ids = {a["sample_id"] for a in annotations}
        progress = len(annotated_ids) / max(len(samples), 1)
        st.progress(progress)
        st.write(f"已完成: {len(annotated_ids)}/{len(samples)}")

        st.header("标注指南")
        if Path(GUIDELINE_FILE).exists():
            with open(GUIDELINE_FILE, "r", encoding="utf-8") as f:
                st.markdown(f.read())
        else:
            st.info("标注指南文件不存在")

    # 主区域
    samples = load_samples()
    unannotated = [s for s in samples if s["id"] not in annotated_ids]

    if not unannotated:
        st.success("所有样本已标注完成！")
        st.balloons()
        return

    current = unannotated[0]

    # 显示样本信息
    st.header(f"样本: {current['id']}")

    col1, col2 = st.columns(2)
    with col1:
        st.subheader("用户问题")
        st.info(current["query"])

    with col2:
        st.subheader("标准答案")
        st.success(current["ground_truth"])

    st.subheader("RAG 回答")
    st.warning(current["answer"])

    with st.expander("检索上下文"):
        for i, ctx in enumerate(current.get("contexts", [])):
            st.text_area(f"上下文 {i+1}", ctx, height=100, disabled=True)

    # 评分区域
    st.markdown("---")
    st.subheader("评分")

    scores = {}
    dimensions = {
        "accuracy": ("准确性", "回答中的信息是否正确"),
        "completeness": ("完整性", "回答是否覆盖了所有关键信息"),
        "relevance": ("相关性", "回答是否直接回应用户的问题"),
        "readability": ("可读性", "回答是否清晰易懂"),
        "overall": ("总体满意度", "作为用户，你对这个回答的满意程度"),
    }

    cols = st.columns(len(dimensions))
    for col, (key, (name, desc)) in zip(cols, dimensions.items()):
        with col:
            scores[key] = st.slider(
                name,
                min_value=1,
                max_value=5,
                value=3,
                help=desc,
            )

    # 备注
    notes = st.text_area("备注 (可选)", placeholder="如有特殊情况请在此备注...")

    # 是否需要审核
    needs_review = st.checkbox("标记为需审核")

    # 提交
    if st.button("提交标注", type="primary"):
        annotation = {
            "sample_id": current["id"],
            "query": current["query"],
            "scores": scores,
            "notes": notes,
            "needs_review": needs_review,
            "annotator": st.session_state.get("annotator", "unknown"),
            "timestamp": datetime.now().isoformat(),
        }
        save_annotation(annotation)
        st.success("标注已保存！")
        st.rerun()


if __name__ == "__main__":
    main()
```

## 5. 标注一致性 (Inter-Annotator Agreement)

```python
"""
标注一致性计算
Cohen's Kappa, Fleiss' Kappa, Krippendorff's Alpha
"""

import numpy as np
from typing import List, Dict, Tuple
from collections import Counter
from itertools import combinations


def cohens_kappa(
    annotator1: List[int],
    annotator2: List[int],
) -> float:
    """
    Cohen's Kappa: 两个标注者之间的一致性

    参数:
        annotator1: 标注者1的评分列表
        annotator2: 标注者2的评分列表

    返回:
        Kappa 值 (-1 到 1, 通常 0.6 以上表示较好的一致性)
    """
    assert len(annotator1) == len(annotator2), "两个标注者的样本数必须相同"

    n = len(annotator1)
    categories = sorted(set(annotator1) | set(annotator2))

    # 构建混淆矩阵
    cm = np.zeros((len(categories), len(categories)), dtype=int)
    cat_to_idx = {c: i for i, c in enumerate(categories)}

    for a1, a2 in zip(annotator1, annotator2):
        cm[cat_to_idx[a1]][cat_to_idx[a2]] += 1

    # 计算观察一致率 (Po)
    po = np.trace(cm) / n

    # 计算期望一致率 (Pe)
    row_sums = cm.sum(axis=1)
    col_sums = cm.sum(axis=0)
    pe = np.sum(row_sums * col_sums) / (n * n)

    # 计算 Kappa
    if pe == 1.0:
        return 1.0
    kappa = (po - pe) / (1 - pe)
    return float(kappa)


def interpret_kappa(kappa: float) -> str:
    """解读 Kappa 值"""
    if kappa >= 0.8:
        return "几乎完全一致 (Almost Perfect)"
    elif kappa >= 0.6:
        return "高度一致 (Substantial)"
    elif kappa >= 0.4:
        return "中等一致 (Moderate)"
    elif kappa >= 0.2:
        return "一般一致 (Fair)"
    else:
        return "一致性差 (Slight/Poor)"


def fleiss_kappa(
    ratings: List[List[int]],
    num_categories: int,
) -> float:
    """
    Fleiss' Kappa: 多个标注者之间的一致性

    参数:
        ratings: 每个样本的评分列表 [[rater1, rater2, ...], ...]
        num_categories: 评分类别数

    返回:
        Kappa 值
    """
    n = len(ratings)
    k = num_categories
    m = len(ratings[0])  # 标注者数量

    # 构建频率矩阵
    freq_matrix = np.zeros((n, k), dtype=int)
    for i, sample_ratings in enumerate(ratings):
        for rating in sample_ratings:
            freq_matrix[i][rating] += 1

    # 计算每个样本的一致率
    p_i = np.sum(freq_matrix * (freq_matrix - 1), axis=1) / (m * (m - 1))
    p_bar = np.mean(p_i)

    # 计算每个类别的比例
    p_j = np.sum(freq_matrix, axis=0) / (n * m)

    # 计算期望一致率
    pe = np.sum(p_j ** 2)

    # 计算 Kappa
    if pe == 1.0:
        return 1.0
    kappa = (p_bar - pe) / (1 - pe)
    return float(kappa)


def compute_per_dimension_agreement(
    annotations: List[Dict],
    dimensions: List[str],
) -> Dict[str, Dict[str, float]]:
    """计算每个评估维度的标注一致性"""
    results = {}

    for dim in dimensions:
        # 提取每个标注者在该维度的评分
        by_annotator = {}
        for ann in annotations:
            annotator = ann["annotator"]
            if annotator not in by_annotator:
                by_annotator[annotator] = {}
            by_annotator[annotator][ann["sample_id"]] = ann["scores"].get(dim, 3)

        annotators = sorted(by_annotator.keys())
        if len(annotators) < 2:
            results[dim] = {"kappa": None, "message": "需要至少2个标注者"}
            continue

        # 计算所有标注者对的 Kappa
        kappas = []
        for a1, a2 in combinations(annotators, 2):
            common_samples = sorted(
                set(by_annotator[a1].keys()) & set(by_annotator[a2].keys())
            )
            if common_samples:
                r1 = [by_annotator[a1][s] for s in common_samples]
                r2 = [by_annotator[a2][s] for s in common_samples]
                kappa = cohens_kappa(r1, r2)
                kappas.append(kappa)

        avg_kappa = np.mean(kappas) if kappas else 0.0
        results[dim] = {
            "avg_kappa": float(avg_kappa),
            "interpretation": interpret_kappa(avg_kappa),
            "pairwise_kappas": kappas,
        }

    return results


# ============================================================
# 使用示例
# ============================================================

if __name__ == "__main__":
    # 两个标注者的示例
    rater1 = [4, 3, 5, 2, 4, 3, 5, 4, 3, 2]
    rater2 = [4, 3, 4, 2, 5, 3, 5, 4, 3, 3]

    kappa = cohens_kappa(rater1, rater2)
    print(f"Cohen's Kappa: {kappa:.4f}")
    print(f"解读: {interpret_kappa(kappa)}")

    # 多个标注者的示例
    multi_ratings = [
        [4, 4, 3],  # 样本1: 三个标注者的评分
        [3, 3, 3],  # 样本2
        [5, 4, 5],  # 样本3
        [2, 2, 2],  # 样本4
        [4, 3, 4],  # 样本5
    ]
    fk = fleiss_kappa(multi_ratings, num_categories=5)
    print(f"\nFleiss' Kappa: {fk:.4f}")
    print(f"解读: {interpret_kappa(fk)}")
```

## 6. 人工评估与自动化指标的关联

```python
"""
人工评估与自动化指标的关联分析
"""

import numpy as np
from typing import List, Dict, Any
from scipy import stats


def compute_correlation(
    human_scores: List[float],
    auto_scores: List[float],
    method: str = "spearman",
) -> Dict[str, float]:
    """
    计算人工评分与自动化指标的相关性

    参数:
        human_scores: 人工评分列表
        auto_scores: 自动化指标分数列表
        method: 相关性计算方法 (pearson, spearman, kendall)

    返回:
        相关系数和 p 值
    """
    if method == "pearson":
        corr, p_value = stats.pearsonr(human_scores, auto_scores)
    elif method == "spearman":
        corr, p_value = stats.spearmanr(human_scores, auto_scores)
    elif method == "kendall":
        corr, p_value = stats.kendalltau(human_scores, auto_scores)
    else:
        raise ValueError(f"Unknown method: {method}")

    return {
        "method": method,
        "correlation": float(corr),
        "p_value": float(p_value),
        "significant": p_value < 0.05,
        "interpretation": interpret_correlation(corr),
    }


def interpret_correlation(corr: float) -> str:
    """解读相关系数"""
    abs_corr = abs(corr)
    if abs_corr >= 0.8:
        strength = "强"
    elif abs_corr >= 0.5:
        strength = "中等"
    elif abs_corr >= 0.3:
        strength = "弱"
    else:
        strength = "极弱"

    direction = "正" if corr >= 0 else "负"
    return f"{strength}{direction}相关"


def analyze_metric_human_alignment(
    eval_results: List[Dict[str, Any]],
    human_dimension: str = "overall",
) -> Dict[str, Any]:
    """
    分析自动化指标与人工评估维度的对齐程度

    参数:
        eval_results: 评估结果列表，每条包含 auto_scores 和 human_scores
        human_dimension: 人工评估维度

    返回:
        各自动化指标与人工维度的相关性分析
    """
    if not eval_results:
        return {}

    auto_metrics = list(eval_results[0].get("auto_scores", {}).keys())
    alignments = {}

    for metric in auto_metrics:
        auto_vals = [r["auto_scores"].get(metric, 0) for r in eval_results]
        human_vals = [r["human_scores"].get(human_dimension, 0) for r in eval_results]

        correlation = compute_correlation(human_vals, auto_vals)
        alignments[metric] = {
            "correlation": correlation,
            "mean_auto": float(np.mean(auto_vals)),
            "mean_human": float(np.mean(human_vals)),
            "std_auto": float(np.std(auto_vals)),
            "std_human": float(np.std(human_vals)),
        }

    return alignments


def generate_alignment_report(alignments: Dict) -> str:
    """生成对齐分析报告"""
    lines = []
    lines.append("=" * 60)
    lines.append("  自动化指标 vs 人工评估 对齐分析报告")
    lines.append("=" * 60)
    lines.append("")
    lines.append(f"  {'指标':<20} {'相关系数':<12} {'显著性':<10} {'解读'}")
    lines.append("  " + "-" * 55)

    for metric, data in sorted(
        alignments.items(),
        key=lambda x: abs(x[1]["correlation"]["correlation"]),
        reverse=True,
    ):
        corr = data["correlation"]
        sig = "✅ 显著" if corr["significant"] else "❌ 不显著"
        lines.append(
            f"  {metric:<20} {corr['correlation']:<12.4f} {sig:<10} {corr['interpretation']}"
        )

    lines.append("")
    lines.append("  建议:")
    best_metric = max(
        alignments.items(),
        key=lambda x: abs(x[1]["correlation"]["correlation"]),
    )
    lines.append(f"  - 与人工评估最对齐的指标: {best_metric[0]}")
    lines.append(f"  - 相关系数: {best_metric[1]['correlation']['correlation']:.4f}")

    return "\n".join(lines)
```

## 7. 人类 vs 自动化评估对比

| 维度 | 人工评估 | 自动化评估 |
|------|---------|-----------|
| **成本** | 高（需要人力） | 低（API 调用费） |
| **速度** | 慢（每条需要几分钟） | 快（批量处理） |
| **一致性** | 受标注者主观影响 | 完全一致 |
| **覆盖维度** | 广（准确性、语气、帮助性等） | 窄（主要是准确性相关） |
| **可扩展性** | 差（人力有限） | 好（可处理百万级数据） |
| **解释性** | 高（有具体反馈） | 低（只有分数） |
| **适用场景** | 质量审核、金标准建立 | 日常监控、回归测试 |
| **更新频率** | 低（定期标注） | 高（每次代码变更后） |

## 9. 工程建议

1. **标注指南是标注质量的核心**：模糊的标注指南会导致标注员理解不一致。每个评估维度都要有明确的定义、评分标准和具体示例，尤其是边界情况的处理规则。
2. **每个样本至少两人标注**：单人标注的误差率通常在 15-20%。双人标注加仲裁能显著提高标注质量，同时通过 Inter-Annotator Agreement 量化标注一致性。
3. **标注工具要高效易用**：标注员每天要处理几百条数据，工具的效率直接影响标注进度和质量。Streamlit 或 Label Studio 都是不错的快速搭建方案。
4. **人工评估结果要反馈到自动评估**：用人工评估的标注数据来校准自动评估指标。如果 RAGAS 的 Faithfulness 分数与人工评分的相关性低于 0.7，说明自动评估需要调优。
