## 3. 反馈收集器

```python
"""
用户反馈收集器
收集显式反馈 (评分、点赞) 和隐式反馈 (停留时间、复制、分享)
"""

import json
import time
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Optional, Any
from datetime import datetime
from pathlib import Path
from collections import Counter
import hashlib


@dataclass
class FeedbackEvent:
    """反馈事件"""
    feedback_id: str
    user_id: str
    session_id: str
    query: str
    answer: str
    contexts: List[str] = field(default_factory=list)
    explicit_feedback: Dict[str, Any] = field(default_factory=dict)
    implicit_feedback: Dict[str, Any] = field(default_factory=dict)
    timestamp: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)

    def __post_init__(self):
        if not self.feedback_id:
            content = f"{self.user_id}:{self.query}:{self.timestamp}"
            self.feedback_id = hashlib.md5(content.encode()).hexdigest()[:12]
        if not self.timestamp:
            self.timestamp = datetime.now().isoformat()


class FeedbackCollector:
    """反馈收集器"""

    def __init__(self, storage_path: str = "feedback_data"):
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.events: List[FeedbackEvent] = []
        self._load_existing()

    def _load_existing(self) -> None:
        """加载已有反馈数据"""
        path = self.storage_path / "feedback.jsonl"
        if path.exists():
            with open(path, "r", encoding="utf-8") as f:
                for line in f:
                    data = json.loads(line.strip())
                    self.events.append(FeedbackEvent(**data))

    def collect_explicit(
        self,
        user_id: str,
        session_id: str,
        query: str,
        answer: str,
        rating: Optional[int] = None,
        thumbs_up: Optional[bool] = None,
        thumbs_down: Optional[bool] = None,
        feedback_text: Optional[str] = None,
        dimensions: Optional[Dict[str, int]] = None,
        contexts: Optional[List[str]] = None,
    ) -> FeedbackEvent:
        """收集显式反馈"""
        event = FeedbackEvent(
            feedback_id="",
            user_id=user_id,
            session_id=session_id,
            query=query,
            answer=answer,
            contexts=contexts or [],
            explicit_feedback={
                "rating": rating,
                "thumbs_up": thumbs_up,
                "thumbs_down": thumbs_down,
                "feedback_text": feedback_text,
                "dimensions": dimensions or {},
            },
        )
        self.events.append(event)
        self._persist(event)
        return event

    def collect_implicit(
        self,
        user_id: str,
        session_id: str,
        query: str,
        answer: str,
        dwell_time_seconds: float = 0,
        copied: bool = False,
        shared: bool = False,
        scrolled: bool = False,
        clicked_references: bool = False,
        asked_followup: bool = False,
        contexts: Optional[List[str]] = None,
    ) -> FeedbackEvent:
        """收集隐式反馈"""
        # 推断满意度
        inferred_satisfaction = self._infer_satisfaction(
            dwell_time_seconds, copied, shared, asked_followup
        )

        event = FeedbackEvent(
            feedback_id="",
            user_id=user_id,
            session_id=session_id,
            query=query,
            answer=answer,
            contexts=contexts or [],
            implicit_feedback={
                "dwell_time_seconds": dwell_time_seconds,
                "copied": copied,
                "shared": shared,
                "scrolled": scrolled,
                "clicked_references": clicked_references,
                "asked_followup": asked_followup,
                "inferred_satisfaction": inferred_satisfaction,
            },
        )
        self.events.append(event)
        self._persist(event)
        return event

    def _infer_satisfaction(
        self,
        dwell_time: float,
        copied: bool,
        shared: bool,
        asked_followup: bool,
    ) -> float:
        """基于隐式行为推断满意度 (0-1)"""
        score = 0.5  # 基线

        # 停留时间: 太短说明没看，太长说明没看懂
        if 10 < dwell_time < 120:
            score += 0.1
        elif dwell_time < 5:
            score -= 0.2

        if copied:
            score += 0.15  # 复制说明有用
        if shared:
            score += 0.1   # 分享说明质量高
        if asked_followup:
            score += 0.05  # 追问说明有兴趣

        return max(0.0, min(1.0, score))

    def _persist(self, event: FeedbackEvent) -> None:
        """持久化单条事件"""
        path = self.storage_path / "feedback.jsonl"
        with open(path, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(event), ensure_ascii=False) + "\n")

    def get_negative_feedback(
        self,
        rating_threshold: int = 2,
        limit: int = 50,
    ) -> List[FeedbackEvent]:
        """获取负面反馈"""
        negative = []
        for event in self.events:
            rating = event.explicit_feedback.get("rating")
            thumbs_down = event.explicit_feedback.get("thumbs_down")
            if (rating is not None and rating <= rating_threshold) or thumbs_down:
                negative.append(event)
        return negative[:limit]

    def get_statistics(self) -> Dict[str, Any]:
        """获取反馈统计"""
        total = len(self.events)
        if total == 0:
            return {"total": 0}

        ratings = [
            e.explicit_feedback["rating"]
            for e in self.events
            if e.explicit_feedback.get("rating") is not None
        ]
        thumbs_up = sum(
            1 for e in self.events
            if e.explicit_feedback.get("thumbs_up")
        )
        thumbs_down = sum(
            1 for e in self.events
            if e.explicit_feedback.get("thumbs_down")
        )

        return {
            "total_feedback": total,
            "explicit_ratings": len(ratings),
            "avg_rating": sum(ratings) / len(ratings) if ratings else None,
            "thumbs_up": thumbs_up,
            "thumbs_down": thumbs_down,
            "satisfaction_rate": thumbs_up / max(thumbs_up + thumbs_down, 1),
        }


# ============================================================
# 使用示例
# ============================================================

if __name__ == "__main__":
    collector = FeedbackCollector("test_feedback")

    # 模拟收集反馈
    collector.collect_explicit(
        user_id="user_001",
        session_id="sess_001",
        query="什么是 RAG？",
        answer="RAG 是检索增强生成技术...",
        rating=4,
        thumbs_up=True,
    )

    collector.collect_implicit(
        user_id="user_002",
        session_id="sess_002",
        query="如何优化检索？",
        answer="可以通过混合检索...",
        dwell_time_seconds=45,
        copied=True,
    )

    stats = collector.get_statistics()
    print(f"反馈统计: {json.dumps(stats, indent=2, ensure_ascii=False)}")
```

## 4. 问题分析器

```python
"""
反馈问题分析器
自动分类问题类型，生成改进建议
"""

import json
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any
from collections import Counter, defaultdict
from enum import Enum


class IssueCategory(Enum):
    """问题类别"""
    HALLUCINATION = "hallucination"         # 幻觉: 回答包含不存在的信息
    IRRELEVANT = "irrelevant"               # 不相关: 答非所问
    INCOMPLETE = "incomplete"               # 不完整: 遗漏关键信息
    OUTDATED = "outdated"                   # 过时: 信息已过期
    WRONG_FORMAT = "wrong_format"           # 格式差: 回答格式不佳
    SLOW_RESPONSE = "slow_response"         # 响应慢: 耗时太长
    RETRIEVAL_MISS = "retrieval_miss"       # 检索失败: 没检索到相关文档
    KNOWLEDGE_GAP = "knowledge_gap"         # 知识缺失: 知识库中没有相关信息


@dataclass
class IssueReport:
    """问题报告"""
    issue_id: str
    category: str
    severity: str          # high, medium, low
    query: str
    answer: str
    description: str
    suggested_fix: str
    frequency: int = 1
    examples: List[str] = field(default_factory=list)


class IssueAnalyzer:
    """问题分析器"""

    def __init__(self, llm_client: Optional[Any] = None):
        self.llm_client = llm_client
        self.issue_patterns: Dict[str, List[Dict]] = defaultdict(list)

    def analyze_feedback_batch(
        self,
        feedback_events: List[Any],
    ) -> List[IssueReport]:
        """批量分析反馈，生成问题报告"""
        issues = []

        for event in feedback_events:
            # 判断是否为负面反馈
            is_negative = self._is_negative(event)
            if not is_negative:
                continue

            # 分类问题
            category, severity = self._classify_issue(event)

            # 生成修复建议
            fix = self._suggest_fix(category, event)

            report = IssueReport(
                issue_id=event.feedback_id,
                category=category.value,
                severity=severity,
                query=event.query,
                answer=event.answer,
                description=self._generate_description(category, event),
                suggested_fix=fix,
            )
            issues.append(report)
            self.issue_patterns[category.value].append({
                "query": event.query,
                "feedback_id": event.feedback_id,
            })

        return issues

    def _is_negative(self, event: Any) -> bool:
        """判断是否为负面反馈"""
        rating = event.explicit_feedback.get("rating")
        thumbs_down = event.explicit_feedback.get("thumbs_down")
        inferred = event.implicit_feedback.get("inferred_satisfaction", 0.5)

        if rating is not None and rating <= 2:
            return True
        if thumbs_down:
            return True
        if inferred < 0.3:
            return True
        return False

    def _classify_issue(self, event: Any) -> tuple:
        """对问题进行分类"""
        query = event.query.lower()
        answer = event.answer.lower()
        contexts = event.contexts
        feedback_text = event.explicit_feedback.get("feedback_text", "")

        # 基于规则的简单分类
        if not contexts or all(len(c.strip()) < 10 for c in contexts):
            return IssueCategory.RETRIEVAL_MISS, "high"

        if "不知道" in answer or "无法回答" in answer:
            return IssueCategory.KNOWLEDGE_GAP, "medium"

        # 检查是否答非所问
        query_words = set(query.split())
        answer_words = set(answer.split())
        overlap = len(query_words & answer_words) / max(len(query_words), 1)
        if overlap < 0.1:
            return IssueCategory.IRRELEVANT, "high"

        # 检查回答长度
        if len(answer) < 50:
            return IssueCategory.INCOMPLETE, "medium"

        # 检查是否包含幻觉标记
        if "可能" in answer or "据说" in answer or "也许" in answer:
            return IssueCategory.HALLUCINATION, "medium"

        # 默认
        return IssueCategory.INCOMPLETE, "low"

    def _suggest_fix(self, category: IssueCategory, event: Any) -> str:
        """根据问题类型生成修复建议"""
        fixes = {
            IssueCategory.HALLUCINATION: "优化 Prompt，要求模型只基于上下文回答；增加 Faithfulness 检查",
            IssueCategory.IRRELEVANT: "检查检索质量，优化查询改写；调整 Prompt 明确回答范围",
            IssueCategory.INCOMPLETE: "优化分块策略确保信息完整；增加检索数量 (top_k)",
            IssueCategory.OUTDATED: "更新知识库中的过时文档；建立定期更新机制",
            IssueCategory.WRONG_FORMAT: "在 Prompt 中明确格式要求；添加输出格式化步骤",
            IssueCategory.SLOW_RESPONSE: "优化检索性能；考虑使用更快的模型；添加缓存",
            IssueCategory.RETRIEVAL_MISS: "优化嵌入模型；添加关键词检索作为补充；扩展知识库",
            IssueCategory.KNOWLEDGE_GAP: "补充缺失知识到知识库；添加兜底回答策略",
        }
        return fixes.get(category, "需要进一步分析")

    def _generate_description(self, category: IssueCategory, event: Any) -> str:
        """生成问题描述"""
        return f"用户查询 '{event.query[:50]...' 时，系统回答被标记为 {category.value}"

    def get_top_issues(self, n: int = 10) -> List[Dict[str, Any]]:
        """获取最高频问题"""
        issue_counts = []
        for category, occurrences in self.issue_patterns.items():
            issue_counts.append({
                "category": category,
                "count": len(occurrences),
                "sample_queries": [o["query"] for o in occurrences[:3]],
            })
        return sorted(issue_counts, key=lambda x: x["count"], reverse=True)[:n]

    def generate_improvement_plan(self) -> str:
        """生成改进计划"""
        top_issues = self.get_top_issues(5)
        lines = []
        lines.append("=" * 60)
        lines.append("  RAG 系统改进计划")
        lines.append("=" * 60)

        for i, issue in enumerate(top_issues, 1):
            lines.append(f"\n  {i}. {issue['category']} (出现 {issue['count']} 次)")
            lines.append(f"     示例查询: {issue['sample_queries'][0] if issue['sample_queries'] else 'N/A'}")

        return "\n".join(lines)
```

## 5. 知识库更新器

```python
"""
知识库更新管理器
支持增量更新、版本控制和回滚
"""

import json
import hashlib
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Optional, Any
from datetime import datetime
from pathlib import Path
from enum import Enum


class UpdateAction(Enum):
    """更新动作"""
    ADD = "add"
    UPDATE = "update"
    DELETE = "delete"
    REINDEX = "reindex"


@dataclass
class UpdateRecord:
    """更新记录"""
    update_id: str
    action: str
    doc_id: str
    doc_content: str
    reason: str
    timestamp: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)
    status: str = "pending"    # pending, applied, rolled_back

    def __post_init__(self):
        if not self.timestamp:
            self.timestamp = datetime.now().isoformat()
        if not self.update_id:
            content = f"{self.action}:{self.doc_id}:{self.timestamp}"
            self.update_id = hashlib.md5(content.encode()).hexdigest()[:12]


class KnowledgeBaseUpdater:
    """知识库更新管理器"""

    def __init__(
        self,
        vector_store: Any,
        storage_path: str = "kb_updates",
    ):
        self.vector_store = vector_store
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.update_history: List[UpdateRecord] = []
        self._load_history()

    def _load_history(self) -> None:
        """加载更新历史"""
        path = self.storage_path / "update_history.jsonl"
        if path.exists():
            with open(path, "r", encoding="utf-8") as f:
                for line in f:
                    data = json.loads(line.strip())
                    self.update_history.append(UpdateRecord(**data))

    def add_document(
        self,
        doc_id: str,
        content: str,
        metadata: Optional[Dict] = None,
        reason: str = "new_document",
    ) -> UpdateRecord:
        """添加新文档"""
        record = UpdateRecord(
            update_id="",
            action=UpdateAction.ADD.value,
            doc_id=doc_id,
            doc_content=content,
            reason=reason,
            metadata=metadata or {},
        )

        # 执行更新
        try:
            self.vector_store.add_document(doc_id, content, metadata)
            record.status = "applied"
        except Exception as e:
            record.status = "failed"
            record.metadata["error"] = str(e)

        self.update_history.append(record)
        self._persist(record)
        return record

    def update_document(
        self,
        doc_id: str,
        new_content: str,
        reason: str = "content_update",
    ) -> UpdateRecord:
        """更新已有文档"""
        record = UpdateRecord(
            update_id="",
            action=UpdateAction.UPDATE.value,
            doc_id=doc_id,
            doc_content=new_content,
            reason=reason,
        )

        try:
            self.vector_store.update_document(doc_id, new_content)
            record.status = "applied"
        except Exception as e:
            record.status = "failed"
            record.metadata["error"] = str(e)

        self.update_history.append(record)
        self._persist(record)
        return record

    def delete_document(
        self,
        doc_id: str,
        reason: str = "content_removed",
    ) -> UpdateRecord:
        """删除文档"""
        record = UpdateRecord(
            update_id="",
            action=UpdateAction.DELETE.value,
            doc_id=doc_id,
            doc_content="",
            reason=reason,
        )

        try:
            self.vector_store.delete_document(doc_id)
            record.status = "applied"
        except Exception as e:
            record.status = "failed"
            record.metadata["error"] = str(e)

        self.update_history.append(record)
        self._persist(record)
        return record

    def batch_update(
        self,
        updates: List[Dict[str, Any]],
    ) -> List[UpdateRecord]:
        """批量更新"""
        records = []
        for update in updates:
            action = update.get("action", "add")
            if action == "add":
                record = self.add_document(
                    update["doc_id"],
                    update["content"],
                    update.get("metadata"),
                    update.get("reason", "batch_add"),
                )
            elif action == "update":
                record = self.update_document(
                    update["doc_id"],
                    update["content"],
                    update.get("reason", "batch_update"),
                )
            elif action == "delete":
                record = self.delete_document(
                    update["doc_id"],
                    update.get("reason", "batch_delete"),
                )
            else:
                continue
            records.append(record)
        return records

    def rollback(self, update_id: str) -> bool:
        """回滚指定更新"""
        for record in self.update_history:
            if record.update_id == update_id and record.status == "applied":
                if record.action == UpdateAction.ADD.value:
                    self.vector_store.delete_document(record.doc_id)
                    record.status = "rolled_back"
                    return True
                elif record.action == UpdateAction.DELETE.value:
                    self.vector_store.add_document(
                        record.doc_id,
                        record.doc_content,
                        record.metadata,
                    )
                    record.status = "rolled_back"
                    return True
        return False

    def _persist(self, record: UpdateRecord) -> None:
        """持久化更新记录"""
        path = self.storage_path / "update_history.jsonl"
        with open(path, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(record), ensure_ascii=False) + "\n")

    def get_statistics(self) -> Dict[str, Any]:
        """获取更新统计"""
        total = len(self.update_history)
        by_action = Counter(r.action for r in self.update_history)
        by_status = Counter(r.status for r in self.update_history)

        return {
            "total_updates": total,
            "by_action": dict(by_action),
            "by_status": dict(by_status),
        }
```

## 6. Pipeline 编排器

```python
"""
持续优化 Pipeline 编排器
将反馈收集、问题分析、知识库更新串联为自动化流程
"""

import json
import time
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any, Callable
from datetime import datetime
from pathlib import Path
from enum import Enum


class PipelineStage(Enum):
    """Pipeline 阶段"""
    COLLECT = "collect"
    ANALYZE = "analyze"
    PLAN = "plan"
    EXECUTE = "execute"
    VERIFY = "verify"


@dataclass
class PipelineRun:
    """Pipeline 运行记录"""
    run_id: str
    started_at: str
    completed_at: str = ""
    status: str = "running"
    stages_completed: List[str] = field(default_factory=list)
    metrics: Dict[str, Any] = field(default_factory=dict)
    errors: List[str] = field(default_factory=list)


class OptimizationPipeline:
    """持续优化 Pipeline"""

    def __init__(
        self,
        feedback_collector: Any,
        issue_analyzer: Any,
        kb_updater: Any,
        eval_pipeline: Optional[Any] = None,
        config: Optional[Dict] = None,
    ):
        self.feedback_collector = feedback_collector
        self.issue_analyzer = issue_analyzer
        self.kb_updater = kb_updater
        self.eval_pipeline = eval_pipeline
        self.config = config or {
            "min_negative_feedback": 10,
            "auto_update_enabled": False,
            "verification_required": True,
        }
        self.run_history: List[PipelineRun] = []

    def run(self) -> PipelineRun:
        """执行完整优化 Pipeline"""
        run = PipelineRun(
            run_id=f"opt_{int(time.time())}",
            started_at=datetime.now().isoformat(),
        )

        try:
            # Stage 1: 收集反馈
            print("[Pipeline] Stage 1/5: 收集反馈...")
            feedback_stats = self._stage_collect()
            run.stages_completed.append("collect")
            run.metrics["feedback"] = feedback_stats

            # Stage 2: 分析问题
            print("[Pipeline] Stage 2/5: 分析问题...")
            issues = self._stage_analyze()
            run.stages_completed.append("analyze")
            run.metrics["issues_found"] = len(issues)

            # Stage 3: 生成改进计划
            print("[Pipeline] Stage 3/5: 生成改进计划...")
            plan = self._stage_plan(issues)
            run.stages_completed.append("plan")
            run.metrics["plan"] = plan

            # Stage 4: 执行改进
            print("[Pipeline] Stage 4/5: 执行改进...")
            if self.config.get("auto_update_enabled"):
                updates = self._stage_execute(plan)
                run.stages_completed.append("execute")
                run.metrics["updates_applied"] = len(updates)
            else:
                run.metrics["updates_applied"] = 0
                run.metrics["note"] = "自动更新已禁用，需手动审批"

            # Stage 5: 验证效果
            if self.config.get("verification_required") and self.eval_pipeline:
                print("[Pipeline] Stage 5/5: 验证改进效果...")
                verification = self._stage_verify()
                run.stages_completed.append("verify")
                run.metrics["verification"] = verification

            run.status = "completed"
            run.completed_at = datetime.now().isoformat()

        except Exception as e:
            run.status = "failed"
            run.errors.append(str(e))

        self.run_history.append(run)
        return run

    def _stage_collect(self) -> Dict[str, Any]:
        """收集阶段"""
        stats = self.feedback_collector.get_statistics()
        negative = self.feedback_collector.get_negative_feedback()
        return {
            "total_feedback": stats.get("total_feedback", 0),
            "negative_count": len(negative),
            "satisfaction_rate": stats.get("satisfaction_rate", 0),
        }

    def _stage_analyze(self) -> List[Any]:
        """分析阶段"""
        negative_feedback = self.feedback_collector.get_negative_feedback()
        issues = self.issue_analyzer.analyze_feedback_batch(negative_feedback)
        return issues

    def _stage_plan(self, issues: List[Any]) -> Dict[str, Any]:
        """计划阶段"""
        plan = {
            "total_issues": len(issues),
            "by_category": {},
            "actions": [],
        }

        for issue in issues:
            cat = issue.category
            if cat not in plan["by_category"]:
                plan["by_category"][cat] = 0
            plan["by_category"][cat] += 1

            plan["actions"].append({
                "category": cat,
                "severity": issue.severity,
                "fix": issue.suggested_fix,
                "query": issue.query,
            })

        return plan

    def _stage_execute(self, plan: Dict[str, Any]) -> List[Any]:
        """执行阶段"""
        updates = []
        for action in plan.get("actions", []):
            if action["category"] in ("retrieval_miss", "knowledge_gap"):
                # 需要补充知识库
                updates.append({
                    "action": "add",
                    "doc_id": f"auto_{action['category']}_{int(time.time())}",
                    "content": action.get("fix", ""),
                    "reason": f"自动补充: {action['category']}",
                })

        if updates:
            self.kb_updater.batch_update(updates)

        return updates

    def _stage_verify(self) -> Dict[str, Any]:
        """验证阶段"""
        if not self.eval_pipeline:
            return {"status": "no_eval_pipeline"}

        # 运行评估
        # report = self.eval_pipeline.run(...)
        return {"status": "verification_skipped", "reason": "需要集成评估 Pipeline"}

    def get_latest_run(self) -> Optional[PipelineRun]:
        """获取最近一次运行"""
        return self.run_history[-1] if self.run_history else None


# ============================================================
# 使用示例
# ============================================================

if __name__ == "__main__":
    print("Pipeline 编排器示例 (需要集成各组件)")
    print("  1. FeedbackCollector: 收集用户反馈")
    print("  2. IssueAnalyzer: 分析问题类型")
    print("  3. KnowledgeBaseUpdater: 更新知识库")
    print("  4. OptimizationPipeline: 编排整个流程")
```

## 7. 手动 vs 半自动 vs 全自动优化对比

| 维度 | 手动优化 | 半自动优化 | 全自动优化 |
|------|---------|-----------|-----------|
| **触发方式** | 人工定期检查 | 自动检测 + 人工审批 | 全自动触发 |
| **问题发现** | 人工阅读反馈 | 自动分类 + 人工确认 | 自动分类和优先级排序 |
| **改进执行** | 手动修改代码/配置 | 自动生成方案 + 人工执行 | 自动执行改进 |
| **验证方式** | 人工验证 | 自动评估 + 人工确认 | 自动评估和回滚 |
| **适用阶段** | 早期探索期 | 成长期 | 成熟期 |
| **风险** | 低 | 中 | 需要完善的安全机制 |
| **效率** | 低 | 中 | 高 |
| **人力成本** | 高 | 中 | 低 |

## 8. 常见错误

```
┌─────────────────────────────────────────────────────────────────┐
│                       常见错误                                  │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  ❌ 错误1: 只收集不分析                                          │
│     堆积大量反馈数据但从不分析                                   │
│     ✅ 正确: 建立定期分析机制，至少每周一次                       │
│                                                                 │
│  ❌ 错误2: 过度依赖自动化                                        │
│     全自动更新可能导致知识库质量下降                              │
│     ✅ 正确: 关键更新需要人工审批                                 │
│                                                                 │
│  ❌ 错误3: 不验证改进效果                                        │
│     更新后不评估就认为已经改善                                    │
│     ✅ 正确: 每次更新前后都要评估对比                             │
│                                                                 │
│  ❌ 错误4: 忽略数据质量                                          │
│     垃圾反馈数据会导致错误的优化方向                              │
│     ✅ 正确: 过滤 spam，验证反馈有效性                            │
│                                                                 │
│  ❌ 错误5: 知识库更新不版本化                                    │
│     出问题时无法回滚                                             │
│     ✅ 正确: 记录每次更新，支持回滚                               │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```
