## 3. 流量分配器实现

```python
"""
A/B 测试流量分配器
支持基于 user_id 的确定性分流
"""

import hashlib
import json
import time
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any
from datetime import datetime
from pathlib import Path


@dataclass
class ExperimentConfig:
    """实验配置"""
    experiment_id: str
    name: str
    description: str
    variants: List[Dict[str, Any]]  # [{"id": "control", "weight": 50}, {"id": "treatment", "weight": 50}]
    start_time: Optional[str] = None
    end_time: Optional[str] = None
    enabled: bool = True
    whitelist: List[str] = field(default_factory=list)
    blacklist: List[str] = field(default_factory=list)
    metadata: Dict[str, Any] = field(default_factory=dict)


class TrafficSplitter:
    """流量分配器"""

    def __init__(self):
        self.experiments: Dict[str, ExperimentConfig] = {}
        self.assignment_log: List[Dict] = []

    def register_experiment(self, config: ExperimentConfig) -> None:
        """注册实验"""
        total_weight = sum(v["weight"] for v in config.variants)
        if abs(total_weight - 100) > 0.01:
            raise ValueError(f"变体权重之和必须为 100，当前为 {total_weight}")
        self.experiments[config.experiment_id] = config

    def assign(
        self,
        user_id: str,
        experiment_id: str,
    ) -> Dict[str, Any]:
        """为用户分配实验变体"""
        exp = self.experiments.get(experiment_id)
        if not exp or not exp.enabled:
            return {"variant_id": "control", "reason": "experiment_not_found"}

        # 白名单检查
        if exp.whitelist and user_id in exp.whitelist:
            variant = exp.variants[-1]  # 白名单用户进入实验组
            return {"variant_id": variant["id"], "reason": "whitelist"}

        # 黑名单检查
        if exp.blacklist and user_id in exp.blacklist:
            return {"variant_id": "control", "reason": "blacklist"}

        # 基于 hash 的确定性分流
        hash_input = f"{experiment_id}:{user_id}"
        hash_value = int(hashlib.md5(hash_input.encode()).hexdigest(), 16)
        bucket = hash_value % 100

        # 分配变体
        cumulative = 0
        for variant in exp.variants:
            cumulative += variant["weight"]
            if bucket < cumulative:
                result = {
                    "variant_id": variant["id"],
                    "experiment_id": experiment_id,
                    "user_id": user_id,
                    "bucket": bucket,
                    "timestamp": datetime.now().isoformat(),
                }
                self.assignment_log.append(result)
                return result

        # fallback
        return {"variant_id": "control", "reason": "fallback"}

    def get_variant_config(
        self,
        user_id: str,
        experiment_id: str,
    ) -> Dict[str, Any]:
        """获取用户对应的变体配置"""
        assignment = self.assign(user_id, experiment_id)
        exp = self.experiments.get(experiment_id)
        if not exp:
            return {}

        variant_id = assignment["variant_id"]
        for variant in exp.variants:
            if variant["id"] == variant_id:
                return {
                    "variant_id": variant_id,
                    "config": variant.get("config", {}),
                    **assignment,
                }
        return assignment

    def get_assignment_stats(self, experiment_id: str) -> Dict[str, Any]:
        """获取分配统计"""
        exp_assignments = [
            a for a in self.assignment_log
            if a.get("experiment_id") == experiment_id
        ]

        variant_counts = {}
        for a in exp_assignments:
            vid = a["variant_id"]
            variant_counts[vid] = variant_counts.get(vid, 0) + 1

        return {
            "experiment_id": experiment_id,
            "total_assignments": len(exp_assignments),
            "variant_distribution": variant_counts,
        }


# ============================================================
# 使用示例
# ============================================================

if __name__ == "__main__":
    splitter = TrafficSplitter()

    # 注册一个 A/B 测试
    config = ExperimentConfig(
        experiment_id="rag_prompt_v2",
        name="Prompt 优化实验",
        description="测试新的 system prompt 是否提升回答质量",
        variants=[
            {
                "id": "control",
                "weight": 50,
                "config": {"prompt_template": "default_v1"},
            },
            {
                "id": "treatment",
                "weight": 50,
                "config": {"prompt_template": "optimized_v2"},
            },
        ],
    )
    splitter.register_experiment(config)

    # 模拟用户分流
    for i in range(20):
        user_id = f"user_{i:04d}"
        result = splitter.assign(user_id, "rag_prompt_v2")
        print(f"  {user_id} → {result['variant_id']} (bucket={result['bucket']})")

    # 查看分配统计
    stats = splitter.get_assignment_stats("rag_prompt_v2")
    print(f"\n分配统计: {json.dumps(stats, indent=2)}")
```

## 4. 指标收集器

```python
"""
A/B 测试指标收集器
收集显式和隐式用户反馈
"""

import json
import time
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any
from datetime import datetime
from pathlib import Path
from collections import defaultdict


@dataclass
class MetricEvent:
    """指标事件"""
    user_id: str
    experiment_id: str
    variant_id: str
    event_type: str       # "explicit_rating", "implicit_feedback", "system_metric"
    event_name: str       # "thumbs_up", "dwell_time", "latency", etc.
    value: float
    timestamp: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)

    def __post_init__(self):
        if not self.timestamp:
            self.timestamp = datetime.now().isoformat()


class MetricCollector:
    """指标收集器"""

    def __init__(self, storage_path: str = "ab_metrics"):
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.events: List[MetricEvent] = []
        self._counters: Dict[str, Dict[str, List[float]]] = defaultdict(lambda: defaultdict(list))

    def track(
        self,
        user_id: str,
        experiment_id: str,
        variant_id: str,
        event_type: str,
        event_name: str,
        value: float,
        metadata: Optional[Dict] = None,
    ) -> None:
        """记录一个指标事件"""
        event = MetricEvent(
            user_id=user_id,
            experiment_id=experiment_id,
            variant_id=variant_id,
            event_type=event_type,
            event_name=event_name,
            value=value,
            metadata=metadata or {},
        )
        self.events.append(event)

        key = f"{experiment_id}:{variant_id}"
        self._counters[key][event_name].append(value)

    def track_rating(
        self,
        user_id: str,
        experiment_id: str,
        variant_id: str,
        rating: int,
        dimension: str = "overall",
    ) -> None:
        """记录用户评分"""
        self.track(
            user_id=user_id,
            experiment_id=experiment_id,
            variant_id=variant_id,
            event_type="explicit_rating",
            event_name=f"rating_{dimension}",
            value=float(rating),
        )

    def track_implicit(
        self,
        user_id: str,
        experiment_id: str,
        variant_id: str,
        dwell_time: float,
        copied: bool = False,
        shared: bool = False,
    ) -> None:
        """记录隐式反馈"""
        self.track(user_id, experiment_id, variant_id, "implicit", "dwell_time", dwell_time)
        if copied:
            self.track(user_id, experiment_id, variant_id, "implicit", "copy_action", 1.0)
        if shared:
            self.track(user_id, experiment_id, variant_id, "implicit", "share_action", 1.0)

    def track_system(
        self,
        experiment_id: str,
        variant_id: str,
        latency_ms: float,
        error: bool = False,
    ) -> None:
        """记录系统指标"""
        self.track("system", experiment_id, variant_id, "system", "latency_ms", latency_ms)
        if error:
            self.track("system", experiment_id, variant_id, "system", "error", 1.0)

    def get_variant_metrics(
        self,
        experiment_id: str,
        event_name: str,
    ) -> Dict[str, Dict[str, float]]:
        """获取各变体的指标统计"""
        result = {}
        for key, events in self._counters.items():
            exp_id, variant_id = key.split(":", 1)
            if exp_id != experiment_id:
                continue
            values = events.get(event_name, [])
            if not values:
                continue
            import numpy as np
            result[variant_id] = {
                "count": len(values),
                "mean": float(np.mean(values)),
                "std": float(np.std(values)),
                "median": float(np.median(values)),
                "min": float(np.min(values)),
                "max": float(np.max(values)),
            }
        return result

    def save(self) -> None:
        """持久化事件数据"""
        path = self.storage_path / "events.jsonl"
        with open(path, "a", encoding="utf-8") as f:
            for event in self.events:
                f.write(json.dumps(event.__dict__, ensure_ascii=False) + "\n")
        self.events.clear()

    def load(self) -> None:
        """加载历史事件"""
        path = self.storage_path / "events.jsonl"
        if not path.exists():
            return
        with open(path, "r", encoding="utf-8") as f:
            for line in f:
                data = json.loads(line.strip())
                event = MetricEvent(**data)
                self.events.append(event)
                key = f"{event.experiment_id}:{event.variant_id}"
                self._counters[key][event.event_name].append(event.value)


# ============================================================
# 使用示例
# ============================================================

if __name__ == "__main__":
    import numpy as np

    collector = MetricCollector()

    # 模拟数据收集
    np.random.seed(42)
    for i in range(100):
        user_id = f"user_{i:04d}"
        variant = "control" if i % 2 == 0 else "treatment"

        # 评分 (treatment 组略高)
        base = 3.5 if variant == "control" else 4.0
        rating = int(np.clip(np.random.normal(base, 0.8), 1, 5))
        collector.track_rating(user_id, "exp_001", variant, rating)

        # 停留时间
        dwell = np.random.normal(30 if variant == "control" else 45, 10)
        collector.track_implicit(user_id, "exp_001", variant, max(dwell, 5))

    # 查看结果
    rating_stats = collector.get_variant_metrics("exp_001", "rating_overall")
    print("评分统计:")
    for v, s in rating_stats.items():
        print(f"  {v}: mean={s['mean']:.2f}, std={s['std']:.2f}, n={s['count']}")

    dwell_stats = collector.get_variant_metrics("exp_001", "dwell_time")
    print("\n停留时间统计:")
    for v, s in dwell_stats.items():
        print(f"  {v}: mean={s['mean']:.1f}s, std={s['std']:.1f}s, n={s['count']}")
```

## 5. 统计显著性计算

```python
"""
A/B 测试统计显著性计算
支持 t-test, 卡方检验, Bootstrap 置信区间
"""

import numpy as np
from typing import List, Dict, Any, Tuple
from dataclasses import dataclass
from scipy import stats


@dataclass
class TestResult:
    """统计检验结果"""
    metric_name: str
    control_mean: float
    treatment_mean: float
    lift: float              # 相对提升百分比
    p_value: float
    confidence_level: float
    is_significant: bool
    confidence_interval: Tuple[float, float]
    sample_size_control: int
    sample_size_treatment: int
    test_method: str
    interpretation: str


def t_test(
    control: List[float],
    treatment: List[float],
    confidence_level: float = 0.95,
    metric_name: str = "metric",
) -> TestResult:
    """
    独立样本 t 检验

    参数:
        control: 对照组数据
        treatment: 实验组数据
        confidence_level: 置信水平
        metric_name: 指标名称

    返回:
        TestResult 对象
    """
    alpha = 1 - confidence_level
    t_stat, p_value = stats.ttest_ind(control, treatment, equal_var=False)

    control_mean = float(np.mean(control))
    treatment_mean = float(np.mean(treatment))
    lift = (treatment_mean - control_mean) / control_mean * 100 if control_mean != 0 else 0

    # 计算置信区间
    diff = treatment_mean - control_mean
    se = np.sqrt(np.var(control) / len(control) + np.var(treatment) / len(treatment))
    ci_lower = diff - stats.t.ppf(1 - alpha / 2, len(control) + len(treatment) - 2) * se
    ci_upper = diff + stats.t.ppf(1 - alpha / 2, len(control) + len(treatment) - 2) * se

    is_sig = p_value < alpha

    if is_sig and lift > 0:
        interpretation = f"实验组显著优于对照组 (提升 {lift:.1f}%)"
    elif is_sig and lift < 0:
        interpretation = f"实验组显著差于对照组 (下降 {abs(lift):.1f}%)"
    else:
        interpretation = "差异不显著，无法拒绝零假设"

    return TestResult(
        metric_name=metric_name,
        control_mean=control_mean,
        treatment_mean=treatment_mean,
        lift=lift,
        p_value=float(p_value),
        confidence_level=confidence_level,
        is_significant=is_sig,
        confidence_interval=(float(ci_lower), float(ci_upper)),
        sample_size_control=len(control),
        sample_size_treatment=len(treatment),
        test_method="welch_t_test",
        interpretation=interpretation,
    )


def chi_square_test(
    control_success: int,
    control_total: int,
    treatment_success: int,
    treatment_total: int,
    confidence_level: float = 0.95,
    metric_name: str = "conversion",
) -> TestResult:
    """
    卡方检验 (适用于二元指标，如转化率)

    参数:
        control_success: 对照组成功数
        control_total: 对照组总数
        treatment_success: 实验组成功数
        treatment_total: 实验组总数
    """
    alpha = 1 - confidence_level

    contingency = np.array([
        [control_success, control_total - control_success],
        [treatment_success, treatment_total - treatment_success],
    ])
    chi2, p_value, dof, expected = stats.chi2_contingency(contingency)

    control_rate = control_success / control_total
    treatment_rate = treatment_success / treatment_total
    lift = (treatment_rate - control_rate) / control_rate * 100 if control_rate != 0 else 0

    # 置信区间 (使用正态近似)
    se = np.sqrt(
        control_rate * (1 - control_rate) / control_total +
        treatment_rate * (1 - treatment_rate) / treatment_total
    )
    diff = treatment_rate - control_rate
    z = stats.norm.ppf(1 - alpha / 2)
    ci_lower = diff - z * se
    ci_upper = diff + z * se

    is_sig = p_value < alpha

    return TestResult(
        metric_name=metric_name,
        control_mean=control_rate,
        treatment_mean=treatment_rate,
        lift=lift,
        p_value=float(p_value),
        confidence_level=confidence_level,
        is_significant=is_sig,
        confidence_interval=(float(ci_lower), float(ci_upper)),
        sample_size_control=control_total,
        sample_size_treatment=treatment_total,
        test_method="chi_square",
        interpretation=f"转化率 {'显著' if is_sig else '不显著'}变化: {control_rate:.2%} → {treatment_rate:.2%}",
    )


def bootstrap_confidence_interval(
    control: List[float],
    treatment: List[float],
    n_bootstrap: int = 10000,
    confidence_level: float = 0.95,
) -> Dict[str, float]:
    """
    Bootstrap 置信区间估计
    适用于分布未知或样本量较小的情况
    """
    np.random.seed(42)
    control = np.array(control)
    treatment = np.array(treatment)

    diffs = []
    for _ in range(n_bootstrap):
        c_sample = np.random.choice(control, size=len(control), replace=True)
        t_sample = np.random.choice(treatment, size=len(treatment), replace=True)
        diffs.append(np.mean(t_sample) - np.mean(c_sample))

    alpha = 1 - confidence_level
    ci_lower = np.percentile(diffs, 100 * alpha / 2)
    ci_upper = np.percentile(diffs, 100 * (1 - alpha / 2))

    return {
        "mean_diff": float(np.mean(diffs)),
        "ci_lower": float(ci_lower),
        "ci_upper": float(ci_upper),
        "confidence_level": confidence_level,
        "significant": ci_lower > 0 or ci_upper < 0,
    }


def calculate_sample_size(
    baseline_rate: float,
    minimum_detectable_effect: float,
    alpha: float = 0.05,
    power: float = 0.80,
) -> int:
    """
    计算所需样本量

    参数:
        baseline_rate: 基线指标值 (如当前满意度 0.7)
        minimum_detectable_effect: 最小可检测效应 (如 0.05 表示 5% 提升)
        alpha: 显著性水平
        power: 统计功效

    返回:
        每组所需样本量
    """
    effect_size = minimum_detectable_effect / np.sqrt(baseline_rate * (1 - baseline_rate))
    z_alpha = stats.norm.ppf(1 - alpha / 2)
    z_beta = stats.norm.ppf(power)

    n = ((z_alpha + z_beta) ** 2 * 2) / (effect_size ** 2)
    return int(np.ceil(n))

if __name__ == "__main__":

    # 模拟数据

    print(f"提升: {result.lift:.1f}%")
    print(f"p 值: {result.p_value:.4f}")
    print(f"显著: {result.is_significant}")
    print(f"解读: {result.interpretation}")

    # 卡方检验
    chi_result = chi_square_test(
```

```python
"""
灰度发布控制器
支持渐进式流量扩大和自动回滚
"""

import json
import time
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any, Callable
from datetime import datetime
from enum import Enum
from pathlib import Path


class RolloutStage(Enum):
    """灰度阶段"""
    CANARY = "canary"          # 金丝雀: 1-5% 流量
    SMALL = "small"            # 小流量: 5-20% 流量
    MEDIUM = "medium"          # 中流量: 20-50% 流量
    LARGE = "large"            # 大流量: 50-90% 流量
    FULL = "full"              # 全量: 100% 流量


@dataclass
class RolloutConfig:
    """灰度发布配置"""
    experiment_id: str
    stages: List[Dict[str, Any]] = field(default_factory=lambda: [
        {"stage": "canary", "traffic_pct": 5, "duration_hours": 2, "auto_promote": True},
        {"stage": "small", "traffic_pct": 20, "duration_hours": 12, "auto_promote": True},
        {"stage": "medium", "traffic_pct": 50, "duration_hours": 24, "auto_promote": True},
        {"stage": "large", "traffic_pct": 90, "duration_hours": 48, "auto_promote": True},
        {"stage": "full", "traffic_pct": 100, "duration_hours": 0, "auto_promote": False},
    ])
    rollback_thresholds: Dict[str, float] = field(default_factory=lambda: {
        "error_rate_max": 0.05,
        "latency_p99_max_ms": 5000,
        "satisfaction_min": 3.0,
    })
    check_interval_minutes: int = 30


@dataclass
class RolloutState:
    """灰度发布状态"""
    experiment_id: str
    current_stage_idx: int = 0
    current_traffic_pct: int = 0
    started_at: str = ""
    last_check_at: str = ""
    stage_started_at: str = ""
    total_requests: int = 0
    total_errors: int = 0
    status: str = "pending"    # pending, running, paused, completed, rolled_back
    history: List[Dict] = field(default_factory=list)


class GradualRolloutController:
    """灰度发布控制器"""

    def __init__(
        self,
        config: RolloutConfig,
        splitter: Any,       # TrafficSplitter 实例
        collector: Any,      # MetricCollector 实例
        storage_path: str = "rollout_state",
    ):
        self.config = config
        self.splitter = splitter
        self.collector = collector
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.state = RolloutState(experiment_id=config.experiment_id)

    def start(self) -> Dict[str, Any]:
        """启动灰度发布"""
        self.state.status = "running"
        self.state.started_at = datetime.now().isoformat()
        self.state.current_stage_idx = 0
        self._enter_stage(0)
        self._save_state()
        return {"status": "started", "stage": self._current_stage()["stage"]}

    def _current_stage(self) -> Dict:
        return self.config.stages[self.state.current_stage_idx]

    def _enter_stage(self, idx: int) -> None:
        """进入指定灰度阶段"""
        stage = self.config.stages[idx]
        self.state.current_stage_idx = idx
        self.state.current_traffic_pct = stage["traffic_pct"]
        self.state.stage_started_at = datetime.now().isoformat()
        self.state.history.append({
            "action": "enter_stage",
            "stage": stage["stage"],
            "traffic_pct": stage["traffic_pct"],
            "timestamp": datetime.now().isoformat(),
        })

        # 更新流量分配
        exp = self.splitter.experiments.get(self.config.experiment_id)
        if exp:
            treatment_pct = stage["traffic_pct"]
            control_pct = 100 - treatment_pct
            exp.variants[0]["weight"] = control_pct
            exp.variants[1]["weight"] = treatment_pct

    def check_health(self) -> Dict[str, Any]:
        """检查当前阶段健康状态"""
        self.state.last_check_at = datetime.now().isoformat()
        thresholds = self.config.rollback_thresholds

        # 获取指标
        metrics = self._collect_current_metrics()

        # 检查阈值
        issues = []
        if metrics.get("error_rate", 0) > thresholds.get("error_rate_max", 1.0):
            issues.append(f"错误率 {metrics['error_rate']:.2%} 超过阈值")
        if metrics.get("latency_p99", 0) > thresholds.get("latency_p99_max_ms", 99999):
            issues.append(f"P99 延迟 {metrics['latency_p99']:.0f}ms 超过阈值")
        if metrics.get("satisfaction", 5) < thresholds.get("satisfaction_min", 0):
            issues.append(f"满意度 {metrics['satisfaction']:.2f} 低于阈值")

        health = {
            "healthy": len(issues) == 0,
            "metrics": metrics,
            "issues": issues,
            "stage": self._current_stage()["stage"],
            "traffic_pct": self.state.current_traffic_pct,
        }

        # 自动回滚
        if issues:
            self._rollback(f"健康检查失败: {'; '.join(issues)}")
            health["action"] = "rolled_back"

        return health

    def try_promote(self) -> Dict[str, Any]:
        """尝试推进到下一阶段"""
        stage = self._current_stage()

        # 检查是否已在最终阶段
        if self.state.current_stage_idx >= len(self.config.stages) - 1:
            return {"promoted": False, "reason": "already_at_final_stage"}

        # 检查是否满足推进条件
        health = self.check_health()
        if not health["healthy"]:
            return {"promoted": False, "reason": "health_check_failed", "details": health}

        # 检查持续时间
        if self.state.stage_started_at:
            started = datetime.fromisoformat(self.state.stage_started_at)
            hours_elapsed = (datetime.now() - started).total_seconds() / 3600
            required_hours = stage.get("duration_hours", 0)
            if hours_elapsed < required_hours:
                return {
                    "promoted": False,
                    "reason": "duration_not_met",
                    "elapsed_hours": round(hours_elapsed, 1),
                    "required_hours": required_hours,
                }

        # 推进
        next_idx = self.state.current_stage_idx + 1
        self._enter_stage(next_idx)
        self._save_state()

        return {
            "promoted": True,
            "new_stage": self._current_stage()["stage"],
            "new_traffic_pct": self.state.current_traffic_pct,
        }

    def _rollback(self, reason: str) -> None:
        """回滚到对照组"""
        self.state.status = "rolled_back"
        self.state.history.append({
            "action": "rollback",
            "reason": reason,
            "timestamp": datetime.now().isoformat(),
        })

        # 将全部流量切回对照组
        exp = self.splitter.experiments.get(self.config.experiment_id)
        if exp:
            exp.variants[0]["weight"] = 100
            exp.variants[1]["weight"] = 0

        self._save_state()

    def _collect_current_metrics(self) -> Dict[str, float]:
        """收集当前指标"""
        exp_id = self.config.experiment_id
        treatment_metrics = self.collector.get_variant_metrics(exp_id, "rating_overall")
        latency_metrics = self.collector.get_variant_metrics(exp_id, "latency_ms")
        error_metrics = self.collector.get_variant_metrics(exp_id, "error")

        return {
            "satisfaction": treatment_metrics.get("treatment", {}).get("mean", 5.0),
            "latency_p99": latency_metrics.get("treatment", {}).get("mean", 0) * 1.5,
            "error_rate": error_metrics.get("treatment", {}).get("mean", 0),
        }

    def _save_state(self) -> None:
        """持久化状态"""
        path = self.storage_path / f"{self.config.experiment_id}.json"
        with open(path, "w", encoding="utf-8") as f:
            json.dump(self.state.__dict__, f, ensure_ascii=False, indent=2)

    def get_status(self) -> Dict[str, Any]:
        """获取当前状态"""
        return {
            "experiment_id": self.state.experiment_id,
            "status": self.state.status,
            "current_stage": self._current_stage()["stage"],
            "traffic_pct": self.state.current_traffic_pct,
            "history": self.state.history,
        }


# ============================================================
# 使用示例
# ============================================================

if __name__ == "__main__":
```

## 7. A/B 测试设计对比

| 设计类型 | 适用场景 | 优点 | 缺点 | RAG 应用示例 |
|---------|---------|------|------|-------------|
| **经典 A/B** | 单一变量对比 | 简单直观，分析容易 | 一次只能测一个变量 | 对比两个 Prompt 模板 |
| **A/B/N** | 多个变体同时对比 | 效率高，一次测多组 | 样本量需求大 | 对比 3 种检索策略 |
| **分层实验** | 多个独立实验并行 | 充分利用流量 | 需要确保正交性 | 同时测 Prompt 和检索 |
| **互斥实验** | 相互影响的实验 | 避免干扰 | 流量被分割 | 测不同 LLM 模型 |
| **多臂老虎机** | 动态优化 | 自动分配更多流量到优胜组 | 实现复杂 | 动态调整参数 |
| **前后对比** | 无法分流的场景 | 不需要分流 | 受时间因素影响 | 知识库更新效果 |

## 8. 常见错误

```
┌─────────────────────────────────────────────────────────────────┐
│                       常见错误                                  │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  ❌ 错误1: 样本量不足就下结论                                    │
│     p 值不显著可能是样本不够                                     │
│     ✅ 正确: 先计算所需样本量，达到后再分析                       │
│                                                                 │
│  ❌ 错误2: 多次偷看数据 (peeking problem)                        │
│     反复检查显著性会导致假阳性                                    │
│     ✅ 正确: 预设检查时间点，或使用序贯检验                       │
│                                                                 │
│  ❌ 错误3: 忽略网络效应                                          │
│     用户之间互相影响，违反独立性假设                              │
│     ✅ 正确: 按用户分组，不按请求分组                             │
│                                                                 │
│  ❌ 错误4: 只看平均值不看分布                                    │
│     平均值相同但分布可能完全不同                                  │
│     ✅ 正确: 分析分位数、方差和分布形态                           │
│                                                                 │
│  ❌ 错误5: 灰度发布不设回滚阈值                                  │
│     出问题时无法自动止损                                         │
│     ✅ 正确: 预设明确的回滚条件和自动回滚机制                     │
│                                                                 │
│  ❌ 错误6: 实验时间太短                                          │
│     新鲜效应会让初期数据偏高                                     │
│     ✅ 正确: 至少运行 1-2 周，覆盖完整周期                       │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```
