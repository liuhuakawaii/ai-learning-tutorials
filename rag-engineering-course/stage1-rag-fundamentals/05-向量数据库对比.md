## 2. 主流向量数据库深度对比

### 2.1 Milvus

Milvus 是一个高度灵活、可扩展的开源向量数据库，专为大规模向量搜索设计。

```python
# Milvus 完整示例
from pymilvus import (
    connections, Collection, FieldSchema,
    CollectionSchema, DataType, utility
)

# 连接 Milvus
connections.connect("default", host="localhost", port="19530")

# 定义 Schema
fields = [
    FieldSchema(name="id", dtype=DataType.INT64, is_primary=True, auto_id=True),
    FieldSchema(name="text", dtype=DataType.VARCHAR, max_length=65535),
    FieldSchema(name="embedding", dtype=DataType.FLOAT_VECTOR, dim=1536),
    FieldSchema(name="source", dtype=DataType.VARCHAR, max_length=512),
    FieldSchema(name="chunk_id", dtype=DataType.INT64),
]
schema = CollectionSchema(fields, description="RAG document chunks")
collection = Collection("rag_chunks", schema)

# 创建 HNSW 索引
index_params = {
    "metric_type": "COSINE",
    "index_type": "HNSW",
    "params": {"M": 16, "efConstruction": 200}
}
collection.create_index("embedding", index_params)

# 插入数据
import numpy as np
data = [
    ["RAG是检索增强生成技术", "向量数据库存储embedding"],
    [[np.random.randn(1536).tolist()], [np.random.randn(1536).tolist()]],
    ["paper_001", "doc_002"],
    [0, 1],
]
collection.insert(data)
collection.load()

# 查询（带元数据过滤）
search_params = {"metric_type": "COSINE", "params": {"ef": 100}}
query_vector = np.random.randn(1536).tolist()

results = collection.search(
    data=[query_vector],
    anns_field="embedding",
    param=search_params,
    limit=5,
    expr='source == "paper_001"',  # 元数据过滤
    output_fields=["text", "source"]
)

for hits in results:
    for hit in hits:
        print(f"ID: {hit.id}, Score: {hit.score}, Text: {hit.entity.get('text')}")
```

**Milvus 核心特点：**
- 支持多种索引：HNSW、IVF_FLAT、IVF_SQ8、IVF_PQ、DISKANN
- 支持标量过滤 + 向量搜索的混合查询
- 支持分布式部署，水平扩展
- 云原生架构，支持 Kubernetes

### 2.2 Qdrant

Qdrant 是一个高性能的向量数据库，专注于精确过滤和丰富的负载（payload）支持。

```python
# Qdrant 完整示例
from qdrant_client import QdrantClient
from qdrant_client.models import (
    Distance, VectorParams, PointStruct,
    Filter, FieldCondition, MatchValue
)
import numpy as np

# 连接 Qdrant
client = QdrantClient(host="localhost", port=6333)

# 创建 Collection
client.create_collection(
    collection_name="rag_chunks",
    vectors_config=VectorParams(
        size=1536,
        distance=Distance.COSINE
    ),
)

# 插入数据（带 payload）
points = [
    PointStruct(
        id=1,
        vector=np.random.randn(1536).tolist(),
        payload={
            "text": "RAG是检索增强生成技术",
            "source": "paper_001",
            "chunk_id": 0,
            "page": 1,
            "tags": ["AI", "NLP"]
        }
    ),
    PointStruct(
        id=2,
        vector=np.random.randn(1536).tolist(),
        payload={
            "text": "向量数据库存储embedding",
            "source": "doc_002",
            "chunk_id": 1,
            "page": 3,
            "tags": ["database", "vector"]
        }
    ),
]
client.upsert(collection_name="rag_chunks", points=points)

# 查询（带精确过滤）
query_vector = np.random.randn(1536).tolist()

results = client.search(
    collection_name="rag_chunks",
    query_vector=query_vector,
    query_filter=Filter(
        must=[
            FieldCondition(
                key="source",
                match=MatchValue(value="paper_001")
            )
        ]
    ),
    limit=5
)

for result in results:
    print(f"ID: {result.id}, Score: {result.score}")
    print(f"Text: {result.payload['text']}")
```

**Qdrant 核心特点：**
- Payload 过滤性能优异，支持嵌套对象过滤
- 支持多向量（Multi-vector）存储
- 内存映射（mmap）支持超大数据集
- 提供 Qdrant Cloud 托管服务

### 2.3 ChromaDB

ChromaDB 是一个轻量级的开源向量数据库，非常适合快速原型开发和本地实验。

```python
# ChromaDB 完整示例
import chromadb
from chromadb.utils import embedding_functions
import numpy as np

# 创建客户端（本地持久化）
client = chromadb.PersistentClient(path="./chroma_db")

# 使用 OpenAI Embedding（也可用其他 embedding 函数）
openai_ef = embedding_functions.OpenAIEmbeddingFunction(
    api_key="your-api-key",
    model_name="text-embedding-3-small"
)

# 创建 Collection
collection = client.get_or_create_collection(
    name="rag_chunks",
    embedding_function=openai_ef,
    metadata={"hnsw:space": "cosine"}
)

# 插入数据
collection.add(
    documents=["RAG是检索增强生成技术", "向量数据库存储embedding"],
    metadatas=[
        {"source": "paper_001", "chunk_id": 0, "page": 1},
        {"source": "doc_002", "chunk_id": 1, "page": 3}
    ],
    ids=["chunk_1", "chunk_2"]
)

# 查询（带元数据过滤）
results = collection.query(
    query_texts=["什么是RAG?"],
    n_results=5,
    where={"source": "paper_001"},
    include=["documents", "metadatas", "distances"]
)

for i, doc in enumerate(results["documents"][0]):
    print(f"Text: {doc}")
    print(f"Metadata: {results['metadatas'][0][i]}")
    print(f"Distance: {results['distances'][0][i]}")
```

**ChromaDB 核心特点：**
- 零配置，pip install 即用
- 内置 embedding 函数支持
- 适合本地开发和测试
- 生产环境性能有限，适合中小规模

### 2.4 Weaviate

Weaviate 是一个 AI-native 的向量数据库，内置向量化模块和 GraphQL API。

```python
# Weaviate 完整示例
import weaviate
import numpy as np

# 连接 Weaviate
client = weaviate.Client(url="http://localhost:8080")

# 创建 Class（Schema）
class_obj = {
    "class": "RAGChunk",
    "vectorizer": "none",  # 使用自定义向量
    "properties": [
        {"name": "text", "dataType": ["text"]},
        {"name": "source", "dataType": ["text"]},
        {"name": "chunkId", "dataType": ["int"]},
    ]
}

# 删除旧 schema 并创建
client.schema.delete_class("RAGChunk")
client.schema.create_class(class_obj)

# 插入数据
for i, (text, source) in enumerate([
    ("RAG是检索增强生成技术", "paper_001"),
    ("向量数据库存储embedding", "doc_002"),
]):
    data_object = {"text": text, "source": source, "chunkId": i}
    client.data_object.create(
        data_object=data_object,
        class_name="RAGChunk",
        vector=np.random.randn(1536).tolist()
    )

# 查询（带 where 过滤）
result = (
    client.query
    .get("RAGChunk", ["text", "source"])
    .with_near_vector({"vector": np.random.randn(1536).tolist()})
    .with_where({
        "path": ["source"],
        "operator": "Equal",
        "valueText": "paper_001"
    })
    .with_limit(5)
    .do()
)

for item in result["data"]["Get"]["RAGChunk"]:
    print(f"Text: {item['text']}, Source: {item['source']}")
```

**Weaviate 核心特点：**
- 内置向量化模块（text2vec-transformers 等）
- GraphQL API，查询灵活
- 支持多租户（Multi-tenancy）
- Hybrid Search（BM25 + 向量）

### 2.5 Pinecone

Pinecone 是一个全托管的向量数据库服务，无需运维，适合快速上线。

```python
# Pinecone 完整示例
from pinecone import Pinecone, ServerlessSpec
import numpy as np

# 初始化
pc = Pinecone(api_key="your-api-key")

# 创建索引
index_name = "rag-chunks"
if index_name not in pc.list_indexes().names():
    pc.create_index(
        name=index_name,
        dimension=1536,
        metric="cosine",
        spec=ServerlessSpec(cloud="aws", region="us-east-1")
    )

# 获取索引
index = pc.Index(index_name)

# 插入数据
vectors = [
    {
        "id": "chunk_1",
        "values": np.random.randn(1536).tolist(),
        "metadata": {
            "text": "RAG是检索增强生成技术",
            "source": "paper_001",
            "chunk_id": 0
        }
    },
    {
        "id": "chunk_2",
        "values": np.random.randn(1536).tolist(),
        "metadata": {
            "text": "向量数据库存储embedding",
            "source": "doc_002",
            "chunk_id": 1
        }
    },
]
index.upsert(vectors=vectors)

# 查询（带元数据过滤）
query_vector = np.random.randn(1536).tolist()

results = index.query(
    vector=query_vector,
    top_k=5,
    include_metadata=True,
    filter={"source": {"$eq": "paper_001"}}
)

for match in results["matches"]:
    print(f"ID: {match['id']}, Score: {match['score']}")
    print(f"Text: {match['metadata']['text']}")
```

**Pinecone 核心特点：**
- 全托管，零运维
- Serverless 按用量计费
- 低延迟，适合生产环境
- 不支持私有部署

## 3. 索引算法深度解析

### 3.1 HNSW（Hierarchical Navigable Small World）

HNSW 是目前最流行的近似最近邻（ANN）索引算法。

```
┌─────────────────────────────────────────────────────────────────┐
│                     HNSW 索引结构                                │
│                                                                 │
│  Layer 2 (最稀疏):    ┌───┐                                     │
│                       │ A │                                     │
│                       └─┬─┘                                     │
│                         │                                       │
│  Layer 1:         ┌─────┼─────┐                                 │
│                   │     │     │                                 │
│                 ┌─┴─┐ ┌─┴─┐ ┌─┴─┐                              │
│                 │ B │ │ C │ │ D │                               │
│                 └─┬─┘ └─┬─┘ └─┬─┘                              │
│                   │     │     │                                 │
│  Layer 0 (最密):  ├─────┼─────┤                                 │
│                 ┌─┴─┐ ┌─┴─┐ ┌─┴─┐ ┌───┐ ┌───┐                 │
│                 │ E │ │ F │ │ G │ │ H │ │ I │                  │
│                 └───┘ └───┘ └───┘ └───┘ └───┘                  │
│                                                                 │
│  搜索过程: 从顶层开始贪心搜索，逐层下降细化                        │
│  时间复杂度: O(log N)                                            │
│  空间复杂度: O(N * M)  M = 每层最大连接数                         │
└─────────────────────────────────────────────────────────────────┘
```

**HNSW 关键参数：**
- `M`：每个节点的最大连接数（通常 16-64），影响图的密度
- `efConstruction`：构建时的搜索宽度（通常 100-500），影响索引质量
- `ef`：查询时的搜索宽度（通常 50-200），影响查询精度

**权衡：**
- `M` 越大，图越密，召回率越高，但内存占用越大
- `efConstruction` 越大，索引构建越慢，但质量越好
- `ef` 越大，查询越慢，但召回率越高

```python
# HNSW 参数调优示例
from pymilvus import Collection

# 低内存场景
index_params_low_mem = {
    "index_type": "HNSW",
    "params": {"M": 8, "efConstruction": 100}
}

# 高精度场景
index_params_high_acc = {
    "index_type": "HNSW",
    "params": {"M": 32, "efConstruction": 400}
}

# 查询时调整 ef
search_params = {"params": {"ef": 200}}  # ef 越大精度越高
```

### 3.2 IVF（Inverted File Index）

IVF 通过聚类将向量空间划分为多个区域，查询时只搜索最近的几个区域。

```
┌─────────────────────────────────────────────────────────────────┐
│                      IVF 索引结构                                │
│                                                                 │
│  原始向量空间:                                                   │
│                                                                 │
│     ·  ·          Cluster 1 (centroid C1)                       │
│    · ·· ·                                                       │
│     ·  ·     · ·                                                │
│             ·  ·    Cluster 2 (centroid C2)                     │
│              · ·                                                │
│     ·   ·                                                       │
│    ·  · ·       Cluster 3 (centroid C3)                         │
│     ·   ·                                                       │
│                                                                 │
│  倒排索引:                                                       │
│  ┌─────────────────────────────────────────┐                    │
│  │ C1 → [vec_0, vec_3, vec_7, vec_12]      │                    │
│  │ C2 → [vec_1, vec_5, vec_8, vec_15]      │                    │
│  │ C3 → [vec_2, vec_4, vec_9, vec_11]      │                    │
│  └─────────────────────────────────────────┘                    │
│                                                                 │
│  查询: 先找最近的 nprobe 个 centroid，再在这些 cluster 中搜索      │
└─────────────────────────────────────────────────────────────────┘
```

**IVF 关键参数：**
- `nlist`：聚类中心数量（通常 sqrt(N) 到 4*sqrt(N)）
- `nprobe`：查询时搜索的聚类数量，影响精度和速度

```python
# IVF 索引示例（Milvus）
index_params_ivf = {
    "index_type": "IVF_FLAT",
    "metric_type": "COSINE",
    "params": {"nlist": 1024}
}

# 查询时调整 nprobe
search_params_ivf = {
    "metric_type": "COSINE",
    "params": {"nprobe": 32}  # nprobe 越大精度越高
}
```

### 3.3 PQ（Product Quantization）

PQ 通过向量压缩降低内存占用，适合超大规模数据集。

```
┌─────────────────────────────────────────────────────────────────┐
│                   PQ 量化原理                                    │
│                                                                 │
│  原始向量 (1536维):                                              │
│  [0.12, -0.34, 0.56, ..., -0.23]                               │
│                                                                 │
│  分段量化:                                                       │
│  ┌──────────────┬──────────────┬──────────────┬──────────────┐  │
│  │ 子向量 1     │ 子向量 2     │ 子向量 3     │ 子向量 4     │  │
│  │ (384维)      │ (384维)      │ (384维)      │ (384维)      │  │
│  └──────┬───────┴──────┬───────┴──────┬───────┴──────┬───────┘  │
│         ▼              ▼              ▼              ▼           │
│  ┌──────────────┬──────────────┬──────────────┬──────────────┐  │
│  │ Codebook 1   │ Codebook 2   │ Codebook 3   │ Codebook 4   │  │
│  │ → code: 42   │ → code: 107  │ → code: 23   │ → code: 198  │  │
│  └──────────────┴──────────────┴──────────────┴──────────────┘  │
│                                                                 │
│  压缩后: 4 个 uint8 = 4 字节（原 1536*4 = 6144 字节）            │
│  压缩比: 1536:1                                                 │
└─────────────────────────────────────────────────────────────────┘
```

## 5. 元数据过滤设计

元数据过滤是提升 RAG 检索精度的关键技术。

### 5.1 Schema 设计最佳实践

```python
# 好的 Schema 设计
{
    "id": "chunk_001",
    "embedding": [0.12, -0.34, ...],  # 向量
    "payload": {
        # 检索过滤字段
        "source": "paper_2024_rag",        # 文档来源
        "doc_type": "research_paper",       # 文档类型
        "chunk_index": 5,                   # 块序号
        "page_number": 12,                  # 页码
        "language": "zh",                   # 语言
        "created_at": "2024-01-15",         # 创建时间
        "tags": ["RAG", "NLP", "LLM"],      # 标签

        # 存储字段
        "text": "RAG是检索增强生成技术...",  # 原文
        "title": "RAG技术综述",             # 文档标题
        "section": "第二章 系统架构",        # 章节
    }
}
```

### 5.2 常见过滤模式

```python
# Qdrant 过滤示例
from qdrant_client.models import Filter, FieldCondition, MatchValue

# 模式1: 按文档来源过滤
filter_by_source = Filter(
    must=[FieldCondition(key="source", match=MatchValue(value="paper_2024"))]
)

# 模式2: 按文档类型过滤
filter_by_type = Filter(
    must=[FieldCondition(key="doc_type", match=MatchValue(value="research_paper"))]
)

# 模式3: 多条件组合过滤
filter_combined = Filter(
    must=[
        FieldCondition(key="source", match=MatchValue(value="paper_2024")),
        FieldCondition(key="language", match=MatchValue(value="zh")),
    ]
)

# 模式4: 标签过滤（任一匹配）
filter_tags = Filter(
    should=[
        FieldCondition(key="tags", match=MatchValue(value="RAG")),
        FieldCondition(key="tags", match=MatchValue(value="NLP")),
    ]
)
```

## 6. 性能基准测试

### 6.1 测试框架

```python
import time
import numpy as np
from typing import List, Dict

class VectorDBBenchmark:
    def __init__(self, db_client, collection_name: str):
        self.client = db_client
        self.collection = collection_name
        self.results = []

    def benchmark_insert(
        self,
        vectors: List[List[float]],
        batch_size: int = 1000
    ) -> Dict:
        total = len(vectors)
        start = time.time()

        for i in range(0, total, batch_size):
            batch = vectors[i:i + batch_size]
            self._insert_batch(batch)

        elapsed = time.time() - start
        result = {
            "operation": "insert",
            "total_vectors": total,
            "elapsed_seconds": round(elapsed, 2),
            "vectors_per_second": round(total / elapsed, 0),
            "batch_size": batch_size,
        }
        self.results.append(result)
        return result

    def benchmark_search(
        self,
        query_vectors: List[List[float]],
        top_k: int = 10
    ) -> Dict:
        latencies = []

        for qv in query_vectors:
            start = time.time()
            self._search(qv, top_k)
            latencies.append(time.time() - start)

        result = {
            "operation": "search",
            "total_queries": len(query_vectors),
            "avg_latency_ms": round(np.mean(latencies) * 1000, 2),
            "p50_latency_ms": round(np.percentile(latencies, 50) * 1000, 2),
            "p99_latency_ms": round(np.percentile(latencies, 99) * 1000, 2),
            "qps": round(len(query_vectors) / sum(latencies), 0),
        }
        self.results.append(result)
        return result

    def benchmark_filtered_search(
        self,
        query_vectors: List[List[float]],
        filter_expr: dict,
        top_k: int = 10
    ) -> Dict:
        latencies = []

        for qv in query_vectors:
            start = time.time()
            self._search_with_filter(qv, filter_expr, top_k)
            latencies.append(time.time() - start)

        result = {
            "operation": "filtered_search",
            "total_queries": len(query_vectors),
            "avg_latency_ms": round(np.mean(latencies) * 1000, 2),
            "p99_latency_ms": round(np.percentile(latencies, 99) * 1000, 2),
            "qps": round(len(query_vectors) / sum(latencies), 0),
        }
        self.results.append(result)
        return result

    def report(self) -> str:
        lines = ["=" * 60, f"Benchmark Report: {self.collection}", "=" * 60]
        for r in self.results:
            lines.append(f"\n--- {r['operation'].upper()} ---")
            for k, v in r.items():
                if k != "operation":
                    lines.append(f"  {k}: {v}")
        return "\n".join(lines)
```


## 7. 常见错误与避坑指南

### 错误 1: 选择错误的索引类型

```python
# ❌ 错误：小数据集用 IVF
# IVF 需要足够的数据来训练聚类中心
index_params = {"index_type": "IVF_FLAT", "params": {"nlist": 1024}}
# 数据量 < nlist * 39 时，IVF 效果很差

# ✅ 正确：小数据集用 HNSW 或 FLAT
index_params = {"index_type": "HNSW", "params": {"M": 16, "efConstruction": 200}}
# 或者
index_params = {"index_type": "FLAT"}  # 暴力搜索，100%召回
```

### 错误 2: 忽略距离度量选择

```python
# ❌ 错误：不做归一化就用 COSINE
# 如果 embedding 模型输出未归一化，COSINE 计算可能不准确

# ✅ 正确：确认 embedding 模型的输出格式
# OpenAI text-embedding-3 已归一化，可直接用 COSINE
# 有些模型需要手动归一化
import numpy as np

def normalize_vector(v):
    norm = np.linalg.norm(v)
    if norm > 0:
        return (np.array(v) / norm).tolist()
    return v
```

### 错误 3: 不使用元数据过滤

```python
# ❌ 错误：只做向量搜索，不过滤
results = collection.search(query_vector, limit=10)
# 可能返回来自无关文档的结果

# ✅ 正确：结合元数据过滤
results = collection.search(
    query_vector,
    limit=10,
    filter={"source": "relevant_doc.pdf"}  # 缩小搜索范围
)
```

### 错误 4: 批量插入不设置 batch_size

```python
# ❌ 错误：一次性插入百万条数据
collection.insert(all_data)  # 可能 OOM 或超时

# ✅ 正确：分批插入
batch_size = 5000
for i in range(0, len(all_data), batch_size):
    batch = all_data[i:i + batch_size]
    collection.insert(batch)
```

### 错误 5: 忘记创建索引就查询

```python
# ❌ 错误：不创建索引直接搜索
# Milvus 默认使用 FLAT 索引，大数据集会很慢

# ✅ 正确：先创建索引
collection.create_index("embedding", index_params)
collection.load()  # 加载到内存
```
