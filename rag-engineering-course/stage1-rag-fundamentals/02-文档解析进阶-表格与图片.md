## 1. 为什么文档解析很重要？

在 RAG 系统中，文档解析是索引阶段的第一步，其质量直接影响后续的检索效果。

```
┌─────────────────────────────────────────────────────────────┐
│                文档解析在 RAG 中的位置                         │
│                                                             │
│  ┌──────────┐    ┌──────────┐    ┌──────────┐              │
│  │  原始    │    │  文档    │    │  文本    │              │

│  └──────────┘    └──────────┘    └──────────┘              │
│                                                             │

│                                                             │
│  ⚠️ 如果解析阶段丢失了信息，后续无法恢复                      │
└─────────────────────────────────────────────────────────────┘
```

**常见的文档解析挑战：**

- PDF 中的表格可能是图片格式，无法直接提取文本
- 复杂的页面布局（多栏、图文混排）难以正确解析
- 扫描件需要 OCR 识别
- 不同文档格式需要不同的解析工具

## 3. PDF 表格解析

### 3.1 使用 PyMuPDF 提取表格

```python
"""
使用 PyMuPDF (fitz) 从 PDF 中提取表格
PyMuPDF 1.23.0+ 支持内置表格提取功能
"""

import fitz  # PyMuPDF
import json

def extract_tables_from_pdf(pdf_path: str) -> list[dict]:
    """
    从 PDF 中提取所有表格
    
    Args:
        pdf_path: PDF 文件路径
    
    Returns:
        包含表格数据的字典列表
    """
    doc = fitz.open(pdf_path)
    all_tables = []
    
    for page_num in range(len(doc)):
        page = doc[page_num]
        
        # 查找页面中的表格
        tables = page.find_tables()
        
        for table_idx, table in enumerate(tables):
            # 提取表头（第一行）
            headers = table.rows[0].cells if table.rows else []
            header_texts = [cell.text.strip() for cell in headers]
            
            # 提取数据行
            rows_data = []
            for row in table.rows[1:]:
                row_texts = [cell.text.strip() for cell in row.cells]
                rows_data.append(dict(zip(header_texts, row_texts)))
            
            all_tables.append({
                "page": page_num + 1,
                "table_index": table_idx,
                "headers": header_texts,
                "rows": rows_data,
                "row_count": len(rows_data),
                "col_count": len(header_texts)
            })
    
    doc.close()
    return all_tables


def tables_to_markdown(tables: list[dict]) -> str:
    """将表格转换为 Markdown 格式，便于后续处理"""
    md_output = []
    
    for table in tables:
        md_output.append(f"\n### 表格 (第 {table['page']} 页, 表 {table['table_index'] + 1})\n")
        
        # 表头
        headers = table["headers"]
        md_output.append("| " + " | ".join(headers) + " |")
        md_output.append("| " + " | ".join(["---"] * len(headers)) + " |")
        
        # 数据行
        for row in table["rows"]:
            row_values = [row.get(h, "") for h in headers]
            md_output.append("| " + " | ".join(row_values) + " |")
    
    return "\n".join(md_output)


# 使用示例
if __name__ == "__main__":
    tables = extract_tables_from_pdf("example.pdf")
    print(json.dumps(tables, ensure_ascii=False, indent=2))
    
    markdown = tables_to_markdown(tables)
    print(markdown)
```

### 3.2 使用 pdfplumber 提取复杂表格

```python
"""
使用 pdfplumber 提取复杂表格
pdfplumber 在处理复杂表格布局时表现更好
"""

import pdfplumber
import pandas as pd

def extract_tables_with_pdfplumber(pdf_path: str) -> list[pd.DataFrame]:
    """
    使用 pdfplumber 提取 PDF 表格
    
    pdfplumber 提供了更精细的表格检测控制：
    - 自定义分隔线检测
    - 处理合并单元格
    - 处理跨页表格
    """
    tables = []
    
    with pdfplumber.open(pdf_path) as pdf:
        for page_num, page in enumerate(pdf.pages):
            # 提取表格
            page_tables = page.extract_tables(
                table_settings={
                    "vertical_strategy": "lines",      # 使用线条检测垂直分隔
                    "horizontal_strategy": "lines",    # 使用线条检测水平分隔
                    "min_words_vertical": 2,           # 最小垂直词数
                    "min_words_horizontal": 2,         # 最小水平词数
                }
            )
            
            for table_data in page_tables:
                if table_data and len(table_data) > 1:
                    # 第一行作为表头
                    df = pd.DataFrame(table_data[1:], columns=table_data[0])
                    df["_source_page"] = page_num + 1
                    tables.append(df)
    
    return tables


def clean_table_data(df: pd.DataFrame) -> pd.DataFrame:
    """清理表格数据"""
    # 去除空行
    df = df.dropna(how="all")
    
    # 去除空白字符
    for col in df.columns:
        if df[col].dtype == object:
            df[col] = df[col].str.strip()
    
    # 重置索引
    df = df.reset_index(drop=True)
    
    return df


# 使用示例
tables = extract_tables_with_pdfplumber("financial_report.pdf")
for i, table in enumerate(tables):
    cleaned = clean_table_data(table)
    print(f"\n表格 {i + 1}:")
    print(cleaned.to_string())
```

## 4. HTML 与 Excel 表格解析

### 4.1 HTML 表格解析

```python
"""
从 HTML 中提取表格数据
支持网页和本地 HTML 文件
"""

from bs4 import BeautifulSoup
import pandas as pd
import requests

def extract_tables_from_html(html_content: str) -> list[pd.DataFrame]:
    """
    从 HTML 内容中提取所有表格
    
    Args:
        html_content: HTML 字符串
    
    Returns:
        DataFrame 列表
    """
    soup = BeautifulSoup(html_content, "html.parser")
    tables = soup.find_all("table")
    
    result = []
    for table in tables:
        # 提取表头
        headers = []
        header_row = table.find("thead")
        if header_row:
            headers = [th.get_text(strip=True) for th in header_row.find_all(["th", "td"])]
        
        # 提取数据行
        rows = []
        body = table.find("tbody") or table
        for tr in body.find_all("tr"):
            cells = [td.get_text(strip=True) for td in tr.find_all(["td", "th"])]
            if cells:
                rows.append(cells)
        
        if rows:
            if headers:
                df = pd.DataFrame(rows, columns=headers)
            else:
                df = pd.DataFrame(rows)
            result.append(df)
    
    return result


def extract_tables_from_url(url: str) -> list[pd.DataFrame]:
    """从网页 URL 提取表格"""
    response = requests.get(url)
    response.encoding = response.apparent_encoding
    return extract_tables_from_html(response.text)


def tables_to_structured_text(tables: list[pd.DataFrame]) -> str:
    """将表格转换为结构化文本，适合 RAG 分块"""
    output = []
    
    for i, df in enumerate(tables):
        output.append(f"\n--- 表格 {i + 1} ---\n")
        
        for _, row in df.iterrows():
            row_text = " | ".join([f"{col}: {val}" for col, val in row.items() if val])
            output.append(row_text)
    
    return "\n".join(output)


# 使用示例
tables = extract_tables_from_url("https://example.com/data-page")
for table in tables:
    print(table.head())
```

### 4.2 Excel 表格解析

```python
"""
从 Excel 文件中提取数据
支持多个工作表和复杂格式
"""

import pandas as pd
from pathlib import Path

def extract_tables_from_excel(
    excel_path: str,
    sheet_names: list[str] | None = None
) -> dict[str, pd.DataFrame]:
    """
    从 Excel 文件中提取表格
    
    Args:
        excel_path: Excel 文件路径
        sheet_names: 指定工作表名称，None 表示全部
    
    Returns:
        工作表名到 DataFrame 的映射
    """
    xls = pd.ExcelFile(excel_path)
    
    if sheet_names is None:
        sheet_names = xls.sheet_names
    
    tables = {}
    for sheet in sheet_names:
        df = pd.read_excel(xls, sheet_name=sheet)
        if not df.empty:
            tables[sheet] = df
    
    return tables


def excel_to_rag_friendly_text(excel_path: str) -> str:
    """
    将 Excel 转换为 RAG 友好的文本格式
    每行数据转换为独立的文本块
    """
    tables = extract_tables_from_excel(excel_path)
    output = []
    
    for sheet_name, df in tables.items():
        output.append(f"\n=== 工作表: {sheet_name} ===\n")
        
        for idx, row in df.iterrows():
            row_parts = []
            for col in df.columns:
                value = row[col]
                if pd.notna(value):
                    row_parts.append(f"{col}={value}")
            
            if row_parts:
                output.append(f"行 {idx + 1}: " + ", ".join(row_parts))
    
    return "\n".join(output)


# 使用示例
tables = extract_tables_from_excel("sales_data.xlsx", ["Q1", "Q2"])
for name, df in tables.items():
    print(f"\n工作表: {name}")
    print(f"行数: {len(df)}, 列数: {len(df.columns)}")
    print(df.head())
```

## 5. 图片提取与处理

### 5.1 从 PDF 中提取图片

```python
"""
从 PDF 中提取图片并进行处理
支持图片保存和 OCR 识别
"""

import fitz  # PyMuPDF
from pathlib import Path
import hashlib

def extract_images_from_pdf(
    pdf_path: str,
    output_dir: str = "./extracted_images",
    min_size: tuple[int, int] = (100, 100)
) -> list[dict]:
    """
    从 PDF 中提取图片
    
    Args:
        pdf_path: PDF 文件路径
        output_dir: 图片输出目录
        min_size: 最小图片尺寸 (宽, 高)，过滤小图片
    
    Returns:
        图片信息列表
    """
    output_path = Path(output_dir)
    output_path.mkdir(parents=True, exist_ok=True)
    
    doc = fitz.open(pdf_path)
    images = []
    
    for page_num in range(len(doc)):
        page = doc[page_num]
        image_list = page.get_images(full=True)
        
        for img_idx, img_info in enumerate(image_list):
            xref = img_info[0]
            
            # 提取图片
            base_image = doc.extract_image(xref)
            image_bytes = base_image["image"]
            image_ext = base_image["ext"]
            width = base_image["width"]
            height = base_image["height"]
            
            # 过滤小图片（可能是图标等）
            if width < min_size[0] or height < min_size[1]:
                continue
            
            # 生成唯一文件名
            img_hash = hashlib.md5(image_bytes).hexdigest()[:8]
            filename = f"page{page_num+1}_img{img_idx}_{img_hash}.{image_ext}"
            filepath = output_path / filename
            
            # 保存图片
            with open(filepath, "wb") as f:
                f.write(image_bytes)
            
            images.append({
                "page": page_num + 1,
                "index": img_idx,
                "filename": filename,
                "filepath": str(filepath),
                "width": width,
                "height": height,
                "format": image_ext,
                "size_bytes": len(image_bytes)
            })
    
    doc.close()
    return images


# 使用示例
images = extract_images_from_pdf("report.pdf", min_size=(200, 200))
print(f"提取了 {len(images)} 张图片")
for img in images:
    print(f"  第 {img['page']} 页: {img['filename']} ({img['width']}x{img['height']})")
```

### 5.2 图片 OCR 识别

```python
"""
使用 PaddleOCR 或 Tesseract 进行图片文字识别
PaddleOCR 对中文识别效果更好
"""

from pathlib import Path

def ocr_with_paddleocr(image_path: str, lang: str = "ch") -> list[dict]:
    """
    使用 PaddleOCR 识别图片中的文字
    
    Args:
        image_path: 图片文件路径
        lang: 语言，"ch" 中文, "en" 英文
    
    Returns:
        识别结果列表
    """
    from paddleocr import PaddleOCR
    
    ocr = PaddleOCR(use_angle_cls=True, lang=lang, show_log=False)
    result = ocr.ocr(image_path, cls=True)
    
    texts = []
    for line in result[0]:
        bbox = line[0]  # 边界框坐标
        text = line[1][0]  # 识别的文字
        confidence = line[1][1]  # 置信度
        
        texts.append({
            "text": text,
            "confidence": confidence,
            "bbox": bbox
        })
    
    return texts


def ocr_with_tesseract(image_path: str, lang: str = "chi_sim+eng") -> str:
    """
    使用 Tesseract OCR 识别图片文字
    
    Args:
        image_path: 图片文件路径
        lang: 语言包，chi_sim 简体中文
    
    Returns:
        识别的文字
    """
    import pytesseract
    from PIL import Image
    
    image = Image.open(image_path)
    text = pytesseract.image_to_string(image, lang=lang)
    
    return text.strip()


def images_to_text_batch(image_dir: str, engine: str = "paddleocr") -> dict[str, str]:
    """
    批量识别目录中的所有图片
    
    Args:
        image_dir: 图片目录
        engine: OCR 引擎，"paddleocr" 或 "tesseract"
    
    Returns:
        文件名到识别文字的映射
    """
    image_dir = Path(image_dir)
    results = {}
    
    for image_path in image_dir.glob("*"):
        if image_path.suffix.lower() in [".png", ".jpg", ".jpeg", ".bmp", ".tiff"]:
            try:
                if engine == "paddleocr":
                    texts = ocr_with_paddleocr(str(image_path))
                    results[image_path.name] = " ".join([t["text"] for t in texts])
                else:
                    results[image_path.name] = ocr_with_tesseract(str(image_path))
            except Exception as e:
                results[image_path.name] = f"[OCR 失败: {e}]"
    
    return results
```

## 6. 多模态文档处理

### 6.1 使用 Unstructured 处理混合文档

```python
"""
使用 Unstructured 库处理多模态文档
自动识别文本、表格、图片等元素
"""

from unstructured.partition.auto import partition
from unstructured.documents.elements import (
    NarrativeText,
    Title,
    Table,
    Image,
    ListItem
)

def parse_document_unstructured(file_path: str) -> list[dict]:
    """
    使用 Unstructured 自动解析文档
    
    支持的格式: PDF, Word, HTML, Email, Markdown 等
    
    Returns:
        解析结果列表，每个元素包含类型和内容
    """
    elements = partition(filename=file_path)
    
    parsed = []
    for element in elements:
        element_type = type(element).__name__
        
        parsed_element = {
            "type": element_type,
            "text": str(element),
            "metadata": element.metadata.to_dict() if element.metadata else {}
        }
        
        # 根据元素类型添加额外信息
        if isinstance(element, Table):
            parsed_element["is_table"] = True
        elif isinstance(element, Image):
            parsed_element["is_image"] = True
        elif isinstance(element, Title):
            parsed_element["is_heading"] = True
        
        parsed.append(parsed_element)
    
    return parsed


def group_elements_by_type(elements: list[dict]) -> dict[str, list[dict]]:
    """将解析结果按类型分组"""
    grouped = {
        "titles": [],
        "texts": [],
        "tables": [],
        "images": [],
        "lists": [],
        "other": []
    }
    
    for elem in elements:
        if elem.get("is_heading"):
            grouped["titles"].append(elem)
        elif elem.get("is_table"):
            grouped["tables"].append(elem)
        elif elem.get("is_image"):
            grouped["images"].append(elem)
        elif elem["type"] == "ListItem":
            grouped["lists"].append(elem)
        elif elem["type"] == "NarrativeText":
            grouped["texts"].append(elem)
        else:
            grouped["other"].append(elem)
    
    return grouped


def elements_to_rag_chunks(elements: list[dict], chunk_size: int = 500) -> list[dict]:
    """
    将解析结果转换为 RAG 分块
    表格和图片作为独立块，文本按 chunk_size 分割
    """
    chunks = []
    current_text = ""
    current_metadata = {}
    
    for elem in elements:
        if elem.get("is_table") or elem.get("is_image"):
            # 先保存之前的文本块
            if current_text:
                chunks.append({
                    "text": current_text.strip(),
                    "type": "text",
                    "metadata": current_metadata
                })
                current_text = ""
            
            # 表格和图片作为独立块
            chunks.append({
                "text": elem["text"],
                "type": "table" if elem.get("is_table") else "image",
                "metadata": elem.get("metadata", {})
            })
        else:
            # 文本累积
            current_text += elem["text"] + "\n"
            current_metadata.update(elem.get("metadata", {}))
            
            # 达到 chunk_size 时保存
            if len(current_text) >= chunk_size:
                chunks.append({
                    "text": current_text.strip(),
                    "type": "text",
                    "metadata": current_metadata
                })
                current_text = ""
                current_metadata = {}
    
    # 保存剩余文本
    if current_text:
        chunks.append({
            "text": current_text.strip(),
            "type": "text",
            "metadata": current_metadata
        })
    
    return chunks


# 使用示例
elements = parse_document_unstructured("complex_report.pdf")
grouped = group_elements_by_type(elements)

print(f"标题: {len(grouped['titles'])} 个")
print(f"段落: {len(grouped['texts'])} 个")
print(f"表格: {len(grouped['tables'])} 个")
print(f"图片: {len(grouped['images'])} 个")

chunks = elements_to_rag_chunks(elements, chunk_size=500)
print(f"\n生成了 {len(chunks)} 个 RAG 分块")
```

## 9. 工程建议

1. **建立文档类型自动检测流水线**：不要手动判断每个文件是文本型还是扫描型 PDF。用 PyMuPDF 检测页面文本密度，低于阈值的自动走 OCR 路径，高于阈值的走文本提取路径。
2. **表格转换为 Markdown 而非纯文本**：表格如果只提取纯文本会丢失行列对应关系。转换为 Markdown 格式后，LLM 能正确理解表格结构，检索时也能保留更多语义。
3. **图片提取设置合理阈值**：过小的图片通常是 logo 或装饰元素，不包含有效信息。建议设置最小 100x100 像素的过滤阈值，避免无意义的图片污染知识库。
4. **保留解析元数据**：每条提取结果都应该记录源文件名、页码、元素类型（文本/表格/图片）。这些元数据在后续检索过滤和答案引用时至关重要。
