class ProgrammingLanguage(BaseModel):

## 3. Schema 约束提示

### 3.1 在 Prompt 中嵌入 JSON Schema

```python
import json

ANALYSIS_SCHEMA = {
    "type": "object",
    "properties": {
        "summary": {"type": "string", "description": "一句话总结"},
        "sentiment": {"type": "string", "enum": ["positive", "negative", "neutral"]},
        "key_points": {
            "type": "array",
            "items": {"type": "string"},
            "minItems": 1,
            "maxItems": 5,
        },
        "confidence": {"type": "number", "minimum": 0, "maximum": 1},
    },
    "required": ["summary", "sentiment", "key_points", "confidence"],
}

def schema_constrained_prompt(content: str, schema: dict) -> str:
    return f"""请分析以下内容，并严格按照指定的 JSON Schema 输出结果。

## 待分析内容
{content}

## 输出 JSON Schema
```json
{json.dumps(schema, ensure_ascii=False, indent=2)}
```

## 要求
1. 输出必须是合法的 JSON
2. 必须包含所有 required 字段
3. 字段值必须符合 schema 中定义的类型和约束
4. 不要输出 JSON 以外的任何内容

请输出 JSON："""
```

## 4. 正则引导的输出解析

```python
import re
from typing import Optional

class OutputParser:
    """通用输出解析器"""

    @staticmethod
    def extract_json(text: str) -> Optional[dict]:
        """从文本中提取 JSON"""
        # 尝试直接解析
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            pass

        # 尝试从代码块中提取
        patterns = [
            r'```json\s*(.*?)\s*```',
            r'```\s*(.*?)\s*```',
            r'\{[^{}]*(?:\{[^{}]*\}[^{}]*)*\}',
        ]

        for pattern in patterns:
            match = re.search(pattern, text, re.DOTALL)
            if match:
                try:
                    candidate = match.group(1) if '```' in pattern else match.group(0)
                    return json.loads(candidate)
                except (json.JSONDecodeError, IndexError):
                    continue

        return None

    @staticmethod
    def extract_xml_tags(text: str, tag: str) -> list[str]:
        """提取指定 XML 标签的内容"""
        pattern = f'<{tag}>(.*?)</{tag}>'
        return re.findall(pattern, text, re.DOTALL)

    @staticmethod
    def extract_table(text: str) -> list[list[str]]:
        """提取 Markdown 表格数据"""
        rows = []
        for line in text.strip().split('\n'):
            line = line.strip()
            if line.startswith('|') and line.endswith('|'):
                cells = [c.strip() for c in line.split('|')[1:-1]]
                if not all(set(c) <= {'-', ' ', ':'} for c in cells):
                    rows.append(cells)
        return rows
```

## 5. 格式修复策略

```
┌───────────────────────────────────────────────────────┐
│              格式修复决策树                              │
├───────────────────────────────────────────────────────┤
│                                                       │

│           ▼             ▼                             │

└───────────────────────────────────────────────────────┘
```

### 5.1 格式修复实现

```python
class FormatRepairer:
    """格式修复器"""

    @staticmethod
    def repair_json(text: str) -> Optional[str]:
        """尝试修复常见的 JSON 格式问题"""
        # 移除 BOM 和零宽字符
        text = text.strip().lstrip('\ufeff')

        # 修复中文引号
        text = text.replace('\u201c', '"').replace('\u201d', '"')
        text = text.replace('\u2018', "'").replace('\u2019', "'")

        # 修复尾部逗号
        text = re.sub(r',\s*([}\]])', r'\1', text)

        # 修复缺失的引号（简单场景）
        text = re.sub(
            r'(?<=[\{,])\s*(\w+)\s*:',
            r' "\1":',
            text
        )

        # 尝试补全未闭合的括号
        open_braces = text.count('{') - text.count('}')
        if open_braces > 0:
            text += '}' * open_braces

        open_brackets = text.count('[') - text.count(']')
        if open_brackets > 0:
            text += ']' * open_brackets

        return text

    @staticmethod
    def extract_and_repair(text: str) -> Optional[dict]:
        """提取并修复 JSON"""
        parser = OutputParser()

        # 先尝试直接解析
        result = parser.extract_json(text)
        if result:
            return result

        # 尝试修复
        repaired = FormatRepairer.repair_json(text)
        if repaired:
            try:
                return json.loads(repaired)
            except json.JSONDecodeError:
                pass

        # 从代码块中提取后修复
        for pattern in [r'```json\s*(.*?)\s*```', r'```\s*(.*?)\s*```']:
            match = re.search(pattern, text, re.DOTALL)
            if match:
                repaired = FormatRepairer.repair_json(match.group(1))
                try:
                    return json.loads(repaired)
                except json.JSONDecodeError:
                    continue

        return None
```

## 6. 代码实战：完整的输出控制系统

### 6.1 Pydantic Schema 驱动的输出校验

```python
from pydantic import BaseModel, Field, field_validator
from typing import TypeVar, Generic
from openai import OpenAI

T = TypeVar('T', bound=BaseModel)

class StructuredOutputEngine(Generic[T]):
    """基于 Pydantic 的结构化输出引擎"""

    def __init__(self, model: str = "gpt-4o"):
        self.client = OpenAI()
        self.model = model
        self.parser = OutputParser()
        self.repairer = FormatRepairer()

    def extract_schema(self, pydantic_model: type[T]) -> dict:
        """从 Pydantic Model 提取 JSON Schema"""
        return pydantic_model.model_json_schema()

    def build_prompt(self, task: str, schema: dict) -> str:
        return f"""{task}

请严格按照以下 JSON Schema 输出：
```json
{json.dumps(schema, ensure_ascii=False, indent=2)}
```

要求：
1. 输出合法 JSON
2. 包含所有 required 字段
3. 遵守类型和约束限制
4. 只输出 JSON，无其他内容"""

    def call_with_retry(
        self,
        messages: list[dict],
        pydantic_model: type[T],
        max_retries: int = 3,
    ) -> T:
        """带重试的结构化调用"""
        schema = self.extract_schema(pydantic_model)
        last_error = None
        last_output = None

        for attempt in range(max_retries):
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                temperature=0.1 if attempt > 0 else 0.3,
            )
            raw_output = response.choices[0].message.content
            last_output = raw_output

            # 解析
            data = self.parser.extract_json(raw_output)

            # 修复
            if data is None:
                data = self.repairer.extract_and_repair(raw_output)

            if data is None:
                last_error = "无法从输出中提取有效 JSON"
                messages.append({"role": "assistant", "content": raw_output})
                messages.append({
                    "role": "user",
                    "content": "你的输出不是合法 JSON，请重新输出，只输出 JSON，无其他内容。",
                })
                continue

            # 校验
            try:
                validated = pydantic_model.model_validate(data)
                return validated
            except Exception as e:
                last_error = f"Schema 校验失败: {e}"
                messages.append({"role": "assistant", "content": raw_output})
                messages.append({
                    "role": "user",
                    "content": f"JSON 校验失败，错误：{e}\n请修正后重新输出。",
                })
                continue

        raise RuntimeError(
            f"重试 {max_retries} 次后仍失败。最后错误: {last_error}\n最后输出: {last_output}"
        )
```

### 6.2 使用示例：产品评测分析

```python
class ProductReview(BaseModel):
    """产品评测结果"""
    product_name: str = Field(..., description="产品名称")
    overall_score: float = Field(..., ge=0, le=10, description="综合评分 0-10")
    pros: list[str] = Field(..., min_length=1, max_length=5, description="优点列表")
    cons: list[str] = Field(..., min_length=1, max_length=5, description="缺点列表")
    recommendation: Literal["强烈推荐", "推荐", "一般", "不推荐"] = Field(
        ..., description="推荐等级"
    )
    target_audience: str = Field(..., description="适合的目标用户群")

    @field_validator("overall_score")
    @classmethod
    def score_precision(cls, v):
        return round(v, 1)


def demo_structured_output():
    engine = StructuredOutputEngine[ProductReview](model="gpt-4o")

    messages = [
        {
            "role": "user",
            "content": engine.build_prompt(
                task="请对 MacBook Pro M4 进行产品评测",
                schema=engine.extract_schema(ProductReview),
            ),
        }
    ]

    review = engine.call_with_retry(messages, ProductReview)

    print(f"产品：{review.product_name}")
    print(f"评分：{review.overall_score}/10")
    print(f"推荐：{review.recommendation}")
    print(f"优点：{', '.join(review.pros)}")
    print(f"缺点：{', '.join(review.cons)}")
    print(f"适合：{review.target_audience}")
```

## 7. 常见错误

### ❌ 错误 1：过度依赖 Prompt 约束格式

```python
# 不可靠：仅靠 Prompt 要求 JSON
prompt = "输出JSON格式"  # LLM 可能输出多余文字

# 可靠：Prompt + API 参数 + 后处理三重保障
response = client.chat.completions.create(
    model="gpt-4o",
    messages=[...],
    response_format={"type": "json_object"},  # API 层保证
)
data = OutputParser.extract_json(response.choices[0].message.content)  # 后处理兜底
```

### ❌ 错误 2：Schema 过于复杂

```python
# 错误：嵌套层级过深，LLM 容易出错
schema = {
    "type": "object",
    "properties": {
        "level1": {
            "type": "object",
            "properties": {
                "level2": {
                    "type": "object",
                    "properties": {
                        "level3": {  # 嵌套太深！
                            ...
                        }
                    }
                }
            }
        }
    }
}

# 正确：扁平化 schema，或拆分为多次调用
```

### ❌ 错误 3：不处理格式修复就重试

```python
# 错误：直接重试，浪费 token
for i in range(3):
    response = call_llm(prompt)  # 每次输出可能有同样问题

# 正确：先修复，修复失败再重试
data = repairer.extract_and_repair(response)
if data is None:
    # 重试时追加错误反馈
    messages.append({"role": "user", "content": f"错误：{error}"})
```

## 8. 格式控制策略对比

| 策略 | 可靠性 | 延迟 | Token 消耗 | 适用场景 |
|------|--------|------|-----------|---------|
| Prompt 约束 | 低 | 无额外 | 少 | 简单任务 |
| JSON Mode | 高 | 无额外 | 少 | 通用 JSON 输出 |
| Structured Output | 最高 | 无额外 | 少 | 严格 Schema |
| 格式修复 | 中 | 重试时增加 | 中 | 兜底方案 |
| Regex 解析 | 中 | 低 | 无 | 自由格式提取 |

---

## 9. 总结

- 输出格式控制是 LLM 应用可靠性的关键环节
- Structured Output 是最可靠的方案，推荐优先使用
- 格式修复作为兜底方案，能处理大部分边界情况
- Pydantic 提供了从 Schema 定义到运行时校验的完整链路
