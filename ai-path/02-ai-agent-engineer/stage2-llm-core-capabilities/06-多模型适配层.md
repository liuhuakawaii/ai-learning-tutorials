# 06 多模型适配层

> 绑死一家模型提供商，就是在给自己的业务埋雷。

## 场景引入

你的 AI 应用一直用的 GPT-4o，某天 OpenAI 服务宕机了两小时，你的整个应用跟着瘫痪。老板问："能不能切到 Claude？"你发现代码里到处都是 `openai.ChatCompletion.create()` 的调用，切换意味着改几十个文件。更现实的问题是：简单分类任务用 GPT-4o 太贵了，能不能自动用便宜模型？国内客户要求数据不出境，能不能用国产模型？你需要一个统一的模型适配层来解决这些问题。

## 学习目标

- 设计统一的多模型适配层
- 支持 OpenAI、Anthropic Claude、国产大模型
- 实现模型路由、降级、负载均衡
- 掌握模型选型的决策框架

## 前置要求

- 已完成阶段 2 第 1-5 课
- 理解不同 LLM API 的差异

## 为什么需要多模型适配

1. **成本优化**：简单任务用便宜模型，复杂任务用强模型
2. **可靠性**：一家挂了能切到另一家
3. **合规**：某些场景必须用国产模型
4. **性能**：不同任务适合不同模型

from dataclasses import dataclass, field
from enum import Enum

class ModelProvider(str, Enum):
    OPENAI = "openai"
    ANTHROPIC = "anthropic"
    ZHIPU = "zhipu"        # 智谱 GLM
    QWEN = "qwen"          # 通义千问
    DEEPSEEK = "deepseek"  # DeepSeek

@dataclass
class LLMMessage:
    role: str       # system / user / assistant / tool
    content: str
    name: str | None = None
    tool_calls: list | None = None
    tool_call_id: str | None = None

@dataclass
class LLMConfig:
    model: str
    temperature: float = 0.7
    max_tokens: int = 4096
    top_p: float = 1.0
    stop: list[str] | None = None
    
@dataclass
class LLMResponse:
    content: str
    model: str
    provider: ModelProvider
    input_tokens: int
    output_tokens: int
    cost: float
    latency: float
    
@dataclass
class LLMStreamChunk:
    content: str
    is_done: bool = False
    usage: dict | None = None
```

from abc import ABC, abstractmethod

class BaseLLMProvider(ABC):
    """LLM Provider 基类"""
    
    @abstractmethod
    async def chat(
        self,
        messages: list[LLMMessage],
        config: LLMConfig,
    ) -> LLMResponse:
        """同步调用"""
        ...
    
    @abstractmethod
    async def stream_chat(
        self,
        messages: list[LLMMessage],
        config: LLMConfig,
    ):
        """流式调用"""
        ...
    
    @abstractmethod
    def count_tokens(self, messages: list[LLMMessage]) -> int:
        """计算 token 数量"""
        ...
```

```python
# backend/app/services/llm/openai_provider.py
from openai import AsyncOpenAI
from .base import BaseLLMProvider

class OpenAIProvider(BaseLLMProvider):
    def __init__(self, api_key: str, base_url: str | None = None):
        self.client = AsyncOpenAI(api_key=api_key, base_url=base_url)
    
    async def chat(
        self, messages: list[LLMMessage], config: LLMConfig,
    ) -> LLMResponse:
        import time
        start = time.perf_counter()
        
        response = await self.client.chat.completions.create(
            model=config.model,
            messages=[{"role": m.role, "content": m.content} for m in messages],
            temperature=config.temperature,
            max_tokens=config.max_tokens,
            top_p=config.top_p,
            stop=config.stop,
        )
        
        usage = response.usage
        return LLMResponse(
            content=response.choices[0].message.content,
            model=response.model,
            provider=ModelProvider.OPENAI,
            input_tokens=usage.prompt_tokens,
            output_tokens=usage.completion_tokens,
            cost=self._calculate_cost(config.model, usage.prompt_tokens, usage.completion_tokens),
            latency=time.perf_counter() - start,
        )
    
    async def stream_chat(self, messages, config):
        stream = await self.client.chat.completions.create(
            model=config.model,
            messages=[{"role": m.role, "content": m.content} for m in messages],
            temperature=config.temperature,
            max_tokens=config.max_tokens,
            stream=True,
        )
        async for chunk in stream:
            if chunk.choices and chunk.choices[0].delta.content:
                yield LLMStreamChunk(content=chunk.choices[0].delta.content)
        yield LLMStreamChunk(content="", is_done=True)
    
    def _calculate_cost(self, model, input_tokens, output_tokens):
        pricing = {
            "gpt-4o": {"input": 2.5, "output": 10.0},
            "gpt-4o-mini": {"input": 0.15, "output": 0.6},
        }
        p = pricing.get(model, {"input": 0, "output": 0})
        return (input_tokens * p["input"] + output_tokens * p["output"]) / 1_000_000
```

# backend/app/services/llm/anthropic_provider.py
from anthropic import AsyncAnthropic
from .base import BaseLLMProvider

class AnthropicProvider(BaseLLMProvider):
    def __init__(self, api_key: str):
        self.client = AsyncAnthropic(api_key=api_key)
    
    async def chat(
        self, messages: list[LLMMessage], config: LLMConfig,
    ) -> LLMResponse:
        import time
        start = time.perf_counter()
        
        # Anthropic 的 system 是单独参数
        system_msg = ""
        chat_messages = []
        for m in messages:
            if m.role == "system":
                system_msg = m.content
            else:
                chat_messages.append({"role": m.role, "content": m.content})
        
        response = await self.client.messages.create(
            model=config.model,
            system=system_msg,
            messages=chat_messages,
            max_tokens=config.max_tokens,
            temperature=config.temperature,
        )
        
        return LLMResponse(
            content=response.content[0].text,
            model=response.model,
            provider=ModelProvider.ANTHROPIC,
            input_tokens=response.usage.input_tokens,
            output_tokens=response.usage.output_tokens,
            cost=self._calculate_cost(config.model, response.usage.input_tokens, response.usage.output_tokens),
            latency=time.perf_counter() - start,
        )

## 统一 LLM 服务

```python
# backend/app/services/llm/service.py
class LLMService:
    """统一 LLM 服务——模型路由、降级、负载均衡"""
    
    def __init__(self, config: dict):
        self.providers: dict[ModelProvider, BaseLLMProvider] = {}
        self.model_map: dict[str, tuple[ModelProvider, str]] = {}
        self._init_providers(config)
    
    def _init_providers(self, config: dict):
        if config.get("openai_api_key"):
            self.providers[ModelProvider.OPENAI] = OpenAIProvider(
                api_key=config["openai_api_key"],
                base_url=config.get("openai_base_url"),
            )
        
        if config.get("anthropic_api_key"):
            self.providers[ModelProvider.ANTHROPIC] = AnthropicProvider(
                api_key=config["anthropic_api_key"],
            )
        
        # 模型名 → Provider 映射
        self.model_map = {
            "gpt-4o": (ModelProvider.OPENAI, "gpt-4o"),
            "gpt-4o-mini": (ModelProvider.OPENAI, "gpt-4o-mini"),
            "claude-sonnet": (ModelProvider.ANTHROPIC, "claude-sonnet-4-20250514"),
            "claude-haiku": (ModelProvider.ANTHROPIC, "claude-3-5-haiku-20241022"),
        }
    
    async def chat(
        self,
        messages: list[LLMMessage],
        model: str = "gpt-4o",
        fallback_models: list[str] | None = None,
        **kwargs,
    ) -> LLMResponse:
        """调用 LLM，支持自动降级"""
        config = LLMConfig(model=model, **kwargs)
        
        # 尝试主模型
        models_to_try = [model] + (fallback_models or [])
        
        for model_name in models_to_try:
            try:
                provider, actual_model = self._resolve_model(model_name)
                config.model = actual_model
                return await provider.chat(messages, config)
            except Exception as e:
                logger.warning(f"Model {model_name} failed: {e}")
                if model_name == models_to_try[-1]:
                    raise
        
        raise Exception("All models failed")
    
    def _resolve_model(self, model: str) -> tuple[BaseLLMProvider, str]:
        if model in self.model_map:
            provider_name, actual_model = self.model_map[model]
            return self.providers[provider_name], actual_model
        
        raise ValueError(f"Unknown model: {model}")
```

## 模型选型指南

| 场景 | 推荐模型 | 理由 |
|------|----------|------|
| 简单分类/提取 | GPT-4o-mini / Claude Haiku | 便宜、快、够用 |
| 复杂推理 | GPT-4o / Claude Sonnet | 强、准确 |
| 代码生成 | Claude Sonnet / DeepSeek Coder | 代码能力强 |
| 中文写作 | GLM-4 / Qwen | 中文优化 |
| 长文档处理 | Claude Sonnet (200K) | 上下文窗口最大 |
| 合规场景 | 国产模型 | 数据不出境 |

### 练习 2

**思路**：在 `LLMService` 中实现模型路由（根据名称找 Provider）和自动降级（主模型失败后尝试备选模型），记录每个模型的调用成功率和延迟。

**答案**：

```python
# backend/app/services/llm/service.py
import logging
from dataclasses import dataclass, field
from collections import defaultdict
from .base import BaseLLMProvider, LLMMessage, LLMConfig, LLMResponse

logger = logging.getLogger(__name__)

@dataclass
class ModelMetrics:
    total_calls: int = 0
    successful_calls: int = 0
    total_latency: float = 0.0
    total_cost: float = 0.0

    @property
    def success_rate(self) -> float:
        return self.successful_calls / self.total_calls if self.total_calls > 0 else 0.0

    @property
    def avg_latency(self) -> float:
        return self.total_latency / self.successful_calls if self.successful_calls > 0 else 0.0

class LLMService:
    def __init__(self):
        self.providers: dict[str, BaseLLMProvider] = {}
        self.model_map: dict[str, tuple[str, str]] = {}  # model_name -> (provider_key, actual_model)
        self.metrics: dict[str, ModelMetrics] = defaultdict(ModelMetrics)

    def register_provider(self, key: str, provider: BaseLLMProvider):
        self.providers[key] = provider

    def register_model(self, alias: str, provider_key: str, actual_model: str):
        self.model_map[alias] = (provider_key, actual_model)

    async def chat(
        self,
        messages: list[LLMMessage],
        model: str = "gpt-4o",
        fallback_models: list[str] | None = None,
        **kwargs,
    ) -> LLMResponse:
        models_to_try = [model] + (fallback_models or [])

        for model_name in models_to_try:
            try:
                provider_key, actual_model = self._resolve(model_name)
                provider = self.providers[provider_key]
                config = LLMConfig(model=actual_model, **kwargs)
                response = await provider.chat(messages, config)

                self.metrics[model_name].total_calls += 1
                self.metrics[model_name].successful_calls += 1
                self.metrics[model_name].total_latency += response.latency
                self.metrics[model_name].total_cost += response.cost

                return response
            except Exception as e:
                self.metrics[model_name].total_calls += 1
                logger.warning(f"Model {model_name} failed: {e}")
                if model_name == models_to_try[-1]:
                    raise

        raise Exception("All models failed")

    def _resolve(self, model: str) -> tuple[str, str]:
        if model in self.model_map:
            return self.model_map[model]
        raise ValueError(f"Unknown model: {model}")

    def get_metrics_report(self) -> dict:
        return {
            name: {
                "total_calls": m.total_calls,
                "success_rate": f"{m.success_rate:.1%}",
                "avg_latency": f"{m.avg_latency:.2f}s",
                "total_cost": f"${m.total_cost:.4f}",
            }
            for name, m in self.metrics.items()
        }
```

# 使用示例
service = LLMService()
service.register_provider("openai", OpenAIProvider(api_key="sk-xxx"))
service.register_provider("deepseek", DeepSeekProvider(api_key="sk-xxx"))

- `ModelMetrics` 记录成功率和延迟，为后续的智能路由提供数据支撑
- 常见错误：降级时没有记录日志，上线后不知道哪个模型经常失败

### 练习 3

**思路**：根据任务类型自动选模型（简单任务用便宜模型），记录每日成本，超过预算时自动降级到更便宜的模型。

**答案**：

```python
# backend/app/services/llm/cost_optimizer.py
import asyncio
from datetime import datetime
from enum import Enum
from .service import LLMService
from .base import LLMMessage, LLMResponse

class TaskComplexity(str, Enum):
    SIMPLE = "simple"      # 分类、提取、格式转换
    MEDIUM = "medium"      # 问答、摘要
    COMPLEX = "complex"    # 推理、代码生成、长文写作

class CostOptimizer:
    MODEL_BY_COMPLEXITY = {
        TaskComplexity.SIMPLE: ["gpt-4o-mini", "deepseek-chat"],
        TaskComplexity.MEDIUM: ["gpt-4o-mini", "gpt-4o"],
        TaskComplexity.COMPLEX: ["gpt-4o", "deepseek-chat"],
    }

    def __init__(self, llm_service: LLMService, daily_budget: float = 10.0):
        self.llm = llm_service
        self.daily_budget = daily_budget
        self.daily_cost: float = 0.0
        self.last_reset_date: str = ""

    def _reset_if_new_day(self):
        today = datetime.now().strftime("%Y-%m-%d")
        if today != self.last_reset_date:
            self.daily_cost = 0.0
            self.last_reset_date = today

    def _is_over_budget(self) -> bool:
        return self.daily_cost >= self.daily_budget

    async def chat(
        self,
        messages: list[LLMMessage],
        complexity: TaskComplexity = TaskComplexity.MEDIUM,
        **kwargs,
    ) -> LLMResponse:
        self._reset_if_new_day()

        models = self.MODEL_BY_COMPLEXITY[complexity]

        if self._is_over_budget():
            # 超预算：强制用最便宜的模型
            models = ["gpt-4o-mini"]
            kwargs["max_tokens"] = min(kwargs.get("max_tokens", 4096), 1024)

        response = await self.llm.chat(
            messages=messages,
            model=models[0],
            fallback_models=models[1:],
            **kwargs,
        )

        self.daily_cost += response.cost
        return response

    def get_budget_status(self) -> dict:
        self._reset_if_new_day()
        return {
            "daily_budget": self.daily_budget,
            "daily_cost": round(self.daily_cost, 4),
            "remaining": round(self.daily_budget - self.daily_cost, 4),
            "usage_percent": f"{(self.daily_cost / self.daily_budget) * 100:.1f}%",
        }
```

```python
# 使用示例
optimizer = CostOptimizer(llm_service, daily_budget=5.0)

# 简单分类 → 自动用 gpt-4o-mini（便宜）
result = await optimizer.chat(
    messages=[LLMMessage(role="user", content="这句话的情感是正面还是负面：'这个产品真好用'")],
    complexity=TaskComplexity.SIMPLE,
)

# 复杂推理 → 自动用 gpt-4o（强）
result = await optimizer.chat(
    messages=[LLMMessage(role="user", content="分析这段代码的性能瓶颈并给出优化方案...")],
    complexity=TaskComplexity.COMPLEX,
)

# 查看预算状态
print(optimizer.get_budget_status())
# {'daily_budget': 5.0, 'daily_cost': 0.0023, 'remaining': 4.9977, 'usage_percent': '0.0%'}
```

**要点**：
- `TaskComplexity` 枚举让调用方明确声明任务复杂度，比自动判断更可靠
- 超预算后不仅降级模型，还限制 `max_tokens` 进一步控制成本
- 常见错误：预算只在内存中记录，服务重启后丢失——生产环境应该持久化到数据库或 Redis

## 本节要点

- 多模型适配是企业级 AI 应用的必需
- 统一接口让切换模型不需要改业务代码
- 自动降级提升系统可靠性
- 模型选型要根据场景平衡成本、速度和质量

## 常见误区

| 错误 | 原因 | 解决 |
|------|------|------|
| API 格式不兼容 | 不同 Provider 的消息格式不同 | 在 Provider 内部做格式转换 |
| Token 计算不一致 | 不同模型的 tokenizer 不同 | 每个 Provider 实现自己的 count_tokens |
| 降级不生效 | 没配置 fallback_models | 在调用时传入降级列表 |
| 成本统计不准 | 汇率或定价没更新 | 定期更新定价表 |

## 工程建议

- 模型选型要根据场景做决策：简单分类/提取用便宜模型（gpt-4o-mini、claude-haiku），复杂推理用强模型，中文写作用国产模型
- 对关键业务配置 fallback 模型列表，主模型失败时自动降级，降级策略要记录日志便于后续分析
- 定价表要定期更新，建议每月检查一次各厂商的最新定价，避免成本统计出现偏差
- 切换模型时要重新测试 Prompt 效果，不同模型对同一 Prompt 的表现可能差异很大
- 建议维护一份模型能力矩阵文档，记录每个模型在各场景下的表现，为选型提供数据支撑
