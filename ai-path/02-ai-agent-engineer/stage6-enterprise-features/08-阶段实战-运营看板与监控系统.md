# 08 阶段实战：运营看板与监控系统

> 没有看板的系统是黑盒——有了看板才能看见、管好、优化。

## 场景引入

平台运行了一个月，老板问"现在有多少活跃用户？每个模型花了多少钱？哪个 Agent 最受欢迎？"你打开数据库手动跑了好几条 SQL，拼凑出一个大概的数字。第二天又有人问同样的问题，你意识到没有运营看板，每次回答都要重复劳动。你需要一个实时的监控看板，让所有人都能自己看到系统运行状态。

## 学习目标

- 整合阶段 6 所有技能，构建完整的运营看板
- 实现用量统计、成本分析、质量监控
- 体验企业级监控系统的完整功能

## 1. 后端：统计 API

```python
# backend/app/api/v1/stats.py
from fastapi import APIRouter, Depends
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, func
from app.core.database import get_db
from app.api.deps import get_current_user
from app.models.user import User
from app.models.session import ChatSession, Message
from app.models import Agent, Skill, KnowledgeBase, Workflow

router = APIRouter()


@router.get("/overview")
async def get_overview(
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    session_count = (await db.execute(select(func.count()).select_from(ChatSession))).scalar() or 0
    message_count = (await db.execute(select(func.count()).select_from(Message))).scalar() or 0
    agent_count = (await db.execute(select(func.count()).select_from(Agent))).scalar() or 0
    skill_count = (await db.execute(select(func.count()).select_from(Skill))).scalar() or 0
    kb_count = (await db.execute(select(func.count()).select_from(KnowledgeBase))).scalar() or 0
    workflow_count = (await db.execute(select(func.count()).select_from(Workflow))).scalar() or 0

    token_result = await db.execute(
        select(
            func.coalesce(func.sum(Message.input_tokens), 0),
            func.coalesce(func.sum(Message.output_tokens), 0),
            func.coalesce(func.sum(Message.cost), 0.0),
        )
    )
    row = token_result.one()

    return {
        "sessions": session_count,
        "messages": message_count,
        "agents": agent_count,
        "skills": skill_count,
        "knowledge_bases": kb_count,
        "workflows": workflow_count,
        "total_input_tokens": row[0],
        "total_output_tokens": row[1],
        "total_cost": round(float(row[2]), 4),
    }


@router.get("/usage")
async def get_usage_stats(
    days: int = 7,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    from datetime import datetime, timedelta, timezone

    cutoff = datetime.now(timezone.utc) - timedelta(days=days)

    daily_result = await db.execute(
        select(
            func.date(Message.created_at).label("date"),
            func.count().label("count"),
            func.coalesce(func.sum(Message.input_tokens + Message.output_tokens), 0).label("tokens"),
        )
        .where(Message.created_at >= cutoff)
        .group_by(func.date(Message.created_at))
        .order_by(func.date(Message.created_at))
    )

    return {
        "period_days": days,
        "daily": [
            {"date": str(row.date), "messages": row.count, "tokens": row.tokens}
            for row in daily_result.all()
        ],
    }


@router.get("/models")
async def get_model_stats(
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    result = await db.execute(
        select(
            Message.model,
            func.count().label("count"),
            func.coalesce(func.sum(Message.input_tokens), 0).label("input_tokens"),
            func.coalesce(func.sum(Message.output_tokens), 0).label("output_tokens"),
            func.coalesce(func.sum(Message.cost), 0.0).label("cost"),
        )
        .where(Message.model.isnot(None))
        .group_by(Message.model)
        .order_by(func.count().desc())
    )

    return [
        {
            "model": row.model,
            "count": row.count,
            "input_tokens": row.input_tokens,
            "output_tokens": row.output_tokens,
            "cost": round(float(row.cost), 4),
        }
        for row in result.all()
    ]
```

## 2. 后端：Prometheus 指标端点

```python
# backend/app/core/metrics.py
import time
from fastapi import FastAPI, Request, Response

_metrics = {
    "requests_total": 0,
    "requests_by_path": {},
    "request_duration_seconds": [],
    "errors_total": 0,
}


def setup_metrics(app: FastAPI):
    @app.middleware("http")
    async def metrics_middleware(request: Request, call_next):
        start = time.perf_counter()
        _metrics["requests_total"] += 1

        path = request.url.path
        # 将带参数的路径归一化
        path_key = path
        for prefix in ["/api/v1/chat/sessions/", "/api/v1/agents/", "/api/v1/knowledge/"]:
            if path.startswith(prefix) and len(path) > len(prefix):
                path_key = prefix + "{id}"
                break

        _metrics["requests_by_path"][path_key] = _metrics["requests_by_path"].get(path_key, 0) + 1

        try:
            response = await call_next(request)
            duration = time.perf_counter() - start
            _metrics["request_duration_seconds"].append(duration)
            # 只保留最近 10000 条
            if len(_metrics["request_duration_seconds"]) > 10000:
                _metrics["request_duration_seconds"] = _metrics["request_duration_seconds"][-5000:]
            return response
        except Exception:
            _metrics["errors_total"] += 1
            raise

    @app.get("/metrics")
    async def metrics():
        durations = _metrics["request_duration_seconds"]
        avg = sum(durations) / len(durations) if durations else 0
        p95 = sorted(durations)[int(len(durations) * 0.95)] if len(durations) > 1 else 0

        lines = [
            f"http_requests_total {_metrics['requests_total']}",
            f"http_request_duration_seconds_avg {avg:.3f}",
            f"http_request_duration_seconds_p95 {p95:.3f}",
            f"http_errors_total {_metrics['errors_total']}",
        ]
        for p, count in sorted(_metrics["requests_by_path"].items()):
            lines.append(f'http_requests_by_path{{path="{p}"}} {count}')

        return Response(content="\n".join(lines), media_type="text/plain")
```

## 常见误区

| 误区 | 原因 | 解决 |
|------|------|------|
| 看板数据和实际对不上 | 统计口径不一致 | 统一用数据库聚合，避免前端二次计算 |
| 监控指标太多反而没人看 | 没有分层 | 核心指标放首页，详细指标放二级页面 |
| 告警风暴 | 阈值设置不合理 | 设置静默期和告警聚合 |

## 工程建议

看板的统计查询要做缓存，聚合 SQL 在数据量大时很慢，建议每 5 分钟刷新一次而非实时查询。Prometheus 指标要控制基数，带参数的路径必须归一化（如 `/api/v1/agents/{id}`），否则指标量会随用户增长爆炸。Grafana Dashboard 建议按角色分组——开发关注延迟和错误率，运营关注用量和成本，老板只看趋势图。告警通知要分级：P0 电话、P1 企微/钉钉、P2 邮件，避免所有告警都走同一条通道导致重要告警被淹没。
