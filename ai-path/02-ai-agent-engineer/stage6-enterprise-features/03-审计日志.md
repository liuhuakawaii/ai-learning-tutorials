# 03 审计日志

> 出了问题能查、能追溯、能定责——审计日志是企业合规的基础。

## 场景引入

某天客户投诉说自己的知识库文档被删除了，但团队里没有人承认操作过。翻遍数据库和服务器日志，只找到一条 DELETE 请求记录，却不知道是谁、在什么时间、通过哪个界面执行的。没有审计日志的系统，出了问题就像在黑箱里找线索。你需要一套完整的操作审计体系。

## 学习目标

- 设计审计日志的数据模型
- 实现操作记录和合规追溯
- 支持日志查询和导出

## 审计中间件

```python
class AuditMiddleware:
    """审计日志中间件"""
    
    async def __call__(self, request: Request, call_next):
        response = await call_next(request)
        
        # 记录关键操作
        if request.method in ("POST", "PUT", "DELETE"):
            await self._log_action(request, response)
        
        return response
    
    async def _log_action(self, request: Request, response: Response):
        user = getattr(request.state, "user", None)
        
        log = AuditLog(
            id=str(uuid.uuid4()),
            user_id=user.id if user else "anonymous",
            action=self._method_to_action(request.method),
            resource_type=self._extract_resource_type(request.url.path),
            resource_id=self._extract_resource_id(request.url.path),
            ip_address=request.client.host,
            user_agent=request.headers.get("user-agent", ""),
        )
        
        db.add(log)
```

## 参考答案

### 练习 1

**思路**：审计日志系统的核心是"自动记录 + 结构化存储 + 可查询"。用 FastAPI 中间件自动拦截所有写操作（POST/PUT/DELETE），记录操作人、动作、资源类型、变更前后数据。关键是要在中间件中捕获请求体和响应体，对比变更前后的数据差异。

**答案**：

```python
import uuid
import json
from datetime import datetime, timezone
from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware
from sqlalchemy import Column, String, JSON, DateTime, Text, func
from sqlalchemy.ext.declarative import declarative_base

Base = declarative_base()

class AuditLog(Base):
    __tablename__ = "audit_logs"

    id = Column(String(36), primary_key=True)
    user_id = Column(String(36), index=True, nullable=False)
    action = Column(String(50), index=True, nullable=False)
    resource_type = Column(String(50), index=True)
    resource_id = Column(String(36), index=True)
    before_data = Column(JSON, nullable=True)
    after_data = Column(JSON, nullable=True)
    ip_address = Column(String(50))
    user_agent = Column(String(500))
    request_path = Column(String(500))
    request_method = Column(String(10))
    status_code = Column(String(10))
    created_at = Column(DateTime, server_default=func.now())

    SENSITIVE_FIELDS = {"password", "password_hash", "token", "secret", "api_key", "credit_card"}

    def to_dict(self, include_sensitive=False):
        d = {c.name: getattr(self, c.name) for c in self.__table__.columns}
        if not include_sensitive:
            d.pop("user_agent", None)
        return d


class AuditMiddleware(BaseHTTPMiddleware):
    """审计日志中间件——自动记录所有写操作"""

    def __init__(self, app, db_session_factory):
        super().__init__(app)
        self.db_session_factory = db_session_factory
        self.audit_methods = {"POST", "PUT", "PATCH", "DELETE"}
        self.skip_paths = {"/api/v1/health", "/api/v1/auth/refresh"}

    async def dispatch(self, request: Request, call_next):
        if request.method not in self.audit_methods:
            return await call_next(request)
        if request.url.path in self.skip_paths:
            return await call_next(request)

        # 读取请求体
        body = await request.body()
        request_body = None
        if body:
            try:
                request_body = json.loads(body)
            except json.JSONDecodeError:
                pass

        # 执行请求
        response = await call_next(request)

        # 异步记录审计日志（不阻塞响应）
        try:
            user = getattr(request.state, "user", None)
            await self._save_log(
                user_id=user["id"] if user else "anonymous",
                action=self._method_to_action(request.method),
                resource_type=self._extract_resource_type(request.url.path),
                resource_id=self._extract_resource_id(request.url.path),
                request_body=request_body,
                ip_address=request.client.host if request.client else "unknown",
                user_agent=request.headers.get("user-agent", ""),
                request_path=request.url.path,
                request_method=request.method,
                status_code=str(response.status_code),
            )
        except Exception:
            pass  # 审计日志失败不能影响正常请求

        return response

    def _method_to_action(self, method: str) -> str:
        return {"POST": "create", "PUT": "update", "PATCH": "update", "DELETE": "delete"}.get(method, "unknown")

    def _extract_resource_type(self, path: str) -> str:
        parts = [p for p in path.split("/") if p]
        api_parts = [p for p in parts if p not in ("api", "v1")]
        return api_parts[0] if api_parts else "unknown"

    def _extract_resource_id(self, path: str) -> str | None:
        parts = [p for p in path.split("/") if p]
        if len(parts) >= 3:
            return parts[2]  # /api/v1/{type}/{id}
        return None

    async def _save_log(self, **kwargs):
        async with self.db_session_factory() as session:
            # 脱敏处理
            after_data = kwargs.pop("request_body", None)
            if after_data:
                after_data = self._sanitize(after_data)
            log = AuditLog(id=str(uuid.uuid4()), after_data=after_data, **kwargs)
            session.add(log)
            await session.commit()

    def _sanitize(self, data: dict) -> dict:
        """移除敏感字段"""
        if not isinstance(data, dict):
            return data
        return {k: "***" if k in AuditLog.SENSITIVE_FIELDS else v for k, v in data.items()}


# 查询 API
from fastapi import APIRouter, Depends, Query

router = APIRouter(prefix="/api/v1/audit-logs")

@router.get("")
async def query_audit_logs(
    user_id: str | None = None,
    action: str | None = None,
    resource_type: str | None = None,
    resource_id: str | None = None,
    start_date: str | None = None,
    end_date: str | None = None,
    page: int = Query(1, ge=1),
    size: int = Query(50, ge=1, le=200),
    user=Depends(require_permission("admin:audit")),
):
    """查询审计日志——需要 admin:audit 权限"""
    query = select(AuditLog)
    if user_id:
        query = query.where(AuditLog.user_id == user_id)
    if action:
        query = query.where(AuditLog.action == action)
    if resource_type:
        query = query.where(AuditLog.resource_type == resource_type)
    if resource_id:
        query = query.where(AuditLog.resource_id == resource_id)
    if start_date:
        query = query.where(AuditLog.created_at >= start_date)
    if end_date:
        query = query.where(AuditLog.created_at <= end_date)

    total = await db.scalar(select(func.count()).select_from(query.subquery()))
    rows = await db.execute(query.order_by(AuditLog.created_at.desc()).offset((page-1)*size).limit(size))
    return {"items": [dict(row) for row in rows], "total": total, "page": page, "size": size}
```

**要点**：
- 审计日志写入必须异步化，不能阻塞主请求链路——用后台任务或消息队列
- 敏感字段（password、token 等）在写入前必须脱敏，`_sanitize` 方法用字段名黑名单过滤
- 常见错误：在中间件中用 `await request.body()` 后没有重新构造 request body——FastAPI 的 body 只能读一次，后续处理会拿不到 body

### 练习 2

**思路**：日志导出的核心是支持 CSV 和 JSON 两种格式，按时间范围筛选，流式生成文件避免内存溢出。用 `StreamingResponse` 返回文件流，前端用 Blob 下载。

**答案**：

```python
import csv
import io
import json
from datetime import datetime
from fastapi import APIRouter, Query, Depends
from fastapi.responses import StreamingResponse

router = APIRouter(prefix="/api/v1/audit-logs")

@router.get("/export")
async def export_audit_logs(
    format: str = Query("csv", regex="^(csv|json)$"),
    start_date: str = Query(..., description="开始日期 YYYY-MM-DD"),
    end_date: str = Query(..., description="结束日期 YYYY-MM-DD"),
    user_id: str | None = None,
    action: str | None = None,
    resource_type: str | None = None,
    user=Depends(require_permission("admin:audit")),
):
    """导出审计日志"""
    query = select(AuditLog).where(
        AuditLog.created_at >= start_date,
        AuditLog.created_at <= end_date,
    )
    if user_id:
        query = query.where(AuditLog.user_id == user_id)
    if action:
        query = query.where(AuditLog.action == action)
    if resource_type:
        query = query.where(AuditLog.resource_type == resource_type)

    query = query.order_by(AuditLog.created_at.desc())
    rows = (await db.execute(query)).all()

    if format == "csv":
        return _export_csv(rows, start_date, end_date)
    else:
        return _export_json(rows, start_date, end_date)


def _export_csv(rows, start_date: str, end_date: str) -> StreamingResponse:
    """导出为 CSV"""
    output = io.StringIO()
    writer = csv.writer(output)

    headers = ["ID", "用户ID", "操作", "资源类型", "资源ID", "IP地址", "请求路径", "状态码", "变更后数据", "时间"]
    writer.writerow(headers)

    for row in rows:
        writer.writerow([
            row.id,
            row.user_id,
            row.action,
            row.resource_type,
            row.resource_id,
            row.ip_address,
            row.request_path,
            row.status_code,
            json.dumps(row.after_data, ensure_ascii=False) if row.after_data else "",
            str(row.created_at),
        ])

    output.seek(0)
    filename = f"audit_logs_{start_date}_{end_date}.csv"
    return StreamingResponse(
        iter([output.getvalue()]),
        media_type="text/csv; charset=utf-8-sig",  # utf-8-sig 让 Excel 正确识别中文
        headers={"Content-Disposition": f'attachment; filename="{filename}"'},
    )


def _export_json(rows, start_date: str, end_date: str) -> StreamingResponse:
    """导出为 JSON"""
    data = [
        {
            "id": row.id,
            "user_id": row.user_id,
            "action": row.action,
            "resource_type": row.resource_type,
            "resource_id": row.resource_id,
            "ip_address": row.ip_address,
            "request_path": row.request_path,
            "status_code": row.status_code,
            "after_data": row.after_data,
            "created_at": str(row.created_at),
        }
        for row in rows
    ]

    content = json.dumps(data, ensure_ascii=False, indent=2)
    filename = f"audit_logs_{start_date}_{end_date}.json"
    return StreamingResponse(
        iter([content]),
        media_type="application/json; charset=utf-8",
        headers={"Content-Disposition": f'attachment; filename="{filename}"'},
    )
```

**要点**：
- CSV 用 `utf-8-sig` 编码（带 BOM），否则 Excel 打开中文会乱码
- 大量数据时应该用流式查询 + 分批写入，不要一次性加载所有行到内存
- 常见错误：导出时没有限制时间范围——如果用户不传日期参数，可能导出全量数据导致内存溢出。必须要求 `start_date` 和 `end_date` 为必填

## 常见误区

| 错误 | 原因 | 解决 |
|------|------|------|
| 日志太多性能差 | 没有索引和分区 | 按时间分区，建立索引 |
| 敏感信息泄露 | 日志记录了密码 | 过滤敏感字段 |
| 日志丢失 | 同步写入失败 | 异步写入 + 消息队列 |

## 工程建议

审计日志表的数据量增长很快，建议从一开始就按月分区，避免单表过大导致查询变慢。敏感字段（密码、Token、银行卡号）在写入日志前必须脱敏，可以维护一个字段黑名单自动过滤。日志写入建议采用异步方式，通过消息队列缓冲，既不拖慢主请求链路，也避免写入失败导致日志丢失。审计日志本身要设置为只追加、不可删除，满足合规要求。

## 本节要点

- 审计日志记录"谁在什么时间做了什么"
- 变更前后的数据对比是审计的核心
- 日志要支持多维度查询和导出
- 审计日志本身不能被删除或修改
