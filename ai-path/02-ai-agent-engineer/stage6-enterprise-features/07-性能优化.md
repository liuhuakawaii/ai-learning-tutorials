# 07 性能优化

> 用户不会等超过 3 秒——性能是用户体验的基础。

## 场景引入

平台用户量增长到一千人后，用户开始抱怨"发消息要等五六秒才有回复"。排查发现，每次 LLM 调用都要重新查询用户权限和 Agent 配置，数据库连接池经常被打满，而且没有任何缓存策略。当 LLM 服务商偶尔响应变慢时，整个系统直接超时崩溃。你需要一套从缓存到限流的完整性能优化方案。

## 学习目标

- 实现多层缓存策略
- 优化数据库查询和连接池
- 设计限流和降级机制

## 缓存策略

```python
class CacheService:
    """多层缓存"""
    
    def __init__(self, redis: redis.Redis):
        self.redis = redis
        self.local_cache = {}  # 本地缓存（进程内）
    
    async def get_or_set(self, key: str, factory, ttl: int = 300):
        """获取缓存，不存在则创建"""
        # 1. 本地缓存
        if key in self.local_cache:
            return self.local_cache[key]
        
        # 2. Redis 缓存
        cached = await self.redis.get(key)
        if cached:
            value = json.loads(cached)
            self.local_cache[key] = value
            return value
        
        # 3. 创建新值
        value = await factory()
        
        # 写入缓存
        await self.redis.set(key, json.dumps(value, ensure_ascii=False, default=str), ex=ttl)
        self.local_cache[key] = value
        
        return value
```

```python
class RateLimiter:
    """限流器"""
    
    async def check(self, key: str, limit: int, window: int) -> bool:
        current = await self.redis.incr(key)
        if current == 1:
            await self.redis.expire(key, window)
        return current <= limit

class CircuitBreaker:
    """熔断器"""
    
    def __init__(self, failure_threshold: int = 5, recovery_timeout: int = 60):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.failures = 0
        self.last_failure_time = 0
        self.state = "closed"  # closed / open / half_open
    
    async def call(self, func, *args, **kwargs):
        if self.state == "open":
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = "half_open"
            else:
                raise Exception("Circuit breaker is open")
        
        try:
            result = await func(*args, **kwargs)
            if self.state == "half_open":
                self.state = "closed"
                self.failures = 0
            return result
        except Exception as e:
            self.failures += 1
            self.last_failure_time = time.time()
            if self.failures >= self.failure_threshold:
                self.state = "open"
            raise
```

## 参考答案

### 练习 1

**思路**：多层缓存的核心是"本地缓存 → Redis → 数据库"三级读取。本地缓存（进程内 dict）最快但无法跨进程共享；Redis 缓存跨进程但有网络开销；数据库最慢但数据最权威。缓存失效策略用 TTL + 随机化偏移量防止雪崩。

**答案**：

```python
import json
import hashlib
import random
import time
from typing import Any, Callable

class MultiLayerCache:
    """多层缓存：本地 → Redis → 数据库"""

    def __init__(self, redis_client, db):
        self.redis = redis_client
        self.db = db
        self.local_cache: dict[str, tuple[Any, float]] = {}  # key → (value, expire_ts)
        self.local_ttl = 60  # 本地缓存 60 秒

    async def get_or_set(
        self,
        key: str,
        factory: Callable,
        ttl: int = 300,
        use_local: bool = True,
    ) -> Any:
        """获取缓存，不存在则通过 factory 创建"""
        # 1. 本地缓存
        if use_local and key in self.local_cache:
            value, expire_ts = self.local_cache[key]
            if time.time() < expire_ts:
                return value
            else:
                del self.local_cache[key]

        # 2. Redis 缓存
        cached = await self.redis.get(key)
        if cached:
            value = json.loads(cached)
            if use_local:
                self.local_cache[key] = (value, time.time() + self.local_ttl)
            return value

        # 3. 数据库查询
        value = await factory()

        # 写入 Redis（带随机化 TTL 防雪崩）
        jitter = random.randint(-int(ttl * 0.1), int(ttl * 0.1))
        actual_ttl = max(60, ttl + jitter)
        await self.redis.set(key, json.dumps(value, ensure_ascii=False, default=str), ex=actual_ttl)

        # 写入本地缓存
        if use_local:
            self.local_cache[key] = (value, time.time() + self.local_ttl)

        return value

    async def invalidate(self, key: str):
        """使缓存失效"""
        self.local_cache.pop(key, None)
        await self.redis.delete(key)

    async def invalidate_pattern(self, pattern: str):
        """批量使缓存失效（按模式匹配）"""
        # 本地缓存
        keys_to_delete = [k for k in self.local_cache if self._match_pattern(k, pattern)]
        for k in keys_to_delete:
            del self.local_cache[k]

        # Redis
        cursor = 0
        while True:
            cursor, keys = await self.redis.scan(cursor, match=pattern, count=100)
            if keys:
                await self.redis.delete(*keys)
            if cursor == 0:
                break

    def _match_pattern(self, key: str, pattern: str) -> bool:
        import fnmatch
        return fnmatch.fnmatch(key, pattern)


# 使用示例
cache = MultiLayerCache(redis_client, db)

async def get_user_profile(user_id: str) -> dict:
    """带缓存的用户信息查询"""
    return await cache.get_or_set(
        key=f"user_profile:v1:{user_id}",
        factory=lambda: db.get("users", user_id),
        ttl=300,
    )

async def update_user_profile(user_id: str, data: dict):
    """更新用户信息并清除缓存"""
    await db.update("users", user_id, data)
    await cache.invalidate(f"user_profile:v1:{user_id}")

    pool_recycle=3600,     # 连接最大存活时间（秒）

    echo=False,            # 生产环境关闭 SQL 日志
```

**要点**：
- 本地缓存 TTL 要远小于 Redis TTL（60s vs 300s）——本地缓存无法跨进程通知失效，短 TTL 降低不一致风险
- Redis TTL 必须加随机偏移量（±10%），否则大量缓存同时过期会造成缓存雪崩
- 常见错误：缓存 key 不包含版本号——代码逻辑变更后旧缓存仍然命中，返回错误数据。key 中加 `v1:` 前缀，变更时 bump 版本号即可批量失效

### 练习 2

**思路**：限流保护系统不被打垮，熔断保护系统在依赖服务故障时快速失败。限流用 Redis 滑动窗口实现，区分用户级别。熔断器用三态模型（closed → open → half_open），失败次数超阈值时打开熔断，恢复超时后半开尝试。

**答案**：

```python
import time
import asyncio
from enum import Enum
from typing import Callable, Any

class RateLimiter:
    """Redis 滑动窗口限流器"""

    def __init__(self, redis_client):
        self.redis = redis_client

    async def check(self, key: str, limit: int, window_seconds: int) -> dict:
        """滑动窗口限流检查"""
        now = time.time()
        window_start = now - window_seconds

        pipe = self.redis.pipeline()
        # 移除窗口外的记录
        pipe.zremrangebyscore(key, 0, window_start)
        # 添加当前请求
        pipe.zadd(key, {str(now): now})
        # 统计窗口内的请求数
        pipe.zcard(key)
        # 设置 key 过期
        pipe.expire(key, window_seconds)
        results = await pipe.execute()

        current_count = results[2]
        allowed = current_count <= limit

        return {
            "allowed": allowed,
            "current": current_count,
            "limit": limit,
            "remaining": max(0, limit - current_count),
            "retry_after": window_seconds if not allowed else 0,
        }


class RateLimitMiddleware:
    """限流中间件——区分用户级别"""

    RATE_LIMITS = {
        "free": {"limit": 30, "window": 60},       # 30 次/分钟
        "pro": {"limit": 120, "window": 60},        # 120 次/分钟
        "enterprise": {"limit": 600, "window": 60}, # 600 次/分钟
    }

    def __init__(self, rate_limiter: RateLimiter):
        self.limiter = rate_limiter

    async def check_user(self, user_id: str, user_role: str) -> dict:
        config = self.RATE_LIMITS.get(user_role, self.RATE_LIMITS["free"])
        return await self.limiter.check(
            key=f"rate_limit:user:{user_id}",
            limit=config["limit"],
            window=config["window"],
        )

    async def check_skill(self, user_id: str, skill_id: str, limit: int = 10, window: int = 60) -> dict:
        return await self.limiter.check(
            key=f"rate_limit:skill:{user_id}:{skill_id}",
            limit=limit,
            window=window,
        )


# --- 熔断器 ---
class CircuitState(str, Enum):
    CLOSED = "closed"         # 正常：允许请求通过
    OPEN = "open"             # 熔断：直接拒绝请求
    HALF_OPEN = "half_open"   # 半开：允许少量请求探测

class CircuitBreaker:
    """熔断器——保护下游服务"""

    def __init__(
        self,
        name: str,
        failure_threshold: int = 5,
        recovery_timeout: int = 60,
        half_open_max_calls: int = 3,
    ):
        self.name = name
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_calls = half_open_max_calls

        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.success_count = 0
        self.last_failure_time = 0
        self.half_open_calls = 0

    async def call(self, func: Callable, *args, **kwargs) -> Any:
        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
                self.half_open_calls = 0
                self.success_count = 0
            else:
                raise CircuitOpenError(f"熔断器 {self.name} 已打开，拒绝请求")

        if self.state == CircuitState.HALF_OPEN:
            if self.half_open_calls >= self.half_open_max_calls:
                raise CircuitOpenError(f"熔断器 {self.name} 半开状态已满")
            self.half_open_calls += 1

        try:
            result = await func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise

    def _on_success(self):
        if self.state == CircuitState.HALF_OPEN:
            self.success_count += 1
            if self.success_count >= self.half_open_max_calls:
                self.state = CircuitState.CLOSED
                self.failure_count = 0
        elif self.state == CircuitState.CLOSED:
            self.failure_count = 0  # 成功后重置失败计数

    def _on_failure(self):
        self.failure_count += 1
        self.last_failure_time = time.time()
        if self.state == CircuitState.HALF_OPEN:
            self.state = CircuitState.OPEN
        elif self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN

    def get_status(self) -> dict:
        return {
            "name": self.name,
            "state": self.state.value,
            "failure_count": self.failure_count,
            "last_failure": self.last_failure_time,
        }


class CircuitOpenError(Exception):
    pass


# --- 降级策略 ---
class LLMWithFallback:
    """带降级的 LLM 调用"""

    def __init__(self):
        self.primary_breaker = CircuitBreaker("gpt-4o", failure_threshold=3, recovery_timeout=120)
        self.fallback_breaker = CircuitBreaker("gpt-4o-mini", failure_threshold=5, recovery_timeout=60)

    async def chat(self, messages: list[dict], **kwargs) -> str:
        # 1. 尝试主模型
        try:
            return await self.primary_breaker.call(
                self._call_llm, "gpt-4o", messages, **kwargs
            )
        except (CircuitOpenError, Exception):
            pass

        # 2. 降级到备用模型
        try:
            return await self.fallback_breaker.call(
                self._call_llm, "gpt-4o-mini", messages, **kwargs
            )
        except (CircuitOpenError, Exception):
            pass

        # 3. 最终降级：返回兜底回复
        return "抱歉，AI 服务暂时不可用，请稍后再试。"

    async def _call_llm(self, model: str, messages: list[dict], **kwargs) -> str:
        # 实际 LLM 调用逻辑
        response = await llm_client.chat(model=model, messages=messages, **kwargs)
        return response.content
```

**要点**：
- 限流用滑动窗口而非固定窗口——固定窗口在窗口边界处会有两倍突发流量
- 熔断器的半开状态要限制探测请求数量（`half_open_max_calls`），否则大量请求同时涌入会再次压垮下游
- 常见错误：熔断器打开后不恢复——必须有 `recovery_timeout` 机制，否则一次故障后熔断器永远打开，服务永远不会恢复

## 常见误区

| 错误 | 原因 | 解决 |
|------|------|------|
| 缓存雪崩 | 大量缓存同时过期 | 随机化过期时间 |
| 缓存穿透 | 查询不存在的数据 | 布隆过滤器 |
| 连接池耗尽 | 连接没归还 | 用 async with |

## 工程建议

缓存的 key 设计要包含版本号，方便代码变更时批量失效旧缓存。Redis 缓存建议设置随机化的过期时间（如基础 TTL ± 10%），防止大量缓存同时过期造成雪崩。限流策略要区分用户级别——免费用户和付费用户使用不同的限流阈值。熔断器的状态变更要记录日志并触发告警，方便运维及时发现下游服务异常。数据库查询优化优先用 `select` 指定字段，避免 `SELECT *` 带来的网络和序列化开销。
