# 01 认证与授权

> 没有认证的系统是裸奔——没有授权的系统是无差别放行。

## 场景引入

你的 AI Agent 平台上线第一天，就有用户反馈：登录后 Token 两小时就过期，正在编辑的 Agent 配置全丢了。更糟的是，有用户发现只要猜对 URL 就能访问别人的知识库——系统完全没有权限隔离。你需要一套既能保证安全、又不影响用户体验的认证授权体系。

## 学习目标

- 实现 JWT 认证和 Token 刷新
- 设计 RBAC 权限模型
- 支持多组织空间和 API Key 管理

## JWT 认证

```python
# backend/app/core/security.py
from datetime import datetime, timedelta
from jose import JWTError, jwt
from passlib.context import CryptContext

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

class AuthService:
    def __init__(self, secret_key: str):
        self.secret_key = secret_key
        self.access_token_expire = timedelta(hours=2)
        self.refresh_token_expire = timedelta(days=30)
    
    def create_tokens(self, user_id: str) -> dict:
        """创建 access_token 和 refresh_token"""
        access_token = jwt.encode(
            {"sub": user_id, "type": "access", "exp": datetime.utcnow() + self.access_token_expire},
            self.secret_key,
            algorithm="HS256",
        )
        refresh_token = jwt.encode(
            {"sub": user_id, "type": "refresh", "exp": datetime.utcnow() + self.refresh_token_expire},
            self.secret_key,
            algorithm="HS256",
        )
        return {"access_token": access_token, "refresh_token": refresh_token}
    
    def decode_token(self, token: str) -> dict | None:
        try:
            return jwt.decode(token, self.secret_key, algorithms=["HS256"])
        except JWTError:
            return None
```

## RBAC 权限模型

```python
# 权限定义
PERMISSIONS = {
    "chat:read": "查看对话",
    "chat:write": "发送消息",
    "agent:read": "查看 Agent",
    "agent:write": "编辑 Agent",
    "agent:publish": "发布 Agent",
    "knowledge:read": "查看知识库",
    "knowledge:write": "编辑知识库",
    "workflow:read": "查看工作流",
    "workflow:write": "编辑工作流",
    "workflow:execute": "执行工作流",
    "admin:users": "管理用户",
    "admin:settings": "管理设置",
}

# 角色定义
ROLES = {
    "viewer": ["chat:read", "agent:read", "knowledge:read", "workflow:read"],
    "user": ["chat:*", "agent:read", "knowledge:read", "workflow:read"],
    "editor": ["chat:*", "agent:*", "knowledge:*", "workflow:*"],
    "admin": ["*"],
}

class RBACService:
    async def check_permission(self, user_id: str, permission: str) -> bool:
        user_role = await self._get_user_role(user_id)
        role_permissions = ROLES.get(user_role, [])
        
        # 通配符匹配
        for rp in role_permissions:
            if rp == "*" or rp == permission:
                return True
            if rp.endswith(":*") and permission.startswith(rp[:-2]):
                return True
        
        return False
```

            text("SELECT * FROM api_keys WHERE key_hash = :hash AND is_active = true"),

        )
        row = result.first()
        return dict(row) if row else None

## 参考答案

### 练习 1

**思路**：JWT 认证的核心是双 Token 机制——短命的 access_token（2h）用于接口认证，长命的 refresh_token（30d）用于无感续期。登录时签发两个 Token，access_token 过期后用 refresh_token 换新的，避免用户频繁登录。多设备管理需要记录每个设备的 refresh_token，支持单独吊销。

**答案**：

```python
import uuid
import hashlib
from datetime import datetime, timedelta, timezone
from jose import JWTError, jwt
from passlib.context import CryptContext
from fastapi import APIRouter, HTTPException, Depends

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
router = APIRouter(prefix="/api/v1/auth")

class AuthService:
    def __init__(self, secret_key: str):
        self.secret_key = secret_key
        self.algorithm = "HS256"
        self.access_expire = timedelta(hours=2)
        self.refresh_expire = timedelta(days=30)

    def create_tokens(self, user_id: str, device_id: str = "web") -> dict:
        """签发 access_token 和 refresh_token"""
        now = datetime.now(timezone.utc)
        access_token = jwt.encode(
            {"sub": user_id, "type": "access", "device": device_id, "exp": now + self.access_expire},
            self.secret_key,
            algorithm=self.algorithm,
        )
        refresh_token = jwt.encode(
            {"sub": user_id, "type": "refresh", "device": device_id, "exp": now + self.refresh_expire},
            self.secret_key,
            algorithm=self.algorithm,
        )
        return {
            "access_token": access_token,
            "refresh_token": refresh_token,
            "expires_in": int(self.access_expire.total_seconds()),
        }

    def decode_token(self, token: str) -> dict | None:
        try:
            return jwt.decode(token, self.secret_key, algorithms=[self.algorithm])
        except JWTError:
            return None

    def hash_password(self, password: str) -> str:
        return pwd_context.hash(password)

    def verify_password(self, plain: str, hashed: str) -> bool:
        return pwd_context.verify(plain, hashed)


# --- 注册 ---
@router.post("/register")
async def register(body: RegisterRequest, db=Depends(get_db)):
    existing = await db.query("users", email=body.email)
    if existing:
        raise HTTPException(400, "邮箱已注册")

    user_id = str(uuid.uuid4())
    await db.insert("users", {
        "id": user_id,
        "email": body.email,
        "password_hash": AuthService.hash_password(body.password),
        "name": body.name,
        "role": "user",
    })
    return {"ok": True, "user_id": user_id}


# --- 登录 ---
@router.post("/login")
async def login(body: LoginRequest, db=Depends(get_db), auth: AuthService = Depends(get_auth_service)):
    user = await db.query("users", email=body.email)
    if not user or not auth.verify_password(body.password, user["password_hash"]):
        raise HTTPException(401, "邮箱或密码错误")

    tokens = auth.create_tokens(user["id"], device_id=body.device_id)

    # 存储 refresh_token 到设备表（支持多设备管理）
    await db.insert("user_devices", {
        "id": str(uuid.uuid4()),
        "user_id": user["id"],
        "device_id": body.device_id,
        "device_name": body.device_name,
        "refresh_token_hash": hashlib.sha256(tokens["refresh_token"].encode()).hexdigest(),
        "last_active": datetime.now(timezone.utc),
    })

    return tokens


# --- Token 刷新 ---
@router.post("/refresh")
async def refresh_token(body: RefreshRequest, db=Depends(get_db), auth: AuthService = Depends(get_auth_service)):
    payload = auth.decode_token(body.refresh_token)
    if not payload or payload.get("type") != "refresh":
        raise HTTPException(401, "refresh_token 无效或已过期")

    user_id = payload["sub"]
    device_id = payload.get("device", "web")

    # 验证 refresh_token 是否在设备表中（未被吊销）
    token_hash = hashlib.sha256(body.refresh_token.encode()).hexdigest()
    device = await db.query("user_devices", user_id=user_id, device_id=device_id, refresh_token_hash=token_hash)
    if not device:
        raise HTTPException(401, "refresh_token 已被吊销")

    # 签发新 Token（滑动窗口：同时更新 refresh_token）
    new_tokens = auth.create_tokens(user_id, device_id)

    await db.update("user_devices", device["id"], {
        "refresh_token_hash": hashlib.sha256(new_tokens["refresh_token"].encode()).hexdigest(),
        "last_active": datetime.now(timezone.utc),
    })

    return new_tokens


# --- 多设备管理 ---
@router.get("/devices")
async def list_devices(user=Depends(get_current_user), db=Depends(get_db)):
    """列出用户所有登录设备"""
    devices = await db.query_list("user_devices", user_id=user["id"])
    return [{"device_id": d["device_id"], "device_name": d["device_name"], "last_active": d["last_active"]} for d in devices]


@router.delete("/devices/{device_id}")
async def revoke_device(device_id: str, user=Depends(get_current_user), db=Depends(get_db)):
    """吊销指定设备的 Token"""
    await db.delete("user_devices", user_id=user["id"], device_id=device_id)
    return {"ok": True, "message": f"设备 {device_id} 已被吊销"}
```

**要点**：
- access_token 和 refresh_token 必须分开存储和验证——access_token 无状态校验即可，refresh_token 需要查库确认未被吊销
- 刷新 Token 时要同时签发新的 refresh_token（滑动窗口），避免同一个 refresh_token 长期有效增加泄露风险
- 常见错误：把 refresh_token 也做成了无状态——一旦泄露无法吊销，必须在服务端维护 refresh_token 的有效状态

### 练习 2

**思路**：RBAC 的核心是"用户 → 角色 → 权限"三层映射。用 FastAPI 的依赖注入系统统一做权限检查，每个需要鉴权的接口注入一个 `require_permission("xxx")` 依赖。通配符匹配（如 `agent:*` 表示 agent 相关所有权限）让角色定义更灵活。

**答案**：

```python
from fastapi import Depends, HTTPException, Request

# 权限定义
PERMISSIONS = {
    "chat:read": "查看对话",
    "chat:write": "发送消息",
    "chat:delete": "删除对话",
    "agent:read": "查看 Agent",
    "agent:write": "编辑 Agent",
    "agent:publish": "发布 Agent",
    "knowledge:read": "查看知识库",
    "knowledge:write": "编辑知识库",
    "workflow:read": "查看工作流",
    "workflow:write": "编辑工作流",
    "workflow:execute": "执行工作流",
    "admin:users": "管理用户",
    "admin:settings": "管理设置",
}

# 角色 → 权限映射
ROLES = {
    "viewer": ["chat:read", "agent:read", "knowledge:read", "workflow:read"],
    "user": ["chat:read", "chat:write", "agent:read", "knowledge:read", "workflow:read", "workflow:execute"],
    "editor": ["chat:*", "agent:*", "knowledge:*", "workflow:*"],
    "admin": ["*"],
}


class RBACService:
    def __init__(self, db):
        self.db = db

    async def get_user_permissions(self, user_id: str) -> set[str]:
        """获取用户的所有权限（展开通配符）"""
        user = await self.db.get("users", user_id)
        role = user.get("role", "user")
        role_perms = ROLES.get(role, [])

        expanded = set()
        for perm in role_perms:
            if perm == "*":
                return set(PERMISSIONS.keys())
            if perm.endswith(":*"):
                prefix = perm[:-2]
                expanded.update(p for p in PERMISSIONS if p.startswith(f"{prefix}:"))
            else:
                expanded.add(perm)
        return expanded

    async def check_permission(self, user_id: str, permission: str) -> bool:
        perms = await self.get_user_permissions(user_id)
        return permission in perms

    async def check_skill_permission(self, user_id: str, skill_id: str) -> bool:
        """Skill 级别的权限控制"""
        skill = await self.db.get("skills", skill_id)
        if not skill:
            return False
        user = await self.db.get("users", user_id)

        # 1. 检查角色
        if skill.get("allowed_roles") and user["role"] not in skill["allowed_roles"]:
            return False

        # 2. 检查用户白名单
        if skill.get("allowed_users") and user_id not in skill["allowed_users"]:
            return False

        return True


def require_permission(permission: str):
    """FastAPI 依赖：要求指定权限"""
    async def checker(request: Request, user=Depends(get_current_user)):
        rbac: RBACService = request.app.state.rbac
        if not await rbac.check_permission(user["id"], permission):
            raise HTTPException(403, f"权限不足，需要: {permission}")
        return user
    return checker


# --- 使用示例 ---
@router.get("/agents")
async def list_agents(user=Depends(require_permission("agent:read"))):
    """查看 Agent 列表——需要 agent:read 权限"""
    return await db.query_list("agents", owner_id=user["id"])


@router.post("/agents")
async def create_agent(body: CreateAgentRequest, user=Depends(require_permission("agent:write"))):
    """创建 Agent——需要 agent:write 权限"""
    return await db.insert("agents", {**body.dict(), "owner_id": user["id"]})


@router.post("/agents/{agent_id}/publish")
async def publish_agent(agent_id: str, user=Depends(require_permission("agent:publish"))):
    """发布 Agent——需要 agent:publish 权限"""
    await db.update("agents", agent_id, {"status": "published"})
    return {"ok": True}


@router.delete("/users/{target_user_id}")
async def delete_user(target_user_id: str, user=Depends(require_permission("admin:users"))):
    """删除用户——需要 admin:users 权限"""
    await db.delete("users", target_user_id)
    return {"ok": True}
```

**要点**：
- 权限检查通过 FastAPI 依赖注入统一处理，避免在每个接口里手动写 `if` 判断——遗漏一个就是安全漏洞
- 通配符匹配要在 `get_user_permissions` 阶段展开，不要在 `check_permission` 时逐个匹配——后者性能差且容易出 bug
- 常见错误：只检查角色名而不检查具体权限——用户角色变更后，旧的权限缓存没有清除，导致权限不一致

## 常见误区

| 错误 | 原因 | 解决 |
|------|------|------|
| Token 过期频繁 | 有效期太短 | access 2h + refresh 30d |
| 权限检查遗漏 | 没统一用依赖注入 | 封装 get_current_user 依赖 |
| API Key 泄露 | 明文存储 | 只存 hash，创建时返回一次 |

## 工程建议

生产环境中，Token 密钥必须从环境变量读取，绝对不能硬编码在代码里。API Key 的权限要遵循最小权限原则，为每个 Key 只分配必要的权限集，而不是一律给 admin。多租户场景下，务必在每个数据库查询中加入 org_id 过滤条件，防止跨组织的数据泄露。Token 刷新建议采用滑动窗口策略——用户活跃时自动续期，长期不活跃则要求重新登录。

## 本节要点

- JWT 是无状态认证的标准方案
- RBAC 让权限管理可配置、可扩展
- API Key 让外部系统也能安全接入
- Token 刷新避免用户频繁登录
