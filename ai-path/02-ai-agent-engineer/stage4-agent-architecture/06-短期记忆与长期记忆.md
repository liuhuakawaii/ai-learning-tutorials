# 06 短期记忆与长期记忆

> 没有记忆的 Agent 每次对话都是陌生人——记忆让 AI 真正理解用户。

## 场景引入

用户第三次和 Agent 对话时说"还是按上次的格式来"，Agent 却完全不知道"上次的格式"是什么——它没有跨会话的记忆。每次对话都像第一次见面，用户不得不重复自己的偏好和背景信息。记忆系统让 Agent 从"健忘的工具"变成"了解你的助手"。

## 学习目标

- 设计 Agent 的短期记忆（会话内）和长期记忆（跨会话）
- 实现会话摘要、用户画像、向量记忆
- 理解记忆的存储、检索和更新策略

## 记忆类型

| 类型 | 作用域 | 存储 | 生命周期 |
|------|--------|------|----------|
| 工作记忆 | 当前推理步骤 | 内存 | 毫秒级 |
| 短期记忆 | 当前会话 | Redis | 小时级 |
| 长期记忆 | 跨会话 | 数据库 + 向量库 | 永久 |

## 短期记忆

```python
class ShortTermMemory:
    """短期记忆——会话级"""
    
    def __init__(self, redis: redis.Redis, session_id: str):
        self.redis = redis
        self.session_id = session_id
        self.key = f"memory:short:{session_id}"
        self.max_items = 100
        self.ttl = 3600 * 24  # 24 小时
    
    async def add(self, item: dict):
        """添加记忆项"""
        data = json.dumps(item, ensure_ascii=False)
        await self.redis.rpush(self.key, data)
        await self.redis.ltrim(self.key, -self.max_items, -1)
        await self.redis.expire(self.key, self.ttl)
    
    async def get_recent(self, n: int = 10) -> list[dict]:
        """获取最近 N 条记忆"""
        data = await self.redis.lrange(self.key, -n, -1)
        return [json.loads(d) for d in data]
    
    async def get_all(self) -> list[dict]:
        """获取所有短期记忆"""
        data = await self.redis.lrange(self.key, 0, -1)
        return [json.loads(d) for d in data]
    
    async def clear(self):
        """清除短期记忆"""
        await self.redis.delete(self.key)
```

## 长期记忆

```python
class LongTermMemory:
    """长期记忆——跨会话"""
    
    def __init__(self, db: AsyncSession, embedding: EmbeddingService):
        self.db = db
        self.embedding = embedding
    
    async def store(self, user_id: str, content: str, memory_type: str, metadata: dict = None):
        """存储长期记忆"""
        embedding = await self.embedding.embed(content)
        
        await self.db.execute(
            text("""
                INSERT INTO long_term_memories 
                (id, user_id, content, memory_type, embedding, metadata)
                VALUES (:id, :user_id, :content, :type, :embedding, :metadata)
            """),
            {
                "id": str(uuid.uuid4()),
                "user_id": user_id,
                "content": content,
                "type": memory_type,
                "embedding": str(embedding),
                "metadata": json.dumps(metadata or {}, ensure_ascii=False),
            },
        )
    
    async def recall(
        self,
        user_id: str,
        query: str,
        memory_type: str | None = None,
        top_k: int = 5,
    ) -> list[dict]:
        """根据查询召回相关记忆"""
        query_embedding = await self.embedding.embed(query)
        
        sql = """
            SELECT content, memory_type, metadata,
                   1 - (embedding <=> :query_embedding) AS score
            FROM long_term_memories
            WHERE user_id = :user_id
        """
        params = {
            "user_id": user_id,
            "query_embedding": str(query_embedding),
        }
        
        if memory_type:
            sql += " AND memory_type = :type"
            params["type"] = memory_type
        
        sql += " ORDER BY embedding <=> :query_embedding LIMIT :top_k"
        params["top_k"] = top_k
        
        result = await self.db.execute(text(sql), params)
        return [dict(row) for row in result]
```

## 参考答案

### 练习 1

**思路**：基于课程中的 ShortTermMemory 类，实现对话历史和工具调用结果的存储，以及最近 N 条记忆的检索。核心是理解 Redis List 的操作。

**答案**：

```python
import json
import redis.asyncio as redis

class ShortTermMemory:
    def __init__(self, redis_client: redis.Redis, session_id: str):
        self.redis = redis_client
        self.session_id = session_id
        self.key = f"memory:short:{session_id}"
        self.max_items = 100
        self.ttl = 3600 * 24

    async def add_message(self, role: str, content: str):
        item = {"type": "message", "role": role, "content": content, "timestamp": self._now()}
        await self._push(item)

    async def add_tool_call(self, tool_name: str, arguments: dict, result: str):
        item = {
            "type": "tool_call", "tool": tool_name,
            "arguments": arguments, "result": result, "timestamp": self._now(),
        }
        await self._push(item)

    async def get_recent(self, n: int = 10) -> list[dict]:
        data = await self.redis.lrange(self.key, -n, -1)
        return [json.loads(d) for d in data]

    async def get_all(self) -> list[dict]:
        data = await self.redis.lrange(self.key, 0, -1)
        return [json.loads(d) for d in data]

    async def clear(self):
        await self.redis.delete(self.key)

    async def _push(self, item: dict):
        await self.redis.rpush(self.key, json.dumps(item, ensure_ascii=False))
        await self.redis.ltrim(self.key, -self.max_items, -1)
        await self.redis.expire(self.key, self.ttl)

    def _now(self) -> str:
        from datetime import datetime
        return datetime.utcnow().isoformat()


# 使用示例
async def test_short_term_memory():
    r = redis.Redis(host="localhost", port=6379, decode_responses=True)
    memory = ShortTermMemory(r, session_id="session-001")

    await memory.add_message("user", "星辰科技专业版多少钱？")
    await memory.add_message("assistant", "专业版定价 299 元/月。")
    await memory.add_tool_call("search", {"query": "产品价格"}, "专业版 299 元/月，基础版 99 元/月")

    recent = await memory.get_recent(5)
    for item in recent:
        print(f"[{item['type']}] {item.get('content', item.get('tool', ''))}")
```

**要点**：
- 短期记忆用 Redis List 存储，RPUSH 追加，LRANGE 读取，LTRIM 限制长度
- 区分 message 和 tool_call 两种类型，方便后续组装上下文
- TTL 设为 24 小时，过期自动清除，避免存储无限增长
- 常见错误：不设 max_items，长对话导致内存溢出；不设 TTL，僵尸 session 占用存储

### 练习 2

**思路**：实现会话结束时自动生成摘要、存入长期记忆、下次对话时召回相关记忆的完整流程。

**答案**：

```python
class LongTermMemory:
    def __init__(self, db, embedding_service):
        self.db = db
        self.embedding = embedding_service

    async def store(self, user_id: str, content: str, memory_type: str, metadata: dict = None):
        emb = await self.embedding.embed(content)
        await self.db.execute(
            text("""
                INSERT INTO long_term_memories (id, user_id, content, memory_type, embedding, metadata)
                VALUES (:id, :uid, :content, :type, :emb, :meta)
            """),
            {
                "id": str(uuid.uuid4()), "uid": user_id,
                "content": content, "type": memory_type,
                "emb": str(emb), "meta": json.dumps(metadata or {}),
            },
        )

    async def recall(self, user_id: str, query: str, memory_type: str = None, top_k: int = 5) -> list[dict]:
        query_emb = await self.embedding.embed(query)
        sql = """
            SELECT content, memory_type, metadata, 1 - (embedding <=> :qemb) AS score
            FROM long_term_memories WHERE user_id = :uid
        """
        params = {"uid": user_id, "qemb": str(query_emb), "top_k": top_k}
        if memory_type:
            sql += " AND memory_type = :type"
            params["type"] = memory_type
        sql += " ORDER BY embedding <=> :qemb LIMIT :top_k"
        result = await self.db.execute(text(sql), params)
        return [dict(row) for row in result]


class ConversationSummarizer:
    async def summarize(self, messages: list[dict], llm) -> str:
        conversation = "\n".join(f"{m['role']}: {m['content']}" for m in messages if m.get("content"))
        prompt = f"请将以下对话压缩成简洁摘要，保留核心需求、关键决策和未完成任务：\n\n{conversation}\n\n摘要："
        response = await llm.chat(messages=[{"role": "user", "content": prompt}], model="gpt-4o-mini")
        return response.content


# 完整流程
async def end_session(session_id: str, user_id: str, short_memory: ShortTermMemory, long_memory: LongTermMemory, llm):
    # 1. 获取会话历史
    messages = await short_memory.get_all()
    if not messages:
        return

    # 2. 生成摘要
    summarizer = ConversationSummarizer()
    summary = await summarizer.summarize(messages, llm)

    # 3. 存入长期记忆
    await long_memory.store(user_id, summary, "conversation", {"session_id": session_id})

    # 4. 清除短期记忆
    await short_memory.clear()


async def start_session(user_id: str, query: str, long_memory: LongTermMemory, llm):
    # 召回相关记忆
    memories = await long_memory.recall(user_id, query, top_k=3)
    memory_context = "\n".join(f"- {m['content']}" for m in memories)

    prompt = f"用户历史偏好和对话记录：\n{memory_context}\n\n当前问题：{query}"
    return prompt
```

**要点**：
- 会话结束时生成摘要是关键时机，不要在每轮对话后都生成（太频繁、浪费 token）
- 长期记忆用向量检索，下次对话时根据当前问题召回最相关的记忆
- 常见错误：摘要太长（和原始对话差不多长），失去压缩意义；摘要太短（一句话），丢失关键信息

### 练习 3

**思路**：从对话中提取用户偏好（如"喜欢简洁回答"、"关注价格"），更新用户画像，在 System Prompt 中注入画像。

**答案**：

```python
class UserProfileManager:
    async def update_profile(self, user_id: str, conversation_summary: str, llm, memory: LongTermMemory):
        existing = await memory.recall(user_id, "用户画像", "user_profile", top_k=1)
        current_profile = existing[0]["content"] if existing else "暂无画像"

        prompt = f"""根据以下对话摘要，更新用户画像。

现有画像：
{current_profile}

最新对话摘要：
{conversation_summary}

请输出更新后的用户画像 JSON：
{{
    "name": "称呼",
    "preferences": ["偏好1", "偏好2"],
    "communication_style": "沟通风格",
    "focus_areas": ["关注领域1", "关注领域2"],
    "history_notes": "历史备注"
}}"""

        response = await llm.chat(
            messages=[{"role": "user", "content": prompt}],
            model="gpt-4o-mini",
            response_format={"type": "json_object"},
        )
        new_profile = response.content
        await memory.store(user_id, new_profile, "user_profile")


def build_system_prompt(user_profile: dict) -> str:
    return f"""你是一个智能助手。

用户画像：
- 称呼：{user_profile.get('name', '用户')}
- 偏好：{', '.join(user_profile.get('preferences', []))}
- 沟通风格：{user_profile.get('communication_style', '正常')}
- 关注领域：{', '.join(user_profile.get('focus_areas', []))}

请根据用户画像调整你的回答风格和内容。"""
```

**要点**：
- 用户画像用增量合并而非全量覆盖，避免丢失历史积累的偏好
- 画像更新要严格按 user_id 隔离，跨用户记忆泄露是严重安全事故
- 在 System Prompt 中注入画像，让 LLM 自动调整回答风格
- 常见错误：画像更新太频繁（每轮对话都更新），导致画像不稳定；应该在会话结束时统一更新

## 工程建议

- 短期记忆的 TTL 建议设为 24 小时，过长会浪费存储，过短会导致用户中断对话后丢失上下文
- 长期记忆存储前要做去重——相同信息反复存储会导致召回时冗余结果干扰排序
- 用户画像更新建议采用"增量合并"而非"全量覆盖"，避免丢失历史积累的偏好信息
- 记忆检索要做好用户隔离，严格按 user_id 过滤，跨用户记忆泄露是严重的安全事故

## 本节要点

- 短期记忆保证会话内连贯，长期记忆保证跨会话连贯
- 会话摘要把长对话压缩成有用信息
- 用户画像让 AI 越来越了解用户
- 向量检索让记忆召回更精准

## 常见误区

| 错误 | 原因 | 解决 |
|------|------|------|
| 记忆太多太杂 | 没有过滤和排序 | 按相关性和时间排序 |
| 用户画像不准 | 提取逻辑太简单 | 用 LLM 结构化提取 |
| 记忆冲突 | 新旧信息矛盾 | 更新时合并而非覆盖 |
| 记忆泄露 | 不同用户记忆串了 | 严格按 user_id 隔离 |
