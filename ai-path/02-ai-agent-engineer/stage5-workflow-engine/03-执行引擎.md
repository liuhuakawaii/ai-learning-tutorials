# 03 执行引擎

> 编辑器只是画图——执行引擎才是真正运行工作流的地方。

## 场景引入

你在可视化编辑器里画好了一个工作流：输入 → LLM 分析 → 条件判断 → 工具调用 → 输出。点下"执行"按钮后，系统需要知道先执行哪个节点、后执行哪个节点、节点之间怎么传递数据、某个节点超时了怎么办。如果执行顺序错了（比如在 LLM 调用之前就执行了依赖其输出的工具节点），整个工作流就废了。执行引擎就是解决这些问题的核心模块。

## 学习目标

- 实现 DAG 拓扑排序和执行调度
- 设计执行状态管理和错误处理
- 支持重试和超时

## 执行引擎核心

```python
# backend/app/workflow/engine.py
from collections import defaultdict, deque
import asyncio

class WorkflowEngine:
    """工作流执行引擎"""
    
    def __init__(self, llm: LLMService, tools: ToolRegistry):
        self.llm = llm
        self.tools = tools
        self.node_handlers: dict[str, NodeHandler] = {
            "llm": LLMNodeHandler(llm),
            "tool": ToolNodeHandler(tools),
            "condition": ConditionNodeHandler(),
            "human": HumanNodeHandler(),
            "input": InputNodeHandler(),
            "output": OutputNodeHandler(),
        }
    
    async def execute(
        self,
        workflow: WorkflowDefinition,
        inputs: dict,
        execution_id: str,
    ) -> dict:
        """执行工作流"""
        # 1. 拓扑排序
        execution_order = self._topological_sort(workflow)
        
        # 2. 初始化执行上下文
        context = ExecutionContext(
            execution_id=execution_id,
            workflow_id=workflow.id,
            variables={**workflow.variables, **inputs},
            node_results={},
        )
        
        # 3. 按顺序执行节点
        for node_id in execution_order:
            node = self._get_node(workflow, node_id)
            
            try:
                # 更新状态
                context.current_node = node_id
                await self._update_status(execution_id, "running", node_id)
                
                # 获取节点输入
                node_inputs = self._get_node_inputs(workflow, node, context)
                
                # 执行节点
                handler = self.node_handlers[node.type]
                result = await handler.execute(node, node_inputs, context)
                
                # 保存结果
                context.node_results[node_id] = result
                
                await self._update_status(execution_id, "completed", node_id)
            
            except HumanApprovalRequired as e:
                # 需要人工审批
                context.pending_approval = {
                    "node_id": node_id,
                    "content": e.content,
                }
                await self._update_status(execution_id, "waiting_for_human", node_id)
                return context.to_dict()
            
            except Exception as e:
                # 节点执行失败
                await self._update_status(execution_id, "failed", node_id, str(e))
                raise
        
        # 4. 返回最终结果
        return context.get_output()
    
    def _topological_sort(self, workflow: WorkflowDefinition) -> list[str]:
        """拓扑排序"""
        # 构建邻接表和入度表
        graph = defaultdict(list)
        in_degree = defaultdict(int)
        
        for edge in workflow.edges:
            graph[edge.source].append(edge.target)
            in_degree[edge.target] += 1
        
        # 所有节点初始化入度为 0
        for node in workflow.nodes:
            if node.id not in in_degree:
                in_degree[node.id] = 0
        
        # BFS 拓扑排序
        queue = deque([node_id for node_id, degree in in_degree.items() if degree == 0])
        order = []
        
        while queue:
            node_id = queue.popleft()
            order.append(node_id)
            
            for neighbor in graph[node_id]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    queue.append(neighbor)
        
        if len(order) != len(workflow.nodes):
            raise ValueError("Workflow contains a cycle")
        
        return order
```

class LLMNodeHandler:
    """LLM 节点处理器"""
    
    def __init__(self, llm: LLMService):
        self.llm = llm
    
    async def execute(
        self, node: NodeConfig, inputs: dict, context: ExecutionContext
    ) -> dict:
        # 渲染 Prompt（支持变量替换）
        prompt = self._render_prompt(node.config["prompt"], context.variables)
        
        # 调用 LLM
        response = await self.llm.chat(
            messages=[{"role": "user", "content": prompt}],
            model=node.config.get("model", "gpt-4o"),
            temperature=node.config.get("temperature", 0.7),
        )
        
        return {"content": response.content, "usage": response.__dict__}

class ConditionNodeHandler:
    """条件节点处理器"""
    
    async def execute(
        self, node: NodeConfig, inputs: dict, context: ExecutionContext
    ) -> dict:
        condition = node.config["expression"]
        
        # 安全地评估条件表达式
        result = self._evaluate_condition(condition, context.variables)
        
        # 根据结果选择输出端口
        return {"branch": "true" if result else "false", "value": result}
    
    def _evaluate_condition(self, expression: str, variables: dict) -> bool:
        """安全地评估条件表达式"""
        # 生产环境推荐使用 py-expression-eval 或 simpleeval 库
        # 这里用简单的白名单方式解析常见比较操作
        import re
        try:
            # 替换变量
            for key, value in variables.items():
                expression = expression.replace(f"{{{key}}}", repr(value))

            # 只允许安全的比较操作
            pattern = r"^(.+?)\s*(==|!=|>=|<=|>|<)\s*(.+)$"

## 执行上下文

```python
class ExecutionContext:
    """执行上下文"""
    
    def __init__(
        self,
        execution_id: str,
        workflow_id: str,
        variables: dict,
        node_results: dict,
    ):
        self.execution_id = execution_id
        self.workflow_id = workflow_id
        self.variables = variables
        self.node_results = node_results
        self.current_node: str | None = None
        self.pending_approval: dict | None = None
    
    def get_node_output(self, node_id: str) -> any:
        """获取节点输出"""
        return self.node_results.get(node_id, {})
    
    def get_output(self) -> dict:
        """获取工作流最终输出"""
        # 找到 output 节点的结果
        for node_id, result in self.node_results.items():
            if result.get("type") == "output":
                return result
        return self.node_results
    
    def to_dict(self) -> dict:
        return {
            "execution_id": self.execution_id,
            "workflow_id": self.workflow_id,
            "variables": self.variables,
            "node_results": {k: str(v) for k, v in self.node_results.items()},
        }
```

## 参考答案

### 练习 1

**思路**：拓扑排序的核心是 BFS（Kahn 算法）——从入度为 0 的节点开始，每次取出一个节点后将其邻居的入度减 1，减到 0 的入队。最终如果排序结果数量不等于节点总数，说明存在环路。

**答案**：

```python
from collections import defaultdict, deque
from dataclasses import dataclass, field

@dataclass
class Edge:
    source: str
    target: str

@dataclass
class WorkflowGraph:
    nodes: list[str]
    edges: list[Edge]

def topological_sort(workflow: WorkflowGraph) -> list[str]:
    """BFS 拓扑排序（Kahn 算法）"""
    # 1. 构建邻接表和入度表
    adjacency: dict[str, list[str]] = defaultdict(list)
    in_degree: dict[str, int] = {node: 0 for node in workflow.nodes}

    for edge in workflow.edges:
        adjacency[edge.source].append(edge.target)
        in_degree[edge.target] = in_degree.get(edge.target, 0) + 1

    # 2. 将所有入度为 0 的节点加入队列
    queue = deque([node_id for node_id, degree in in_degree.items() if degree == 0])
    order: list[str] = []

    # 3. BFS
    while queue:
        node_id = queue.popleft()
        order.append(node_id)
        for neighbor in adjacency[node_id]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    # 4. 检测环路
    if len(order) != len(workflow.nodes):
        cycle_nodes = [n for n in workflow.nodes if n not in order]
        raise ValueError(f"工作流存在环路，无法完成拓扑排序。环路涉及节点: {cycle_nodes}")

    return order


# --- 验证 ---
if __name__ == "__main__":
    # 正常 DAG
    graph = WorkflowGraph(
        nodes=["input", "llm", "condition", "tool_a", "tool_b", "output"],
        edges=[
            Edge("input", "llm"),
            Edge("llm", "condition"),
            Edge("condition", "tool_a"),
            Edge("condition", "tool_b"),
            Edge("tool_a", "output"),
            Edge("tool_b", "output"),
        ],
    )
    order = topological_sort(graph)
    print("执行顺序:", order)
    assert order.index("input") < order.index("llm")
    assert order.index("llm") < order.index("condition")
    assert order.index("condition") < order.index("tool_a")
    assert order.index("condition") < order.index("tool_b")

    # 环路检测
    cyclic = WorkflowGraph(
        nodes=["A", "B", "C"],
        edges=[Edge("A", "B"), Edge("B", "C"), Edge("C", "A")],
    )
    try:
        topological_sort(cyclic)
        print("ERROR: 应该检测到环路")
    except ValueError as e:
        print("正确检测到环路:", e)
```

**要点**：
- 入度为 0 的节点是"没有前置依赖"的起点，这是 BFS 的起点
- 排序结果数量 != 节点总数时一定存在环路，这是最可靠的检测方式
- 常见错误：只检查 `in_degree` 中存在的节点而遗漏孤立节点（入度始终为 0 但没有边连接的节点）

### 练习 2

**思路**：执行引擎的核心流程是：拓扑排序确定执行顺序 → 按顺序逐个执行节点 → 从 ExecutionContext 读取上游节点输出作为当前节点输入 → 异常时记录状态并中断。关键是节点间数据传递：每个节点执行完后将结果写入 context，下游节点通过边的连接关系从 context 读取上游输出。

**答案**：

```python
import asyncio
from dataclasses import dataclass, field
from typing import Any
from enum import Enum

class ExecutionStatus(str, Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"

@dataclass
class NodeResult:
    node_id: str
    status: ExecutionStatus
    output: Any = None
    error: str | None = None
    duration_ms: float = 0

@dataclass
class ExecutionContext:
    execution_id: str
    workflow_id: str
    variables: dict = field(default_factory=dict)
    node_results: dict[str, NodeResult] = field(default_factory=dict)

    def get_node_output(self, node_id: str) -> Any:
        result = self.node_results.get(node_id)
        return result.output if result else None

    def get_final_output(self) -> dict:
        return {k: v.output for k, v in self.node_results.items()}

class NodeHandler:
    """节点处理器基类"""
    async def execute(self, node: dict, inputs: dict, context: ExecutionContext) -> Any:
        raise NotImplementedError

class WorkflowEngine:
    """基础执行引擎"""

    def __init__(self, handlers: dict[str, NodeHandler]):
        self.handlers = handlers

    async def execute(
        self,
        workflow: dict,       # {"nodes": [...], "edges": [...]}
        inputs: dict,
        execution_id: str,
    ) -> ExecutionContext:
        nodes = {n["id"]: n for n in workflow["nodes"]}
        edges = workflow["edges"]

        # 1. 拓扑排序
        order = self._topological_sort(list(nodes.keys()), edges)

        # 2. 初始化上下文
        context = ExecutionContext(
            execution_id=execution_id,
            workflow_id=workflow.get("id", ""),
            variables=dict(inputs),
        )

        # 3. 按顺序执行
        for node_id in order:
            node = nodes[node_id]
            import time
            start = time.monotonic()
            try:
                # 收集输入：找到所有指向当前节点的边，取出源节点的输出
                node_inputs = {}
                for edge in edges:
                    if edge["target"] == node_id:
                        source_output = context.get_node_output(edge["source"])
                        if source_output is not None:
                            node_inputs[edge["source"]] = source_output

                # 执行节点
                handler = self.handlers.get(node["type"])
                if not handler:
                    raise ValueError(f"未知节点类型: {node['type']}")

                output = await asyncio.wait_for(
                    handler.execute(node, node_inputs, context),
                    timeout=node.get("timeout", 60),
                )

                duration = (time.monotonic() - start) * 1000
                context.node_results[node_id] = NodeResult(
                    node_id=node_id,
                    status=ExecutionStatus.COMPLETED,
                    output=output,
                    duration_ms=duration,
                )

            except asyncio.TimeoutError:
                duration = (time.monotonic() - start) * 1000
                context.node_results[node_id] = NodeResult(
                    node_id=node_id,
                    status=ExecutionStatus.FAILED,
                    error=f"节点执行超时（>{node.get('timeout', 60)}s）",
                    duration_ms=duration,
                )
                raise RuntimeError(f"节点 {node_id} 执行超时")

            except Exception as e:
                duration = (time.monotonic() - start) * 1000
                context.node_results[node_id] = NodeResult(
                    node_id=node_id,
                    status=ExecutionStatus.FAILED,
                    error=str(e),
                    duration_ms=duration,
                )
                raise

        return context

    def _topological_sort(self, node_ids: list[str], edges: list[dict]) -> list[str]:
        from collections import defaultdict, deque
        adjacency = defaultdict(list)
        in_degree = {nid: 0 for nid in node_ids}
        for edge in edges:
            adjacency[edge["source"]].append(edge["target"])
            in_degree[edge["target"]] += 1

        queue = deque([nid for nid, d in in_degree.items() if d == 0])
        order = []
        while queue:
            nid = queue.popleft()
            order.append(nid)
            for neighbor in adjacency[nid]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    queue.append(neighbor)

        if len(order) != len(node_ids):
            raise ValueError("工作流存在环路")
        return order
```

**要点**：
- 节点间数据传递的核心是：执行完的节点把输出存入 `context.node_results`，下游节点通过遍历 edges 找到源节点的输出
- 必须为每个节点设置超时，否则一个 LLM 调用卡住会阻塞整个工作流
- 常见错误：直接把所有上游输出作为 inputs 传给节点——应该只传"有边连接的"上游节点输出，否则会传入无关数据

## 常见误区

| 错误 | 原因 | 解决 |
|------|------|------|
| 死循环 | DAG 包含环路 | 拓扑排序前检测环路 |
| 数据丢失 | 节点输出没传递到下一节点 | 用 ExecutionContext 管理 |
| 超时 | 某个节点执行太久 | 设置节点超时 |

## 工程建议

- 每个节点都要设置超时时间，避免单个节点卡住整个工作流执行
- 执行上下文要独立管理——每次执行用独立的 ExecutionContext，避免并发执行时数据串扰
- 执行日志要持久化存储，记录每个节点的输入、输出、耗时和状态，方便事后排查
- 执行引擎要设计为无状态服务，执行状态存入数据库，支持暂停恢复和分布式部署
