# 05 人工审批节点

> 高风险操作必须停下来等人类确认——这是 AI 安全的底线。

## 场景引入

你的 AI 工作流要执行一个批量操作：根据 LLM 的分析结果，自动给 500 个用户发送退款通知。如果完全自动化，一旦 LLM 误判，就会给不该退款的用户发通知，造成资损和客诉。你需要在关键步骤加入人工审批——工作流执行到这一步自动暂停，通知审批人确认后才继续。审批节点是高风险操作的安全阀。

## 学习目标

- 实现人工审批节点
- 设计审批通知和超时机制
- 支持审批委托和批量审批

## 审批节点设计

```python
class HumanApprovalHandler:
    """人工审批节点处理器"""
    
    async def execute(self, node, inputs, context):
        approval_config = node.config
        
        # 创建审批请求
        approval_request = {
            "id": str(uuid.uuid4()),
            "execution_id": context.execution_id,
            "node_id": node.id,
            "title": approval_config.get("title", "需要审批"),
            "description": self._render_description(approval_config, context),
            "content": inputs,
            "approvers": approval_config.get("approvers", []),
            "timeout_minutes": approval_config.get("timeout", 60),
            "created_at": datetime.utcnow().isoformat(),
        }
        
        # 保存审批请求
        await self._save_approval(approval_request)
        
        # 发送通知
        await self._notify_approvers(approval_request)
        
        # 抛出暂停异常
        raise HumanApprovalRequired(approval_request)
```

```python
class ApprovalTimeoutWorker:
    """审批超时处理"""
    
    async def check_timeouts(self):
        """检查超时的审批请求"""
        result = await db.execute(
            text("""
                SELECT * FROM approval_requests
                WHERE status = 'pending'
                AND created_at + timeout_minutes * interval '1 minute' < now()
            """)
        )
        
        for approval in result:
            timeout_action = approval["timeout_action"]
            
            if timeout_action == "auto_approve":
                await self._auto_approve(approval)
            elif timeout_action == "auto_reject":
                await self._auto_reject(approval)
            elif timeout_action == "escalate":
                await self._escalate(approval)
```

## 参考答案

### 练习 1

**思路**：审批节点的核心是"暂停-通知-恢复"三步。执行到审批节点时抛出特殊异常让引擎暂停；同时创建审批记录并通知审批人；审批人操作后通过 API 恢复工作流执行。关键是审批状态要持久化到数据库，不能只存在内存里（否则服务重启就丢失了）。

**答案**：

```python
import uuid
from datetime import datetime, timezone
from enum import Enum

class ApprovalStatus(str, Enum):
    PENDING = "pending"
    APPROVED = "approved"
    REJECTED = "rejected"
    TIMEOUT = "timeout"

class HumanApprovalRequired(Exception):
    """审批暂停异常——引擎捕获后暂停工作流"""
    def __init__(self, approval_request: dict):
        self.approval_request = approval_request
        super().__init__(f"需要人工审批: {approval_request['title']}")


class HumanApprovalHandler:
    """人工审批节点处理器"""

    async def execute(self, node: dict, inputs: dict, context) -> dict:
        config = node.config

        approval_request = {
            "id": str(uuid.uuid4()),
            "execution_id": context.execution_id,
            "node_id": node["id"],
            "title": config.get("title", "需要审批"),
            "description": self._render_description(config, context),
            "content": inputs,
            "approvers": config.get("approvers", []),
            "timeout_minutes": config.get("timeout_minutes", 60),
            "timeout_action": config.get("timeout_action", "reject"),
            "status": ApprovalStatus.PENDING.value,
            "created_at": datetime.now(timezone.utc).isoformat(),
        }

        # 持久化到数据库
        await self._save_to_db(approval_request)

        # 通知审批人
        await self._notify_approvers(approval_request)

        # 抛出暂停异常，引擎会捕获并保存执行状态
        raise HumanApprovalRequired(approval_request)

    def _render_description(self, config: dict, context) -> str:
        template = config.get("description_template", "请审批以下操作")
        for key, value in context.variables.items():
            template = template.replace(f"{{{key}}}", str(value))
        return template

    async def _save_to_db(self, request: dict):
        """保存审批请求到数据库"""
        # INSERT INTO approval_requests (id, execution_id, node_id, title, ...)
        print(f"[DB] 保存审批请求: {request['id']}")

    async def _notify_approvers(self, request: dict):
        """通知审批人（邮件/企业微信/钉钉）"""
        for approver_id in request["approvers"]:
            print(f"[通知] 向 {approver_id} 发送审批通知: {request['title']}")


# 审批 API
from fastapi import APIRouter, HTTPException, Depends

router = APIRouter(prefix="/api/v1/approvals")

@router.get("/pending")
async def list_pending_approvals(user_id: str = Depends(get_current_user)):
    """获取当前用户的待审批列表"""
    approvals = await db.query(
        "SELECT * FROM approval_requests WHERE :uid = ANY(approvers) AND status = 'pending'",
        uid=user_id,
    )
    return approvals


@router.post("/{approval_id}/approve")
async def approve(approval_id: str, comment: str = "", user_id: str = Depends(get_current_user)):
    """审批通过"""
    approval = await db.get("approval_requests", approval_id)
    if not approval:
        raise HTTPException(404, "审批请求不存在")
    if user_id not in approval["approvers"]:
        raise HTTPException(403, "您不是该请求的审批人")
    if approval["status"] != ApprovalStatus.PENDING.value:
        raise HTTPException(400, f"审批已处理: {approval['status']}")

    # 更新审批状态
    await db.update("approval_requests", approval_id, {
        "status": ApprovalStatus.APPROVED.value,
        "decided_by": user_id,
        "comment": comment,
        "decided_at": datetime.now(timezone.utc).isoformat(),
    })

    # 恢复工作流执行
    await resume_workflow_execution(approval["execution_id"], {
        "approved": True,
        "approver": user_id,
        "comment": comment,
    })

    return {"ok": True, "message": "审批通过"}


@router.post("/{approval_id}/reject")
async def reject(approval_id: str, comment: str = "", user_id: str = Depends(get_current_user)):
    """审批拒绝"""
    approval = await db.get("approval_requests", approval_id)
    if not approval:
        raise HTTPException(404, "审批请求不存在")

    await db.update("approval_requests", approval_id, {
        "status": ApprovalStatus.REJECTED.value,
        "decided_by": user_id,
        "comment": comment,
        "decided_at": datetime.now(timezone.utc).isoformat(),
    })

    await resume_workflow_execution(approval["execution_id"], {
        "approved": False,
        "approver": user_id,
        "comment": comment,
    })

    return {"ok": True, "message": "审批已拒绝"}
```

**要点**：
- 审批请求必须持久化到数据库，不能只存内存——服务重启后要能恢复未完成的审批
- 抛出 `HumanApprovalRequired` 异常是让引擎暂停的关键机制，引擎捕获后保存当前执行状态
- 常见错误：审批通过后忘记调用 `resume_workflow_execution`，导致工作流永远卡在审批节点

### 练习 2

**思路**：超时处理需要一个定时任务（Worker）扫描超时的审批请求，根据配置的策略执行不同动作。超时策略有三种：自动通过（低风险场景）、自动拒绝（默认安全策略）、升级（转交更高级别审批人）。同时需要发送超时通知。

**答案**：

```python
import asyncio
from datetime import datetime, timezone, timedelta

class TimeoutAction(str, Enum):
    AUTO_APPROVE = "auto_approve"
    AUTO_REJECT = "auto_reject"
    ESCALATE = "escalate"

class ApprovalTimeoutWorker:
    """审批超时检查 Worker——定时扫描并处理超时审批"""

    def __init__(self, check_interval_seconds: int = 60):
        self.check_interval = check_interval_seconds
        self._running = False

    async def start(self):
        """启动定时检查"""
        self._running = True
        while self._running:
            try:
                await self.check_timeouts()
            except Exception as e:
                print(f"[超时检查] 异常: {e}")
            await asyncio.sleep(self.check_interval)

    def stop(self):
        self._running = False

    async def check_timeouts(self):
        """扫描超时的审批请求"""
        now = datetime.now(timezone.utc)
        expired = await db.query("""
            SELECT * FROM approval_requests
            WHERE status = 'pending'
            AND created_at + (timeout_minutes || ' minutes')::interval < :now
        """, now=now)

        for approval in expired:
            action = approval.get("timeout_action", "auto_reject")
            print(f"[超时] 审批 {approval['id']} 超时，执行策略: {action}")

            if action == TimeoutAction.AUTO_APPROVE:
                await self._auto_approve(approval, now)
            elif action == TimeoutAction.AUTO_REJECT:
                await self._auto_reject(approval, now)
            elif action == TimeoutAction.ESCALATE:
                await self._escalate(approval, now)

    async def _auto_approve(self, approval: dict, now: datetime):
        """超时自动通过"""
        await db.update("approval_requests", approval["id"], {
            "status": ApprovalStatus.APPROVED.value,
            "decided_by": "system_timeout",
            "comment": "审批超时，自动通过",
            "decided_at": now.isoformat(),
        })
        await resume_workflow_execution(approval["execution_id"], {
            "approved": True,
            "approver": "system_timeout",
            "comment": "审批超时，自动通过",
        })
        await self._send_timeout_notification(approval, "auto_approved")

    async def _auto_reject(self, approval: dict, now: datetime):
        """超时自动拒绝（默认安全策略）"""
        await db.update("approval_requests", approval["id"], {
            "status": ApprovalStatus.REJECTED.value,
            "decided_by": "system_timeout",
            "comment": "审批超时，自动拒绝",
            "decided_at": now.isoformat(),
        })
        await resume_workflow_execution(approval["execution_id"], {
            "approved": False,
            "approver": "system_timeout",
            "comment": "审批超时，自动拒绝",
        })
        await self._send_timeout_notification(approval, "auto_rejected")

    async def _escalate(self, approval: dict, now: datetime):
        """超时升级——转交上级审批人"""
        escalated_approvers = approval.get("escalated_approvers", [])
        if not escalated_approvers:
            # 没有配置升级审批人，降级为自动拒绝
            await self._auto_reject(approval, now)
            return

        # 更新审批人，重置计时
        await db.update("approval_requests", approval["id"], {
            "approvers": escalated_approvers,
            "timeout_minutes": approval.get("escalation_timeout_minutes", 30),
            "created_at": now.isoformat(),  # 重置开始时间
            "title": f"[升级] {approval['title']}",
        })

        # 通知新的审批人
        for approver_id in escalated_approvers:
            print(f"[升级通知] 向 {approver_id} 发送升级审批通知")
        await self._send_timeout_notification(approval, "escalated")

    async def _send_timeout_notification(self, approval: dict, action: str):
        """发送超时通知给相关人"""
        message = f"审批 {approval['id']} 已超时，处理方式: {action}"
        # 通知原始审批人
        for approver_id in approval["approvers"]:
            print(f"[通知] {approver_id}: {message}")
        # 通知工作流发起人
        print(f"[通知] 工作流发起人: {message}")
```

**要点**：
- 默认策略应该是 `auto_reject`（自动拒绝）——安全优先，宁可中断也不能未授权执行高风险操作
- 升级策略要处理"没有配置升级审批人"的边界情况，否则审批会永远卡住
- 常见错误：超时检查没有幂等性——同一批过期审批如果 Worker 重启后被重复扫描，会重复触发恢复操作。需要用 `status` 条件更新（CAS）来保证只处理一次

## 常见误区

| 错误 | 原因 | 解决 |
|------|------|------|
| 审批后不恢复 | 没触发工作流恢复 | 审批完成后调用 resume |
| 通知不到人 | 没配置通知渠道 | 集成邮件/企业微信 |
| 审批记录丢失 | 没持久化 | 存入数据库 |

## 工程建议

- 审批超时策略要明确：默认自动拒绝还是自动升级，不能让工作流永远卡在审批节点
- 审批通知要集成实际的消息渠道（邮件、企业微信、钉钉），不能只写数据库等人来查
- 审批操作要保证幂等性——同一个审批请求被重复提交时不能多次触发工作流恢复
- 审批数据要和工作流执行数据分开存储，审批记录需要独立的查询和审计能力
