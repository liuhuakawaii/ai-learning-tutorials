# 04 条件分支与循环

> 没有条件分支的工作流是"流水线"——有了条件才是"智能流程"。

## 场景引入

你的 AI 客服工作流已经能跑通基本流程了，但产品经理提出了新需求：当用户情绪评分低于 0.3 时自动转人工，高于 0.8 时直接发送满意度调查，中间的才走自动回复流程。同时，运营团队有一个待审核内容列表，需要批量对每条内容执行审核工作流。没有条件分支和循环，这些需求根本无法实现——工作流就只能是死板的流水线。

## 学习目标

- 实现条件分支节点
- 支持循环（foreach、while）
- 设计安全的表达式引擎

前端根据分支结果选择执行路径：

```python
def get_next_nodes(self, workflow, current_node_id, result):
    """根据节点结果获取下一个节点"""
    edges = [e for e in workflow.edges if e.source == current_node_id]
    
    if result.get("branch"):
        # 条件节点：根据分支选择
        branch = result["branch"]
        matching_edge = next((e for e in edges if e.source_handle == branch), None)
        return [matching_edge.target] if matching_edge else []
    else:
        # 普通节点：执行所有后续节点
        return [e.target for e in edges]
```

### While 循环

```python
class WhileNodeHandler:
    """While 循环节点"""
    
    async def execute(self, node, inputs, context):
        condition = node.config["condition"]
        max_iterations = node.config.get("max_iterations", 10)
        
        results = []
        iteration = 0
        
        while iteration < max_iterations:
            # 检查条件
            if not ExpressionEngine.evaluate(condition, context.variables):
                break
            
            # 执行循环体
            sub_workflow = node.config["sub_workflow"]
            result = await self._execute_sub_workflow(sub_workflow, context)
            results.append(result)
            
            # 更新变量
            context.variables.update(result.get("variables", {}))
            iteration += 1
        
        return {"iterations": iteration, "results": results}
```

## 表达式引擎

```python
class ExpressionEngine:
    """安全的表达式引擎——用白名单替代 eval()"""

    ALLOWED_OPERATORS = {
        "==": operator.eq,
        "!=": operator.ne,
        ">": operator.gt,
        "<": operator.lt,
        ">=": operator.ge,
        "<=": operator.le,
        "and": lambda a, b: a and b,  # 逻辑与（不是位运算 operator.and_）
        "or": lambda a, b: a or b,    # 逻辑或（不是位运算 operator.or_）
        "not": operator.not_,
        "in": lambda a, b: a in b,
        "contains": lambda a, b: b in a,
    }

    @classmethod
    def evaluate(cls, expression: str, variables: dict) -> bool:
        """安全地评估表达式——用白名单匹配替代 eval()"""
        import re
        # 替换变量
        for key, value in variables.items():
            expression = expression.replace(f"{{{key}}}", repr(value))

        # 解析简单比较表达式: left op right
        pattern = r"^(.+?)\s*(==|!=|>=|<=|>|<)\s*(.+)$"
        match = re.match(pattern, expression.strip())
        if match:
            left, op, right = match.groups()
            left_val = ast.literal_eval(left.strip())
            right_val = ast.literal_eval(right.strip())
            return bool(cls.ALLOWED_OPERATORS[op](left_val, right_val))

        # 布尔值直接返回
        if expression.strip().lower() in ("true", "1"):
            return True
        if expression.strip().lower() in ("false", "0", ""):
            return False

        raise ValueError(f"不支持的表达式: {expression}")
```

## 参考答案

### 练习 1

**思路**：条件分支节点的核心是安全地评估表达式并返回分支标识。不能用 `eval()`，而是用白名单解析器只允许比较操作符和逻辑操作符。前端根据 `branch` 字段（`"true"` / `"false"`）选择对应的边来决定下一个执行节点。

**答案**：

```python
import ast
import operator
import re

class ExpressionEngine:
    """安全的表达式引擎"""

    COMPARISON_OPS = {
        "==": operator.eq,
        "!=": operator.ne,
        ">": operator.gt,
        "<": operator.lt,
        ">=": operator.ge,
        "<=": operator.le,
    }

    LOGIC_OPS = {
        "and": lambda a, b: a and b,
        "or": lambda a, b: a or b,
    }

    @classmethod
    def evaluate(cls, expression: str, variables: dict) -> bool:
        """安全评估表达式，支持比较和逻辑组合"""
        # 替换变量占位符 {var_name}
        resolved = expression
        for key, value in variables.items():
            resolved = resolved.replace(f"{{{key}}}", repr(value))

        # 处理逻辑组合：拆分 and/or
        for logic_op, logic_fn in cls.LOGIC_OPS.items():
            # 按逻辑操作符分割（只处理顶层，不处理括号内的）
            parts = cls._split_by_logic(resolved, logic_op)
            if len(parts) > 1:
                results = [cls._eval_comparison(p.strip()) for p in parts]
                return logic_fn(*results)

        return cls._eval_comparison(resolved.strip())

    @classmethod
    def _eval_comparison(cls, expr: str) -> bool:
        """评估单个比较表达式"""
        # 尝试匹配 left op right
        pattern = r"^(.+?)\s*(==|!=|>=|<=|>|<)\s*(.+)$"
        match = re.match(pattern, expr)
        if not match:
            # 布尔值
            if expr.lower() in ("true", "1"):
                return True
            if expr.lower() in ("false", "0", ""):
                return False
            raise ValueError(f"无法解析表达式: {expr}")

        left_str, op, right_str = match.groups()
        left_val = ast.literal_eval(left_str.strip())
        right_val = ast.literal_eval(right_str.strip())
        return bool(cls.COMPARISON_OPS[op](left_val, right_val))

    @classmethod
    def _split_by_logic(cls, expr: str, logic_op: str) -> list[str]:
        """按逻辑操作符分割，避免拆分字符串内部"""
        parts = []
        current = ""
        tokens = expr.split()
        for token in tokens:
            if token == logic_op:
                parts.append(current)
                current = ""
            else:
                current += (" " if current else "") + token
        if current:
            parts.append(current)
        return parts if len(parts) > 1 else [expr]


class ConditionNodeHandler:
    """条件分支节点处理器"""

    async def execute(self, node: dict, inputs: dict, context) -> dict:
        expression = node.config["expression"]
        result = ExpressionEngine.evaluate(expression, context.variables)
        return {
            "branch": "true" if result else "false",
            "value": result,
        }


def get_next_nodes(workflow, current_node_id: str, result: dict) -> list[str]:
    """根据条件分支结果获取下一个节点"""
    edges = [e for e in workflow["edges"] if e["source"] == current_node_id]

    if "branch" in result:
        branch = result["branch"]
        matching = next((e for e in edges if e.get("source_handle") == branch), None)
        return [matching["target"]] if matching else []
    else:
        return [e["target"] for e in edges]


# --- 验证 ---
if __name__ == "__main__":
    ctx_variables = {"score": 0.85, "status": "active"}

    # 简单条件
    assert ExpressionEngine.evaluate("{score} > 0.5", ctx_variables) is True
    assert ExpressionEngine.evaluate("{score} < 0.3", ctx_variables) is False
    assert ExpressionEngine.evaluate("{status} == 'active'", ctx_variables) is True

    # 逻辑组合
    assert ExpressionEngine.evaluate("{score} > 0.5 and {status} == 'active'", ctx_variables) is True
    assert ExpressionEngine.evaluate("{score} > 0.9 or {status} == 'active'", ctx_variables) is True
    assert ExpressionEngine.evaluate("{score} > 0.9 and {status} == 'inactive'", ctx_variables) is False

    print("所有测试通过")
```

**要点**：
- 表达式引擎必须用白名单方式解析，绝对不能用 `eval()`——用户输入的表达式可能是 `__import__('os').system('rm -rf /')`
- 逻辑组合（and/or）的解析要注意运算优先级，简单实现可以先按操作符分割再逐个评估
- 常见错误：变量替换时直接拼接字符串而不用 `repr()`，导致字符串类型的值缺少引号，`ast.literal_eval` 解析失败

### 练习 2

**思路**：循环的核心是遍历集合或重复执行直到条件不满足。foreach 需要将列表中的每个元素注入子工作流的上下文；while 需要每轮重新评估条件。两者都必须有最大迭代次数限制，防止无限循环。

**答案**：

```python
import asyncio
from typing import Any

class ForeachNodeHandler:
    """Foreach 循环节点处理器"""

    async def execute(self, node: dict, inputs: dict, context) -> dict:
        list_var = node.config["list_variable"]
        items = context.variables.get(list_var, [])
        sub_workflow = node.config["sub_workflow"]
        max_items = node.config.get("max_items", 100)

        if len(items) > max_items:
            raise ValueError(f"列表长度 {len(items)} 超过限制 {max_items}")

        results = []
        errors = []
        for index, item in enumerate(items[:max_items]):
            loop_vars = {
                **context.variables,
                "loop_item": item,
                "loop_index": index,
                "loop_total": len(items),
            }
            try:
                sub_engine = WorkflowEngine(handlers=context.handlers)
                sub_context = await sub_engine.execute(
                    sub_workflow,
                    inputs=loop_vars,
                    execution_id=f"{context.execution_id}_foreach_{index}",
                )
                results.append({"index": index, "item": item, "output": sub_context.get_final_output()})
            except Exception as e:
                errors.append({"index": index, "item": item, "error": str(e)})
                if node.config.get("fail_fast", False):
                    raise

        return {"results": results, "errors": errors, "total": len(items)}


class WhileNodeHandler:
    """While 循环节点处理器"""

    async def execute(self, node: dict, inputs: dict, context) -> dict:
        condition = node.config["condition"]
        sub_workflow = node.config["sub_workflow"]
        max_iterations = node.config.get("max_iterations", 10)

        results = []
        iteration = 0

        while iteration < max_iterations:
            # 每轮重新评估条件
            should_continue = ExpressionEngine.evaluate(condition, context.variables)
            if not should_continue:
                break

            sub_engine = WorkflowEngine(handlers=context.handlers)
            sub_context = await sub_engine.execute(
                sub_workflow,
                inputs=dict(context.variables),
                execution_id=f"{context.execution_id}_while_{iteration}",
            )
            result = sub_context.get_final_output()
            results.append({"iteration": iteration, "output": result})

            # 更新上下文变量（子工作流可能修改了变量）
            context.variables.update(result.get("variables", {}))
            iteration += 1

        reached_limit = iteration >= max_iterations
        return {
            "iterations": iteration,
            "results": results,
            "reached_max_iterations": reached_limit,
        }
```

**要点**：
- `max_iterations` 是 while 循环的安全阀，必须设置且不能太大——建议默认 10，最大不超过 100
- foreach 要限制列表长度（`max_items`），防止用户传入超大列表导致资源耗尽
- 常见错误：while 循环的条件变量没有在循环体内更新，导致条件永远为真——子工作流的输出必须回写到 `context.variables`

## 常见误区

| 错误 | 原因 | 解决 |
|------|------|------|
| 无限循环 | 条件永远为真 | 设置 max_iterations |
| 表达式注入 | 用户输入被 eval | 用安全的表达式解析器 |
| 分支不生效 | source_handle 配置错误 | 检查边的 handle 配置 |

## 工程建议

- 表达式引擎要用安全的白名单解析器，绝对不能用 eval()——用户输入的表达式可能包含恶意代码
- while 循环必须设置 max_iterations，防止条件永远为真导致无限循环
- 条件表达式要有完善的单元测试，覆盖各种边界情况（空值、类型不匹配、特殊字符）
- 循环节点要支持中断和取消——当循环体内的子工作流出错时，能快速停止而不是继续执行
