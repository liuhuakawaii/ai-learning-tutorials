# 02 可视化编辑器

> 让非开发人员也能编排 AI 工作流——可视化是关键。

## 场景引入

运营团队需要配置一个内容审核工作流：先用 LLM 判断内容是否违规，违规的自动标记，边界 case 转人工审核。如果每次调整流程都要找开发改代码、改配置文件、重新部署，一个需求从提出到上线要一周。你需要一个可视化的 DAG 编辑器，让运营人员自己拖拽节点、连线、配置参数，像画流程图一样编排 AI 任务。

## 学习目标

- 使用 Vue Flow 实现 DAG 可视化编辑器
- 实现节点拖拽、连线、配置
- 设计属性面板和节点库

## 技术选型

| 库 | 特点 | 适用 |
|------|------|------|
| **Vue Flow** | Vue 3 原生、活跃维护 | **本课程推荐** |
| X6 (AntV) | 功能强大、文档完善 | 备选 |
| Rete.js | 节点编辑器框架 | 复杂场景 |

## 基础编辑器

```vue
<!-- frontend/src/views/WorkflowEditor.vue -->
<template>
  <div class="workflow-editor">
    <!-- 节点库 -->
    <div class="node-palette">
      <div class="palette-title">节点库</div>
      <div
        v-for="nodeType in nodeTypes"
        :key="nodeType.type"
        class="palette-item"
        draggable="true"
        @dragstart="onDragStart($event, nodeType)"
      >
        <n-icon :size="20"><component :is="nodeType.icon" /></n-icon>
        <span>{{ nodeType.label }}</span>
      </div>
    </div>

    <!-- 画布 -->
    <VueFlow
      v-model:nodes="nodes"
      v-model:edges="edges"
      :default-viewport="{ zoom: 1, x: 0, y: 0 }"
      @nodes-change="onNodesChange"
      @edges-change="onEdgesChange"
      @connect="onConnect"
      @drop="onDrop"
      @dragover.prevent
    >
      <!-- 自定义节点 -->
      <template #node-llm="props">
        <LLMNode v-bind="props" />
      </template>
      <template #node-tool="props">
        <ToolNode v-bind="props" />
      </template>
      <template #node-condition="props">
        <ConditionNode v-bind="props" />
      </template>
    </VueFlow>

    <!-- 属性面板 -->
    <div v-if="selectedNode" class="properties-panel">
      <div class="panel-title">{{ selectedNode.data.label }}</div>
      <component
        :is="getPropertiesComponent(selectedNode.type)"
        :node="selectedNode"
        @update="updateNodeConfig"
      />
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, markRaw } from 'vue'
import { VueFlow, useVueFlow } from '@vue-flow/core'
import '@vue-flow/core/dist/style.css'

const nodeTypes = [
  { type: 'input', label: '输入', icon: InputIcon },
  { type: 'llm', label: 'LLM 调用', icon: ChatIcon },
  { type: 'tool', label: '工具调用', icon: ToolIcon },
  { type: 'condition', label: '条件判断', icon: BranchIcon },
  { type: 'human', label: '人工审批', icon: PersonIcon },
  { type: 'output', label: '输出', icon: OutputIcon },
]

const nodes = ref([])
const edges = ref([])
const selectedNode = ref(null)

function onDragStart(event: DragEvent, nodeType: any) {
  event.dataTransfer.setData('application/vueflow', nodeType.type)
}

function onDrop(event: DragEvent) {
  const type = event.dataTransfer.getData('application/vueflow')
  const position = { x: event.offsetX, y: event.offsetY }
  
  nodes.value.push({
    id: `node-${Date.now()}`,
    type,
    position,
    data: { label: `新${nodeType.label}节点`, config: {} },
  })
}

function onConnect(params: any) {
  edges.value.push({
    id: `edge-${Date.now()}`,
    source: params.source,
    target: params.target,
    sourceHandle: params.sourceHandle,
  })
}
</script>
```

## 参考答案

### 练习 1

**思路**：基于 Vue Flow 的 `useVueFlow` 组合式 API 实现核心交互——拖拽通过 HTML5 Drag API 的 `dragstart` / `drop` 事件完成节点创建，连线通过 `@connect` 事件完成边的添加，选中通过 `@nodeClick` 事件记录当前节点，删除通过键盘 Delete 键或右键菜单触发。

**答案**：

```vue
<!-- frontend/src/views/WorkflowEditor.vue -->
<template>
  <div class="workflow-editor">
    <div class="node-palette">
      <div class="palette-title">节点库</div>
      <div
        v-for="nodeType in nodeTypes"
        :key="nodeType.type"
        class="palette-item"
        draggable="true"
        @dragstart="onDragStart($event, nodeType)"
      >
        <span>{{ nodeType.label }}</span>
      </div>
    </div>

    <VueFlow
      v-model:nodes="nodes"
      v-model:edges="edges"
      :default-viewport="{ zoom: 1, x: 0, y: 0 }"
      @connect="onConnect"
      @node-click="onNodeClick"
      @pane-click="onPaneClick"
      @drop="onDrop"
      @dragover.prevent
    >
      <template #node-llm="props">
        <LLMNode v-bind="props" />
      </template>
      <template #node-tool="props">
        <ToolNode v-bind="props" />
      </template>
      <template #node-condition="props">
        <ConditionNode v-bind="props" />
      </template>
    </VueFlow>

    <div v-if="selectedNode" class="properties-panel">
      <div class="panel-header">
        <span>{{ selectedNode.data.label }}</span>
        <n-button size="small" type="error" @click="deleteSelectedNode">删除节点</n-button>
      </div>
      <component
        :is="getPropertiesComponent(selectedNode.type)"
        :node="selectedNode"
        @update="updateNodeConfig"
      />
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { VueFlow, useVueFlow, type Node, type Edge } from '@vue-flow/core'
import '@vue-flow/core/dist/style.css'

const nodeTypes = [
  { type: 'input', label: '输入' },
  { type: 'llm', label: 'LLM 调用' },
  { type: 'tool', label: '工具调用' },
  { type: 'condition', label: '条件判断' },
  { type: 'human', label: '人工审批' },
  { type: 'output', label: '输出' },
]

const nodes = ref<Node[]>([])
const edges = ref<Edge[]>([])
const selectedNode = ref<Node | null>(null)

const { removeNodes, removeEdges, getConnectedEdges } = useVueFlow()

function onDragStart(event: DragEvent, nodeType: { type: string; label: string }) {
  event.dataTransfer!.setData('application/vueflow', nodeType.type)
  event.dataTransfer!.effectAllowed = 'move'
}

function onDrop(event: DragEvent) {
  const type = event.dataTransfer!.getData('application/vueflow')
  if (!type) return

  const bounds = (event.currentTarget as HTMLElement).getBoundingClientRect()
  const position = { x: event.clientX - bounds.left, y: event.clientY - bounds.top }

  const label = nodeTypes.find(n => n.type === type)?.label ?? type
  nodes.value.push({
    id: `node-${Date.now()}`,
    type,
    position,
    data: { label: `新${label}节点`, config: {} },
  })
}

function onConnect(params: any) {
  edges.value.push({
    id: `edge-${Date.now()}`,
    source: params.source,
    target: params.target,
    sourceHandle: params.sourceHandle,
    targetHandle: params.targetHandle,
  })
}

function onNodeClick({ node }: { node: Node }) {
  selectedNode.value = node
}

function onPaneClick() {
  selectedNode.value = null
}

function deleteSelectedNode() {
  if (!selectedNode.value) return
  const nodeId = selectedNode.value.id
  const connected = getConnectedEdges(nodeId)
  removeEdges(connected)
  removeNodes([nodeId])
  selectedNode.value = null
}
</script>
```

**要点**：
- 拖拽坐标需要用 `event.clientX - bounds.left` 转换为画布坐标，直接用 `offsetX` 在缩放和平移后会偏移
- 常见错误：删除节点时只 `removeNodes` 不删除关联的边，导致画面上残留悬空连线，后端执行时也会报错
- `@pane-click`（点击空白区域）用来取消选中，否则用户以为节点还在编辑状态

### 练习 2

**思路**：每种节点类型对应一个属性面板组件，通过 `type -> component` 映射动态渲染。面板内用 Naive UI 的表单组件收集配置，校验通过后 emit `update` 事件同步到节点的 `data.config`。

**答案**：

```vue
<!-- frontend/src/components/panels/LLMPropertiesPanel.vue -->
<template>
  <n-form :model="form" label-placement="top" size="small">
    <n-form-item label="模型" path="model" :rule="{ required: true, message: '请选择模型' }">
      <n-select
        v-model:value="form.model"
        :options="modelOptions"
        placeholder="选择 LLM 模型"
      />
    </n-form-item>
    <n-form-item label="Prompt 模板" path="prompt_template" :rule="{ required: true, message: 'Prompt 不能为空' }">
      <n-input
        v-model:value="form.prompt_template"
        type="textarea"
        :rows="4"
        placeholder="输入 Prompt，支持 ${variable} 插值"
      />
    </n-form-item>
    <n-form-item label="Temperature">
      <n-slider v-model:value="form.temperature" :min="0" :max="2" :step="0.1" />
    </n-form-item>
    <n-button type="primary" @click="onSave">保存配置</n-button>
  </n-form>
</template>

<script setup lang="ts">
import { reactive } from 'vue'

const props = defineProps<{ node: any }>()
const emit = defineEmits<{ update: [config: any] }>()

const modelOptions = [
  { label: 'GPT-4o', value: 'gpt-4o' },
  { label: 'GPT-4o-mini', value: 'gpt-4o-mini' },
  { label: 'Claude 3.5 Sonnet', value: 'claude-3-5-sonnet' },
]

const form = reactive({
  model: props.node.data.config?.model ?? 'gpt-4o',
  prompt_template: props.node.data.config?.prompt_template ?? '',
  temperature: props.node.data.config?.temperature ?? 0.7,
})

function onSave() {
  if (!form.prompt_template.trim()) return
  emit('update', { ...form })
}
</script>
```

```vue
<!-- frontend/src/components/panels/ToolPropertiesPanel.vue -->
<template>
  <n-form :model="form" label-placement="top" size="small">
    <n-form-item label="工具" path="tool_name" :rule="{ required: true, message: '请选择工具' }">
      <n-select
        v-model:value="form.tool_name"
        :options="toolOptions"
        placeholder="选择工具"
      />
    </n-form-item>
    <n-form-item label="参数映射">
      <div v-for="(mapping, idx) in form.params_mapping" :key="idx" class="mapping-row">
        <n-input v-model:value="mapping.key" placeholder="参数名" style="width: 40%" />
        <n-input v-model:value="mapping.value" placeholder="${变量名}" style="width: 50%" />
        <n-button size="small" @click="form.params_mapping.splice(idx, 1)">删</n-button>
      </div>
      <n-button size="small" dashed @click="form.params_mapping.push({ key: '', value: '' })">
        添加参数
      </n-button>
    </n-form-item>
    <n-button type="primary" @click="onSave">保存配置</n-button>
  </n-form>
</template>

<script setup lang="ts">
import { reactive } from 'vue'

const props = defineProps<{ node: any }>()
const emit = defineEmits<{ update: [config: any] }>()

const toolOptions = [
  { label: 'web_search - 搜索', value: 'web_search' },
  { label: 'calculator - 计算', value: 'calculator' },
  { label: 'query_order - 查订单', value: 'query_order' },
]

const savedMapping = props.node.data.config?.params_mapping ?? {}
const form = reactive({
  tool_name: props.node.data.config?.tool_name ?? '',
  params_mapping: Object.entries(savedMapping).map(([key, value]) => ({ key, value: value as string })),
})

function onSave() {
  const params_mapping: Record<string, string> = {}
  for (const m of form.params_mapping) {
    if (m.key) params_mapping[m.key] = m.value
  }
  emit('update', { tool_name: form.tool_name, params_mapping })
}
</script>
```

```vue
<!-- frontend/src/components/panels/ConditionPropertiesPanel.vue -->
<template>
  <n-form :model="form" label-placement="top" size="small">
    <n-form-item label="条件表达式" path="expression" :rule="{ required: true, message: '表达式不能为空' }">
      <n-input
        v-model:value="form.expression"
        type="textarea"
        :rows="3"
        placeholder='如: intent == "refund" or confidence > 0.8'
      />
    </n-form-item>
    <n-form-item label="true 分支标签">
      <n-input v-model:value="form.true_label" placeholder="是" />
    </n-form-item>
    <n-form-item label="false 分支标签">
      <n-input v-model:value="form.false_label" placeholder="否" />
    </n-form-item>
    <n-button type="primary" @click="onSave">保存配置</n-button>
  </n-form>
</template>

<script setup lang="ts">
import { reactive } from 'vue'

const props = defineProps<{ node: any }>()
const emit = defineEmits<{ update: [config: any] }>()

const form = reactive({
  expression: props.node.data.config?.expression ?? '',
  true_label: props.node.data.config?.true_label ?? '是',
  false_label: props.node.data.config?.false_label ?? '否',
})

function onSave() {
  if (!form.expression.trim()) return
  emit('update', { ...form })
}
</script>
```

**要点**：
- 属性面板的核心是"读节点 config 初始化表单 → 用户编辑 → emit update 回写到节点 data"，保持单向数据流
- 常见错误：属性面板直接修改 `props.node.data.config`（mutate props），Vue 会报警告且状态不可预测；正确做法是 emit 事件让父组件更新
- 条件节点的表达式要在前端做基础校验（非空、括号匹配），但真正的语义校验（变量是否存在）应由后端在保存时执行

## 常见误区

- 一开始就追求复杂的自定义节点样式，忽略了基础的拖拽和连线功能的稳定性
- 属性面板不做表单校验，用户输入非法配置后执行时才报错
- 忽略编辑器的撤销/重做和自动保存，用户刷新页面就丢失所有工作
- 节点配置的序列化格式和后端不一致，保存后执行引擎无法解析

## 工程建议

- 先实现核心交互（拖拽、连线、删除），再逐步完善属性面板和自定义节点样式
- 属性面板要配合 JSON Schema 自动生成表单，并做实时校验和错误提示
- 编辑器状态要做好持久化——支持撤销重做、自动保存、导入导出
- 节点配置的序列化和反序列化要保持前后端一致，建议用统一的 Schema 定义

## 本节要点

- Vue Flow 是 Vue 3 生态最好的 DAG 编辑器
- 自定义节点让编辑器更直观
- 属性面板是配置节点的关键交互
- 拖拽 + 连线 = 非技术用户也能编排工作流
