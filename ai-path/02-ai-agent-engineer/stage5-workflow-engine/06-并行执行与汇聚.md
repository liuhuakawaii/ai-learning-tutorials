# 06 并行执行与汇聚

> 串行太慢——并行执行能让工作流速度提升数倍。

## 场景引入

你的 AI 数据分析平台有一个工作流：用户上传一份报告，需要同时调用三个外部 API——一个提取关键词、一个生成摘要、一个翻译成英文。如果串行执行，每个 API 调用需要 3 秒，总共要 9 秒。但这三个任务互不依赖，完全可以并行执行，总耗时只需要 3 秒。并行执行和结果汇聚是提升工作流性能的关键手段。

## 学习目标

- 实现并行执行（扇出/扇入）
- 设计结果汇聚和失败策略
- 处理并行执行的超时和错误

## 并行执行模型

```python
class ParallelNodeHandler:
    """并行执行节点"""
    
    async def execute(self, node, inputs, context):
        branches = node.config["branches"]  # 并行分支配置
        strategy = node.config.get("strategy", "wait_all")  # wait_all / wait_any / race
        
        tasks = []
        for branch in branches:
            sub_workflow = branch["workflow"]
            task = asyncio.create_task(
                self._execute_branch(sub_workflow, inputs, context)
            )
            tasks.append((branch["name"], task))
        
        if strategy == "wait_all":
            # 等待所有分支完成
            results = {}
            for name, task in tasks:
                try:
                    results[name] = await asyncio.wait_for(task, timeout=300)
                except asyncio.TimeoutError:
                    results[name] = {"error": "timeout"}
            return {"results": results}
        
        elif strategy == "wait_any":
            # 等待任意一个完成
            done, pending = await asyncio.wait(
                [t for _, t in tasks], return_when=asyncio.FIRST_COMPLETED
            )
            # 取消未完成的
            for task in pending:
                task.cancel()
            return {"result": done.pop().result()}
        
        elif strategy == "race":
            # 竞争：取最快的
            return await self._race(tasks)
```

## 结果汇聚

```python
class AggregatorNodeHandler:
    """结果汇聚节点"""
    
    async def execute(self, node, inputs, context):
        strategy = node.config.get("aggregation_strategy", "concatenate")
        
        if strategy == "concatenate":
            # 拼接所有结果
            all_results = []
            for key, value in inputs.items():
                if isinstance(value, dict) and "results" in value:
                    all_results.extend(value["results"])
            return {"aggregated": all_results}
        
        elif strategy == "merge":
            # 合并字典
            merged = {}
            for key, value in inputs.items():
                if isinstance(value, dict):
                    merged.update(value)
            return {"aggregated": merged}
        
        elif strategy == "llm_merge":
            # 用 LLM 合并结果
            prompt = f"请合并以下结果：\n\n{json.dumps(inputs, ensure_ascii=False)}"
            response = await self.llm.chat(
                messages=[{"role": "user", "content": prompt}],
                model="gpt-4o-mini",
            )
            return {"aggregated": response.content}
```

## 失败策略

```python
class FailureStrategy(str, Enum):
    FAIL_FAST = "fail_fast"          # 一个失败就全部失败
    CONTINUE = "continue"            # 忽略失败继续
    RETRY = "retry"                  # 重试失败的分支

async def execute_with_strategy(self, tasks, strategy, max_retries=3):
    if strategy == FailureStrategy.FAIL_FAST:
        # 任意一个失败就抛异常
        results = await asyncio.gather(*[t for _, t in tasks], return_exceptions=False)
        return dict(zip([n for n, _ in tasks], results))
    
    elif strategy == FailureStrategy.CONTINUE:
        # 忽略失败
        results = {}
        for name, task in tasks:
            try:
                results[name] = await task
            except Exception as e:
                results[name] = {"error": str(e)}
        return results
    
    elif strategy == FailureStrategy.RETRY:
        # 重试失败的
        results = {}
        for name, task in tasks:
            for attempt in range(max_retries):
                try:
                    results[name] = await task
                    break
                except Exception:
                    if attempt == max_retries - 1:
                        results[name] = {"error": "max retries exceeded"}
                    await asyncio.sleep(2 ** attempt)
        return results
```

## 参考答案

### 练习 1

**思路**：并行执行的核心是用 `asyncio.create_task` 同时启动多个分支，然后根据汇聚策略决定何时收集结果。`wait_all` 等所有分支完成；`wait_any` 取第一个完成的结果并取消其余；`race` 竞争最快的结果。每个分支需要独立的超时控制。

**答案**：

```python
import asyncio
from enum import Enum
from typing import Any

class ParallelStrategy(str, Enum):
    WAIT_ALL = "wait_all"
    WAIT_ANY = "wait_any"
    RACE = "race"

class ParallelNodeHandler:
    """并行执行节点处理器"""

    async def execute(self, node: dict, inputs: dict, context) -> dict:
        branches = node.config["branches"]
        strategy = ParallelStrategy(node.config.get("strategy", "wait_all"))
        branch_timeout = node.config.get("branch_timeout", 300)

        # 为每个分支创建异步任务
        tasks: list[tuple[str, asyncio.Task]] = []
        for branch in branches:
            task = asyncio.create_task(
                self._execute_branch(branch, inputs, context),
                name=branch["name"],
            )
            tasks.append((branch["name"], task))

        if strategy == ParallelStrategy.WAIT_ALL:
            return await self._wait_all(tasks, branch_timeout)
        elif strategy == ParallelStrategy.WAIT_ANY:
            return await self._wait_any(tasks, branch_timeout)
        elif strategy == ParallelStrategy.RACE:
            return await self._race(tasks, branch_timeout)

    async def _wait_all(self, tasks: list[tuple[str, asyncio.Task]], timeout: float) -> dict:
        """等待所有分支完成，部分超时则标记错误"""
        results = {}
        for name, task in tasks:
            try:
                results[name] = await asyncio.wait_for(asyncio.shield(task), timeout=timeout)
            except asyncio.TimeoutError:
                task.cancel()
                results[name] = {"error": f"分支 {name} 执行超时（>{timeout}s）"}
            except Exception as e:
                results[name] = {"error": str(e)}

        all_success = all("error" not in v for v in results.values() if isinstance(v, dict))
        return {"results": results, "all_success": all_success}

    async def _wait_any(self, tasks: list[tuple[str, asyncio.Task]], timeout: float) -> dict:
        """等待任意一个完成，取消其余"""
        task_objs = [t for _, t in tasks]
        names = [n for n, _ in tasks]

        done, pending = await asyncio.wait(
            task_objs, return_when=asyncio.FIRST_COMPLETED, timeout=timeout
        )

        # 取消未完成的
        for task in pending:
            task.cancel()
        for task in pending:
            try:
                await task
            except asyncio.CancelledError:
                pass

        if not done:
            return {"result": None, "error": "所有分支均超时"}

        completed_task = done.pop()
        idx = task_objs.index(completed_task)
        return {"result": completed_task.result(), "completed_branch": names[idx]}

    async def _race(self, tasks: list[tuple[str, asyncio.Task]], timeout: float) -> dict:
        """竞争：取最快的成功结果，忽略失败"""
        task_objs = [t for _, t in tasks]
        names = [n for n, _ in tasks]

        for coro in asyncio.as_completed(task_objs, timeout=timeout):
            try:
                result = await coro
                # 取消其余
                for task in task_objs:
                    if not task.done():
                        task.cancel()
                return {"result": result, "winner": names[task_objs.index(coro)]}
            except Exception:
                continue

        return {"result": None, "error": "所有分支均失败或超时"}

    async def _execute_branch(self, branch: dict, inputs: dict, context) -> Any:
        """执行单个并行分支"""
        sub_workflow = branch["workflow"]
        engine = WorkflowEngine(handlers=context.handlers)
        sub_context = await engine.execute(
            sub_workflow,
            inputs={**inputs, **branch.get("extra_inputs", {})},
            execution_id=f"{context.execution_id}_parallel_{branch['name']}",
        )
        return sub_context.get_final_output()


class AggregatorNodeHandler:
    """结果汇聚节点处理器"""

    async def execute(self, node: dict, inputs: dict, context) -> dict:
        strategy = node.config.get("strategy", "concatenate")

        if strategy == "concatenate":
            all_results = []
            for value in inputs.values():
                if isinstance(value, dict) and "results" in value:
                    all_results.extend(value["results"])
                else:
                    all_results.append(value)
            return {"aggregated": all_results}

        elif strategy == "merge":
            merged = {}
            for value in inputs.values():
                if isinstance(value, dict):
                    merged.update(value)
            return {"aggregated": merged}

        elif strategy == "llm_merge":
            prompt = f"请合并以下并行执行的结果为一个连贯的输出：\n\n{json.dumps(inputs, ensure_ascii=False, indent=2)}"
            response = await context.llm.chat(
                messages=[{"role": "user", "content": prompt}],
                model="gpt-4o-mini",
            )
            return {"aggregated": response.content}

        raise ValueError(f"未知汇聚策略: {strategy}")
```

**要点**：
- `asyncio.wait_for` 包裹 `asyncio.shield` 可以在超时时取消任务但不丢失已完成的结果
- `wait_any` 和 `race` 在取到结果后必须取消其余分支，否则后台任务会继续消耗资源
- 常见错误：直接用 `asyncio.gather` 而不处理单个分支的超时——一个慢分支会拖住所有分支

### 练习 2

**思路**：三种失败策略的核心区别在于"一个分支失败后其他分支怎么办"。fail_fast 用 `asyncio.gather(return_exceptions=False)` 立即抛异常；continue 逐个 try-catch 记录错误但不中断；retry 用指数退避重试。关键是要正确处理已启动但未完成的后台任务。

**答案**：

```python
import asyncio
from enum import Enum

class FailureStrategy(str, Enum):
    FAIL_FAST = "fail_fast"
    CONTINUE = "continue"
    RETRY = "retry"

class ResilientParallelExecutor:
    """带失败策略的并行执行器"""

    async def execute(
        self,
        tasks: list[tuple[str, callable]],
        strategy: FailureStrategy,
        max_retries: int = 3,
        retry_delay_base: float = 1.0,
    ) -> dict:
        if strategy == FailureStrategy.FAIL_FAST:
            return await self._fail_fast(tasks)
        elif strategy == FailureStrategy.CONTINUE:
            return await self._continue_on_error(tasks)
        elif strategy == FailureStrategy.RETRY:
            return await self._retry_on_error(tasks, max_retries, retry_delay_base)

    async def _fail_fast(self, tasks: list[tuple[str, callable]]) -> dict:
        """快速失败：任意一个失败就立即抛异常，取消其余"""
        task_objs = {name: asyncio.create_task(coro()) for name, coro in tasks}

        try:
            # return_exceptions=False 会在第一个异常时抛出
            results_list = await asyncio.gather(
                *task_objs.values(), return_exceptions=False
            )
            return dict(zip(task_objs.keys(), results_list))
        except Exception:
            # 取消所有未完成的任务
            for task in task_objs.values():
                if not task.done():
                    task.cancel()
            for task in task_objs.values():
                try:
                    await task
                except asyncio.CancelledError:
                    pass
            raise

    async def _continue_on_error(self, tasks: list[tuple[str, callable]]) -> dict:
        """继续执行：记录错误但不中断其他分支"""
        results = {}

        async def run_one(name: str, coro):
            try:
                results[name] = await coro()
            except Exception as e:
                results[name] = {"error": str(e), "error_type": type(e).__name__}

        await asyncio.gather(*[run_one(n, c) for n, c in tasks])

        failed = [n for n, v in results.items() if isinstance(v, dict) and "error" in v]
        return {"results": results, "failed_count": len(failed), "failed_branches": failed}

    async def _retry_on_error(
        self,
        tasks: list[tuple[str, callable]],
        max_retries: int,
        retry_delay_base: float,
    ) -> dict:
        """重试：失败后指数退避重试"""
        results = {}

        async def run_with_retry(name: str, coro):
            last_error = None
            for attempt in range(max_retries + 1):
                try:
                    results[name] = await coro()
                    return
                except Exception as e:
                    last_error = e
                    if attempt < max_retries:
                        delay = retry_delay_base * (2 ** attempt)
                        print(f"[重试] 分支 {name} 第 {attempt + 1} 次失败，{delay}s 后重试: {e}")
                        await asyncio.sleep(delay)

            results[name] = {
                "error": str(last_error),
                "retries_exhausted": True,
                "attempts": max_retries + 1,
            }

        await asyncio.gather(*[run_with_retry(n, c) for n, c in tasks])

        failed = [n for n, v in results.items() if isinstance(v, dict) and "error" in v]
        return {"results": results, "failed_count": len(failed), "failed_branches": failed}


# --- 验证 ---
async def main():
    executor = ResilientParallelExecutor()

    async def success_task():
        await asyncio.sleep(0.1)
        return {"status": "ok", "data": "success"}

    async def fail_task():
        raise ValueError("模拟失败")

    async def slow_task():
        await asyncio.sleep(10)
        return {"status": "ok"}

    tasks = [
        ("branch_a", success_task),
        ("branch_b", success_task),
    ]

    # 测试 continue 策略
    tasks_with_fail = [
        ("branch_a", success_task),
        ("branch_b", fail_task),
    ]
    result = await executor.execute(tasks_with_fail, FailureStrategy.CONTINUE)
    print(f"Continue 策略: {result}")
    assert result["failed_count"] == 1

    # 测试 retry 策略
    result = await executor.execute(tasks_with_fail, FailureStrategy.RETRY, max_retries=2)
    print(f"Retry 策略: {result}")

    print("所有测试通过")

if __name__ == "__main__":
    asyncio.run(main())
```

**要点**：
- `fail_fast` 必须在抛异常后取消所有未完成的任务——否则后台协程会泄漏，持续消耗资源
- `retry` 要用指数退避（1s → 2s → 4s），避免立即重试导致下游服务雪崩
- 常见错误：`retry` 重试的是已经 resolve 的 coroutine 对象——Python 的 coroutine 只能 await 一次，重试时需要重新调用函数创建新的 coroutine

## 常见误区

| 错误 | 原因 | 解决 |
|------|------|------|
| 结果丢失 | 没有汇聚节点 | 加入 Aggregator 节点 |
| 资源耗尽 | 并行太多 | 限制并发数 |
| 超时 | 分支执行太慢 | 设置分支超时 |

## 工程建议

- 并行分支数量要设上限，根据系统资源和下游服务承受能力合理控制并发度
- 每个并行分支都要设置独立的超时时间，避免一个慢分支拖住整个工作流
- 汇聚策略要根据业务场景选择——简单拼接适合日志收集，字典合并适合数据聚合，LLM 合并适合非结构化结果
- 并行执行的每个分支要有独立的执行 ID，方便追踪和排查单个分支的问题
