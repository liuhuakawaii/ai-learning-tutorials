## 一、从零构建 LLM-as-Judge

### 1.1 最简实现

```python
from openai import OpenAI
from dataclasses import dataclass

client = OpenAI()

@dataclass
class EvalResult:
    score: int          # 1-5
    reasoning: str      # 评估理由
    passed: bool        # 是否通过阈值

def judge_answer(
    question: str,
    answer: str,
    reference: str = None,
    threshold: int = 3
) -> EvalResult:
    """基础版 LLM-as-Judge"""
    
    ref_section = ""
    if reference:
        ref_section = f"\n参考答案：{reference}"
    
    prompt = f"""你是一个严格的 AI 回答质量评估员。

问题：{question}
模型回答：{answer}{ref_section}

请从以下维度评估回答质量：
1. 准确性：信息是否正确
2. 完整性：是否覆盖了问题的关键方面
3. 相关性：是否切题，没有跑题

评分标准：
1 分：完全错误、不相关、或拒绝回答
2 分：有部分正确信息，但有重大错误或遗漏
3 分：基本正确，有小的不准确或不完整
4 分：正确且完整，表述清晰
5 分：优秀，准确、完整、表述专业

请严格按以下 JSON 格式输出，不要输出其他内容：
{{"score": <1-5>, "reasoning": "<简短评估理由>"}}"""

    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
        temperature=0,
        response_format={"type": "json_object"}
    )
    
    result = response.choices[0].message.content
    import json
    data = json.loads(result)
    
    return EvalResult(
        score=data["score"],
        reasoning=data["reasoning"],
        passed=data["score"] >= threshold
    )
```

## 二、评分 Prompt 的设计技巧

### 2.1 关键设计原则

```
评分 Prompt 的四要素：

1. 角色定义
   "你是一个严格的 AI 回答质量评估员"
   → 让模型进入"评估模式"而不是"聊天模式"

2. 评分维度
   明确告诉模型从哪些角度评估
   → 减少评估的随机性

3. 评分标准
   每个分数对应什么质量水平
   → 让评分有据可依

4. 输出格式
   JSON 格式，包含 score 和 reasoning
   → 方便程序解析
```

### 2.2 高级评分 Prompt 模板

```python
def build_judge_prompt(
    question: str,
    answer: str,
    context: str = None,
    reference: str = None,
    criteria: list[str] = None
) -> str:
    """构建高质量的评估 Prompt"""
    
    sections = [f"## 问题\n{question}", f"## 模型回答\n{answer}"]
    
    if context:
        sections.insert(1, f"## 参考资料\n{context}")
    if reference:
        sections.append(f"## 参考答案\n{reference}")
    
    criteria_text = ""
    if criteria:
        criteria_text = "\n".join(f"- {c}" for c in criteria)
    else:
        criteria_text = """- 准确性：信息是否正确，是否有幻觉
- 完整性：是否覆盖了问题的关键方面
- 相关性：是否切题，没有无关信息
- 清晰度：表述是否清楚、结构是否合理"""
    
    return f"""你是一个专业的 AI 回答质量评估员。请严格按照评分标准进行评估。

{chr(10).join(sections)}

## 评估维度
{criteria_text}

## 评分标准
| 分数 | 含义 | 判断标准 |
|------|------|----------|
| 1 | 差 | 完全错误、不相关、或拒绝回答 |
| 2 | 较差 | 有部分正确信息，但有重大错误 |
| 3 | 一般 | 基本正确，有明显不足 |
| 4 | 良好 | 正确且完整，表述清晰 |
| 5 | 优秀 | 准确、完整、表述专业、有额外价值 |

## 输出要求
请以 JSON 格式输出评估结果：
```json
{{
  "score": 1-5,
  "reasoning": "简短评估理由（50字以内）",
  "strengths": ["优点1", "优点2"],
  "weaknesses": ["不足1", "不足2"]
}}
```"""
```

### 2.3 避免常见陷阱

```python
# 陷阱 1：评分标准模糊
# 差：
"请评估回答的质量"
# 好：
"请从准确性、完整性、相关性三个维度评估，每个维度 1-5 分"

# 陷阱 2：没有参考信息
# 差：只给问题和回答，让模型"猜"对不对
# 好：提供参考答案或参考资料，让模型有判断依据

# 陷阱 3：没有要求输出理由
# 差：只要求输出分数
# 好：要求输出分数 + 理由（便于调试和校准）

# 陷阱 4：温度设得太高
# 差：temperature=0.7（评估结果不稳定）
# 好：temperature=0（评估结果可复现）
```

## 三、多维度评估实现

### 3.1 并行评估多个维度

```python
import asyncio
from openai import AsyncOpenAI

async_client = AsyncOpenAI()

async def evaluate_dimension(
    question: str,
    answer: str,
    dimension: str,
    criteria: str
) -> dict:
    """评估单个维度"""
    prompt = f"""评估以下回答在"{dimension}"维度的表现。

问题：{question}
回答：{answer}

评估标准：{criteria}

请输出 JSON：{{"score": 1-5, "reasoning": "理由"}}"""

    response = await async_client.chat.completions.create(
        model="gpt-4o-mini",  # 用便宜的模型做评估
        messages=[{"role": "user", "content": prompt}],
        temperature=0,
        response_format={"type": "json_object"}
    )
    
    return json.loads(response.choices[0].message.content)


async def multi_dimension_eval(question: str, answer: str) -> dict:
    """多维度并行评估"""
    dimensions = {
        "accuracy": "回答中的事实信息是否正确？是否有编造的内容？",
        "completeness": "回答是否覆盖了问题的所有关键方面？",
        "relevancy": "回答是否切题？是否有无关信息？",
        "clarity": "回答的表述是否清晰？结构是否合理？",
    }
    
    tasks = [
        evaluate_dimension(question, answer, dim, criteria)
        for dim, criteria in dimensions.items()
    ]
    
    results = await asyncio.gather(*tasks)
    
    return {
        dim: result
        for dim, result in zip(dimensions.keys(), results)
    }
```

### 3.2 综合评分计算

```python
def compute_weighted_score(eval_results: dict) -> float:
    """计算加权综合评分"""
    weights = {
        "accuracy": 0.40,
        "completeness": 0.25,
        "relevancy": 0.20,
        "clarity": 0.15,
    }
    
    total = sum(
        eval_results[dim]["score"] * weights[dim]
        for dim in weights
    )
    
    return round(total, 2)

# 使用示例
results = asyncio.run(multi_dimension_eval(
    question="什么是 RAG？",
    answer="RAG（Retrieval-Augmented Generation）是一种结合检索和生成的技术..."
))

overall = compute_weighted_score(results)
print(f"综合评分：{overall}/5.0")
print(f"各维度：{json.dumps(results, indent=2, ensure_ascii=False)}")
```

import pandas as pd
import numpy as np

def batch_evaluate(dataset: list[dict]) -> pd.DataFrame:
    """批量评估并生成统计报告"""
    results = []
    
    for case in dataset:
        eval_result = judge_answer(
            question=case["question"],
            answer=case["answer"],
            reference=case.get("reference")
        )
        results.append({
            "id": case["id"],
            "question": case["question"][:50],
            "score": eval_result.score,
            "passed": eval_result.passed,
            "reasoning": eval_result.reasoning,
        })
    
    df = pd.DataFrame(results)
    return df


def generate_report(df: pd.DataFrame) -> str:
    """生成评估报告"""
    report = f"""
# 评估报告

## 总体统计
- 总用例数：{len(df)}
- 通过率：{df['passed'].mean():.1%}
- 平均分：{df['score'].mean():.2f}/5.0
- 标准差：{df['score'].std():.2f}

## 分数分布
{df['score'].value_counts().sort_index().to_string()}

## 低分案例（需重点关注）
{df[df['score'] <= 2][['id', 'question', 'score', 'reasoning']].to_string()}

## 高分案例（优秀表现）
{df[df['score'] >= 4][['id', 'question', 'score', 'reasoning']].to_string()}
"""
    return report

## 五、评估器的校准与优化

### 5.1 人工校准流程

```python
def calibrate_judge(
    dataset: list[dict],
    human_scores: list[int],
    sample_size: int = 50
) -> dict:
    """校准评估器，对比人工评分"""
    
    # 随机采样
    indices = np.random.choice(len(dataset), min(sample_size, len(dataset)), replace=False)
    
    judge_scores = []
    for idx in indices:
        result = judge_answer(
            question=dataset[idx]["question"],
            answer=dataset[idx]["answer"],
            reference=dataset[idx].get("reference")
        )
        judge_scores.append(result.score)
    
    human_sample = [human_scores[i] for i in indices]
    
    # 计算一致性
    agreement = sum(1 for j, h in zip(judge_scores, human_sample) if abs(j - h) <= 1) / len(judge_scores)
    
    # 计算相关性
    correlation = np.corrcoef(judge_scores, human_sample)[0, 1]
    
    return {
        "agreement_rate": agreement,  # ±1 分内的一致率
        "correlation": correlation,   # 相关系数
        "avg_diff": np.mean(np.array(judge_scores) - np.array(human_sample)),  # 平均分差
        "judge_scores": judge_scores,
        "human_scores": human_sample,
    }
```

### 5.2 常见校准问题

```
问题 1：评估器偏高/偏低
  现象：judge 平均分比人工高/低 1 分以上
  解决：调整评分标准中的分数描述，或加偏移量

问题 2：评估器区分度不够
  现象：所有回答都得 3-4 分，没有 1 分和 5 分
  解决：在 Prompt 中加入更多极端案例的描述

问题 3：对特定类型回答偏差大
  现象：对简短回答系统性偏低，对冗长回答系统性偏高
  解决：在评分标准中明确"简洁但完整"的要求

问题 4：评估结果不稳定
  现象：同一回答多次评估分数不同
  解决：确保 temperature=0，使用 JSON 格式输出
```

## 六、完整评估 Pipeline

```python
class EvalPipeline:
    """完整的评估 Pipeline"""
    
    def __init__(self, dataset_path: str):
        with open(dataset_path, "r", encoding="utf-8") as f:
            self.dataset = json.load(f)
        self.results = []
    
    def run(self, get_answer_fn) -> pd.DataFrame:
        """运行评估
        
        Args:
            get_answer_fn: 接受 question 返回 answer 的函数
                           （可以是你的 RAG/Agent 系统）
        """
        for case in self.dataset:
            # 1. 获取模型回答
            answer = get_answer_fn(case["question"])
            
            # 2. 评估
            eval_result = judge_answer(
                question=case["question"],
                answer=answer,
                reference=case.get("reference")
            )
            
            # 3. 记录结果
            self.results.append({
                "id": case["id"],
                "question": case["question"],
                "answer": answer,
                "reference": case.get("reference"),
                "score": eval_result.score,
                "passed": eval_result.passed,
                "reasoning": eval_result.reasoning,
            })
        
        return pd.DataFrame(self.results)
    
    def report(self) -> str:
        """生成报告"""
        df = pd.DataFrame(self.results)
        return generate_report(df)
    
    def save(self, path: str):
        """保存结果"""
        df = pd.DataFrame(self.results)
        df.to_csv(path, index=False, encoding="utf-8-sig")
        print(f"评估结果已保存到 {path}")

# 使用示例
def my_rag_system(question: str) -> str:
    """你的 RAG 系统"""
    # 这里替换为你的实际系统
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": question}]
    )
    return response.choices[0].message.content

# 运行评估
pipeline = EvalPipeline("eval_dataset.json")
df = pipeline.run(my_rag_system)
print(pipeline.report())
pipeline.save("eval_results.csv")
```

## 常见误区

1. **评分 Prompt 写得太模糊**：只写"请评估回答质量"，没有明确维度和评分标准，导致 LLM 评委输出随机、不可复现。
2. **不做人工校准就直接用**：假设 LLM 评委的分数天然准确。实际上必须用 50-100 条人工标注数据做校准，计算一致率和相关性，确保评委可信。
3. **用太弱的模型做评委**：为了省成本用 GPT-3.5 做评估，结果评委本身就分不清事实和幻觉，评估结果毫无意义。评委模型至少要和被评估模型同等水平。
4. **忽略评估结果的不稳定性**：temperature 不为 0、没有用 JSON 格式约束输出、没有要求输出理由——这些都会导致同一输入多次评估结果不同。

## 工程建议

1. **评分 Prompt 必须包含四要素**：角色定义、评分维度、分数标准（每档有具体描述）、输出格式（JSON + reasoning）。缺少任何一个都会降低评估质量。
2. **先用 50 条人工标注数据做校准**：跑完校准后计算一致率（±1 分内）和相关系数。一致率低于 75% 就要调整评分 Prompt，不要急着上线。
3. **多维度评估用并行调用**：把准确性、完整性、相关性等维度拆成独立的 API 调用并行执行，既提高速度又避免单次 Prompt 过于复杂导致评估质量下降。
4. **评估报告要包含低分案例的详细信息**：不要只输出统计数字。每个低分 case 都要记录问题、回答、分数和理由，方便人工复查和问题定位。

## 参考答案

### 练习一

**思路**：代码生成场景的评估需要关注三个维度：正确性（逻辑是否正确）、可运行性（代码能否执行）、代码风格（是否符合规范）。需要设计专门的评分 Prompt，并为每个维度设定明确的评分标准。

**答案**：

```python
from openai import OpenAI
from dataclasses import dataclass

client = OpenAI()

@dataclass
class CodeEvalResult:
    correctness: int      # 1-5
    runnability: int       # 1-5
    style: int             # 1-5
    overall: float         # 加权综合分
    reasoning: str

def judge_code_answer(
    question: str,
    code_answer: str,
    reference: str = None,
    language: str = "python"
) -> CodeEvalResult:
    """代码生成场景的 LLM-as-Judge"""

    ref_section = f"\n参考答案：\n```{language}\n{reference}\n```" if reference else ""

    prompt = f"""你是一个严格的代码质量评估员。请评估以下代码回答的质量。

问题：{question}

模型生成的代码：
```{language}
{code_answer}
```
{ref_section}

请从以下三个维度评估：

1. **正确性（Correctness）**：
   - 代码逻辑是否正确解决了问题？
   - 是否有明显的 bug 或逻辑错误？
   - 边界情况是否处理得当？

2. **可运行性（Runnability）**：
   - 代码是否能直接运行（无语法错误、无缺失依赖）？
   - 是否需要额外的上下文或导入才能运行？

3. **代码风格（Style）**：
   - 命名是否清晰、有业务语义？
   - 结构是否合理、可读性是否好？
   - 是否遵循 {language} 的惯用写法？

评分标准（每个维度 1-5 分）：
1 分：完全不可用
2 分：有重大问题，需要大量修改
3 分：基本可用，有小问题
4 分：质量良好，可直接使用
5 分：优秀，超出预期

请以 JSON 格式输出：
{{"correctness": <1-5>, "runnability": <1-5>, "style": <1-5>, "reasoning": "<评估理由>"}}"""

    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
        temperature=0,
        response_format={"type": "json_object"}
    )

    data = json.loads(response.choices[0].message.content)

    overall = (
        data["correctness"] * 0.50 +
        data["runnability"] * 0.30 +
        data["style"] * 0.20
    )

    return CodeEvalResult(
        correctness=data["correctness"],
        runnability=data["runnability"],
        style=data["style"],
        overall=round(overall, 2),
        reasoning=data["reasoning"]
    )
```

**要点**：
- 代码评估的权重应该是：正确性 > 可运行性 > 风格（正确性是核心）
- 需要在 Prompt 中提供语言信息，因为不同语言的风格标准不同
- 常见错误：只评估"代码能不能跑"而忽略"逻辑是否正确"——能运行但逻辑错误的代码比不能运行的更危险

### 练习二

**思路**：校准的核心是对比 LLM 评委和人工评分的一致性。需要准备 20 条真实数据，分别用人工和 LLM 评估，然后计算一致率（±1 分内）和相关系数。

**答案**：

```python
import numpy as np
from openai import OpenAI

client = OpenAI()

# 准备 20 条真实问答数据
test_data = [
    {"id": i, "question": f"问题{i}", "answer": f"回答{i}", "reference": f"参考{i}"}
    for i in range(1, 21)
]

# 人工评分（假设已收集）
human_scores = [4, 3, 5, 2, 4, 3, 4, 5, 1, 3, 4, 2, 5, 3, 4, 4, 2, 3, 5, 4]

# LLM 评委评分
def get_llm_score(question, answer, reference):
    prompt = f"""请对以下回答评分（1-5 分）。
问题：{question}
参考答案：{reference}
回答：{answer}
评分标准：1=差 2=较差 3=一般 4=良好 5=优秀
请只输出数字。"""
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
        temperature=0
    )
    return int(response.choices[0].message.content.strip())

llm_scores = []
for item in test_data:
    score = get_llm_score(item["question"], item["answer"], item["reference"])
    llm_scores.append(score)

# 计算一致性指标
llm_arr = np.array(llm_scores)
human_arr = np.array(human_scores)

# 一致率（±1 分内）
agreement = np.sum(np.abs(llm_arr - human_arr) <= 1) / len(llm_arr)
print(f"一致率（±1 分内）：{agreement:.1%}")

# 相关系数
correlation = np.corrcoef(llm_arr, human_arr)[0, 1]
print(f"相关系数：{correlation:.3f}")

# 平均分差
avg_diff = np.mean(llm_arr - human_arr)
print(f"平均分差：{avg_diff:+.2f}（正=LLM偏高，负=LLM偏低）")

# 判断标准
if agreement >= 0.80:
    print("✅ 一致率达标（≥80%），LLM 评委可用")
elif agreement >= 0.70:
    print("⚠️ 一致率偏低（70-80%），需要调整评分 Prompt")
else:
    print("❌ 一致率过低（<70%），需要更换评委模型或重新设计评分标准")
```

**要点**：
- 一致率（±1 分内）是核心指标，目标 ≥ 80%
- 相关系数反映评分趋势是否一致，目标 ≥ 0.7
- 平均分差反映系统性偏差：正偏差说明 LLM 评委偏宽松，负偏差说明偏严格
- 常见错误：只用"完全一致率"（分数完全相同）来评判——1-5 分制下完全一致太严格，±1 分内的一致率更合理

### 练习三

**思路**：检测评估器对不同长度回答的系统性偏差，需要将测试数据按长度分组，对比两组的平均分和分布差异。

**答案**：

```python
import pandas as pd
import numpy as np

# 假设已有评估结果
eval_data = [
    {"id": 1, "answer_length": 30, "score": 2},
    {"id": 2, "answer_length": 45, "score": 3},
    {"id": 3, "answer_length": 20, "score": 2},
    {"id": 4, "answer_length": 500, "score": 5},
    {"id": 5, "answer_length": 600, "score": 4},
    {"id": 6, "answer_length": 40, "score": 3},
    {"id": 7, "answer_length": 550, "score": 5},
    {"id": 8, "answer_length": 25, "score": 2},
    # ... 更多数据
]

df = pd.DataFrame(eval_data)
df["length_group"] = pd.cut(df["answer_length"], bins=[0, 50, 500, float("inf")],
                             labels=["短回答(<50字)", "中等(50-500字)", "长回答(>500字)"])

# 分组统计
group_stats = df.groupby("length_group")["score"].agg(["mean", "std", "count"])
print("各长度组评分统计：")
print(group_stats)

# 偏差检测
short_avg = df[df["answer_length"] < 50]["score"].mean()
long_avg = df[df["answer_length"] > 500]["score"].mean()
diff = long_avg - short_avg

print(f"\n短回答平均分：{short_avg:.2f}")
print(f"长回答平均分：{long_avg:.2f}")
print(f"分差：{diff:+.2f}")

if abs(diff) > 0.5:
    print(f"⚠️ 检测到长度偏差：{'长回答偏高' if diff > 0 else '短回答偏高'}")
    print("修正方案：")
    print("  1. 在评分 Prompt 中明确：'简洁但完整的回答应得高分，冗长但有冗余信息的回答应扣分'")
    print("  2. 在评分标准中加入：'信息密度'维度——有效信息量 / 回答总字数")
    print("  3. 对不同长度组分别设定阈值，而非使用统一阈值")
else:
    print("✅ 未检测到显著长度偏差")
```

**要点**：
- LLM 评委普遍存在"长度偏差"——倾向于给更长的回答更高分
- 修正方法核心：在评分标准中强调"简洁但完整"和"信息密度"
- 常见错误：不做长度偏差检测就直接使用评估结果——如果你的系统倾向于生成简短回答，评估分数会被系统性低估
