## 一、Token 消耗追踪

### 1.1 基础追踪

```python
class TokenTracker:
    """Token 消耗追踪器"""
    
    def __init__(self):
        self.records = []
    
    def track(self, operation: str, input_tokens: int, output_tokens: int, model: str):
        """记录 Token 消耗"""
        self.records.append({
            "operation": operation,
            "input_tokens": input_tokens,
            "output_tokens": output_tokens,
            "total_tokens": input_tokens + output_tokens,
            "model": model,
            "timestamp": datetime.now().isoformat()
        })
    
    def get_summary(self) -> dict:
        """获取汇总"""
        if not self.records:
            return {}
        
        total_input = sum(r["input_tokens"] for r in self.records)
        total_output = sum(r["output_tokens"] for r in self.records)
        
        return {
            "total_calls": len(self.records),
            "total_input_tokens": total_input,
            "total_output_tokens": total_output,
            "total_tokens": total_input + total_output,
            "by_operation": self._group_by_operation()
        }
    
    def _group_by_operation(self) -> dict:
        """按操作分组"""
        groups = {}
        for r in self.records:
            op = r["operation"]
            if op not in groups:
                groups[op] = {"count": 0, "tokens": 0}
            groups[op]["count"] += 1
            groups[op]["tokens"] += r["total_tokens"]
        return groups
```

## 二、延迟分析

### 2.1 调用链路追踪

```python
class LatencyTracker:
    """延迟追踪器"""
    
    def __init__(self):
        self.spans = []
    
    def start_span(self, name: str) -> str:
        """开始一个 span"""
        span_id = str(uuid.uuid4())
        self.spans.append({
            "id": span_id,
            "name": name,
            "start_time": time.time(),
            "end_time": None,
            "duration": None
        })
        return span_id
    
    def end_span(self, span_id: str):
        """结束一个 span"""
        for span in self.spans:
            if span["id"] == span_id:
                span["end_time"] = time.time()
                span["duration"] = span["end_time"] - span["start_time"]
                break
    
    def get_summary(self) -> dict:
        """获取延迟汇总"""
        completed = [s for s in self.spans if s["duration"] is not None]
        
        if not completed:
            return {}
        
        total_duration = sum(s["duration"] for s in completed)
        
        return {
            "total_spans": len(completed),
            "total_duration": total_duration,
            "avg_duration": total_duration / len(completed),
            "by_name": self._group_by_name(completed)
        }
    
    def _group_by_name(self, spans: list) -> dict:
        """按名称分组"""
        groups = {}
        for s in spans:
            name = s["name"]
            if name not in groups:
                groups[name] = {"count": 0, "total_duration": 0}
            groups[name]["count"] += 1
            groups[name]["total_duration"] += s["duration"]
        return groups
```

class CostAlert:
    """成本预警"""
    
    def __init__(self, daily_limit: float, monthly_limit: float):
        self.daily_limit = daily_limit
        self.monthly_limit = monthly_limit
        self.daily_cost = 0.0
        self.monthly_cost = 0.0
    
    def add_cost(self, cost: float):
        """添加成本"""
        self.daily_cost += cost
        self.monthly_cost += cost

            self._send_alert("daily", self.daily_cost, self.daily_limit)
        
        if self.monthly_cost > self.monthly_limit * 0.8:
            self._send_alert("monthly", self.monthly_cost, self.monthly_limit)
    
    def _send_alert(self, period: str, current: float, limit: float):
        """发送预警"""
        print(f"⚠️ {period} 成本预警：${current:.2f} / ${limit:.2f}")

## 常见误区

1. **只看总 Token 数，不看输入/输出比例**：输入 Token（Prompt）和输出 Token（生成）的单价差异巨大。一个 Agent 如果 Prompt 设计臃肿导致输入 Token 暴涨，优化 Prompt 比换模型更有效。
2. **把平均延迟当唯一指标**：平均延迟 3 秒看起来不错，但 P99 延迟可能是 15 秒。用户体感由最慢的请求决定，必须关注尾部延迟。
3. **盲目上缓存不评估命中率**：缓存能省钱，但 Agent 的用户查询高度个性化，缓存命中率可能很低。投入精力做缓存但命中率不到 10%，不如把精力放在 Prompt 精简上。
4. **成本优化只在上线前做一次**：模型定价会变、用户行为会变、Agent 功能会迭代。成本监控和优化必须是持续的过程，不是一次性任务。

## 工程建议

1. **按用户请求维度统计成本**：不只看全局总费用，要算到每个请求的平均成本。建立"单次请求成本"指标，设阈值告警——比如单次请求超过 $0.05 就触发预警。
2. **延迟优化优先解决"最长的那一环"**：用链路追踪找到耗时最大的环节（通常是 LLM 调用或外部工具调用），针对性优化。并行化可以互相独立的工具调用，缓存高频重复查询。
3. **模型路由是成本优化的杀手锏**：简单任务用便宜模型（gpt-4o-mini），复杂任务用强模型（gpt-4o）。通过 query 复杂度分类器做自动路由，可以降低 40-60% 的成本而几乎不影响质量。
4. **建立成本看板并设周度 Review**：在 Grafana 或类似工具中展示每日/每周成本趋势、按功能模块的成本分布、单次请求成本的 P50/P95/P99。每周 Review 一次，及时发现异常增长。

## 参考答案

### 练习一

**思路**：Token 和延迟追踪需要在每个 LLM 调用和工具调用处埋点。核心是用一个上下文管理器自动记录每次调用的 token 数、延迟和成本，最后汇总输出报告。

**答案**：

```python
import time
from dataclasses import dataclass, field
from datetime import datetime
from contextlib import contextmanager
from typing import Optional


PRICING = {
    "gpt-4o": {"input": 2.50, "output": 10.00},
    "gpt-4o-mini": {"input": 0.15, "output": 0.60},
}


@dataclass
class CallRecord:
    operation: str
    model: str
    input_tokens: int
    output_tokens: int
    latency: float
    cost: float
    timestamp: str


class AgentCostTracker:
    """Agent 成本和延迟追踪器"""

    def __init__(self):
        self.records: list[CallRecord] = []
        self._span_stack: list[dict] = []

    @contextmanager
    def track(self, operation: str, model: str = "gpt-4o"):
        """上下文管理器：自动追踪延迟"""
        start = time.time()
        self._span_stack.append({"operation": operation, "start": start})
        try:
            yield self
        finally:
            duration = time.time() - start
            self._span_stack[-1]["duration"] = duration

    def record(
        self,
        operation: str,
        model: str,
        input_tokens: int,
        output_tokens: int,
        latency: float,
    ):
        pricing = PRICING.get(model, PRICING["gpt-4o-mini"])
        cost = (
            input_tokens * pricing["input"] / 1_000_000
            + output_tokens * pricing["output"] / 1_000_000
        )
        self.records.append(CallRecord(
            operation=operation,
            model=model,
            input_tokens=input_tokens,
            output_tokens=output_tokens,
            latency=latency,
            cost=cost,
            timestamp=datetime.now().isoformat(),
        ))

    def get_summary(self) -> dict:
        if not self.records:
            return {}
        total_input = sum(r.input_tokens for r in self.records)
        total_output = sum(r.output_tokens for r in self.records)
        total_cost = sum(r.cost for r in self.records)
        total_latency = sum(r.latency for r in self.records)

        by_operation = {}
        for r in self.records:
            op = r.operation
            by_operation.setdefault(op, {"count": 0, "tokens": 0, "cost": 0, "latency": 0})
            by_operation[op]["count"] += 1
            by_operation[op]["tokens"] += r.input_tokens + r.output_tokens
            by_operation[op]["cost"] += r.cost
            by_operation[op]["latency"] += r.latency

        return {
            "total_calls": len(self.records),
            "total_input_tokens": total_input,
            "total_output_tokens": total_output,
            "total_tokens": total_input + total_output,
            "total_cost": total_cost,
            "total_latency": total_latency,
            "avg_cost_per_call": total_cost / len(self.records),
            "avg_latency_per_call": total_latency / len(self.records),
            "by_operation": by_operation,
        }


# 使用示例
tracker = AgentCostTracker()

# 追踪 RAG 检索 + LLM 调用
with tracker.track("rag_retrieval", "gpt-4o-mini"):
    time.sleep(0.3)
    tracker.record("query_rewrite", "gpt-4o-mini", 200, 50, 0.3)

with tracker.track("llm_generation", "gpt-4o"):
    time.sleep(1.5)
    tracker.record("answer_generation", "gpt-4o", 1500, 400, 1.5)

summary = tracker.get_summary()
print(f"总调用次数: {summary['total_calls']}")
print(f"总 Token: {summary['total_tokens']}")
print(f"总成本: ${summary['total_cost']:.4f}")
print(f"总延迟: {summary['total_latency']:.2f}s")
```

**要点**：
- 追踪必须在每个 LLM 调用处埋点，遗漏一个就无法得到准确的成本分布
- 使用上下文管理器可以自动记录延迟，避免手动计时的遗漏和误差
- 常见错误：只追踪了总 Token 数，没有区分 input/output，因为两者的单价差异很大（input 通常便宜 4-5 倍）

### 练习二

**思路**：用追踪器的数据按操作分组，找出 Token 消耗最多和成本最高的操作。重点关注 input/output 比例，如果 input 占比过高说明 Prompt 有优化空间。

**答案**：

```python
import json


def analyze_token_distribution(tracker) -> dict:
    """分析 Token 消耗分布，找出优化方向"""
    summary = tracker.get_summary()
    by_op = summary["by_operation"]

    # 按 Token 消耗排序
    sorted_ops = sorted(by_op.items(), key=lambda x: x[1]["tokens"], reverse=True)

    print("=== Token 消耗分布 ===\n")
    print(f"{'操作':<25} {'调用次数':<10} {'Token数':<12} {'成本':<12} {'延迟':<10}")
    print("-" * 70)
    for op, stats in sorted_ops:
        print(f"{op:<25} {stats['count']:<10} {stats['tokens']:<12} ${stats['cost']:<11.4f} {stats['latency']:<10.2f}s")

    # 找出消耗最大的操作
    top_consumer = sorted_ops[0]
    print(f"\nToken 消耗最大的操作: {top_consumer[0]} ({top_consumer[1]['tokens']} tokens)")

    # 分析 input/output 比例
    total_input = summary["total_input_tokens"]
    total_output = summary["total_output_tokens"]
    input_ratio = total_input / summary["total_tokens"]
    print(f"\nInput/Output 比例: {input_ratio:.1%} / {1-input_ratio:.1%}")

    recommendations = []
    if input_ratio > 0.75:
        recommendations.append("Input Token 占比过高，建议精简 System Prompt 和上下文")
    if top_consumer[1]["tokens"] / summary["total_tokens"] > 0.5:
        recommendations.append(f"'{top_consumer[0]}' 占总 Token 的 {top_consumer[1]['tokens']/summary['total_tokens']:.0%}，是重点优化目标")
    if top_consumer[1]["count"] > 3:
        recommendations.append(f"'{top_consumer[0]}' 调用 {top_consumer[1]['count']} 次，考虑缓存或批处理")

    return {
        "top_consumer": top_consumer[0],
        "input_ratio": input_ratio,
        "recommendations": recommendations,
    }
```

**要点**：
- Token 消耗分析的关键是按操作分组，而不是只看总数
- Input/Output 比例超过 75% 说明 Prompt 过于臃肿，优化 Prompt 比换模型更有效
- 常见错误：只关注 Token 总数，不看分布，导致优化方向错误——可能最大的成本来源不是你以为的那个操作

### 练习三

**思路**：降低 30% 成本需要从三个杠杆入手：模型降级（简单任务用便宜模型）、Prompt 精简（减少 input token）、缓存（减少重复调用）。按优先级逐步实施，每一步都量化效果。

**答案**：

```python
import json


class CostOptimizer:
    """成本优化方案"""

    def __init__(self, tracker):
        self.tracker = tracker
        self.optimizations = []

    def analyze_and_optimize(self) -> dict:
        summary = self.tracker.get_summary()
        current_cost = summary["total_cost"]
        target_cost = current_cost * 0.7
        savings_needed = current_cost - target_cost

        plan = {
            "current_cost": current_cost,
            "target_cost": target_cost,
            "savings_needed": savings_needed,
            "strategies": [],
        }

        by_op = summary["by_operation"]

        # 策略 1：模型降级——将非关键操作从 gpt-4o 降到 gpt-4o-mini
        downgrade_candidates = [
            (op, stats) for op, stats in by_op.items()
            if "rewrite" in op or "classify" in op or "extract" in op
        ]
        model_savings = 0
        for op, stats in downgrade_candidates:
            # gpt-4o-mini 成本约为 gpt-4o 的 1/16
            current_op_cost = stats["cost"]
            new_op_cost = current_op_cost / 16
            saved = current_op_cost - new_op_cost
            model_savings += saved

        if model_savings > 0:
            plan["strategies"].append({
                "name": "模型降级",
                "description": f"将 {len(downgrade_candidates)} 个非关键操作从 gpt-4o 降到 gpt-4o-mini",
                "estimated_savings": model_savings,
                "affected_operations": [op for op, _ in downgrade_candidates],
                "risk": "低——简单任务用 gpt-4o-mini 质量损失很小",
            })

        # 策略 2：Prompt 精简——减少 input token
        total_input = summary["total_input_tokens"]
        prompt_savings_estimate = total_input * 0.2 * 2.50 / 1_000_000  # 假设减少 20% input
        plan["strategies"].append({
            "name": "Prompt 精简",
            "description": "精简 System Prompt，去除冗余指令，预计减少 20% input token",
            "estimated_savings": prompt_savings_estimate,
            "affected_operations": ["所有 LLM 调用"],
            "risk": "低——精简 Prompt 不影响质量，但需要验证",
        })

        # 策略 3：语义缓存——减少重复调用
        total_calls = summary["total_calls"]
        cache_hit_estimate = total_calls * 0.15  # 假设 15% 命中率
        cache_savings = cache_hit_estimate * summary["avg_cost_per_call"]
        plan["strategies"].append({
            "name": "语义缓存",
            "description": f"对相似查询缓存结果，预计命中率 15%，减少约 {int(cache_hit_estimate)} 次调用",
            "estimated_savings": cache_savings,
            "affected_operations": ["重复查询类操作"],
            "risk": "中——缓存过期可能导致回答不准确，需要合理的 TTL 策略",
        })

        total_estimated_savings = sum(s["estimated_savings"] for s in plan["strategies"])
        plan["total_estimated_savings"] = total_estimated_savings
        plan["estimated_new_cost"] = current_cost - total_estimated_savings
        plan["target_met"] = plan["estimated_new_cost"] <= target_cost

        return plan


# 输出优化方案
optimizer = CostOptimizer(tracker)
plan = optimizer.analyze_and_optimize()
print(f"当前成本: ${plan['current_cost']:.4f}")
print(f"目标成本: ${plan['target_cost']:.4f}")
print(f"预计可节省: ${plan['total_estimated_savings']:.4f}")
print(f"目标是否达成: {'✅' if plan['target_met'] else '❌ 需要更多优化'}")
for s in plan["strategies"]:
    print(f"\n  策略: {s['name']}")
    print(f"  说明: {s['description']}")
    print(f"  预计节省: ${s['estimated_savings']:.4f}")
    print(f"  风险: {s['risk']}")
```

**要点**：
- 模型降级是最快见效的优化手段——简单任务用 gpt-4o-mini 成本降低约 94% 且质量损失很小
- 语义缓存需要评估命中率，如果命中率不到 10% 则投入产出比不高
- 常见错误：只盯着模型价格，忽略 Prompt 精简——减少 30% input token 有时比换模型效果更好
