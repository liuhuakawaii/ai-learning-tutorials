# 02 检索评估指标——Context Precision / Recall / Relevancy 的计算

> 检索质量决定了 RAG 的上限。检索不到正确的文档，再好的生成模型也无济于事。

## 场景引入

你的 RAG 系统回答质量不稳定，有些问题回答得很好，有些则完全偏题。你用 top_k=5 做检索，但没有评估过这 5 条结果中有多少是真正相关的。当你引入 P@K 和 R@K 指标后才发现，Precision 只有 0.4——检索到的 5 条文档里平均只有 2 条是相关的，剩下 3 条噪声严重干扰了生成质量。

## 学习目标

- 深入理解三种检索评估指标的计算方法
- 掌握用 LLM 自动评估检索质量的技巧
- 学会诊断和优化检索问题

## 一、指标计算详解

### 1.1 带标签的检索评估

```python
from dataclasses import dataclass

@dataclass
class RetrievalLabel:
    """检索评估标签"""
    query: str
    retrieved_doc_ids: list[str]      # 检索到的文档 ID
    relevant_doc_ids: list[str]       # 真正相关的文档 ID
    graded_relevance: dict[str, int]  # 文档 ID → 相关性评分 (0-3)

# 相关性评分标准：
# 0 = 不相关
# 1 = 略微相关
# 2 = 部分相关
# 3 = 完全相关

def precision_at_k(retrieved: list[str], relevant: set[str], k: int) -> float:
    """计算 P@K：前 K 个检索结果的精确率"""
    top_k = retrieved[:k]
    hits = sum(1 for doc in top_k if doc in relevant)
    return hits / k

def recall_at_k(retrieved: list[str], relevant: set[str], k: int) -> float:
    """计算 R@K：前 K 个检索结果的召回率"""
    top_k = retrieved[:k]
    hits = sum(1 for doc in top_k if doc in relevant)
    return hits / len(relevant) if relevant else 1.0

def average_precision(retrieved: list[str], relevant: set[str]) -> float:
    """计算 AP（Average Precision）"""
    hits = 0
    sum_precision = 0.0
    
    for i, doc in enumerate(retrieved, 1):
        if doc in relevant:
            hits += 1
            sum_precision += hits / i
    
    return sum_precision / len(relevant) if relevant else 0.0

def ndcg_at_k(retrieved: list[str], graded: dict[str, int], k: int) -> float:
    """计算 NDCG@K（Normalized Discounted Cumulative Gain）"""
    import math
    
    # DCG
    dcg = 0.0
    for i, doc in enumerate(retrieved[:k], 1):
        rel = graded.get(doc, 0)
        dcg += (2**rel - 1) / math.log2(i + 1)
    
    # Ideal DCG
    ideal_rels = sorted(graded.values(), reverse=True)[:k]
    idcg = sum((2**rel - 1) / math.log2(i + 1) for i, rel in enumerate(ideal_rels, 1))
    
    return dcg / idcg if idcg > 0 else 0.0
```

### 1.2 批量计算

```python
def evaluate_retrieval_batch(labels: list[RetrievalLabel], k_values: list[int] = [1, 3, 5]) -> dict:
    """批量评估检索质量"""
    
    results = {f"P@{k}": [] for k in k_values}
    results.update({f"R@{k}": [] for k in k_values})
    results["AP"] = []
    results["MRR"] = []
    
    for label in labels:
        relevant = set(label.relevant_doc_ids)
        
        for k in k_values:
            results[f"P@{k}"].append(precision_at_k(label.retrieved_doc_ids, relevant, k))
            results[f"R@{k}"].append(recall_at_k(label.retrieved_doc_ids, relevant, k))
        
        results["AP"].append(average_precision(label.retrieved_doc_ids, relevant))
        
        # MRR
        for rank, doc in enumerate(label.retrieved_doc_ids, 1):
            if doc in relevant:
                results["MRR"].append(1.0 / rank)
                break
        else:
            results["MRR"].append(0.0)
    
    # 计算平均值
    return {metric: sum(values) / len(values) for metric, values in results.items()}
```

## 二、无标签情况下的检索评估

### 2.1 用 LLM 判断相关性

```python
def llm_judge_relevance(query: str, doc: str, client) -> int:
    """用 LLM 判断文档与查询的相关性"""
    prompt = f"""请判断以下文档与查询的相关性。

查询：{query}
文档：{doc}

评分标准：
0 = 完全不相关
1 = 略微相关（有共同主题但不能回答问题）
2 = 部分相关（包含部分答案）
3 = 完全相关（包含完整答案）

请只输出数字（0-3）。"""

    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": prompt}],
        temperature=0
    )
    return int(response.choices[0].message.content.strip())

def auto_generate_labels(dataset: list[dict], client) -> list[RetrievalLabel]:
    """自动生成检索评估标签"""
    labels = []
    
    for item in dataset:
        query = item["question"]
        retrieved = item["retrieved_docs"]
        
        # 用 LLM 判断每个文档的相关性
        relevant_ids = []
        graded = {}
        for doc_id, doc_content in zip(item["doc_ids"], retrieved):
            relevance = llm_judge_relevance(query, doc_content, client)
            graded[doc_id] = relevance
            if relevance >= 2:  # 阈值：部分相关以上
                relevant_ids.append(doc_id)
        
        labels.append(RetrievalLabel(
            query=query,
            retrieved_doc_ids=item["doc_ids"],
            relevant_doc_ids=relevant_ids,
            graded_relevance=graded
        ))
    
    return labels
```

## 三、检索问题诊断

### 3.1 问题分类

```
检索问题诊断：

问题 1：Precision 低，Recall 高
  含义：检索到了很多文档，但很多不相关
  原因：
    - 检索数量太多（top_k 设置过大）
    - 相似度阈值太低
    - embedding 模型区分度不够
  优化：
    - 减少 top_k
    - 提高相似度阈值
    - 添加 reranker
    - 优化 embedding 模型

问题 2：Precision 高，Recall 低
  含义：检索到的文档都很相关，但遗漏了很多
  原因：
    - 检索数量太少
    - 查询太具体
    - 文档没有充分索引
  优化：
    - 增加 top_k
    - 查询扩展（query expansion）
    - 混合检索（关键词 + 语义）

问题 3：Precision 和 Recall 都低
  含义：检索系统整体失效
  原因：
    - embedding 模型质量差
    - 文档和查询的语义差距大
    - 索引构建有问题
  优化：
    - 更换 embedding 模型
    - 优化文档切分策略
    - 检查索引构建流程
```

## 四、混合检索的评估

### 4.1 多路检索的评估

```python
class HybridRetrievalEvaluator:
    """混合检索评估器"""
    
    def __init__(self, client):
        self.client = client
    
    def evaluate_retrieval_methods(
        self,
        query: str,
        methods_results: dict[str, list[str]],
        ground_truth: list[str]
    ) -> dict:
        """对比不同检索方法的效果"""
        
        relevant = set(ground_truth)
        comparison = {}
        
        for method_name, retrieved in methods_results.items():
            comparison[method_name] = {
                "P@3": precision_at_k(retrieved, relevant, 3),
                "R@3": recall_at_k(retrieved, relevant, 3),
                "P@5": precision_at_k(retrieved, relevant, 5),
                "R@5": recall_at_k(retrieved, relevant, 5),
                "AP": average_precision(retrieved, relevant),
            }
        
        return comparison

# 使用示例
evaluator = HybridRetrievalEvaluator(client)

methods_results = {
    "vector_search": ["doc1", "doc3", "doc5", "doc7", "doc9"],
    "bm25": ["doc2", "doc3", "doc4", "doc6", "doc8"],
    "hybrid": ["doc1", "doc2", "doc3", "doc5", "doc6"],
}

comparison = evaluator.evaluate_retrieval_methods(
    query="公司的年假政策是什么？",
    methods_results=methods_results,
    ground_truth=["doc1", "doc3", "doc6"]
)

# 输出对比结果
for method, metrics in comparison.items():
    print(f"{method}: P@3={metrics['P@3']:.2f}, R@3={metrics['R@3']:.2f}")
```

## 五、Reranker 的评估

### 5.1 评估 Reranker 效果

```python
def evaluate_reranker(
    query: str,
    initial_results: list[str],
    reranked_results: list[str],
    relevant: set[str],
    client
) -> dict:
    """评估 Reranker 的效果"""
    
    initial_metrics = {
        "P@3": precision_at_k(initial_results, relevant, 3),
        "R@3": recall_at_k(initial_results, relevant, 3),
    }
    
    reranked_metrics = {
        "P@3": precision_at_k(reranked_results, relevant, 3),
        "R@3": recall_at_k(reranked_results, relevant, 3),
    }
    
    return {
        "initial": initial_metrics,
        "reranked": reranked_metrics,
        "improvement": {
            "P@3": reranked_metrics["P@3"] - initial_metrics["P@3"],
            "R@3": reranked_metrics["R@3"] - initial_metrics["R@3"],
        }
    }
```

## 常见误区

1. **只看 Precision 不看 Recall**：P@5 很高说明检索到的都相关，但可能遗漏了大量相关文档。单独看一个指标会误导优化方向。

2. **用固定阈值判断相关性**：不同查询对"相关"的定义不同。一个模糊查询可能有多个合理答案，一个精确查询只有唯一答案，不能用同一个阈值。

3. **忽略排序质量**：Precision@K 只看"有没有相关文档"，不看"排在第几位"。MRR 和 NDCG 才能反映排序质量，对用户体验影响更大。

4. **LLM 自动标签不做校验**：用 LLM 生成相关性标签时不做人工抽样检查，标签噪声会导致评估结论失真。

## 工程建议

1. **同时跟踪 P@K 和 R@K**：建议用 P@3 + R@5 组合，前者保证生成质量，后者保证信息覆盖。

2. **用 NDCG 替代简单 Precision 作为核心指标**：NDCG 同时考虑了相关性和排序位置，更能反映用户实际体验。

3. **定期校准 LLM 标签**：每周人工抽样 50 条 LLM 标注结果，计算与人工标注的一致率，低于 90% 就需要调整 Prompt。

4. **混合检索要做消融实验**：分别测试纯向量、纯关键词、混合检索的效果，用数据证明混合策略的收益。

## 参考答案

### 练习一

**思路**：按 P@K、R@K、AP 的定义逐步计算。P@3 是前 3 个结果中有多少相关，R@3 是所有相关文档中前 3 个找到了多少，AP 需要逐位计算。

**答案**：

```
检索结果：[A, B, C, D, E]
相关文档：{B, D, F}
```

**P@3 计算**：
- 前 3 个结果：[A, B, C]
- 其中相关的：B（1 个）
- P@3 = 1/3 ≈ **0.333**

**R@3 计算**：
- 前 3 个结果：[A, B, C]
- 其中相关的：B（1 个）
- 相关文档总数：3 个（B, D, F）
- R@3 = 1/3 ≈ **0.333**

**AP（Average Precision）计算**：
- 逐位检查：
  - 位置 1：A，不相关 → 跳过
  - 位置 2：B，相关 → 累计命中 1 个，Precision = 1/2 = 0.5
  - 位置 3：C，不相关 → 跳过
  - 位置 4：D，相关 → 累计命中 2 个，Precision = 2/4 = 0.5
  - 位置 5：E，不相关 → 跳过
- AP = (0.5 + 0.5) / 3（相关文档总数）= 1.0/3 ≈ **0.333**

**结果汇总**：

| 指标 | 值 | 含义 |
|------|-----|------|
| P@3 | 0.333 | 前 3 个结果中只有 1/3 相关 |
| R@3 | 0.333 | 3 个相关文档只找到了 1 个 |
| AP | 0.333 | 相关文档的平均排序精度不高 |

**要点**：
- P@3 和 R@3 巧合相同是因为相关文档恰好有 3 个，且前 3 个结果中只有 1 个相关
- AP 考虑了排序位置——如果 B 排在第 1 位而不是第 2 位，AP 会更高
- 常见错误：把 R@K 算成"前 K 个中相关的 / K"——R@K 的分母是所有相关文档数，不是 K

### 练习二

**思路**：P@5 高但 R@5 低，说明检索到的文档质量不错但覆盖面不够。需要在不降低 Precision 的前提下提高 Recall。

**答案**：

**诊断分析**：

| 指标 | 值 | 含义 |
|------|-----|------|
| P@5 | 0.8 | 检索到的 5 个文档中 4 个是相关的（质量不错） |
| R@5 | 0.4 | 相关文档只找到了 40%（遗漏严重） |

**问题本质**：检索太保守。系统只返回了最匹配的 5 个文档，但遗漏了大量"次相关但重要"的文档。

**优化方案**：

1. **增加 top_k + Reranker**：
   - 先用 top_k=20 召回更多候选文档
   - 再用 Reranker（如 bge-reranker）精排，取 top_5
   - 这样既提高 Recall，又不降低 Precision

2. **混合检索（Hybrid Search）**：
   - 向量检索擅长语义匹配，BM25 擅长关键词匹配
   - 两路检索合并去重，能覆盖更多相关文档
   - 示例：向量检索 top_10 + BM25 top_10 → 合并去重 → Reranker → top_5

3. **查询扩展（Query Expansion）**：
   - 用 LLM 将用户问题扩展为多个子问题
   - 例如："公司的年假政策" → ["年假天数", "年假申请条件", "年假过期规则"]
   - 多路检索合并结果

4. **优化文档切分**：
   - 如果相关文档被切成了多个片段，可能只有部分片段被匹配到
   - 使用更大的 chunk_size 或重叠切分（overlap）

**要点**：
- P@5=0.8 + R@5=0.4 说明相关文档至少有 10 个（找到了 4 个，占 40%），但只检索了 5 个
- 盲目增加 top_k 会降低 Precision——必须配合 Reranker
- 常见错误：只优化一个指标——提高 Recall 时如果不关注 Precision，会引入大量噪声，反而降低生成质量

### 练习三

**思路**：实现一个用 LLM 自动判断文档与查询相关性的标签生成器。需要设计好评分标准，处理边界情况，并支持批量处理。

**答案**：

```python
from openai import OpenAI
from dataclasses import dataclass
import json
import time

client = OpenAI()

@dataclass
class RelevanceLabel:
    doc_id: str
    query: str
    score: int          # 0-3
    label: str          # "irrelevant" / "slightly" / "partial" / "fully"
    reasoning: str

def llm_judge_relevance(query: str, doc_content: str, doc_id: str) -> RelevanceLabel:
    """用 LLM 判断文档与查询的相关性"""

    prompt = f"""请判断以下文档与查询的相关性。

查询：{query}
文档内容：{doc_content}

评分标准：
0 = 完全不相关：文档主题与查询无关
1 = 略微相关：有共同主题，但不能回答查询
2 = 部分相关：包含查询的部分答案
3 = 完全相关：包含查询的完整答案

请以 JSON 格式输出：
{{"score": <0-3>, "label": "<irrelevant/slightly/partial/fully>", "reasoning": "<50字以内>"}}"""

    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": prompt}],
        temperature=0,
        response_format={"type": "json_object"}
    )

    result = json.loads(response.choices[0].message.content)

    return RelevanceLabel(
        doc_id=doc_id,
        query=query,
        score=result["score"],
        label=result["label"],
        reasoning=result["reasoning"]
    )


def auto_generate_labels(
    queries: list[str],
    retrieved_docs: list[list[dict]],
    relevance_threshold: int = 2
) -> list[dict]:
    """批量自动生成检索评估标签

    Args:
        queries: 查询列表
        retrieved_docs: 每个查询对应的检索结果列表
            [{"id": "doc1", "content": "..."}, ...]
        relevance_threshold: 相关性阈值（>= 此值视为相关）
    """
    all_labels = []

    for i, (query, docs) in enumerate(zip(queries, retrieved_docs)):
        print(f"处理查询 {i+1}/{len(queries)}: {query[:40]}...")

        query_labels = []
        relevant_ids = []

        for doc in docs:
            label = llm_judge_relevance(query, doc["content"], doc["id"])
            query_labels.append(label)

            if label.score >= relevance_threshold:
                relevant_ids.append(doc["id"])

            time.sleep(0.1)  # 限速

        all_labels.append({
            "query": query,
            "retrieved_doc_ids": [d["id"] for d in docs],
            "relevant_doc_ids": relevant_ids,
            "graded_relevance": {l.doc_id: l.score for l in query_labels},
            "labels": query_labels
        })

    return all_labels


# 使用示例
queries = ["公司的年假政策是什么？", "如何申请报销？"]
retrieved_docs = [
    [
        {"id": "doc1", "content": "年假政策：入职满一年后享有10天年假。"},
        {"id": "doc2", "content": "公司简介：XX公司成立于2018年。"},
        {"id": "doc3", "content": "请假流程：在OA系统中提交申请。"},
    ],
    [
        {"id": "doc4", "content": "报销流程：登录OA系统填写报销单。"},
        {"id": "doc5", "content": "年假政策：入职满一年后享有10天年假。"},
    ]
]

labels = auto_generate_labels(queries, retrieved_docs)

# 输出结果
for label in labels:
    print(f"\n查询：{label['query']}")
    print(f"  相关文档：{label['relevant_doc_ids']}")
    print(f"  详细评分：")
    for l in label["labels"]:
        print(f"    {l.doc_id}: {l.score} ({l.label}) - {l.reasoning}")
```

**要点**：
- 用 gpt-4o-mini 做相关性判断性价比最高，成本低且准确度够用
- 批量处理时要加限速（time.sleep），避免触发 API 限流
- 生成的标签必须人工抽样校验——建议每 100 条随机抽 10 条人工审核，一致率低于 90% 需要调整 Prompt
- 常见错误：直接用 LLM 生成的标签不做校验——LLM 可能对某些领域（如医疗、法律）的相关性判断有偏差
