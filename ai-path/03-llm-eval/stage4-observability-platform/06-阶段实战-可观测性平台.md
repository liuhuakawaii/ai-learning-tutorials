## 第一步：统一接入层

```python
# integration.py

import time
import uuid
from datetime import datetime
from core.logger import StructuredLogger
from core.tracer import LangfuseTracer
from core.metrics import MetricsCollector
from core.cost_monitor import CostMonitor
from alerts.alerter import AlertManager

class ObservabilityPlatform:
    def __init__(self, config: dict):
        self.logger = StructuredLogger(config)
        self.tracer = LangfuseTracer(config)
        self.metrics = MetricsCollector()
        self.cost_monitor = CostMonitor(config.get("pricing", {}))
        self.alerter = AlertManager(config.get("alert_rules", []))

    def track_request(self, request_id: str, user_id: str, feature: str):
        """开始追踪一个请求"""
        self.tracer.start_trace(request_id, user_id, feature)
        self.metrics.increment("requests_total", {"feature": feature})

    def track_llm_call(self, request_id: str, model: str, input_tokens: int,
                       output_tokens: int, latency_ms: float, success: bool,
                       error: str = None):
        """记录一次 LLM 调用"""
        cost = self.cost_monitor.calculate(model, input_tokens, output_tokens)

        self.logger.log_llm_call(request_id, model, input_tokens, output_tokens,
                                 latency_ms, success, error, cost)
        self.tracer.log_generation(request_id, model, input_tokens, output_tokens,
                                   latency_ms)
        self.metrics.record_histogram("llm_latency_ms", latency_ms, {"model": model})
        self.metrics.increment("llm_tokens_total", {"model": model, "type": "input"},
                               input_tokens)
        self.metrics.increment("llm_tokens_total", {"model": model, "type": "output"},
                               output_tokens)
        self.metrics.set_gauge("llm_cost_dollars", cost, {"model": model})

        self.alerter.check({"latency_ms": latency_ms, "cost": cost, "success": success})

    def track_quality_score(self, request_id: str, score: float):
        """记录质量评分"""
        self.tracer.log_score(request_id, "quality", score)
        self.metrics.record_histogram("quality_score", score)
        if score < 0.7:
            self.alerter.trigger("low_quality", {"request_id": request_id, "score": score})

    def get_dashboard_data(self) -> dict:
        """获取 Dashboard 数据"""
        return {
            "overview": self.metrics.get_overview(),
            "cost_report": self.cost_monitor.get_report(),
            "active_alerts": self.alerter.get_active(),
        }
```

## 第二步：成本监控

```python
# core/cost_monitor.py

from collections import defaultdict
from datetime import datetime

class CostMonitor:
    PRICING = {
        "gpt-4o": {"input": 2.50, "output": 10.00},
        "gpt-4o-mini": {"input": 0.15, "output": 0.60},
        "claude-3-5-sonnet": {"input": 3.00, "output": 15.00},
    }

    def __init__(self, custom_pricing: dict = None):
        if custom_pricing:
            self.PRICING.update(custom_pricing)
        self.usage = defaultdict(lambda: {"input_tokens": 0, "output_tokens": 0, "cost": 0})
        self.daily_costs = defaultdict(float)

    def calculate(self, model: str, input_tokens: int, output_tokens: int) -> float:
        pricing = self.PRICING.get(model, self.PRICING["gpt-4o-mini"])
        cost = (input_tokens * pricing["input"] + output_tokens * pricing["output"]) / 1_000_000

        self.usage[model]["input_tokens"] += input_tokens
        self.usage[model]["output_tokens"] += output_tokens
        self.usage[model]["cost"] += cost

        today = datetime.now().strftime("%Y-%m-%d")
        self.daily_costs[today] += cost

        return round(cost, 6)

    def get_report(self) -> dict:
        total_cost = sum(u["cost"] for u in self.usage.values())
        return {
            "total_cost": round(total_cost, 4),
            "by_model": {m: round(u["cost"], 4) for m, u in self.usage.items()},
            "daily_costs": dict(self.daily_costs),
        }
```

## 第三步：告警管理

```python
# alerts/alerter.py

from datetime import datetime

class AlertManager:
    def __init__(self, rules: list[dict]):
        self.rules = rules
        self.active_alerts = []

    def check(self, data: dict):
        for rule in self.rules:
            metric = rule["metric"]
            value = data.get(metric)
            if value is None:
                continue

            if rule["direction"] == "above" and value > rule["threshold"]:
                self.trigger(rule["name"], {"value": value, "threshold": rule["threshold"]})
            elif rule["direction"] == "below" and value < rule["threshold"]:
                self.trigger(rule["name"], {"value": value, "threshold": rule["threshold"]})

    def trigger(self, alert_name: str, details: dict):
        alert = {
            "name": alert_name,
            "details": details,
            "timestamp": datetime.now().isoformat(),
        }
        self.active_alerts.append(alert)
        print(f"🚨 告警: {alert_name} - {details}")

    def get_active(self) -> list[dict]:
        return self.active_alerts
```

st.set_page_config(page_title="AI 可观测性 Dashboard", layout="wide")

## 分阶段落地

不要一步到位。按这个顺序：

1. **第一周**：接入 Langfuse 做基础追踪 + 一个概览 Dashboard
2. **第二周**：加成本监控和预算预警
3. **第三周**：加质量评估和告警规则
4. **持续**：根据实际需求迭代

每阶段验证可用再进入下一阶段，比一次性搭建一个复杂但未经验证的系统更可靠。

## 常见问题

**过度追求架构完美？** 花三周设计"完美"架构，应用已经带着问题跑了三周没有监控。先用 Langfuse + 最简 Dashboard 跑起来。

**告警太多导致疲劳？** 严格分级：P0 电话、P1 即时消息、P2 邮件、P3 Dashboard 展示。

**只建平台不建流程？** 平台搭好了但没人看 Dashboard、没人响应告警。可观测性的价值在于"有人看、有人反应、有人改进"。

## 练习

1. 搭建一个包含成本监控的可观测性平台，集成到你的 AI 应用
2. 配置至少 2 个告警规则（如 Token 超限、延迟过高）
3. 连续 3 天检查 Dashboard，记录发现的问题
