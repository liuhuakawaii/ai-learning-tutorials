```python
class AnnotationTask:
    """标注任务"""
    
    def __init__(self, task_id: str, data: dict):
        self.task_id = task_id
        self.data = data
        self.annotations = []
        self.status = "pending"  # pending / in_progress / completed
    
    def add_annotation(self, annotator_id: str, labels: dict):
        """添加标注"""
        self.annotations.append({
            "annotator_id": annotator_id,
            "labels": labels,
            "timestamp": datetime.now().isoformat()
        })
    
    def get_consensus(self) -> dict:
        """获取共识标注"""
        if not self.annotations:
            return None
        
        # 多数投票
        label_counts = {}
        for ann in self.annotations:
            for label, value in ann["labels"].items():
                if label not in label_counts:
                    label_counts[label] = {}
                label_counts[label][value] = label_counts[label].get(value, 0) + 1
        
        consensus = {}
        for label, counts in label_counts.items():
            consensus[label] = max(counts, key=counts.get)
        
        return consensus

class AnnotationManager:
    """标注管理器"""
    
    def __init__(self):
        self.tasks = {}
        self.annotators = {}
    
    def create_batch(self, items: list[dict], annotators_per_item: int = 2):
        """创建标注批次"""
        for i, item in enumerate(items):
            task_id = f"task_{i}"
            self.tasks[task_id] = AnnotationTask(task_id, item)
    
    def assign_tasks(self, annotator_id: str, count: int):
        """分配任务给标注员"""
        pending = [t for t in self.tasks.values() if t.status == "pending"]
        assigned = pending[:count]
        
        for task in assigned:
            task.status = "in_progress"
            # 记录分配
        
        return [t.task_id for t in assigned]
```

## 二、标注一致性

### 2.1 Cohen's Kappa

```python
def calculate_cohens_kappa(annotations_a: list, annotations_b: list) -> float:
    """计算 Cohen's Kappa 系数"""
    
    # 计算一致率
    agreements = sum(1 for a, b in zip(annotations_a, annotations_b) if a == b)
    po = agreements / len(annotations_a)
    
    # 计算期望一致率
    unique_labels = set(annotations_a + annotations_b)
    pe = 0
    for label in unique_labels:
        pa = annotations_a.count(label) / len(annotations_a)
        pb = annotations_b.count(label) / len(annotations_b)
        pe += pa * pb
    
    # 计算 Kappa
    kappa = (po - pe) / (1 - pe) if pe < 1 else 1.0
    
    return kappa

# Kappa 解读：
# < 0.2: 极差
# 0.2-0.4: 差
# 0.4-0.6: 中等
# 0.6-0.8: 良好
# > 0.8: 优秀
```

## 常见误区

1. **标注指南写得太笼统**：只告诉标注员"判断回答质量好不好"，不同人理解完全不同。标注指南必须有明确的评分标准、边界案例示例和判断流程，否则标注一致性极低。

2. **不计算标注一致性就直接用数据**：两个标注员对同一批数据的判断差异很大，说明标注标准有问题。必须先用 Cohen's Kappa 等指标检验一致性，达标后再大规模标注。

3. **把人工标注当成一次性任务**：标注不是做一次就完了。模型迭代后评估标准可能变化，标注数据也需要更新。建立持续的标注-反馈-迭代机制才是关键。

4. **标注员选择不当导致数据偏差**：让不了解业务领域的标注员标注专业问答，会导致大量误判。标注员的专业背景必须与标注任务匹配。

## 工程建议

1. **标注指南要配具体示例**：每个评分等级至少配 3 个边界案例示例，让标注员对"3 分和 4 分的区别"有统一理解。示例比文字描述有效得多。

2. **用"金标准"数据集校准标注员**：在正式标注前，先用已知正确答案的金标准数据测试标注员，Kappa 达标后再开放正式标注任务。这能及早发现理解偏差。

3. **反馈回路要与评估系统打通**：将人工标注结果自动写入评估系统的数据库，让人工标注的发现能直接用于更新自动评估的校准阈值，形成人机协同的质量保障体系。

4. **标注任务优先级与业务影响挂钩**：不要均匀标注所有用例，优先标注用户投诉多、评估分数低、业务影响大的用例，让有限的标注资源产生最大价值。

### 练习一

**思路**：标注系统需要三个核心模块——任务管理（创建、分配、状态追踪）、标注界面（展示数据、收集标注）、结果汇总（共识计算、一致性检查）。关键是支持多人标注同一条数据以计算一致性。

**答案**：

```python
import uuid
from datetime import datetime
from dataclasses import dataclass, field
from typing import Optional


@dataclass
class AnnotationLabel:
    quality: int          # 1-5
    relevance: int        # 1-5
    faithfulness: int     # 1-5
    has_hallucination: bool
    comment: str = ""


@dataclass
class Annotation:
    annotator_id: str
    labels: AnnotationLabel
    timestamp: str
    duration_seconds: float = 0


@dataclass
class AnnotationTask:
    task_id: str
    data: dict
    status: str = "pending"  # pending / assigned / completed
    assigned_to: list[str] = field(default_factory=list)
    annotations: list[Annotation] = field(default_factory=list)
    required_annotations: int = 2

    def is_complete(self) -> bool:
        return len(self.annotations) >= self.required_annotations

    def get_consensus(self) -> dict:
        if not self.annotations:
            return None
        # 多数投票
        quality_scores = [a.labels.quality for a in self.annotations]
        relevance_scores = [a.labels.relevance for a in self.annotations]
        faith_scores = [a.labels.faithfulness for a in self.annotations]
        halluc_votes = [a.labels.has_hallucination for a in self.annotations]

        return {
            "quality": round(sum(quality_scores) / len(quality_scores), 1),
            "relevance": round(sum(relevance_scores) / len(relevance_scores), 1),
            "faithfulness": round(sum(faith_scores) / len(faith_scores), 1),
            "has_hallucination": sum(halluc_votes) > len(halluc_votes) / 2,
            "annotator_count": len(self.annotations),
        }


class AnnotationSystem:
    """标注系统"""

    def __init__(self):
        self.tasks: dict[str, AnnotationTask] = {}
        self.annotators: dict[str, dict] = {}

    def register_annotator(self, annotator_id: str, name: str, expertise: list[str]):
        self.annotators[annotator_id] = {
            "name": name,
            "expertise": expertise,
            "total_annotations": 0,
        }

    def create_batch(self, items: list[dict], required_per_item: int = 2) -> list[str]:
        """创建标注批次"""
        task_ids = []
        for item in items:
            task_id = str(uuid.uuid4())[:8]
            self.tasks[task_id] = AnnotationTask(
                task_id=task_id,
                data=item,
                required_annotations=required_per_item,
            )
            task_ids.append(task_id)
        return task_ids

    def assign_tasks(self, annotator_id: str, count: int) -> list[str]:
        """分配任务给标注员"""
        pending = [t for t in self.tasks.values()
                   if t.status == "pending" and annotator_id not in t.assigned_to]
        assigned = pending[:count]
        for task in assigned:
            task.assigned_to.append(annotator_id)
            task.status = "assigned"
        return [t.task_id for t in assigned]

    def submit_annotation(
        self,
        task_id: str,
        annotator_id: str,
        labels: dict,
        duration: float = 0,
    ):
        """提交标注"""
        task = self.tasks.get(task_id)
        if not task:
            raise ValueError(f"任务不存在: {task_id}")

        annotation = Annotation(
            annotator_id=annotator_id,
            labels=AnnotationLabel(**labels),
            timestamp=datetime.now().isoformat(),
            duration_seconds=duration,
        )
        task.annotations.append(annotation)

        if task.is_complete():
            task.status = "completed"

        self.annotators[annotator_id]["total_annotations"] += 1

    def get_batch_report(self) -> dict:
        total = len(self.tasks)
        completed = sum(1 for t in self.tasks.values() if t.status == "completed")
        pending = sum(1 for t in self.tasks.values() if t.status == "pending")
        return {
            "total_tasks": total,
            "completed": completed,
            "pending": pending,
            "in_progress": total - completed - pending,
            "completion_rate": completed / total if total > 0 else 0,
        }

# 注册标注员
system.register_annotator("ann_001", "张三", ["技术文档", "客服"])
system.register_annotator("ann_002", "李四", ["技术文档", "法律"])

# 创建标注批次
items = [
    {"question": "什么是 RAG？", "answer": "RAG 是检索增强生成..."},
    {"question": "如何优化 Prompt？", "answer": "Prompt 优化可以从..."},
    {"question": "向量数据库怎么选？", "answer": "向量数据库选型..."},
]
task_ids = system.create_batch(items, required_per_item=2)

# 分配任务
assigned_1 = system.assign_tasks("ann_001", 3)
assigned_2 = system.assign_tasks("ann_002", 3)

# 提交标注
system.submit_annotation(task_ids[0], "ann_001", {
    "quality": 4, "relevance": 5, "faithfulness": 4, "has_hallucination": False,
})
system.submit_annotation(task_ids[0], "ann_002", {
    "quality": 3, "relevance": 4, "faithfulness": 4, "has_hallucination": False,
})

# 查看共识
task = system.tasks[task_ids[0]]
print(f"任务 {task.task_id} 共识: {task.get_consensus()}")
print(f"批次报告: {system.get_batch_report()}")
```

**要点**：
- 标注系统必须支持多人标注同一条数据，这是计算一致性的前提
- `required_per_item` 参数控制每条数据需要几人标注，通常 2-3 人
- 常见错误：每条数据只分配给一个人标注，导致无法计算一致性，也无法发现标注质量问题

### 练习二

**思路**：Cohen's Kappa 衡量两个标注员的一致程度，排除随机一致的影响。计算步骤：先算观察一致率（Po），再算期望一致率（Pe），最后用公式 (Po - Pe) / (1 - Pe)。

**答案**：

```python
def calculate_cohens_kappa(annotator_a: list[str], annotator_b: list[str]) -> dict:
    """计算 Cohen's Kappa 系数"""

    assert len(annotator_a) == len(annotator_b), "两个标注序列长度必须一致"
    n = len(annotator_a)

    # 计算观察一致率 Po
    agreements = sum(1 for a, b in zip(annotator_a, annotator_b) if a == b)
    po = agreements / n

    # 计算期望一致率 Pe
    unique_labels = sorted(set(annotator_a + annotator_b))
    pe = 0
    for label in unique_labels:
        pa = annotator_a.count(label) / n
        pb = annotator_b.count(label) / n
        pe += pa * pb

    # 计算 Kappa
    kappa = (po - pe) / (1 - pe) if pe < 1 else 1.0

    # 解读
    if kappa < 0.2:
        interpretation = "极差——标注标准需要完全重写"
    elif kappa < 0.4:
        interpretation = "差——标注员理解差异大，需要重新培训"
    elif kappa < 0.6:
        interpretation = "中等——有一定一致性，但标注指南需细化"
    elif kappa < 0.8:
        interpretation = "良好——一致性较高，可用于生产"
    else:
        interpretation = "优秀——标注质量很高"

    # 逐类别一致性
    per_label = {}
    for label in unique_labels:
        both = sum(1 for a, b in zip(annotator_a, annotator_b) if a == label and b == label)
        either = sum(1 for a, b in zip(annotator_a, annotator_b) if a == label or b == label)
        per_label[label] = round(both / either, 2) if either > 0 else 1.0

    return {
        "kappa": round(kappa, 3),
        "po": round(po, 3),
        "pe": round(pe, 3),
        "interpretation": interpretation,
        "sample_size": n,
        "per_label_agreement": per_label,
    }


# 模拟两个标注员的标注结果
annotator_a = ["good", "good", "bad", "good", "bad", "good", "bad", "bad", "good", "good"]
annotator_b = ["good", "bad", "bad", "good", "bad", "good", "good", "bad", "good", "good"]

result = calculate_cohens_kappa(annotator_a, annotator_b)

print("=== Cohen's Kappa 分析 ===\n")
print(f"样本数: {result['sample_size']}")
print(f"观察一致率 Po: {result['po']}")
print(f"期望一致率 Pe: {result['pe']}")
print(f"Kappa 系数: {result['kappa']}")
print(f"解读: {result['interpretation']}")
print(f"\n逐类别一致率:")
for label, agree in result["per_label_agreement"].items():
    print(f"  {label}: {agree}")
```

**要点**：
- Kappa 比简单一致率（Po）更可靠——它排除了"碰巧一致"的影响
- Kappa < 0.6 时应暂停标注，先优化标注指南和培训标注员
- 常见错误：用简单一致率代替 Kappa，当标注分布不均衡时（如 90% 都是 good），简单一致率会虚高

### 练习三

**思路**：用户反馈收集需要低摩擦（用户点一下就能反馈）、结构化（便于分析）和可关联（反馈要关联到具体的 Trace）。关键是设计好反馈的触发时机和数据结构。

**答案**：

```python
import uuid
from datetime import datetime
from dataclasses import dataclass
from typing import Optional


@dataclass
class UserFeedback:
    feedback_id: str
    trace_id: str
    user_id: str
    feedback_type: str      # "thumbs_up" / "thumbs_down" / "rating" / "text"
    rating: Optional[int]   # 1-5
    text: Optional[str]     # 自由文本
    category: Optional[str] # "wrong" / "incomplete" / "irrelevant" / "harmful" / "other"
    timestamp: str


class FeedbackCollector:
    """用户反馈收集器"""

    def __init__(self):
        self.feedback_store: list[UserFeedback] = []

    def collect_thumbs(
        self,
        trace_id: str,
        user_id: str,
        is_positive: bool,
        reason: str = None,
    ) -> UserFeedback:
        """收集点赞/踩"""
        feedback = UserFeedback(
            feedback_id=str(uuid.uuid4())[:8],
            trace_id=trace_id,
            user_id=user_id,
            feedback_type="thumbs_up" if is_positive else "thumbs_down",
            rating=None,
            text=reason,
            category=None,
            timestamp=datetime.now().isoformat(),
        )
        self.feedback_store.append(feedback)
        return feedback

    def collect_rating(
        self,
        trace_id: str,
        user_id: str,
        rating: int,
        comment: str = None,
    ) -> UserFeedback:
        """收集评分"""
        feedback = UserFeedback(
            feedback_id=str(uuid.uuid4())[:8],
            trace_id=trace_id,
            user_id=user_id,
            feedback_type="rating",
            rating=rating,
            text=comment,
            category=None,
            timestamp=datetime.now().isoformat(),
        )
        self.feedback_store.append(feedback)
        return feedback

    def collect_detailed(
        self,
        trace_id: str,
        user_id: str,
        category: str,
        text: str,
    ) -> UserFeedback:
        """收集详细反馈（踩之后的追问）"""
        feedback = UserFeedback(
            feedback_id=str(uuid.uuid4())[:8],
            trace_id=trace_id,
            user_id=user_id,
            feedback_type="detailed",
            rating=None,
            text=text,
            category=category,
            timestamp=datetime.now().isoformat(),
        )
        self.feedback_store.append(feedback)
        return feedback

    def get_stats(self) -> dict:
        if not self.feedback_store:
            return {"total": 0}

        total = len(self.feedback_store)
        thumbs_up = sum(1 for f in self.feedback_store if f.feedback_type == "thumbs_up")
        thumbs_down = sum(1 for f in self.feedback_store if f.feedback_type == "thumbs_down")
        ratings = [f.rating for f in self.feedback_store if f.rating is not None]

        by_category = {}
        for f in self.feedback_store:
            if f.category:
                by_category.setdefault(f.category, 0)
                by_category[f.category] += 1

        return {
            "total": total,
            "thumbs_up": thumbs_up,
            "thumbs_down": thumbs_down,
            "satisfaction_rate": thumbs_up / (thumbs_up + thumbs_down) if (thumbs_up + thumbs_down) > 0 else None,
            "avg_rating": sum(ratings) / len(ratings) if ratings else None,
            "by_category": by_category,
        }


# 使用示例
collector = FeedbackCollector()

# 收集反馈
collector.collect_thumbs("trace_001", "user_001", True)
collector.collect_thumbs("trace_002", "user_002", False, "回答不准确")
collector.collect_rating("trace_003", "user_001", 4, "回答不错但可以更详细")
collector.collect_detailed("trace_002", "user_002", "wrong", "关于 RAG 的定义回答错误")

stats = collector.get_stats()
print("=== 用户反馈统计 ===\n")
print(f"总反馈数: {stats['total']}")
print(f"满意度: {stats['satisfaction_rate']:.1%}")
print(f"平均评分: {stats['avg_rating']}")
print(f"问题分布: {stats['by_category']}")
```

**要点**：
- 反馈要低摩擦——点赞/踩一步完成，详细反馈是可选的追问，不要强制用户填表
- 反馈必须关联到 `trace_id`，这样才能将用户反馈与自动评估指标对比，发现自动评估的盲区
- 常见错误：反馈收集后不与评估系统打通，导致人工反馈无法用于校准自动评估阈值
