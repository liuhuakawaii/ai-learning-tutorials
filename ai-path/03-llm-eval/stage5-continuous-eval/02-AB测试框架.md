### 1.1 实验框架

```python
class ABTest:
    """A/B 测试框架"""
    
    def __init__(self, name: str, variants: list[dict]):
        self.name = name
        self.variants = variants  # [{"name": "A", "config": {...}}, ...]
        self.results = {v["name"]: [] for v in variants}
    
    def assign_variant(self, user_id: str) -> str:
        """分配变体（确定性）"""
        hash_val = hash(f"{self.name}_{user_id}") % len(self.variants)
        return self.variants[hash_val]["name"]
    
    def record_result(self, variant: str, metrics: dict):
        """记录结果"""
        self.results[variant].append(metrics)
    
    def analyze(self) -> dict:
        """分析结果"""
        analysis = {}
        
        for variant_name, results in self.results.items():
            if not results:
                continue
            
            df = pd.DataFrame(results)
            analysis[variant_name] = {
                "count": len(df),
                "avg_quality": df["quality"].mean(),
                "avg_latency": df["latency"].mean(),
                "success_rate": df["success"].mean(),
            }
        
        return analysis
    
    def check_significance(self, metric: str, confidence: float = 0.95) -> dict:
        """检查统计显著性"""
        from scipy import stats
        
        variant_a = [r[metric] for r in self.results[self.variants[0]["name"]]]
        variant_b = [r[metric] for r in self.results[self.variants[1]["name"]]]
        
        t_stat, p_value = stats.ttest_ind(variant_a, variant_b)
        
        return {
            "metric": metric,
            "t_statistic": t_stat,
            "p_value": p_value,
            "significant": p_value < (1 - confidence),
            "variant_a_mean": np.mean(variant_a),
            "variant_b_mean": np.mean(variant_b),
            "improvement": (np.mean(variant_b) - np.mean(variant_a)) / np.mean(variant_a) * 100
        }
```

## 三、样本量计算

```python
def calculate_sample_size(
    baseline_rate: float,
    minimum_detectable_effect: float,
    alpha: float = 0.05,
    power: float = 0.8
) -> int:
    """计算所需样本量"""
    from scipy import stats
    
    z_alpha = stats.norm.ppf(1 - alpha / 2)
    z_beta = stats.norm.ppf(power)
    
    p1 = baseline_rate
    p2 = baseline_rate * (1 + minimum_detectable_effect)
    
    n = (
        (z_alpha * np.sqrt(2 * p1 * (1 - p1)) +
         z_beta * np.sqrt(p1 * (1 - p1) + p2 * (1 - p2))) ** 2 /
        (p2 - p1) ** 2
    )
    
    return int(np.ceil(n))

# 示例：检测 5% 的质量提升
sample_size = calculate_sample_size(
    baseline_rate=0.8,
    minimum_detectable_effect=0.05
)
print(f"每组需要 {sample_size} 个样本")
```

## 四、实验报告

```python
def generate_ab_report(test: ABTest) -> str:
    """生成 A/B 测试报告"""
    
    analysis = test.analyze()
    
    report = f"# A/B 测试报告：{test.name}\n\n"
    
    report += "## 变体对比\n\n"
    report += "| 变体 | 样本数 | 平均质量 | 平均延迟 | 成功率 |\n"
    report += "|------|--------|----------|----------|--------|\n"
    
    for name, stats in analysis.items():
        report += f"| {name} | {stats['count']} | {stats['avg_quality']:.3f} | {stats['avg_latency']:.2f}s | {stats['success_rate']:.1%} |\n"
    
    # 显著性检验
    if len(analysis) == 2:
        sig = test.check_significance("quality")
        report += f"\n## 统计显著性\n\n"
        report += f"- p-value: {sig['p_value']:.4f}\n"
        report += f"- 显著: {'是' if sig['significant'] else '否'}\n"
        report += f"- 改进: {sig['improvement']:.1f}%\n"
    
    return report
```

## 常见误区

1. **样本量不够就下结论**：跑了一天数据、各组 50 个样本就宣布胜者，这是最常见的错误。样本量不足时，观察到的差异很可能是随机波动，必须做统计显著性检验。

2. **只看平均值不看分布**：A 组平均质量 0.82、B 组 0.80，看似 A 赢了，但 B 组在长尾用例上可能明显更稳定。除了均值，还要看方差、分位数和退化案例分布。

3. **同时改变多个变量**：同时换了 Prompt 又改了 temperature，即使效果变好了也无法归因。A/B 测试的核心原则是每次只变一个变量，否则实验结果无法解释。

4. **忽略实验的"学习效应"**：用户在使用过程中会适应系统的回答风格，前几次交互的数据和后续数据的分布可能不同。实验要跑足够长的时间以覆盖这种适应期。

## 工程建议

1. **先算样本量再开实验**：在启动 A/B 测试前，用历史数据估算所需样本量和实验时长。避免跑了一周发现样本不够、又多跑两周，浪费时间和 API 成本。

2. **用确定性哈希分配变体**：基于 user_id 做哈希分配，保证同一用户始终看到同一变体，避免用户体验不一致。同时便于按用户维度分析长期效果。

3. **建立实验看板自动化报告**：实验运行期间自动生成中间报告（样本量、各项指标趋势、显著性变化），让团队无需手动拉数据就能跟踪实验进展。

4. **实验结束后留存完整记录**：记录实验假设、配置、样本量、结果和结论，形成实验知识库。避免未来重复做类似的实验，也方便新成员了解历史决策。

### 练习一

**思路**：Prompt A/B 测试需要定义两个变体（当前版本 vs 优化版本），在同一数据集上运行，用相同的评估指标对比。关键是每次只改一个变量，并用统计显著性检验判断差异是否真实。

**答案**：

```python
import json
import numpy as np
from scipy import stats


class PromptABTest:
    """Prompt A/B 测试框架"""

    def __init__(self, name: str, prompt_a: str, prompt_b: str):
        self.name = name
        self.prompts = {"A": prompt_a, "B": prompt_b}
        self.results = {"A": [], "B": []}

    def run(self, dataset: list[dict], llm_fn, eval_fn):
        """在同一数据集上运行两个 Prompt"""
        for item in dataset:
            for variant, prompt in self.prompts.items():
                # 构建完整 prompt
                full_prompt = f"{prompt}\n\n用户问题: {item['question']}"
                # 调用 LLM
                answer = llm_fn(full_prompt)
                # 评估质量
                score = eval_fn(item["question"], answer, item.get("reference"))
                self.results[variant].append({
                    "question": item["question"],
                    "answer": answer,
                    "quality": score,
                })

    def analyze(self) -> dict:
        """分析结果"""
        scores_a = [r["quality"] for r in self.results["A"]]
        scores_b = [r["quality"] for r in self.results["B"]]

        t_stat, p_value = stats.ttest_ind(scores_a, scores_b)
        improvement = (np.mean(scores_b) - np.mean(scores_a)) / np.mean(scores_a) * 100

        return {
            "variant_a": {
                "count": len(scores_a),
                "mean": np.mean(scores_a),
                "std": np.std(scores_a),
                "median": np.median(scores_a),
            },
            "variant_b": {
                "count": len(scores_b),
                "mean": np.mean(scores_b),
                "std": np.std(scores_b),
                "median": np.median(scores_b),
            },
            "improvement_pct": improvement,
            "t_statistic": t_stat,
            "p_value": p_value,
            "significant": p_value < 0.05,
            "winner": "B" if np.mean(scores_b) > np.mean(scores_a) and p_value < 0.05 else "A",
        }
```

### 练习二

**思路**：样本量计算需要考虑基线质量、最小可检测效应（MDE）、显著性水平（α）和统计功效（1-β）。MDE 越小需要的样本量越大。

**答案**：

```python
import numpy as np
from scipy import stats


def calculate_sample_size(
    baseline_mean: float,
    baseline_std: float,
    minimum_detectable_effect: float,
    alpha: float = 0.05,
    power: float = 0.8,
) -> dict:
    """计算两组对比所需的样本量"""

    z_alpha = stats.norm.ppf(1 - alpha / 2)
    z_beta = stats.norm.ppf(power)

    # 效应量（Cohen's d）
    effect_size = minimum_detectable_effect / baseline_std

    # 样本量公式（两组 t 检验）
    n = ((z_alpha + z_beta) / effect_size) ** 2 * 2

    n_ceil = int(np.ceil(n))

    # 计算不同 MDE 下的样本量
    mde_range = [0.03, 0.05, 0.08, 0.10, 0.15]
    sample_sizes = {}
    for mde in mde_range:
        es = mde / baseline_std
        n_mde = int(np.ceil(((z_alpha + z_beta) / es) ** 2 * 2))
        sample_sizes[f"{mde:.0%}"] = n_mde

    return {
        "input": {
            "baseline_mean": baseline_mean,
            "baseline_std": baseline_std,
            "mde": minimum_detectable_effect,
            "alpha": alpha,
            "power": power,
        },
        "result": {
            "sample_size_per_group": n_ceil,
            "total_samples": n_ceil * 2,
            "effect_size_cohens_d": round(effect_size, 3),
        },
        "sensitivity": sample_sizes,
    }


# 计算检测 10% 质量提升所需的样本量
result = calculate_sample_size(
    baseline_mean=0.80,
    baseline_std=0.15,
    minimum_detectable_effect=0.08,  # 10% of 0.80 = 0.08
)

print("=== 样本量计算结果 ===\n")
print(f"基线均值: {result['input']['baseline_mean']}")
print(f"基线标准差: {result['input']['baseline_std']}")
print(f"最小可检测效应: {result['input']['mde']}")
print(f"显著性水平 α: {result['input']['alpha']}")
print(f"统计功效 1-β: {result['input']['power']}")
print(f"\n每组需要样本: {result['result']['sample_size_per_group']}")
print(f"总计需要样本: {result['result']['total_samples']}")
print(f"效应量 (Cohen's d): {result['result']['effect_size_cohens_d']}")

print("\n不同 MDE 下的样本量需求:")
for mde, n in result["sensitivity"].items():
    print(f"  MDE {mde}: 每组 {n} 个样本")
```

**要点**：
- 检测 10% 的质量提升（0.08 的绝对差异），在基线标准差 0.15 的情况下，每组约需要 56 个样本
- MDE 越小需要的样本量越大——检测 3% 的提升可能需要 500+ 样本，成本很高
- 常见错误：不计算样本量就开实验，跑了一周发现样本不够，又多跑两周浪费时间和 API 成本

### 练习三

**思路**：分析 A/B 测试结果需要看均值差异、效应量、p 值和置信区间。p < 0.05 只是及格线，还要看效应量是否具有实际意义。

**答案**：

```python
import numpy as np
from scipy import stats


def analyze_ab_test_result(scores_a: list[float], scores_b: list[float]) -> dict:
    """分析 A/B 测试结果，判断统计显著性"""

    n_a, n_b = len(scores_a), len(scores_b)
    mean_a, mean_b = np.mean(scores_a), np.mean(scores_b)
    std_a, std_b = np.std(scores_a, ddof=1), np.std(scores_b, ddof=1)

    # t 检验
    t_stat, p_value = stats.ttest_ind(scores_a, scores_b)

    # 效应量（Cohen's d）
    pooled_std = np.sqrt(((n_a - 1) * std_a**2 + (n_b - 1) * std_b**2) / (n_a + n_b - 2))
    cohens_d = (mean_b - mean_a) / pooled_std

    # 95% 置信区间
    se = pooled_std * np.sqrt(1/n_a + 1/n_b)
    ci_lower = (mean_b - mean_a) - 1.96 * se
    ci_upper = (mean_b - mean_a) + 1.96 * se

    # 改进百分比
    improvement_pct = (mean_b - mean_a) / mean_a * 100

    # 效应量解读
    if abs(cohens_d) < 0.2:
        effect_label = "微小"
    elif abs(cohens_d) < 0.5:
        effect_label = "小"
    elif abs(cohens_d) < 0.8:
        effect_label = "中等"
    else:
        effect_label = "大"

    # 决策建议
    if p_value < 0.05 and abs(cohens_d) >= 0.2:
        decision = "✅ B 显著优于 A，建议采用 Prompt B"
    elif p_value < 0.05 and abs(cohens_d) < 0.2:
        decision = "⚠️ 统计显著但效应量太小，实际改进不明显，建议继续观察"
    elif p_value >= 0.05 and n_a + n_b < 100:
        decision = "⚠️ 样本量可能不足，建议增加样本后重新测试"
    else:
        decision = "❌ 无显著差异，保持当前版本"

    return {
        "sample_sizes": {"A": n_a, "B": n_b},
        "means": {"A": round(mean_a, 4), "B": round(mean_b, 4)},
        "stds": {"A": round(std_a, 4), "B": round(std_b, 4)},
        "improvement_pct": round(improvement_pct, 2),
        "t_statistic": round(t_stat, 4),
        "p_value": round(p_value, 4),
        "significant_005": p_value < 0.05,
        "cohens_d": round(cohens_d, 3),
        "effect_label": effect_label,
        "confidence_interval": f"[{ci_lower:.4f}, {ci_upper:.4f}]",
        "decision": decision,
    }


# 模拟 A/B 测试结果
np.random.seed(42)
scores_a = np.random.normal(0.78, 0.12, 80).tolist()
scores_b = np.random.normal(0.83, 0.11, 80).tolist()

result = analyze_ab_test_result(scores_a, scores_b)

print("=== A/B 测试分析报告 ===\n")
print(f"样本量: A={result['sample_sizes']['A']}, B={result['sample_sizes']['B']}")
print(f"均值: A={result['means']['A']}, B={result['means']['B']}")
print(f"改进: {result['improvement_pct']}%")
print(f"t 统计量: {result['t_statistic']}")
print(f"p 值: {result['p_value']}")
print(f"统计显著 (α=0.05): {'是' if result['significant_005'] else '否'}")
print(f"效应量: {result['cohens_d']} ({result['effect_label']})")
print(f"95% 置信区间: {result['confidence_interval']}")
print(f"\n决策: {result['decision']}")
```

**要点**：
- p < 0.05 只是及格线，还要看效应量——Cohen's d < 0.2 表示即使统计显著，实际改进也很小
- 置信区间比 p 值更有用——它告诉你真实差异的范围，而不只是"有没有差异"
- 常见错误：只看 p 值就下结论，忽略了效应量和置信区间，导致采纳了"统计显著但实际无意义"的改进
