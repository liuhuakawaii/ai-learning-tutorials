# 04 回归测试——确保优化不引入新问题的策略

> 每次优化都可能引入新问题。回归测试是你的"安全网"。

## 场景引入

你的 RAG 应用在"法律合同"类问题上表现很好，忠实度达到 0.9。为了提升"技术文档"类问题的回答质量，你优化了检索策略和 Prompt。上线后技术文档类问题确实改善了，但法律合同类问题的忠实度骤降到 0.7——优化引入了回归，而你没有一套机制来检测它。回归测试就是解决这个问题的"安全网"。

```python
class RegressionTestSuite:
    """回归测试套件"""
    
    def __init__(self, dataset_path: str):
        with open(dataset_path, "r") as f:
            self.dataset = json.load(f)
        self.baseline = None
        self.current = None
    
    def run_baseline(self, pipeline_fn):
        """运行基线测试"""
        self.baseline = self._run_tests(pipeline_fn)
        return self.baseline
    
    def run_current(self, pipeline_fn):
        """运行当前版本测试"""
        self.current = self._run_tests(pipeline_fn)
        return self.current
    
    def _run_tests(self, pipeline_fn) -> list[dict]:
        """运行测试"""
        results = []
        for item in self.dataset:
            result = pipeline_fn(item["question"])
            quality = evaluate_quality(item["question"], result["answer"], item.get("reference"))
            results.append({
                "id": item["id"],
                "question": item["question"],
                "answer": result["answer"],
                "quality": quality,
                "latency": result.get("latency", 0)
            })
        return results
    
    def compare(self) -> dict:
        """对比基线和当前版本"""
        if not self.baseline or not self.current:
            raise ValueError("请先运行基线和当前版本测试")
        
        baseline_df = pd.DataFrame(self.baseline)
        current_df = pd.DataFrame(self.current)
        
        # 计算变化
        quality_change = current_df["quality"].mean() - baseline_df["quality"].mean()
        latency_change = current_df["latency"].mean() - baseline_df["latency"].mean()
        
        # 找出退化的案例
        regressions = []
        for b, c in zip(self.baseline, self.current):
            if c["quality"] < b["quality"] - 0.1:  # 质量下降超过 0.1
                regressions.append({
                    "id": b["id"],
                    "question": b["question"][:50],
                    "baseline_quality": b["quality"],
                    "current_quality": c["quality"],
                    "change": c["quality"] - b["quality"]
                })
        
        return {
            "quality_change": quality_change,
            "latency_change": latency_change,
            "regressions": regressions,
            "regression_count": len(regressions),
            "passed": len(regressions) == 0 and quality_change >= -0.02
        }
```

```python
def run_regression_test(config: dict) -> bool:

    # 对比结果

    # 输出报告

    print(f"延迟变化: {comparison['latency_change']:+.3f}s")
    print(f"退化案例: {comparison['regression_count']}")
    
    if comparison["regressions"]:
        print("\n退化案例:")
        for reg in comparison["regressions"]:
            print(f"  - {reg['question']}: {reg['baseline_quality']:.3f} → {reg['current_quality']:.3f}")

        print("\n✅ 回归测试通过")
        return True
    else:
        print("\n❌ 回归测试失败")
        return False
```

## 三、回归测试策略

```
回归测试策略：

1. 全量回归
   - 每次优化后运行全部测试
   - 耗时长，但最全面
   - 适用于重大优化

2. 增量回归
   - 只运行受影响的测试用例
   - 快速，但可能遗漏
   - 适用于小优化

3. 采样回归
   - 随机采样部分测试用例
   - 平衡速度和覆盖率
   - 适用于日常开发

4. 分层回归
   - 按重要性分层
   - 核心用例每次运行
   - 非核心用例定期运行
```

## 常见误区

1. **基线数据集一成不变**：基线是回归测试的参照物，但如果基线数据集本身就有问题或不够全面，回归测试就失去了意义。基线需要定期扩充和更新，覆盖新发现的典型用例。

2. **只关注整体平均分，忽略单条退化**：整体质量从 0.82 变成 0.81 看似问题不大，但可能有一类特定问题从 0.9 掉到了 0.5。回归测试必须逐条对比，找出退化的具体用例并分析退化模式。

3. **退化阈值设得太松或太严**：阈值太松会放过真正的回归，太严会导致频繁误报、团队疲劳。建议根据业务容忍度设定，核心场景用更严格的阈值，非核心场景适当放宽。

4. **回归测试只在发布前跑一次**：等到发布前才发现回归，修复成本很高。应该在每次代码/Prompt 变更时都运行回归测试，尽早发现问题。

## 工程建议

1. **分层管理测试用例**：将回归测试集分为"核心层"（必须每次 CI 都跑）和"扩展层"（每日定时跑）。核心层用例少而精、运行快，扩展层覆盖面广但耗时，两者配合兼顾速度和覆盖率。

2. **退化案例自动归档并关联优化任务**：回归测试发现退化用例后，自动将其归档到"退化案例库"，并关联到对应的优化任务。这能确保每个退化都被跟踪到修复，不会遗漏。

3. **基线版本化管理**：每次发布后将当前评估结果保存为新基线，并打上版本标签。这样不仅知道"相对上一个版本退化了多少"，还能追溯任意历史版本之间的差异。

4. **回归测试结果可视化趋势**：用图表展示每次回归测试的质量指标变化趋势，让团队一眼看出整体质量是在上升还是下降，而不是每次都看一份静态报告。

### 练习一

**思路**：回归测试套件需要分层管理——核心层（每次 CI 都跑）和扩展层（每日定时跑）。测试数据集要覆盖不同场景类型，每条用例有明确的评估指标和退化阈值。

**答案**：

```python
import json
from dataclasses import dataclass
from typing import Optional


@dataclass
class RegressionTestCase:
    id: str
    question: str
    category: str           # 场景分类
    reference: str          # 参考答案
    layer: str              # "core" / "extended"
    expected_min_quality: float
    expected_max_latency: float = 10.0


class RegressionTestSuite:
    """回归测试套件"""

    def __init__(self):
        self.test_cases: list[RegressionTestCase] = []
        self.baseline: dict = {}
        self.current_results: dict = {}

    def add_case(self, case: RegressionTestCase):
        self.test_cases.append(case)

    def load_from_file(self, path: str):
        with open(path, "r") as f:
            data = json.load(f)
        for item in data:
            self.add_case(RegressionTestCase(**item))

    def save_to_file(self, path: str):
        data = [
            {
                "id": c.id, "question": c.question, "category": c.category,
                "reference": c.reference, "layer": c.layer,
                "expected_min_quality": c.expected_min_quality,
                "expected_max_latency": c.expected_max_latency,
            }
            for c in self.test_cases
        ]
        with open(path, "w") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def run_baseline(self, pipeline_fn) -> dict:
        """运行基线测试"""
        self.baseline = self._run_all(pipeline_fn)
        return self.baseline

    def run_current(self, pipeline_fn, layer: str = "core") -> dict:
        """运行当前版本测试"""
        cases = [c for c in self.test_cases if c.layer == layer] if layer != "all" else self.test_cases
        self.current_results = self._run_subset(pipeline_fn, cases)
        return self.current_results

    def _run_all(self, pipeline_fn) -> dict:
        results = {}
        for case in self.test_cases:
            result = pipeline_fn(case.question)
            results[case.id] = {
                "question": case.question,
                "answer": result["answer"],
                "quality": result.get("quality", 0),
                "latency": result.get("latency", 0),
            }
        return results

    def _run_subset(self, pipeline_fn, cases) -> dict:
        results = {}
        for case in cases:
            result = pipeline_fn(case.question)
            results[case.id] = {
                "question": case.question,
                "answer": result["answer"],
                "quality": result.get("quality", 0),
                "latency": result.get("latency", 0),
            }
        return results

    def compare(self, regression_threshold: float = 0.1) -> dict:
        """对比基线和当前版本"""
        regressions = []
        improvements = []

        for case_id, current in self.current_results.items():
            baseline = self.baseline.get(case_id)
            if not baseline:
                continue

            quality_change = current["quality"] - baseline["quality"]
            latency_change = current["latency"] - baseline["latency"]

            if quality_change < -regression_threshold:
                regressions.append({
                    "id": case_id,
                    "question": current["question"][:50],
                    "baseline_quality": baseline["quality"],
                    "current_quality": current["quality"],
                    "change": quality_change,
                })
            elif quality_change > regression_threshold:
                improvements.append({
                    "id": case_id,
                    "question": current["question"][:50],
                    "change": quality_change,
                })

        avg_baseline = sum(r["quality"] for r in self.baseline.values()) / len(self.baseline) if self.baseline else 0
        avg_current = sum(r["quality"] for r in self.current_results.values()) / len(self.current_results) if self.current_results else 0

        return {
            "overall_change": avg_current - avg_baseline,
            "regressions": regressions,
            "regression_count": len(regressions),
            "improvements": improvements,
            "improvement_count": len(improvements),
            "passed": len(regressions) == 0 and (avg_current - avg_baseline) >= -0.02,
        }

test_cases = [

    RegressionTestCase("core_003", "向量数据库的选型标准是什么？", "vector_db", "向量数据库选型需要考虑性能和成本", "core", 0.70),
    RegressionTestCase("ext_001", "法律合同中的不可抗力条款如何理解？", "legal", "不可抗力条款是指...", "extended", 0.80),
    RegressionTestCase("ext_002", "技术文档中的 API 版本管理最佳实践", "tech_doc", "API 版本管理...", "extended", 0.75),
]

    suite.add_case(case)
```

**要点**：
- 核心层用例少而精（5-10 条），每次 CI 都跑，运行时间控制在 2 分钟内
- 扩展层覆盖面广（30-50 条），每日定时跑，不阻断 CI

### 练习二

**思路**：运行回归测试后，重点分析退化案例——质量下降超过阈值的用例。对每个退化案例分析是检索问题、Prompt 问题还是模型问题。

**答案**：

```python
import json


def analyze_regressions(comparison_result: dict) -> dict:
    """分析退化案例，找出退化模式"""

    regressions = comparison_result.get("regressions", [])
    if not regressions:
        return {"status": "no_regressions", "message": "无退化案例"}

    print("=== 退化案例分析 ===\n")
    print(f"退化案例数: {len(regressions)}\n")

    # 按退化程度排序
    regressions.sort(key=lambda x: x["change"])

    for reg in regressions:
        print(f"案例 {reg['id']}: {reg['question']}...")
        print(f"  基线质量: {reg['baseline_quality']:.3f}")
        print(f"  当前质量: {reg['current_quality']:.3f}")
        print(f"  变化: {reg['change']:+.3f}")
        print()

    # 分析退化模式
    avg_degradation = sum(r["change"] for r in regressions) / len(regressions)
    max_degradation = min(r["change"] for r in regressions)

    patterns = []
    if len(regressions) >= 3:
        patterns.append("多条用例同时退化，可能是系统性问题（如 Prompt 变更或模型更新）")
    if max_degradation < -0.2:
        patterns.append("存在严重退化（>0.2），需要紧急修复")
    if avg_degradation < -0.1:
        patterns.append("平均退化超过 0.1，整体质量下降明显")

    return {
        "status": "regressions_found",
        "regression_count": len(regressions),
        "avg_degradation": avg_degradation,
        "max_degradation": max_degradation,
        "patterns": patterns,
        "action_items": [
            "1. 检查最近的 Prompt 变更是否影响了退化用例的场景",
            "2. 检查检索策略是否发生了变化",
            "3. 检查模型版本是否有更新",
            "4. 对严重退化用例单独调试定位根因",
        ],
    }


# 模拟回归测试结果
mock_comparison = {
```

**要点**：
- 退化分析要区分"系统性退化"（多条用例同时退化）和"个别退化"（单条用例退化）
- 系统性退化通常是 Prompt 变更或模型更新导致，个别退化可能是特定场景的边界问题
- 常见错误：只看整体平均分变化（-0.05），忽略了严重退化的单条用例（-0.23），导致"平均分还行但个别场景崩了"
