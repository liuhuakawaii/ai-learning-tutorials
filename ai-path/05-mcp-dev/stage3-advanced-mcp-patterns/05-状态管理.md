# 05 状态管理——有状态 Server 的设计与会话管理

> 有些 MCP Server 需要维护状态，比如数据库连接、会话信息等。

## 场景引入

你的 MCP Server 需要连接 PostgreSQL 数据库。每次 tools/call 都新建连接、用完关闭，性能很差。你想用连接池复用连接，但连接池是 Server 级别的状态，怎么在多个 Tool 调用之间共享？更复杂的是，如果用户要执行一个多步事务（BEGIN → INSERT → COMMIT），三次 tools/call 必须用同一个数据库连接——这就是有状态 Server 要解决的问题。

## 二、会话管理

```python
from mcp.server import Server
from mcp.types import TextContent
import uuid

class SessionManager:
    """会话管理器"""
    
    def __init__(self):
        self.sessions = {}
    
    def create_session(self) -> str:
        """创建会话"""
        session_id = str(uuid.uuid4())
        self.sessions[session_id] = {
            "id": session_id,
            "created_at": datetime.now(),
            "data": {}
        }
        return session_id
    
    def get_session(self, session_id: str) -> dict:
        """获取会话"""
        return self.sessions.get(session_id)
    
    def update_session(self, session_id: str, data: dict):
        """更新会话"""
        if session_id in self.sessions:
            self.sessions[session_id]["data"].update(data)

class StatefulServer:
    """有状态 Server"""
    
    def __init__(self):
        self.server = Server("stateful-server")
        self.session_manager = SessionManager()
        self._register_tools()
    
    def _register_tools(self):
        """注册工具"""
        
        @self.server.tool()
        async def create_session() -> list[TextContent]:
            """创建会话"""
            session_id = self.session_manager.create_session()
            return [TextContent(type="text", text=session_id)]
        
        @self.server.tool()
        async def get_session_data(session_id: str) -> list[TextContent]:
            """获取会话数据"""
            session = self.session_manager.get_session(session_id)
            if session:
                return [TextContent(type="text", text=json.dumps(session["data"]))]
            return [TextContent(type="text", text="会话不存在")]
```

## 三、状态持久化

```python
import json

class PersistentStateManager:
    """持久化状态管理"""
    
    def __init__(self, storage_path: str):
        self.storage_path = storage_path
        self.state = self._load()
    
    def _load(self) -> dict:
        """加载状态"""
        try:
            with open(self.storage_path, "r") as f:
                return json.load(f)
        except FileNotFoundError:
            return {}
    
    def _save(self):
        """保存状态"""
        with open(self.storage_path, "w") as f:
            json.dump(self.state, f)
    
    def get(self, key: str) -> any:
        """获取状态"""
        return self.state.get(key)
    
    def set(self, key: str, value: any):
        """设置状态"""
        self.state[key] = value
        self._save()
```

## 常见误区

```
误区 1：有状态 Server 不能水平扩展
  把状态存在外部存储（Redis、数据库），Server 本身仍然无状态。
  这样可以水平扩展，每个实例共享同一份状态。

误区 2：会话状态越多越好
  只存储必要的上下文信息。过多的会话状态会占用内存，
  也会让调试变得困难——你不知道某个状态是哪次调用设置的。

误区 3：连接池不需要关闭
  Server shutdown 时必须优雅关闭所有连接池。
  不关闭连接池会导致数据库连接泄漏，最终耗尽连接数。

误区 4：状态持久化就是写文件
  文件持久化适合配置和小型数据，不适合高频更新的状态。
  高频状态用 Redis 或数据库，低频状态用文件。
```

---

## 工程建议

```
1. 连接池作为 Server 级别共享资源
  在 Server 初始化时创建连接池，所有 Tool 共享。
  不要在每个 Tool 内部创建和销毁连接。

2. 会话用 TTL 自动过期
  设置会话过期时间（如 30 分钟），过期自动清理。
  防止长时间不活跃的会话占用内存。

3. 事务状态用 session_id 关联
  多步事务通过 session_id 关联同一数据库连接。
  在 Tool 的参数中暴露 session_id，让 AI 能跨调用维持事务。

4. Server shutdown 要清理资源
  注册 shutdown hook，按顺序：停止接收新请求 → 等待进行中的请求完成 → 关闭连接池。
  不要直接 kill 进程，那会导致事务未提交、连接未释放。
```

## 参考答案

### 练习一：会话管理器

**思路**：会话管理器的核心是维护 session_id 到会话数据的映射，支持创建、获取、更新、删除操作。关键是加入 TTL 自动过期机制，防止长时间不活跃的会话占用内存。会话数据应存储在内存中，高频访问不需要持久化。

**答案**：
```python
import uuid
import time
import json
from datetime import datetime
from mcp.server import Server
from mcp.types import TextContent

class SessionManager:
    def __init__(self, ttl_seconds: int = 1800):
        self.sessions: dict[str, dict] = {}
        self.ttl_seconds = ttl_seconds

    def create_session(self, initial_data: dict = None) -> str:
        session_id = str(uuid.uuid4())
        self.sessions[session_id] = {
            "id": session_id,
            "created_at": time.time(),
            "last_accessed": time.time(),
            "data": initial_data or {}
        }
        return session_id

    def get_session(self, session_id: str) -> dict | None:
        session = self.sessions.get(session_id)
        if not session:
            return None
        if self._is_expired(session):
            self.delete_session(session_id)
            return None
        session["last_accessed"] = time.time()
        return session

    def update_session(self, session_id: str, data: dict) -> bool:
        session = self.get_session(session_id)
        if not session:
            return False
        session["data"].update(data)
        session["last_accessed"] = time.time()
        return True

    def delete_session(self, session_id: str):
        self.sessions.pop(session_id, None)

    def cleanup_expired(self) -> int:
        expired = [sid for sid, s in self.sessions.items() if self._is_expired(s)]
        for sid in expired:
            self.sessions.pop(sid)
        return len(expired)

    def _is_expired(self, session: dict) -> bool:
        return time.time() - session["last_accessed"] > self.ttl_seconds

session_mgr = SessionManager()
server = Server("session-server")

@server.tool()
async def create_session(initial_data: str = "{}") -> list[TextContent]:
    data = json.loads(initial_data)
    session_id = session_mgr.create_session(data)
    return [TextContent(type="text", text=json.dumps({"session_id": session_id}))]

@server.tool()
async def get_session_data(session_id: str) -> list[TextContent]:
    session = session_mgr.get_session(session_id)
    if not session:
        return [TextContent(type="text", text=json.dumps({"error": "会话不存在或已过期"}))]
    return [TextContent(type="text", text=json.dumps(session["data"], ensure_ascii=False))]

@server.tool()
async def update_session_data(session_id: str, data: str) -> list[TextContent]:
    update = json.loads(data)
    success = session_mgr.update_session(session_id, update)
    if not success:
        return [TextContent(type="text", text=json.dumps({"error": "更新失败，会话不存在或已过期"}))]
    return [TextContent(type="text", text=json.dumps({"status": "updated"}))]
```

**要点**：
- 会话必须有 TTL 自动过期机制，否则长时间不活跃的会话会持续占用内存直到 OOM
- 每次访问会话时更新 `last_accessed` 时间，实现滑动过期（活跃的会话不会过期）
- 常见错误：只在创建时设置过期时间，不更新访问时间，导致用户正在使用的会话突然过期

### 练习二：状态持久化

**思路**：状态持久化的核心是在内存中维护状态的同时写入外部存储。关键设计决策：写入频率（每次 set 都写 vs 定期批量写）、存储介质（文件 vs Redis vs 数据库）、并发安全（文件锁）。低频状态用文件，高频状态用 Redis。

**答案**：
```python
import json
import os
import threading
from mcp.server import Server
from mcp.types import TextContent

class PersistentStateManager:
    def __init__(self, storage_path: str, auto_save: bool = True):
        self.storage_path = storage_path
        self.auto_save = auto_save
        self.state: dict = self._load()
        self._lock = threading.Lock()

    def _load(self) -> dict:
        try:
            with open(self.storage_path, "r", encoding="utf-8") as f:
                return json.load(f)
        except (FileNotFoundError, json.JSONDecodeError):
            return {}

    def _save(self):
        tmp_path = self.storage_path + ".tmp"
        with open(tmp_path, "w", encoding="utf-8") as f:
            json.dump(self.state, f, ensure_ascii=False, indent=2)
        os.replace(tmp_path, self.storage_path)

    def get(self, key: str, default=None):
        with self._lock:
            return self.state.get(key, default)

    def set(self, key: str, value):
        with self._lock:
            self.state[key] = value
            if self.auto_save:
                self._save()

    def delete(self, key: str) -> bool:
        with self._lock:
            if key in self.state:
                del self.state[key]
                if self.auto_save:
                    self._save()
                return True
            return False

    def flush(self):
        with self._lock:
            self._save()

    def keys(self) -> list[str]:
        with self._lock:
            return list(self.state.keys())

state_mgr = PersistentStateManager("./server_state.json")
server = Server("stateful-server")

@server.tool()
async def get_state(key: str) -> list[TextContent]:
    value = state_mgr.get(key)
    if value is None:
        return [TextContent(type="text", text=json.dumps({"error": f"键 {key} 不存在"}))]
    return [TextContent(type="text", text=json.dumps({"key": key, "value": value}, ensure_ascii=False))]

@server.tool()
async def set_state(key: str, value: str) -> list[TextContent]:
    state_mgr.set(key, json.loads(value))
    return [TextContent(type="text", text=json.dumps({"status": "saved", "key": key}))]
```

**要点**：
- 写文件时先写临时文件再 `os.replace`，避免写入中途崩溃导致状态文件损坏
- 必须加锁保护读写操作，否则并发访问会导致数据竞争和状态不一致
- 常见错误：每次 `set` 都直接 `open` 写文件，在高频更新场景下 IO 开销巨大；高频状态应批量写或用 Redis

### 练习三：连接池管理

**思路**：连接池作为 Server 级别共享资源，所有 Tool 共用同一个池。关键是：延迟初始化（首次使用时创建）、按连接字符串缓存（不同数据库用不同池）、Server shutdown 时优雅关闭所有连接池。不在每个 Tool 内部创建和销毁连接。

**答案**：
```python
import asyncio
import json
from mcp.server import Server
from mcp.types import TextContent

class ConnectionPoolManager:
    def __init__(self, max_size: int = 20, min_size: int = 5):
        self.pools: dict[str, object] = {}
        self.max_size = max_size
        self.min_size = min_size
        self._initialized = set()

    async def get_pool(self, connection_string: str):
        if connection_string not in self.pools:
            self.pools[connection_string] = await self._create_pool(connection_string)
        return self.pools[connection_string]

    async def _create_pool(self, connection_string: str):
        class MockPool:
            def __init__(self, conn_str, max_size, min_size):
                self.conn_str = conn_str
                self.max_size = max_size
                self.min_size = min_size
                self._connections = []

            async def acquire(self):
                return {"connection": id(self), "pool": self.conn_str}

            async def release(self, conn):
                pass

            async def close(self):
                self._connections.clear()

            def size(self):
                return self.min_size

        pool = MockPool(connection_string, self.max_size, self.min_size)
        return pool

    async def close_all(self):
        for conn_str, pool in self.pools.items():
            try:
                await pool.close()
            except Exception:
                pass
        self.pools.clear()

    def list_pools(self) -> list[dict]:
        return [{"connection_string": k, "size": v.size()} for k, v in self.pools.items()]

pool_manager = ConnectionPoolManager()
server = Server("pool-server")

@server.tool()
async def execute_query(connection_string: str, sql: str) -> list[TextContent]:
    pool = await pool_manager.get_pool(connection_string)
    conn = await pool.acquire()
    try:
        result = {"sql": sql, "rows": [{"id": 1, "name": "test"}]}
        return [TextContent(type="text", text=json.dumps(result, ensure_ascii=False))]
    finally:
        await pool.release(conn)

@server.tool()
async def list_connection_pools() -> list[TextContent]:
    pools = pool_manager.list_pools()
    return [TextContent(type="text", text=json.dumps(pools, ensure_ascii=False))]

async def shutdown():
    await pool_manager.close_all()
```

**要点**：
- 连接池必须在 Server 级别初始化，所有 Tool 共享；在每个 Tool 内部创建连接池会导致连接数失控
- `finally` 块中释放连接是必须的，否则异常时连接不会归还池中，最终耗尽连接数
- 常见错误：Server shutdown 时不关闭连接池，导致数据库连接泄漏；应在 shutdown hook 中按顺序：停止接收新请求 → 等待进行中请求完成 → 关闭连接池
