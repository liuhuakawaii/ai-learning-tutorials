## 一、评估方案设计

```
微调模型评估三维度：

┌─────────────────────────────────────────────────────────────┐
│  1. 通用能力评估（不能变差）                                  │
│     - MMLU：知识广度                                         │
│     - HumanEval：代码能力                                    │
│     - GSM8K：数学推理                                        │
│     → 与基座模型对比，下降幅度 < 3% 为合格                    │
│                                                              │
│  2. 领域任务评估（必须提升）                                  │
│     - 自建测试集（50-200 条）                                │
│     - 准确率 / F1 / BLEU / ROUGE                            │
│     - 人工评估（盲评打分 1-5）                                │
│     → 比基座模型提升 > 10% 为合格                            │
│                                                              │
│  3. 工程指标评估（生产要求）                                  │
│     - 推理延迟（P50 / P95 / P99）                            │
│     - 吞吐量（tokens/s）                                    │
│     - 显存占用                                               │
│     - 首 token 延迟（TTFT）                                  │
│     → 满足业务 SLA 要求                                      │
└─────────────────────────────────────────────────────────────┘
```

## 二、自动化评估脚本

```python
import json
import time
import torch
from pathlib import Path
from transformers import AutoTokenizer, AutoModelForCausalLM
from peft import PeftModel

class ModelEvaluator:
    """微调模型评估器"""

    def __init__(self, base_model: str, lora_path: str):
        self.tokenizer = AutoTokenizer.from_pretrained(base_model, trust_remote_code=True)
        base = AutoModelForCausalLM.from_pretrained(
            base_model, torch_dtype=torch.bfloat16, device_map="auto"
        )
        self.model = PeftModel.from_pretrained(base, lora_path)
        self.model.eval()

    def evaluate_domain(self, test_data: list[dict]) -> dict:
        """领域任务评估"""
        correct = 0
        total = len(test_data)
        results = []

        for item in test_data:
            prompt = item["prompt"]
            expected = item["expected"]

            inputs = self.tokenizer(prompt, return_tensors="pt").to(self.model.device)
            with torch.no_grad():
                outputs = self.model.generate(**inputs, max_new_tokens=256, temperature=0)

            predicted = self.tokenizer.decode(outputs[0][inputs.input_ids.shape[1]:], skip_special_tokens=True)
            is_correct = self._compare_answer(predicted, expected)

            if is_correct:
                correct += 1
            results.append({
                "prompt": prompt[:100],
                "expected": expected[:100],
                "predicted": predicted[:100],
                "correct": is_correct,
            })

        return {
            "accuracy": correct / total,
            "correct": correct,
            "total": total,
            "details": results[:10],  # 只保留前 10 条详情
        }

    def measure_performance(self, prompt: str = "你好，请介绍一下自己。", runs: int = 5) -> dict:
        """性能基准测试"""
        latencies = []
        tokens_per_sec = []

        for _ in range(runs):
            inputs = self.tokenizer(prompt, return_tensors="pt").to(self.model.device)
            start = time.time()
            with torch.no_grad():
                outputs = self.model.generate(**inputs, max_new_tokens=256)
            elapsed = time.time() - start

            new_tokens = outputs.shape[1] - inputs.input_ids.shape[1]
            latencies.append(elapsed)
            tokens_per_sec.append(new_tokens / elapsed)

        latencies.sort()
        return {
            "latency_p50": latencies[len(latencies) // 2],
            "latency_p95": latencies[int(len(latencies) * 0.95)],
            "tokens_per_sec": sum(tokens_per_sec) / len(tokens_per_sec),
            "gpu_memory_gb": torch.cuda.max_memory_allocated() / 1e9,
        }

    def _compare_answer(self, predicted: str, expected: str) -> bool:
        """简单答案匹配（可根据任务自定义）"""
        return expected.strip().lower() in predicted.strip().lower()
```

## 三、评估报告生成

```python
from datetime import datetime

def generate_report(
    model_name: str,
    base_model: str,
    domain_results: dict,
    perf_results: dict,
    baseline_domain: dict | None = None,
) -> str:
    """生成 Markdown 评估报告"""

    improvement = ""
    if baseline_domain:
        diff = domain_results["accuracy"] - baseline_domain["accuracy"]
        improvement = f"- 领域提升：**{diff:+.2%}**（基座 {baseline_domain['accuracy']:.2%} → 微调 {domain_results['accuracy']:.2%}）"

    report = f"""# 微调模型评估报告

> 生成时间：{datetime.now().strftime('%Y-%m-%d %H:%M')}
> 模型：{model_name}
> 基座：{base_model}

## 领域任务评估

- 测试集大小：{domain_results['total']} 条
- 准确率：**{domain_results['accuracy']:.2%}**
{improvement}

### 样例（前 5 条）

| # | 预期 | 预测 | 正确 |
|---|------|------|------|
"""

    for i, d in enumerate(domain_results["details"][:5], 1):
        status = "✅" if d["correct"] else "❌"
        report += f"| {i} | {d['expected'][:30]} | {d['predicted'][:30]} | {status} |\n"

    report += f"""
## 性能指标

| 指标 | 值 |
|------|-----|
| P50 延迟 | {perf_results['latency_p50']:.3f}s |
| P95 延迟 | {perf_results['latency_p95']:.3f}s |
| 吞吐量 | {perf_results['tokens_per_sec']:.1f} tokens/s |
| 显存占用 | {perf_results['gpu_memory_gb']:.1f} GB |

## 结论

- 领域任务准确率 {'✅ 达标' if domain_results['accuracy'] > 0.7 else '❌ 需改进'}
- 推理性能 {'✅ 满足要求' if perf_results['latency_p95'] < 2.0 else '⚠️ 需优化'}
"""

    return report
```

## 四、模型合并与量化

```python
from peft import PeftModel
from transformers import AutoModelForCausalLM, AutoTokenizer

def merge_lora(base_model: str, lora_path: str, output_path: str):
    """合并 LoRA 权重到基座模型"""
    base = AutoModelForCausalLM.from_pretrained(base_model, torch_dtype="auto")
    model = PeftModel.from_pretrained(base, lora_path)
    merged = model.merge_and_unload()  # 合并 LoRA 到基座

    merged.save_pretrained(output_path)
    tokenizer = AutoTokenizer.from_pretrained(base_model)
    tokenizer.save_pretrained(output_path)
    print(f"Merged model saved to {output_path}")


def quantize_gguf(model_path: str, output_path: str, quant: str = "Q4_K_M"):
    """导出 GGUF 格式（用于 llama.cpp / Ollama）"""
    # 使用 llama.cpp 的 convert 脚本
    import subprocess
    subprocess.run([
        "python", "convert_hf_to_gguf.py",
        model_path,
        "--outfile", f"{output_path}/model.gguf",
        "--outtype", quant,
    ], check=True)
    print(f"GGUF model saved to {output_path}/model.gguf ({quant})")
```

## 六、完整流程串联

```python
async def full_pipeline(base_model: str, lora_path: str, test_data_path: str):
    """完整的评估-合并-部署流程"""

    # 1. 评估
    print("Step 1: Evaluating...")
    evaluator = ModelEvaluator(base_model, lora_path)

    with open(test_data_path) as f:
        test_data = json.load(f)

    domain_results = evaluator.evaluate_domain(test_data)
    perf_results = evaluator.measure_performance()

    # 2. 生成报告
    print("Step 2: Generating report...")
    report = generate_report(
        model_name="my-finetuned-model",
        base_model=base_model,
        domain_results=domain_results,
        perf_results=perf_results,
    )
    Path("eval-report.md").write_text(report)

    # 3. 判断是否达标
    if domain_results["accuracy"] < 0.7:
        print("❌ 领域准确率不达标，需要继续训练或调整数据")
        return

    # 4. 合并 LoRA
    print("Step 3: Merging LoRA...")
    merge_lora(base_model, lora_path, "./merged-model")

    # 5. 导出 GGUF（可选）
    print("Step 4: Exporting GGUF...")
    quantize_gguf("./merged-model", "./gguf-model", quant="Q4_K_M")

    # 6. 部署
    print("Step 5: Ready for deployment!")
    print("  - vLLM: vllm serve ./merged-model")
    print("  - Ollama: ollama create my-model -f Modelfile && ollama run my-model")
    print("  - Docker: docker build -t my-model . && docker run -p 8000:8000 my-model")
```

## 常见误区

1. **评估报告只有分数没有分析**：报告里列出准确率 85%、延迟 0.5s，但没有分析这意味着什么、是否达标、哪些方面需要改进。数字本身没有意义，分析才有价值。

2. **合并和量化一步到位**：应该先合并测试，再量化测试，每步都验证效果。如果合并后就有问题，量化后问题可能更严重，排查更困难。

3. **部署后不做线上验证**：模型部署到生产环境后，还需要用真实请求验证效果。离线评估和线上表现可能有差距（如延迟、并发处理能力）。

4. **忽略课程项目的验证脚本**：final-project 的 check.py 验证了项目结构和关键文件的完整性，应该先运行验证确保项目基础正确。

## 工程建议

1. **评估报告要面向决策者**：报告的读者可能是你的老板或产品经理。用结论驱动的写法——先说结论（达标/不达标），再给数据支撑，最后说下一步建议。

2. **全流程自动化脚本**：从评估到合并到量化到部署，写成一个完整的自动化脚本。一键运行就能完成从 checkpoint 到可部署模型的全流程。

3. **部署方案要匹配场景**：内部工具用 Ollama 最简单，面向用户的服务用 vLLM + Docker 最可靠，边缘设备用 GGUF + llama.cpp 最轻量。

4. **建立持续评估机制**：模型上线后定期用新数据评估效果，发现退化及时迭代。一次微调不是终点，持续优化才是。
