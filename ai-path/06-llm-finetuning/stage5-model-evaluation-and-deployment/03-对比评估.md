# 03 对比评估——微调模型 vs 基础模型 vs API 模型的多维度对比

> 对比评估帮你做出合理的 build vs buy 决策。

## 场景引入

你的微调模型效果不错，但老板问了一个关键问题："我们自己微调的模型和直接用 GPT-4 API 相比，到底哪个更划算？"你发现这不只是质量对比的问题——还要考虑延迟、成本、数据隐私、维护难度等多个维度。一个全面的对比评估框架，能帮你做出最有说服力的 build vs buy 决策。

1. 质量

   - 完整性

   - 延迟
   - 吞吐量
   - 并发能力

3. 成本
   - API 费用
   - 部署成本
   - 维护成本

4. 其他
   - 数据隐私
   - 定制化能力
   - 可用性

## 五、决策框架

```python
def make_decision(results: dict, priorities: dict) -> str:
    """做出决策"""
    
    scores = {}
    
    for model_name, model_results in results.items():
        score = 0
        for dimension, weight in priorities.items():
            score += model_results[dimension] * weight
        scores[model_name] = score
    
    return max(scores.items(), key=lambda x: x[1])[0]
```

## 常见误区

1. **只对比质量不对比成本**：微调模型质量比 API 好 5%，但成本高 3 倍，这个提升值得吗？对比评估必须包含成本维度，做综合性价比分析。

2. **用不同 prompt 测试不同模型**：同一个问题用不同的 prompt 表述，模型输出可能差异很大。对比评估时所有模型必须用完全相同的输入。

3. **忽略延迟对用户体验的影响**：API 模型可能质量更好，但延迟 3 秒；本地模型延迟 0.5 秒。对于实时交互场景，延迟差异可能比质量差异更重要。

4. **对比评估只做一次就下结论**：模型的效果和成本会随着用量、数据分布、模型版本的变化而变化。对比评估应该定期重做。

---

## 工程建议

1. **决策矩阵要量化**：不要用"好/一般/差"这种模糊评价，要给每个维度打分（1-10），乘以权重后算总分。量化决策比主观判断更可靠。

2. **考虑隐性成本**：本地部署的隐性成本包括运维人力、GPU 闲置成本、故障恢复成本。API 的隐性成本包括供应商锁定、数据隐私风险。全面计算 TCO。

3. **先小规模试点再全面切换**：不要一次性把所有流量切到新模型。先用 10% 的流量做 A/B 测试，确认效果和稳定性后再逐步扩大。

4. **建立退出机制**：选择本地方案时要考虑退出成本——如果效果不达标或成本超预期，能否快速切回 API？预留回退路径。

## 参考答案

### 练习一：对比题

**思路**：在相同条件下（相同输入、相同硬件、相同评估指标）对比多个模型的质量、延迟和成本。

**答案**：
```python
import time
import json
from openai import OpenAI

def measure_latency(client, model: str, messages: list, runs: int = 10) -> dict:
    """测量模型延迟"""
    latencies = []
    
    for _ in range(runs):
        start = time.time()
        response = client.chat.completions.create(
            model=model,
            messages=messages
        )
        latency = time.time() - start
        latencies.append(latency)
    
    return {
        "avg": sum(latencies) / len(latencies),
        "min": min(latencies),
        "max": max(latencies),
        "p95": sorted(latencies)[int(len(latencies) * 0.95)]
    }


def compare_models(eval_dataset: list[dict]) -> dict:
    """对比多个模型的表现"""
    
    models = {
        "gpt-4o": {"client": OpenAI(), "model": "gpt-4o"},
        "local_finetuned": {"client": OpenAI(base_url="http://localhost:8000/v1", api_key="none"), "model": "local"}
    }
    
    results = {}
    
    for model_name, config in models.items():
        correct = 0
        total_latency = []
        total_tokens = 0
        
        for sample in eval_dataset:
            messages = [{"role": "user", "content": sample["question"]}]
            
            start = time.time()
            response = config["client"].chat.completions.create(
                model=config["model"],
                messages=messages
            )
            latency = time.time() - start
            
            answer = response.choices[0].message.content
            if answer.strip() == sample["reference"].strip():
                correct += 1
            
            total_latency.append(latency)
            total_tokens += response.usage.total_tokens if response.usage else 0
        
        results[model_name] = {
            "accuracy": correct / len(eval_dataset),
            "avg_latency": sum(total_latency) / len(total_latency),
            "total_tokens": total_tokens,
            "cost": total_tokens * 0.0000025  # 简化计算
        }
    
    # 输出对比表
    print(f"{'模型':<20} {'准确率':>10} {'平均延迟':>10} {'总Token':>10} {'成本':>10}")
    print("-" * 65)
    for name, r in results.items():
        print(f"{name:<20} {r['accuracy']:>9.2%} {r['avg_latency']:>9.2f}s {r['total_tokens']:>10} ${r['cost']:>8.4f}")
    
    return results


if __name__ == "__main__":
    eval_data = [
        {"question": "什么是高血压？", "reference": "高血压是血压持续升高的一种慢性疾病"},
        {"question": "糖尿病的常见症状？", "reference": "多饮、多尿、多食、体重减轻"},
    ]
    results = compare_models(eval_data)
```

**要点**：
- 所有模型必须用完全相同的输入和评估集，不能用不同 prompt
- 延迟测量要取多次平均值，排除首次冷启动的影响
- 成本计算要包含所有 token（输入 + 输出），不能只算输出
- 常见错误：只对比质量不对比延迟和成本，得出片面结论

### 练习二：分析题

**思路**：从多维度对比结果中识别各模型的优劣势，找到最适合业务场景的选择。

**答案**：
```python
def analyze_comparison(results: dict, priorities: dict = None) -> dict:
    """分析对比结果，找出最优选择"""
    
    if priorities is None:
        priorities = {"accuracy": 0.4, "latency": 0.3, "cost": 0.3}
    
    # 归一化各维度分数（0-1）
    dimensions = ["accuracy", "avg_latency", "cost"]
    normalized = {}
    
    for dim in dimensions:
        values = [r[dim] for r in results.values()]
        min_val, max_val = min(values), max(values)
        range_val = max_val - min_val if max_val != min_val else 1
        
        for model_name, r in results.items():
            if model_name not in normalized:
                normalized[model_name] = {}
            
            if dim == "avg_latency" or dim == "cost":
                # 延迟和成本越低越好，反转归一化
                normalized[model_name][dim] = 1 - (r[dim] - min_val) / range_val
            else:
                normalized[model_name][dim] = (r[dim] - min_val) / range_val
    
    # 计算加权总分
    scores = {}
    for model_name, norms in normalized.items():
        score = sum(norms[dim] * priorities[dim] for dim in dimensions)
        scores[model_name] = score
    
    # 分析
    analysis = {
        "scores": scores,
        "best_overall": max(scores.items(), key=lambda x: x[1])[0],
        "best_accuracy": max(results.items(), key=lambda x: x[1]["accuracy"])[0],
        "best_latency": min(results.items(), key=lambda x: x[1]["avg_latency"])[0],
        "best_cost": min(results.items(), key=lambda x: x[1]["cost"])[0],
    }
    
    print("=== 对比分析结果 ===\n")
    print(f"综合最优: {analysis['best_overall']} (加权得分: {scores[analysis['best_overall']]:.3f})")
    print(f"质量最优: {analysis['best_accuracy']} (准确率: {results[analysis['best_accuracy']]['accuracy']:.2%})")
    print(f"速度最优: {analysis['best_latency']} (延迟: {results[analysis['best_latency']]['avg_latency']:.2f}s)")
    print(f"成本最优: {analysis['best_cost']} (成本: ${results[analysis['best_cost']]['cost']:.4f})")
    
    print(f"\n各模型加权得分:")
    for name, score in sorted(scores.items(), key=lambda x: -x[1]):
        print(f"  {name}: {score:.3f}")
    
    return analysis


if __name__ == "__main__":
    sample_results = {
        "gpt-4o": {"accuracy": 0.92, "avg_latency": 1.2, "cost": 0.05},
        "local_7b": {"accuracy": 0.78, "avg_latency": 0.3, "cost": 0.005},
        "claude-sonnet": {"accuracy": 0.90, "avg_latency": 1.5, "cost": 0.06}
    }
    analysis = analyze_comparison(sample_results)
```

**要点**：
- 归一化是关键步骤，不同维度的量纲不同（准确率是百分比，延迟是秒，成本是美元），必须归一化后才能加权
- 延迟和成本是越低越好，归一化时需要反转
- 权重设置要反映业务优先级，不同场景的权重差异很大
- 常见错误：不做归一化直接加权，导致数值大的维度主导结果

### 练习三：决策题

**思路**：基于对比分析结果，结合业务约束（数据隐私、预算、上线时间），用决策矩阵做出 build vs buy 决策。

**答案**：
```python
def make_build_vs_buy_decision(
    comparison_results: dict,
    business_constraints: dict
) -> dict:
    """做出 build vs buy 决策"""
    
    # 定义决策维度和权重
    dimensions = {
        "quality": {"weight": 0.25, "higher_is_better": True},
        "latency": {"weight": 0.20, "higher_is_better": False},
        "cost": {"weight": 0.20, "higher_is_better": False},
        "data_privacy": {"weight": 0.15, "higher_is_better": True},
        "customization": {"weight": 0.10, "higher_is_better": True},
        "time_to_market": {"weight": 0.10, "higher_is_better": True}
    }
    
    # 模型评分（1-10）
    model_scores = {
        "api_gpt4o": {
            "quality": 9, "latency": 6, "cost": 5,
            "data_privacy": 3, "customization": 3, "time_to_market": 10
        },
        "local_finetuned": {
            "quality": 7, "latency": 9, "cost": 7,
            "data_privacy": 10, "customization": 10, "time_to_market": 4
        }
    }
    
    # 计算加权总分
    final_scores = {}
    for model, scores in model_scores.items():
        weighted_score = sum(
            scores[dim] * dimensions[dim]["weight"]
            for dim in dimensions
        )
        final_scores[model] = weighted_score
    
    # 应用业务约束
    if business_constraints.get("data_privacy_required"):
        # 如果数据隐私是硬性要求，API 方案直接排除
        if final_scores.get("api_gpt4o", 0) > final_scores.get("local_finetuned", 0):
            print("⚠️ 数据隐私是硬性要求，即使 API 分数更高也必须选择本地部署")
    
    if business_constraints.get("budget_limited"):
        print("⚠️ 预算有限，建议先用 API 验证业务可行性")
    
    if business_constraints.get("urgent_timeline"):
        print("⚠️ 上线时间紧迫，建议先用 API 快速上线，后续再考虑本地部署")
    
    # 最终决策
    best_option = max(final_scores.items(), key=lambda x: x[1])
    
    decision = {
        "recommendation": best_option[0],
        "scores": final_scores,
        "decision_matrix": {}
    }
    
    # 输出决策矩阵
    print("\n=== 决策矩阵 ===\n")
    print(f"{'维度':<15} {'权重':>6} {'API GPT-4o':>12} {'本地微调':>12}")
    print("-" * 50)
    for dim, config in dimensions.items():
        api_score = model_scores["api_gpt4o"][dim]
        local_score = model_scores["local_finetuned"][dim]
        print(f"{dim:<15} {config['weight']:>5.0%} {api_score:>12} {local_score:>12}")
        decision["decision_matrix"][dim] = {"api": api_score, "local": local_score}
    
    print(f"\n{'加权总分':<15} {'':>6} {final_scores['api_gpt4o']:>12.2f} {final_scores['local_finetuned']:>12.2f}")
    print(f"\n建议方案: {best_option[0]}")
    
    return decision


if __name__ == "__main__":
    constraints = {
        "data_privacy_required": True,
        "budget_limited": False,
        "urgent_timeline": False
    }
    decision = make_build_vs_buy_decision({}, constraints)
```

**要点**：
- 决策矩阵必须量化，不能用"好/一般/差"这种模糊评价
- 业务约束（数据隐私、预算、时间）可能推翻纯分数排名
- 不同场景的权重差异很大：实时交互场景延迟权重高，合规场景隐私权重高
- 常见错误：只看质量分数做决策，忽略延迟、成本、隐私等非功能性需求
