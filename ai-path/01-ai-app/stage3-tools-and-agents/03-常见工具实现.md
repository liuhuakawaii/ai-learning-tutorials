## 一、搜索工具

### 1.1 知识库搜索

```typescript
// lib/tools/search.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'
import { retrieve } from '../retrieval'

const SearchSchema = z.object({
  query: z.string().min(1).describe('搜索查询，包含关键词'),
  max_results: z.number().int().min(1).max(20).default(5).describe('最大返回结果数'),
  document_id: z.string().optional().describe('限定在某个文档中搜索'),
})

toolRegistry.register({
  name: 'search_knowledge_base',
  description: '在知识库中搜索与查询相关的文档片段。当用户询问基于已上传文档的问题时使用。',
  schema: SearchSchema,
  handler: async (args) => {
    const { query, max_results, document_id } = args

    const filter = document_id ? { docId: document_id } : undefined
    const results = await retrieve(query, max_results, filter)

    return {
      success: true,
      data: {
        results: results.map(r => ({
          content: r.content,
          source: r.metadata.source,
          similarity: r.similarity,
        })),
        total: results.length,
      },
    }
  },
})
```

### 1.2 网页搜索（调用外部 API）

```typescript
// lib/tools/web-search.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'

const WebSearchSchema = z.object({
  query: z.string().min(1).describe('搜索查询'),
  num_results: z.number().int().min(1).max(10).default(3).describe('返回结果数'),
})

toolRegistry.register({
  name: 'web_search',
  description: '在互联网上搜索信息。当需要获取实时信息或模型不知道的知识时使用。',
  schema: WebSearchSchema,
  handler: async (args) => {
    const { query, num_results } = args

    try {
      // 使用搜索 API（如 SerpAPI、Bing API 等）
      const response = await fetch(
        `https://api.search.com/search?q=${encodeURIComponent(query)}&limit=${num_results}`,
        {
          headers: {
            'Authorization': `Bearer ${process.env.SEARCH_API_KEY}`,
          },
        }
      )

      if (!response.ok) {
        throw new Error('搜索服务不可用')
      }

      const data = await response.json()

      return {
        success: true,
        data: {
          results: data.results.map((r: any) => ({
            title: r.title,
            url: r.url,
            snippet: r.snippet,
          })),
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `搜索失败: ${error}`,
      }
    }
  },
})
```

## 二、数据库查询工具

### 2.1 查询用户数据

```typescript
// lib/tools/database.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'
import { pool } from '../db'

const QueryUserSchema = z.object({
  user_id: z.string().describe('用户 ID'),
  fields: z.array(z.enum(['name', 'email', 'orders', 'preferences']))
    .describe('要查询的字段'),
})

toolRegistry.register({
  name: 'query_user_info',
  description: '查询用户信息。当需要获取用户的订单、偏好等信息时使用。',
  schema: QueryUserSchema,
  handler: async (args) => {
    const { user_id, fields } = args

    try {
      // 白名单校验字段名，防止 SQL 注入
      const allowedFields = ['id', 'name', 'email', 'phone', 'address', 'created_at']
      const validFields = fields.filter((f: string) => allowedFields.includes(f))
      if (validFields.length === 0) {
        return { success: false, error: '没有有效的查询字段' }
      }
      const selectFields = validFields.join(', ')
      const result = await pool.query(
        `SELECT ${selectFields} FROM users WHERE id = $1`,
        [user_id]
      )

      if (result.rows.length === 0) {
        return {
          success: false,
          error: '用户不存在',
        }
      }

      return {
        success: true,
        data: result.rows[0],
      }
    } catch (error) {
      return {
        success: false,
        error: `查询失败: ${error}`,
      }
    }
  },
})
```

## 三、文件读写工具

### 3.1 读取文件

```typescript
// lib/tools/file.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'
import fs from 'fs/promises'
import path from 'path'

const ReadFileSchema = z.object({
  path: z.string().describe('文件路径（相对于项目根目录）'),
  encoding: z.enum(['utf-8', 'base64']).default('utf-8'),
})

toolRegistry.register({
  name: 'read_file',
  description: '读取文件内容。当需要查看配置文件、日志文件等时使用。',
  schema: ReadFileSchema,
  handler: async (args) => {
    const { path: filePath, encoding } = args

    // 安全检查：限制只能访问特定目录
    const allowedDirs = ['uploads', 'data', 'logs']
    const normalizedPath = path.normalize(filePath)

    if (!allowedDirs.some(dir => normalizedPath.startsWith(dir))) {
      return {
        success: false,
        error: '不允许访问该路径',
      }
    }

    try {
      const content = await fs.readFile(normalizedPath, encoding)
      const stats = await fs.stat(normalizedPath)

      return {
        success: true,
        data: {
          content,
          size: stats.size,
          modified: stats.mtime,
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `读取文件失败: ${error}`,
      }
    }
  },
})
```

```typescript
const WriteFileSchema = z.object({

  content: z.string().describe('文件内容'),
  create_dirs: z.boolean().default(true).describe('是否自动创建目录'),

  name: 'write_file',

    const { path: filePath, content, create_dirs } = args

    // 安全检查
    const allowedDirs = ['uploads', 'output', 'reports']

    if (!allowedDirs.some(dir => normalizedPath.startsWith(dir))) {
      return {
        success: false,

      if (create_dirs) {
        await fs.mkdir(path.dirname(normalizedPath), { recursive: true })
      }

      await fs.writeFile(normalizedPath, content, 'utf-8')

      return {
        success: true,
        data: {
          path: normalizedPath,
          size: content.length,
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `写入文件失败: ${error}`,
      }
    }
  },
})
```

```typescript
// 查询笔记
const QueryNotesSchema = z.object({
  query: z.string().optional().describe('搜索关键词'),
  tags: z.array(z.string()).optional().describe('按标签过滤'),
  limit: z.number().int().min(1).max(50).default(10),
})

toolRegistry.register({
  name: 'query_notes',
  description: '查询笔记。当用户询问之前保存的笔记时使用。',
  schema: QueryNotesSchema,
  handler: async (args) => {
    const { query, tags, limit } = args

    try {
      let sql = 'SELECT * FROM notes WHERE 1=1'
      const params: any[] = []
      let paramIndex = 1

      if (query) {
        sql += ` AND (title ILIKE $${paramIndex} OR content ILIKE $${paramIndex})`
        params.push(`%${query}%`)
        paramIndex++
      }

      if (tags && tags.length > 0) {
        sql += ` AND tags ?| $${paramIndex}`
        params.push(tags)
        paramIndex++
      }

      sql += ` ORDER BY created_at DESC LIMIT $${paramIndex}`
      params.push(limit)

      const result = await pool.query(sql, params)

      return {
        success: true,
        data: {
          notes: result.rows,
          total: result.rows.length,
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `查询笔记失败: ${error}`,
      }
    }
  },
})
```

### 5.2 实现权限检查

```typescript
function checkPermission(
  userId: string,
  toolName: string,
  args: any
): { allowed: boolean; reason?: string } {
  // 读取权限
  if (toolName === 'read_file') {
    // 只能读取自己的文件
    if (!args.path.startsWith(`uploads/${userId}/`)) {
      return { allowed: false, reason: '只能读取自己的文件' }
    }
  }

  // 写入权限
  if (toolName === 'write_file') {
    if (!args.path.startsWith(`output/${userId}/`)) {
      return { allowed: false, reason: '只能写入自己的目录' }
    }
  }

  // 数据库权限
  if (toolName === 'query_user_info') {
    if (args.user_id !== userId) {
      return { allowed: false, reason: '只能查询自己的信息' }
    }
  }

  return { allowed: true }
}
```

## 参考答案

### 练习一：实现搜索工具

**思路**：搜索工具的核心是返回带相似度分数的结果，让模型能判断信息的可信度。需要支持按文档 ID 过滤、控制返回数量，并做好空结果和异常的处理。

**答案**：

```typescript
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'

const KnowledgeSearchSchema = z.object({
  query: z.string().min(1).describe('搜索查询，应包含用户问题的关键词'),
  max_results: z.number().int().min(1).max(20).default(5).describe('最大返回结果数'),
  document_id: z.string().optional().describe('限定在某个文档中搜索，不传则搜索全部'),
  min_similarity: z.number().min(0).max(1).default(0.3).describe('最低相似度阈值，低于此值的结果会被过滤'),
})

interface SearchResult {
  content: string
  source: string
  similarity: number
}

// 模拟向量搜索函数
async function searchKnowledgeBase(
  query: string,
  maxResults: number,
  documentId?: string,
  minSimilarity?: number,
): Promise<SearchResult[]> {
  // 实际项目中：将 query 转为向量，在向量数据库中做相似度检索
  // const embedding = await embed(query)
  // const results = await vectorDB.search(embedding, { filter: { docId: documentId } })

  // 这里用模拟数据演示
  const allResults: SearchResult[] = [
    { content: 'AI Agent 是能够自主完成任务的智能系统', source: 'doc-ai-basics.md', similarity: 0.92 },
    { content: '工具调用是 Agent 与外部世界交互的桥梁', source: 'doc-tools.md', similarity: 0.85 },
    { content: '向量数据库用于存储和检索文档嵌入', source: 'doc-vectordb.md', similarity: 0.71 },
  ]

  let filtered = allResults

  // 按文档过滤
  if (documentId) {
    filtered = filtered.filter(r => r.source === documentId)
  }

  // 按相似度过滤
  filtered = filtered.filter(r => r.similarity >= (minSimilarity || 0.3))

  // 按相似度降序排列，取前 N 条
  return filtered
    .sort((a, b) => b.similarity - a.similarity)
    .slice(0, maxResults)
}

toolRegistry.register({
  name: 'search_knowledge_base',
  description: '在知识库中搜索与查询相关的文档片段。当用户询问基于已上传文档的问题时使用。返回结果按相似度降序排列。',
  schema: KnowledgeSearchSchema,
  handler: async (args) => {
    const { query, max_results, document_id, min_similarity } = args

    try {
      const results = await searchKnowledgeBase(query, max_results, document_id, min_similarity)

      if (results.length === 0) {
        return {
          success: true,
          data: { results: [], total: 0, message: '未找到相关文档，请尝试换个关键词' },
        }
      }

      return {
        success: true,
        data: {
          results: results.map(r => ({
            content: r.content,
            source: r.source,
            similarity: Math.round(r.similarity * 100) / 100, // 保留两位小数
          })),
          total: results.length,
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `搜索失败: ${error instanceof Error ? error.message : '未知错误'}`,
      }
    }
  },
})
```

**要点**：
- 相似度分数是搜索结果的核心——没有分数，模型无法判断哪些信息更可靠
- 用 `min_similarity` 阈值过滤低质量结果，避免模型被无关内容误导
- 空结果要返回明确提示（"未找到相关文档"），而不是空数组让模型困惑
- 常见错误：返回 50 条结果不做截断——模型上下文有限，大量结果会挤占生成回答的 token 空间

**要点**：
- 保存和查询是两个独立工具，遵循单一职责原则
- 查询支持关键词 + 标签的组合过滤，提高检索精度
- 标签用数组存储，查询时用 `?|` 操作符做 OR 匹配
- 常见错误：把保存和查询合并成一个 `manage_notes` 工具——模型需要理解 action 参数，增加调用出错的概率

### 练习三：权限设计

**思路**：权限设计的核心原则是"最小权限"——每个工具只允许完成其职责范围内的操作。按操作类型（读/写/调用）分级，高风险操作需要用户确认。

**答案**：

**1. 需要用户确认的操作（高风险）**
- 发送邮件/消息：发送后不可撤回，必须确认收件人和内容
- 删除数据（文件、记录、笔记）：删除后难以恢复
- 支付/转账操作：涉及资金，必须二次确认
- 修改用户权限：影响系统安全

**2. 只读不写的操作（低风险）**
- 搜索知识库：只读操作，不修改任何数据
- 查询用户信息：只读，不修改
- 查询订单状态：只读
- 读取配置文件：只读

**3. 防止越权访问的实现**

```typescript
function checkPermission(
  userId: string,
  toolName: string,
  args: any,
): { allowed: boolean; reason?: string } {
  // 文件访问：只能操作自己目录下的文件
  if (toolName === 'read_file' || toolName === 'write_file') {
    const allowedPrefixes = [`uploads/${userId}/`, `output/${userId}/`]
    const normalizedPath = path.normalize(args.path)
    if (!allowedPrefixes.some(prefix => normalizedPath.startsWith(prefix))) {
      return { allowed: false, reason: '只能访问自己的文件目录' }
    }
  }

  // 数据库查询：只能查询自己的数据
  if (toolName === 'query_user_info' || toolName === 'query_orders') {
    if (args.user_id !== userId) {
      return { allowed: false, reason: '只能查询自己的信息' }
    }
  }

  // 笔记操作：只能操作自己的笔记
  if (toolName === 'query_notes') {
    // 查询时自动注入 userId 过滤条件，模型无法绕过
    args.user_id_filter = userId
  }

  // 高风险操作：需要确认
  const highRiskTools = ['send_email', 'delete_file', 'create_payment']
  if (highRiskTools.includes(toolName)) {
    return { allowed: true, requiresConfirmation: true } as any
  }

  return { allowed: true }
}
```

**要点**：
- 权限检查要在工具执行之前，不能在工具内部做（防止工具实现遗漏检查）
- 文件路径必须用 `path.normalize()` 处理，防止 `../../../etc/passwd` 路径穿越攻击
- 用户 ID 应该从 session/token 中获取，不能让模型传入（防止 prompt injection 冒充他人）
- 常见错误：只在前端做权限校验——模型可能被 prompt injection 操控绕过前端，必须在后端工具层面做校验

## 常见误区

1. **搜索工具不做结果排序**：返回一堆没有相关度评分的结果，模型无法判断哪些信息更可靠。搜索结果必须带相似度或相关度分数，模型才能优先引用高置信度的内容。
2. **数据库查询直接拼接 SQL 字符串**：模型传入的参数可能包含恶意输入。即使参数来自模型而非用户，也要用参数化查询（`$1, $2`）防止 SQL 注入，因为 prompt injection 可能操控模型生成恶意参数。
3. **文件工具不限制访问范围**：如果不做路径白名单校验，模型可能被诱导读取 `/etc/passwd` 或写入系统目录。必须用 `path.normalize()` 处理路径，并限制在允许的目录范围内。
4. **认为工具越多越好**：给系统注册 30 个工具，模型在选择时会"挑花眼"，调用准确率反而下降。按场景动态加载需要的工具集，比全量注册更有效。

---

## 工程建议

1. **搜索工具要做分页和截断**：返回 5 条结果比返回 50 条更高效。模型的上下文有限，大量搜索结果会挤占生成回答的 token 空间。设置 `max_results` 默认值为 5。
2. **数据库工具用白名单校验字段名**：不要让模型传入任意字段名做 SELECT。定义一个 `allowedFields` 数组，只允许查询预定义的字段，从根本上杜绝 SQL 注入。
3. **文件工具限制大小和类型**：读取文件时检查文件大小（如限制 1MB），写入时限制文件类型（如只允许 `.md` 和 `.txt`）。防止模型读取超大日志文件导致内存溢出，或写入可执行文件带来安全风险。
4. **每个工具都要有超时机制**：外部 API 调用可能无限等待。给每个工具设置合理的超时时间（如搜索 10 秒、数据库查询 5 秒），超时后返回明确的错误信息而不是卡死。

本课的核心要点：

1. **搜索工具**：知识库搜索、网页搜索
2. **数据库工具**：查询用户信息、订单状态
3. **文件工具**：读写文件，需要安全检查
4. **笔记工具**：保存和查询笔记
5. **权限控制**：读取、写入、调用都需要权限检查
