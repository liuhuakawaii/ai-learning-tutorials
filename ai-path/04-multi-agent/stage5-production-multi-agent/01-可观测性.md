# 可观测性：多 Agent 系统的追踪、日志、指标设计

> 前置知识：stage1-4 的多 Agent 开发经验
> 预计时长：40 分钟

## 为什么多 Agent 的可观测性比单 Agent 难得多

单 Agent 的请求链路是线性的：用户输入 → LLM 调用 → 工具调用 → 输出。出问题了看日志，一条链就能还原。

多 Agent 的请求链路是图状的：用户输入 → Supervisor 决策 → Agent A → Supervisor 决策 → Agent B → 可能回到 Agent A → Agent C → 输出。一个请求可能经过 5-10 个节点，每个节点都有自己的 LLM 调用、工具调用和状态变更。

没有可观测性，你面对的问题是：
- 用户说"结果不对"，你不知道是哪个 Agent 出了问题
- 你不知道 Supervisor 为什么做了那个路由决策
- 你不知道 Token 消耗主要花在哪个 Agent
- 你不知道哪些 Agent 经常失败、为什么失败

## 三层可观测性

### 第一层：Trace（链路追踪）

Trace 记录一个请求的完整执行路径。每个 Agent 执行时创建一个 Span，所有 Span 组成一棵树。

```python
import time
import uuid
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any
from contextlib import contextmanager


@dataclass
class Span:
    trace_id: str
    span_id: str
    parent_id: str | None
    agent_name: str
    input_data: dict
    output_data: dict | None = None
    start_time: float = 0.0
    end_time: float = 0.0
    duration_ms: float = 0.0
    status: str = "running"  # running / success / error
    error: str | None = None
    metadata: dict = field(default_factory=dict)


class Tracer:
    def __init__(self):
        self.spans: list[Span] = []
        self._current_span: Span | None = None

    @contextmanager
    def start_span(self, agent_name: str, input_data: dict,
                   parent_id: str | None = None):
        span = Span(
            trace_id=self._get_trace_id(),
            span_id=str(uuid.uuid4())[:8],
            parent_id=parent_id,
            agent_name=agent_name,
            input_data=input_data,
            start_time=time.time(),
        )
        self.spans.append(span)
        parent = self._current_span
        self._current_span = span
        try:
            yield span
            span.status = "success"
        except Exception as e:
            span.status = "error"
            span.error = str(e)
            raise
        finally:
            span.end_time = time.time()
            span.duration_ms = (span.end_time - span.start_time) * 1000
            self._current_span = parent

    def get_trace(self, trace_id: str) -> list[Span]:
        return [s for s in self.spans if s.trace_id == trace_id]

    def _get_trace_id(self) -> str:
        if self._current_span:
            return self._current_span.trace_id
        return str(uuid.uuid4())[:8]
```

使用方式：

```python
tracer = Tracer()

def researcher_agent(state, tracer):
    with tracer.start_span("researcher", {"query": state["query"]}) as span:
        result = do_research(state["query"])
        span.output_data = {"research": result[:200]}
        return result
```

关键设计点：
- `trace_id` 贯穿整个请求链路，从入口到出口
- `parent_id` 记录父子关系，支持嵌套 Agent（一个 Agent 内部调用另一个 Agent）
- `input_data` 和 `output_data` 只记录摘要（前 200 字符），避免日志过大

### 第二层：Structured Logging（结构化日志）

追踪告诉你"经过了哪些节点"，日志告诉你"每个节点内部发生了什么"。

```python
import structlog


def setup_logging():
    structlog.configure(
        processors=[
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.add_log_level,
            structlog.processors.JSONRenderer(),
        ],
    )


class AgentLogger:
    def __init__(self, agent_name: str):
        self.logger = structlog.get_logger().bind(agent=agent_name)

    def log_llm_call(self, model: str, prompt_tokens: int,
                     completion_tokens: int, duration_ms: float):
        self.logger.info(
            "llm_call",
            model=model,
            prompt_tokens=prompt_tokens,
            completion_tokens=completion_tokens,
            duration_ms=duration_ms,
        )

    def log_tool_call(self, tool_name: str, input_summary: str,
                      success: bool, duration_ms: float):
        self.logger.info(
            "tool_call",
            tool_name=tool_name,
            input_summary=input_summary[:100],
            success=success,
            duration_ms=duration_ms,
        )

    def log_routing_decision(self, decision: str, reason: str):
        self.logger.info(
            "routing_decision",
            decision=decision,
            reason=reason[:200],
        )

    def log_error(self, error_type: str, error_message: str,
                  context: dict = None):
        self.logger.error(
            "agent_error",
            error_type=error_type,
            error_message=error_message,
            context=context or {},
        )
```

结构化日志的核心原则：
- 每条日志是一个 JSON 对象，不是一行字符串
- 包含 agent 名称、时间戳、日志级别
- LLM 调用记录 model、token 数量、耗时
- 路由决策记录决策结果和原因

### 第三层：Metrics（指标）

指标是聚合数据，告诉你系统的整体健康状况。

```python
from collections import defaultdict
import statistics


class MetricsCollector:
    def __init__(self):
        self.agent_durations: dict[str, list[float]] = defaultdict(list)
        self.agent_success: dict[str, list[bool]] = defaultdict(list)
        self.token_usage: dict[str, list[int]] = defaultdict(list)
        self.routing_counts: dict[str, int] = defaultdict(int)

    def record_agent_execution(self, agent_name: str,
                               duration_ms: float, success: bool):
        self.agent_durations[agent_name].append(duration_ms)
        self.agent_success[agent_name].append(success)

    def record_token_usage(self, agent_name: str, tokens: int):
        self.token_usage[agent_name].append(tokens)

    def record_routing(self, from_agent: str, to_agent: str):
        self.routing_counts[f"{from_agent}→{to_agent}"] += 1

    def get_agent_summary(self, agent_name: str) -> dict:
        durations = self.agent_durations.get(agent_name, [])
        successes = self.agent_success.get(agent_name, [])
        tokens = self.token_usage.get(agent_name, [])

        return {
            "total_calls": len(durations),
            "success_rate": (
                sum(successes) / len(successes) if successes else 0
            ),
            "avg_duration_ms": (
                statistics.mean(durations) if durations else 0
            ),
            "p95_duration_ms": (
                statistics.quantiles(durations, n=20)[18]
                if len(durations) >= 20 else 0
            ),
            "total_tokens": sum(tokens),
            "avg_tokens": (
                statistics.mean(tokens) if tokens else 0
            ),
        }

    def get_system_summary(self) -> dict:
        return {
            agent: self.get_agent_summary(agent)
            for agent in self.agent_durations
        }
```

## 关键指标

多 Agent 系统应该监控的核心指标：

**每个 Agent 的指标**：
- 调用次数
- 成功率
- 平均/P95 延迟
- Token 消耗（输入 + 输出）
- 错误类型分布

**系统级指标**：
- 端到端请求延迟（从用户输入到最终输出）
- 总 Token 消耗和成本
- 路由分布（哪些 Agent 被调用最多）
- 重试次数
- 人工审批等待时间

## 参考答案

### 练习一

```python
def print_trace_tree(spans: list[Span]):
    if not spans:
        return

    trace_id = spans[0].trace_id
    total_ms = sum(s.duration_ms for s in spans)
    print(f"trace: {trace_id} (total: {total_ms:.0f}ms)")

    # 按 parent_id 组织树结构
    children: dict[str | None, list[Span]] = defaultdict(list)
    for s in spans:
        children[s.parent_id].append(s)

    def print_node(parent_id, prefix=""):
        nodes = children.get(parent_id, [])
        for i, node in enumerate(nodes):
            is_last = i == len(nodes) - 1
            connector = "└──" if is_last else "├──"
            status = "✓" if node.status == "success" else "✗"
            print(f"{prefix}{connector} {node.agent_name} "
                  f"({node.duration_ms:.0f}ms) {status}")
            extension = "    " if is_last else "│   "
            print_node(node.span_id, prefix + extension)

    print_node(None)
```

### 练习二

```python
class MetricsCollectorWithBudget(MetricsCollector):
    def __init__(self, token_budget: int = 10000):
        super().__init__()
        self.token_budget = token_budget
        self._request_tokens: dict[str, int] = {}

    def record_token_usage(self, request_id: str, agent_name: str,
                           tokens: int):
        super().record_token_usage(agent_name, tokens)
        self._request_tokens[request_id] = (
            self._request_tokens.get(request_id, 0) + tokens
        )
        self.check_budget(request_id)

    def check_budget(self, request_id: str):
        used = self._request_tokens.get(request_id, 0)
        if used > self.token_budget:
            raise BudgetExceededError(
                f"请求 {request_id} Token 消耗 "
                f"{used} 超过预算 {self.token_budget}"
            )
```

设计决策：预算检查在 record_token_usage 时触发，而不是在 Agent 执行前预估。原因是 LLM 的输出 token 数无法准确预估，只能在调用后检查。这意味超预算时已经消耗了额外的 token，但总消耗是可控的。
