# 阶段实战：为多 Agent 系统实现完整的通信与记忆层

> 预计时长：4 小时
> 目标：构建一个集成消息传递、共享黑板、长期记忆的多 Agent 通信层，并用三个 Agent 协作完成任务

## 你将构建什么

一个三层架构的多 Agent 通信系统：

```
┌─────────────────────────────────────────────┐
│                 Agent 层                     │
│  Researcher  │  Analyst  │  Writer          │
├─────────────────────────────────────────────┤
│               通信层                         │
│  MessageBroker  │  Blackboard               │
├─────────────────────────────────────────────┤
│               记忆层                         │
│  ShortTermMemory (会话内)                    │
│  LongTermMemory  (跨会话，基于向量检索)      │
└─────────────────────────────────────────────┘
```

核心设计约束：
- 通信层和记忆层解耦，Agent 通过统一接口访问
- 每个 Agent 可以独立地读写共享状态
- 记忆层支持"存入"和"检索"两种操作
- 所有操作都有结构化日志

## Step 1：记忆层

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any
import json
import hashlib


@dataclass
class MemoryEntry:
    id: str
    content: str
    metadata: dict
    timestamp: str = field(
        default_factory=lambda: datetime.now().isoformat()
    )
    relevance_score: float = 0.0


class ShortTermMemory:
    """会话内记忆，基于滑动窗口。"""

    def __init__(self, max_entries: int = 50):
        self.entries: list[MemoryEntry] = []
        self.max_entries = max_entries

    def store(self, content: str, metadata: dict = None):
        entry = MemoryEntry(
            id=self._generate_id(content),
            content=content,
            metadata=metadata or {},
        )
        self.entries.append(entry)
        # 超出窗口时移除最旧的
        if len(self.entries) > self.max_entries:
            self.entries = self.entries[-self.max_entries:]

    def retrieve(self, query: str, top_k: int = 5) -> list[MemoryEntry]:
        """基于关键词匹配的简单检索。"""
        query_lower = query.lower()
        scored = []
        for entry in self.entries:
            # 简单的关键词匹配打分
            score = sum(
                1 for word in query_lower.split()
                if word in entry.content.lower()
            )
            if score > 0:
                scored.append((score, entry))
        scored.sort(key=lambda x: x[0], reverse=True)
        return [entry for _, entry in scored[:top_k]]

    def clear(self):
        self.entries.clear()

    def _generate_id(self, content: str) -> str:
        return hashlib.md5(
            f"{content}{datetime.now().isoformat()}".encode()
        ).hexdigest()[:8]


class LongTermMemory:
    """跨会话记忆，基于向量检索（简化版用关键词匹配）。"""

    def __init__(self):
        self.entries: list[MemoryEntry] = []

    def store(self, content: str, metadata: dict = None):
        entry = MemoryEntry(
            id=self._generate_id(content),
            content=content,
            metadata=metadata or {},
        )
        self.entries.append(entry)

    def retrieve(self, query: str, top_k: int = 5) -> list[MemoryEntry]:
        """实际项目中这里应该用向量数据库（如 ChromaDB）。"""
        query_lower = query.lower()
        scored = []
        for entry in self.entries:
            score = sum(
                1 for word in query_lower.split()
                if word in entry.content.lower()
            )
            if score > 0:
                entry.relevance_score = score
                scored.append(entry)
        scored.sort(key=lambda x: x.relevance_score, reverse=True)
        return scored[:top_k]

    def _generate_id(self, content: str) -> str:
        return hashlib.md5(
            f"{content}{datetime.now().isoformat()}".encode()
        ).hexdigest()[:8]
```

为什么区分短期和长期？短期记忆是"工作记忆"，存放当前会话的中间结果，用滑动窗口控制大小。长期记忆是"知识库"，存放跨会话的经验和知识，用向量检索做相关性匹配。

## Step 2：通信层

```python
import asyncio
import threading
from dataclasses import dataclass


@dataclass
class Message:
    sender: str
    receiver: str
    action: str
    payload: dict
    timestamp: str = field(
        default_factory=lambda: datetime.now().isoformat()
    )


class MessageBroker:
    """异步消息代理。"""

    def __init__(self):
        self._queues: dict[str, asyncio.Queue] = {}

    def register(self, agent_name: str):
        self._queues[agent_name] = asyncio.Queue()

    async def send(self, message: Message):
        queue = self._queues.get(message.receiver)
        if queue is None:
            raise ValueError(f"未知接收者: {message.receiver}")
        await queue.put(message)

    async def receive(self, agent_name: str) -> Message:
        return await self._queues[agent_name].get()


class Blackboard:
    """线程安全的共享黑板。"""

    def __init__(self):
        self._data: dict[str, dict] = {}
        self._lock = threading.Lock()

    def write(self, key: str, value: Any, author: str):
        with self._lock:
            self._data[key] = {
                "value": value,
                "author": author,
                "timestamp": datetime.now().isoformat(),
            }

    def read(self, key: str) -> Any | None:
        entry = self._data.get(key)
        return entry["value"] if entry else None

    def read_all(self) -> dict[str, Any]:
        with self._lock:
            return {k: v["value"] for k, v in self._data.items()}
```

## Step 3：通信层整合

```python
class CommunicationLayer:
    """统一的通信层，封装消息代理、黑板和记忆。"""

    def __init__(self):
        self.broker = MessageBroker()
        self.blackboard = Blackboard()
        self.short_term = ShortTermMemory()
        self.long_term = LongTermMemory()

    def register_agent(self, agent_name: str):
        self.broker.register(agent_name)

    async def send_message(self, sender: str, receiver: str,
                           action: str, payload: dict):
        msg = Message(sender=sender, receiver=receiver,
                      action=action, payload=payload)
        await self.broker.send(msg)

    async def receive_message(self, agent_name: str) -> Message:
        return await self.broker.receive(agent_name)

    def update_state(self, key: str, value: Any, agent_name: str):
        self.blackboard.write(key, value, agent_name)

    def get_state(self, key: str) -> Any | None:
        return self.blackboard.read(key)

    def remember(self, content: str, metadata: dict = None,
                 long_term: bool = False):
        self.short_term.store(content, metadata)
        if long_term:
            self.long_term.store(content, metadata)

    def recall(self, query: str, top_k: int = 5) -> list[MemoryEntry]:
        # 先查短期记忆
        results = self.short_term.retrieve(query, top_k)
        if len(results) < top_k:
            # 补充长期记忆
            long_results = self.long_term.retrieve(
                query, top_k - len(results)
            )
            results.extend(long_results)
        return results
```

## Step 4：Agent 实现

```python
class CommunicatingAgent:
    def __init__(self, name: str, role: str,
                 comm: CommunicationLayer, llm):
        self.name = name
        self.role = role
        self.comm = comm
        self.llm = llm

    async def execute(self, task: str) -> str:
        # 1. 检索相关记忆
        memories = self.comm.recall(task, top_k=3)
        memory_text = "\n".join(
            f"- {m.content}" for m in memories
        ) if memories else "无相关记忆"

        # 2. 读取共享状态
        shared = self.comm.blackboard.read_all()
        shared_text = json.dumps(
            shared, ensure_ascii=False, indent=2
        ) if shared else "无共享状态"

        # 3. 执行任务
        prompt = f"""你是{self.role}。

任务：{task}

相关记忆：
{memory_text}

共享状态：
{shared_text}

请完成任务并输出结果。"""

        response = self.llm.invoke(prompt)
        result = response.content

        # 4. 存储记忆
        self.comm.remember(
            f"[{self.name}] 任务: {task}\n结果: {result[:200]}",
            metadata={"agent": self.name, "task": task},
            long_term=True,
        )

        # 5. 更新共享状态
        self.comm.update_state(
            f"{self.name}_result", result, self.name
        )

        return result
```

## Step 5：组装运行

```python
import asyncio
from langchain_openai import ChatOpenAI


async def main():
    llm = ChatOpenAI(model="gpt-4o-mini")
    comm = CommunicationLayer()

    # 注册 Agent
    for name in ["researcher", "analyst", "writer"]:
        comm.register_agent(name)

    researcher = CommunicatingAgent(
        "researcher", "研究专家", comm, llm
    )
    analyst = CommunicatingAgent(
        "analyst", "分析专家", comm, llm
    )
    writer = CommunicatingAgent(
        "writer", "写作专家", comm, llm
    )

    # 顺序执行（也可以并行）
    topic = "2025 年多 Agent 系统的发展趋势"

    print("=== 研究阶段 ===")
    research = await researcher.execute(topic)
    print(research[:200])

    print("\n=== 分析阶段 ===")
    analysis = await analyst.execute(
        f"基于研究结果进行分析: {research[:300]}"
    )
    print(analysis[:200])

    print("\n=== 写作阶段 ===")
    report = await writer.execute(
        f"基于分析结果撰写报告: {analysis[:300]}"
    )
    print(report[:200])

    print("\n=== 记忆检索测试 ===")
    memories = comm.recall("多 Agent 趋势", top_k=3)
    for m in memories:
        print(f"  [{m.metadata.get('agent', '?')}] {m.content[:80]}")


if __name__ == "__main__":
    asyncio.run(main())
```

## 验收标准

1. 三个 Agent 能通过通信层协作完成任务
2. 每个 Agent 执行后，结果写入黑板，其他 Agent 可以读取
3. 长期记忆能在第二个 Agent 执行时检索到第一个 Agent 存入的内容
4. 所有通信都有日志记录
5. 代码结构清晰，通信层和 Agent 层解耦

## 扩展挑战

1. **并行 Agent + 冲突解决**：让 researcher 和 analyst 并行执行，当两者在黑板写入冲突时，用 LLM 仲裁
2. **记忆压缩**：当短期记忆超过阈值时，用 LLM 将多条记忆压缩成一条摘要
3. **消息超时**：给 MessageBroker 加超时机制，超时未消费的消息进入死信队列
