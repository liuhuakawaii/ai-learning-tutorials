# 阶段实战：实现带审批流的多 Agent 内容生成系统

> 预计时长：4 小时
> 目标：用 LangGraph 构建一个内容生成系统，包含研究→写作→审核→人工审批→发布的完整流程

## 架构设计

这个系统的核心挑战不是"怎么生成内容"，而是"怎么在正确的地方停下来让人确认"。

```
用户提交主题
    │
    ▼
研究 Agent ──→ 写作 Agent ──→ 自动审核 Agent
                                    │
                            ┌───────┴───────┐
                            │   质量检查     │
                            │   通过？       │
                            └───────┬───────┘
                              是 /     \ 否
                              │         │
                              ▼         ▼
                         人工审批    写作 Agent（重写）
                              │
                        ┌─────┴─────┐
                        │  审批结果  │
                        └─────┬─────┘
                       批准 /    \ 拒绝
                         │        │
                         ▼        ▼
                      发布      终止
```

关键设计决策：
- 自动审核在人工审批之前，过滤掉明显的低质量内容
- 审核不通过时回到写作 Agent 而不是研究 Agent（保留搜索结果）
- 人工审批使用 `interrupt_before` 机制
- 最多重写 3 次，防止死循环

## 完整实现

### State 定义

```python
from typing import TypedDict, Annotated
from operator import add
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.memory import MemorySaver
from langchain_openai import ChatOpenAI


class ContentState(TypedDict):
    topic: str
    research: str
    draft: str
    auto_review: str
    auto_approved: bool
    human_approved: bool
    human_feedback: str
    final_content: str
    rewrite_count: int
    current_step: str
    errors: Annotated[list[str], add]
```

### Agent 节点

```python
llm = ChatOpenAI(model="gpt-4o-mini")


def research_agent(state: ContentState) -> dict:
    """研究 Agent：收集主题相关信息。"""
    prompt = f"""你是一个研究专家。针对以下主题，收集关键信息和数据。

主题：{state['topic']}

输出：
1. 3-5 个核心事实
2. 相关数据和统计
3. 该主题的关键背景信息"""

    response = llm.invoke(prompt)
    return {
        "research": response.content,
        "current_step": "auto_review_check",
    }


def writing_agent(state: ContentState) -> dict:
    """写作 Agent：基于研究结果撰写内容。"""
    feedback_hint = ""
    if state.get("human_feedback"):
        feedback_hint = f"\n\n审稿人反馈：{state['human_feedback']}"
    elif state.get("auto_review"):
        feedback_hint = f"\n\n自动审核反馈：{state['auto_review']}"

    prompt = f"""你是一个技术写作专家。基于以下研究结果撰写一篇技术文章。

研究结果：
{state['research']}{feedback_hint}

文章要求：
1. 标题明确
2. 开头直接切入主题（不要"在当今..."这种开头）
3. 每个论点有数据支撑
4. 结论清晰
5. 总字数 600-1000 字"""

    response = llm.invoke(prompt)
    return {
        "draft": response.content,
        "rewrite_count": state.get("rewrite_count", 0) + 1,
        "current_step": "auto_review",
    }


def auto_review_agent(state: ContentState) -> dict:
    """自动审核 Agent：评估内容质量。"""
    prompt = f"""你是一个内容审核专家。评估以下文章的质量。

文章：
{state['draft']}

评估标准：
1. 是否有明确的主题和论点
2. 数据和事实是否有支撑
3. 逻辑是否连贯
4. 字数是否在 600-1000 字范围内

输出：
- 判定：通过 / 不通过
- 不通过的原因和具体修改建议"""

    response = llm.invoke(prompt)
    approved = "通过" in response.content and "不通过" not in response.content
    return {
        "auto_review": response.content,
        "auto_approved": approved,
        "current_step": "routing",
    }


def publish_agent(state: ContentState) -> dict:
    """发布 Agent：最终处理并发布。"""
    return {
        "final_content": state["draft"],
        "current_step": "done",
    }
```

### 路由逻辑

```python
def route_after_auto_review(state: ContentState) -> str:
    """自动审核后的路由。"""
    if state.get("auto_approved"):
        return "human_approval"
    if state.get("rewrite_count", 0) >= 3:
        # 重写 3 次仍不通过，强制进入人工审批
        return "human_approval"
    return "writer"


def route_after_human_approval(state: ContentState) -> str:
    """人工审批后的路由。"""
    if state.get("human_approved"):
        return "publish"
    return END
```

### 图构建

```python
graph = StateGraph(ContentState)

graph.add_node("researcher", research_agent)
graph.add_node("writer", writing_agent)
graph.add_node("auto_review", auto_review_agent)
graph.add_node("human_approval", lambda s: {})  # 中断点
graph.add_node("publish", publish_agent)

graph.set_entry_point("researcher")
graph.add_edge("researcher", "writer")
graph.add_edge("writer", "auto_review")

graph.add_conditional_edges("auto_review", route_after_auto_review, {
    "human_approval": "human_approval",
    "writer": "writer",
})

graph.add_conditional_edges("human_approval", route_after_human_approval, {
    "publish": "publish",
    END: END,
})

graph.add_edge("publish", END)

checkpointer = MemorySaver()
app = graph.compile(
    checkpointer=checkpointer,
    interrupt_before=["human_approval"],
)
```

### 运行流程

```python
def run_content_pipeline(topic: str):
    config = {"configurable": {"thread_id": f"content-{topic[:10]}"}}

    # 第一阶段：研究 → 写作 → 自动审核 → 暂停等待人工审批
    result = app.invoke({
        "topic": topic,
        "research": "", "draft": "", "auto_review": "",
        "auto_approved": False, "human_approved": False,
        "human_feedback": "", "final_content": "",
        "rewrite_count": 0, "current_step": "", "errors": [],
    }, config)

    print("=" * 60)
    print(f"主题: {topic}")
    print(f"\n研究结果:\n{result['research'][:300]}...")
    print(f"\n草稿:\n{result['draft'][:300]}...")
    print(f"\n自动审核: {'通过' if result['auto_approved'] else '未通过'}")
    print(f"重写次数: {result['rewrite_count']}")

    # 此时图暂停在 human_approval 节点
    # 模拟人工审批
    print("\n" + "=" * 60)
    print("等待人工审批...")
    decision = input("输入 'approve' 批准，'reject' 拒绝，'edit:xxx' 修改: ")

    if decision == "approve":
        app.update_state(config, {
            "human_approved": True,
            "human_feedback": "内容质量OK",
        })
    elif decision.startswith("edit:"):
        new_content = decision[5:]
        app.update_state(config, {
            "human_approved": True,
            "human_feedback": f"已修改: {new_content}",
            "draft": new_content,
        })
    else:
        app.update_state(config, {
            "human_approved": False,
            "human_feedback": "内容不符合要求",
        })
        print("已拒绝，流程终止")
        return

    # 第二阶段：从暂停点继续执行
    result = app.invoke(None, config)
    print(f"\n最终内容:\n{result['final_content'][:300]}...")
    print(f"\n状态: {result['current_step']}")


if __name__ == "__main__":
    run_content_pipeline("2025 年多 Agent 系统的发展趋势")
```

## 为什么这样设计

**自动审核在人工审批之前**：人工审批是昂贵的（需要人的时间），自动审核是便宜的（一次 LLM 调用）。让自动审核先过滤掉明显的低质量内容，减少人工审批的负担。

**审核不通过时回到 writer 而不是 researcher**：搜索结果通常不需要因为写作质量差而重新获取。回到 writer 可以保留 research 字段，只重写 draft。

**重写 3 次后强制进入人工审批**：如果自动审核连续 3 次不通过，说明自动审核的标准和 writer 的能力之间有系统性偏差。这时候需要人类介入判断，是审核标准太严格，还是写作质量确实有问题。

**interrupt_before 而不是 interrupt_after**：interrupt_before 在节点执行前暂停，意味着人工审批节点本身不需要任何代码——它只是一个暂停标记。审批逻辑在外部（update_state）完成。

## 验收标准

1. 完整流程能跑通：研究→写作→自动审核→人工审批→发布
2. 自动审核不通过时能自动重写（最多 3 次）
3. 人工审批能暂停并等待外部输入
4. 审批者的修改能正确传递到后续节点
5. 代码结构清晰，路由逻辑和业务逻辑分离

## 扩展挑战

1. **审批面板**：用 Streamlit 或 FastAPI 搭建一个简单的审批 UI，展示草稿内容，支持批准/拒绝/修改
2. **多级审批**：添加第二级审批（如主编审批），低风险内容只需要一级审批，高风险内容需要两级
3. **审批历史**：记录每次审批的完整上下文（谁审批的、什么时候、修改了什么），支持回溯
