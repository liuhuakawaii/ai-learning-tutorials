> **目标**：把图片理解、OCR、信息提取、描述生成串成一条完整的商品分析 Pipeline

## 当前卡点

你已经会单独调 Vision API 做图片理解、OCR、图表分析、图片生成。但真实需求不是五个独立功能——运营每天上传 500 张商品图，需要自动完成：识别商品类别 → 提取品牌和规格 → 生成营销文案。三步串起来，中间还要处理异常、控制成本、保证输出格式一致。

卡在哪里？每一步单独都能跑，串起来之后问题就来了：
- 一步失败，后面全废
- 每张图调 3 次 API，500 张图就是 1500 次调用，成本和速度都扛不住
- 有的图是白底精修图，有的是手机随手拍，处理策略应该不同

## 系统设计

把分析流程拆成三步，每步独立可替换：

```
图片输入
  │
  ▼
[预处理] ── 判断图片类型，选择 detail 等级
  │
  ▼
[分类 + OCR] ── 一次调用同时完成分类和文字提取
  │
  ▼
[文案生成] ── 根据前一步结果生成营销文案
  │
  ▼
结构化 JSON 输出
```

关键设计判断：
- **分类和 OCR 合成一次调用**，因为 GPT-4o 可以在一次请求里同时做分类和文字提取，拆成两次调用浪费钱
- **文案生成用前一步的结构化结果**而不是原始图片，因为用 JSON 比用图片便宜 100 倍
- **预处理阶段决定 detail 等级**：精修图用 low（便宜），手机拍的用 high（准确）

## 核心实现

```python
import json
import base64
from pathlib import Path
from openai import OpenAI


class ProductAnalyzer:
    def __init__(self, model: str = "gpt-4o"):
        self.client = OpenAI()
        self.model = model

    def analyze(self, image_path: str) -> dict:
        """完整分析一张商品图"""
        image_b64 = self._load_image(image_path)
        detail = self._guess_detail_level(image_path)

        # 第一步：分类 + OCR 合并为一次调用
        product_info = self._classify_and_extract(image_b64, detail)

        # 第二步：用结构化结果生成文案（不传图片）
        marketing_copy = self._generate_copy(product_info)

        return {
            **product_info,
            "marketing_copy": marketing_copy,
            "source_image": str(image_path),
        }

    def _load_image(self, path: str) -> str:
        with open(path, "rb") as f:
            return base64.b64encode(f.read()).decode()

    def _guess_detail_level(self, path: str) -> str:
        """根据文件大小粗略判断是否需要高分辨率"""
        size_mb = Path(path).stat().st_size / (1024 * 1024)
        return "high" if size_mb > 2 else "low"

    def _classify_and_extract(self, image_b64: str, detail: str) -> dict:
        """一次调用完成分类和文字提取"""
        prompt = """分析这张商品图片，输出 JSON：
{
  "category": "商品大类（如：服装、电子产品、食品）",
  "subcategory": "细分类目",
  "brand": "品牌名（看不清填 null）",
  "color": "主色调",
  "material": "材质（看不清填 null）",
  "text_on_image": ["图片中所有可见文字"],
  "key_features": ["卖点特征，最多 5 个"],
  "image_quality": "good/fair/poor（图片质量）"
}"""

        response = self.client.chat.completions.create(
            model=self.model,
            messages=[{
                "role": "user",
                "content": [
                    {"type": "text", "text": prompt},
                    {"type": "image_url", "image_url": {
                        "url": f"data:image/jpeg;base64,{image_b64}",
                        "detail": detail
                    }}
                ]
            }],
            response_format={"type": "json_object"}
        )
        return json.loads(response.choices[0].message.content)

    def _generate_copy(self, product_info: dict) -> str:
        """用结构化信息生成营销文案（不传图片，省 token）"""
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",  # 文案生成用便宜模型就够
            messages=[
                {"role": "system", "content": "你是电商文案写手。根据商品信息写一段 50 字以内的卖点文案，突出核心优势，语气活泼。"},
                {"role": "user", "content": json.dumps(product_info, ensure_ascii=False)}
            ]
        )
        return response.choices[0].message.content
```

几个设计判断说明：

- **`_guess_detail_level`** 用文件大小而不是图片分辨率，因为读文件头判断分辨率要引入 Pillow 依赖，文件大小已经够用——大文件通常是高分辨率图
- **文案生成用 gpt-4o-mini**：从结构化 JSON 生成文案不需要强推理，用大模型浪费钱
- **`image_quality` 字段**：让下游知道这张图的质量，quality=poor 的结果可以自动标记为"需要人工复核"

## 批量处理

一张图能跑通之后，批量处理的核心问题是速度和容错：

```python
import asyncio
from dataclasses import dataclass


@dataclass
class AnalysisResult:
    image_path: str
    success: bool
    data: dict | None = None
    error: str | None = None


async def batch_analyze(image_paths: list[str], concurrency: int = 5) -> list[AnalysisResult]:
    """并发分析多张图片，带错误隔离"""
    analyzer = ProductAnalyzer()
    semaphore = asyncio.Semaphore(concurrency)

    async def analyze_one(path: str) -> AnalysisResult:
        async with semaphore:
            try:
                result = await asyncio.to_thread(analyzer.analyze, path)
                return AnalysisResult(image_path=path, success=True, data=result)
            except Exception as e:
                return AnalysisResult(image_path=path, success=False, error=str(e))

    tasks = [analyze_one(p) for p in image_paths]
    return await asyncio.gather(*tasks)
```

注意 `concurrency=5`——不是越高越好。OpenAI API 的并发限制通常是 500 RPM，但图片请求的 token 消耗大，5 并发已经够稳。设太高容易触发限流。

## 验证系统

跑通之后，检查这几件事：

```python
def validate_result(result: dict) -> list[str]:
    """检查分析结果是否合理"""
    warnings = []

    if not result.get("category"):
        warnings.append("缺少商品分类")
    if result.get("text_on_image") and len(result["text_on_image"]) > 0:
        # 有文字提取结果时，检查是否可能编造
        if result.get("image_quality") == "poor":
            warnings.append("图片质量差但提取了文字，可能有编造，建议人工复核")
    if not result.get("marketing_copy"):
        warnings.append("文案生成失败")

    return warnings
```

```python
def generate_report(results: list[AnalysisResult]) -> str:
    total = len(results)
    success = [r for r in results if r.success]
    failed = [r for r in results if not r.success]
    needs_review = [r for r in success if r.data.get("image_quality") == "poor"]

    categories = {}
    for r in success:
        cat = r.data.get("category", "unknown")
        categories[cat] = categories.get(cat, 0) + 1

    lines = [
        "# 批量分析报告\n",
        f"- 总数: {total}",
        f"- 成功: {len(success)}",
        f"- 失败: {len(failed)}",
        f"- 需人工复核: {len(needs_review)}\n",
        "## 分类分布",
    ]
    for cat, count in sorted(categories.items(), key=lambda x: -x[1]):
        lines.append(f"- {cat}: {count}")

    if failed:
        lines.append("\n## 失败列表")
        for r in failed:
            lines.append(f"- {r.image_path}: {r.error}")

    return "\n".join(lines)
```
