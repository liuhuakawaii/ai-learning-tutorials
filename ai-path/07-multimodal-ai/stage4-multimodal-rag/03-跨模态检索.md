```python
def hybrid_search(query: str, query_type: str, target_type: str, top_k: int = 5) -> list:
    """混合检索"""
    if query_type == "text":
        embedding = get_text_embedding(query)
    else:
        embedding = get_image_embedding(query)
    
    where_filter = {"type": target_type} if target_type else None
    
    results = collection.query(
        query_embeddings=[embedding],
        n_results=top_k,
        where=where_filter
    )
    
    return results
```

## 六、评估

```python
def evaluate_retrieval(query: str, expected_results: list, actual_results: list) -> dict:
    """评估检索效果"""
    # 计算准确率
    correct = len(set(expected_results) & set(actual_results))
    precision = correct / len(actual_results) if actual_results else 0
    recall = correct / len(expected_results) if expected_results else 0
    
    return {
        "precision": precision,
        "recall": recall,
        "f1": 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0
    }
```

## 常见误区

1. **只做向量检索不做排序**：向量相似度最高的结果不一定是用户最想要的，需要结合业务规则（销量、评分、价格）做重排序。
2. **忽略相似度阈值的设定**：没有阈值过滤会返回大量不相关的结果，阈值太高又会漏掉相关结果，需要根据业务场景调优。
3. **不做检索结果的多样性控制**：返回的 top-k 结果可能非常相似（如同一款衣服的不同角度），需要做多样性打散。
4. **评估指标只看准确率不看召回率**：准确率高但召回率低意味着漏掉了很多相关结果，需要同时关注两个指标。

## 工程建议

1. **实现混合检索**：向量检索 + 元数据过滤（类别、价格区间、品牌），先过滤再排序，提高检索精度。
2. **做 A/B 测试优化阈值**：用真实用户行为数据（点击、购买）优化相似度阈值和排序权重。
3. **缓存热门查询结果**：高频查询的结果做缓存，减少向量检索的计算开销。
4. **建立检索质量监控**：定期统计无结果率、点击率、首屏命中率等指标，持续优化检索效果。

## 参考答案

### 练习一：文字搜图片——实现用文字搜图片功能

**思路**：将文本查询编码为向量，在图片向量集合中做相似度检索，返回 Top-K 最匹配的图片结果。

**答案**：
```python
import chromadb
from transformers import CLIPProcessor, CLIPModel
from PIL import Image
import torch
import numpy as np

model = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")

def get_text_embedding(text: str) -> list:
    inputs = processor(text=text, return_tensors="pt", padding=True, truncation=True)
    with torch.no_grad():
        features = model.get_text_features(**inputs)
    features = features / features.norm(dim=-1, keepdim=True)
    return features.squeeze().tolist()

def get_image_embedding(image_path: str) -> list:
    image = Image.open(image_path).convert("RGB")
    inputs = processor(images=image, return_tensors="pt")
    with torch.no_grad():
        features = model.get_image_features(**inputs)
    features = features / features.norm(dim=-1, keepdim=True)
    return features.squeeze().tolist()

def search_images_by_text(collection, query: str, top_k: int = 5, threshold: float = None) -> list:
    """用文字搜图片"""
    query_emb = get_text_embedding(query)
    where_filter = {"type": "image"}

    results = collection.query(
        query_embeddings=[query_emb],
        n_results=top_k,
        where=where_filter
    )

    output = []
    for i in range(len(results["ids"][0])):
        distance = results["distances"][0][i]
        if threshold is not None and distance > threshold:
            continue
        output.append({
            "id": results["ids"][0][i],
            "path": results["metadatas"][0][i]["path"],
            "distance": distance
        })
    return output

# 使用示例
client = chromadb.PersistentClient(path="./chroma_db")
collection = client.get_collection("multimodal")

results = search_images_by_text(collection, "红色碎花连衣裙", top_k=5, threshold=0.5)
for r in results:
    print(f"  图片: {r['path']}, 距离: {r['distance']:.4f}")
```

**要点**：
- 检索时必须用 `where={"type": "image"}` 过滤，否则会返回文本结果
- 设置相似度阈值过滤低质量结果，避免返回不相关的图片
- 常见错误：不归一化向量导致余弦相似度计算不准确

### 练习二：图片搜文字——实现用图片搜文字功能

**思路**：将查询图片编码为向量，在文本向量集合中做相似度检索，返回与图片内容最匹配的文本。

**答案**：
```python
def search_texts_by_image(collection, image_path: str, top_k: int = 5, threshold: float = None) -> list:
    """用图片搜文字"""
    query_emb = get_image_embedding(image_path)
    where_filter = {"type": "text"}

    results = collection.query(
        query_embeddings=[query_emb],
        n_results=top_k,
        where=where_filter
    )

    output = []
    for i in range(len(results["ids"][0])):
        distance = results["distances"][0][i]
        if threshold is not None and distance > threshold:
            continue
        output.append({
            "id": results["ids"][0][i],
            "content": results["metadatas"][0][i]["content"],
            "distance": distance
        })
    return output

def cross_modal_search(collection, query, query_type: str, target_type: str, top_k: int = 5) -> list:
    """通用跨模态检索"""
    if query_type == "text":
        query_emb = get_text_embedding(query)
    else:
        query_emb = get_image_embedding(query)

    results = collection.query(
        query_embeddings=[query_emb],
        n_results=top_k,
        where={"type": target_type}
    )

    output = []
    for i in range(len(results["ids"][0])):
        meta = results["metadatas"][0][i]
        item = {
            "id": results["ids"][0][i],
            "distance": results["distances"][0][i],
            "metadata": meta
        }
        if target_type == "text":
            item["content"] = meta.get("content")
        else:
            item["path"] = meta.get("path")
        output.append(item)
    return output

# 使用示例
results = search_texts_by_image(collection, "product_001.jpg", top_k=3)
for r in results:
    print(f"  文本: {r['content']}, 距离: {r['distance']:.4f}")

# 通用跨模态检索
results = cross_modal_search(collection, "蓝色条纹衬衫", "text", "image", top_k=5)
for r in results:
    print(f"  图片: {r['path']}, 距离: {r['distance']:.4f}")
```

**要点**：
- 图片搜文字和文字搜图片使用相同的向量空间，但检索方向相反
- `cross_modal_search` 是通用接口，通过 `query_type` 和 `target_type` 参数灵活组合
- 常见错误：图片路径如果是 URL 需要先下载再编码，不能直接传 URL 字符串给 CLIP

### 练习三：评估题——评估检索效果

**思路**：构建标注数据集（query + expected results），计算 Precision、Recall、F1 等指标，分析检索质量。

**答案**：
```python
def evaluate_retrieval(collection, test_cases: list) -> dict:
    """评估检索效果"""
    precisions = []
    recalls = []
    f1_scores = []
    mrr_scores = []  # Mean Reciprocal Rank

    for case in test_cases:
        query = case["query"]
        expected_ids = set(case["expected_ids"])
        query_type = case.get("query_type", "text")
        target_type = case.get("target_type", "image")
        top_k = case.get("top_k", 10)

        # 执行检索
        if query_type == "text":
            query_emb = get_text_embedding(query)
        else:
            query_emb = get_image_embedding(query)

        results = collection.query(
            query_embeddings=[query_emb],
            n_results=top_k,
            where={"type": target_type}
        )

        result_ids = results["ids"][0]

        # Precision@K
        relevant_retrieved = len(expected_ids & set(result_ids))
        precision = relevant_retrieved / len(result_ids) if result_ids else 0
        precisions.append(precision)

        # Recall@K
        recall = relevant_retrieved / len(expected_ids) if expected_ids else 0
        recalls.append(recall)

        # F1
        if precision + recall > 0:
            f1 = 2 * precision * recall / (precision + recall)
        else:
            f1 = 0
        f1_scores.append(f1)

        # MRR（第一个相关结果的排名倒数）
        mrr = 0
        for rank, rid in enumerate(result_ids, 1):
            if rid in expected_ids:
                mrr = 1 / rank
                break
        mrr_scores.append(mrr)

    return {
        "precision@k": sum(precisions) / len(precisions),
        "recall@k": sum(recalls) / len(recalls),
        "f1@k": sum(f1_scores) / len(f1_scores),
        "mrr": sum(mrr_scores) / len(mrr_scores),
        "test_cases": len(test_cases)
    }

# 测试数据集
test_cases = [
    {
        "query": "红色碎花连衣裙",
        "expected_ids": ["img_dress_001", "img_dress_002"],
        "query_type": "text",
        "target_type": "image",
        "top_k": 5
    },
    {
        "query": "黑色皮鞋",
        "expected_ids": ["img_shoe_001"],
        "query_type": "text",
        "target_type": "image",
        "top_k": 5
    }
]

metrics = evaluate_retrieval(collection, test_cases)
print(f"Precision@5: {metrics['precision@k']:.4f}")
print(f"Recall@5: {metrics['recall@k']:.4f}")
print(f"F1@5: {metrics['f1@k']:.4f}")
print(f"MRR: {metrics['mrr']:.4f}")
```

**要点**：
- MRR（Mean Reciprocal Rank）衡量第一个相关结果的排名位置，对用户体验影响最大
- 测试数据集需要覆盖正例和负例，且数量至少 50+ 才有统计意义
- 常见错误：只看 Precision 不看 Recall，高 Precision 但低 Recall 意味着漏掉了大量相关结果
