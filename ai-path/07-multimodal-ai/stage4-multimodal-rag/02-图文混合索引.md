# 02 图文混合索引——将图片和文本存储到同一个向量空间

> 图文混合索引是多模态 RAG 的基础。

## 场景引入

你要为一个多模态知识库系统设计存储层：知识库中既有纯文本的技术文档，也有包含大量图表的产品手册，还有产品实拍图片。用户可能用文字查询（"安装步骤"），也可能用图片查询（上传一张零件照片找对应的说明书）。你需要把图片向量和文本向量存储在同一个向量数据库中，支持跨模态检索。选择 ChromaDB、Milvus 还是 pgvector？索引如何设计才能兼顾检索速度和准确率？这些都是构建多模态索引需要解决的核心问题。

- 掌握图文混合索引的设计方法
- 理解向量数据库的使用

## 五、索引管理

```python
class MultimodalIndex:
    """多模态索引"""
    
    def __init__(self):
        self.client = chromadb.Client()
        self.collection = self.client.create_collection("multimodal")
    
    def add_image(self, image_path: str, metadata: dict = None):
        """添加图片"""
        embedding = get_image_embedding(image_path)
        self.collection.add(
            ids=[f"img_{hash(image_path)}"],
            embeddings=[embedding],
            metadatas=[{"type": "image", "path": image_path, **(metadata or {})}]
        )
    
    def add_text(self, text: str, metadata: dict = None):
        """添加文本"""
        embedding = get_text_embedding(text)
        self.collection.add(
            ids=[f"txt_{hash(text)}"],
            embeddings=[embedding],
            metadatas=[{"type": "text", "content": text, **(metadata or {})}]
        )
    
    def search(self, query: str, query_type: str = "text", top_k: int = 5):
        """检索"""
        if query_type == "text":
            embedding = get_text_embedding(query)
        else:
            embedding = get_image_embedding(query)
        
        return self.collection.query(
            query_embeddings=[embedding],
            n_results=top_k
        )
```

## 常见误区

1. **把所有向量存到一个 collection**：图片向量和文本向量的维度可能不同、检索策略不同，混存会导致检索效率低下，建议按模态分 collection 或加 metadata 标记。
2. **不做向量数据库的持久化**：默认的内存模式重启后数据丢失，生产环境必须配置持久化存储。
3. **忽略索引构建的时间成本**：百万级向量的索引构建可能需要几小时，需要做增量索引而不是全量重建。
4. **不做检索结果的去重和过滤**：同一张图片可能被多次索引，不做去重会返回重复结果。

## 工程建议

1. **原型用 ChromaDB，生产用 Milvus**：ChromaDB 简单易用适合快速验证，Milvus 高性能适合生产环境，pgvector 适合已有 PostgreSQL 的团队。
2. **用 metadata 做预过滤**：存储时给每个向量加 metadata（类型、来源、时间），检索时先用 metadata 过滤再做向量匹配，大幅提高检索效率。
3. **做增量索引**：新增内容只索引增量部分，不做全量重建，节省时间和计算资源。
4. **定期做索引质量评估**：用标准查询测试检索准确率，发现向量质量问题及时重新索引。

## 参考答案

### 练习一：索引题——构建一个图文混合索引

**思路**：使用 ChromaDB 创建一个支持图文混合存储的向量集合，通过 metadata 区分模态类型，选择余弦距离作为相似度度量。

**答案**：
```python
import chromadb
from transformers import CLIPProcessor, CLIPModel
from PIL import Image
import torch
import numpy as np

model = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")

def get_image_embedding(image_path: str) -> list:
    image = Image.open(image_path).convert("RGB")
    inputs = processor(images=image, return_tensors="pt")
    with torch.no_grad():
        features = model.get_image_features(**inputs)
    features = features / features.norm(dim=-1, keepdim=True)
    return features.squeeze().tolist()

def get_text_embedding(text: str) -> list:
    inputs = processor(text=text, return_tensors="pt", padding=True, truncation=True)
    with torch.no_grad():
        features = model.get_text_features(**inputs)
    features = features / features.norm(dim=-1, keepdim=True)
    return features.squeeze().tolist()

class MultimodalIndex:
    """图文混合索引"""

    def __init__(self, collection_name: str = "multimodal", persist_dir: str = "./chroma_db"):
        self.client = chromadb.PersistentClient(path=persist_dir)
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"}
        )

    def add_image(self, image_path: str, metadata: dict = None):
        """添加图片到索引"""
        embedding = get_image_embedding(image_path)
        doc_id = f"img_{hash(image_path)}"
        meta = {"type": "image", "path": image_path, **(metadata or {})}
        self.collection.add(ids=[doc_id], embeddings=[embedding], metadatas=[meta])

    def add_text(self, text: str, metadata: dict = None):
        """添加文本到索引"""
        embedding = get_text_embedding(text)
        doc_id = f"txt_{abs(hash(text))}"
        meta = {"type": "text", "content": text, **(metadata or {})}
        self.collection.add(ids=[doc_id], embeddings=[embedding], metadatas=[meta])

    def search(self, query_embedding: list, top_k: int = 5, modality: str = None):
        """检索，支持按模态过滤"""
        where = {"type": modality} if modality else None
        return self.collection.query(
            query_embeddings=[query_embedding],
            n_results=top_k,
            where=where
        )

# 使用示例
index = MultimodalIndex()
index.add_image("product_001.jpg", {"category": "服装", "price": 299})
index.add_image("product_002.jpg", {"category": "鞋靴", "price": 599})
index.add_text("红色碎花连衣裙，适合春夏穿着", {"category": "服装"})
index.add_text("黑色皮质运动鞋，透气舒适", {"category": "鞋靴"})

query_emb = get_text_embedding("红色连衣裙")
results = index.search(query_emb, top_k=3)
print(f"检索到 {len(results['ids'][0])} 条结果")
```

**要点**：
- 使用 `PersistentClient` 而非默认的内存 `Client()`，确保数据持久化
- metadata 中应存储业务字段（类别、价格），支持后续的预过滤检索
- 常见错误：图片和文本的 Embedding 模型不同，导致向量空间不对齐，检索结果混乱

### 练习二：检索题——实现跨模态检索

**思路**：在已构建的图文混合索引上，分别实现"文搜图"和"图搜文"两种方向的检索，并对结果做后处理。

**答案**：
```python
def search_images_by_text(index: MultimodalIndex, query: str, top_k: int = 5) -> list:
    """用文字搜图片"""
    query_emb = get_text_embedding(query)
    results = index.search(query_emb, top_k=top_k, modality="image")

    output = []
    for i, doc_id in enumerate(results["ids"][0]):
        output.append({
            "id": doc_id,
            "path": results["metadatas"][0][i]["path"],
            "distance": results["distances"][0][i],
            "metadata": results["metadatas"][0][i]
        })
    return output

def search_texts_by_image(index: MultimodalIndex, image_path: str, top_k: int = 5) -> list:
    """用图片搜文字"""
    query_emb = get_image_embedding(image_path)
    results = index.search(query_emb, top_k=top_k, modality="text")

    output = []
    for i, doc_id in enumerate(results["ids"][0]):
        output.append({
            "id": doc_id,
            "content": results["metadatas"][0][i]["content"],
            "distance": results["distances"][0][i],
            "metadata": results["metadatas"][0][i]
        })
    return output

def hybrid_search(index: MultimodalIndex, query, query_type: str, top_k: int = 5, threshold: float = 0.3):
    """混合检索，支持相似度阈值过滤"""
    if query_type == "text":
        query_emb = get_text_embedding(query)
    else:
        query_emb = get_image_embedding(query)

    results = index.search(query_emb, top_k=top_k)

    # 按阈值过滤（cosine distance，越小越相似）
    filtered = []
    for i, doc_id in enumerate(results["ids"][0]):
        if results["distances"][0][i] <= threshold:
            filtered.append({
                "id": doc_id,
                "metadata": results["metadatas"][0][i],
                "distance": results["distances"][0][i]
            })
    return filtered

# 使用示例
text_results = search_images_by_text(index, "红色碎花连衣裙")
for r in text_results:
    print(f"图片: {r['path']}, 距离: {r['distance']:.4f}")

image_results = search_texts_by_image(index, "product_001.jpg")
for r in image_results:
    print(f"文本: {r['content']}, 距离: {r['distance']:.4f}")
```

**要点**：
- ChromaDB 的 cosine distance = 1 - cosine_similarity，值越小表示越相似
- 必须设置相似度阈值过滤不相关结果，否则 top-k 中可能包含噪声
- 常见错误：混淆 distance 和 similarity 的方向，导致排序反了

### 练习三：管理题——实现索引管理功能

**思路**：在 MultimodalIndex 基础上增加增删改查和统计功能，支持索引的完整生命周期管理。

**答案**：
```python
class ManagedMultimodalIndex(MultimodalIndex):
    """带管理功能的多模态索引"""

    def add_batch_images(self, image_paths: list, metadata_list: list = None):
        """批量添加图片"""
        embeddings = []
        ids = []
        metadatas = []

        for i, path in enumerate(image_paths):
            emb = get_image_embedding(path)
            meta = metadata_list[i] if metadata_list else {}
            embeddings.append(emb)
            ids.append(f"img_{hash(path)}")
            metadatas.append({"type": "image", "path": path, **meta})

        self.collection.add(ids=ids, embeddings=embeddings, metadatas=metadatas)
        print(f"批量添加 {len(image_paths)} 张图片")

    def delete_by_type(self, modality: str):
        """按类型删除"""
        results = self.collection.get(where={"type": modality})
        if results["ids"]:
            self.collection.delete(ids=results["ids"])
            print(f"删除 {len(results['ids'])} 条 {modality} 类型记录")

    def update_metadata(self, doc_id: str, new_metadata: dict):
        """更新元数据"""
        self.collection.update(ids=[doc_id], metadatas=[new_metadata])

    def get_stats(self) -> dict:
        """获取索引统计"""
        total = self.collection.count()
        image_count = len(self.collection.get(where={"type": "image"})["ids"])
        text_count = len(self.collection.get(where={"type": "text"})["ids"])
        return {
            "total": total,
            "images": image_count,
            "texts": text_count
        }

    def rebuild_index(self, persist_dir: str = "./chroma_db"):
        """重建索引"""
        self.client.delete_collection(self.collection.name)
        self.collection = self.client.get_or_create_collection(
            name="multimodal",
            metadata={"hnsw:space": "cosine"}
        )
        print("索引已重建")

# 使用示例
index = ManagedMultimodalIndex()
index.add_batch_images(
    ["img1.jpg", "img2.jpg", "img3.jpg"],
    [{"category": "A"}, {"category": "B"}, {"category": "A"}]
)
stats = index.get_stats()
print(f"索引统计: 总计 {stats['total']}, 图片 {stats['images']}, 文本 {stats['texts']}")
```

**要点**：
- 批量操作比逐条添加效率高得多，减少网络往返
- 删除操作需要先查询再删除（ChromaDB 不支持直接按条件删除）
- 常见错误：不检查 doc_id 是否存在就调用 update，静默失败导致数据不一致
