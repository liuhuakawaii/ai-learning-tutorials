## 四、缓存策略

```python
class MultimodalCache:
    """多模态缓存"""
    
    def __init__(self):
        self.cache = {}
    
    def get(self, key: str):
        """获取缓存"""
        return self.cache.get(key)
    
    def set(self, key: str, value, ttl: int = 3600):
        """设置缓存"""
        self.cache[key] = {
            "value": value,
            "expires": time.time() + ttl
        }
    
    def get_or_compute(self, key: str, compute_fn):
        """获取或计算"""
        cached = self.get(key)
        if cached and cached["expires"] > time.time():
            return cached["value"]
        
        value = compute_fn()
        self.set(key, value)
        return value
```

## 常见误区

1. **不做 Token 估算就上线**：多模态 API 的 token 消耗远高于纯文本，不做成本预估就上线可能导致月度账单远超预算。
2. **所有任务都用最贵的模型**：简单的文本分类用 GPT-4o 是浪费，应该按任务复杂度分级使用模型。
3. **不做缓存**：相同查询每次都调 API，浪费大量成本和延迟，简单的缓存策略就能节省 50%+ 的 API 调用。
4. **只关注延迟不关注吞吐量**：优化了单次请求的延迟，但没有考虑并发场景下的吞吐量，高并发时系统会崩溃。

## 工程建议

1. **上线前做成本预算**：估算每种 API 调用的 token 消耗 × 单价 × 预期调用量，设置月度预算告警。
2. **按任务复杂度分级用模型**：简单任务用 GPT-4o-mini，复杂推理用 GPT-4o，代码任务用 Claude，按需选择最经济的模型。
3. **实现多级缓存**：内存缓存（热点数据）→ Redis 缓存（共享数据）→ 语义缓存（相似查询），逐级扩大缓存覆盖范围。
4. **建立性能监控看板**：实时监控 API 调用量、token 消耗、延迟分布、缓存命中率、错误率，设置告警阈值。

### 练习一：成本题——计算多模态调用的成本

**思路**：根据 OpenAI 官方定价，分别计算文本、图片、音频的 Token 消耗和对应费用，支持批量请求的成本预估和告警。

**答案**：
```python
from dataclasses import dataclass
from typing import Optional

@dataclass
class PricingConfig:
    """定价配置"""
    text_input_per_1k: float = 0.0025    # GPT-4o text input per 1K tokens
    text_output_per_1k: float = 0.01     # GPT-4o text output per 1K tokens
    image_low_res: float = 85            # tokens per image (low res)
    image_high_res: float = 170          # tokens per image (high res)
    audio_per_minute: float = 50         # tokens per minute of audio

class CostCalculator:
    """多模态成本计算器"""

    def __init__(self, pricing: PricingConfig = None):
        self.pricing = pricing or PricingConfig()

    def calculate_request_cost(
        self,
        text_input_tokens: int,
        text_output_tokens: int,
        image_count: int = 0,
        image_resolution: str = "low",
        audio_minutes: float = 0
    ) -> dict:
        """计算单次请求成本"""
        image_tokens = image_count * (
            self.pricing.image_high_res if image_resolution == "high" else self.pricing.image_low_res
        )
        audio_tokens = audio_minutes * self.pricing.audio_per_minute

        total_input_tokens = text_input_tokens + image_tokens + audio_tokens
        input_cost = total_input_tokens / 1000 * self.pricing.text_input_per_1k
        output_cost = text_output_tokens / 1000 * self.pricing.text_output_per_1k

        return {
            "breakdown": {
                "text_input_tokens": text_input_tokens,
                "image_tokens": image_tokens,
                "audio_tokens": audio_tokens,
                "total_input_tokens": total_input_tokens,
                "output_tokens": text_output_tokens
            },
            "cost": {
                "input_cost_usd": round(input_cost, 6),
                "output_cost_usd": round(output_cost, 6),
                "total_cost_usd": round(input_cost + output_cost, 6)
            }
        }

    def estimate_monthly_cost(
        self,
        daily_requests: int,
        avg_text_tokens: int = 500,
        avg_output_tokens: int = 200,
        avg_images_per_request: float = 0.5,
        avg_audio_minutes: float = 0.1
    ) -> dict:
        """估算月度成本"""
        single = self.calculate_request_cost(
            text_input_tokens=avg_text_tokens,
            text_output_tokens=avg_output_tokens,
            image_count=avg_images_per_request,
            audio_minutes=avg_audio_minutes
        )

        daily_cost = single["cost"]["total_cost_usd"] * daily_requests
        monthly_cost = daily_cost * 30

        return {
            "per_request": single["cost"]["total_cost_usd"],
            "daily_cost_usd": round(daily_cost, 2),
            "monthly_cost_usd": round(monthly_cost, 2),
            "daily_requests": daily_requests
        }

    def compare_models(self, text_input_tokens: int, text_output_tokens: int) -> dict:
        """对比不同模型的成本"""
        models = {
            "gpt-4o": {"input": 0.0025, "output": 0.01},
            "gpt-4o-mini": {"input": 0.00015, "output": 0.0006},
            "claude-3-5-sonnet": {"input": 0.003, "output": 0.015}
        }

        comparison = {}
        for model, price in models.items():
            cost = (text_input_tokens / 1000 * price["input"] +
                    text_output_tokens / 1000 * price["output"])
            comparison[model] = round(cost, 6)

        return comparison

# 使用示例
calculator = CostCalculator()

# 单次请求成本
cost = calculator.calculate_request_cost(
    text_input_tokens=300,
    text_output_tokens=200,
    image_count=2,
    image_resolution="high",
    audio_minutes=0.5
)
print(f"单次请求成本: ${cost['cost']['total_cost_usd']:.4f}")
print(f"  文本 Token: {cost['breakdown']['text_input_tokens']}")
print(f"  图片 Token: {cost['breakdown']['image_tokens']}")
print(f"  音频 Token: {cost['breakdown']['audio_tokens']}")

# 月度估算
monthly = calculator.estimate_monthly_cost(daily_requests=1000)
print(f"月度成本: ${monthly['monthly_cost_usd']:.2f}")

# 模型对比
comparison = calculator.compare_models(500, 200)
for model, cost in comparison.items():
    print(f"  {model}: ${cost:.6f}")
```

**要点**：
- 图片 Token 消耗取决于分辨率，低分辨率 85 tokens，高分辨率 170 tokens，差异巨大
- 月度预估需要考虑请求量的增长趋势，不能只用当前量线性计算
- 常见错误：忽略输出 Token 的价格（通常是输入的 2-4 倍），导致成本预估偏低

### 练习二：缓存题——实现缓存策略

**思路**：实现多级缓存（精确匹配缓存 + 语义缓存），精确缓存用哈希键快速查找，语义缓存用向量相似度匹配"差不多的"查询，两者结合最大化缓存命中率。

**答案**：
```python
import hashlib
import time
import numpy as np
from typing import Callable, Optional

class ExactCache:
    """精确匹配缓存"""

    def __init__(self, ttl: int = 3600, max_size: int = 10000):
        self.cache = {}
        self.ttl = ttl
        self.max_size = max_size
        self.hits = 0
        self.misses = 0

    def _make_key(self, query: str, params: dict = None) -> str:
        raw = query + str(sorted((params or {}).items()))
        return hashlib.md5(raw.encode()).hexdigest()

    def get(self, query: str, params: dict = None) -> Optional[dict]:
        key = self._make_key(query, params)
        if key in self.cache:
            entry = self.cache[key]
            if entry["expires"] > time.time():
                self.hits += 1
                return entry["value"]
            del self.cache[key]
        self.misses += 1
        return None

    def set(self, query: str, value: dict, params: dict = None):
        if len(self.cache) >= self.max_size:
            self._evict()
        key = self._make_key(query, params)
        self.cache[key] = {"value": value, "expires": time.time() + self.ttl}

    def _evict(self):
        """淘汰过期和最旧的条目"""
        now = time.time()
        expired = [k for k, v in self.cache.items() if v["expires"] <= now]
        for k in expired:
            del self.cache[k]
        if len(self.cache) >= self.max_size:
            oldest_key = min(self.cache, key=lambda k: self.cache[k]["expires"])
            del self.cache[oldest_key]

    @property
    def hit_rate(self):
        total = self.hits + self.misses
        return self.hits / total if total > 0 else 0

class SemanticCache:
    """语义缓存"""

    def __init__(self, similarity_threshold: float = 0.92, ttl: int = 3600):
        self.entries = []  # list of {query, embedding, value, expires}
        self.threshold = similarity_threshold
        self.ttl = ttl

    def get(self, query: str, query_embedding: list) -> Optional[dict]:
        now = time.time()
        best_match = None
        best_similarity = 0

        for entry in self.entries:
            if entry["expires"] <= now:
                continue
            similarity = self._cosine_similarity(query_embedding, entry["embedding"])
            if similarity > best_similarity:
                best_similarity = similarity
                best_match = entry

        if best_match and best_similarity >= self.threshold:
            return best_match["value"]
        return None

    def set(self, query: str, query_embedding: list, value: dict):
        self.entries.append({
            "query": query,
            "embedding": query_embedding,
            "value": value,
            "expires": time.time() + self.ttl
        })
        self._cleanup()

    def _cosine_similarity(self, a, b):
        a, b = np.array(a), np.array(b)
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))

    def _cleanup(self):
        now = time.time()
        self.entries = [e for e in self.entries if e["expires"] > now]

class MultimodalCacheManager:
    """多级缓存管理器"""

    def __init__(self, get_embedding_fn: Callable):
        self.exact_cache = ExactCache(ttl=3600)
        self.semantic_cache = SemanticCache(ttl=1800)
        self.get_embedding = get_embedding_fn

    def get(self, query: str, params: dict = None) -> Optional[dict]:
        """多级缓存查找"""
        # L1: 精确匹配
        result = self.exact_cache.get(query, params)
        if result:
            return result

        # L2: 语义匹配
        embedding = self.get_embedding(query)
        result = self.semantic_cache.get(query, embedding)
        if result:
            return result

        return None

    def set(self, query: str, value: dict, params: dict = None):
        """写入缓存"""
        self.exact_cache.set(query, value, params)
        embedding = self.get_embedding(query)
        self.semantic_cache.set(query, embedding, value)

    def get_stats(self) -> dict:
        return {
            "exact_cache": {
                "size": len(self.exact_cache.cache),
                "hit_rate": self.exact_cache.hit_rate
            },
            "semantic_cache": {
                "size": len(self.semantic_cache.entries)
            }
        }

# 使用示例
def dummy_get_embedding(text: str) -> list:
    import hashlib
    h = hashlib.md5(text.encode()).hexdigest()
    return [int(h[i:i+2], 16) / 255.0 for i in range(0, 32, 2)]

cache = MultimodalCacheManager(get_embedding_fn=dummy_get_embedding)

# 首次查询 - miss
result = cache.get("红色连衣裙怎么搭配")
print(f"首次查询: {result}")

# 写入缓存
cache.set("红色连衣裙怎么搭配", {"answer": "搭配白色鞋子..."})

# 精确命中
result = cache.get("红色连衣裙怎么搭配")
print(f"精确命中: {result}")

# 语义命中（相似查询）
result = cache.get("红色碎花裙如何搭配")
print(f"语义命中: {result}")

print(f"缓存统计: {cache.get_stats()}")
```

**要点**：
- 精确缓存响应快（O(1)），但只能匹配完全相同的查询
- 语义缓存能匹配相似查询，但需要计算 Embedding，有额外成本
- 语义相似度阈值需要根据业务调优：太低会返回不相关的缓存，太高则命中率低
- 常见错误：语义缓存不做过期清理，条目无限增长导致内存溢出

### 练习三：监控题——实现性能监控

**思路**：实现一个完整的性能监控系统，记录每次请求的 Token 消耗、延迟、成本、缓存命中率等指标，支持告警和报告生成。

**答案**：
```python
import time
import json
from collections import defaultdict
from datetime import datetime, timedelta
from typing import Optional

class PerformanceMonitor:
    """性能监控器"""

    def __init__(self, alert_thresholds: dict = None):
        self.records = []
        self.alerts = []
        self.thresholds = alert_thresholds or {
            "max_latency_ms": 5000,
            "max_daily_cost_usd": 100,
            "min_cache_hit_rate": 0.3
        }
        self.counters = defaultdict(int)

    def record_request(self, request_data: dict):
        """记录请求"""
        record = {
            "timestamp": datetime.now().isoformat(),
            "model": request_data.get("model", "unknown"),
            "text_tokens": request_data.get("text_tokens", 0),
            "image_tokens": request_data.get("image_tokens", 0),
            "audio_tokens": request_data.get("audio_tokens", 0),
            "output_tokens": request_data.get("output_tokens", 0),
            "latency_ms": request_data.get("latency_ms", 0),
            "cost_usd": request_data.get("cost_usd", 0),
            "cache_hit": request_data.get("cache_hit", False),
            "success": request_data.get("success", True),
            "error": request_data.get("error")
        }
        self.records.append(record)
        self.counters["total_requests"] += 1

        if record["cache_hit"]:
            self.counters["cache_hits"] += 1
        if not record["success"]:
            self.counters["errors"] += 1

        self._check_alerts(record)

    def _check_alerts(self, record: dict):
        """检查是否触发告警"""
        if record["latency_ms"] > self.thresholds["max_latency_ms"]:
            self.alerts.append({
                "type": "high_latency",
                "timestamp": record["timestamp"],
                "value": record["latency_ms"],
                "threshold": self.thresholds["max_latency_ms"]
            })

    def get_summary(self, period_hours: int = 24) -> dict:
        """获取监控摘要"""
        cutoff = datetime.now() - timedelta(hours=period_hours)
        recent = [r for r in self.records if datetime.fromisoformat(r["timestamp"]) > cutoff]

        if not recent:
            return {"message": "无数据"}

        latencies = [r["latency_ms"] for r in recent]
        costs = [r["cost_usd"] for r in recent]

        return {
            "period_hours": period_hours,
            "total_requests": len(recent),
            "success_rate": sum(1 for r in recent if r["success"]) / len(recent),
            "cache_hit_rate": sum(1 for r in recent if r["cache_hit"]) / len(recent),
            "latency": {
                "avg_ms": sum(latencies) / len(latencies),
                "p50_ms": sorted(latencies)[len(latencies) // 2],
                "p95_ms": sorted(latencies)[int(len(latencies) * 0.95)],
                "max_ms": max(latencies)
            },
            "cost": {
                "total_usd": sum(costs),
                "avg_usd": sum(costs) / len(costs)
            },
            "tokens": {
                "total_text": sum(r["text_tokens"] for r in recent),
                "total_image": sum(r["image_tokens"] for r in recent),
                "total_audio": sum(r["audio_tokens"] for r in recent),
                "total_output": sum(r["output_tokens"] for r in recent)
            },
            "alerts_count": len(self.alerts)
        }

    def generate_report(self, period_hours: int = 24) -> str:
        """生成监控报告"""
        summary = self.get_summary(period_hours)

        return f"""# 性能监控报告

时间范围：最近 {period_hours} 小时
生成时间：{datetime.now().strftime("%Y-%m-%d %H:%M")}

## 请求概览
- 总请求数：{summary["total_requests"]}
- 成功率：{summary["success_rate"]:.1%}
- 缓存命中率：{summary["cache_hit_rate"]:.1%}

## 延迟指标
- 平均延迟：{summary["latency"]["avg_ms"]:.0f}ms
- P50 延迟：{summary["latency"]["p50_ms"]:.0f}ms
- P95 延迟：{summary["latency"]["p95_ms"]:.0f}ms
- 最大延迟：{summary["latency"]["max_ms"]:.0f}ms

## 成本统计
- 总成本：${summary["cost"]["total_usd"]:.4f}
- 平均成本：${summary["cost"]["avg_usd"]:.6f}

## Token 消耗
- 文本 Token：{summary["tokens"]["total_text"]:,}
- 图片 Token：{summary["tokens"]["total_image"]:,}
- 音频 Token：{summary["tokens"]["total_audio"]:,}

## 告警
- 触发告警次数：{summary["alerts_count"]}
"""

# 使用示例

```

**要点**：
- P95 延迟比平均延迟更有参考意义，它反映了长尾请求的体验
- 告警阈值需要根据业务场景调整，不能用固定值
- 常见错误：只监控平均值不看分位数，平均延迟 1s 但 P95 是 10s 的情况很常见
