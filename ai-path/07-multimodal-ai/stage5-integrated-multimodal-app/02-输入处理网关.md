# 02 输入处理网关——统一处理文本、图片、语音、文档的输入

> 输入处理网关是多模态系统的入口。

## 场景引入

你的多模态系统需要处理来自不同渠道的输入：用户可能上传一张 JPG 图片、一段 MP3 语音、一份 PDF 合同，或者直接输入文字。每种输入的预处理逻辑完全不同——图片需要压缩和格式转换、语音需要降噪和采样率标准化、文档需要解析和分块。你需要设计一个统一的输入处理网关，对外暴露一致的接口，对内路由到不同的处理器。

## 学习目标

- 掌握输入处理网关的设计
- 理解不同输入的预处理方法
- 学会构建统一的输入处理层

## 六、统一网关

```python
class InputGateway:
    """输入网关"""
    
    def __init__(self):
        self.processors = {
            "text": TextProcessor(),
            "image": ImageProcessor(),
            "audio": AudioProcessor(),
            "document": DocumentProcessor()
        }
    
    def process(self, input_data) -> dict:
        """处理输入"""
        # 识别类型
        input_type = self._identify_type(input_data)
        
        # 选择处理器
        processor = self.processors.get(input_type)
        if not processor:
            raise ValueError(f"不支持的输入类型：{input_type}")
        
        # 处理
        return processor.process(input_data)
    
    def _identify_type(self, input_data) -> str:
        """识别输入类型"""
        if isinstance(input_data, str):
            if input_data.endswith((".pdf", ".docx", ".doc")):
                return "document"
            elif input_data.endswith((".jpg", ".png", ".gif")):
                return "image"
            elif input_data.endswith((".mp3", ".wav", ".m4a")):
                return "audio"
            else:
                return "text"
        return "unknown"
```

## 常见误区

1. **不做输入验证就处理**：用户上传的文件可能是损坏的、格式不支持的、或超大文件，不做前置验证会导致处理失败或资源浪费。
2. **忽略文件大小限制**：大图片、长音频、大 PDF 会消耗大量内存和处理时间，必须设置合理的文件大小限制。
3. **同步处理所有输入**：大文件的预处理（如音频降噪、PDF 解析）耗时很长，同步处理会阻塞请求，应该用异步任务队列。
4. **不做标准化输出**：不同处理器的输出格式不统一，下游需要写适配代码，应该定义统一的输出 schema。

## 工程建议

1. **前置校验拦截非法输入**：检查文件格式、大小、完整性，不合法的输入直接返回明确的错误信息，不要进入处理流程。
2. **大文件用异步处理**：超过阈值的文件（如 >10MB 图片、>5 分钟音频）走异步任务队列，返回任务 ID 让用户轮询结果。
3. **定义统一的输出 schema**：所有处理器输出相同结构的 JSON（type, content, metadata），下游统一处理。
4. **做好资源限制和熔断**：每种处理器设置并发上限和超时时间，避免单种输入占用过多资源影响其他处理器。

### 练习一：网关题——实现输入处理网关

**思路**：设计一个统一的输入网关，通过策略模式注册多种处理器，自动识别输入类型并路由到对应处理器，所有处理器输出统一的 JSON schema。

**答案**：
```python
import os
import mimetypes
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Optional

@dataclass
class StandardOutput:
    """统一输出格式"""
    type: str
    content: str
    metadata: dict
    raw_data: any = None

class BaseProcessor(ABC):
    """处理器基类"""

    @abstractmethod
    def can_process(self, input_data) -> bool:
        pass

    @abstractmethod
    def process(self, input_data) -> StandardOutput:
        pass

class TextProcessor(BaseProcessor):
    def can_process(self, input_data):
        return isinstance(input_data, str) and not self._is_file_path(input_data)

    def _is_file_path(self, s: str) -> bool:
        return os.path.exists(s) or "." in s.split("/")[-1]

    def process(self, input_data) -> StandardOutput:
        return StandardOutput(
            type="text",
            content=input_data,
            metadata={"char_count": len(input_data), "language": "zh"}
        )

class ImageProcessor(BaseProcessor):
    SUPPORTED_EXTENSIONS = {".jpg", ".jpeg", ".png", ".gif", ".webp", ".bmp"}

    def can_process(self, input_data):
        if isinstance(input_data, str):
            ext = os.path.splitext(input_data)[1].lower()
            return ext in self.SUPPORTED_EXTENSIONS
        return False

    def process(self, input_data) -> StandardOutput:
        from PIL import Image
        image = Image.open(input_data).convert("RGB")
        return StandardOutput(
            type="image",
            content=input_data,
            metadata={
                "width": image.size[0],
                "height": image.size[1],
                "format": image.format,
                "file_size_kb": os.path.getsize(input_data) / 1024
            },
            raw_data=image
        )

class AudioProcessor(BaseProcessor):
    SUPPORTED_EXTENSIONS = {".mp3", ".wav", ".m4a", ".ogg", ".flac"}

    def __init__(self):
        self._model = None

    def _get_model(self):
        if self._model is None:
            import whisper
            self._model = whisper.load_model("base")
        return self._model

    def can_process(self, input_data):
        if isinstance(input_data, str):
            ext = os.path.splitext(input_data)[1].lower()
            return ext in self.SUPPORTED_EXTENSIONS
        return isinstance(input_data, bytes)

    def process(self, input_data) -> StandardOutput:
        model = self._get_model()
        result = model.transcribe(input_data)
        return StandardOutput(
            type="audio",
            content=result["text"],
            metadata={
                "duration": result.get("duration", 0),
                "language": result.get("language", "unknown"),
                "file_size_kb": os.path.getsize(input_data) / 1024 if isinstance(input_data, str) else 0
            }
        )

class DocumentProcessor(BaseProcessor):
    SUPPORTED_EXTENSIONS = {".pdf", ".docx", ".doc", ".txt"}

    def can_process(self, input_data):
        if isinstance(input_data, str):
            ext = os.path.splitext(input_data)[1].lower()
            return ext in self.SUPPORTED_EXTENSIONS
        return False

    def process(self, input_data) -> StandardOutput:
        elements = parse_document(input_data)  # 假设已有解析函数
        text = "\n".join([e.text for e in elements if hasattr(e, "text")])
        tables = [e.text for e in elements if getattr(e, "category", "") == "Table"]
        images = [e.metadata.image_path for e in elements if getattr(e, "category", "") == "Image"]

        return StandardOutput(
            type="document",
            content=text,
            metadata={
                "page_count": len(set(getattr(e.metadata, "page_number", 0) for e in elements)),
                "table_count": len(tables),
                "image_count": len(images),
                "tables": tables,
                "images": images
            }
        )

class InputGateway:
    """输入处理网关"""

    def __init__(self):
        self.processors: list[BaseProcessor] = []
        self._register_defaults()

    def _register_defaults(self):
        self.processors.append(TextProcessor())
        self.processors.append(ImageProcessor())
        self.processors.append(AudioProcessor())
        self.processors.append(DocumentProcessor())

    def register(self, processor: BaseProcessor):
        self.processors.append(processor)

    def process(self, input_data) -> StandardOutput:
        """自动识别并处理输入"""
        for processor in self.processors:
            if processor.can_process(input_data):
                return processor.process(input_data)
        raise ValueError(f"不支持的输入类型: {type(input_data)}")

# 使用示例
gateway = InputGateway()
result = gateway.process("今天天气怎么样？")
print(f"类型: {result.type}, 内容: {result.content}")

result = gateway.process("product_photo.jpg")
print(f"类型: {result.type}, 尺寸: {result.metadata['width']}x{result.metadata['height']}")
```

**要点**：
- 所有处理器输出 `StandardOutput` 统一格式，下游不需要写适配代码
- 模型（如 Whisper）延迟加载，避免网关初始化时加载所有模型
- 常见错误：文件类型判断只看扩展名不检查文件实际内容，可能被伪造扩展名欺骗

### 练习二：处理器题——实现一个图片处理器

**思路**：实现一个功能完整的图片处理器，支持格式转换、尺寸调整、质量压缩、元数据提取和错误处理。

**答案**：
```python
from PIL import Image, ExifTags
import io
import os

class AdvancedImageProcessor:
    """高级图片处理器"""

    MAX_SIZE = (2048, 2048)
    QUALITY = 85
    MAX_FILE_SIZE_MB = 10

    def process(self, image_input) -> dict:
        """处理图片的完整流程"""
        # 1. 加载图片
        image = self._load_image(image_input)

        # 2. 校验
        self._validate(image, image_input)

        # 3. 提取元数据
        metadata = self._extract_metadata(image)

        # 4. 预处理
        image = self._preprocess(image)

        # 5. 转为标准格式
        buffer = io.BytesIO()
        image.save(buffer, format="JPEG", quality=self.QUALITY)
        buffer.seek(0)

        return {
            "type": "image",
            "data": buffer.getvalue(),
            "metadata": {
                **metadata,
                "processed_size": image.size,
                "output_format": "JPEG",
                "output_size_kb": len(buffer.getvalue()) / 1024
            }
        }

    def _load_image(self, image_input) -> Image.Image:
        """加载图片"""
        if isinstance(image_input, str):
            if image_input.startswith("http"):
                import requests
                response = requests.get(image_input, timeout=10)
                response.raise_for_status()
                return Image.open(io.BytesIO(response.content)).convert("RGB")
            else:
                return Image.open(image_input).convert("RGB")
        elif isinstance(image_input, bytes):
            return Image.open(io.BytesIO(image_input)).convert("RGB")
        elif isinstance(image_input, Image.Image):
            return image_input.convert("RGB")
        raise ValueError(f"不支持的图片输入类型: {type(image_input)}")

    def _validate(self, image: Image.Image, source):
        """校验图片"""
        # 检查尺寸
        w, h = image.size
        if w < 32 or h < 32:
            raise ValueError(f"图片尺寸过小: {w}x{h}，最小 32x32")
        if w > 8192 or h > 8192:
            raise ValueError(f"图片尺寸过大: {w}x{h}，最大 8192x8192")

        # 检查文件大小
        if isinstance(source, str) and os.path.exists(source):
            size_mb = os.path.getsize(source) / (1024 * 1024)
            if size_mb > self.MAX_FILE_SIZE_MB:
                raise ValueError(f"文件过大: {size_mb:.1f}MB，最大 {self.MAX_FILE_SIZE_MB}MB")

    def _preprocess(self, image: Image.Image) -> Image.Image:
        """预处理：调整尺寸、去除 EXIF 旋转"""
        # 处理 EXIF 旋转
        try:
            exif = image._getexif()
            if exif:
                for tag, value in exif.items():
                    if ExifTags.TAGS.get(tag) == "Orientation":
                        if value == 3:
                            image = image.rotate(180, expand=True)
                        elif value == 6:
                            image = image.rotate(270, expand=True)
                        elif value == 8:
                            image = image.rotate(90, expand=True)
        except (AttributeError, KeyError):
            pass

        # 调整尺寸
        image.thumbnail(self.MAX_SIZE, Image.Resampling.LANCZOS)
        return image

    def _extract_metadata(self, image: Image.Image) -> dict:
        """提取图片元数据"""
        metadata = {
            "original_size": image.size,
            "mode": image.mode,
            "format": image.format
        }

        # 提取 EXIF 信息
        try:
            exif = image._getexif()
            if exif:
                for tag, value in exif.items():
                    tag_name = ExifTags.TAGS.get(tag, tag)
                    if tag_name in ("Make", "Model", "DateTime", "GPSInfo"):
                        metadata[tag_name] = str(value)
        except (AttributeError, KeyError):
            pass

        return metadata

# 使用示例
processor = AdvancedImageProcessor()
result = processor.process("product_photo.jpg")
print(f"处理后尺寸: {result['metadata']['processed_size']}")
print(f"输出大小: {result['metadata']['output_size_kb']:.1f}KB")
```

**要点**：
- 图片必须统一转为 RGB 模式，RGBA 图片直接保存 JPEG 会报错
- EXIF 旋转信息必须处理，否则手机拍的图片可能显示方向不对
- 常见错误：不做尺寸校验直接处理超大图片，导致内存溢出

### 练习三：统一题——实现统一的输入处理接口

**思路**：在网关基础上实现统一的异步处理接口，支持批量输入、超时控制、错误隔离和处理结果缓存。

**答案**：
```python
import asyncio
import hashlib
import time
from typing import Any, Dict, List

class UnifiedInputProcessor:
    """统一输入处理器"""

    def __init__(self, gateway: InputGateway):
        self.gateway = gateway
        self._cache: Dict[str, StandardOutput] = {}
        self._cache_ttl = 3600

    def process_sync(self, input_data: Any) -> StandardOutput:
        """同步处理"""
        cache_key = self._compute_cache_key(input_data)
        cached = self._get_cache(cache_key)
        if cached:
            return cached

        result = self.gateway.process(input_data)
        self._set_cache(cache_key, result)
        return result

    async def process_async(self, input_data: Any, timeout: float = 30.0) -> StandardOutput:
        """异步处理"""
        cache_key = self._compute_cache_key(input_data)
        cached = self._get_cache(cache_key)
        if cached:
            return cached

        try:
            result = await asyncio.wait_for(
                asyncio.to_thread(self.gateway.process, input_data),
                timeout=timeout
            )
            self._set_cache(cache_key, result)
            return result
        except asyncio.TimeoutError:
            return StandardOutput(
                type="error",
                content=f"处理超时（{timeout}s）",
                metadata={"input_type": type(input_data).__name__}
            )

    async def process_batch(self, inputs: List[Any], max_concurrent: int = 5) -> List[StandardOutput]:
        """批量处理"""
        semaphore = asyncio.Semaphore(max_concurrent)

        async def process_one(input_data):
            async with semaphore:
                return await self.process_async(input_data)

        tasks = [process_one(inp) for inp in inputs]
        return await asyncio.gather(*tasks, return_exceptions=False)

    def _compute_cache_key(self, input_data: Any) -> str:
        """计算缓存键"""
        if isinstance(input_data, str):
            if os.path.exists(input_data):
                # 文件路径：用路径 + 修改时间
                mtime = os.path.getmtime(input_data)
                return hashlib.md5(f"{input_data}:{mtime}".encode()).hexdigest()
            return hashlib.md5(input_data.encode()).hexdigest()
        return hashlib.md5(str(input_data).encode()).hexdigest()

    def _get_cache(self, key: str) -> Optional[StandardOutput]:
        """获取缓存"""
        if key in self._cache:
            entry = self._cache[key]
            if entry.get("expires", 0) > time.time():
                return entry["value"]
            del self._cache[key]
        return None

    def _set_cache(self, key: str, value: StandardOutput):
        """设置缓存"""
        self._cache[key] = {"value": value, "expires": time.time() + self._cache_ttl}

# 使用示例
gateway = InputGateway()
processor = UnifiedInputProcessor(gateway)

# 同步
result = processor.process_sync("你好")
print(f"同步结果: {result.content}")

# 异步批量
async def main():
    inputs = ["你好", "photo1.jpg", "photo2.jpg", "audio.mp3"]
    results = await processor.process_batch(inputs, max_concurrent=2)
    for i, r in enumerate(results):
        if isinstance(r, Exception):
            print(f"  [{i}] 错误: {r}")
        else:
            print(f"  [{i}] 类型: {r.type}, 内容: {r.content[:50]}")

asyncio.run(main())
```

**要点**：
- 缓存键需要考虑文件内容变化（用修改时间），避免缓存过期数据
- 批量处理用信号量控制并发，避免同时处理太多大文件导致内存溢出
- 常见错误：不处理异步超时，某个大文件卡住会阻塞整个批量任务
