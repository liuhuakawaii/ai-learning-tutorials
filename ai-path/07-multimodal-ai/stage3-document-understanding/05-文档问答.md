# 05 文档问答——基于文档内容的智能问答系统

> 文档问答是文档处理的核心应用场景。

## 场景引入

你要为公司内部构建一个"智能 HR 问答系统"：员工上传员工手册后，可以用自然语言提问，如"年假有多少天？""试用期可以请假吗？""报销流程是什么？"系统需要从文档中找到相关内容并给出准确回答，同时标注引用来源。这看似简单的 RAG 应用，实际面临的问题包括：文档分块不合理导致检索不到相关内容、相似问题的表述差异大导致匹配失败、回答引用了错误的文档段落——文档问答的准确率直接取决于每个环节的工程质量。

## 学习目标

- 掌握文档问答系统的设计
- 理解检索增强生成（RAG）在文档问答中的应用
- 学会构建文档问答系统

---

## 一、系统架构

```
文档问答系统：

文档 → 解析 → 分块 → 向量化 → 存储
                                │
用户问题 → 向量化 → 检索 → LLM 生成 → 回答
```

---

## 二、核心实现

```python
class DocumentQA:
    """文档问答"""
    
    def __init__(self):
        self.chunks = []
        self.embeddings = []
        self.client = OpenAI()
    
    def add_document(self, doc_path: str):
        """添加文档"""
        # 解析文档
        elements = parse_document(doc_path)
        
        # 分块
        chunks = semantic_chunking(elements)
        
        # 向量化
        for chunk in chunks:
            embedding = self._get_embedding(chunk)
            self.chunks.append(chunk)
            self.embeddings.append(embedding)
    
    def ask(self, question: str) -> str:
        """提问"""
        # 检索相关块
        question_embedding = self._get_embedding(question)
        relevant_chunks = self._retrieve(question_embedding, top_k=3)
        
        # 生成答案
        context = "\n".join(relevant_chunks)
        answer = self._generate_answer(question, context)
        
        return answer
    
    def _get_embedding(self, text: str) -> list:
        """获取向量"""
        response = self.client.embeddings.create(
            model="text-embedding-3-small",
            input=text
        )
        return response.data[0].embedding
    
    def _retrieve(self, query_embedding: list, top_k: int) -> list:
        """检索"""
        similarities = []
        for i, emb in enumerate(self.embeddings):
            sim = cosine_similarity(query_embedding, emb)
            similarities.append((sim, i))
        
        similarities.sort(reverse=True)
        return [self.chunks[i] for _, i in similarities[:top_k]]
    
    def _generate_answer(self, question: str, context: str) -> str:
        """生成答案"""
        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[{
                "role": "user",
                "content": f"基于以下文档内容回答问题：\n\n{context}\n\n问题：{question}"
            }]
        )
        return response.choices[0].message.content
```

## 常见误区

1. **只用向量检索不用关键词检索**：纯向量检索对精确匹配（如"年假5天"中的"5天"）效果不好，混合检索（向量 + 关键词）能显著提高准确率。
2. **不做引用溯源**：用户需要知道回答来自文档的哪个段落，不做溯源会降低用户信任度，也无法验证回答的准确性。
3. **忽略查询改写**：用户的口语化问题（"请假咋办"）和文档中的书面表述（"请假流程"）差异大，不做查询改写会导致检索失败。
4. **top_k 设置不合理**：top_k 太小会漏掉相关段落，太大会引入噪音，需要根据文档特点调优。

## 工程建议

1. **实现混合检索**：向量检索 + BM25 关键词检索，用 RRF（Reciprocal Rank Fusion）合并结果，兼顾语义匹配和精确匹配。
2. **每个回答必须带引用**：标注回答来自哪个文档的哪个段落，方便用户验证和复核。
3. **做查询改写和扩展**：用 LLM 将用户的口语化问题改写为更接近文档表述的查询，或生成多个变体查询提高召回率。
4. **建立评估数据集**：收集典型问题和标准答案，定期评估问答准确率，发现和修复 bad case。

### 练习一：构建一个文档问答系统

**思路**：实现完整的 RAG Pipeline：文档解析 → 分块 → 向量化 → 检索 → LLM 生成，支持引用溯源。

**答案**：

```python
"""文档问答系统：完整 RAG Pipeline。"""
from openai import OpenAI
from unstructured.partition.pdf import partition_pdf
import numpy as np
import json


class DocumentQASystem:
    """基于 RAG 的文档问答系统。"""

    def __init__(self):
        self.client = OpenAI()
        self.chunks = []
        self.embeddings = []

    def ingest(self, file_path: str, chunk_size: int = 800):
        """导入文档：解析 → 分块 → 向量化。"""
        ext = file_path.rsplit(".", 1)[-1].lower()
        if ext == "pdf":
            elements = partition_pdf(file_path, strategy="hi_res")
        else:
            raise ValueError(f"暂不支持 {ext} 格式")

        # 语义分块
        self.chunks = self._chunk(elements, chunk_size)

        # 批量向量化
        texts = [c["text"] for c in self.chunks]
        resp = self.client.embeddings.create(model="text-embedding-3-small", input=texts)
        self.embeddings = [item.embedding for item in resp.data]

        return len(self.chunks)

    def query(self, question: str, top_k: int = 3) -> dict:
        """提问并返回答案和来源。"""
        # 检索
        q_emb = self._embed(question)
        indices = self._cosine_search(q_emb, top_k)
        sources = [self.chunks[i] for i in indices]

        # 生成
        context = "\n\n".join(f"[来源 {s['index'] + 1}] {s['text']}" for s in sources)
        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[
                {"role": "system", "content": "根据文档内容回答，标注引用来源编号。无相关信息则说明。"},
                {"role": "user", "content": f"文档：\n{context}\n\n问题：{question}"},
            ],
        )

        return {
            "answer": response.choices[0].message.content,
            "sources": [{"index": s["index"], "preview": s["text"][:150]} for s in sources],
        }

    def _chunk(self, elements, max_size):
        chunks, buf, size = [], [], 0
        for el in elements:
            t = el.text.strip() if el.text else ""
            if not t:
                continue
            if size + len(t) > max_size and buf:
                chunks.append({"text": "\n".join(buf), "index": len(chunks)})
                buf, size = [t], len(t)
            else:
                buf.append(t)
                size += len(t)
        if buf:
            chunks.append({"text": "\n".join(buf), "index": len(chunks)})
        return chunks

    def _embed(self, text):
        return self.client.embeddings.create(model="text-embedding-3-small", input=text).data[0].embedding

    def _cosine_search(self, q_emb, top_k):
        q = np.array(q_emb)
        sims = [(np.dot(q, np.array(e)) / (np.linalg.norm(q) * np.linalg.norm(np.array(e))), i)
                for i, e in enumerate(self.embeddings)]
        sims.sort(reverse=True)
        return [i for _, i in sims[:top_k]]


if __name__ == "__main__":
    qa = DocumentQASystem()
    qa.ingest("employee_handbook.pdf")
    result = qa.query("试用期员工可以请假吗？")
    print(f"答案: {result['answer']}")
    for s in result["sources"]:
        print(f"  来源 {s['index'] + 1}: {s['preview']}...")
```

**要点**：
- `ingest` 阶段的向量化应批量调用 API，逐条调用会产生大量网络开销
- prompt 中必须约束 LLM 只用文档内容回答，否则会用训练数据中的知识"编造"看似合理但文档中没有的答案
- 检索不到相关内容时应明确告知用户，而不是生成模糊的回答
- 常见错误：分块太大（>2000字）导致检索精度下降，分块太小（<100字）导致语义不完整

### 练习二：优化检索准确率

**思路**：实现混合检索（向量 + BM25 关键词），用 RRF 合并结果，加入查询改写提升召回率。

**答案**：

```python
"""混合检索优化：向量检索 + BM25 + 查询改写。"""
from openai import OpenAI
import numpy as np
from rank_bm25 import BM25Okapi
import jieba


class HybridRetriever:
    """混合检索器：向量 + BM25 + 查询改写。"""

    def __init__(self, chunks: list, embeddings: list):
        self.client = OpenAI()
        self.chunks = chunks
        self.embeddings = embeddings

        # 初始化 BM25
        tokenized = [list(jieba.cut(c["text"])) for c in chunks]
        self.bm25 = BM25Okapi(tokenized)

    def search(self, query: str, top_k: int = 5) -> list:
        """混合检索。"""
        # 1. 查询改写
        expanded_queries = self._rewrite_query(query)

        # 2. 向量检索
        vector_results = set()
        for q in expanded_queries:
            q_emb = self._embed(q)
            indices = self._vector_search(q_emb, top_k)
            vector_results.update(indices)

        # 3. BM25 检索
        bm25_results = set()
        for q in expanded_queries:
            tokens = list(jieba.cut(q))
            scores = self.bm25.get_scores(tokens)
            top_indices = np.argsort(scores)[::-1][:top_k]
            bm25_results.update(top_indices)

        # 4. RRF 合并
        rrf_scores = {}
        k = 60  # RRF 常数

        for rank, idx in enumerate(sorted(vector_results)):
            rrf_scores[idx] = rrf_scores.get(idx, 0) + 1 / (k + rank + 1)

        for rank, idx in enumerate(sorted(bm25_results)):
            rrf_scores[idx] = rrf_scores.get(idx, 0) + 1 / (k + rank + 1)

        # 排序取 top_k
        sorted_indices = sorted(rrf_scores.keys, key=lambda i: rrf_scores[i], reverse=True)[:top_k]
        return [{"chunk": self.chunks[i], "score": rrf_scores[i]} for i in sorted_indices]

    def _rewrite_query(self, query: str) -> list:
        """查询改写：扩展为多个变体。"""
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{
                "role": "user",
                "content": f"将以下问题改写为 3 个不同表述，每行一个，不要编号：\n{query}",
            }],
            temperature=0.7,
        )
        variants = response.choices[0].message.content.strip().split("\n")
        return [query] + [v.strip() for v in variants if v.strip()]

    def _embed(self, text):
        return self.client.embeddings.create(
            model="text-embedding-3-small", input=text
        ).data[0].embedding

    def _vector_search(self, q_emb, top_k):
        q = np.array(q_emb)
        sims = []
        for i, e in enumerate(self.embeddings):
            sim = np.dot(q, np.array(e)) / (np.linalg.norm(q) * np.linalg.norm(np.array(e)))
            sims.append((sim, i))
        sims.sort(reverse=True)
        return [i for _, i in sims[:top_k]]


if __name__ == "__main__":
    # 假设已有 chunks 和 embeddings
    retriever = HybridRetriever(chunks=[], embeddings=[])
    results = retriever.search("请假咋办", top_k=5)
    for r in results:
        print(f"  分数: {r['score']:.4f} | {r['chunk']['text'][:80]}...")
```

**要点**：
- 纯向量检索对精确匹配效果差（如"年假5天"中的"5天"），BM25 可弥补这一短板
- RRF（Reciprocal Rank Fusion）合并两个排序列表，k=60 是论文推荐值
- 查询改写可将口语化问题（"请假咋办"）扩展为书面表述（"请假流程"），显著提升召回率
- 常见错误：只用向量检索不用关键词检索，用户问具体数字或名称时检索不到相关内容

### 练习三：创建 Web 界面

**思路**：用 Gradio 搭建文档问答的 Web 界面，支持文件上传、提问、展示答案和引用来源。

**答案**：

```python
"""文档问答 Web 界面：Gradio 实现。"""
import gradio as gr
from document_qa_system import DocumentQASystem


qa_system = DocumentQASystem()
current_doc = None


def upload_document(file):
    """上传并处理文档。"""
    global current_doc
    if file is None:
        return "请上传文档"

    try:
        chunk_count = qa_system.ingest(file.name)
        current_doc = file.name
        return f"文档已加载：{file.name}（共 {chunk_count} 个段落）"
    except Exception as e:
        return f"文档处理失败: {str(e)}"


def ask_question(question):
    """提问并返回结果。"""
    if not current_doc:
        return "请先上传文档", ""

    if not question.strip():
        return "请输入问题", ""

    try:
        result = qa_system.query(question)

        # 格式化来源
        sources_text = ""
        for s in result["sources"]:
            sources_text += f"**来源 {s['index'] + 1}**\n> {s['preview']}...\n\n"

        return result["answer"], sources_text
    except Exception as e:
        return f"查询失败: {str(e)}", ""


with gr.Blocks(title="文档智能问答") as demo:
    gr.Markdown("# 文档智能问答系统")
    gr.Markdown("上传文档后，用自然语言提问，系统会从文档中找到相关内容并回答。")

    with gr.Row():
        with gr.Column(scale=1):
            file_input = gr.File(label="上传文档", file_types=[".pdf"])
            upload_btn = gr.Button("处理文档", variant="primary")
            upload_status = gr.Textbox(label="状态", interactive=False)

        with gr.Column(scale=2):
            question_input = gr.Textbox(label="输入问题", placeholder="例如：公司的年假政策是什么？")
            ask_btn = gr.Button("提问", variant="primary")
            answer_output = gr.Markdown(label="回答")
            sources_output = gr.Markdown(label="引用来源")

    upload_btn.click(fn=upload_document, inputs=file_input, outputs=upload_status)
    ask_btn.click(fn=ask_question, inputs=question_input, outputs=[answer_output, sources_output])

if __name__ == "__main__":
    demo.launch(server_name="0.0.0.0", server_port=7860)
```

**要点**：
- 引用来源必须展示给用户，方便验证答案准确性，这也是 RAG 系统的核心价值
- 文件上传后应做持久化存储，用户刷新页面后不需要重新上传
- 用 `gr.Markdown` 展示答案比 `gr.Textbox` 效果好，支持格式化和引用高亮
- 常见错误：每次提问都重新解析文档，应该在上传时一次性处理并缓存结果
