## 分词器

索引和查询必须使用同一个分词器，否则会出现不匹配问题——这是搜索引擎最常见的 bug 来源。

```python
import re, json
from collections import defaultdict
from dataclasses import dataclass, field

DEFAULT_STOPWORDS = {
    "的", "了", "是", "在", "我", "有", "和", "就", "不", "人", "都", "一",
    "很", "到", "说", "要", "去", "你", "会", "着", "没有", "好", "自己",
    "the", "a", "an", "is", "are", "was", "in", "on", "at", "to",
}


class Tokenizer:
    def __init__(self, stopwords=None, custom_words=None):
        self.stopwords = stopwords or DEFAULT_STOPWORDS
        try:
            import jieba
            self.jieba = jieba
            for w in (custom_words or []):
                jieba.add_word(w)
        except ImportError:
            self.jieba = None

    def tokenize(self, text):
        tokens = (self.jieba.cut_for_search(text) if self.jieba
                  else re.findall(r"[\u4e00-\u9fff]+|[a-zA-Z0-9]+", text))
        return [t.strip().lower() for t in tokens
                if t.strip() and t.strip().lower() not in self.stopwords]
```

## 倒排索引

```python
@dataclass
class Posting:
    doc_id: int
    term_freq: int = 0
    positions: list[int] = field(default_factory=list)


class InvertedIndex:
    def __init__(self):
        self.index = defaultdict(list)
        self.doc_store = {}
        self.doc_count = 0
        self.doc_lengths = {}

    def add_document(self, doc_id, terms, metadata=None):
        self.doc_store[doc_id] = metadata or {}
        self.doc_count += 1
        self.doc_lengths[doc_id] = len(terms)
        term_positions = defaultdict(list)
        for pos, term in enumerate(terms):
            term_positions[term].append(pos)
        for term, positions in term_positions.items():
            self.index[term].append(Posting(doc_id, len(positions), positions))

    def get_postings(self, term):
        return self.index.get(term, [])
```

## 布尔查询

```python
class BooleanRetriever:
    def __init__(self, index):
        self.index = index

    def search_and(self, terms):
        if not terms:
            return []
        lists = [sorted(p.doc_id for p in self.index.get_postings(t)) for t in terms]
        if any(not pl for pl in lists):
            return []
        result = lists[0]
        for pl in lists[1:]:
            result = self._merge(result, pl)
            if not result:
                break
        return result

    def search_or(self, terms):
        ids = set()
        for t in terms:
            for p in self.index.get_postings(t):
                ids.add(p.doc_id)
        return sorted(ids)

    def search_not(self, exclude, base=None):
        if base is None:
            base = list(self.index.doc_store.keys())
        exclude_ids = {p.doc_id for t in exclude for p in self.index.get_postings(t)}
        return [d for d in base if d not in exclude_ids]

    def _merge(self, a, b):
        result, i, j = [], 0, 0
        while i < len(a) and j < len(b):
            if a[i] == b[j]:
                result.append(a[i]); i += 1; j += 1
            elif a[i] < b[j]:
                i += 1
            else:
                j += 1
        return result
```

## 查询解析器

```python
class QueryParser:
    def __init__(self, tokenizer):
        self.tokenizer = tokenizer

    def parse(self, query):
        include, exclude = [], []
        phrases = re.findall(r'"([^"]+)"', query)
        remaining = re.sub(r'"[^"]*"', '', query)
        for token in remaining.split():
            if token.startswith("-") and len(token) > 1:
                exclude.extend(self.tokenizer.tokenize(token[1:]))
            elif token.upper() != "OR":
                include.extend(self.tokenizer.tokenize(token))
        for phrase in phrases:
            include.extend(self.tokenizer.tokenize(phrase))
        return {"include": include, "exclude": exclude, "phrases": phrases}
```

## 主类

```python
class SearchEngine:
    def __init__(self, stopwords=None, custom_words=None):
        self.tokenizer = Tokenizer(stopwords, custom_words)
        self.index = InvertedIndex()
        self.retriever = BooleanRetriever(self.index)
        self.parser = QueryParser(self.tokenizer)

    def add_document(self, doc_id, title, body):
        terms = self.tokenizer.tokenize(f"{title} {body}")
        self.index.add_document(doc_id, terms, {"title": title, "body": body})

    def search(self, query):
        parsed = self.parser.parse(query)
        include, exclude = parsed["include"], parsed["exclude"]
        if not include and not exclude:
            return []
        doc_ids = (self.retriever.search_and(include) if include
                   else list(self.index.doc_store.keys()))
        if exclude:
            doc_ids = self.retriever.search_not(exclude, doc_ids)
        return [{"doc_id": d, "title": self.index.doc_store[d].get("title", ""),
                 "snippet": self.index.doc_store[d].get("body", "")[:100]}
                for d in doc_ids]

    def save(self, path):
        data = {"doc_count": self.index.doc_count,
                "index": {t: [{"doc_id": p.doc_id, "tf": p.term_freq,
                               "pos": p.positions} for p in ps]
                          for t, ps in self.index.index.items()},
                "doc_store": {str(k): v for k, v in self.index.doc_store.items()},
                "doc_lengths": {str(k): v for k, v in self.index.doc_lengths.items()}}
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    @classmethod
    def load(cls, path, stopwords=None):
        engine = cls(stopwords)
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        engine.index.doc_count = data["doc_count"]
        for term, postings in data["index"].items():
            engine.index.index[term] = [Posting(p["doc_id"], p["tf"], p["pos"])
                                        for p in postings]
        engine.index.doc_store = {int(k): v for k, v in data["doc_store"].items()}
        engine.index.doc_lengths = {int(k): v for k, v in data.get("doc_lengths", {}).items()}
        return engine
```

## 扩展：BM25 排序

布尔查询只返回"包含/不包含"的结果，没有排序。BM25 综合词频和逆文档频率评估相关性：

```python
import math

def bm25_score(query_terms, doc_id, index, k1=1.5, b=0.75):
    score = 0.0
    avg_dl = sum(index.doc_lengths.values()) / index.doc_count
    dl = index.doc_lengths.get(doc_id, 0)
    for term in query_terms:
        tf = next((p.term_freq for p in index.get_postings(term)
                   if p.doc_id == doc_id), 0)
        if tf == 0:
            continue
        df = len(index.get_postings(term))
        idf = math.log((index.doc_count - df + 0.5) / (df + 0.5) + 1)
        score += idf * (tf * (k1 + 1)) / (tf + k1 * (1 - b + b * dl / avg_dl))
    return score
```

### 练习二：添加搜索建议

当查询词不在索引中时，用编辑距离推荐相似词项，返回距离最小的前 3 个。

```python
def suggest(engine, term, max_suggestions=3):
    if term in engine.index.index:
        return [term]
    distances = []
    for t in engine.index.index:
        if abs(len(t) - len(term)) > 2:
            continue
        d = levenshtein_distance(term, t)
        if d <= 2:
            distances.append((d, t))
    distances.sort()
    return [t for _, t in distances[:max_suggestions]]
```

### 练习一

```python
def search_phrase(self, phrase_tokens):
    if not phrase_tokens:
        return []
    first = {p.doc_id: p.positions
             for p in self.index.get_postings(phrase_tokens[0])}
    result = []
    for doc_id, positions in first.items():
        for start in positions:
            match = True
            for offset in range(1, len(phrase_tokens)):
                postings = {p.doc_id: p.positions
                            for p in self.index.get_postings(phrase_tokens[offset])}
                if doc_id not in postings or (start + offset) not in postings[doc_id]:
                    match = False
                    break
            if match:
                result.append(doc_id)
                break
    return sorted(result)
```

### 练习二

```python
def levenshtein_distance(s1, s2):
    if len(s1) < len(s2):
        return levenshtein_distance(s2, s1)
    if len(s2) == 0:
        return len(s1)
    prev = range(len(s2) + 1)
    for i, c1 in enumerate(s1):
        curr = [i + 1]
        for j, c2 in enumerate(s2):
            curr.append(min(prev[j+1] + 1, curr[j] + 1, prev[j] + (c1 != c2)))
        prev = curr
    return prev[-1]
```
