# 阶段实战：基于 ES 的完整搜索服务

## 场景引入

经过前面 5 课的学习，你已经掌握了 Elasticsearch 的架构、查询执行、索引策略、聚合分析和性能优化。现在是时候把这些知识整合起来，构建一个完整的搜索服务了。

本课将带你从零开始构建一个电商搜索服务：创建索引、导入数据、配置中文分词、实现搜索查询、完成聚合分析，最终用 FastAPI 暴露 HTTP 接口。这不是一个 demo——而是一个可以直接部署到生产环境的搜索服务骨架，包含了错误处理、日志记录、健康检查、搜索建议等生产级功能。

在真实项目中，搜索服务往往是最先被用户感知的系统。用户打开电商 APP，第一件事就是搜索商品。搜索体验的好坏直接影响转化率。据 Amazon 统计，搜索用户的购买转化率是浏览用户的 5-10 倍。所以搜索服务不仅要在功能上正确，还要在性能上优秀。

```python
                "original_price": {"type": "float"},

                "brand": {"type": "keyword"},
                "tags": {"type": "keyword"},
                "stock": {"type": "integer"},
                "rating": {"type": "float"},
                "sales_count": {"type": "integer"},
                "in_stock": {"type": "boolean"},

                "updated_at": {"type": "date", "format": "yyyy-MM-dd HH:mm:ss||yyyy-MM-dd||epoch_millis"},
                "search_all": {
                    "type": "text",
                    "analyzer": "product_index_analyzer"
                },
                "suggest": {
                    "type": "completion",
                    "analyzer": "ik_max_word",
                    "search_analyzer": "ik_smart"
                }
            }
        }
    }

    if es.indices.exists(index=INDEX_NAME):
        logger.info(f"索引 {INDEX_NAME} 已存在")

    es.indices.update_aliases(body={
        "actions": [{"add": {"index": INDEX_NAME, "alias": INDEX_ALIAS}}]
    })
    logger.info(f"索引 {INDEX_NAME} 创建成功，别名 {INDEX_ALIAS}")
```

```python
            "_id": str(i + 1),
            "_source": {

                "in_stock": product["stock"] > 0,
                "created_at": now,
                "updated_at": now,

    success, failed = bulk(es, actions, chunk_size=100)
```

## 第五步：搜索服务

```python
# search_service.py
from elasticsearch import Elasticsearch
from config import INDEX_ALIAS, DEFAULT_PAGE_SIZE, MAX_PAGE_SIZE, SEARCH_TIMEOUT
from models import SearchRequest, SearchResponse
import logging

logger = logging.getLogger(__name__)

def search_products(es: Elasticsearch, request: SearchRequest) -> SearchResponse:
    """商品搜索"""
    must_clauses = []
    filter_clauses = []

    # 关键词搜索
    if request.keyword:
        must_clauses.append({
            "multi_match": {
                "query": request.keyword,
                "fields": ["name^3", "description", "tags^2", "brand^2"],
                "fuzziness": "AUTO",
                "prefix_length": 2
            }
        })

    # 分类筛选
    if request.category:
        filter_clauses.append({"term": {"category": request.category}})

    # 品牌筛选
    if request.brand:
        filter_clauses.append({"term": {"brand": request.brand}})

    # 价格范围
    if request.min_price is not None or request.max_price is not None:
        price_range = {}
        if request.min_price is not None:
            price_range["gte"] = request.min_price
        if request.max_price is not None:
            price_range["lte"] = request.max_price
        filter_clauses.append({"range": {"price": price_range}})

    # 库存筛选
    if request.in_stock is not None:
        filter_clauses.append({"term": {"in_stock": request.in_stock}})

    # 构建查询
    if must_clauses or filter_clauses:
        query = {
            "bool": {
                "must": must_clauses if must_clauses else [{"match_all": {}}],
                "filter": filter_clauses
            }
        }
    else:
        query = {"match_all": {}}

    # 排序
    sort_options = {
        "_score": [{"_score": "desc"}],
        "price_asc": [{"price": "asc"}],
        "price_desc": [{"price": "desc"}],
        "rating": [{"rating": "desc"}, {"_score": "desc"}],
        "sales": [{"sales_count": "desc"}],
        "newest": [{"created_at": "desc"}]
    }
    sort = sort_options.get(request.sort_by, [{"_score": "desc"}])

    # 分页
    page_size = min(request.page_size, MAX_PAGE_SIZE)
    from_offset = (request.page - 1) * page_size

    # 聚合
    aggs = {
        "categories": {"terms": {"field": "category", "size": 20}},
        "brands": {"terms": {"field": "brand", "size": 20}},
        "price_ranges": {
            "range": {
                "field": "price",
                "ranges": [
                    {"key": "0-500", "to": 500},
                    {"key": "500-2000", "from": 500, "to": 2000},
                    {"key": "2000-5000", "from": 2000, "to": 5000},
                    {"key": "5000+", "from": 5000}
                ]
            }
        },
        "price_stats": {"stats": {"field": "price"}}
    }

    # 执行查询
    response = es.search(
        index=INDEX_ALIAS,
        body={
            "query": query,
            "sort": sort,
            "from": from_offset,
            "size": page_size,
            "highlight": {
                "fields": {
                    "name": {"number_of_fragments": 1},
                    "description": {"fragment_size": 200, "number_of_fragments": 1}
                },
                "pre_tags": ["<em>"],
                "post_tags": ["</em>"]
            },
            "aggs": aggs,
            "_source": ["name", "description", "price", "original_price",
                        "category", "brand", "tags", "stock", "rating",
                        "sales_count", "in_stock", "created_at"]
        },
        timeout=SEARCH_TIMEOUT
    )

    # 解析结果
    results = []
    for hit in response["hits"]["hits"]:
        src = hit["_source"]
        highlight = hit.get("highlight", {})
        results.append({
            "id": hit["_id"],
            "name": highlight.get("name", [src["name"]])[0],
            "description": highlight.get("description", [src.get("description", "")])[0],
            "price": src["price"],
            "original_price": src.get("original_price"),
            "category": src["category"],
            "brand": src["brand"],
            "tags": src.get("tags", []),
            "stock": src.get("stock", 0),
            "rating": src.get("rating", 0),
            "sales_count": src.get("sales_count", 0),
            "in_stock": src.get("in_stock", True),
            "score": hit["_score"]
        })

    # 解析聚合
    aggregations = {}
    if "aggregations" in response:
        aggs_data = response["aggregations"]
        aggregations = {
            "categories": [{"name": b["key"], "count": b["doc_count"]}
                          for b in aggs_data["categories"]["buckets"]],
            "brands": [{"name": b["key"], "count": b["doc_count"]}
                      for b in aggs_data["brands"]["buckets"]],
            "price_distribution": [{"range": b["key"], "count": b["doc_count"]}
                                   for b in aggs_data["price_ranges"]["buckets"]],
            "price_stats": aggs_data["price_stats"]
        }

    return SearchResponse(
        total=response["hits"]["total"]["value"],
        page=request.page,
        page_size=page_size,
        results=results,
        aggregations=aggregations,
        took_ms=response["took"]
    )


def get_suggestions(es: Elasticsearch, keyword: str, size: int = 5):
    """搜索建议"""
    response = es.search(
        index=INDEX_ALIAS,
        body={
            "suggest": {
                "product_suggest": {
                    "prefix": keyword,
                    "completion": {
                        "field": "suggest",
                        "size": size,
                        "fuzzy": {"fuzziness": "AUTO"}
                    }
                }
            }
        }
    )

    suggestions = []
    for option in response["suggest"]["product_suggest"][0]["options"]:
        suggestions.append({
            "text": option["text"],
            "score": option["_score"]
        })

    return suggestions
```

```python
def get_aggregations(es: Elasticsearch, keyword=None, category=None):
    """获取聚合数据"""
    filters = []
    if keyword:
        filters.append({"multi_match": {"query": keyword, "fields": ["name", "description"]}})
    if category:
        filters.append({"term": {"category": category}})

    if filters:
        query = {"bool": {"must": [{"match_all": {}}], "filter": filters}}
    else:
        query = {"match_all": {}}

    response = es.search(
        index=INDEX_ALIAS,
        body={
            "size": 0,
            "query": query,
            "aggs": {
                "categories": {
                    "terms": {"field": "category", "size": 20},
                    "aggs": {
                        "avg_price": {"avg": {"field": "price"}},
                        "avg_rating": {"avg": {"field": "rating"}}
                    }
                },
                "brands": {"terms": {"field": "brand", "size": 20}},
                "price_ranges": {
                    "range": {
                        "field": "price",
                        "ranges": [
                            {"key": "0-500", "to": 500},
                            {"key": "500-2000", "from": 500, "to": 2000},
                            {"key": "2000-5000", "from": 2000, "to": 5000},
                            {"key": "5000+", "from": 5000}
                        ]
                    }
                },
                "price_stats": {"stats": {"field": "price"}},
                "rating_distribution": {
                    "histogram": {"field": "rating", "interval": 0.5, "min_doc_count": 1}
                }
            }
        }
    )

    aggs = response["aggregations"]
    return {
        "categories": [
            {
                "name": b["key"],
                "count": b["doc_count"],
                "avg_price": round(b["avg_price"]["value"], 2),
                "avg_rating": round(b["avg_rating"]["value"], 2)
            }
            for b in aggs["categories"]["buckets"]
        ],
        "brands": [
            {"name": b["key"], "count": b["doc_count"]}
            for b in aggs["brands"]["buckets"]
        ],
        "price_distribution": [
            {"range": b["key"], "count": b["doc_count"]}
            for b in aggs["price_ranges"]["buckets"]
        ],
        "price_stats": aggs["price_stats"],
        "rating_distribution": [
            {"rating": b["key"], "count": b["doc_count"]}
            for b in aggs["rating_distribution"]["buckets"]
        ]
    }
```

```python
# app.py
from fastapi import FastAPI, Query, HTTPException
from typing import Optional
from es_client import get_es_client
from index_manager import create_product_index
from data_loader import load_sample_data
from search_service import search_products, get_suggestions, get_aggregations
from models import SearchRequest, SearchResponse
from config import INDEX_NAME
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

app = FastAPI(
    title="电商搜索服务",
    description="基于 Elasticsearch 的电商商品搜索 API",
    version="1.0.0"
)

@app.on_event("startup")
async def startup():
    """应用启动时初始化"""
    try:
        es = get_es_client()
        create_product_index(es)
        load_sample_data(es, INDEX_NAME)
        logger.info("搜索服务启动完成")
    except Exception as e:
        logger.error(f"启动失败: {e}")
        raise

@app.get("/search", response_model=SearchResponse)
async def search(
    keyword: Optional[str] = Query(None, description="搜索关键词"),
    category: Optional[str] = Query(None, description="分类筛选"),
    brand: Optional[str] = Query(None, description="品牌筛选"),
    min_price: Optional[float] = Query(None, ge=0, description="最低价格"),
    max_price: Optional[float] = Query(None, ge=0, description="最高价格"),
    in_stock: Optional[bool] = Query(None, description="是否有货"),
    sort_by: str = Query("_score", description="排序方式: _score, price_asc, price_desc, rating, sales, newest"),
    page: int = Query(1, ge=1, description="页码"),
    page_size: int = Query(10, ge=1, le=50, description="每页数量")
):
    """商品搜索接口"""
    try:
        es = get_es_client()
        request = SearchRequest(
            keyword=keyword,
            category=category,
            brand=brand,
            min_price=min_price,
            max_price=max_price,
            in_stock=in_stock,
            sort_by=sort_by,
            page=page,
            page_size=page_size
        )
        return search_products(es, request)
    except Exception as e:
        logger.error(f"搜索失败: {e}")
        raise HTTPException(status_code=500, detail=f"搜索服务异常: {str(e)}")

@app.get("/suggest")

@app.get("/aggregations")
async def aggregations(
    keyword: Optional[str] = None,
    category: Optional[str] = None
):
    """聚合分析接口"""
    try:
        es = get_es_client()
        return get_aggregations(es, keyword, category)
    except Exception as e:
        logger.error(f"聚合查询失败: {e}")
        raise HTTPException(status_code=500, detail=f"聚合服务异常: {str(e)}")

@app.get("/health")
async def health():
    """健康检查接口"""
    try:
        es = get_es_client()
        cluster_health = es.cluster.health()
        return {
            "status": "healthy",
            "cluster_status": cluster_health["status"],
            "nodes": cluster_health["number_of_nodes"],
            "active_shards": cluster_health["active_shards"]
        }
    except Exception as e:
        return {"status": "unhealthy", "error": str(e)}

@app.get("/product/{product_id}")
async def get_product(product_id: str):
    """获取单个商品详情"""
    try:
        es = get_es_client()
        response = es.get(index=INDEX_NAME, id=product_id)
        return {"id": response["_id"], **response["_source"]}
    except Exception as e:
        if "not_found" in str(e).lower():
            raise HTTPException(status_code=404, detail="商品不存在")
        raise HTTPException(status_code=500, detail=str(e))
```

## 常见误区

1. **启动时不检查 ES 连接**：应在 startup 事件中验证连接，否则用户会看到莫名的 500 错误。

2. **数据导入后不 refresh**：搜索可能查不到刚导入的数据。导入完成后应手动 refresh。

3. **忽略搜索高亮**：高亮能显著提升用户体验，让用户快速定位匹配内容。

4. **聚合查询没有 size=0**：应设置 `size: 0` 避免返回不必要文档。

5. **硬编码索引名**：使用别名方便后续索引重建。

6. **忽略异常处理**：ES 连接失败、查询超时等异常应被捕获并返回友好错误信息。

7. **搜索参数不做校验**：使用 Pydantic 的 Query 参数校验防止非法输入。

8. **忘记设置超时**：搜索查询应设置 timeout，防止单个慢查询拖垮服务。

## 工程建议

1. **使用连接池**：elasticsearch-py 默认使用连接池，复用 HTTP 连接。

2. **异常处理**：捕获 `ConnectionError`、`RequestError`、`NotFoundError`，返回友好错误信息。

3. **搜索参数校验**：使用 Pydantic 的 `Query` 参数校验防止非法输入。

### 练习四

**思路**：实现一个基于字典的简单缓存，支持 TTL 和最大容量。

**答案**：

```python
import time
from collections import OrderedDict

class TTLCache:
    def __init__(self, max_size=1000, ttl_seconds=60):
        self.cache = OrderedDict()
        self.max_size = max_size
        self.ttl = ttl_seconds

    def get(self, key):
        if key in self.cache:
            value, timestamp = self.cache[key]
            if time.time() - timestamp < self.ttl:
                # 移到末尾（最近使用）
                self.cache.move_to_end(key)
                return value
            else:
                del self.cache[key]
        return None

    def set(self, key, value):
        if key in self.cache:
            del self.cache[key]
        elif len(self.cache) >= self.max_size:
            self.cache.popitem(last=False)
        self.cache[key] = (value, time.time())

# 在搜索服务中使用
search_cache = TTLCache(max_size=1000, ttl_seconds=60)

def cached_search(es, request):
    cache_key = f"{request.keyword}:{request.category}:{request.sort_by}:{request.page}"
    cached = search_cache.get(cache_key)
    if cached:
        return cached
    result = search_products(es, request)
    search_cache.set(cache_key, result)
    return result
```

**要点**：缓存键应包含所有影响搜索结果的参数；TTL 应根据数据更新频率设置。

### 练习五

**答案**：

```python
from elasticsearch import Elasticsearch
from datetime import datetime

es = Elasticsearch("http://localhost:9200")

# 创建搜索日志索引
def create_search_log_index(es):
    es.indices.create(index="search_logs", body={
        "mappings": {
            "properties": {
                "keyword": {"type": "keyword"},
                "category": {"type": "keyword"},
                "result_count": {"type": "integer"},
                "took_ms": {"type": "integer"},
                "timestamp": {"type": "date"},
                "user_id": {"type": "keyword"}
            }
        }
    }, ignore=[400])

# 记录搜索日志
def log_search(keyword, category, result_count, took_ms, user_id=None):
    es.index(index="search_logs", document={
        "keyword": keyword,
        "category": category,
        "result_count": result_count,
        "took_ms": took_ms,
        "timestamp": datetime.now().isoformat(),
        "user_id": user_id
    })

# 分析热门搜索词
def analyze_hot_keywords(es, days=7, top_n=20):
    response = es.search(index="search_logs", body={
        "size": 0,
        "query": {"range": {"timestamp": {"gte": f"now-{days}d"}}},
        "aggs": {
            "hot_keywords": {
                "terms": {"field": "keyword", "size": top_n},
                "aggs": {
                    "avg_results": {"avg": {"field": "result_count"}},
                    "avg_latency": {"avg": {"field": "took_ms"}}
                }
            },
            "zero_result_keywords": {
                "filter": {"term": {"result_count": 0}},
                "aggs": {
                    "keywords": {"terms": {"field": "keyword", "size": 20}}
                }
            }
        }
    })

    aggs = response["aggregations"]
    return {
        "hot_keywords": [
            {
                "keyword": b["key"],
                "count": b["doc_count"],
                "avg_results": round(b["avg_results"]["value"], 1),
                "avg_latency": round(b["avg_latency"]["value"], 1)
            }
            for b in aggs["hot_keywords"]["buckets"]
        ],
        "zero_result_keywords": [
            b["key"]
            for b in aggs["zero_result_keywords"]["keywords"]["buckets"]
        ]
    }
```

**要点**：无结果搜索词是优化搜索质量的重要信号，应定期分析并添加同义词或调整分词器。
