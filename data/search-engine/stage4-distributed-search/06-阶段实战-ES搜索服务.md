```python
from config import ES_HOST

_es_client = None

def get_es_client():
    global _es_client
    if _es_client is None:
        _es_client = Elasticsearch(ES_HOST, retry_on_timeout=True,
                                   max_retries=3, request_timeout=30)
        if not _es_client.ping():
            raise ConnectionError(f"无法连接: {ES_HOST}")
    return _es_client
```

```python
from pydantic import BaseModel, Field
from typing import Optional, List

class SearchRequest(BaseModel):

    category: Optional[str] = Field(None, description="分类筛选")

    page: int
    page_size: int
    results: list
    aggregations: Optional[dict] = None
    took_ms: int
```

## 索引管理

```python
# index_manager.py
from elasticsearch import Elasticsearch
from config import INDEX_NAME, INDEX_ALIAS

def create_product_index(es: Elasticsearch):
    index_body = {
        "settings": {
            "number_of_shards": 2, "number_of_replicas": 1,
            "analysis": {
                "filter": {
                    "product_synonym": {
                        "type": "synonym",
                        "synonyms": ["手机,移动电话,智能机",
                                     "笔记本,笔记本电脑,laptop",
                                     "耳机,earphone,headphone"]
                    }
                },
                "analyzer": {
                    "product_index_analyzer": {
                        "type": "custom", "tokenizer": "ik_max_word",
                        "filter": ["lowercase", "product_synonym"]
                    },
                    "product_search_analyzer": {
                        "type": "custom", "tokenizer": "ik_smart",
                        "filter": ["lowercase"]
                    }
                }
            }
        },
        "mappings": {
            "properties": {
                "name": {
                    "type": "text",
                    "analyzer": "product_index_analyzer",
                    "search_analyzer": "product_search_analyzer",
                    "fields": {"keyword": {"type": "keyword"}},
                    "copy_to": "search_all"
                },
                "description": {"type": "text", "analyzer": "product_index_analyzer",
                                "doc_values": False, "copy_to": "search_all"},
                "price": {"type": "float"},
                "category": {"type": "keyword"},
                "brand": {"type": "keyword"},
                "tags": {"type": "keyword"},
                "stock": {"type": "integer"},
                "rating": {"type": "float"},
                "sales_count": {"type": "integer"},
                "in_stock": {"type": "boolean"},
                "search_all": {"type": "text", "analyzer": "product_index_analyzer"},
                "suggest": {"type": "completion", "analyzer": "ik_max_word",
                            "search_analyzer": "ik_smart"}
            }
        }
    }
    if not es.indices.exists(index=INDEX_NAME):
        es.indices.create(index=INDEX_NAME, body=index_body)
        es.indices.update_aliases(body={
            "actions": [{"add": {"index": INDEX_NAME, "alias": INDEX_ALIAS}}]
        })
```

关键设计选择：
- 索引时用 `ik_max_word`（最细粒度，提高召回），搜索时用 `ik_smart`（粗粒度，提高精确率）
- `copy_to: search_all` 将 name 和 description 合并到一个字段，简化全文搜索
- 同义词配置：手机=移动电话=智能机。同义词更新后需要 reindex 才能生效

```python
from elasticsearch.helpers import bulk
from datetime import datetime

SAMPLE_PRODUCTS = [
    {"name": "Apple iPhone 15 Pro Max", "description": "苹果旗舰手机，A17 Pro芯片",
     "price": 9999, "category": "electronics", "brand": "Apple",
     "tags": ["手机", "旗舰"], "stock": 500, "rating": 4.8, "sales_count": 15000},
    {"name": "华为 Mate 60 Pro", "description": "华为旗舰手机，麒麟9000S芯片",
     "price": 6999, "category": "electronics", "brand": "华为",
     "tags": ["手机", "旗舰"], "stock": 300, "rating": 4.7, "sales_count": 12000},
    {"name": "索尼 WH-1000XM5", "description": "旗舰降噪蓝牙耳机，30小时续航",
     "price": 2499, "category": "electronics", "brand": "索尼",
     "tags": ["耳机", "降噪"], "stock": 600, "rating": 4.8, "sales_count": 20000},
    # ... 更多商品
]

def load_sample_data(es, index):
    if es.count(index=index)["count"] > 0:
        return
    actions = []
    for i, product in enumerate(SAMPLE_PRODUCTS):
        now = datetime.now().isoformat()
        actions.append({
            "_index": index, "_id": str(i + 1),
            "_source": {**product, "in_stock": product["stock"] > 0,
                        "created_at": now, "updated_at": now,
                        "suggest": {"input": [product["name"], product["brand"]],
                                    "weight": int(product["rating"] * 100)}}
        })
    bulk(es, actions, chunk_size=100)
    es.indices.refresh(index=index)
```

导入完成后必须 `refresh`，否则搜索可能查不到刚导入的数据。

## 搜索服务

```python
# search_service.py
from elasticsearch import Elasticsearch
from config import INDEX_ALIAS, MAX_PAGE_SIZE, SEARCH_TIMEOUT
from models import SearchRequest, SearchResponse

def search_products(es: Elasticsearch, request: SearchRequest) -> SearchResponse:
    must_clauses = []
    filter_clauses = []

    if request.keyword:
        must_clauses.append({
            "multi_match": {
                "query": request.keyword,
                "fields": ["name^3", "description", "tags^2", "brand^2"],
                "fuzziness": "AUTO", "prefix_length": 2
            }
        })

    if request.category:
        filter_clauses.append({"term": {"category": request.category}})
    if request.brand:
        filter_clauses.append({"term": {"brand": request.brand}})
    if request.min_price is not None or request.max_price is not None:
        price_range = {}
        if request.min_price is not None:
            price_range["gte"] = request.min_price
        if request.max_price is not None:
            price_range["lte"] = request.max_price
        filter_clauses.append({"range": {"price": price_range}})

    query = ({"bool": {"must": must_clauses if must_clauses else [{"match_all": {}}],
                        "filter": filter_clauses}}
             if must_clauses or filter_clauses else {"match_all": {}})

    sort_options = {
        "_score": [{"_score": "desc"}], "price_asc": [{"price": "asc"}],
        "price_desc": [{"price": "desc"}], "rating": [{"rating": "desc"}],
        "sales": [{"sales_count": "desc"}],
    }
    sort = sort_options.get(request.sort_by, [{"_score": "desc"}])
    page_size = min(request.page_size, MAX_PAGE_SIZE)

    aggs = {
        "categories": {"terms": {"field": "category", "size": 20}},
        "brands": {"terms": {"field": "brand", "size": 20}},
        "price_ranges": {"range": {"field": "price", "ranges": [
            {"key": "0-500", "to": 500}, {"key": "500-2000", "from": 500, "to": 2000},
            {"key": "2000-5000", "from": 2000, "to": 5000}, {"key": "5000+", "from": 5000}
        ]}},
    }

    response = es.search(
        index=INDEX_ALIAS,
        body={"query": query, "sort": sort,
              "from": (request.page - 1) * page_size, "size": page_size,
              "highlight": {"fields": {"name": {}, "description": {"fragment_size": 200}},
                            "pre_tags": ["<em>"], "post_tags": ["</em>"]},
              "aggs": aggs,
              "_source": ["name", "description", "price", "category", "brand",
                          "tags", "rating", "sales_count", "in_stock"]},
        timeout=SEARCH_TIMEOUT)

    results = []
    for hit in response["hits"]["hits"]:
        src = hit["_source"]
        hl = hit.get("highlight", {})
        results.append({
            "id": hit["_id"],
            "name": hl.get("name", [src["name"]])[0],
            "description": hl.get("description", [src.get("description", "")])[0],
            "price": src["price"], "category": src["category"],
            "brand": src["brand"], "rating": src.get("rating", 0),
            "score": hit["_score"]
        })

    aggregations = {}
    if "aggregations" in response:
        aggs_data = response["aggregations"]
        aggregations = {
            "categories": [{"name": b["key"], "count": b["doc_count"]}
                           for b in aggs_data["categories"]["buckets"]],
            "brands": [{"name": b["key"], "count": b["doc_count"]}
                       for b in aggs_data["brands"]["buckets"]],
        }

    return SearchResponse(
        total=response["hits"]["total"]["value"], page=request.page,
        page_size=page_size, results=results, aggregations=aggregations,
        took_ms=response["took"])
```

## FastAPI 接口

```python
# app.py
from fastapi import FastAPI, Query, HTTPException
from typing import Optional
from es_client import get_es_client
from index_manager import create_product_index
from data_loader import load_sample_data
from search_service import search_products
from models import SearchRequest, SearchResponse
from config import INDEX_NAME

app = FastAPI(title="电商搜索服务", version="1.0.0")

@app.on_event("startup")
async def startup():
    es = get_es_client()
    create_product_index(es)
    load_sample_data(es, INDEX_NAME)

@app.get("/search", response_model=SearchResponse)
async def search(
    keyword: Optional[str] = None, category: Optional[str] = None,
    brand: Optional[str] = None, min_price: Optional[float] = None,
    max_price: Optional[float] = None, sort_by: str = "_score",
    page: int = 1, page_size: int = 10
):
    try:
        es = get_es_client()
        request = SearchRequest(keyword=keyword, category=category, brand=brand,
                                min_price=min_price, max_price=max_price,
                                sort_by=sort_by, page=page, page_size=page_size)
        return search_products(es, request)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/suggest")
async def suggest(keyword: str = Query(..., min_length=1), size: int = 5):
    es = get_es_client()
    response = es.search(index=INDEX_ALIAS, body={
        "suggest": {"product_suggest": {
            "prefix": keyword,
            "completion": {"field": "suggest", "size": size,
                           "fuzzy": {"fuzziness": "AUTO"}}
        }}
    })
    return [{"text": o["text"], "score": o["_score"]}
            for o in response["suggest"]["product_suggest"][0]["options"]]

@app.get("/health")
async def health():
    try:
        es = get_es_client()
        h = es.cluster.health()
        return {"status": "healthy", "cluster_status": h["status"],
                "nodes": h["number_of_nodes"]}
    except Exception as e:
        return {"status": "unhealthy", "error": str(e)}
```

## 常见误区

**启动时不检查 ES 连接。** 应在 startup 事件中验证连接，否则用户会看到莫名的 500 错误。

**数据导入后不 refresh。** 搜索可能查不到刚导入的数据。

**搜索和分面用两次请求。** 分面聚合应该和搜索放在同一个请求中。

**忘记设置超时。** 搜索查询应设置 timeout，防止单个慢查询拖垮服务。

## 工程建议

- 使用别名：方便后续索引重建，零停机切换
- 异常处理：捕获 `ConnectionError`、`RequestError`，返回友好错误信息
- 健康检查：配合 K8s readiness probe
- 搜索结果缓存：对热门搜索词的结果进行缓存，TTL 5-10 分钟

### 练习二：搜索结果缓存

实现一个 TTL 缓存层，对热门搜索词的结果缓存 60 秒。

```python
import time
from collections import OrderedDict

class TTLCache:
    def __init__(self, max_size=1000, ttl_seconds=60):
        self.cache = OrderedDict()
        self.max_size = max_size
        self.ttl = ttl_seconds

    def get(self, key):
        if key in self.cache:
            value, timestamp = self.cache[key]
            if time.time() - timestamp < self.ttl:
                self.cache.move_to_end(key)
                return value
            del self.cache[key]
        return None

    def set(self, key, value):
        if key in self.cache:
            del self.cache[key]
        elif len(self.cache) >= self.max_size:
            self.cache.popitem(last=False)
        self.cache[key] = (value, time.time())
```

### 练习三：搜索日志分析

添加搜索日志记录功能，然后用 ES 聚合分析热门搜索词和无结果搜索词。

```python
def log_search(es, keyword, result_count, took_ms):
    es.index(index="search_logs", document={
        "keyword": keyword, "result_count": result_count,
        "took_ms": took_ms, "timestamp": datetime.now().isoformat()
    })

def analyze_zero_results(es):
    response = es.search(index="search_logs", body={
        "size": 0,
        "query": {"term": {"result_count": 0}},
        "aggs": {"keywords": {"terms": {"field": "keyword.keyword", "size": 20}}}
    })
    return [b["key"] for b in response["aggregations"]["keywords"]["buckets"]]
```

无结果搜索词是优化搜索质量的重要信号，应定期分析并添加同义词或调整分词器。
