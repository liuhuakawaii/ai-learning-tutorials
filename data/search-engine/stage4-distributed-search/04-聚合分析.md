# 聚合分析

## 场景引入

你的电商平台产品经理提出需求：在搜索页面右侧展示商品分类分布、价格区间分布、品牌 Top 10，还要支持按时间查看销量趋势。这些都不是简单的搜索查询，而是需要对搜索结果进行统计分析。

Elasticsearch 的聚合（Aggregation）功能就是为这类需求设计的。它能在搜索的同时完成复杂的统计计算，基于倒排索引和列式存储，性能远优于"先查出来再程序里统计"。

聚合的历史可以追溯到 Elasticsearch 0.90 版本（2013 年），最初只支持简单的 terms 和 stats 聚合。到 2.x 版本引入了 pipeline aggregation，5.x 版本引入了 composite aggregation 支持高基数分页遍历。如今聚合已经成为 ELK Stack 中 Kibana 可视化的核心基础——Kibana 的每一个图表背后都是一个 ES 聚合查询。

理解聚合不仅能让你实现搜索页面的筛选面板，还能让你用 Elasticsearch 做实时数据分析，替代传统的 OLAP 数据库。

## 学习目标

完成本课学习后，你将能够：

1. 使用桶聚合（Bucket Aggregation）对文档分组统计
2. 使用指标聚合（Metric Aggregation）计算统计值
3. 使用管道聚合（Pipeline Aggregation）对聚合结果再聚合
4. 用 Python 实现各种聚合查询
5. 掌握聚合性能优化技巧
6. 理解聚合的分布式执行原理和精度问题

## 聚合的基本结构

```
┌─────────────────────────────────────────────────────────────┐
│                    聚合类型                                   │
│                                                             │
│  Bucket Aggregation（桶聚合）                                │
│    把文档分成不同的"桶"，类似 SQL 的 GROUP BY                 │
│    terms / histogram / date_histogram / range               │
│                                                             │
│  Metric Aggregation（指标聚合）                              │
│    对一组文档计算统计指标                                     │
│    avg / max / min / sum / percentiles / cardinality        │
│                                                             │
│  Pipeline Aggregation（管道聚合）                            │
│    对其他聚合的结果进行再聚合                                 │
│    avg_bucket / max_bucket / derivative                     │
└─────────────────────────────────────────────────────────────┘
```

聚合查询的基本结构：

```json
{
  "size": 0,
  "query": { ... },
  "aggs": {
    "聚合名称": {
      "聚合类型": { ... },
      "aggs": {
        "子聚合名称": { ... }
      }
    }
  }
}
```

`size: 0` 表示不返回搜索结果，只返回聚合结果，减少数据传输。

## 桶聚合（Bucket Aggregation）

### Terms 聚合

按字段值分桶，类似 SQL 的 `GROUP BY`。

```bash
GET /products/_search
{
  "size": 0,
  "aggs": {
    "categories": {
      "terms": { "field": "category", "size": 10 }
    }
  }
}
```

返回结果包含 `buckets` 数组，每个桶有 `key` 和 `doc_count`。

**Terms 聚合的精度问题**：

Terms 聚合默认采用近似算法：每个分片取 Top N（由 `shard_size` 控制），然后协调节点合并。分片多时结果可能不精确。

```python
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

def accurate_terms_aggregation(es, index, field, size=10):
    """高精度 terms 聚合"""
    response = es.search(
        index=index,
        body={
            "size": 0,
            "aggs": {
                "accurate_terms": {
                    "terms": {
                        "field": field,
                        "size": size,
                        "shard_size": size * 10,  # 每个分片取更多结果
                        "show_term_doc_count_error": True
                    }
                }
            }
        }
    )

    buckets = response["aggregations"]["accurate_terms"]["buckets"]
    for b in buckets:
        error_upper = b.get("doc_count_error_upper_bound", 0)
        print(f"  {b['key']}: {b['doc_count']} 条 (误差上限: {error_upper})")

    return buckets

accurate_terms_aggregation(es, "products", "brand", size=10)
```

```python
es = Elasticsearch("http://localhost:9200")

    response = es.search(
        index=index,

            "query": {
                "range": {
                    "created_at": {
                        "gte": start_date,
                        "lte": end_date
                    }
                }
            },
            "aggs": {
                "monthly": {
                    "date_histogram": {
                        "field": "created_at",
                        "calendar_interval": "month",
                        "format": "yyyy-MM"
                    },
                    "aggs": {
                        "revenue": {"sum": {"field": "amount"}},

                        "unique_buyers": {"cardinality": {"field": "user_id"}}

                }
```

```python
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

def search_filter_panel(es, index, query=None):
    """搜索页面筛选面板数据"""
    base_query = query or {"match_all": {}}

    response = es.search(
        index=index,
        body={
            "size": 0,
            "query": base_query,
            "aggs": {
                "categories": {
                    "terms": {"field": "category", "size": 20},
                    "aggs": {
                        "avg_price": {"avg": {"field": "price"}},
                        "price_range": {
                            "range": {
                                "field": "price",
                                "ranges": [
                                    {"key": "0-100", "to": 100},
                                    {"key": "100-500", "from": 100, "to": 500},
                                    {"key": "500+", "from": 500}
                                ]
                            }
                        }
                    }
                },
                "brands": {
                    "terms": {"field": "brand", "size": 20}
                },
                "price_stats": {
                    "stats": {"field": "price"}
                },
                "price_percentiles": {
                    "percentiles": {"field": "price", "percents": [25, 50, 75, 95]}
                }
            }
        }
    )

    aggs = response["aggregations"]

    panel = {
        "total_hits": response["hits"]["total"]["value"],
        "categories": [
            {
                "name": b["key"],
                "count": b["doc_count"],
                "avg_price": round(b["avg_price"]["value"], 2),
                "price_distribution": {
                    pr["key"]: pr["doc_count"]
                    for pr in b["price_range"]["buckets"]
                }
            }
            for b in aggs["categories"]["buckets"]
        ],
        "brands": [
            {"name": b["key"], "count": b["doc_count"]}
            for b in aggs["brands"]["buckets"]
        ],
        "price_stats": aggs["price_stats"],
        "price_percentiles": aggs["price_percentiles"]["values"]
    }

    return panel

panel = search_filter_panel(es, "products", {"match": {"name": "手机"}})
print(f"总命中: {panel['total_hits']}")
print(f"分类数: {len(panel['categories'])}")
print(f"品牌数: {len(panel['brands'])}")
print(f"价格中位数: {panel['price_percentiles']['50.0']}")
```

### Composite 聚合

Composite 聚合支持分页遍历所有桶，适合高基数字段（如用户 ID、商品 ID）：

```python
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

def composite_aggregation_walk(es, index, field, page_size=1000):
    """使用 composite 聚合遍历所有桶"""
    all_buckets = []
    after_key = None

    while True:
        body = {
            "size": 0,
            "aggs": {
                "my_buckets": {
                    "composite": {
                        "size": page_size,
                        "sources": [
                            {field: {"terms": {"field": field}}}
                        ]
                    }
                }
            }
        }

        if after_key:
            body["aggs"]["my_buckets"]["composite"]["after"] = after_key

        response = es.search(index=index, body=body)
        agg = response["aggregations"]["my_buckets"]

        for bucket in agg["buckets"]:
            all_buckets.append({
                "key": bucket["key"][field],
                "count": bucket["doc_count"]
            })

        if "after_key" in agg:
            after_key = agg["after_key"]
        else:
            break

    print(f"共遍历 {len(all_buckets)} 个桶")
    return all_buckets

# 遍历所有品牌
brands = composite_aggregation_walk(es, "products", "brand")
```

### Top Hits 聚合

获取每个桶中的实际文档，适合获取每个品牌的最新商品：

```python
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

def top_products_per_brand(es, index, top_n=3):
    """每个品牌的 Top N 商品"""
    response = es.search(
        index=index,
        body={
            "size": 0,
            "aggs": {
                "by_brand": {
                    "terms": {"field": "brand", "size": 20},
                    "aggs": {
                        "top_products": {
                            "top_hits": {
                                "size": top_n,
                                "sort": [{"rating": "desc"}],
                                "_source": ["name", "price", "rating"]
                            }
                        },
                        "avg_rating": {"avg": {"field": "rating"}}
                    }
                }
            }
        }
    )

    results = {}
    for bucket in response["aggregations"]["by_brand"]["buckets"]:
        brand = bucket["key"]
        products = []
        for hit in bucket["top_products"]["hits"]["hits"]:
            products.append({
                "name": hit["_source"]["name"],
                "price": hit["_source"]["price"],
                "rating": hit["_source"]["rating"]
            })
        results[brand] = {
            "count": bucket["doc_count"],
            "avg_rating": round(bucket["avg_rating"]["value"], 2),
            "top_products": products
        }

    return results

results = top_products_per_brand(es, "products", top_n=3)
for brand, data in results.items():
    print(f"\n{brand} ({data['count']} 商品, 平均评分 {data['avg_rating']}):")
    for p in data["top_products"]:
        print(f"  - {p['name']} (¥{p['price']}, ★{p['rating']})")
```

### 导数聚合（Derivative）

计算相邻桶之间的变化率，适合分析增长趋势：

```python
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

def revenue_growth_analysis(es, index, start_date, end_date):
    """分析月收入增长趋势"""
    response = es.search(
        index=index,
        body={
            "size": 0,
            "query": {"range": {"created_at": {"gte": start_date, "lte": end_date}}},
            "aggs": {
                "monthly": {
                    "date_histogram": {
                        "field": "created_at",
                        "calendar_interval": "month"
                    },
                    "aggs": {
                        "revenue": {"sum": {"field": "amount"}},
                        "revenue_deriv": {
                            "derivative": {"buckets_path": "revenue"}
                        },
                        "revenue_pct_change": {
                            "serial_diff": {"buckets_path": "revenue", "lag": 1}
                        }
                    }
                }
            }
        }
    )

    results = []
    for bucket in response["aggregations"]["monthly"]["buckets"]:
        month = bucket["key_as_string"]
        revenue = round(bucket["revenue"]["value"], 2)
        deriv = bucket.get("revenue_deriv", {}).get("value", 0)
        pct_change = bucket.get("revenue_pct_change", {}).get("value", 0)
        results.append({
            "month": month,
            "revenue": revenue,
            "growth": round(deriv, 2) if deriv else 0,
            "change": round(pct_change, 2) if pct_change else 0
        })

    return results

growth = revenue_growth_analysis(es, "orders", "2024-01-01", "2024-12-31")
for g in growth:
    print(f"{g['month']}: 收入 {g['revenue']}, 增长 {g['change']}")
```

### 聚合性能优化清单

```python
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

def optimized_aggregation(es, index, category=None):
    """优化的聚合查询示例"""
    filters = []
    if category:
        filters.append({"term": {"category": category}})

    response = es.search(
        index=index,
        body={
            "size": 0,  # 不返回文档，只返回聚合
            "query": {
                "bool": {
                    "filter": filters  # 使用 filter context，可缓存
                }
            } if filters else {"match_all": {}},
            "aggs": {
                "categories": {
                    "terms": {
                        "field": "category",
                        "size": 10,
                        "shard_size": 50,  # 提高精度
                        "execution_mode": "map"  # 小数据量用 map 模式更快
                    },
                    "aggs": {
                        "avg_price": {
                            "avg": {"field": "price"}
                        }
                    }
                }
            }
        },
        params={"request_cache": "true"}  # 启用请求缓存
    )

    return response["aggregations"]

# 使用 preference 确保缓存命中
response = es.search(
    index="products",
    body={"size": 0, "aggs": {"categories": {"terms": {"field": "category"}}}},
    preference="user_123"  # 同一用户路由到同一分片
)
```

## 常见误区

1. **聚合结果代表全量数据**：聚合默认在查询匹配的文档集上执行，没写 query 才全量聚合。

2. **terms 聚合结果是精确的**：默认在每个分片上取 Top N 再合并，分片多时可能不精确。需要精确结果时增大 `shard_size`。

3. **cardinality 是精确去重**：使用 HyperLogLog++ 概率算法，存在约 2% 误差。需要精确去重时使用 `precision_threshold` 参数。

```python
# 提高 cardinality 精度
response = es.search(index="products", body={
    "size": 0,
    "aggs": {
        "exact_brands": {
            "cardinality": {
                "field": "brand",
                "precision_threshold": 40000  # 最大精确值
            }
        }
    }
})
```

4. **忽略聚合的内存消耗**：高基数字段的 terms 聚合（如 user_id）会消耗大量内存。

5. **聚合不需要排序**：桶聚合可以用 `order` 参数排序，但排序字段必须是桶内的子聚合。

6. **过度嵌套聚合**：嵌套层数过多会导致性能急剧下降，通常不超过 3 层。

## 工程建议

1. **`size: 0` 只返回聚合结果**：不需要搜索结果时减少数据传输。

2. **合理设置 shard_size**：高精度场景增大此值，但会增加内存消耗。

```python
# 根据数据量动态调整 shard_size
def adaptive_terms_aggregation(es, index, field, size=10):
    """根据总文档数自适应调整 shard_size"""
    doc_count = es.count(index=index)["count"]
    # 文档数越多，shard_size 需要越大以保证精度
    shard_size = min(size * 10, max(size * 3, doc_count // 1000))

    return es.search(index=index, body={
        "size": 0,
        "aggs": {
            "terms": {
                "terms": {
                    "field": field,
                    "size": size,
                    "shard_size": shard_size
                }
            }
        }
    })
```

3. **使用 composite 聚合遍历高基数桶**：支持分页遍历所有桶，不会因为桶数过多导致 OOM。

4. **预聚合 vs 实时聚合**：历史数据可预先计算，避免每次实时聚合。

5. **使用 filter 聚合替代多个查询**：一次查询获取多个条件下的统计值。

```python
def multi_filter_aggregation(es, index):
    """使用 filter 聚合一次获取多组统计"""
    response = es.search(index=index, body={
        "size": 0,
        "aggs": {
            "in_stock": {
                "filter": {"term": {"in_stock": True}},
                "aggs": {
                    "count": {"value_count": {"field": "price"}},
                    "avg_price": {"avg": {"field": "price"}}
                }
            },
            "out_of_stock": {
                "filter": {"term": {"in_stock": False}},
                "aggs": {
                    "count": {"value_count": {"field": "price"}},
                    "avg_price": {"avg": {"field": "price"}}
                }
            },
            "discounted": {
                "filter": {"range": {"price": {"lt": 100}}},
                "aggs": {
                    "count": {"value_count": {"field": "price"}}
                }
            }
        }
    })
    return response["aggregations"]
```

## 小结

Elasticsearch 聚合是分布式数据分析的核心能力。桶聚合按维度分组，指标聚合计算统计值，管道聚合对聚合结果再聚合。通过嵌套组合可实现复杂多维分析。实际应用中需关注性能——限制返回桶数量、使用 filter 缓存、composite 分页遍历高基数字段。理解聚合的分布式执行原理和精度问题，是用好聚合的关键。cardinality 使用概率算法有约 2% 误差，terms 聚合受 shard_size 影响可能不精确。在生产环境中，应根据业务需求在精度和性能之间找到平衡。

### 练习四

**思路**：使用 filter 聚合，基于当前搜索条件一次性获取所有筛选面板数据。

**答案**：

```python
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

def search_with_facets(es, index, keyword=None, category=None, min_price=None, max_price=None):
    """带筛选面板的搜索"""
    must = []
    filters = []

    if keyword:
        must.append({"multi_match": {"query": keyword, "fields": ["name^3", "description"]}})
    if category:
        filters.append({"term": {"category": category}})
    if min_price is not None or max_price is not None:
        price_range = {}
        if min_price is not None:
            price_range["gte"] = min_price
        if max_price is not None:
            price_range["lte"] = max_price
        filters.append({"range": {"price": price_range}})

    query = {"bool": {"must": must or [{"match_all": {}}], "filter": filters}}

    response = es.search(index=index, body={
        "size": 10,
        "query": query,
        "aggs": {
            "categories": {"terms": {"field": "category", "size": 20}},
            "brands": {"terms": {"field": "brand", "size": 20}},
            "price_ranges": {
                "range": {
                    "field": "price",
                    "ranges": [
                        {"key": "0-100", "to": 100},
                        {"key": "100-500", "from": 100, "to": 500},
                        {"key": "500-2000", "from": 500, "to": 2000},
                        {"key": "2000+", "from": 2000}
                    ]
                }
            },
            "rating_distribution": {
                "histogram": {"field": "rating", "interval": 0.5, "min_doc_count": 1}
            }
        }
    })

    return {
        "total": response["hits"]["total"]["value"],
        "results": [hit["_source"] for hit in response["hits"]["hits"]],
        "facets": response["aggregations"]
    }
```

**要点**：聚合在查询匹配的文档集上执行，所以筛选面板会随搜索条件变化。

**答案**：

```python
from elasticsearch import Elasticsearch
from datetime import datetime

es = Elasticsearch("http://localhost:9200")

def dashboard_data(es, index):
    """电商仪表盘数据"""
    today = datetime.now().strftime("%Y-%m-%d")

    response = es.search(index=index, body={
        "size": 0,
        "query": {"range": {"created_at": {"gte": today}}},
        "aggs": {
            "order_count": {"value_count": {"field": "order_id"}},
            "total_revenue": {"sum": {"field": "amount"}},
            "avg_order_value": {"avg": {"field": "amount"}},
            "order_status": {"terms": {"field": "status", "size": 10}},
            "top_products": {
                "terms": {"field": "product_name.keyword", "size": 10, "order": {"total_sales": "desc"}},
                "aggs": {"total_sales": {"sum": {"field": "amount"}}}
            },
            "hourly_trend": {
                "date_histogram": {
                    "field": "created_at",
                    "calendar_interval": "hour",
                    "format": "HH:00"
                },
                "aggs": {"revenue": {"sum": {"field": "amount"}}}
            }
        }
    })

    aggs = response["aggregations"]
    return {
        "today_orders": aggs["order_count"]["value"],
        "today_revenue": round(aggs["total_revenue"]["value"], 2),
        "avg_order_value": round(aggs["avg_order_value"]["value"], 2),
        "order_status": {b["key"]: b["doc_count"] for b in aggs["order_status"]["buckets"]},
        "top_products": [
            {"name": b["key"], "sales": round(b["total_sales"]["value"], 2)}
            for b in aggs["top_products"]["buckets"]
        ],
        "hourly_trend": [
            {"hour": b["key_as_string"], "revenue": round(b["revenue"]["value"], 2)}
            for b in aggs["hourly_trend"]["buckets"]
        ]
    }
```

**要点**：一次查询获取所有仪表盘数据，减少网络 round trip。
