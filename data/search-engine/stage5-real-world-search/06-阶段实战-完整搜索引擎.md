## 整体架构设计

一个生产级搜索引擎的架构分为四层：

1. **接入层**：API 网关、限流、认证、请求路由。
2. **业务层**：搜索服务、建议服务、纠错服务、质量监控。
3. **存储层**：Elasticsearch（索引和检索）、Redis（缓存和计数器）、对象存储（大文档）。
4. **基础设施层**：日志收集、监控告警、CI/CD。

数据流向：用户请求 → API 网关 → 搜索服务 → ES 查询 → 结果处理（高亮、分面、AI 摘要）→ 响应返回。每一步都有监控和降级策略。

```
用户 → FastAPI → SearchService → Elasticsearch
                → SuggestService → Redis Cache
                → CorrectionService
                → MetricsService → Redis
                → AIService → OpenAI API (可选)
```

```python
from typing import Optional
from enum import Enum

class SortBy(str, Enum):

    PRICE_ASC = "price_asc"
    PRICE_DESC = "price_desc"
    SALES = "sales"
    RATING = "rating"

class SearchRequest(BaseModel):

    filters: dict = Field(default_factory=dict)
    sort_by: SortBy = SortBy.RELEVANCE
```

### Pydantic 模型的设计考量

`SearchRequest` 中的字段验证不是装饰，而是防御性编程的必要手段：

- `min_length=1`：防止空查询导致 ES 报错。
- `max_length=200`：超长查询可能是恶意输入或 bug。
- `ge=1`（page）：页码从 1 开始，防止负数。
- `le=100`（page_size）：限制单次返回数量，防止恶意请求拉取全量数据。

`SortBy` 用枚举类型限制可选值，FastAPI 自动返回 422 错误如果传入非法值。

```python
# app/services/indexer.py
from elasticsearch import Elasticsearch
from elasticsearch.helpers import bulk
from app.config import settings

es = Elasticsearch(settings.ES_HOST)

class IndexManager:
    def __init__(self, index_name: str = "main"):
        self.index = f"{settings.ES_INDEX_PREFIX}_{index_name}"

    def create_index(self, recreate: bool = False):
        if recreate:
            es.indices.delete(index=self.index, ignore=[404])
        mapping = {"mappings": {"properties": {
            "title": {"type": "text", "analyzer": "ik_max_word", "search_analyzer": "ik_smart"},
            "content": {"type": "text", "analyzer": "ik_max_word"},
            "category": {"type": "keyword"}, "tags": {"type": "keyword"},
            "price": {"type": "float"}, "sales_count": {"type": "integer"},
            "rating": {"type": "float"}, "created_at": {"type": "date"}
        }}, "settings": {"number_of_shards": 1, "number_of_replicas": 0}}
        es.indices.create(index=self.index, body=mapping, ignore=400)

    def bulk_index(self, documents: list[dict]):
        actions = [{"_index": self.index, "_id": d.get("id"), "_source": d} for d in documents]
        return bulk(es, actions, request_timeout=30)

    def get_stats(self) -> dict:
        stats = es.indices.stats(index=self.index)
        return {"doc_count": stats["indices"][self.index]["total"]["docs"]["count"],
                "size_bytes": stats["indices"][self.index]["total"]["store"]["size_in_bytes"]}
```

### 索引别名管理

生产环境中，直接用索引名查询是不推荐的。应该使用别名，这样可以在不停机的情况下切换索引：

```python
class IndexAliasManager:
    def __init__(self, alias_name: str = "main"):
        self.alias = f"{settings.ES_INDEX_PREFIX}_{alias_name}"

    def switch_alias(self, new_index: str):
        """零停机切换索引"""
        old_indices = es.indices.get_alias(index=self.alias, ignore=[404])

        actions = []
        for old_index in old_indices:
            actions.append({"remove": {"index": old_index, "alias": self.alias}})
        actions.append({"add": {"index": new_index, "alias": self.alias}})

        es.indices.update_aliases({"actions": actions})

    def reindex_with_alias(self, new_mapping: dict = None):
        """重建索引并切换别名"""
        import time
        timestamp = int(time.time())
        new_index = f"{self.alias}_v{timestamp}"

        # 创建新索引
        if new_mapping:
            es.indices.create(index=new_index, body=new_mapping)
        else:
            es.indices.create(index=new_index)

        # 从旧索引 reindex 到新索引
        old_indices = es.indices.get_alias(index=self.alias, ignore=[404])
        for old_index in old_indices:
            es.reindex({"source": {"index": old_index}, "dest": {"index": new_index}},
                       wait_for_completion=True)

        # 切换别名
        self.switch_alias(new_index)

        # 删除旧索引
        for old_index in old_indices:
            es.indices.delete(index=old_index, ignore=[404])

        return new_index
```

别名管理的好处：mapping 变更时不需要停机；数据迁移时可以平滑切换；回滚只需再次切换别名到旧索引。

## 核心搜索服务

```python
# app/services/searcher.py
from app.services.indexer import es
from app.models import SearchRequest, SearchResponse, SortBy
from app.config import settings
import time

class SearchService:
    SORT_MAP = {SortBy.RELEVANCE: ["_score"], SortBy.PRICE_ASC: [{"price": "asc"}],
                SortBy.PRICE_DESC: [{"price": "desc"}], SortBy.SALES: [{"sales_count": "desc"}],
                SortBy.RATING: [{"rating": "desc"}]}

    def search(self, request: SearchRequest) -> SearchResponse:
        start = time.time()
        filters = []
        for field, value in request.filters.items():
            if field == "price_range":
                filters.append({"range": {"price": value}})
            elif isinstance(value, list):
                filters.append({"terms": {field: value}})
            else:
                filters.append({"term": {field: value}})

        body = {"query": {"bool": {
                    "must": [{"multi_match": {"query": request.query,
                              "fields": ["title^3", "content", "tags^2"],
                              "minimum_should_match": "75%"}}],
                    "filter": filters}},
                "sort": self.SORT_MAP.get(request.sort_by, ["_score"]),
                "from": (request.page - 1) * request.page_size,
                "size": request.page_size,
                "aggs": {"categories": {"terms": {"field": "category", "size": 15}},
                         "tags": {"terms": {"field": "tags", "size": 20}}}}

        if request.enable_highlight:
            body["highlight"] = {"fields": {"title": {}, "content": {"fragment_size": 150}},
                                 "pre_tags": ["<mark>"], "post_tags": ["</mark>"]}

        result = es.search(index=f"{settings.ES_INDEX_PREFIX}_*", body=body,
                           timeout=f"{settings.SEARCH_TIMEOUT}s")
        return SearchResponse(
            query=request.query,
            total_hits=result["hits"]["total"]["value"],
            results=[{"id": h["_id"], "score": h["_score"], **h["_source"],
                       "highlights": h.get("highlight", {})} for h in result["hits"]["hits"]],
            facets={n: [{"value": b["key"], "count": b["doc_count"]} for b in a.get("buckets", [])]
                    for n, a in result.get("aggregations", {}).items()},
            took_ms=int((time.time() - start) * 1000))

search_service = SearchService()
```

搜索服务的核心是多字段加权搜索：标题权重 3 倍、标签 2 倍、内容 1 倍。

### 搜索请求的生命周期

一个搜索请求从进入到返回，经历以下步骤：

1. **输入验证**：Pydantic 自动验证参数类型和范围。
2. **查询预处理**：去除特殊字符、trim 空白、分词。
3. **纠错检查**：调用 CorrectionService 检查拼写。
4. **缓存查询**：检查 Redis 缓存是否有结果。
5. **构建查询**：根据请求参数构建 ES 查询 DSL。
6. **执行搜索**：发送请求到 ES，设置超时。
7. **结果处理**：提取搜索结果、高亮、分面数据。
8. **AI 增强**（可选）：生成 AI 摘要、重排序。
9. **记录指标**：异步记录搜索事件到 MetricsService。
10. **返回响应**：构建 SearchResponse 返回给用户。

```python
class SearchPipeline:
    def __init__(self, search_service, correction_service, metrics_service, cache):
        self.search_service = search_service
        self.correction_service = correction_service
        self.metrics_service = metrics_service
        self.cache = cache

    async def execute(self, request: SearchRequest) -> SearchResponse:
        # 步骤1：输入清洗
        cleaned_query = self._clean_query(request.query)
        request = request.model_copy(update={"query": cleaned_query})

        # 步骤2：检查缓存
        cache_key = self._build_cache_key(request)
        cached = self.cache.get(cache_key)
        if cached:
            return cached

        # 步骤3：拼写纠错
        corrected = self.correction_service.check_spelling(request.query)
        if corrected["has_correction"]:
            request = request.model_copy(update={"query": corrected["corrected"]})

        # 步骤4：执行搜索
        response = self.search_service.search(request)
        response.corrected_query = corrected["corrected"] if corrected["has_correction"] else None

        # 步骤5：缓存结果
        self.cache.set(cache_key, response, ttl=settings.CACHE_TTL)

        # 步骤6：记录指标（异步）
        self.metrics_service.log_search(query=request.query, result_count=response.total_hits)

        return response

    def _clean_query(self, query: str) -> str:
        import re
        query = query.strip()
        query = re.sub(r'[^\w\s\u4e00-\u9fff]', '', query)
        return query[:200]

    def _build_cache_key(self, request: SearchRequest) -> str:
        import hashlib, json
        raw = json.dumps({
            "q": request.query, "f": request.filters,
            "s": request.sort_by, "p": request.page,
            "ps": request.page_size
        }, sort_keys=True)
        return f"search:{hashlib.md5(raw.encode()).hexdigest()}"
```

Pipeline 模式让搜索流程清晰可测试，每一步都可以独立替换或跳过。

## 搜索建议与纠错

```python
# app/services/suggester.py
from app.services.indexer import es
from app.config import settings

class SuggestService:
    def get_suggestions(self, prefix: str, size: int = 8):
        result = es.search(index=f"{settings.ES_INDEX_PREFIX}_*", body={
            "suggest": {"s": {"prefix": prefix, "completion": {
                "field": "suggest_field", "size": size,
                "skip_duplicates": True, "fuzzy": {"fuzziness": "AUTO"}}}}
        })
        return [o["text"] for o in result["suggest"]["s"][0]["options"]]

suggest_service = SuggestService()

class CorrectionService:
    def check_spelling(self, query: str) -> dict:
        tokens = query.lower().split()
        corrections, has_correction = [], False
        for token in tokens:
            result = es.search(index=f"{settings.ES_INDEX_PREFIX}_dictionary",
                body={"suggest": {"spell": {"text": token, "term": {
                    "field": "term", "suggest_mode": "popular", "max_edits": 2}}}, "size": 0})
            options = result["suggest"]["spell"][0]["options"]
            if options and options[0]["text"] != token:
                corrections.append({"original": token, "suggestion": options[0]["text"]})
                has_correction = True
            else:
                corrections.append({"original": token, "suggestion": token})
        return {"original": query, "corrected": " ".join(c["suggestion"] for c in corrections),
                "has_correction": has_correction}

correction_service = CorrectionService()
```

## 质量监控服务

```python
# app/services/metrics.py
import json, redis
from datetime import datetime
from app.config import settings

redis_client = redis.Redis(host=settings.REDIS_HOST, port=settings.REDIS_PORT, db=1)

class MetricsService:
    def log_search(self, query: str, result_count: int):
        redis_client.lpush("search_events", json.dumps(
            {"query": query, "result_count": result_count, "ts": datetime.utcnow().isoformat()}))
        redis_client.ltrim("search_events", 0, 9999)
        if result_count == 0:
            redis_client.zincrby("zero_result_queries", 1, query)
        redis_client.zincrby("query_frequency", 1, query)

    def log_click(self, query: str, doc_id: str, position: int):
        redis_client.lpush("click_events", json.dumps(
            {"query": query, "doc_id": doc_id, "position": position}))

    def get_stats(self) -> dict:
        zero = redis_client.zrevrange("zero_result_queries", 0, 19, withscores=True)
        popular = redis_client.zrevrange("query_frequency", 0, 19, withscores=True)
        return {"total_searches": redis_client.llen("search_events"),
                "top_zero_queries": [{"query": q.decode(), "count": int(s)} for q, s in zero],
                "top_popular_queries": [{"query": q.decode(), "count": int(s)} for q, s in popular]}

metrics_service = MetricsService()
```

### 指标的持久化

Redis 存储的是实时数据，需要定期持久化到 ES 用于长期分析：

```python
def persist_metrics_to_es():
    """定期将 Redis 指标持久化到 ES"""
    import json
    from datetime import datetime

    # 读取搜索事件
    events = redis_client.lrange("search_events", 0, -1)
    if not events:
        return

    index_name = f"search_metrics-{datetime.now().strftime('%Y.%m.%d')}"
    actions = []
    for event_bytes in events:
        event = json.loads(event_bytes)
        actions.append({
            "_index": index_name,
            "_source": {
                "@timestamp": event["ts"],
                "query": event["query"],
                "result_count": event["result_count"]
            }
        })

    if actions:
        from elasticsearch.helpers import bulk
        bulk(es, actions, request_timeout=30)

    # 读取点击事件
    clicks = redis_client.lrange("click_events", 0, -1)
    click_actions = []
    for click_bytes in clicks:
        click = json.loads(click_bytes)
        click_actions.append({
            "_index": f"search_clicks-{datetime.now().strftime('%Y.%m.%d')}",
            "_source": click
        })

    if click_actions:
        from elasticsearch.helpers import bulk
        bulk(es, click_actions, request_timeout=30)
```

这个函数应该由定时任务（如 Celery Beat 或 cron）每小时执行一次。持久化后可以清空 Redis 中的旧数据。

### 中间件与异常处理

生产环境的 API 需要完善的中间件和异常处理：

```python
from fastapi import Request, HTTPException
from fastapi.responses import JSONResponse
from fastapi.middleware.cors import CORSMiddleware
import time
import logging

logger = logging.getLogger("search-api")

# CORS 中间件
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 请求计时中间件
@app.middleware("http")
async def timing_middleware(request: Request, call_next):
    start = time.time()
    response = await call_next(request)
    duration = (time.time() - start) * 1000
    response.headers["X-Response-Time"] = f"{duration:.2f}ms"

    if duration > 500:
        logger.warning(f"慢请求: {request.url.path} 耗时 {duration:.2f}ms")

    return response

# 全局异常处理
@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
    logger.error(f"未处理异常: {request.url.path} - {exc}", exc_info=True)
    return JSONResponse(
        status_code=500,
        content={"error": "内部服务错误", "detail": str(exc)}
    )

@app.exception_handler(HTTPException)
async def http_exception_handler(request: Request, exc: HTTPException):
    return JSONResponse(
        status_code=exc.status_code,
        content={"error": exc.detail}
    )

# 限流中间件
from collections import defaultdict
import time as time_module

rate_limit_store = defaultdict(list)

@app.middleware("http")
async def rate_limit_middleware(request: Request, call_next):
    client_ip = request.client.host
    now = time_module.time()

    # 清理 1 分钟前的记录
    rate_limit_store[client_ip] = [
        t for t in rate_limit_store[client_ip] if now - t < 60
    ]

    if len(rate_limit_store[client_ip]) >= settings.RATE_LIMIT_PER_MINUTE:
        return JSONResponse(
            status_code=429,
            content={"error": "请求过于频繁，请稍后再试"}
        )

    rate_limit_store[client_ip].append(now)
    return await call_next(request)
```

中间件的执行顺序很重要：CORS → 限流 → 计时 → 业务逻辑。限流基于客户端 IP，每分钟最多 60 次请求。

### 健康检查与监控

```python
@app.get("/api/health/detailed")
async def detailed_health():
    checks = {}

    # ES 健康检查
    try:
        es_health = es.cluster.health()
        checks["elasticsearch"] = {
            "status": es_health["status"],
            "nodes": es_health["number_of_nodes"]
        }
    except Exception as e:
        checks["elasticsearch"] = {"status": "unhealthy", "error": str(e)}

    # Redis 健康检查
    try:
        redis_client.ping()
        checks["redis"] = {"status": "healthy"}
    except Exception as e:
        checks["redis"] = {"status": "unhealthy", "error": str(e)}

    # 搜索性能检查
    try:
        import time
        start = time.time()
        es.search(index=f"{settings.ES_INDEX_PREFIX}_*", body={"query": {"match_all": {}}, "size": 1})
        latency = (time.time() - start) * 1000
        checks["search_latency"] = {
            "ms": round(latency, 2),
            "status": "healthy" if latency < 200 else "degraded"
        }
    except Exception as e:
        checks["search_latency"] = {"status": "unhealthy", "error": str(e)}

    overall = all(c.get("status") in ("healthy", "degraded") for c in checks.values() if isinstance(c, dict))

    return {
        "status": "healthy" if overall else "unhealthy",
        "checks": checks
    }
```

健康检查覆盖三个维度：ES 集群状态、Redis 连接、搜索延迟。可以接入 Kubernetes 的 liveness 和 readiness 探针。

## 常见误区

1. **搜索和管理 API 不分离**：搜索接口需要高可用低延迟，管理接口需要写权限，应该分开部署。
2. **不做请求验证**：用户输入可能包含特殊字符或超长字符串，必须做输入验证。
3. **忽略超时设置**：ES 查询可能超时，必须设置合理的超时时间。
4. **日志和监控缺失**：没有搜索日志就无法分析用户行为和优化质量。
5. **不做降级处理**：当 AI 服务不可用时，应该降级到传统搜索。
6. **缓存策略不当**：缓存粒度太粗会导致脏数据，太细则缓存命中率低。
7. **不做容量规划**：索引增长后分片数不够，查询性能会急剧下降。

## 工程建议

1. **用 Pydantic 做请求验证**：FastAPI + Pydantic 自动验证参数，减少错误处理代码。
2. **搜索结果缓存**：对高频查询做缓存，TTL 设为 5-10 分钟。
3. **异步日志记录**：搜索日志应该异步写入，不阻塞搜索响应。
4. **API 版本化**：搜索 API 应该带版本号（如 `/api/v1/search`）。
5. **限流和熔断**：对搜索 API 做限流保护，对 AI 服务调用做熔断降级。
6. **灰度发布**：新版排序算法先对 5% 流量生效，观察指标后再全量。
7. **文档和 SDK**：为搜索 API 提供 OpenAPI 文档和 Python/JS SDK。
8. **备份与恢复**：定期备份 ES 索引数据，测试恢复流程。

## 小结

一个完整的搜索引擎包含索引管理、多模式检索、排序、体验优化和质量监控五大模块。FastAPI 提供高性能 API 框架，Elasticsearch 处理核心索引和检索，Redis 缓存建议和指标。生产环境中还需要关注部署、监控、限流和降级。

### 练习一

**思路**：用 Redis List 存储每个用户的搜索历史，LPUSH 添加，LTRIM 限制长度。

**答案**：

```python
import redis, json
redis_client = redis.Redis(host="localhost", port=6379, db=1)

def record_history(user_id: str, query: str):
    key = f"search_history:{user_id}"
    redis_client.lpush(key, json.dumps({"query": query}))
    redis_client.ltrim(key, 0, 19)
    redis_client.expire(key, 86400 * 30)

def get_history(user_id: str, limit: int = 20):
    key = f"search_history:{user_id}"
    return [json.loads(item) for item in redis_client.lrange(key, 0, limit - 1)]

def clear_history(user_id: str):
    redis_client.delete(f"search_history:{user_id}")
```

**要点**：Redis List 的 LPUSH + LTRIM 天然适合"最近 N 条"场景。

### 练习二

**思路**：用查询参数的哈希作为缓存键，搜索前先查缓存。

**答案**：

```python
import hashlib, json, redis
redis_client = redis.Redis(host="localhost", port=6379, db=2)

def cache_key(query: str, filters: dict, sort_by: str, page: int) -> str:
    raw = json.dumps({"q": query, "f": filters, "s": sort_by, "p": page}, sort_keys=True)
    return f"search_cache:{hashlib.md5(raw.encode()).hexdigest()}"

def get_cached(key: str):
    cached = redis_client.get(key)
    return json.loads(cached) if cached else None

def set_cache(key: str, result: dict, ttl: int = 300):
    redis_client.setex(key, ttl, json.dumps(result))
```

**要点**：缓存键用参数的 MD5 哈希保证唯一性。

### 练习四

**思路**：用令牌桶算法实现限流，每个用户一个桶，定期补充令牌。

**答案**：

```python
import time
import redis

redis_client = redis.Redis(host="localhost", port=6379, db=3)

class TokenBucketRateLimiter:
    def __init__(self, max_tokens: int = 60, refill_rate: float = 1.0):
        self.max_tokens = max_tokens
        self.refill_rate = refill_rate  # 每秒补充的令牌数

    def is_allowed(self, user_id: str) -> bool:
        key = f"rate_limit:{user_id}"
        now = time.time()

        pipe = redis_client.pipeline()
        pipe.hgetall(key)
        result = pipe.execute()[0]

        if result:
            tokens = float(result.get(b"tokens", self.max_tokens))
            last_refill = float(result.get(b"last_refill", now))
        else:
            tokens = self.max_tokens
            last_refill = now

        # 补充令牌
        elapsed = now - last_refill
        tokens = min(self.max_tokens, tokens + elapsed * self.refill_rate)

        if tokens >= 1:
            tokens -= 1
            allowed = True
        else:
            allowed = False

        # 更新状态
        pipe = redis_client.pipeline()
        pipe.hset(key, mapping={"tokens": tokens, "last_refill": now})
        pipe.expire(key, 120)
        pipe.execute()

        return allowed

rate_limiter = TokenBucketRateLimiter(max_tokens=60, refill_rate=1.0)

# 在 FastAPI 中使用
@app.middleware("http")
async def token_bucket_rate_limit(request: Request, call_next):
    user_id = request.headers.get("X-User-ID", request.client.host)

    if not rate_limiter.is_allowed(user_id):
        return JSONResponse(
            status_code=429,
            content={"error": "请求频率超限，请稍后再试"},
            headers={"Retry-After": "1"}
        )

    return await call_next(request)
```

**要点**：令牌桶算法比固定窗口更平滑：每秒补充 1 个令牌，最多积累 60 个。短时间的突发请求可以消耗积累的令牌，但持续高频请求会被限制。Redis 的原子操作保证分布式环境下的正确性。
