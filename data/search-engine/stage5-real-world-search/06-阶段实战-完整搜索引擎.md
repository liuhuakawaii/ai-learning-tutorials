## 架构设计

```
用户 → FastAPI → SearchService → Elasticsearch
                → SuggestService → Redis Cache
                → CorrectionService
                → MetricsService → Redis
```

四层架构：接入层（API 网关、限流）、业务层（搜索/建议/纠错/监控）、存储层（ES + Redis）、基础设施层（日志、监控、CI/CD）。

```python
from pydantic import BaseModel, Field
from typing import Optional
from enum import Enum

class SortBy(str, Enum):
    RELEVANCE = "relevance"
    PRICE_ASC = "price_asc"
    PRICE_DESC = "price_desc"
    SALES = "sales"
    RATING = "rating"

class SearchRequest(BaseModel):
    query: str = Field(..., min_length=1, max_length=200)
    filters: dict = Field(default_factory=dict)
    sort_by: SortBy = SortBy.RELEVANCE
    page: int = Field(default=1, ge=1)
    page_size: int = Field(default=20, ge=1, le=100)
    enable_highlight: bool = True

class SearchResponse(BaseModel):
    query: str
    corrected_query: Optional[str] = None
    total_hits: int
    results: list[dict]
    facets: dict = Field(default_factory=dict)
    suggestions: list[str] = Field(default_factory=list)
    took_ms: int
```

Pydantic 模型的字段验证不是装饰：`min_length=1` 防止空查询，`max_length=200` 防止恶意输入，`le=100` 限制单次返回数量。

## 索引管理

```python
# app/services/indexer.py
from elasticsearch import Elasticsearch
from elasticsearch.helpers import bulk
from app.config import settings

es = Elasticsearch(settings.ES_HOST)

class IndexManager:
    def __init__(self, index_name="main"):
        self.index = f"{settings.ES_INDEX_PREFIX}_{index_name}"

    def create_index(self, recreate=False):
        if recreate:
            es.indices.delete(index=self.index, ignore=[404])
        mapping = {"mappings": {"properties": {
            "title": {"type": "text", "analyzer": "ik_max_word",
                      "search_analyzer": "ik_smart"},
            "content": {"type": "text", "analyzer": "ik_max_word"},
            "category": {"type": "keyword"},
            "tags": {"type": "keyword"},
            "price": {"type": "float"},
            "sales_count": {"type": "integer"},
            "rating": {"type": "float"},
            "created_at": {"type": "date"}
        }}, "settings": {"number_of_shards": 1, "number_of_replicas": 0}}
        es.indices.create(index=self.index, body=mapping, ignore=400)

    def bulk_index(self, documents):
        actions = [{"_index": self.index, "_id": d.get("id"), "_source": d}
                   for d in documents]
        return bulk(es, actions, request_timeout=30)
```

### 索引别名管理

生产环境中直接用索引名查询是不推荐的。应该使用别名，这样可以在不停机的情况下切换索引：

```python
class IndexAliasManager:
    def __init__(self, alias_name="main"):
        self.alias = f"{settings.ES_INDEX_PREFIX}_{alias_name}"

    def switch_alias(self, new_index):
        old_indices = es.indices.get_alias(index=self.alias, ignore=[404])
        actions = [{"remove": {"index": old, "alias": self.alias}}
                   for old in old_indices]
        actions.append({"add": {"index": new_index, "alias": self.alias}})
        es.indices.update_aliases({"actions": actions})

    def reindex_with_alias(self, new_mapping=None):
        import time
        new_index = f"{self.alias}_v{int(time.time())}"
        if new_mapping:
            es.indices.create(index=new_index, body=new_mapping)
        else:
            es.indices.create(index=new_index)
        old_indices = es.indices.get_alias(index=self.alias, ignore=[404])
        for old in old_indices:
            es.reindex({"source": {"index": old}, "dest": {"index": new_index}},
                       wait_for_completion=True)
        self.switch_alias(new_index)
        for old in old_indices:
            es.indices.delete(index=old, ignore=[404])
        return new_index
```

## 核心搜索服务

```python
# app/services/searcher.py
from app.services.indexer import es
from app.models import SearchRequest, SearchResponse, SortBy
from app.config import settings
import time

class SearchService:
    SORT_MAP = {
        SortBy.RELEVANCE: ["_score"],
        SortBy.PRICE_ASC: [{"price": "asc"}],
        SortBy.PRICE_DESC: [{"price": "desc"}],
        SortBy.SALES: [{"sales_count": "desc"}],
        SortBy.RATING: [{"rating": "desc"}],
    }

    def search(self, request: SearchRequest) -> SearchResponse:
        start = time.time()
        filters = []
        for field, value in request.filters.items():
            if field == "price_range":
                filters.append({"range": {"price": value}})
            elif isinstance(value, list):
                filters.append({"terms": {field: value}})
            else:
                filters.append({"term": {field: value}})

        body = {
            "query": {"bool": {
                "must": [{"multi_match": {
                    "query": request.query,
                    "fields": ["title^3", "content", "tags^2"],
                    "minimum_should_match": "75%"
                }}],
                "filter": filters
            }},
            "sort": self.SORT_MAP.get(request.sort_by, ["_score"]),
            "from": (request.page - 1) * request.page_size,
            "size": request.page_size,
            "aggs": {
                "categories": {"terms": {"field": "category", "size": 15}},
                "tags": {"terms": {"field": "tags", "size": 20}}
            }
        }

        if request.enable_highlight:
            body["highlight"] = {
                "fields": {"title": {}, "content": {"fragment_size": 150}},
                "pre_tags": ["<mark>"], "post_tags": ["</mark>"]
            }

        result = es.search(index=f"{settings.ES_INDEX_PREFIX}_*", body=body,
                           timeout=f"{settings.SEARCH_TIMEOUT}s")

        return SearchResponse(
            query=request.query,
            total_hits=result["hits"]["total"]["value"],
            results=[{"id": h["_id"], "score": h["_score"], **h["_source"],
                       "highlights": h.get("highlight", {})}
                     for h in result["hits"]["hits"]],
            facets={n: [{"value": b["key"], "count": b["doc_count"]}
                        for b in a.get("buckets", [])]
                    for n, a in result.get("aggregations", {}).items()},
            took_ms=int((time.time() - start) * 1000))
```

## 搜索建议与纠错

```python
# app/services/suggester.py
from app.services.indexer import es
from app.config import settings

class SuggestService:
    def get_suggestions(self, prefix, size=8):
        result = es.search(index=f"{settings.ES_INDEX_PREFIX}_*", body={
            "suggest": {"s": {"prefix": prefix, "completion": {
                "field": "suggest_field", "size": size,
                "skip_duplicates": True, "fuzzy": {"fuzziness": "AUTO"}}}}
        })
        return [o["text"] for o in result["suggest"]["s"][0]["options"]]


class CorrectionService:
    def check_spelling(self, query):
        tokens = query.lower().split()
        corrections, has_correction = [], False
        for token in tokens:
            result = es.search(
                index=f"{settings.ES_INDEX_PREFIX}_dictionary",
                body={"suggest": {"spell": {"text": token, "term": {
                    "field": "term", "suggest_mode": "popular",
                    "max_edits": 2}}}, "size": 0})
            options = result["suggest"]["spell"][0]["options"]
            if options and options[0]["text"] != token:
                corrections.append(options[0]["text"])
                has_correction = True
            else:
                corrections.append(token)
        return {"original": query, "corrected": " ".join(corrections),
                "has_correction": has_correction}
```

## 质量监控

```python
# app/services/metrics.py
import json, redis
from datetime import datetime
from app.config import settings

redis_client = redis.Redis(host=settings.REDIS_HOST, port=settings.REDIS_PORT, db=1)

class MetricsService:
    def log_search(self, query, result_count):
        redis_client.lpush("search_events", json.dumps(
            {"query": query, "result_count": result_count,
             "ts": datetime.utcnow().isoformat()}))
        redis_client.ltrim("search_events", 0, 9999)
        if result_count == 0:
            redis_client.zincrby("zero_result_queries", 1, query)
        redis_client.zincrby("query_frequency", 1, query)

    def get_stats(self):
        zero = redis_client.zrevrange("zero_result_queries", 0, 19, withscores=True)
        popular = redis_client.zrevrange("query_frequency", 0, 19, withscores=True)
        return {
            "total_searches": redis_client.llen("search_events"),
            "top_zero_queries": [{"query": q.decode(), "count": int(s)}
                                 for q, s in zero],
            "top_popular_queries": [{"query": q.decode(), "count": int(s)}
                                    for q, s in popular]
        }
```

## FastAPI 入口

```python
# app/main.py
from fastapi import FastAPI, Query
from app.models import SearchRequest, SearchResponse
from app.services.searcher import SearchService
from app.services.suggester import SuggestService
from app.services.corrector import CorrectionService
from app.services.metrics import MetricsService

app = FastAPI(title="搜索引擎 API", version="1.0.0")
search_service = SearchService()
suggest_service = SuggestService()
correction_service = CorrectionService()
metrics_service = MetricsService()

@app.post("/api/search", response_model=SearchResponse)
async def search(request: SearchRequest):
    corrected = correction_service.check_spelling(request.query)
    if corrected["has_correction"]:
        request = request.model_copy(update={"query": corrected["corrected"]})
    response = search_service.search(request)
    response.corrected_query = (corrected["corrected"]
                                if corrected["has_correction"] else None)
    metrics_service.log_search(request.query, response.total_hits)
    if response.total_hits == 0:
        response.suggestions = suggest_service.get_suggestions(request.query)
    return response

@app.post("/api/suggest")
async def suggest(prefix: str, size: int = Query(default=8, le=20)):
    return {"suggestions": suggest_service.get_suggestions(prefix, size)}

@app.get("/api/metrics/stats")
async def get_stats():
    return metrics_service.get_stats()

@app.get("/api/health")
async def health():
    from app.services.indexer import es
    try:
        es.info()
        return {"status": "healthy"}
    except Exception as e:
        return {"status": "unhealthy", "error": str(e)}
```

### 中间件

```python
from fastapi import Request
from fastapi.responses import JSONResponse
from fastapi.middleware.cors import CORSMiddleware
import time, logging

logger = logging.getLogger("search-api")

app.add_middleware(CORSMiddleware, allow_origins=["*"],
                   allow_methods=["*"], allow_headers=["*"])

@app.middleware("http")
async def timing_middleware(request: Request, call_next):
    start = time.time()
    response = await call_next(request)
    duration = (time.time() - start) * 1000
    response.headers["X-Response-Time"] = f"{duration:.2f}ms"
    if duration > 500:
        logger.warning(f"慢请求: {request.url.path} {duration:.2f}ms")
    return response
```

## 常见误区

**搜索和管理 API 不分离。** 搜索接口需要高可用低延迟，管理接口需要写权限，应该分开部署。

**不做请求验证。** 用户输入可能包含特殊字符或超长字符串。

**不做降级处理。** 当 AI 服务不可用时，应该降级到传统搜索。

**缓存策略不当。** 缓存粒度太粗会导致脏数据，太细则缓存命中率低。

## 工程建议

- 搜索结果缓存：对高频查询做缓存，TTL 5-10 分钟
- 异步日志记录：搜索日志应该异步写入，不阻塞搜索响应
- API 版本化：搜索 API 应该带版本号（如 `/api/v1/search`）
- 限流和熔断：对搜索 API 做限流保护，对 AI 服务调用做熔断降级

### 练习一：搜索结果缓存

用 Redis 实现搜索结果缓存，缓存键为查询+筛选+排序+分页的组合，TTL 为 5 分钟。

```python
import hashlib, json, redis

redis_client = redis.Redis(host="localhost", port=6379, db=2)

def cache_key(query, filters, sort_by, page):
    raw = json.dumps({"q": query, "f": filters, "s": sort_by, "p": page},
                     sort_keys=True)
    return f"search_cache:{hashlib.md5(raw.encode()).hexdigest()}"

def get_cached(key):
    cached = redis_client.get(key)
    return json.loads(cached) if cached else None

def set_cache(key, result, ttl=300):
    redis_client.setex(key, ttl, json.dumps(result))
```

### 练习三

```python
import time
import redis

redis_client = redis.Redis(host="localhost", port=6379, db=3)

class TokenBucketRateLimiter:
    def __init__(self, max_tokens=60, refill_rate=1.0):
        self.max_tokens = max_tokens
        self.refill_rate = refill_rate

    def is_allowed(self, user_id):
        key = f"rate_limit:{user_id}"
        now = time.time()
        pipe = redis_client.pipeline()
        pipe.hgetall(key)
        result = pipe.execute()[0]

        if result:
            tokens = float(result.get(b"tokens", self.max_tokens))
            last_refill = float(result.get(b"last_refill", now))
        else:
            tokens = self.max_tokens
            last_refill = now

        elapsed = now - last_refill
        tokens = min(self.max_tokens, tokens + elapsed * self.refill_rate)

        if tokens >= 1:
            tokens -= 1
            allowed = True
        else:
            allowed = False

        pipe = redis_client.pipeline()
        pipe.hset(key, mapping={"tokens": tokens, "last_refill": now})
        pipe.expire(key, 120)
        pipe.execute()
        return allowed
```

令牌桶算法比固定窗口更平滑：每秒补充 1 个令牌，最多积累 60 个。短时间的突发请求可以消耗积累的令牌，但持续高频请求会被限制。
