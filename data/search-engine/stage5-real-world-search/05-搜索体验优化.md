# 搜索体验优化：让用户更快找到想要的内容

## 场景引入

用户在搜索框输入"pythn"，系统自动提示"您是否要搜索 Python？"。点击搜索后，关键词在结果中被高亮显示。左侧展示分面导航帮助筛选。如果搜索没有结果，页面展示热门推荐和相关搜索建议。

这些细节构成了搜索体验。搜索体验优化不改变检索算法本身，而是通过搜索建议、拼写纠错、高亮、零结果处理等功能，让用户更快、更舒服地找到想要的内容。

## 学习目标

1. 实现搜索建议（auto-suggest）功能
2. 构建拼写纠错（spell check）系统
3. 实现搜索关键词高亮（highlight）
4. 设计零结果页面的降级策略
5. 理解搜索体验优化的性能权衡

## 搜索体验的核心原则

搜索体验优化的核心原则来自 Nielsen Norman Group 的可用性研究：

1. **即时反馈**：用户每输入一个字符，搜索框应该在 100ms 内给出建议。超过这个延迟，用户会感觉系统"卡"了。
2. **容错性**：用户会犯错——拼写错误、用词不精确、甚至输入完全无关的内容。好的搜索体验应该容忍这些错误。
3. **信息 scent**：搜索结果的摘要、高亮、分面等信息应该给用户足够的"气味"，让他们判断是否值得点击。
4. **逃生通道**：当搜索没有结果时，必须提供替代路径（纠错、模糊匹配、热门推荐），不能让用户走进死胡同。

## 搜索建议（Auto-Suggest）

用户输入时实时给出候选词，减少输入成本：

```python
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

def setup_suggest_index():
    mapping = {
        "mappings": {
            "properties": {
                "query_text": {"type": "text", "fields": {"keyword": {"type": "keyword"}}},
                "query_suggest": {"type": "completion", "analyzer": "simple"},
                "frequency": {"type": "integer"},
                "category": {"type": "keyword"}
            }
        }
    }
    es.indices.create(index="search_suggestions", body=mapping, ignore=400)

def get_suggestions(prefix: str, category: str = None, size: int = 8) -> list[dict]:
    body = {
        "suggest": {
            "title_suggest": {
                "prefix": prefix,
                "completion": {
                    "field": "query_suggest",
                    "size": size,
                    "skip_duplicates": True,
                    "fuzzy": {"fuzziness": "AUTO", "min_length": 2}
                }
            }
        }
    }
    if category:
        body["query"] = {"term": {"category": category}}

    result = es.search(index="search_suggestions", body=body)
    return [
        {"text": opt["text"], "score": opt["_score"]}
        for opt in result["suggest"]["title_suggest"][0]["options"]
    ]

def record_query(query: str, category: str = "general"):
    existing = es.search(
        index="search_suggestions",
        body={"query": {"term": {"query_text.keyword": query}}}
    )
    if existing["hits"]["hits"]:
        doc_id = existing["hits"]["hits"][0]["_id"]
        es.update(index="search_suggestions", id=doc_id,
                  body={"script": {"source": "ctx._source.frequency += 1"}})
    else:
        es.index(index="search_suggestions", body={
            "query_text": query,
            "query_suggest": {"input": [query], "weight": 1},
            "frequency": 1, "category": category
        })
```

ES 的 Completion Suggester 基于 FST 数据结构，前缀匹配速度极快。`weight` 让高频查询排名更靠前。`fuzzy` 容忍拼写错误。

### Completion Suggester 的内部原理

ES 的 Completion Suggester 使用 FST（Finite State Transducer）数据结构，这是一种特殊的有限状态自动机：

- **时间复杂度**：前缀匹配的时间复杂度是 O(k)，k 是前缀长度，与词典大小无关。这意味着即使有百万级查询词，响应时间也几乎不变。
- **空间效率**：FST 比哈希表节省约 10-20 倍的内存。
- **局限性**：FST 只支持前缀匹配，不支持中间匹配或后缀匹配。

```python
# FST 的局限性演示
# 输入 "python" 能匹配 "python 教程"、"python 入门"
# 但输入 "教程" 无法匹配 "python 教程"（不是前缀）

# 如果需要支持中间匹配，需要用 match_phrase_prefix
def get_suggestions_mid_match(prefix: str, size: int = 8):
    body = {
        "query": {
            "match_phrase_prefix": {
                "query_text": {
                    "query": prefix,
                    "slop": 2
                }
            }
        },
        "sort": [{"frequency": {"order": "desc"}}],
        "size": size
    }
    result = es.search(index="search_suggestions", body=body)
    return [h["_source"]["query_text"] for h in result["hits"]["hits"]]
```

### 搜索建议的性能优化

搜索建议接口的 QPS 通常是搜索接口的 5-10 倍（用户每输入一个字符都触发一次），需要特别注意性能：

```python
import time
from functools import lru_cache
import hashlib

class SuggestionCache:
    def __init__(self, ttl: int = 300):
        self.cache = {}
        self.ttl = ttl

    def get(self, prefix: str) -> list[dict] | None:
        key = prefix.lower().strip()
        if key in self.cache:
            result, timestamp = self.cache[key]
            if time.time() - timestamp < self.ttl:
                return result
            del self.cache[key]
        return None

    def set(self, prefix: str, result: list[dict]):
        key = prefix.lower().strip()
        self.cache[key] = (result, time.time())

    def cleanup(self):
        now = time.time()
        expired = [k for k, (_, ts) in self.cache.items() if now - ts > self.ttl]
        for k in expired:
            del self.cache[k]

suggest_cache = SuggestionCache(ttl=300)

def get_suggestions_cached(prefix: str, size: int = 8) -> list[dict]:
    cached = suggest_cache.get(prefix)
    if cached is not None:
        return cached

    result = get_suggestions(prefix, size=size)
    suggest_cache.set(prefix, result)
    return result

# 前端防抖：建议在前端实现 debounce（300ms），减少请求次数
# 前端缓存：相同前缀的结果在前端也做缓存，避免重复请求
```

关键优化：服务端缓存热门前缀的结果（5 分钟 TTL），前端实现 debounce（300ms）和本地缓存。这样大部分请求直接从缓存返回，ES 只处理冷查询。

## 拼写纠错（Spell Check）

当用户输入错误时，给出纠正建议：

```python
def spell_check(query: str) -> dict:
    tokens = query.lower().split()
    corrections = []
    has_correction = False

    for token in tokens:
        result = es.search(
            index="dictionary", body={
                "suggest": {
                    "spell": {
                        "text": token,
                        "term": {"field": "term", "suggest_mode": "popular", "max_edits": 2}
                    }
                }, "size": 0
            }
        )
        options = result["suggest"]["spell"][0]["options"]
        if options and options[0]["text"] != token:
            corrections.append({"original": token, "suggestion": options[0]["text"]})
            has_correction = True
        else:
            corrections.append({"original": token, "suggestion": token})

    return {
        "original_query": query,
        "corrected_query": " ".join(c["suggestion"] for c in corrections),
        "has_correction": has_correction
    }

def search_with_correction(query: str, index: str = "products"):
    body = {"query": {"match": {"title": query}}, "size": 5}
    result = es.search(index=index, body=body)

    if result["hits"]["total"]["value"] < 3:
        correction = spell_check(query)
        if correction["has_correction"]:
            corrected_body = {"query": {"match": {"title": correction["corrected_query"]}}, "size": 5}
            corrected_result = es.search(index=index, body=corrected_body)
            return {"results": corrected_result["hits"]["hits"],
                    "correction": correction, "used_correction": True}

    return {"results": result["hits"]["hits"], "correction": None, "used_correction": False}
```

采用"先搜后纠"策略：先用原始查询搜索，只有结果不足时才触发纠错，避免对正确查询的误纠正。

### 编辑距离与模糊匹配

ES 的 Term Suggester 基于编辑距离（Levenshtein Distance）计算相似度：

```python
def levenshtein_distance(s1: str, s2: str) -> int:
    """计算两个字符串的编辑距离"""
    m, n = len(s1), len(s2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]

    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j

    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if s1[i-1] == s2[j-1]:
                dp[i][j] = dp[i-1][j-1]
            else:
                dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])

    return dp[m][n]

def find_closest_terms(query: str, dictionary: list[str], max_distance: int = 2) -> list[dict]:
    """从词典中找出与查询最接近的词"""
    results = []
    for term in dictionary:
        dist = levenshtein_distance(query.lower(), term.lower())
        if 0 < dist <= max_distance:
            results.append({"term": term, "distance": dist})

    return sorted(results, key=lambda x: x["distance"])
```

编辑距离为 1 表示只差一个字符（插入、删除或替换），这覆盖了大部分拼写错误。`max_edits: 2` 允许最多 2 次编辑，但要注意这可能导致误匹配。

### 基于搜索日志的纠错

从搜索日志中学习纠错规则，比通用编辑距离更精准：

```python
def learn_corrections_from_logs(time_range: str = "30d"):
    """从搜索日志中学习用户的纠正行为"""
    # 找出用户搜索后没有点击、然后修改查询重新搜索的模式
    body = {
        "query": {
            "bool": {
                "filter": [
                    {"range": {"@timestamp": {"gte": f"now-{time_range}"}}},
                    {"term": {"event_type": "search"}}
                ]
            }
        },
        "aggs": {
            "sessions": {
                "terms": {"field": "session_id", "size": 10000},
                "aggs": {
                    "queries": {
                        "terms": {"field": "query.keyword", "size": 10, "order": {"_count": "desc"}}
                    }
                }
            }
        },
        "size": 0
    }
    result = es.search(index="search_logs-*", body=body)

    corrections = {}
    for bucket in result["aggregations"]["sessions"]["buckets"]:
        queries = [b["key"] for b in bucket["queries"]["buckets"]]
        if len(queries) >= 2:
            # 如果两个查询相似但不同，可能是纠正行为
            for i in range(len(queries) - 1):
                dist = levenshtein_distance(queries[i], queries[i + 1])
                if 1 <= dist <= 2:
                    if queries[i] not in corrections:
                        corrections[queries[i]] = []
                    corrections[queries[i]].append(queries[i + 1])

    # 统计最频繁的纠正
    frequent_corrections = []
    for original, suggestions in corrections.items():
        most_common = max(set(suggestions), key=suggestions.count)
        frequent_corrections.append({
            "original": original,
            "correction": most_common,
            "count": suggestions.count(most_common)
        })

    return sorted(frequent_corrections, key=lambda x: x["count"], reverse=True)
```

这种方法从用户的真实行为中学习：用户搜了"pythn"没有点击，然后搜了"python"并点击了结果，说明"pythn"应该纠正为"python"。

## 搜索高亮（Highlight）

在搜索结果中高亮匹配的关键词：

```python
def search_with_highlight(query: str, index: str = "products",
                          highlight_fields: list[str] = None):
    if highlight_fields is None:
        highlight_fields = ["title", "content"]

    body = {
        "query": {"multi_match": {"query": query, "fields": highlight_fields}},
        "highlight": {
            "fields": {
                field: {"fragment_size": 150, "number_of_fragments": 3,
                         "pre_tags": ["<mark>"], "post_tags": ["</mark>"]}
                for field in highlight_fields
            }
        },
        "size": 10
    }
    result = es.search(index=index, body=body)

    return [
        {
            "id": hit["_id"], "score": hit["_score"],
            "source": hit["_source"],
            "highlights": hit.get("highlight", {})
        }
        for hit in result["hits"]["hits"]
    ]
```

ES 的 Highlight 自动找到匹配片段并标记关键词。`fragment_size` 控制片段长度，`number_of_fragments` 控制返回几个片段。

### 高亮的性能考量

高亮是 CPU 密集型操作，对长文档（如几万字的文章）做高亮会显著增加查询延迟：

```python
def search_with_optimized_highlight(query: str, index: str = "products"):
    body = {
        "query": {"match": {"content": query}},
        "highlight": {
            "fields": {
                "content": {
                    "fragment_size": 150,
                    "number_of_fragments": 3,
                    "no_match_size": 150,
                    "highlight_query": {
                        "bool": {
                            "should": [
                                {"match": {"content": query}},
                                {"match_phrase": {"content": {"query": query, "slop": 2}}}
                            ]
                        }
                    }
                }
            },
            "order": "score"
        },
        "size": 10
    }
    return es.search(index=index, body=body)
```

优化点：
- `no_match_size: 150`：如果没有匹配，返回文档前 150 字符作为预览，避免返回空高亮。
- `highlight_query`：高亮时使用比搜索更宽松的查询（加入短语匹配），展示更精准的匹配片段。
- `order: "score"`：按相关性排序高亮片段，最相关的片段排在最前面。

### 高亮的前端展示

高亮结果需要正确处理 HTML 转义和样式：

```python
import html

def format_highlights(highlights: dict, max_snippets: int = 3) -> dict:
    """格式化高亮结果，处理 HTML 转义"""
    formatted = {}
    for field, fragments in highlights.items():
        clean_fragments = []
        for fragment in fragments[:max_snippets]:
            # ES 返回的高亮已经包含 <mark> 标签，需要确保其他部分已转义
            clean = html.unescape(fragment)
            clean_fragments.append(clean)
        formatted[field] = clean_fragments
    return formatted

def merge_highlights_with_source(source: dict, highlights: dict) -> dict:
    """将高亮结果合并到源数据中"""
    result = dict(source)
    for field, fragments in highlights.items():
        if fragments:
            result[f"{field}_highlighted"] = fragments[0]
    return result
```

## 分面导航（Faceted Navigation）

在搜索结果页展示可交互的筛选维度：

```python
def search_with_facets(query: str, active_filters: dict = None, index: str = "products"):
    if active_filters is None:
        active_filters = {}

    filter_clauses = []
    for field, value in active_filters.items():
        if isinstance(value, dict) and ("min" in value or "max" in value):
            r = {}
            if "min" in value: r["gte"] = value["min"]
            if "max" in value: r["lte"] = value["max"]
            filter_clauses.append({"range": {field: r}})
        elif isinstance(value, list):
            filter_clauses.append({"terms": {field: value}})
        else:
            filter_clauses.append({"term": {field: value}})

    body = {
        "query": {"bool": {"must": [{"match": {"title": query}}], "filter": filter_clauses}},
        "aggs": {
            "categories": {"terms": {"field": "category", "size": 15}},
            "brands": {"terms": {"field": "brand", "size": 20}},
            "price_ranges": {
                "range": {"field": "price", "ranges": [
                    {"to": 50, "key": "50元以下"},
                    {"from": 50, "to": 200, "key": "50-200元"},
                    {"from": 200, "to": 500, "key": "200-500元"},
                    {"from": 500, "key": "500元以上"}
                ]}
            }
        },
        "size": 20
    }
    result = es.search(index=index, body=body)

    return {
        "hits": result["hits"],
        "facets": {
            name: [{"value": b["key"], "count": b["doc_count"]} for b in agg["buckets"]]
            for name, agg in result["aggregations"].items() if "buckets" in agg
        }
    }
```

分面聚合和搜索在同一个请求中完成。每个分面值标记了是否被选中，前端据此渲染选中状态。

## 零结果处理

当搜索没有结果时，提供有价值的降级内容：

```python
def search_with_fallback(query: str, index: str = "products"):
    body = {"query": {"match": {"title": query}}, "size": 20}
    result = es.search(index=index, body=body)

    if result["hits"]["total"]["value"] > 0:
        return {"has_results": True, "results": result["hits"]["hits"]}

    fallback = {"has_results": False, "query": query, "suggestions": []}

    # 策略1：拼写纠错
    correction = spell_check(query)
    if correction["has_correction"]:
        corrected_result = es.search(index=index, body={
            "query": {"match": {"title": correction["corrected_query"]}}, "size": 10
        })
        if corrected_result["hits"]["total"]["value"] > 0:
            fallback["suggestions"].append({
                "type": "correction", "query": correction["corrected_query"],
                "results": corrected_result["hits"]["hits"]
            })

    # 策略2：模糊匹配
    fuzzy_result = es.search(index=index, body={
        "query": {"multi_match": {"query": query, "fields": ["title", "content"], "fuzziness": "AUTO"}},
        "size": 5
    })
    if fuzzy_result["hits"]["total"]["value"] > 0:
        fallback["suggestions"].append({"type": "fuzzy", "results": fuzzy_result["hits"]["hits"]})

    # 策略3：热门推荐
    popular = es.search(index=index, body={
        "query": {"match_all": {}}, "sort": [{"sales_count": "desc"}], "size": 5
    })
    fallback["suggestions"].append({"type": "popular", "results": popular["hits"]["hits"]})

    return fallback
```

零结果处理采用三级降级：先尝试拼写纠错，再尝试模糊匹配，最后展示热门推荐。

### 零结果的预防

与其在零结果后做降级，不如主动预防零结果：

```python
def predictive_zero_result_handling(query: str, index: str = "products"):
    """预测性零结果处理：在搜索前就判断是否可能零结果"""
    # 检查查询词是否都在索引中
    tokens = query.split()
    missing_tokens = []

    for token in tokens:
        check_body = {
            "query": {"term": {"title.keyword": token}},
            "size": 0
        }
        result = es.search(index=index, body=check_body)
        if result["hits"]["total"]["value"] == 0:
            missing_tokens.append(token)

    # 如果有缺失的词，预先做模糊匹配
    if missing_tokens:
        fuzzy_clauses = [
            {"fuzzy": {"title": {"value": t, "fuzziness": "AUTO"}}}
            for t in missing_tokens
        ]
        body = {
            "query": {"bool": {"should": fuzzy_clauses}},
            "suggest": {
                "spelling": {
                    "text": query,
                    "phrase": {
                        "field": "title",
                        "size": 3
                    }
                }
            },
            "size": 10
        }
        result = es.search(index=index, body=body)

        suggestions = []
        if result["suggest"]["spelling"][0]["options"]:
            suggestions = [o["text"] for o in result["suggest"]["spelling"][0]["options"]]

        return {
            "has_exact_results": False,
            "fuzzy_results": result["hits"]["hits"],
            "spelling_suggestions": suggestions,
            "missing_tokens": missing_tokens
        }

    return {"has_exact_results": True}
```

这种方案在搜索前先检查查询词是否在索引中，如果发现缺失词，直接用模糊匹配和拼写建议，用户看到的不是"零结果"而是"您是否要搜索 XXX"。

## 常见误区

1. **搜索建议延迟太高**：建议接口需要在 100ms 内响应，后端计算太慢会影响输入体验。
2. **纠错太激进**：对正确的查询做误纠正会让用户困惑，应该只在结果不足时才推荐纠正。
3. **高亮截断不合理**：高亮片段在句子中间截断会影响可读性，应尽量在句子边界截断。
4. **分面数量不控制**：上万个品牌值全部展示会让页面崩溃，必须限制返回数量。
5. **零结果页面空白**：搜索无结果时展示空白页面是最差体验，必须提供降级内容。
6. **忽略移动端体验**：移动端搜索框空间有限，建议列表不宜过长。
7. **不做 A/B 测试**：搜索体验的每次改动都应该通过 A/B 测试验证效果。

## 工程建议

1. **搜索建议独立部署**：建议接口 QPS 远高于搜索接口，应该独立部署和缓存。
2. **纠错词典定期更新**：从搜索日志中提取高频拼写错误，定期更新词典。
3. **高亮性能优化**：对长文档做高亮比较耗时，可以用 `no_match_size` 限制返回长度。
4. **分面缓存**：同一查询的分面结果变化不大，可以缓存 5-10 分钟。
5. **零结果监控**：记录所有零结果查询，定期分析是否有索引缺失问题。
6. **前端 debounce**：搜索建议的输入事件应该做 300ms 防抖，减少不必要的请求。
7. **渐进式加载**：先展示搜索结果，AI 摘要和分面数据异步加载。
8. **用户反馈收集**：在搜索结果页添加"搜索结果是否有帮助"的反馈按钮。

## 小结

搜索体验优化围绕"让用户更快找到内容"展开。搜索建议减少输入成本，拼写纠错容忍用户错误，高亮帮助快速定位信息，分面导航缩小选择范围，零结果处理避免用户流失。这些功能不改变检索算法，但对用户感知的搜索质量有巨大影响。

### 练习二

**思路**：按段落拆分文档，计算每个段落与查询的相关性分数，返回 top 3 并高亮。

**答案**：

```python
import re

def context_aware_highlight(document: dict, query: str, max_fragments: int = 3):
    content = document.get("content", "")
    paragraphs = [p.strip() for p in content.split("\n\n") if p.strip()]
    tokens = query.lower().split()

    scored = []
    for para in paragraphs:
        score = sum(para.lower().count(t) for t in tokens)
        if score > 0:
            scored.append((score, para))

    scored.sort(key=lambda x: x[0], reverse=True)
    highlighted = []
    for _, para in scored[:max_fragments]:
        snippet = para[:200]
        for token in tokens:
            snippet = re.compile(re.escape(token), re.IGNORECASE).sub(f"<mark>{token}</mark>", snippet)
        highlighted.append(snippet)
    return highlighted
```

**要点**：按段落拆分后用关键词出现次数作为相关性分数，比 ES 默认的固定长度截断更智能。

### 练习三

**思路**：从搜索日志中统计零结果查询，与索引中的术语对比，识别可通过同义词补充解决的查询。

**答案**：

```python
def analyze_zero_result_queries(time_range: str = "24h"):
    body = {
        "query": {"bool": {"filter": [
            {"term": {"event_type": "search"}},
            {"term": {"result_count": 0}},
            {"range": {"@timestamp": {"gte": f"now-{time_range}"}}}
        ]}},
        "aggs": {"top_queries": {"terms": {"field": "query.keyword", "size": 20, "order": {"_count": "desc"}}}},
        "size": 0
    }
    result = es.search(index="search_logs-*", body=body)

    top_queries = []
    for bucket in result["aggregations"]["top_queries"]["buckets"]:
        query_text = bucket["key"]
        tokens = query_text.split()
        missing = []
        for token in tokens:
            check = es.search(index="dictionary", body={"query": {"term": {"term": token}}})
            if check["hits"]["total"]["value"] == 0:
                missing.append(token)

        top_queries.append({
            "query": query_text,
            "count": bucket["doc_count"],
            "missing_terms": missing,
            "can_fix": len(missing) > 0 and len(missing) < len(tokens)
        })

    return {
        "top_zero_queries": top_queries,
        "fixable": [q for q in top_queries if q["can_fix"]]
    }
```

**要点**：零结果分析的核心是找出用户搜索了什么但没找到。通过与词典对比，识别缺失术语并建议添加同义词。

### 练习四

**思路**：实现两个不同的建议排序策略，用哈希分流用户到不同组，统计点击率。

**答案**：

```python
import hashlib

class SuggestABTest:
    def __init__(self):
        self.variants = {
            "frequency": self._sort_by_frequency,
            "conversion": self._sort_by_conversion
        }
        self.metrics = {v: {"impressions": 0, "clicks": 0} for v in self.variants}

    def assign_variant(self, user_id: str) -> str:
        hash_val = hashlib.md5(user_id.encode()).hexdigest()
        return "frequency" if int(hash_val[:8], 16) % 2 == 0 else "conversion"

    def _sort_by_frequency(self, prefix: str, size: int = 8) -> list[dict]:
        body = {
            "query": {"prefix": {"query_text.keyword": prefix}},
            "sort": [{"frequency": {"order": "desc"}}],
            "size": size
        }
        result = es.search(index="search_suggestions", body=body)
        return [{"text": h["_source"]["query_text"], "score": h["_source"]["frequency"]}
                for h in result["hits"]["hits"]]

    def _sort_by_conversion(self, prefix: str, size: int = 8) -> list[dict]:
        body = {
            "query": {"prefix": {"query_text.keyword": prefix}},
            "sort": [{"conversion_rate": {"order": "desc"}}, {"frequency": {"order": "desc"}}],
            "size": size
        }
        result = es.search(index="search_suggestions", body=body)
        return [{"text": h["_source"]["query_text"],
                 "score": h["_source"].get("conversion_rate", 0)}
                for h in result["hits"]["hits"]]

    def get_suggestions(self, user_id: str, prefix: str, size: int = 8) -> dict:
        variant = self.assign_variant(user_id)
        suggestions = self.variants[variant](prefix, size)
        self.metrics[variant]["impressions"] += 1
        return {"variant": variant, "suggestions": suggestions}

    def log_click(self, user_id: str):
        variant = self.assign_variant(user_id)
        self.metrics[variant]["clicks"] += 1

    def evaluate(self) -> dict:
        results = {}
        for variant, data in self.metrics.items():
            ctr = data["clicks"] / max(data["impressions"], 1)
            results[variant] = {
                "impressions": data["impressions"],
                "clicks": data["clicks"],
                "ctr": round(ctr, 4)
            }

        freq_ctr = results["frequency"]["ctr"]
        conv_ctr = results["conversion"]["ctr"]
        return {
            "variants": results,
            "winner": "conversion" if conv_ctr > freq_ctr else "frequency",
            "lift": round((conv_ctr - freq_ctr) / max(freq_ctr, 0.001), 4)
        }
```

**要点**：通过哈希保证同一用户始终看到同一排序策略。评估时比较两种策略的 CTR，选择更优的方案上线。
