# 阶段实战：多策略检索系统

经过前 5 课的学习，我们掌握了 BM25、向量空间模型、语言模型、语义检索和混合检索。这节课把它们整合到一个系统中，通过对比实验验证各策略的效果差异。

真实生产环境中的搜索系统从来不是单一策略。Elasticsearch 8.x 的 kNN 搜索、Pinecone 的混合检索，底层都是类似架构。

## 策略解耦设计

核心原则：每种检索策略独立实现，通过统一接口调用，融合层负责组合结果。

```
查询输入 → 路由分发 → [BM25] [语义检索] [混合策略] → 融合层(RRF) → 排序输出
```

为什么策略解耦很重要？如果把 BM25 和语义检索混在一起写，就无法独立测试、独立优化、灵活组合。解耦后，新增策略只需实现接口并注册。

```python
from abc import ABC, abstractmethod

class RetrievalStrategy(ABC):
    @abstractmethod
    def index(self, documents: list[str]) -> None:
        pass

    @abstractmethod
    def search(self, query: str, top_k: int = 10) -> list[tuple[str, float]]:
        pass
```

接口返回 `list[tuple[str, float]]`，但不同策略的分数含义不同——BM25 是相关性评分，语义检索是余弦相似度。归一化交给融合层处理。

## BM25 策略

```python
import math
from collections import Counter

class BM25Strategy(RetrievalStrategy):
    def __init__(self, k1=1.5, b=0.75):
        self.k1, self.b = k1, b
        self.corpus, self.doc_tf, self.doc_lens = [], [], []
        self.doc_freqs, self.avgdl, self.doc_count = Counter(), 0, 0
        self.raw_docs = []

    def index(self, documents):
        self.raw_docs = documents
        self.corpus = [list(doc) for doc in documents]
        self.doc_count = len(self.corpus)
        self.avgdl = sum(len(d) for d in self.corpus) / self.doc_count
        self.doc_lens = [len(d) for d in self.corpus]
        self.doc_tf, self.doc_freqs = [], Counter()
        for doc in self.corpus:
            tf = Counter(doc)
            self.doc_tf.append(tf)
            for word in set(doc):
                self.doc_freqs[word] += 1

    def _idf(self, word):
        df = self.doc_freqs.get(word, 0)
        return math.log((self.doc_count - df + 0.5) / (df + 0.5) + 1)

    def _score(self, query_tokens, doc_idx):
        score, tf, dl = 0.0, self.doc_tf[doc_idx], self.doc_lens[doc_idx]
        for word in query_tokens:
            if word not in tf:
                continue
            tf_val = tf[word]
            num = tf_val * (self.k1 + 1)
            den = tf_val + self.k1 * (1 - self.b + self.b * dl / self.avgdl)
            score += self._idf(word) * (num / den)
        return score

    def search(self, query, top_k=10):
        tokens = list(query)
        scores = [(i, self._score(tokens, i)) for i in range(self.doc_count)]
        scores.sort(key=lambda x: -x[1])
        return [(self.raw_docs[idx], score) for idx, score in scores[:top_k]]
```

## 语义检索策略

```python
import numpy as np

class SemanticStrategy(RetrievalStrategy):
    def __init__(self):
        self.model, self.doc_vectors, self.raw_docs = None, None, []

    def index(self, documents):
        from sentence_transformers import SentenceTransformer
        self.raw_docs = documents
        self.model = SentenceTransformer('all-MiniLM-L6-v2')
        self.doc_vectors = self.model.encode(documents).astype('float32')
        norms = np.linalg.norm(self.doc_vectors, axis=1, keepdims=True)
        norms[norms == 0] = 1
        self.doc_vectors = self.doc_vectors / norms

    def search(self, query, top_k=10):
        query_vec = self.model.encode([query]).astype('float32')
        q_norm = np.linalg.norm(query_vec)
        if q_norm > 0:
            query_vec = query_vec / q_norm
        scores = (self.doc_vectors @ query_vec.T).flatten()
        top_idx = np.argsort(-scores)[:top_k]
        return [(self.raw_docs[i], float(scores[i])) for i in top_idx]
```

## 混合检索策略（RRF 融合）

RRF（Reciprocal Rank Fusion）是简单有效的融合方法，不需要分数归一化。核心思想：每个策略的排名转换为分数，加权求和。

```python
from collections import defaultdict

class HybridStrategy(RetrievalStrategy):
    def __init__(self, bm25, semantic, bm25_weight=0.4, semantic_weight=0.6, rrf_k=60):
        self.bm25, self.semantic = bm25, semantic
        self.bm25_w, self.semantic_w = bm25_weight, semantic_weight
        self.rrf_k = rrf_k

    def index(self, documents):
        self.raw_docs = documents

    def search(self, query, top_k=10):
        k = top_k * 3
        bm25_ranking = [doc for doc, _ in self.bm25.search(query, top_k=k)]
        sem_ranking = [doc for doc, _ in self.semantic.search(query, top_k=k)]
        scores = defaultdict(float)
        for ranking, weight in [(bm25_ranking, self.bm25_w),
                                 (sem_ranking, self.semantic_w)]:
            for rank, doc in enumerate(ranking, 1):
                scores[doc] += weight / (self.rrf_k + rank)
        fused = sorted(scores.items(), key=lambda x: -x[1])
        return [(doc, score) for doc, score in fused[:top_k]]
```

## 评估框架

没有评估，任何优化都是盲目的。

```python
def precision_at_k(retrieved, relevant, k):
    return sum(1 for d in retrieved[:k] if d in relevant) / k

def recall_at_k(retrieved, relevant, k):
    return sum(1 for d in retrieved[:k] if d in relevant) / len(relevant) if relevant else 0

def mrr(retrieved, relevant):
    for rank, doc in enumerate(retrieved, 1):
        if doc in relevant:
            return 1.0 / rank
    return 0.0

def ndcg_at_k(retrieved, relevant, k):
    dcg = sum(1.0 / np.log2(i + 2) for i, doc in enumerate(retrieved[:k]) if doc in relevant)
    ideal = sum(1.0 / np.log2(i + 2) for i in range(min(len(relevant), k)))
    return dcg / ideal if ideal > 0 else 0.0

def evaluate_strategy(strategy, queries, ground_truth, k=5):
    p_list, r_list, mrr_list, ndcg_list = [], [], [], []
    for query, relevant in zip(queries, ground_truth):
        retrieved = [doc for doc, _ in strategy.search(query, top_k=k)]
        p_list.append(precision_at_k(retrieved, relevant, k))
        r_list.append(recall_at_k(retrieved, relevant, k))
        mrr_list.append(mrr(retrieved, relevant))
        ndcg_list.append(ndcg_at_k(retrieved, relevant, k))
    return {"P@K": sum(p_list)/len(p_list), "R@K": sum(r_list)/len(r_list),
            "MRR": sum(mrr_list)/len(mrr_list), "NDCG@K": sum(ndcg_list)/len(ndcg_list)}
```

| 指标 | 含义 | 关注点 |
|------|------|--------|
| P@K | Top-K 中相关文档比例 | 精度 |
| R@K | 相关文档被找回的比例 | 召回 |
| MRR | 第一个相关结果排名的倒数 | 用户体验 |
| NDCG@K | 归一化折损累积增益 | 排序质量 |

### 逐查询分析

平均指标掩盖了查询间的差异。逐查询分析能发现各策略的优势和盲区：

```python
def per_query_analysis(query, strategies, relevant, k=5):
    print(f"\n查询: '{query}' (相关文档: {len(relevant)} 个)")
    for name, strategy in strategies.items():
        retrieved = [doc for doc, _ in strategy.search(query, top_k=k)]
        hits = sum(1 for d in retrieved if d in relevant)
        first_rank = next((i+1 for i, d in enumerate(retrieved) if d in relevant), 0)
        print(f"  {name:<15} 命中={hits}  首个排名={first_rank or 'N/A'}")
```

## 系统封装

```python
class MultiStrategySearchSystem:
    def __init__(self):
        self.strategies = {}

    def add_strategy(self, name, strategy):
        self.strategies[name] = strategy

    def build_index(self, documents):
        for name, s in self.strategies.items():
            s.index(documents)

    def search(self, query, strategy_name="hybrid", top_k=10):
        results = self.strategies[strategy_name].search(query, top_k=top_k)
        return [{"rank": i+1, "doc": doc, "score": round(score, 4)}
                for i, (doc, score) in enumerate(results)]

    def compare(self, query, top_k=5):
        return {n: self.search(query, n, top_k) for n in self.strategies}
```

## 常见误区

**过早优化索引性能。** 先确保检索质量达标，再考虑性能。一个返回错误结果的系统再快也没有意义。

**混合权重不做调优。** 直接用等权混合效果通常不是最优，权重调优是混合检索效果的关键。

**只看平均指标。** 平均指标掩盖了查询间的差异，应该同时看逐查询分析。

**评估集和训练集泄露。** 如果用同一份文档做索引和评估，结果会过于乐观。

## 工程建议

- 渐进式上线：先用 BM25，收集数据后逐步引入语义检索和混合检索
- A/B 测试：混合检索权重应通过 A/B 测试验证，离线指标只是参考
- 降级策略：语义检索依赖外部模型，可能因模型加载失败报错，应自动回退到纯 BM25

### 练习一

```python
class QueryExpansionStrategy(RetrievalStrategy):
    def __init__(self, base, top_cooccur=3):
        self.base, self.top_cooccur = base, top_cooccur
        self.cooccurrence = Counter()

    def index(self, documents):
        self.raw_docs = documents
        self.base.index(documents)
        for doc in self.base.corpus:
            words = set(doc)
            for w1 in words:
                for w2 in words:
                    if w1 != w2:
                        self.cooccurrence[(w1, w2)] += 1

    def search(self, query, top_k=10):
        expanded = list(query)
        for word in query:
            cands = sorted(
                [(w2, c) for (w1, w2), c in self.cooccurrence.items()
                 if w1 == word and w2 not in query],
                key=lambda x: -x[1])
            expanded.extend(w for w, _ in cands[:self.top_cooccur])
        scores = [(i, self.base._score(expanded, i))
                  for i in range(len(self.raw_docs))]
        scores.sort(key=lambda x: -x[1])
        return [(self.raw_docs[i], s) for i, s in scores[:top_k]]
```

```python
class FallbackSemanticStrategy(RetrievalStrategy):
    def __init__(self, fallback):
        self.model, self.doc_vectors = None, None
        self.raw_docs, self.fallback = [], fallback
        self.is_healthy = False

    def index(self, documents):
        self.raw_docs = documents
        try:
            from sentence_transformers import SentenceTransformer
            self.model = SentenceTransformer('all-MiniLM-L6-v2')
            self.doc_vectors = self.model.encode(documents).astype('float32')
            norms = np.linalg.norm(self.doc_vectors, axis=1, keepdims=True)
            norms[norms == 0] = 1
            self.doc_vectors = self.doc_vectors / norms
            self.is_healthy = True
        except Exception as e:
            print(f"语义模型加载失败，降级到 BM25: {e}")
            self.is_healthy = False
            self.fallback.index(documents)

    def search(self, query, top_k=10):
        if not self.is_healthy:
            return self.fallback.search(query, top_k)
        try:
            query_vec = self.model.encode([query]).astype('float32')
            q_norm = np.linalg.norm(query_vec)
            if q_norm > 0:
                query_vec = query_vec / q_norm
            scores = (self.doc_vectors @ query_vec.T).flatten()
            top_idx = np.argsort(-scores)[:top_k]
            return [(self.raw_docs[i], float(scores[i])) for i in top_idx]
        except Exception as e:
            print(f"语义检索出错，降级到 BM25: {e}")
            self.is_healthy = False
            return self.fallback.search(query, top_k)
```
