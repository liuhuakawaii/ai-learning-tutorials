## 二、异常数据检测

### 2.1 检测器设计

```python
import pandas as pd
from typing import List, Dict, Any, Optional
from dataclasses import dataclass
from enum import Enum

class AnomalyType(Enum):
    """异常类型"""
    FORMAT = "format"
    RANGE = "range"
    LOGIC = "logic"
    COMPLETENESS = "completeness"
    UNIQUENESS = "uniqueness"
    REFERENCE = "reference"

class Severity(Enum):
    """严重程度"""
    CRITICAL = "critical"
    ERROR = "error"
    WARNING = "warning"
    INFO = "info"

@dataclass
class AnomalyRecord:
    """异常记录"""
    record_index: int
    column: str
    anomaly_type: AnomalyType
    severity: Severity
    current_value: Any
    expected_value: Any
    description: str
    fix_suggestion: Optional[str] = None

class AnomalyDetector:
    """异常检测器"""
    
    def detect(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测异常数据"""
        anomalies = []
        
        # 执行各种检测
        anomalies.extend(self._detect_format_anomalies(data))
        anomalies.extend(self._detect_range_anomalies(data))
        anomalies.extend(self._detect_logic_anomalies(data))
        anomalies.extend(self._detect_completeness_anomalies(data))
        anomalies.extend(self._detect_uniqueness_anomalies(data))
        
        return anomalies
    
    def _detect_format_anomalies(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测格式异常"""
        anomalies = []
        
        # 日期格式检查
        if "publish_date" in data.columns:
            for idx, value in data["publish_date"].items():
                if pd.notna(value) and not isinstance(value, (pd.Timestamp, datetime)):
                    try:
                        pd.to_datetime(value)
                    except:
                        anomalies.append(AnomalyRecord(
                            record_index=idx,
                            column="publish_date",
                            anomaly_type=AnomalyType.FORMAT,
                            severity=Severity.ERROR,
                            current_value=value,
                            expected_value="有效的日期格式",
                            description=f"日期格式无效: {value}",
                            fix_suggestion="使用标准日期格式 YYYY-MM-DD"
                        ))
        
        return anomalies
    
    def _detect_range_anomalies(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测范围异常"""
        anomalies = []
        
        # 薪资范围检查
        if "salary_min" in data.columns:
            for idx, value in data["salary_min"].items():
                if pd.notna(value) and (value < 0 or value > 1000000):
                    anomalies.append(AnomalyRecord(
                        record_index=idx,
                        column="salary_min",
                        anomaly_type=AnomalyType.RANGE,
                        severity=Severity.ERROR,
                        current_value=value,
                        expected_value="0-1000000",
                        description=f"薪资超出合理范围: {value}",
                        fix_suggestion="检查数据源或设置为 NULL"
                    ))
        
        return anomalies
    
    def _detect_logic_anomalies(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测逻辑异常"""
        anomalies = []
        
        # 薪资逻辑检查：min <= max
        if "salary_min" in data.columns and "salary_max" in data.columns:
            mask = (data["salary_min"].notna() & 
                   data["salary_max"].notna() & 
                   (data["salary_min"] > data["salary_max"]))
            
            for idx in data[mask].index:
                anomalies.append(AnomalyRecord(
                    record_index=idx,
                    column="salary_min,salary_max",
                    anomaly_type=AnomalyType.LOGIC,
                    severity=Severity.ERROR,
                    current_value=f"min={data.loc[idx, 'salary_min']}, max={data.loc[idx, 'salary_max']}",
                    expected_value="salary_min <= salary_max",
                    description="最低薪资大于最高薪资",
                    fix_suggestion="交换 min 和 max 值"
                ))
        
        return anomalies
    
    def _detect_completeness_anomalies(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测完整性异常"""
        anomalies = []
        
        required_columns = ["job_id", "title", "company", "city"]
        
        for col in required_columns:
            if col in data.columns:
                null_mask = data[col].isnull()
                for idx in data[null_mask].index:
                    anomalies.append(AnomalyRecord(
                        record_index=idx,
                        column=col,
                        anomaly_type=AnomalyType.COMPLETENESS,
                        severity=Severity.CRITICAL if col == "job_id" else Severity.ERROR,
                        current_value=None,
                        expected_value="非空值",
                        description=f"必填字段 {col} 为空",
                        fix_suggestion="从数据源补充或标记为无效"
                    ))
        
        return anomalies
    
    def _detect_uniqueness_anomalies(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测唯一性异常"""
        anomalies = []
        
        if "job_id" in data.columns:
            duplicated_mask = data["job_id"].duplicated(keep=False)
            for idx in data[duplicated_mask].index:
                anomalies.append(AnomalyRecord(
                    record_index=idx,
                    column="job_id",
                    anomaly_type=AnomalyType.UNIQUENESS,
                    severity=Severity.ERROR,
                    current_value=data.loc[idx, "job_id"],
                    expected_value="唯一值",
                    description=f"job_id 重复: {data.loc[idx, 'job_id']}",
                    fix_suggestion="保留最新记录，删除重复"
                ))
        
        return anomalies
```

## 三、异常数据隔离

### 3.1 隔离机制

```python
from datetime import datetime
import json

class AnomalyQuarantine:
    """异常数据隔离器"""
    
    def __init__(self, engine):
        self.engine = engine
    
    def quarantine(
        self,
        data: pd.DataFrame,
        anomalies: List[AnomalyRecord],
        batch_id: str = None
    ) -> Dict[str, Any]:
        """隔离异常数据"""
        
        if not anomalies:
            return {"quarantined": 0, "message": "无异常数据"}
        
        # 收集需要隔离的记录索引
        critical_indices = set()
        error_indices = set()
        
        for anomaly in anomalies:
            if anomaly.severity == Severity.CRITICAL:
                critical_indices.add(anomaly.record_index)
            elif anomaly.severity == Severity.ERROR:
                error_indices.add(anomaly.record_index)
        
        # 合并需要隔离的索引
        quarantine_indices = critical_indices | error_indices
        
        if not quarantine_indices:
            return {"quarantined": 0, "message": "无需要隔离的数据"}
        
        # 提取隔离数据
        quarantine_data = data.loc[list(quarantine_indices)].copy()
        
        # 添加隔离元信息
        quarantine_data["quarantine_time"] = datetime.now()
        quarantine_data["batch_id"] = batch_id
        quarantine_data["anomaly_summary"] = self._build_anomaly_summary(
            quarantine_indices, anomalies
        )
        
        # 写入隔离表
        quarantine_data.to_sql(
            "quarantine_data",
            self.engine,
            if_exists="append",
            index=False
        )
        
        return {
            "quarantined": len(quarantine_indices),
            "critical_count": len(critical_indices),
            "error_count": len(error_indices),
            "message": f"已隔离 {len(quarantine_indices)} 条记录"
        }
    
    def _build_anomaly_summary(
        self,
        indices: set,
        anomalies: List[AnomalyRecord]
    ) -> Dict[int, List[str]]:
        """构建异常摘要"""
        summary = {}
        
        for anomaly in anomalies:
            if anomaly.record_index in indices:
                if anomaly.record_index not in summary:
                    summary[anomaly.record_index] = []
                summary[anomaly.record_index].append(
                    f"{anomaly.column}: {anomaly.description}"
                )
        
        return json.dumps(summary, ensure_ascii=False)
```

## 四、异常数据修复

### 4.1 自动修复策略

```python
from typing import Callable, Dict, Any

class AutoFixer:
    """自动修复器"""
    
    def __init__(self):
        self.fix_strategies: Dict[str, Callable] = {}
        self._register_default_strategies()
    
    def _register_default_strategies(self):
        """注册默认修复策略"""
        
        # 薪资逻辑修复：交换 min 和 max
        self.register_fix(
            "salary_swap",
            lambda row: self._fix_salary_swap(row)
        )
        
        # 格式标准化
        self.register_fix(
            "date_format",
            lambda row: self._fix_date_format(row)
        )
        
        # 填充默认值
        self.register_fix(
            "fill_defaults",
            lambda row: self._fill_defaults(row)
        )
    
    def register_fix(self, name: str, fix_func: Callable):
        """注册修复策略"""
        self.fix_strategies[name] = fix_func
    
    def auto_fix(
        self,
        data: pd.DataFrame,
        anomalies: List[AnomalyRecord]
    ) -> Dict[str, Any]:
        """自动修复"""
        
        fixed_count = 0
        fix_details = []
        
        for anomaly in anomalies:
            if anomaly.severity in [Severity.CRITICAL]:
                # 致命错误不能自动修复
                continue
            
            fix_result = self._try_fix(data, anomaly)
            
            if fix_result["fixed"]:
                fixed_count += 1
                fix_details.append({
                    "index": anomaly.record_index,
                    "column": anomaly.column,
                    "original": anomaly.current_value,
                    "fixed": fix_result["new_value"],
                    "strategy": fix_result["strategy"]
                })
        
        return {
            "fixed_count": fixed_count,
            "details": fix_details
        }
    
    def _try_fix(self, data: pd.DataFrame, anomaly: AnomalyRecord) -> Dict[str, Any]:
        """尝试修复单个异常"""
        
        idx = anomaly.record_index
        
        # 根据异常类型选择修复策略
        if anomaly.anomaly_type == AnomalyType.LOGIC:
            if "salary" in anomaly.column:
                return self._fix_salary_swap(data, idx)
        
        elif anomaly.anomaly_type == AnomalyType.FORMAT:
            if "date" in anomaly.column:
                return self._fix_date_format(data, idx, anomaly.column)
        
        elif anomaly.anomaly_type == AnomalyType.COMPLETENESS:
            return self._fill_default(data, idx, anomaly.column)
        
        return {"fixed": False}
    
    def _fix_salary_swap(self, data: pd.DataFrame, idx: int) -> Dict[str, Any]:
        """修复薪资逻辑错误：交换 min 和 max"""
        try:
            salary_min = data.loc[idx, "salary_min"]
            salary_max = data.loc[idx, "salary_max"]
            
            data.loc[idx, "salary_min"] = salary_max
            data.loc[idx, "salary_max"] = salary_min
            
            return {
                "fixed": True,
                "new_value": f"min={salary_max}, max={salary_min}",
                "strategy": "salary_swap"
            }
        except:
            return {"fixed": False}
    
    def _fix_date_format(
        self,
        data: pd.DataFrame,
        idx: int,
        column: str
    ) -> Dict[str, Any]:
        """修复日期格式"""
        try:
            value = data.loc[idx, column]
            if pd.isna(value):
                return {"fixed": False}
            
            # 尝试解析日期
            parsed = pd.to_datetime(value)
            data.loc[idx, column] = parsed.date()
            
            return {
                "fixed": True,
                "new_value": str(parsed.date()),
                "strategy": "date_format"
            }
        except:
            return {"fixed": False}
    
    def _fill_default(
        self,
        data: pd.DataFrame,
        idx: int,
        column: str
    ) -> Dict[str, Any]:
        """填充默认值"""
        
        defaults = {
            "experience": "未知",
            "education": "未知",
            "skills": "",
            "description": ""
        }
        
        if column in defaults:
            data.loc[idx, column] = defaults[column]
            return {
                "fixed": True,
                "new_value": defaults[column],
                "strategy": "fill_default"
            }
        
        return {"fixed": False}
```

### 4.2 人工修复接口

```python
class ManualFixer:
    """人工修复接口"""
    
    def __init__(self, engine):
        self.engine = engine
    
    def get_pending_records(self, limit: int = 100) -> pd.DataFrame:
        """获取待处理记录"""
        query = """
            SELECT *
            FROM quarantine_data
            WHERE status = 'pending'
            ORDER BY quarantine_time DESC
            LIMIT :limit
        """
        return pd.read_sql(query, self.engine, params={"limit": limit})
    
    def fix_record(
        self,
        record_id: int,
        fixes: Dict[str, Any],
        fix_by: str,
        note: str = ""
    ) -> bool:
        """修复单条记录"""
        
        try:
            # 更新字段
            set_clauses = [f"{k} = :{k}" for k in fixes.keys()]
            set_clauses.append("status = 'fixed'")
            set_clauses.append("fix_time = NOW()")
            set_clauses.append("fix_by = :fix_by")
            set_clauses.append("fix_note = :fix_note")
            
            query = f"""
                UPDATE quarantine_data
                SET {', '.join(set_clauses)}
                WHERE id = :id
            """
            
            params = {**fixes, "id": record_id, "fix_by": fix_by, "fix_note": note}
            
            with self.engine.connect() as conn:
                conn.execute(text(query), params)
                conn.commit()
            
            return True
            
        except Exception as e:
            print(f"修复失败: {e}")
            return False
    
    def discard_record(self, record_id: int, reason: str) -> bool:
        """丢弃记录"""
        try:
            query = """
                UPDATE quarantine_data
                SET status = 'discarded', fix_note = :reason
                WHERE id = :id
            """
            
            with self.engine.connect() as conn:
                conn.execute(text(query), {"id": record_id, "reason": reason})
                conn.commit()
            
            return True
            
        except Exception as e:
            print(f"丢弃失败: {e}")
            return False
```

## 五、异常数据回放

### 5.1 回放机制

```
┌──────────────────────────────────────────────────────────────┐
│                    回放机制                                    │
├──────────────────────────────────────────────────────────────┤
│                                                              │
│  什么是回放？                                                  │
│  ├── 把修复后的数据重新加载到正式表                          │
│  ├── 保证数据完整性                                          │
│  └── 支持批量处理                                            │
│                                                              │
│  回放流程：                                                   │
│  ├── 1. 从隔离表获取已修复的数据                             │
│  ├── 2. 验证修复后的数据质量                                 │
│  ├── 3. 写入正式表                                           │
│  └── 4. 更新隔离表状态                                       │
│                                                              │
│  注意事项：                                                   │
│  ├── 回放前必须验证数据质量                                  │
│  ├── 使用事务保证原子性                                      │
│  └── 记录回放日志                                            │
│                                                              │
└──────────────────────────────────────────────────────────────┘
```

### 5.2 回放实现

```python
class AnomalyReplay:
    """异常数据回放"""
    
    def __init__(self, engine):
        self.engine = engine
    
    def replay_fixed(
        self,
        target_table: str,
        key_columns: List[str],
        batch_id: str = None
    ) -> Dict[str, Any]:
        """回放已修复的数据"""
        
        # 1. 获取已修复数据
        fixed_data = self._get_fixed_data(batch_id)
        
        if fixed_data.empty:
            return {"replayed": 0, "message": "无待回放数据"}
        
        # 2. 验证数据质量
        validation_result = self._validate_before_replay(fixed_data)
        
        if not validation_result["valid"]:
            return {
                "replayed": 0,
                "message": f"数据验证失败: {validation_result['errors']}"
            }
        
        # 3. 写入正式表
        replayed_count = self._load_to_target(
            fixed_data,
            target_table,
            key_columns
        )
        
        # 4. 更新隔离表状态
        self._update_quarantine_status(fixed_data["id"].tolist())
        
        return {
            "replayed": replayed_count,
            "message": f"成功回放 {replayed_count} 条记录"
        }
    
    def _get_fixed_data(self, batch_id: str = None) -> pd.DataFrame:
        """获取已修复数据"""
        query = """
            SELECT *
            FROM quarantine_data
            WHERE status = 'fixed'
        """
        
        if batch_id:
            query += " AND batch_id = :batch_id"
        
        return pd.read_sql(
            query,
            self.engine,
            params={"batch_id": batch_id} if batch_id else None
        )
    
    def _validate_before_replay(self, data: pd.DataFrame) -> Dict[str, Any]:
        """回放前验证"""
        errors = []
        
        # 检查必填字段
        required_fields = ["job_id", "title", "company"]
        for field in required_fields:
            if field in data.columns:
                null_count = data[field].isnull().sum()
                if null_count > 0:
                    errors.append(f"{field} 有 {null_count} 个空值")
        
        return {
            "valid": len(errors) == 0,
            "errors": errors
        }
    
    def _load_to_target(
        self,
        data: pd.DataFrame,
        target_table: str,
        key_columns: List[str]
    ) -> int:
        """加载到目标表"""
        
        # 移除隔离元信息列
        exclude_columns = [
            "quarantine_time", "anomaly_summary",
            "status", "fix_time", "fix_by", "fix_note"
        ]
        load_columns = [c for c in data.columns if c not in exclude_columns]
        load_data = data[load_columns]
        
        # 使用 Upsert 方式写入
        loaded = 0
        for _, row in load_data.iterrows():
            record = row.to_dict()
            
            columns = ", ".join(record.keys())
            placeholders = ", ".join([f":{key}" for key in record.keys()])
            
            update_columns = [k for k in record.keys() if k not in key_columns]
            update_clause = ", ".join([f"{col} = EXCLUDED.{col}" for col in update_columns])
            conflict_columns = ", ".join(key_columns)
            
            query = f"""
                INSERT INTO {target_table} ({columns})
                VALUES ({placeholders})
                ON CONFLICT ({conflict_columns})
                DO UPDATE SET {update_clause}
            """
            
            try:
                with self.engine.connect() as conn:
                    conn.execute(text(query), record)
                    conn.commit()
                loaded += 1
            except Exception as e:
                print(f"回放失败: {e}")
        
        return loaded
    
    def _update_quarantine_status(self, record_ids: List[int]):
        """更新隔离表状态"""
        if not record_ids:
            return
        
        placeholders = ", ".join([str(id) for id in record_ids])
        
        query = f"""
            UPDATE quarantine_data
            SET status = 'replayed'
            WHERE id IN ({placeholders})
        """
        
        with self.engine.connect() as conn:
            conn.execute(text(query))
            conn.commit()
```

## 六、完整的异常处理流程

```python
class AnomalyPipeline:
    """异常处理管道"""
    
    def __init__(self, engine):
        self.engine = engine
        self.detector = AnomalyDetector()
        self.quarantine = AnomalyQuarantine(engine)
        self.auto_fixer = AutoFixer()
        self.manual_fixer = ManualFixer(engine)
        self.replay = AnomalyReplay(engine)
    
    def process(
        self,
        data: pd.DataFrame,
        batch_id: str = None,
        auto_fix: bool = True,
        target_table: str = None
    ) -> Dict[str, Any]:
        """处理异常数据"""
        
        result = {
            "batch_id": batch_id,
            "total_records": len(data)
        }
        
        # 1. 检测异常
        anomalies = self.detector.detect(data)
        result["anomalies_detected"] = len(anomalies)
        
        if not anomalies:
            result["message"] = "无异常数据"
            return result
        
        # 2. 隔离异常数据
        quarantine_result = self.quarantine.quarantine(data, anomalies, batch_id)
        result["quarantined"] = quarantine_result["quarantined"]
        
        # 3. 自动修复
        if auto_fix:
            fix_result = self.auto_fixer.auto_fix(data, anomalies)
            result["auto_fixed"] = fix_result["fixed_count"]
            result["fix_details"] = fix_result["details"]
        
        # 4. 回放修复数据
        if target_table:
            replay_result = self.replay.replay_fixed(
                target_table,
                key_columns=["job_id"],
                batch_id=batch_id
            )
            result["replayed"] = replay_result["replayed"]
        
        result["message"] = "异常处理完成"
        
        return result
```

## 常见误区

- **异常数据直接删除**：删掉就没了，无法追溯、无法统计、无法恢复。正确做法是先移到隔离表，标记异常原因，后续决定修复还是丢弃。
- **自动修复不验证结果**：修完就直接写回正式表，不检查修复后的数据是否真的合规。修复后必须重新跑一遍质量检查，确认修复有效。
- **隔离表没有索引和状态管理**：隔离表数据越来越多，没有索引查询慢，没有状态字段无法区分"待处理""已修复""已丢弃"，管理混乱。
- **所有异常都尝试自动修复**：致命异常（如主键为空）应该标记为不可修复，等待人工介入。自动修复只适用于格式、默认值等确定性高的场景。

---

## 工程建议

1. **隔离表设计要包含完整上下文**：除了原始数据字段，还要记录异常摘要、隔离时间、批次号、处理状态、修复人、修复备注等元信息。
2. **自动修复策略要可注册、可扩展**：用策略模式注册不同的修复逻辑，新增修复策略时只需注册一个新函数，不用改核心流程。
3. **回放前必须做数据验证**：修复后的数据重新加载到正式表之前，要再跑一遍质量检查，确保修复没有引入新问题。
4. **定期统计异常数据的比例和趋势**：如果某类异常突然增多，可能是上游数据源出了问题，需要及时排查根因而不是只做下游修复。

**思路**：商品价格监控的异常检测规则需要覆盖格式、范围、逻辑、完整性四个维度。每种异常类型对应不同的检测逻辑和修复建议。

**答案**：

```
1. 格式异常：
   - 价格字段包含非数字字符（如 "￥99.00"、"约100元"）
   - 日期格式不标准（如 "2024年1月15日" 而非 "2024-01-15"）
   - 商品ID包含特殊字符或空格

2. 范围异常：

   - 折扣率超出 0-100% 范围
   - 库存数量为负数

3. 逻辑异常：
   - 促销价高于原价
   - 折扣率与促销价不一致（如折扣率 50% 但促销价是原价的 80%）
   - 商品上架时间晚于数据采集时间

4. 完整性异常：
   - 商品ID、商品名称、价格等必填字段为空
   - 有促销价但缺少原价
   - 有商品分类但分类ID不存在（引用完整性）
```

用代码实现：

```python
import pandas as pd
from typing import List
from dataclasses import dataclass


class ProductAnomalyDetector:
    """商品价格异常检测器"""

    def detect(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测所有异常"""
        anomalies = []
        anomalies.extend(self._detect_format(data))
        anomalies.extend(self._detect_range(data))
        anomalies.extend(self._detect_logic(data))
        anomalies.extend(self._detect_completeness(data))
        return anomalies

    def _detect_format(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测格式异常"""
        anomalies = []
        if "price" in data.columns:
            for idx, value in data["price"].items():
                if pd.notna(value) and isinstance(value, str):
                    if not value.replace(".", "").isdigit():
                        anomalies.append(AnomalyRecord(
                            record_index=idx,
                            column="price",
                            anomaly_type=AnomalyType.FORMAT,
                            severity=Severity.ERROR,
                            current_value=value,
                            expected_value="数值格式",
                            description=f"价格格式无效: {value}",
                            fix_suggestion="移除货币符号，转换为数值"
                        ))
        return anomalies

    def _detect_range(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测范围异常"""
        anomalies = []
        if "price" in data.columns:
            for idx, value in data["price"].items():
                if pd.notna(value):
                    try:
                        num_val = float(value)
                        if num_val < 0 or num_val > 1000000:
                            anomalies.append(AnomalyRecord(
                                record_index=idx,
                                column="price",
                                anomaly_type=AnomalyType.RANGE,
                                severity=Severity.ERROR,
                                current_value=value,
                                expected_value="0-1000000",
                                description=f"价格超出范围: {value}",
                                fix_suggestion="检查数据源或标记为异常"
                            ))
                    except (ValueError, TypeError):
                        pass
        return anomalies

    def _detect_logic(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测逻辑异常"""
        anomalies = []
        if "price" in data.columns and "promo_price" in data.columns:
            mask = (data["price"].notna() & data["promo_price"].notna() &
                    (data["promo_price"] > data["price"]))
            for idx in data[mask].index:
                anomalies.append(AnomalyRecord(
                    record_index=idx,
                    column="promo_price",
                    anomaly_type=AnomalyType.LOGIC,
                    severity=Severity.ERROR,
                    current_value=f"promo={data.loc[idx, 'promo_price']}, price={data.loc[idx, 'price']}",
                    expected_value="promo_price <= price",
                    description="促销价高于原价",
                    fix_suggestion="交换价格或标记为异常"
                ))
        return anomalies

    def _detect_completeness(self, data: pd.DataFrame) -> List[AnomalyRecord]:
        """检测完整性异常"""
        anomalies = []
        required = ["product_id", "product_name", "price"]
        for col in required:
            if col in data.columns:
                for idx in data[data[col].isnull()].index:
                    anomalies.append(AnomalyRecord(
                        record_index=idx,
                        column=col,
                        anomaly_type=AnomalyType.COMPLETENESS,
                        severity=Severity.CRITICAL if col == "product_id" else Severity.ERROR,
                        current_value=None,
                        expected_value="非空值",
                        description=f"必填字段 {col} 为空",
                        fix_suggestion="从数据源补充"
                    ))
        return anomalies
```

**要点**：
- 格式异常用类型检查和正则检测，范围异常用数值比较，逻辑异常用字段间关系验证
- 严重程度分级：product_id 为空是 CRITICAL（致命），价格超出范围是 ERROR（严重）
- 每条异常记录都附带修复建议，方便后续自动修复或人工处理

### 练习二

**思路**：自动修复价格字段中的 "￥" 符号，核心是用正则表达式提取数字部分，转换为 float。需要处理各种格式（"￥99.00"、"$199"、"约100元"）。

**答案**：

```python
import re
import pandas as pd
from typing import Dict, Any


def fix_price_format(data: pd.DataFrame, column: str = "price") -> Dict[str, Any]:
    """自动修复价格字段格式

    处理场景：
    - "￥99.00" -> 99.00
    - "$199" -> 199.0
    - "约100元" -> 100.0
    - "1,234.56" -> 1234.56
    - "面议" -> None
    """
    if column not in data.columns:
        return {"fixed_count": 0, "message": f"列 {column} 不存在"}

    result = data.copy()
    fixed_count = 0
    fix_details = []

    for idx, value in result[column].items():
        if pd.isna(value):
            continue

        original = value

        if isinstance(value, (int, float)):
            continue  # 已经是数值，不需要修复

        if isinstance(value, str):
            value = value.strip()

            # 处理"面议"
            if "面议" in value or "薪资不限" in value:
                result.loc[idx, column] = None
                fixed_count += 1
                fix_details.append({
                    "index": idx,
                    "original": original,
                    "fixed": None,
                    "strategy": "set_null"
                })
                continue

            # 提取数字部分
            cleaned = re.sub(r'[￥$€¥,元]', '', value)
            cleaned = re.sub(r'^约\s*', '', cleaned)
            cleaned = cleaned.strip()

            try:
                numeric_value = float(cleaned)
                result.loc[idx, column] = numeric_value
                fixed_count += 1
                fix_details.append({
                    "index": idx,
                    "original": original,
                    "fixed": numeric_value,
                    "strategy": "extract_number"
                })
            except ValueError:
                pass  # 无法解析，保留原值

    return {
        "fixed_count": fixed_count,
        "details": fix_details[:20],
        "data": result
    }

    {"product_id": "P002", "price": "$199"},
    {"product_id": "P003", "price": "约100元"},
    {"product_id": "P004", "price": "1,234.56"},
```

**要点**：
- 修复前先判断类型，已经是数值的跳过，避免不必要的转换
- "面议"等业务空值应设为 None 而非 0，避免误导统计
- 修复后保留原始值和修复策略，方便审计和回溯

### 练习三

**思路**：回放机制的核心流程是：获取已修复数据 → 验证质量 → 写入正式表 → 更新隔离表状态。每一步都需要错误处理，验证失败则阻止回放。

**答案**：

```python
from sqlalchemy import create_engine, text
import pandas as pd
from typing import Dict, Any, List
from datetime import datetime


class DataReplayer:
    """数据回放器"""

    def __init__(self, connection_string: str):
        self.engine = create_engine(connection_string)

    def replay(
        self,
        quarantine_table: str,
        target_table: str,
        key_columns: List[str],
        batch_id: str = None
    ) -> Dict[str, Any]:
        """回放已修复的数据

        流程：
        1. 从隔离表获取已修复数据
        2. 验证数据质量
        3. 使用事务写入正式表
        4. 更新隔离表状态
        """
        # 1. 获取已修复数据
        fixed_data = self._get_fixed_data(quarantine_table, batch_id)
        if fixed_data.empty:
            return {"replayed": 0, "message": "无待回放数据"}

        # 2. 验证数据质量
        validation = self._validate(fixed_data)
        if not validation["valid"]:
            return {
                "replayed": 0,
                "message": f"数据验证失败: {validation['errors']}",
                "validation_errors": validation["errors"]
            }

        # 3. 使用事务写入正式表
        try:
            replayed_count = self._load_to_target(
                fixed_data, target_table, key_columns
            )

            # 4. 更新隔离表状态
            self._update_status(quarantine_table, fixed_data["id"].tolist(), "replayed")

            # 5. 记录回放日志
            self._log_replay(batch_id, replayed_count)

            return {
                "replayed": replayed_count,
                "message": f"成功回放 {replayed_count} 条记录"
            }

        except Exception as e:
            return {
                "replayed": 0,
                "message": f"回放失败: {str(e)}",
                "error": str(e)
            }

    def _get_fixed_data(self, table: str, batch_id: str = None) -> pd.DataFrame:
        """获取已修复数据"""
        query = f"SELECT * FROM {table} WHERE status = 'fixed'"
        params = {}
        if batch_id:
            query += " AND batch_id = :batch_id"
            params["batch_id"] = batch_id
        return pd.read_sql(query, self.engine, params=params)

    def _validate(self, data: pd.DataFrame) -> Dict[str, Any]:
        """回放前验证"""
        errors = []

        # 检查必填字段
        required_fields = ["job_id", "title", "company"]
        for field in required_fields:
            if field in data.columns:
                null_count = data[field].isnull().sum()
                if null_count > 0:
                    errors.append(f"{field} 有 {null_count} 个空值")

        # 检查薪资逻辑
        if "salary_min" in data.columns and "salary_max" in data.columns:
            both = data["salary_min"].notna() & data["salary_max"].notna()
            bad = both & (data["salary_min"] > data["salary_max"])
            if bad.any():
                errors.append(f"salary_min > salary_max: {bad.sum()} 条")

        return {"valid": len(errors) == 0, "errors": errors}

    def _load_to_target(
        self, data: pd.DataFrame, target_table: str, key_columns: List[str]
    ) -> int:
        """使用事务写入正式表"""
        exclude_cols = [
            "id", "quarantine_time", "batch_id", "anomaly_summary",
            "status", "fix_time", "fix_by", "fix_note"
        ]
        load_cols = [c for c in data.columns if c not in exclude_cols]

        conn = self.engine.connect()
        trans = conn.begin()
        loaded = 0

        try:
            for _, row in data.iterrows():
                record = {col: row[col] for col in load_cols if pd.notna(row[col])}
                columns = ", ".join(record.keys())
                placeholders = ", ".join([f":{key}" for key in record.keys()])

                update_cols = [k for k in record.keys() if k not in key_columns]
                update_clause = ", ".join([f"{col} = EXCLUDED.{col}" for col in update_cols])
                conflict_cols = ", ".join(key_columns)

                query = f"""
                    INSERT INTO {target_table} ({columns})
                    VALUES ({placeholders})
                    ON CONFLICT ({conflict_cols})
                    DO UPDATE SET {update_clause}
                """
                conn.execute(text(query), record)
                loaded += 1

            trans.commit()
            return loaded

        except Exception as e:
            trans.rollback()
            raise e
        finally:
            conn.close()

    def _update_status(self, table: str, record_ids: List[int], status: str):
        """更新隔离表状态"""
        if not record_ids:
            return
        ids_str = ", ".join(str(id) for id in record_ids)
        query = f"UPDATE {table} SET status = '{status}' WHERE id IN ({ids_str})"
        with self.engine.connect() as conn:
            conn.execute(text(query))
            conn.commit()

    def _log_replay(self, batch_id: str, count: int):
        """记录回放日志"""
        query = """
            INSERT INTO etl_batch_log (batch_id, source, task_type, start_time, end_time, status, total_records, success_records)
            VALUES (:batch_id, 'quarantine', 'replay', NOW(), NOW(), 'completed', :count, :count)
        """
        try:
            with self.engine.connect() as conn:
                conn.execute(text(query), {"batch_id": f"replay_{batch_id}", "count": count})
                conn.commit()
        except Exception:
            pass  # 日志记录失败不影响主流程


# 使用示例
replayer = DataReplayer("postgresql://user:pass@localhost/db")
result = replayer.replay(
    quarantine_table="quarantine_data",
    target_table="clean_jobs",
    key_columns=["job_id"],
    batch_id="batch_20240115_001"
)
print(f"回放结果: {result['message']}")
```

**要点**：
- 回放前必须验证数据质量，防止修复后的数据仍有问题
- 写入正式表使用事务，失败时自动回滚，保证数据一致性
- 隔离表的元信息列（quarantine_time、status 等）不应写入正式表
