## 二、批次管理

### 2.1 批次号设计

```python
import uuid
from datetime import datetime
from typing import Optional

class BatchManager:
    """批次管理器"""
    
    @staticmethod
    def generate_batch_id(source: str, task_type: str = "etl") -> str:
        """
        生成批次号
        格式：{source}_{task_type}_{timestamp}_{uuid}
        """
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        unique_id = uuid.uuid4().hex[:8]
        return f"{source}_{task_type}_{timestamp}_{unique_id}"
    
    @staticmethod
    def parse_batch_id(batch_id: str) -> dict:
        """解析批次号"""
        parts = batch_id.split("_")
        if len(parts) >= 4:
            return {
                "source": parts[0],
                "task_type": parts[1],
                "date": parts[2],
                "time": parts[3],
                "unique_id": parts[4] if len(parts) > 4 else None
            }
        return {"raw": batch_id}

# 使用示例
batch_id = BatchManager.generate_batch_id("jobs_api", "extract")
# 输出: "jobs_api_extract_20240115_083000_a1b2c3d4"
```

### 2.3 批次生命周期

```python
from enum import Enum
from datetime import datetime
from typing import Optional, Dict, Any

class BatchStatus(Enum):
    """批次状态"""
    PENDING = "pending"      # 待执行
    RUNNING = "running"      # 执行中
    COMPLETED = "completed"  # 已完成
    FAILED = "failed"        # 失败
    CANCELLED = "cancelled"  # 已取消

class BatchLifecycle:
    """批次生命周期管理"""
    
    def __init__(self, engine):
        self.engine = engine
    
    def create_batch(
        self,
        batch_id: str,
        source: str,
        task_type: str,
        task_name: Optional[str] = None
    ) -> Dict[str, Any]:
        """创建批次记录"""
        query = """
            INSERT INTO etl_batch_log 
            (batch_id, source, task_type, task_name, start_time, status)
            VALUES (:batch_id, :source, :task_type, :task_name, NOW(), :status)
        """
        
        with self.engine.connect() as conn:
            conn.execute(text(query), {
                "batch_id": batch_id,
                "source": source,
                "task_type": task_type,
                "task_name": task_name,
                "status": BatchStatus.RUNNING.value
            })
            conn.commit()
        
        return {"batch_id": batch_id, "status": "created"}
    
    def update_progress(
        self,
        batch_id: str,
        total_records: int = None,
        success_records: int = None,
        failed_records: int = None
    ):
        """更新进度"""
        updates = []
        params = {"batch_id": batch_id}
        
        if total_records is not None:
            updates.append("total_records = :total_records")
            params["total_records"] = total_records
        
        if success_records is not None:
            updates.append("success_records = :success_records")
            params["success_records"] = success_records
        
        if failed_records is not None:
            updates.append("failed_records = :failed_records")
            params["failed_records"] = failed_records
        
        if updates:
            query = f"""
                UPDATE etl_batch_log
                SET {', '.join(updates)}, updated_at = NOW()
                WHERE batch_id = :batch_id
            """
            
            with self.engine.connect() as conn:
                conn.execute(text(query), params)
                conn.commit()
    
    def complete_batch(
        self,
        batch_id: str,
        total_records: int,
        success_records: int,
        failed_records: int = 0
    ):
        """完成批次"""
        query = """
            UPDATE etl_batch_log
            SET end_time = NOW(),
                duration = EXTRACT(EPOCH FROM (NOW() - start_time)),
                status = :status,
                total_records = :total_records,
                success_records = :success_records,
                failed_records = :failed_records,
                updated_at = NOW()
            WHERE batch_id = :batch_id
        """
        
        status = BatchStatus.COMPLETED.value if failed_records == 0 else BatchStatus.FAILED.value
        
        with self.engine.connect() as conn:
            conn.execute(text(query), {
                "batch_id": batch_id,
                "status": status,
                "total_records": total_records,
                "success_records": success_records,
                "failed_records": failed_records
            })
            conn.commit()
    
    def fail_batch(self, batch_id: str, error_message: str):
        """标记批次失败"""
        query = """
            UPDATE etl_batch_log
            SET end_time = NOW(),
                duration = EXTRACT(EPOCH FROM (NOW() - start_time)),
                status = :status,
                error_message = :error_message,
                updated_at = NOW()
            WHERE batch_id = :batch_id
        """
        
        with self.engine.connect() as conn:
            conn.execute(text(query), {
                "batch_id": batch_id,
                "status": BatchStatus.FAILED.value,
                "error_message": error_message
            })
            conn.commit()
    
    def get_batch(self, batch_id: str) -> Optional[Dict[str, Any]]:
        """获取批次信息"""
        query = """
            SELECT * FROM etl_batch_log WHERE batch_id = :batch_id
        """
        
        with self.engine.connect() as conn:
            result = conn.execute(text(query), {"batch_id": batch_id})
            row = result.fetchone()
            
            if row:
                return dict(row._mapping)
            return None
    
    def get_recent_batches(
        self,
        source: Optional[str] = None,
        status: Optional[str] = None,
        limit: int = 100
    ) -> pd.DataFrame:
        """获取最近的批次"""
        query = "SELECT * FROM etl_batch_log WHERE 1=1"
        params = {"limit": limit}
        
        if source:
            query += " AND source = :source"
            params["source"] = source
        
        if status:
            query += " AND status = :status"
            params["status"] = status
        
        query += " ORDER BY start_time DESC LIMIT :limit"
        
        return pd.read_sql(query, self.engine, params=params)
```

## 三、ETL 日志

### 3.1 日志级别

```python
import logging
from enum import Enum

class LogLevel(Enum):
    """日志级别"""
    DEBUG = "debug"      # 调试信息
    INFO = "info"        # 一般信息
    WARNING = "warning"  # 警告
    ERROR = "error"      # 错误
    CRITICAL = "critical"  # 严重错误

class ETLLogger:
    """ETL 日志器"""
    
    def __init__(self, batch_id: str, log_file: str = "etl.log"):
        self.batch_id = batch_id
        
        # 配置日志
        self.logger = logging.getLogger(f"etl.{batch_id}")
        self.logger.setLevel(logging.DEBUG)
        
        # 文件处理器
        file_handler = logging.FileHandler(log_file, encoding="utf-8")
        file_handler.setLevel(logging.DEBUG)
        
        # 控制台处理器
        console_handler = logging.StreamHandler()
        console_handler.setLevel(logging.INFO)
        
        # 格式
        formatter = logging.Formatter(
            "%(asctime)s - %(levelname)s - [%(batch_id)s] - %(message)s"
        )
        file_handler.setFormatter(formatter)
        console_handler.setFormatter(formatter)
        
        self.logger.addHandler(file_handler)
        self.logger.addHandler(console_handler)
    
    def _log(self, level: LogLevel, message: str, **kwargs):
        """记录日志"""
        extra = {"batch_id": self.batch_id, **kwargs}
        
        if level == LogLevel.DEBUG:
            self.logger.debug(message, extra=extra)
        elif level == LogLevel.INFO:
            self.logger.info(message, extra=extra)
        elif level == LogLevel.WARNING:
            self.logger.warning(message, extra=extra)
        elif level == LogLevel.ERROR:
            self.logger.error(message, extra=extra)
        elif level == LogLevel.CRITICAL:
            self.logger.critical(message, extra=extra)
    
    def info(self, message: str, **kwargs):
        self._log(LogLevel.INFO, message, **kwargs)
    
    def warning(self, message: str, **kwargs):
        self._log(LogLevel.WARNING, message, **kwargs)
    
    def error(self, message: str, **kwargs):
        self._log(LogLevel.ERROR, message, **kwargs)
    
    def debug(self, message: str, **kwargs):
        self._log(LogLevel.DEBUG, message, **kwargs)
```

### 3.2 结构化日志

```python
import json
from datetime import datetime
from typing import Dict, Any, Optional
from dataclasses import dataclass, asdict

@dataclass
class LogEntry:
    """日志条目"""
    timestamp: str
    level: str
    batch_id: str
    stage: str              # extract, transform, load
    event: str              # start, complete, error, warning
    message: str
    details: Optional[Dict[str, Any]] = None

class StructuredLogger:
    """结构化日志器"""
    
    def __init__(self, batch_id: str, log_file: str = "etl_structured.log"):
        self.batch_id = batch_id
        self.log_file = log_file
    
    def log(self, entry: LogEntry):
        """记录结构化日志"""
        with open(self.log_file, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(entry), ensure_ascii=False) + "\n")
    
    def log_extract_start(self, source: str):
        """记录提取开始"""
        self.log(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="info",
            batch_id=self.batch_id,
            stage="extract",
            event="start",
            message=f"开始从 {source} 提取数据"
        ))
    
    def log_extract_complete(self, record_count: int, duration: float):
        """记录提取完成"""
        self.log(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="info",
            batch_id=self.batch_id,
            stage="extract",
            event="complete",
            message=f"提取完成，共 {record_count} 条记录，耗时 {duration:.2f} 秒",
            details={"record_count": record_count, "duration": duration}
        ))
    
    def log_transform_start(self):
        """记录转换开始"""
        self.log(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="info",
            batch_id=self.batch_id,
            stage="transform",
            event="start",
            message="开始数据转换"
        ))
    
    def log_quality_check(self, passed: int, failed: int):
        """记录质量检查结果"""
        self.log(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="info" if failed == 0 else "warning",
            batch_id=self.batch_id,
            stage="transform",
            event="quality_check",
            message=f"质量检查: 通过 {passed}, 失败 {failed}",
            details={"passed": passed, "failed": failed}
        ))
    
    def log_load_start(self, target_table: str):
        """记录加载开始"""
        self.log(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="info",
            batch_id=self.batch_id,
            stage="load",
            event="start",
            message=f"开始加载到 {target_table}"
        ))
    
    def log_load_complete(self, loaded_count: int):
        """记录加载完成"""
        self.log(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="info",
            batch_id=self.batch_id,
            stage="load",
            event="complete",
            message=f"加载完成，共 {loaded_count} 条记录",
            details={"loaded_count": loaded_count}
        ))
    
    def log_error(self, stage: str, error: str):
        """记录错误"""
        self.log(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="error",
            batch_id=self.batch_id,
            stage=stage,
            event="error",
            message=f"错误: {error}",
            details={"error": error}
        ))
```

### 4.3 血缘追踪实现

```python
class LineageTracker:
    """数据血缘追踪器"""
    
    def __init__(self, engine):
        self.engine = engine
    
    def record_lineage(
        self,
        target_table: str,
        target_column: Optional[str],
        source_table: Optional[str] = None,
        source_column: Optional[str] = None,
        source_file: Optional[str] = None,
        source_api: Optional[str] = None,
        transform_type: str = "direct",
        transform_logic: Optional[str] = None,
        batch_id: Optional[str] = None
    ):
        """记录血缘关系"""
        query = """
            INSERT INTO data_lineage 
            (target_table, target_column, source_table, source_column,
             source_file, source_api, transform_type, transform_logic, batch_id)
            VALUES (:target_table, :target_column, :source_table, :source_column,
                    :source_file, :source_api, :transform_type, :transform_logic, :batch_id)
        """
        
        with self.engine.connect() as conn:
            conn.execute(text(query), {
                "target_table": target_table,
                "target_column": target_column,
                "source_table": source_table,
                "source_column": source_column,
                "source_file": source_file,
                "source_api": source_api,
                "transform_type": transform_type,
                "transform_logic": transform_logic,
                "batch_id": batch_id
            })
            conn.commit()
    
    def get_upstream(self, table_name: str) -> pd.DataFrame:
        """获取上游血缘"""
        query = """
            SELECT DISTINCT source_table, source_column, transform_type, transform_logic
            FROM data_lineage
            WHERE target_table = :table_name
        """
        return pd.read_sql(query, self.engine, params={"table_name": table_name})
    
    def get_downstream(self, table_name: str) -> pd.DataFrame:
        """获取下游血缘"""
        query = """
            SELECT DISTINCT target_table, target_column, transform_type, transform_logic
            FROM data_lineage
            WHERE source_table = :table_name
        """
        return pd.read_sql(query, self.engine, params={"table_name": table_name})
    
    def get_full_lineage(self, table_name: str) -> Dict[str, Any]:
        """获取完整血缘链路"""
        upstream = self.get_upstream(table_name)
        downstream = self.get_downstream(table_name)
        
        return {
            "table": table_name,
            "upstream": upstream.to_dict(orient="records"),
            "downstream": downstream.to_dict(orient="records")
        }
```

### 5.1 报告生成

```python
from datetime import datetime
from typing import Dict, Any, Optional

class ETLReportGenerator:
    """ETL 报告生成器"""
    
    def __init__(self, engine):
        self.engine = engine
    
    def generate_batch_report(self, batch_id: str) -> Dict[str, Any]:
        """生成批次报告"""
        
        # 获取批次信息
        batch_info = self._get_batch_info(batch_id)
        
        if not batch_info:
            return {"error": "批次不存在"}
        
        report = {
            "batch_id": batch_id,
            "report_time": datetime.now().isoformat(),
            "summary": {
                "source": batch_info["source"],
                "task_type": batch_info["task_type"],
                "status": batch_info["status"],
                "start_time": str(batch_info["start_time"]),
                "end_time": str(batch_info.get("end_time", "")),
                "duration": batch_info.get("duration", 0),
                "total_records": batch_info.get("total_records", 0),
                "success_records": batch_info.get("success_records", 0),
                "failed_records": batch_info.get("failed_records", 0)
            },
            "details": self._get_batch_details(batch_id)
        }
        
        return report
    
    def _get_batch_info(self, batch_id: str) -> Optional[Dict]:
        """获取批次信息"""
        query = "SELECT * FROM etl_batch_log WHERE batch_id = :batch_id"
        
        with self.engine.connect() as conn:
            result = conn.execute(text(query), {"batch_id": batch_id})
            row = result.fetchone()
            return dict(row._mapping) if row else None
    
    def _get_batch_details(self, batch_id: str) -> Dict[str, Any]:
        """获取批次详情"""
        details = {}
        
        # 获取质量检查结果
        details["quality_checks"] = self._get_quality_checks(batch_id)
        
        # 获取异常数据统计
        details["anomalies"] = self._get_anomaly_stats(batch_id)
        
        return details
    
    def _get_quality_checks(self, batch_id: str) -> list:
        """获取质量检查结果"""
        # 简化实现
        return []
```

## 六、完整的日志和追踪流程

```python
class ETLWithLogging:
    """带日志的 ETL 流程"""
    
    def __init__(self, engine):
        self.engine = engine
        self.batch_mgr = BatchLifecycle(engine)
        self.lineage = LineageTracker(engine)
        self.reporter = ETLReportGenerator(engine)
    
    def run(
        self,
        source: str,
        task_type: str,
        extract_func,
        transform_func,
        load_func,
        target_table: str
    ) -> Dict[str, Any]:
        """运行带日志的 ETL"""
        
        # 生成批次号
        batch_id = BatchManager.generate_batch_id(source, task_type)
        
        # 创建批次记录
        self.batch_mgr.create_batch(batch_id, source, task_type)
        
        # 初始化日志
        logger = StructuredLogger(batch_id)
        
        try:
            # 1. Extract
            logger.log_extract_start(source)
            start_time = datetime.now()
            
            raw_data = extract_func()
            
            duration = (datetime.now() - start_time).total_seconds()
            logger.log_extract_complete(len(raw_data), duration)
            
            # 2. Transform
            logger.log_transform_start()
            
            clean_data = transform_func(raw_data)
            
            # 记录血缘
            self.lineage.record_lineage(
                target_table=target_table,
                target_column=None,
                source_table=f"raw_{source}",
                transform_type="cleaned",
                transform_logic="数据清洗和标准化",
                batch_id=batch_id
            )
            
            # 3. Load
            logger.log_load_start(target_table)
            
            loaded_count = load_func(clean_data)
            
            logger.log_load_complete(loaded_count)
            
            # 完成批次
            self.batch_mgr.complete_batch(
                batch_id,
                total_records=len(raw_data),
                success_records=loaded_count
            )
            
            # 生成报告
            report = self.reporter.generate_batch_report(batch_id)
            
            return {
                "success": True,
                "batch_id": batch_id,
                "loaded_count": loaded_count,
                "report": report
            }
            
        except Exception as e:
            logger.log_error(task_type, str(e))
            self.batch_mgr.fail_batch(batch_id, str(e))
            
            return {
                "success": False,
                "batch_id": batch_id,
                "error": str(e)
            }
```

## 常见误区

- **用 `print` 当日志**：`print` 输出没有级别、没有时间戳、没有批次号，线上排查时根本找不到有用信息。必须用结构化日志，包含时间、级别、批次号、阶段、事件等字段。
- **只记录错误不记录正常流程**：只在 `except` 里写日志，成功时不记录。导致无法回答"这次运行处理了多少数据""每一步花了多长时间"等问题。
- **血缘关系只靠代码注释维护**：在代码里写 `# 数据来自 raw_jobs`，换个人维护就看不懂了。血缘关系应该存到数据库，和代码解耦，方便查询和可视化。
- **批次号设计太简单**：只用时间戳做批次号，同一秒跑两次就会冲突。批次号必须包含唯一标识（UUID 或随机串）。

---

## 工程建议

1. **结构化日志比纯文本日志更有价值**：用 JSON 格式记录日志，方便用 ELK、Grafana 等工具聚合分析，也方便按字段过滤和统计。
2. **每个 ETL 阶段都记录开始、进度、结束**：Extract 开始 → 提取了 1500 条 → 完成；Transform 开始 → 清洗了 1480 条 → 完成。出了问题能精确定位到哪个阶段。
3. **血缘追踪要记录到字段级别**：不仅记录 `clean_jobs` 来自 `raw_jobs`，还要记录 `salary_avg` 是由 `salary_min` 和 `salary_max` 计算得来的。
4. **定期清理过期日志和批次记录**：日志数据会持续增长，设置保留策略（如保留 90 天），定期归档或清理，避免影响查询性能。

### 练习一

**思路**：批次号需要包含数据源、任务类型、时间戳和唯一标识四部分，用下划线连接。格式为 `{source}_{task_type}_{YYYYMMDD}_{HHMMSS}_{uuid8位}`，既可读又唯一。

**答案**：

```python
import uuid
from datetime import datetime
from typing import Optional


class BatchIdGenerator:
    """批次号生成器"""

    @staticmethod
    def generate(source: str, task_type: str = "etl") -> str:
        """生成批次号

        格式: {source}_{task_type}_{date}_{time}_{uuid}
        示例: jobs_api_extract_20240115_083000_a1b2c3d4

        Args:
            source: 数据源标识（如 jobs_api, price_csv）
            task_type: 任务类型（如 extract, transform, load, etl）
        """
        now = datetime.now()
        date_part = now.strftime("%Y%m%d")
        time_part = now.strftime("%H%M%S")
        uuid_part = uuid.uuid4().hex[:8]

        return f"{source}_{task_type}_{date_part}_{time_part}_{uuid_part}"

    @staticmethod
    def parse(batch_id: str) -> dict:
        """解析批次号

        Returns:
            包含 source, task_type, date, time, uuid 的字典
        """
        parts = batch_id.split("_")
        if len(parts) >= 5:
            return {
                "source": parts[0],
                "task_type": parts[1],
                "date": parts[2],
                "time": parts[3],
                "uuid": "_".join(parts[4:])  # 处理 source 或 task_type 包含下划线的情况
            }
        return {"raw": batch_id, "error": "格式不正确"}

    @staticmethod
    def validate(batch_id: str) -> bool:
        """验证批次号格式"""
        parts = batch_id.split("_")
        if len(parts) < 5:
            return False

        # 验证日期部分
        try:
            datetime.strptime(parts[2], "%Y%m%d")
        except ValueError:
            return False

        # 验证时间部分
        try:
            datetime.strptime(parts[3], "%H%M%S")
        except ValueError:
            return False

        # 验证 UUID 部分（8位十六进制）
        uuid_part = parts[-1]
        if len(uuid_part) != 8:
            return False
        try:
            int(uuid_part, 16)
        except ValueError:
            return False

        return True


# 使用示例
gen = BatchIdGenerator()

# 生成批次号
batch_id = gen.generate("jobs_api", "extract")
print(f"批次号: {batch_id}")
# 输出: jobs_api_extract_20240115_083000_a1b2c3d4

# 解析批次号
info = gen.parse(batch_id)
print(f"数据源: {info['source']}, 任务类型: {info['task_type']}")

# 验证格式
print(gen.validate(batch_id))  # True
print(gen.validate("invalid_batch_id"))  # False
```

**要点**：
- UUID 只取 8 位十六进制，既保证唯一性又保持批次号长度合理
- `parse` 方法需要处理 source 或 task_type 本身包含下划线的情况
- `validate` 方法验证日期、时间、UUID 三部分的格式，方便排查批次号异常

### 练习二

**思路**：结构化日志器需要记录 ETL 各阶段的开始/结束、处理记录数、错误信息和质量检查结果。用 JSON 格式写入日志文件，方便后续用 ELK 等工具分析。

**答案**：

```python
import json
import logging
from datetime import datetime
from typing import Dict, Any, Optional, List
from dataclasses import dataclass, asdict, field


@dataclass
class LogEntry:
    """结构化日志条目"""
    timestamp: str
    level: str
    batch_id: str
    stage: str
    event: str
    message: str
    details: Optional[Dict[str, Any]] = None


class StructuredETLLogger:
    """结构化 ETL 日志器"""

    def __init__(self, batch_id: str, log_file: str = "etl_structured.log"):
        self.batch_id = batch_id
        self.log_file = log_file
        self.entries: List[LogEntry] = []

        # 同时配置标准 logging
        self.logger = logging.getLogger(f"etl.{batch_id}")
        self.logger.setLevel(logging.DEBUG)

        if not self.logger.handlers:
            handler = logging.FileHandler(log_file, encoding="utf-8")
            handler.setFormatter(logging.Formatter(
                "%(asctime)s - %(levelname)s - %(message)s"
            ))
            self.logger.addHandler(handler)

    def _write(self, entry: LogEntry):
        """写入日志"""
        self.entries.append(entry)

        # 写入文件
        with open(self.log_file, "a", encoding="utf-8") as f:
            f.write(json.dumps(asdict(entry), ensure_ascii=False) + "\n")

        # 同时用标准 logging
        level_map = {
            "debug": logging.DEBUG,
            "info": logging.INFO,
            "warning": logging.WARNING,
            "error": logging.ERROR,
            "critical": logging.CRITICAL
        }
        self.logger.log(level_map.get(entry.level, logging.INFO), entry.message)

    def log_stage_start(self, stage: str, message: str = None):
        """记录阶段开始"""
        self._write(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="info",
            batch_id=self.batch_id,
            stage=stage,
            event="start",
            message=message or f"开始 {stage} 阶段"
        ))

    def log_stage_complete(self, stage: str, record_count: int, duration: float):
        """记录阶段完成"""
        self._write(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="info",
            batch_id=self.batch_id,
            stage=stage,
            event="complete",
            message=f"{stage} 完成: {record_count} 条记录, 耗时 {duration:.2f} 秒",
            details={"record_count": record_count, "duration": duration}
        ))

    def log_record_count(self, stage: str, event: str, count: int, label: str = ""):
        """记录处理的记录数"""
        self._write(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="info",
            batch_id=self.batch_id,
            stage=stage,
            event=event,
            message=f"{stage} - {event}: {count} 条{label}",
            details={"count": count}
        ))

    def log_quality_check(self, passed: int, failed: int, warnings: int = 0):
        """记录质量检查结果"""
        level = "info" if failed == 0 else "warning"
        self._write(LogEntry(
            timestamp=datetime.now().isoformat(),
            level=level,
            batch_id=self.batch_id,
            stage="quality",
            event="check",
            message=f"质量检查: 通过 {passed}, 失败 {failed}, 警告 {warnings}",
            details={"passed": passed, "failed": failed, "warnings": warnings}
        ))

    def log_error(self, stage: str, error: str, details: Dict = None):
        """记录错误"""
        self._write(LogEntry(
            timestamp=datetime.now().isoformat(),
            level="error",
            batch_id=self.batch_id,
            stage=stage,
            event="error",
            message=f"错误: {error}",
            details=details or {"error": error}
        ))

    def get_summary(self) -> Dict[str, Any]:
        """获取日志摘要"""
        return {
            "batch_id": self.batch_id,
            "total_entries": len(self.entries),
            "error_count": sum(1 for e in self.entries if e.level == "error"),
            "warning_count": sum(1 for e in self.entries if e.level == "warning"),
            "stages": list(set(e.stage for e in self.entries))
        }


# 使用示例
logger = StructuredETLLogger("jobs_api_etl_20240115_083000_a1b2c3d4")

# 记录 Extract 阶段
logger.log_stage_start("extract", "开始从 jobs_api 提取数据")
logger.log_record_count("extract", "progress", 1500, "已提取")
logger.log_stage_complete("extract", 1500, 12.5)

# 记录 Transform 阶段
logger.log_stage_start("transform")
logger.log_record_count("transform", "cleaned", 1480, "清洗后")
logger.log_record_count("transform", "deduped", 20, "去重移除")
logger.log_quality_check(passed=8, failed=2, warnings=1)
logger.log_stage_complete("transform", 1480, 3.2)

# 记录 Load 阶段
logger.log_stage_start("load", "开始加载到 clean_jobs")
logger.log_record_count("load", "loaded", 1480, "写入")
logger.log_stage_complete("load", 1480, 5.8)

# 记录错误

# 查看摘要

```

**要点**：
- 每条日志包含 timestamp、level、batch_id、stage、event、message 六个核心字段
- `details` 字段存储可选的结构化数据（记录数、耗时、错误详情等）
- 同时写入 JSON 文件和标准 logging，JSON 便于工具分析，logging 便于实时查看

### 练习三

**思路**：血缘追踪器需要记录数据的来源（上游表/文件/API）、经过的转换（清洗/聚合/派生）、去向（下游表）。核心是在 ETL 各阶段调用 `record_lineage` 写入血缘表。

**答案**：

```python
from sqlalchemy import create_engine, text
import pandas as pd
from typing import Dict, Any, Optional, List
from datetime import datetime


class LineageTracker:
    """数据血缘追踪器"""

    def __init__(self, connection_string: str):
        self.engine = create_engine(connection_string)
        self._ensure_table()

    def _ensure_table(self):
        """确保血缘表存在"""
        query = """
            CREATE TABLE IF NOT EXISTS data_lineage (
                id              SERIAL PRIMARY KEY,
                target_table    VARCHAR(100) NOT NULL,
                target_column   VARCHAR(100),
                source_table    VARCHAR(100),
                source_column   VARCHAR(100),
                source_file     VARCHAR(200),
                source_api      VARCHAR(200),
                transform_type  VARCHAR(50),
                transform_logic TEXT,
                batch_id        VARCHAR(100),
                created_at      TIMESTAMP DEFAULT NOW()
            )
        """
        with self.engine.connect() as conn:
            conn.execute(text(query))
            conn.commit()

    def record(
        self,
        target_table: str,
        target_column: str = None,
        source_table: str = None,
        source_column: str = None,
        source_file: str = None,
        source_api: str = None,
        transform_type: str = "direct",
        transform_logic: str = None,
        batch_id: str = None
    ):
        """记录一条血缘关系

        Args:
            target_table: 目标表名
            target_column: 目标字段名
            source_table: 来源表名
            source_column: 来源字段名
            source_file: 来源文件路径
            source_api: 来源 API 地址
            transform_type: 转换类型（direct/derived/aggregated/cleaned）
            transform_logic: 转换逻辑描述
            batch_id: 关联的批次号
        """
        query = """
            INSERT INTO data_lineage
            (target_table, target_column, source_table, source_column,
             source_file, source_api, transform_type, transform_logic, batch_id)
            VALUES (:target_table, :target_column, :source_table, :source_column,
                    :source_file, :source_api, :transform_type, :transform_logic, :batch_id)
        """
        with self.engine.connect() as conn:
            conn.execute(text(query), {
                "target_table": target_table,
                "target_column": target_column,
                "source_table": source_table,
                "source_column": source_column,
                "source_file": source_file,
                "source_api": source_api,
                "transform_type": transform_type,
                "transform_logic": transform_logic,
                "batch_id": batch_id
            })
            conn.commit()

    def get_upstream(self, table_name: str) -> pd.DataFrame:
        """获取上游血缘（数据从哪来）"""
        query = """
            SELECT DISTINCT source_table, source_column, source_file, source_api,
                   transform_type, transform_logic
            FROM data_lineage
            WHERE target_table = :table_name
            ORDER BY source_table
        """
        return pd.read_sql(query, self.engine, params={"table_name": table_name})

    def get_downstream(self, table_name: str) -> pd.DataFrame:
        """获取下游血缘（数据到哪去）"""
        query = """
            SELECT DISTINCT target_table, target_column,
                   transform_type, transform_logic
            FROM data_lineage
            WHERE source_table = :table_name
            ORDER BY target_table
        """
        return pd.read_sql(query, self.engine, params={"table_name": table_name})

    def get_full_lineage(self, table_name: str) -> Dict[str, Any]:
        """获取完整血缘链路"""
        return {
            "table": table_name,
            "upstream": self.get_upstream(table_name).to_dict(orient="records"),
            "downstream": self.get_downstream(table_name).to_dict(orient="records")
        }

    def trace_origin(self, table_name: str, max_depth: int = 10) -> List[str]:
        """追溯数据的最上游来源"""
        origins = []
        visited = set()
        self._dfs_upstream(table_name, origins, visited, max_depth)
        return origins

    def _dfs_upstream(self, table: str, origins: List[str], visited: set, depth: int):
        """深度优先搜索上游"""
        if depth <= 0 or table in visited:
            return
        visited.add(table)

        upstream = self.get_upstream(table)
        if upstream.empty:
            origins.append(table)
            return

        for _, row in upstream.iterrows():
            source = row.get("source_table") or row.get("source_file") or row.get("source_api")
            if source:
                self._dfs_upstream(source, origins, visited, depth - 1)


# 使用示例

# 记录血缘关系

    target_table="raw_jobs",

    target_table="clean_jobs",
    target_column="salary_avg",
    source_table="raw_jobs",
    source_column="salary_min,salary_max",
    transform_type="derived",
    transform_logic="salary_avg = (salary_min + salary_max) / 2",
    batch_id="batch_20240115_001"
)

# 查询血缘
print("clean_jobs 的上游:")
print(tracker.get_upstream("clean_jobs"))

print("\nclean_jobs 的下游:")
print(tracker.get_downstream("clean_jobs"))

print("\nmetrics_city_daily 的完整血缘:")
print(tracker.get_full_lineage("metrics_city_daily"))

print("\n追溯 metrics_city_daily 的最上游来源:")
print(tracker.trace_origin("metrics_city_daily"))
```

**要点**：
- 血缘关系要记录到字段级别，不仅记录表级关系
- `trace_origin` 用深度优先搜索追溯最上游来源，支持多层血缘链路
- `_ensure_table` 确保血缘表存在，避免首次运行时报错
