### 1.2 筛选器实现

```python
from sqlalchemy import text
from typing import Dict, Any, List

class QueryBuilder:
    """查询构建器"""
    
    def __init__(self, base_table: str):
        self.base_table = base_table
        self.conditions = []
        self.params = {}
    
    def add_condition(self, field: str, operator: str, value: Any):
        """添加筛选条件"""
        if value is None:
            return
        
        param_name = f"param_{len(self.params)}"
        
        if operator == "eq":
            self.conditions.append(f"{field} = :{param_name}")
            self.params[param_name] = value
        elif operator == "ne":
            self.conditions.append(f"{field} != :{param_name}")
            self.params[param_name] = value
        elif operator == "gt":
            self.conditions.append(f"{field} > :{param_name}")
            self.params[param_name] = value
        elif operator == "lt":
            self.conditions.append(f"{field} < :{param_name}")
            self.params[param_name] = value
        elif operator == "gte":
            self.conditions.append(f"{field} >= :{param_name}")
            self.params[param_name] = value
        elif operator == "lte":
            self.conditions.append(f"{field} <= :{param_name}")
            self.params[param_name] = value
        elif operator == "in":
            placeholders = ", ".join([f":{param_name}_{i}" for i in range(len(value))])
            self.conditions.append(f"{field} IN ({placeholders})")
            for i, v in enumerate(value):
                self.params[f"{param_name}_{i}"] = v
        elif operator == "like":
            self.conditions.append(f"{field} LIKE :{param_name}")
            self.params[param_name] = f"%{value}%"
        elif operator == "between":
            self.conditions.append(f"{field} BETWEEN :{param_name}_start AND :{param_name}_end")
            self.params[f"{param_name}_start"] = value[0]
            self.params[f"{param_name}_end"] = value[1]
    
    def build_where_clause(self) -> str:
        """构建 WHERE 子句"""
        if not self.conditions:
            return "1=1"
        return " AND ".join(self.conditions)
    
    def build_count_query(self) -> str:
        """构建计数查询"""
        where = self.build_where_clause()
        return f"SELECT COUNT(*) FROM {self.base_table} WHERE {where}"
    
    def build_data_query(
        self,
        columns: List[str],
        sort_by: str,
        sort_order: str,
        limit: int,
        offset: int
    ) -> str:
        """构建数据查询"""
        where = self.build_where_clause()
        cols = ", ".join(columns)
        return f"""
            SELECT {cols}
            FROM {self.base_table}
            WHERE {where}
            ORDER BY {sort_by} {sort_order}
            LIMIT :limit OFFSET :offset
        """
```

### 2.1 排序设计

```python
from enum import Enum
from typing import List, Optional

class SortField(str, Enum):
    """排序字段"""
    PUBLISH_DATE = "publish_date"
    SALARY_AVG = "salary_avg"
    CITY = "city"
    COMPANY = "company"
    CREATED_AT = "created_at"

class SortOrder(str, Enum):
    """排序方向"""
    ASC = "asc"
    DESC = "desc"

class SortOption(BaseModel):
    """排序选项"""
    field: SortField
    order: SortOrder = SortOrder.DESC

# 多字段排序
class MultiSortRequest(BaseModel):
    """多字段排序请求"""
    sort_by: List[SortOption] = Field(
        default=[SortOption(field=SortField.PUBLISH_DATE, order=SortOrder.DESC)],
        description="排序规则"
    )
```

### 2.2 排序实现

```python
def build_order_clause(sort_by: List[SortOption]) -> str:
    """构建排序子句"""
    if not sort_by:
        return "id DESC"
    
    clauses = []
    for sort in sort_by:
        clauses.append(f"{sort.field.value} {sort.order.value}")
    
    return ", ".join(clauses)
```

## 三、聚合接口

### 3.1 聚合查询

```python
from pydantic import BaseModel, Field
from typing import List, Optional
from datetime import date

class AggregateRequest(BaseModel):
    """聚合请求"""
    group_by: List[str] = Field(..., description="分组字段")
    metrics: List[str] = Field(..., description="聚合指标: count, avg, sum, min, max")
    date_from: Optional[date] = Field(None, description="开始日期")
    date_to: Optional[date] = Field(None, description="结束日期")
    city: Optional[str] = Field(None, description="城市筛选")

class AggregateResponse(BaseModel):
    """聚合响应"""
    data: List[dict] = Field(..., description="聚合结果")
    total_groups: int = Field(..., description="分组数量")
    update_time: str = Field(..., description="更新时间")

# 实现
@router.post("/aggregate", response_model=AggregateResponse)
async def aggregate_jobs(request: AggregateRequest):
    """聚合查询"""
    
    # 构建聚合查询
    group_cols = ", ".join(request.group_by)
    
    metric_clauses = []
    for metric in request.metrics:
        if metric == "count":
            metric_clauses.append("COUNT(*) as count")
        elif metric == "avg_salary":
            metric_clauses.append("AVG(salary_avg) as avg_salary")
        elif metric == "max_salary":
            metric_clauses.append("MAX(salary_avg) as max_salary")
        elif metric == "min_salary":
            metric_clauses.append("MIN(salary_avg) as min_salary")
    
    metrics_sql = ", ".join(metric_clauses)
    
    # 构建 WHERE 条件
    conditions = []
    params = {}
    
    if request.date_from:
        conditions.append("publish_date >= :date_from")
        params["date_from"] = request.date_from
    
    if request.date_to:
        conditions.append("publish_date <= :date_to")
        params["date_to"] = request.date_to
    
    if request.city:
        conditions.append("city = :city")
        params["city"] = request.city
    
    where_clause = " AND ".join(conditions) if conditions else "1=1"
    
    sql = f"""
        SELECT {group_cols}, {metrics_sql}
        FROM clean_jobs
        WHERE {where_clause}
        GROUP BY {group_cols}
        ORDER BY count DESC
    """
    
    with engine.connect() as conn:
        result = conn.execute(text(sql), params)
        data = [dict(row._mapping) for row in result.fetchall()]
    
    return AggregateResponse(
        data=data,
        total_groups=len(data),
        update_time=datetime.now().isoformat()
    )
```

```python
from fastapi.responses import StreamingResponse
import csv
import io

@router.get("/export")
async def export_jobs(
    city: Optional[str] = None,
    format: str = Query("csv", enum=["csv", "json"])
):
    """导出数据"""
    
    # 查询数据
    data = await job_service.list_jobs(city=city, page_size=10000)
    
    if format == "csv":
        # CSV 导出
        output = io.StringIO()
        writer = csv.DictWriter(output, fieldnames=data["data"][0].keys())
        writer.writeheader()
        writer.writerows(data["data"])
        
        return StreamingResponse(
            io.BytesIO(output.getvalue().encode('utf-8-sig')),
            media_type="text/csv",
            headers={"Content-Disposition": "attachment; filename=jobs.csv"}
        )
    
    elif format == "json":
        # JSON 导出
        return StreamingResponse(
            io.BytesIO(json.dumps(data["data"], ensure_ascii=False).encode('utf-8')),
            media_type="application/json",
            headers={"Content-Disposition": "attachment; filename=jobs.json"}
        )
```

## 常见误区

- **为每种筛选组合写独立接口**：接口数量会指数增长，维护成本极高，应该设计通用的筛选参数，让一个接口支持多种组合
- **直接拼接用户输入到 SQL 字符串**：这是 SQL 注入的典型漏洞，必须使用参数化查询，所有用户输入都通过参数绑定
- **聚合查询不做任何限制**：用户可能传入大量 group_by 字段导致查询超时，应该限制分组字段数量和查询时间范围
- **导出接口不做数据量限制**：用户可能导出全表数据导致内存溢出，应该限制导出行数或使用流式导出

---

## 工程建议

- 设计 QueryBuilder 类封装查询构建逻辑，支持链式调用添加筛选条件，避免在路由函数中拼接 SQL
- 使用 Enum 约束排序字段和操作符，防止用户传入非法字段名，同时自动生成 API 文档的可选值列表
- 聚合接口限制 group_by 字段数量（如最多 3 个）和时间范围（如最多 365 天），防止恶意查询拖垮数据库
- 大数据量导出使用 StreamingResponse + 流式写入，避免一次性加载所有数据到内存

## 参考答案

### 练习一

**思路**：使用 QueryBuilder 模式封装筛选逻辑，根据参数动态拼接 WHERE 子句，所有参数通过绑定传入防止 SQL 注入。

**答案**：

```python
from fastapi import APIRouter, Query
from typing import List, Optional
from datetime import date
from pydantic import BaseModel, Field
from sqlalchemy import create_engine, text

router = APIRouter()
engine = create_engine("postgresql://user:pass@localhost/jobs_db")

class JobFilterRequest(BaseModel):
    cities: Optional[List[str]] = Field(default=[], description="城市列表")
    salary_min: Optional[float] = Field(None, ge=0, description="最低薪资")
    salary_max: Optional[float] = Field(None, ge=0, description="最高薪资")
    experience: Optional[List[str]] = Field(default=[], description="经验要求")
    education: Optional[List[str]] = Field(default=[], description="学历要求")
    skills: Optional[List[str]] = Field(default=[], description="技能要求")
    date_from: Optional[date] = Field(None, description="开始日期")
    date_to: Optional[date] = Field(None, description="结束日期")
    page: int = Field(1, ge=1, description="页码")
    page_size: int = Field(20, ge=1, le=100, description="每页数量")
    sort_by: str = Field("publish_date", description="排序字段")
    sort_order: str = Field("desc", description="排序方向")

@router.post("/jobs/filter")
async def filter_jobs(request: JobFilterRequest):
    """多条件筛选接口"""

    conditions = []
    params = {
        "limit": request.page_size,
        "offset": (request.page - 1) * request.page_size,
    }

    # 城市列表筛选
    if request.cities:
        placeholders = ", ".join([f":city_{i}" for i in range(len(request.cities))])
        conditions.append(f"city IN ({placeholders})")
        for i, city in enumerate(request.cities):
            params[f"city_{i}"] = city

    # 薪资范围筛选
    if request.salary_min is not None:
        conditions.append("salary_avg >= :salary_min")
        params["salary_min"] = request.salary_min
    if request.salary_max is not None:
        conditions.append("salary_avg <= :salary_max")
        params["salary_max"] = request.salary_max

    # 经验要求筛选
    if request.experience:
        placeholders = ", ".join([f":exp_{i}" for i in range(len(request.experience))])
        conditions.append(f"experience IN ({placeholders})")
        for i, exp in enumerate(request.experience):
            params[f"exp_{i}"] = exp

    # 技能筛选（模糊匹配任意一个技能）
    if request.skills:
        skill_conditions = []
        for i, skill in enumerate(request.skills):
            skill_conditions.append(f"skills ILIKE :skill_{i}")
            params[f"skill_{i}"] = f"%{skill}%"
        conditions.append(f"({' OR '.join(skill_conditions)})")

    # 日期范围筛选
    if request.date_from:
        conditions.append("publish_date >= :date_from")
        params["date_from"] = request.date_from
    if request.date_to:
        conditions.append("publish_date <= :date_to")
        params["date_to"] = request.date_to

    where_clause = " AND ".join(conditions) if conditions else "1=1"

    # 白名单校验排序字段
    allowed_sort = {"publish_date", "salary_avg", "city", "company"}
    sort_by = request.sort_by if request.sort_by in allowed_sort else "publish_date"
    sort_order = "DESC" if request.sort_order == "desc" else "ASC"

    count_sql = f"SELECT COUNT(*) FROM clean_jobs WHERE {where_clause}"
    data_sql = f"""
        SELECT job_id, title, company, city, salary_avg, experience, skills, publish_date
        FROM clean_jobs
        WHERE {where_clause}
        ORDER BY {sort_by} {sort_order}
        LIMIT :limit OFFSET :offset
    """

    with engine.connect() as conn:
        total = conn.execute(text(count_sql), params).scalar()
        result = conn.execute(text(data_sql), params)
        data = [dict(row._mapping) for row in result.fetchall()]

    return {
        "data": data,
        "total": total,
        "page": request.page,
        "page_size": request.page_size,
        "total_pages": (total + request.page_size - 1) // request.page_size,
    }
```

**要点**：
- 所有用户输入通过参数绑定（`:param`），绝不拼接原始字符串
- 列表类型参数用 IN 查询，技能用 ILIKE 模糊匹配
- 排序字段用白名单校验，防止注入

### 练习二

**思路**：使用 GROUP BY 分组 + 聚合函数（COUNT、AVG 等），支持多维度统计。

**答案**：

```python
from fastapi import APIRouter, Query
from typing import Optional
from datetime import date
from sqlalchemy import create_engine, text

router = APIRouter()
engine = create_engine("postgresql://user:pass@localhost/jobs_db")

@router.get("/stats/cities")
async def get_city_stats(
    date_from: Optional[date] = Query(None, description="开始日期"),
    date_to: Optional[date] = Query(None, description="结束日期"),
    limit: int = Query(20, ge=1, le=100),
):
    """按城市统计岗位数量和平均薪资"""
    conditions = []
    params = {"limit": limit}

    if date_from:
        conditions.append("publish_date >= :date_from")
        params["date_from"] = date_from
    if date_to:
        conditions.append("publish_date <= :date_to")
        params["date_to"] = date_to

    where_clause = " AND ".join(conditions) if conditions else "1=1"

    sql = f"""
        SELECT
            city,
            COUNT(*) AS job_count,
            ROUND(AVG(salary_avg), 2) AS avg_salary,
            MAX(salary_avg) AS max_salary,
            MIN(salary_avg) AS min_salary
        FROM clean_jobs
        WHERE {where_clause} AND salary_avg IS NOT NULL
        GROUP BY city
        ORDER BY job_count DESC
        LIMIT :limit
    """

    with engine.connect() as conn:
        result = conn.execute(text(sql), params)
        data = [dict(row._mapping) for row in result.fetchall()]

    return {"data": data, "total_groups": len(data)}


@router.get("/stats/skills")
async def get_skill_stats(

@router.get("/stats/trends")
async def get_trends(

    where_clause = " AND ".join(conditions)

            ROUND(AVG(salary_avg), 2) AS avg_salary

        WHERE {where_clause}
```

**要点**：
- 聚合查询限制返回条数（LIMIT），防止结果集过大
- 技能字段用 `UNNEST(string_to_array)` 拆分后统计
- 时间趋势用 `INTERVAL` 做日期范围过滤

### 练习三

**思路**：使用 StreamingResponse 流式写入，避免一次性加载全部数据到内存；CSV 用 utf-8-sig 编码确保中文在 Excel 中正常显示。

**答案**：

```python
import csv
import io
import json

from fastapi.responses import StreamingResponse
from typing import Optional

@router.get("/jobs/export")
async def export_jobs(

    salary_min: Optional[float] = Query(None, ge=0),
    salary_max: Optional[float] = Query(None, ge=0),
    format: str = Query("csv", enum=["csv", "json"], description="导出格式"),
    limit: int = Query(10000, ge=1, le=50000, description="最大导出行数"),
):
    """导出岗位数据，支持筛选条件和大数据量分批导出"""

    conditions = []
    params = {"limit": limit}

    if city:
        conditions.append("city = :city")
        params["city"] = city
    if salary_min is not None:
        conditions.append("salary_avg >= :salary_min")
        params["salary_min"] = salary_min
    if salary_max is not None:
        conditions.append("salary_avg <= :salary_max")
        params["salary_max"] = salary_max

    where_clause = " AND ".join(conditions) if conditions else "1=1"

    sql = f"""
        SELECT job_id, title, company, city, salary_avg, experience, skills, publish_date
        FROM clean_jobs
        WHERE {where_clause}
        ORDER BY publish_date DESC
        LIMIT :limit
    """

    with engine.connect() as conn:
        result = conn.execute(text(sql), params)
        rows = [dict(row._mapping) for row in result.fetchall()]

    if format == "csv":
        output = io.StringIO()
        if rows:
            writer = csv.DictWriter(output, fieldnames=rows[0].keys())
            writer.writeheader()
            writer.writerows(rows)

        return StreamingResponse(
            io.BytesIO(output.getvalue().encode("utf-8-sig")),
            media_type="text/csv; charset=utf-8",
            headers={"Content-Disposition": "attachment; filename=jobs_export.csv"},
        )

    elif format == "json":
        return StreamingResponse(
            io.BytesIO(json.dumps(rows, ensure_ascii=False, default=str).encode("utf-8")),
            media_type="application/json; charset=utf-8",
            headers={"Content-Disposition": "attachment; filename=jobs_export.json"},
        )
```

**要点**：
- CSV 用 `utf-8-sig` 编码，Excel 打开不会乱码
- `default=str` 处理 date/datetime 等 JSON 不能序列化的类型
- 限制最大导出行数（le=50000），防止内存溢出
