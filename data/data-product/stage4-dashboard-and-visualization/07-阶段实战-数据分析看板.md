## 筛选上下文

所有组件共享筛选状态，用 Context 管理：

```tsx
import { createContext, useContext, useState, useMemo, useCallback } from "react";

interface FilterState {
  department: string;
  source: string;
  keyword: string;
}

const FilterContext = createContext<{
  filters: FilterState;
  setFilters: (f: Partial<FilterState>) => void;
} | null>(null);

function useFilter() {
  const ctx = useContext(FilterContext);
  if (!ctx) throw new Error("useFilter must be inside FilterProvider");
  return ctx;
}
```

## 指标卡组件

数字要大，趋势要一眼看出涨跌。加载时显示骨架屏：

```tsx
function MetricCard({ title, value, change, loading }: {
  title: string; value: string | number; change?: string; loading?: boolean;
}) {
  if (loading) {
    return (
      <div className="border rounded-lg p-4 flex-1 bg-gray-50 animate-pulse">
        <div className="h-4 w-16 bg-gray-200 rounded mb-2" />
        <div className="h-8 w-24 bg-gray-200 rounded" />
      </div>
    );
  }
  return (
    <div className="border rounded-lg p-4 flex-1">
      <div className="text-sm text-gray-500">{title}</div>
      <div className="text-2xl font-bold mt-1">{value}</div>
      {change && (
        <div className={`text-xs mt-1 ${change.startsWith("+") ? "text-green-600" : "text-red-600"}`}>
          {change}
        </div>
      )}
    </div>
  );
}
```

## 导出与状态处理

```tsx
function ExportButton({ data }: { data: Record<string, unknown>[] }) {
  const [exporting, setExporting] = useState(false);
  const handleExport = useCallback(async () => {
    if (data.length === 0) return;
    setExporting(true);
    try {
      const csv = Object.keys(data[0]).join(",") + "\n" +
        data.map((r) => Object.values(r).join(",")).join("\n");
      const blob = new Blob(["\uFEFF" + csv], { type: "text/csv;charset=utf-8" });
      const a = document.createElement("a");
      a.href = URL.createObjectURL(blob);
      a.download = `dashboard-${Date.now()}.csv`;
      a.click();
    } finally { setExporting(false); }
  }, [data]);
  return (
    <button onClick={handleExport} disabled={exporting || data.length === 0}
      className="px-4 py-2 bg-blue-600 text-white rounded text-sm disabled:opacity-50">
      {exporting ? "导出中..." : "导出 CSV"}
    </button>
  );
}

function EmptyState({ message }: { message: string }) {
  return <div className="text-center py-16 text-gray-400">📭 {message}</div>;
}

function ErrorState({ message, onRetry }: { message: string; onRetry: () => void }) {
  return (
    <div className="text-center py-16 text-red-500">
      <div className="mb-3">⚠️ {message}</div>
      <button onClick={onRetry} className="px-4 py-2 border rounded text-sm">重试</button>
    </div>
  );
}
```

## 组装完整页面

把所有组件拼起来。FilterBar 内联在 Dashboard 中：

```tsx
const MOCK_JOBS = [
  { id: 1, title: "前端工程师", company: "字节跳动", salary: "25-40K", source: "boss", department: "技术", city: "北京" },
  { id: 2, title: "后端工程师", company: "阿里巴巴", salary: "30-50K", source: "lagou", department: "技术", city: "杭州" },
  { id: 3, title: "产品经理", company: "腾讯", salary: "20-35K", source: "zhilian", department: "产品", city: "深圳" },
  { id: 4, title: "算法工程师", company: "百度", salary: "35-55K", source: "boss", department: "技术", city: "北京" },
];

export default function Dashboard() {
  const [filters, setFiltersRaw] = useState<FilterState>({ department: "all", source: "all", keyword: "" });
  const setFilters = useCallback((p: Partial<FilterState>) => setFiltersRaw((s) => ({ ...s, ...p })), []);

  const filteredData = useMemo(() => {
    return MOCK_JOBS.filter((job) => {
      if (filters.department !== "all" && job.department !== filters.department) return false;
      if (filters.source !== "all" && job.source !== filters.source) return false;
      if (filters.keyword && !job.title.includes(filters.keyword)) return false;
      return true;
    });
  }, [filters]);

  const cityRanking = useMemo(() => {
    const counts: Record<string, number> = {};
    filteredData.forEach((j) => { counts[j.city] = (counts[j.city] || 0) + 1; });
    return Object.entries(counts).sort(([, a], [, b]) => b - a)
      .map(([name, value], i) => ({ rank: i + 1, name, value }));
  }, [filteredData]);

  return (
    <FilterContext.Provider value={{ filters, setFilters }}>
      <div className="p-6 max-w-6xl mx-auto">
        <div className="flex justify-between items-center mb-6">
          <h1 className="text-xl font-bold">招聘数据分析看板</h1>
          <ExportButton data={filteredData} />
        </div>
        <div className="flex flex-wrap gap-3 mb-6">
          <select value={filters.department} onChange={(e) => setFilters({ department: e.target.value })}
            className="border rounded px-3 py-2 text-sm">
            <option value="all">全部部门</option>
            <option value="技术">技术</option>
            <option value="产品">产品</option>
          </select>
          <select value={filters.source} onChange={(e) => setFilters({ source: e.target.value })}
            className="border rounded px-3 py-2 text-sm">
            <option value="all">全部来源</option>
            <option value="boss">Boss直聘</option>
            <option value="lagou">拉勾</option>
          </select>
          <input type="text" placeholder="搜索职位..." value={filters.keyword}
            onChange={(e) => setFilters({ keyword: e.target.value })}
            className="border rounded px-3 py-2 text-sm flex-1 min-w-[200px]" />
        </div>
        <div className="flex gap-4 mb-6">
          <MetricCard title="总职位数" value={filteredData.length} change="+12%" />
          <MetricCard title="平均薪资" value="24.3K" change="+5%" />
          <MetricCard title="覆盖城市" value={`${new Set(filteredData.map((j) => j.city)).size}个`} />
        </div>
        <div className="flex gap-4 mb-6">
          <RankingTable data={cityRanking} title="城市排名" />
        </div>
        <div className="border rounded-lg p-4">
          <h2 className="text-sm font-semibold mb-3">职位明细</h2>
          <table className="w-full border-collapse text-sm">
            <thead>
              <tr className="border-b-2 text-left text-gray-500">
                <th className="pb-2">职位</th><th className="pb-2">公司</th>
                <th className="pb-2">薪资</th><th className="pb-2">城市</th>
              </tr>
            </thead>
            <tbody>
              {filteredData.length === 0 ? (
                <tr><td colSpan={4} className="py-8 text-center text-gray-400">暂无匹配数据</td></tr>
              ) : filteredData.map((job) => (
                <tr key={job.id} className="border-b hover:bg-gray-50">
                  <td className="py-2">{job.title}</td><td className="py-2">{job.company}</td>
                  <td className="py-2">{job.salary}</td><td className="py-2">{job.city}</td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>
    </FilterContext.Provider>
  );
}
```

## 参考答案

### 练习一

在 Dashboard 中添加 `loading` 和 `error` 状态。加载时返回骨架屏，出错时返回 ErrorState：

```tsx
const [loading, setLoading] = useState(true);
const [error, setError] = useState<string | null>(null);
const fetchData = useCallback(async () => {
  setLoading(true); setError(null);
  try { const res = await fetch("/api/jobs"); if (!res.ok) throw new Error(`${res.status}`); setData(await res.json()); }
  catch (e: any) { setError(e.message); } finally { setLoading(false); }
}, []);
useEffect(() => { fetchData(); }, [fetchData]);
if (loading) return <div className="flex gap-4 p-6"><MetricCard title="" value="" loading /><MetricCard title="" value="" loading /></div>;
if (error) return <ErrorState message={error} onRetry={fetchData} />;
```

### 练习二

用 `localKeyword` 做本地状态，300ms 后同步到 Context：

```tsx
const [localKeyword, setLocalKeyword] = useState(filters.keyword);
useEffect(() => {
  const timer = setTimeout(() => setFilters({ keyword: localKeyword }), 300);
  return () => clearTimeout(timer);
}, [localKeyword]);
```
