## 调度器与 ETL 整合

把调度器和 ETL 管道串在一个入口文件里，任务失败时自动触发告警：

```python
# app/scheduler.py
from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger
from apscheduler.triggers.interval import IntervalTrigger
from app.pipeline import crawl_pipeline, incremental_pipeline
from app.alert import send_alert
from app.database import save_task_log
import uuid, logging

logger = logging.getLogger(__name__)
scheduler = BackgroundScheduler(timezone="Asia/Shanghai")

def run_task(task_type: str, task_id: str, trigger_type: str):
    logger.info(f"任务开始: {task_id} ({task_type})")
    save_task_log(task_id, task_type, "running", trigger_type)
    try:
        if task_type == "crawl":
            crawl_pipeline.run()
        elif task_type == "incremental":
            incremental_pipeline.run()
        elif task_type == "quality_check":
            run_quality_check()
        save_task_log(task_id, task_type, "completed", trigger_type)
    except Exception as e:
        error_msg = f"{type(e).__name__}: {e}"
        save_task_log(task_id, task_type, "failed", trigger_type, error=error_msg)
        send_alert(title=f"任务失败: {task_type}",
                   message=f"任务 {task_id} 失败\n{error_msg}", level="error")

def register_jobs():
    scheduler.add_job(
        lambda: run_task("crawl", f"task_{uuid.uuid4().hex[:12]}", "scheduled"),
        CronTrigger(hour=8, minute=0), id="daily_crawl", replace_existing=True, max_instances=1)
    scheduler.add_job(
        lambda: run_task("incremental", f"task_{uuid.uuid4().hex[:12]}", "scheduled"),
        IntervalTrigger(hours=1), id="hourly_incr", replace_existing=True, max_instances=1)
    scheduler.add_job(
        lambda: run_task("quality_check", f"task_{uuid.uuid4().hex[:12]}", "scheduled"),
        CronTrigger(hour=12, minute=0), id="daily_quality", replace_existing=True)
    scheduler.start()
```

## 告警模块

支持 Webhook（企业微信/钉钉/飞书）和邮件两种通知渠道：

```python
# app/alert.py
import requests, logging, os
from datetime import datetime

logger = logging.getLogger(__name__)
WEBHOOK_URL = os.getenv("ALERT_WEBHOOK_URL", "")

def send_alert(title: str, message: str, level: str = "warning"):
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    logger.warning(f"[{level.upper()}] {title} - {message}")

    if WEBHOOK_URL:
        try:
            requests.post(WEBHOOK_URL, json={
                "msgtype": "markdown",
                "markdown": {"content": f"### {title}\n> 级别: {level} | {timestamp}\n\n{message}"},
            }, timeout=10)
        except Exception as e:
            logger.error(f"Webhook 发送失败: {e}")

    if level == "error" and os.getenv("ALERT_EMAIL"):
        _send_email(title, message, timestamp)

def _send_email(title, message, timestamp):
    import smtplib
    from email.mime.text import MIMEText
    host, user, pwd = os.getenv("SMTP_HOST"), os.getenv("SMTP_USER"), os.getenv("SMTP_PASS")
    if not all([host, user, pwd]): return
    msg = MIMEText(f"{message}\n\n时间: {timestamp}")
    msg["Subject"] = f"[数据产品告警] {title}"
    msg["From"] = user
    msg["To"] = os.getenv("ALERT_EMAIL")
    try:
        with smtplib.SMTP_SSL(host, 465) as s:
            s.login(user, pwd); s.send_message(msg)
    except Exception as e:
        logger.error(f"邮件发送失败: {e}")
```

## 监控 API

面向前端 Dashboard 的系统状态接口：

```python
# app/routers/monitor.py
from fastapi import APIRouter
from sqlalchemy import text
from datetime import date
from app.database import get_db

router = APIRouter()

@router.get("/api/monitor/status")
async def get_system_status():
    db = get_db()
    try:
        db.execute(text("SELECT 1")); db_ok = True
    except Exception:
        db_ok = False
    last_update = db.execute(text("SELECT MAX(updated_at) FROM jobs")).scalar()
    hours = round((datetime.now() - last_update).total_seconds() / 3600, 1) if last_update else None
    return {"database": "healthy" if db_ok else "down", "data_fresh_hours": hours,
            "data_status": "fresh" if hours and hours < 2 else "stale"}

@router.get("/api/monitor/tasks")
async def get_task_overview():
    row = get_db().execute(text("""
        SELECT COUNT(*) FILTER (WHERE status='running') AS running,
               COUNT(*) FILTER (WHERE status='completed') AS completed,
               COUNT(*) FILTER (WHERE status='failed') AS failed
        FROM task_log WHERE created_at::date = :today
    """), {"today": date.today()}).fetchone()
    return dict(row._mapping)

@router.get("/api/monitor/tasks/recent")
async def get_recent_tasks(limit: int = 10):
    rows = get_db().execute(text(
        "SELECT task_id, task_type, status, trigger_type, start_time, end_time, error_message "
        "FROM task_log ORDER BY created_at DESC LIMIT :l"), {"l": limit}).fetchall()
    return [dict(r._mapping) for r in rows]
```

## FastAPI 应用入口

用 `lifespan` 管理调度器生命周期：

```python
# app/main.py
from contextlib import asynccontextmanager
from fastapi import FastAPI
from app.scheduler import register_jobs, scheduler
from app.routers import monitor, tasks

@asynccontextmanager
async def lifespan(app: FastAPI):
    register_jobs()
    yield
    scheduler.shutdown()

app = FastAPI(title="招聘数据产品", version="1.0.0", lifespan=lifespan)
app.include_router(monitor.router, tags=["监控"])
app.include_router(tasks.router, tags=["任务"])

@app.get("/health")
async def health_check():
    return {"status": "ok"}
```

## Docker Compose 编排

三个容器：API、PostgreSQL、Redis：

```yaml
# docker-compose.yml
version: "3.8"
services:
  db:
    image: postgres:15-alpine
    environment:
      POSTGRES_DB: job_data
      POSTGRES_USER: app
      POSTGRES_PASSWORD: ${DB_PASSWORD}
    volumes:
      - pgdata:/var/lib/postgresql/data
      - ./scripts/init.sql:/docker-entrypoint-initdb.d/init.sql
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U app -d job_data"]
      interval: 10s
      retries: 5
  api:
    build: .
    ports: ["8000:8000"]
    environment:
      DATABASE_URL: postgresql://app:${DB_PASSWORD}@db:5432/job_data
      ALERT_WEBHOOK_URL: ${ALERT_WEBHOOK_URL}
    depends_on:
      db: { condition: service_healthy }
    restart: unless-stopped
    command: uvicorn app.main:app --host 0.0.0.0 --port 8000 --workers 1
  redis:
    image: redis:7-alpine
    volumes: [redisdata:/data]
    restart: unless-stopped
volumes:
  pgdata:
  redisdata:
```

`workers=1` 因为调度器在进程内运行，多 worker 会导致任务重复。Dockerfile：

```dockerfile
FROM python:3.11-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY app/ app/ COPY scripts/ scripts/
EXPOSE 8000
CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8000", "--workers", "1"]
```

数据库初始化 `scripts/init.sql`：创建 `jobs`、`task_log`、`alert_log` 三张表，`task_log` 建立 `status` 和 `created_at DESC` 索引。

## 常见误区

- **一步到位上生产**——分阶段上线：先跑通 ETL，再加调度，最后加告警
- **只看功能不看可观测性**——没有监控和告警，出问题只能靠用户反馈
- **忽略回滚方案**——部署前给镜像打 tag，出问题切回旧版本
- **验收标准模糊**——"系统能跑"不是标准，必须有数字：成功率 > 99%、数据新鲜度 < 2h

## 参考答案

### 练习一

```python
@router.post("/tasks/trigger/{task_type}")
async def trigger_task(task_type: str, bg: BackgroundTasks):
    if task_type not in ("crawl", "incremental"):
        return {"error": "无效类型"}
    running = db.execute(text(
        "SELECT task_id FROM task_log WHERE task_type=:t AND status='running' LIMIT 1"),
        {"t": task_type}).fetchone()
    if running:
        return {"error": "任务正在运行中", "running_task_id": running[0]}
    task_id = f"task_{uuid.uuid4().hex[:12]}"
    bg.add_task(run_task, task_type, task_id, "manual")
    return {"task_id": task_id, "status": "queued"}
```
