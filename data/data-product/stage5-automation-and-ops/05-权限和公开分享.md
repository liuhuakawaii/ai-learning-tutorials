### 1.1 实现代码

from functools import wraps
from fastapi import HTTPException, Depends

def require_permission(permission: str):
    """权限检查装饰器"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            user = get_current_user()
            if not has_permission(user, permission):
                raise HTTPException(status_code=403, detail="权限不足")
            return await func(*args, **kwargs)
        return wrapper
    return decorator

import uuid
from datetime import datetime, timedelta

class ShareManager:
    """分享管理器"""
    
    def create_share_link(self, resource_id: str, expires_days: int = 7) -> str:
        """创建分享链接"""
        share_token = uuid.uuid4().hex
        expires_at = datetime.now() + timedelta(days=expires_days)
        
        # 保存到数据库
        save_share_token(share_token, resource_id, expires_at)
        
        return f"/share/{share_token}"
    
    def validate_share_token(self, token: str) -> bool:
        """验证分享令牌"""
        share = get_share_token(token)
        if not share:
            return False
        return share.expires_at > datetime.now()

## 三、数据脱敏

```python
def mask_sensitive_data(data: dict) -> dict:
    """脱敏敏感数据"""
    masked = data.copy()
    
    # 手机号脱敏
    if "phone" in masked:
        masked["phone"] = masked["phone"][:3] + "****" + masked["phone"][-4:]
    
    # 邮箱脱敏
    if "email" in masked:
        parts = masked["email"].split("@")
        masked["email"] = parts[0][:2] + "***@" + parts[1]
    
    return masked
```

## 常见误区

- **一开始就设计 RBAC 权限框架**：对于内部数据产品，简单的角色表 + 权限检查装饰器就够了。过早引入完整的 RBAC 框架会增加不必要的复杂度。
- **分享链接永不过期**：没有设置过期时间的分享链接一旦泄露，数据就永久暴露。所有分享链接都必须有有效期。
- **前端隐藏按钮等于权限控制**：不在页面上显示"导出"按钮不等于用户不能调用导出 API。权限校验必须在后端执行，前端只是体验优化。
- **脱敏只在展示层做**：如果 API 返回了完整手机号，前端做脱敏毫无意义——抓包就能看到原始数据。脱敏必须在数据出库前完成。

---

## 工程建议

1. **权限校验做成中间件**：用装饰器或中间件统一处理权限检查，不要在每个接口里重复写 if-else 判断。
2. **分享链接绑定访问次数**：除了过期时间，还可以限制最大访问次数，防止链接被大量转发后数据被批量爬取。
3. **敏感字段在 SQL 层脱敏**：在数据库查询阶段就对手机号、邮箱等字段做脱敏处理，而不是在 Python 代码里逐条处理，减少数据泄露风险。
4. **权限变更要审计**：记录谁在什么时间给谁分配了什么权限，方便安全审计和问题追溯。

## 参考答案

### 练习一

**思路**：设计一个基于角色的权限模型，用字典定义角色到权限的映射关系。权限检查通过装饰器实现，装饰器从请求中提取当前用户角色，校验是否拥有指定权限。角色和权限的映射关系集中管理，方便扩展。

**答案**：

```python
from enum import Enum
from functools import wraps
from fastapi import HTTPException, Depends, Request

class Role(Enum):
    ADMIN = "admin"
    ANALYST = "analyst"
    VIEWER = "viewer"
    GUEST = "guest"

class Permission(Enum):
    DATA_READ = "data:read"
    DATA_EXPORT = "data:export"
    TASK_MANAGE = "task:manage"
    ALERT_MANAGE = "alert:manage"
    SYSTEM_ADMIN = "system:admin"

ROLE_PERMISSIONS: dict[Role, set[Permission]] = {
    Role.ADMIN: {
        Permission.DATA_READ, Permission.DATA_EXPORT,
        Permission.TASK_MANAGE, Permission.ALERT_MANAGE, Permission.SYSTEM_ADMIN,
    },
    Role.ANALYST: {
        Permission.DATA_READ, Permission.DATA_EXPORT,
    },
    Role.VIEWER: {Permission.DATA_READ},
    Role.GUEST: {Permission.DATA_READ},
}

# 角色继承：ANALYST 包含 VIEWER 的权限，ADMIN 包含所有权限
# 已在 ROLE_PERMISSIONS 中显式定义，无需额外继承逻辑

def get_current_user(request: Request) -> dict:
    """从请求中解析当前用户（示意，实际应从 JWT/Session 中获取）"""
    token = request.headers.get("Authorization", "").replace("Bearer ", "")
    if not token:
        return {"role": Role.GUEST, "user_id": "anonymous"}
    user = decode_token(token)  # 实际实现需解码 JWT
    return user

def require_permission(permission: Permission):
    """权限检查装饰器"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            request: Request = kwargs.get("request") or args[0] if args else None
            if not request:
                for arg in args:
                    if isinstance(arg, Request):
                        request = arg
                        break
            user = get_current_user(request)
            user_role = user.get("role", Role.GUEST)
            allowed = ROLE_PERMISSIONS.get(user_role, set())
            if permission not in allowed:
                raise HTTPException(
                    status_code=403,
                    detail=f"权限不足: 需要 {permission.value}，当前角色 {user_role.value}",
                )
            return await func(*args, **kwargs)
        return wrapper
    return decorator

# 使用示例
from fastapi import APIRouter

router = APIRouter()

@router.get("/api/jobs")
@require_permission(Permission.DATA_READ)
async def list_jobs(request: Request):
    return {"jobs": [...]}

@router.get("/api/export")
@require_permission(Permission.DATA_EXPORT)
async def export_data(request: Request):
    return {"download_url": "..."}

@router.post("/api/tasks/{task_id}/cancel")
@require_permission(Permission.TASK_MANAGE)
async def cancel_task(task_id: str, request: Request):
    return {"message": "任务已取消"}

@router.get("/api/system/health")
@require_permission(Permission.SYSTEM_ADMIN)
async def system_health(request: Request):
    return {"status": "ok"}
```

**要点**：
- `ROLE_PERMISSIONS` 字典集中管理角色-权限映射，新增角色只需添加一行配置
- 权限检查通过装饰器实现，接口代码只需一行 `@require_permission(...)` 即可，不侵入业务逻辑
- 未登录用户默认为 `GUEST` 角色，只有 `data:read` 权限，不能导出或管理任务

### 练习二

**思路**：实现 `ShareManager` 类，支持创建带过期时间和访问次数限制的分享链接。验证时同时检查过期时间和剩余访问次数，任一条件不满足则拒绝访问。分享链接绑定具体资源和权限范围。

**答案**：

```python
import uuid
from datetime import datetime, timedelta
from sqlalchemy import text
from fastapi import APIRouter, HTTPException, Depends

router = APIRouter()

class ShareManager:
    """分享链接管理器"""

    def __init__(self, engine):
        self.engine = engine

    def create_share_link(
        self,
        resource_id: str,
        resource_type: str,
        created_by: str,
        expires_days: int = 7,
        max_access_count: int = None,
        allowed_permissions: list[str] = None,
    ) -> dict:
        """创建分享链接"""
        token = uuid.uuid4().hex
        expires_at = datetime.now() + timedelta(days=expires_days)

        query = """
            INSERT INTO share_links
                (token, resource_id, resource_type, created_by, expires_at, max_access_count, allowed_permissions, created_at)
            VALUES
                (:token, :resource_id, :resource_type, :created_by, :expires_at, :max_count, :permissions, NOW())
        """
        with self.engine.connect() as conn:
            conn.execute(text(query), {
                "token": token,
                "resource_id": resource_id,
                "resource_type": resource_type,
                "created_by": created_by,
                "expires_at": expires_at,
                "max_count": max_access_count,
                "permissions": json.dumps(allowed_permissions or ["data:read"]),
            })
            conn.commit()

        return {
            "token": token,
            "url": f"/share/{token}",
            "expires_at": str(expires_at),
            "max_access_count": max_access_count,
        }

    def validate_and_consume(self, token: str) -> dict:
        """验证分享令牌并消耗一次访问次数"""
        query = "SELECT * FROM share_links WHERE token = :token"
        with self.engine.connect() as conn:
            row = conn.execute(text(query), {"token": token}).fetchone()

        if not row:
            raise ValueError("分享链接不存在")

        if datetime.now() > row.expires_at:
            raise ValueError("分享链接已过期")

        if row.max_access_count is not None and row.access_count >= row.max_access_count:
            raise ValueError("分享链接访问次数已用尽")

        update = "UPDATE share_links SET access_count = access_count + 1, last_accessed_at = NOW() WHERE token = :token"
        with self.engine.connect() as conn:
            conn.execute(text(update), {"token": token})
            conn.commit()

        return {
            "resource_id": row.resource_id,
            "resource_type": row.resource_type,
            "permissions": json.loads(row.allowed_permissions),
        }

# 建表 SQL
SHARE_TABLE_SQL = """
CREATE TABLE IF NOT EXISTS share_links (
    id                   SERIAL PRIMARY KEY,
    token                VARCHAR(64) UNIQUE NOT NULL,
    resource_id          VARCHAR(100) NOT NULL,
    resource_type        VARCHAR(50) NOT NULL,
    created_by           VARCHAR(100) NOT NULL,
    expires_at           TIMESTAMP NOT NULL,
    max_access_count     INTEGER,
    access_count         INTEGER DEFAULT 0,
    allowed_permissions  JSONB DEFAULT '["data:read"]',
    created_at           TIMESTAMP DEFAULT NOW(),
    last_accessed_at     TIMESTAMP
);
CREATE INDEX IF NOT EXISTS idx_share_token ON share_links(token);
"""

# API 接口
@router.post("/api/share")
async def create_share(resource_id: str, resource_type: str, expires_days: int = 7, max_access: int = None):
    user = get_current_user()
    if not has_permission(user, Permission.DATA_EXPORT):
        raise HTTPException(status_code=403, detail="无权创建分享链接")
    manager = ShareManager(engine)
    result = manager.create_share_link(
        resource_id=resource_id,
        resource_type=resource_type,
        created_by=user["user_id"],
        expires_days=expires_days,
        max_access_count=max_access,
    )
    return result

@router.get("/share/{token}")
async def access_share(token: str):
    manager = ShareManager(engine)
    try:
        info = manager.validate_and_consume(token)
    except ValueError as e:
        raise HTTPException(status_code=403, detail=str(e))
    resource = get_resource(info["resource_id"], info["resource_type"])
    return apply_permissions(resource, info["permissions"])

def apply_permissions(resource: dict, permissions: list[str]) -> dict:
    """根据权限范围过滤资源数据"""
    if "data:export" not in permissions:
        resource.pop("export_url", None)
        resource.pop("raw_data", None)
    return resource
```

**要点**：
- 分享链接同时支持过期时间和最大访问次数两种限制，满足不同安全需求
- `validate_and_consume` 方法在验证的同时消耗一次访问计数，原子操作防止并发绕过
- `apply_permissions` 根据分享链接绑定的权限范围过滤返回数据，确保访客只能看到授权内容
