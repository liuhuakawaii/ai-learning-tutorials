## 二、状态机实现

```python
from enum import Enum
from typing import Optional

class TaskStatus(Enum):
    QUEUED = "queued"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"
    RETRYING = "retrying"

class TaskStateMachine:
    """任务状态机"""
    
    # 允许的状态转换
    TRANSITIONS = {
        TaskStatus.QUEUED: [TaskStatus.RUNNING, TaskStatus.CANCELLED],
        TaskStatus.RUNNING: [TaskStatus.COMPLETED, TaskStatus.FAILED, TaskStatus.CANCELLED],
        TaskStatus.FAILED: [TaskStatus.RETRYING, TaskStatus.CANCELLED],
        TaskStatus.RETRYING: [TaskStatus.RUNNING, TaskStatus.CANCELLED],
    }
    
    def can_transition(self, from_status: TaskStatus, to_status: TaskStatus) -> bool:
        """检查是否允许状态转换"""
        allowed = self.TRANSITIONS.get(from_status, [])
        return to_status in allowed
    
    def transition(self, task_id: str, to_status: TaskStatus, error: str = None):
        """执行状态转换"""
        current = self.get_status(task_id)
        
        if not self.can_transition(current, to_status):
            raise ValueError(f"不允许从 {current} 转换到 {to_status}")
        
        # 更新状态
        self.update_status(task_id, to_status, error)
```

## 常见误区

- **用字符串随意管理状态**：直接用 `"running"`、`"done"` 等字符串存储状态，没有枚举约束，容易出现非法状态（如 `"runing"` 拼写错误）。
- **不限制状态转换**：允许从任意状态跳转到任意状态，导致 `completed` 的任务被重新设为 `running`，破坏业务逻辑。
- **状态机只管最终状态**：忽略中间状态（如 `retrying`），导致重试逻辑和状态管理脱节。
- **把业务逻辑和状态转换混在一起**：状态机只负责状态流转，不应该在 `transition()` 方法里执行采集、通知等业务逻辑。

---

## 工程建议

1. **用 Enum 定义状态**：Python 的 `Enum` 能在编译期捕获非法状态值，比字符串安全得多。
2. **状态转换表要覆盖所有合法路径**：画出状态流转图，确保每个转换都有明确的触发条件和边界处理。
3. **记录状态变更历史**：每次状态变更都写一条日志（from_status → to_status + 时间戳），方便回溯任务生命周期。
4. **状态机要线程安全**：如果多个进程可能同时操作同一个任务，状态转换需要加锁或用数据库的乐观锁来防止竞态条件。

## 参考答案

### 练习一

**思路**：用 Python 的 Enum 定义所有任务状态，用字典定义合法的状态转换表。状态机类封装转换逻辑，每次转换前校验是否合法，转换后记录历史日志。重点是只允许画出的状态流转图中定义的转换路径。

**答案**：

```python
from enum import Enum
from datetime import datetime

class TaskStatus(Enum):
    QUEUED = "queued"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"
    RETRYING = "retrying"

class TaskStateMachine:
    """任务状态机"""

    TRANSITIONS = {
        TaskStatus.QUEUED:    [TaskStatus.RUNNING, TaskStatus.CANCELLED],
        TaskStatus.RUNNING:   [TaskStatus.COMPLETED, TaskStatus.FAILED, TaskStatus.CANCELLED],
        TaskStatus.FAILED:    [TaskStatus.RETRYING, TaskStatus.CANCELLED],
        TaskStatus.RETRYING:  [TaskStatus.RUNNING, TaskStatus.CANCELLED],
    }

    def __init__(self, engine):
        self.engine = engine

    def can_transition(self, from_status: TaskStatus, to_status: TaskStatus) -> bool:
        allowed = self.TRANSITIONS.get(from_status, [])
        return to_status in allowed

    def get_status(self, task_id: str) -> TaskStatus:
        query = "SELECT status FROM task_log WHERE task_id = :task_id"
        with self.engine.connect() as conn:
            row = conn.execute(query, {"task_id": task_id}).fetchone()
            if not row:
                raise ValueError(f"任务 {task_id} 不存在")
            return TaskStatus(row[0])

    def transition(self, task_id: str, to_status: TaskStatus, error: str = None) -> dict:
        """执行状态转换，返回转换记录"""
        current = self.get_status(task_id)

        if not self.can_transition(current, to_status):
            raise ValueError(f"不允许从 {current.value} 转换到 {to_status.value}")

        now = datetime.now()
        with self.engine.connect() as conn:
            update_fields = {"status": to_status.value, "task_id": task_id}
            if to_status == TaskStatus.RUNNING:
                conn.execute(
                    "UPDATE task_log SET status = :status, start_time = :now WHERE task_id = :task_id",
                    {**update_fields, "now": now},
                )
            elif to_status in (TaskStatus.COMPLETED, TaskStatus.FAILED, TaskStatus.CANCELLED):
                conn.execute(
                    "UPDATE task_log SET status = :status, end_time = :now, error_message = :error WHERE task_id = :task_id",
                    {**update_fields, "now": now, "error": error},
                )
            else:
                conn.execute(
                    "UPDATE task_log SET status = :status WHERE task_id = :task_id",
                    update_fields,
                )

            conn.execute(
                """INSERT INTO task_status_history (task_id, from_status, to_status, changed_at, reason)
                   VALUES (:task_id, :from_s, :to_s, :now, :reason)""",
                {"task_id": task_id, "from_s": current.value, "to_s": to_status.value, "now": now, "reason": error},
            )
            conn.commit()

        return {"task_id": task_id, "from": current.value, "to": to_status.value, "time": str(now)}

# 状态历史表
HISTORY_TABLE_SQL = """
CREATE TABLE IF NOT EXISTS task_status_history (
    id          SERIAL PRIMARY KEY,
    task_id     VARCHAR(50) NOT NULL,
    from_status VARCHAR(20) NOT NULL,
    to_status   VARCHAR(20) NOT NULL,
    changed_at  TIMESTAMP DEFAULT NOW(),
    reason      TEXT
);
CREATE INDEX IF NOT EXISTS idx_status_history_task ON task_status_history(task_id);
"""

# 使用示例
sm = TaskStateMachine(engine)
sm.transition("task_abc123", TaskStatus.RUNNING)
sm.transition("task_abc123", TaskStatus.FAILED, error="网络超时")
sm.transition("task_abc123", TaskStatus.RETRYING)
sm.transition("task_abc123", TaskStatus.RUNNING)
sm.transition("task_abc123", TaskStatus.COMPLETED)
```

**要点**：
- `TRANSITIONS` 字典严格定义合法转换路径，任何未定义的转换都会抛出 `ValueError`
- 每次转换都往 `task_status_history` 表写一条记录，包含 from_status、to_status、时间和原因，形成完整审计链
- 不同的目标状态触发不同的字段更新（如 RUNNING 更新 start_time，FAILED 更新 end_time 和 error_message）

### 练习二

**思路**：基于 FastAPI 实现任务管理 API，包含查询详情、列表查询、取消和重试四个接口。所有状态变更都通过状态机执行，确保不会出现非法转换。

**答案**：

from fastapi import APIRouter, HTTPException, Query
from pydantic import BaseModel

router = APIRouter()

    task_type: str
    status: str
    trigger_type: str
    start_time: str | None
    end_time: str | None
    error_message: str | None

@router.get("/tasks/{task_id}", response_model=TaskResponse)
async def get_task(task_id: str):
    """查询单个任务详情"""
    query = "SELECT * FROM task_log WHERE task_id = :task_id"
    with engine.connect() as conn:
        row = conn.execute(query, {"task_id": task_id}).fetchone()
    if not row:
        raise HTTPException(status_code=404, detail="任务不存在")
    return {
        "task_id": row.task_id,
        "task_type": row.task_type,
        "status": row.status,
        "trigger_type": row.trigger_type,
        "start_time": str(row.start_time) if row.start_time else None,
        "end_time": str(row.end_time) if row.end_time else None,
        "error_message": row.error_message,
    }

@router.get("/tasks")
async def list_tasks(
    status: str = Query(None, description="按状态筛选"),
    task_type: str = Query(None, description="按任务类型筛选"),
    limit: int = Query(20, ge=1, le=100),
    offset: int = Query(0, ge=0),
):
    """查询任务列表"""
    conditions = []
    params = {"limit": limit, "offset": offset}
    if status:
        conditions.append("status = :status")
        params["status"] = status
    if task_type:
        conditions.append("task_type = :task_type")
        params["task_type"] = task_type

    where = "WHERE " + " AND ".join(conditions) if conditions else ""
    query = f"SELECT * FROM task_log {where} ORDER BY created_at DESC LIMIT :limit OFFSET :offset"

    with engine.connect() as conn:
        rows = conn.execute(query, params).fetchall()
    return [dict(row) for row in rows]
