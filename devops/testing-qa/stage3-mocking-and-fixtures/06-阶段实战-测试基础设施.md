# 阶段实战：测试基础设施

## 问题

项目规模增长到几十个页面、上百个组件时，每个测试文件都重复写着 `render(<Provider><Component /></Provider>)`、手写 mock 数据、用 `toBeTruthy()` 判断复杂结构。测试代码又长又难维护。

本课搭建完整的测试基础设施：自定义 render、全局 Mock、数据工厂、自定义 matcher。

## 自定义 render 函数

```tsx
// src/test-infrastructure/render.tsx
import React, { type ReactElement, type ReactNode } from 'react'
import { render } from '@testing-library/react'
import { Provider } from 'react-redux'
import { BrowserRouter } from 'react-router-dom'
import { configureStore } from '@reduxjs/toolkit'
import { authReducer } from '../store/auth.slice'
import { todosReducer } from '../store/todos.slice'

interface CustomRenderOptions {
  preloadedState?: any
  theme?: 'light' | 'dark'
}

function createWrapper(options: CustomRenderOptions = {}) {
  const { preloadedState } = options
  const store = configureStore({
    reducer: { auth: authReducer, todos: todosReducer },
    preloadedState,
  })

  function Wrapper({ children }: { children: ReactNode }) {
    return (
      <Provider store={store}>
        <BrowserRouter>{children}</BrowserRouter>
      </Provider>
    )
  }
  return { Wrapper, store }
}

export function customRender(ui: ReactElement, options: CustomRenderOptions = {}) {
  const { Wrapper, store } = createWrapper(options)
  return { ...render(ui, { wrapper: Wrapper }), store }
}

export function renderWithAuth(ui: ReactElement, userOverride?: any) {
  return customRender(ui, {
    preloadedState: {
      auth: {
        user: { id: '1', email: 'test@example.com', name: '测试用户', role: 'user', ...userOverride },
        token: 'mock-token',
      },
    },
  })
}
```

一行搞定 Provider 包装：

```tsx
test('显示任务列表', () => {
  renderWithTodos(<TodoList />, [createTodo({ title: '买咖啡' })])
  expect(screen.getByText('买咖啡')).toBeInTheDocument()

```

// src/test-infrastructure/mocks/handlers.ts
import { http, HttpResponse } from 'msw'
import { createTodo, createUser } from '../factories'

export const handlers = [
  http.get('/api/todos', () => HttpResponse.json({ data: [createTodo(), createTodo()], total: 2 })),
  http.post('/api/todos', async ({ request }) => {
    const body = await request.json() as any
    return HttpResponse.json(createTodo(body), { status: 201 })
  }),
  http.post('/api/auth/login', async ({ request }) => {
    const { email, password } = await request.json() as any
    if (password === 'wrong') return HttpResponse.json({ error: '密码错误' }, { status: 401 })
    return HttpResponse.json({ user: createUser({ email }), token: 'mock-jwt-token' })
  }),
]

## 自定义 Matcher

```typescript
// src/test-infrastructure/matchers/index.ts
import { screen } from '@testing-library/react'

function toBeVisibleWithText(text: string | RegExp) {
  try {
    const el = screen.getByText(text)
    return {
      pass: el.offsetParent !== null,
      message: () => `期望元素包含 "${text}" 的可见性不匹配`,
    }
  } catch {
    return { pass: false, message: () => `未找到包含 "${text}" 的元素` }
  }
}

expect.extend({ toBeVisibleWithText })

declare module 'vitest' {
  interface Assertion {
    toBeVisibleWithText(text: string | RegExp): void
  }
}
```

### 练习二

```typescript
function toHaveStoreState(store: any, key: string, expected: any) {
  const keys = key.split('.')
  let actual = store.getState()
  for (const k of keys) {
    if (actual == null) return { pass: false, message: () => `路径 "${key}" 在 "${k}" 处中断` }
    actual = actual[k]
  }
  const pass = JSON.stringify(actual) === JSON.stringify(expected)
  return {
    pass,
    message: () => pass
      ? `期望 store.${key} 不等于 ${JSON.stringify(expected)}`
      : `期望 store.${key} 等于 ${JSON.stringify(expected)}，实际为 ${JSON.stringify(actual)}`,
  }
}
```
