# 阶段实战：工具函数库测试

## 任务

你的团队有一个公共工具函数库，包含日期格式化、货币格式化、防抖、节流、深拷贝、邮箱验证等函数。这个库被多个项目依赖，任何 bug 都会影响所有项目。

本课为这些函数编写完整的测试套件，覆盖正常路径、边界值和异常路径。

## 测试规划策略

每个函数覆盖三类测试：

```typescript
// 1. 正常路径：输入符合预期，验证基本功能
// 2. 边界值：最小值、最大值、空值、特殊值
// 3. 异常路径：无效输入、类型错误、业务规则违反
```

## formatDate

```typescript
// src/utils/formatDate.ts
export function formatDate(
  date: Date | string | number,
  format: string = 'YYYY-MM-DD'
): string {
  const d = new Date(date)
  if (isNaN(d.getTime())) throw new Error('Invalid date')
  const year = d.getFullYear()
  const month = String(d.getMonth() + 1).padStart(2, '0')
  const day = String(d.getDate()).padStart(2, '0')
  const hours = String(d.getHours()).padStart(2, '0')
  const minutes = String(d.getMinutes()).padStart(2, '0')
  const seconds = String(d.getSeconds()).padStart(2, '0')
  return format
    .replace('YYYY', String(year)).replace('MM', month).replace('DD', day)
    .replace('HH', hours).replace('mm', minutes).replace('ss', seconds)
}
```

```typescript
describe('formatDate', () => {
  test('默认格式', () => { expect(formatDate(new Date(2024, 0, 15))).toBe('2024-01-15') })
  test('完整格式', () => {
    expect(formatDate(new Date(2024, 0, 15, 14, 30, 45), 'YYYY-MM-DD HH:mm:ss'))
      .toBe('2024-01-15 14:30:45')
  })
  test('Unix 纪元', () => { expect(formatDate(0)).toBe('1970-01-01') })
  test('闰年 2 月 29 日', () => { expect(formatDate(new Date(2024, 1, 29))).toBe('2024-02-29') })
  test('补零', () => { expect(formatDate(new Date(2024, 0, 5))).toBe('2024-01-05') })
  test('无效日期抛出异常', () => { expect(() => formatDate('invalid')).toThrow('Invalid date') })
})
```

## debounce

```typescript
// src/utils/debounce.ts
export function debounce<T extends (...args: any[]) => any>(
  fn: T, delay: number
): (...args: Parameters<T>) => void {
  let timer: ReturnType<typeof setTimeout> | null = null
  return (...args) => {
    if (timer) clearTimeout(timer)
    timer = setTimeout(() => { fn(...args); timer = null }, delay)
  }
}
```

```typescript
describe('debounce', () => {
  beforeEach(() => { vi.useFakeTimers() })
  afterEach(() => { vi.useRealTimers() })

  test('延迟执行', () => {
    const fn = vi.fn()
    debounce(fn, 300)()
    expect(fn).not.toHaveBeenCalled()
    vi.advanceTimersByTime(300)
    expect(fn).toHaveBeenCalledOnce()
  })

  test('多次调用只执行最后一次', () => {
    const fn = vi.fn()
    const debounced = debounce(fn, 300)
    debounced('first'); debounced('last')
    vi.advanceTimersByTime(300)
    expect(fn).toHaveBeenCalledWith('last')
  })

  test('中途调用重置定时器', () => {
    const fn = vi.fn()
    const debounced = debounce(fn, 300)
    debounced(); vi.advanceTimersByTime(200)
    debounced(); vi.advanceTimersByTime(200)
    expect(fn).not.toHaveBeenCalled()
    vi.advanceTimersByTime(100)
    expect(fn).toHaveBeenCalledOnce()
  })
})
```

## deepClone

```typescript
// src/utils/deepClone.ts
export function deepClone<T>(obj: T): T {
  if (obj === null || typeof obj !== 'object') return obj
  if (obj instanceof Date) return new Date(obj.getTime()) as T
  if (obj instanceof RegExp) return new RegExp(obj.source, obj.flags) as T
  if (Array.isArray(obj)) return obj.map(item => deepClone(item)) as T
  const cloned = {} as T
  for (const key in obj) {
    if (Object.prototype.hasOwnProperty.call(obj, key)) cloned[key] = deepClone(obj[key])
  }
  return cloned
}
```

```typescript
describe('deepClone', () => {
  test('克隆简单对象', () => {
    const original = { name: '张三', age: 25 }
    const cloned = deepClone(original)
    expect(cloned).toEqual(original)
    expect(cloned).not.toBe(original)
  })

  test('克隆嵌套对象', () => {
    const original = { user: { address: { city: '北京' } } }
    const cloned = deepClone(original)
    expect(cloned.user.address).not.toBe(original.user.address)
  })

  test('修改克隆不影响原对象', () => {
    const original = { name: '张三', scores: [90, 85] }
    const cloned = deepClone(original)
    cloned.name = '李四'; cloned.scores.push(100)
    expect(original.name).toBe('张三')
    expect(original.scores).toEqual([90, 85])
  })

  test('克隆 Date', () => {
    const d = new Date(2024, 0, 15)
    const cloned = deepClone(d)
    expect(cloned.getTime()).toBe(d.getTime())
    expect(cloned).not.toBe(d)
  })
})
```

## 常见测试错误

**只测 Happy Path**：`formatDate` 必须测无效输入。`debounce` 必须测中途调用重置。

**忽略浮点数精度**：`expect(0.1 + 0.2).toBeCloseTo(0.3, 2)` 而不是 `toBe`。

### 练习一：为 capitalize 函数编写测试

```typescript
function capitalize(str: string): string {
  if (!str) return ''
  return str.charAt(0).toUpperCase() + str.slice(1)
}
```

覆盖：普通字符串、空字符串、已大写、单个字符、包含空格。

### 练习二：为 chunk 函数编写测试

```typescript
function chunk<T>(array: T[], size: number): T[][] {
  if (size <= 0) throw new Error('size must be greater than 0')
  const result: T[][] = []
  for (let i = 0; i < array.length; i += size) {
    result.push(array.slice(i, i + size))
  }
  return result
}
```

覆盖：正常分组、空数组、size 为 0、size 大于数组长度。

### 练习三：为 throttle 函数编写测试

```typescript
export function throttle<T extends (...args: any[]) => any>(
  fn: T, limit: number
): (...args: Parameters<T>) => void {
  let inThrottle = false
  let lastArgs: Parameters<T> | null = null
  return (...args) => {
    if (!inThrottle) {
      fn(...args); inThrottle = true
      setTimeout(() => {
        inThrottle = false
        if (lastArgs) { fn(...lastArgs); lastArgs = null }
      }, limit)
    } else { lastArgs = args }
  }
}
```

覆盖：立即执行第一次、限制期内只执行第一次、限制期结束后可再次调用、最后一次调用被延迟执行。

describe('throttle', () => {
  beforeEach(() => { vi.useFakeTimers() })
  afterEach(() => { vi.useRealTimers() })

  test('立即执行第一次', () => {
    const fn = vi.fn()
    throttle(fn, 300)()
    expect(fn).toHaveBeenCalledOnce()
  })

  test('限制期内只执行第一次', () => {
    const fn = vi.fn()
    const throttled = throttle(fn, 300)
    throttled('first'); throttled('second')
    expect(fn).toHaveBeenCalledWith('first')
    expect(fn).toHaveBeenCalledTimes(1)
  })

  test('限制期结束后可再次调用', () => {
    const fn = vi.fn()
    const throttled = throttle(fn, 300)
    throttled('first')
    vi.advanceTimersByTime(300)
    throttled('second')
    expect(fn).toHaveBeenCalledTimes(2)
  })

  test('限制期内最后一次调用被延迟执行', () => {
    const fn = vi.fn()
    const throttled = throttle(fn, 300)
    throttled('first'); throttled('delayed')
    vi.advanceTimersByTime(300)
    expect(fn).toHaveBeenCalledTimes(2)
    expect(fn).toHaveBeenLastCalledWith('delayed')
  })
})
