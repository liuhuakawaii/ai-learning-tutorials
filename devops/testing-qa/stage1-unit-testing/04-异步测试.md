# 异步测试

## 场景引入

你的代码中有大量异步操作——API 调用、数据库查询、定时器。这些代码是 bug 高发区，也是测试难点。写得不好，要么断言在异步完成前就执行了（假通过），要么测试挂起直到超时。这一课系统学习异步测试的各种模式。

## 学习目标

- 掌握测试 Promise 和 async/await 的正确方式
- 理解 callback 测试中的 done 参数
- 学会测试 setTimeout/setInterval
- 掌握异步错误处理的测试方法
- 识别和避免竞态条件

## 测试 Promise

### 返回 Promise 模式

```typescript
function fetchUser(id: number): Promise<{ id: number; name: string }> {
  return new Promise((resolve) => {
    setTimeout(() => resolve({ id, name: '张三' }), 100)
  })
}

describe('fetchUser', () => {
  // 关键：return Promise，让测试框架等待完成
  test('返回用户数据', () => {
    return fetchUser(1).then((user) => {
      expect(user.id).toBe(1)
      expect(user.name).toBe('张三')
    })
  })
})
```

不 return Promise，断言可能在 Promise 完成前就执行了：

```typescript
// 错误写法：测试假通过
test('错误示例', () => {
  fetchUser(1).then((user) => {
    expect(user.id).toBe(1) // 可能永远不会执行
  })
  // 测试在这里就结束了
})
```

### async/await 模式（推荐）

```typescript
describe('fetchUser', () => {
  test('返回用户数据', async () => {
    const user = await fetchUser(1)
    expect(user.id).toBe(1)
    expect(user.name).toBe('张三')
  })

  test('同时获取多个用户', async () => {
    const users = await Promise.all([fetchUser(1), fetchUser(2)])
    expect(users).toHaveLength(2)
  })
})
```

## 测试 callback

### done 参数模式

```typescript
function fetchDataCallback(
  url: string,
  callback: (error: Error | null, data?: any) => void
): void {
  setTimeout(() => {
    if (url.includes('error')) callback(new Error('失败'))
    else callback(null, { data: 'test' })
  }, 100)
}

describe('callback 测试', () => {
  test('成功获取数据', (done) => {
    fetchDataCallback('/api/data', (error, data) => {
      expect(error).toBeNull()
      expect(data).toEqual({ data: 'test' })
      done() // 通知测试完成
    })
  })
})
```

### 转换为 Promise（推荐）

```typescript
function fetchDataAsPromise(url: string): Promise<any> {
  return new Promise((resolve, reject) => {
    fetchDataCallback(url, (error, data) => {
      if (error) reject(error)
      else resolve(data)
    })
  })
}

describe('callback 转 Promise', () => {
  test('成功获取数据', async () => {
    const data = await fetchDataAsPromise('/api/data')
    expect(data).toEqual({ data: 'test' })
  })

  test('请求失败', async () => {
    await expect(fetchDataAsPromise('/api/error')).rejects.toThrow('失败')
  })
})
```

## 测试 setTimeout/setInterval

```typescript
import { describe, test, expect, vi, beforeEach, afterEach } from 'vitest'

describe('定时器测试', () => {
  beforeEach(() => { vi.useFakeTimers() })
  afterEach(() => { vi.useRealTimers() })

  test('setTimeout 在指定时间后执行', () => {
    const callback = vi.fn()
    setTimeout(callback, 1000)

    expect(callback).not.toHaveBeenCalled()
    vi.advanceTimersByTime(1000)
    expect(callback).toHaveBeenCalledOnce()
  })

  test('setInterval 每隔指定时间执行', () => {
    const callback = vi.fn()
    setInterval(callback, 500)

    vi.advanceTimersByTime(1500)
    expect(callback).toHaveBeenCalledTimes(3)
  })
})
```

### 测试防抖函数

```typescript
function debounce<T extends (...args: any[]) => any>(
  fn: T, delay: number
): (...args: Parameters<T>) => void {
  let timer: ReturnType<typeof setTimeout> | null = null
  return (...args) => {
    if (timer) clearTimeout(timer)
    timer = setTimeout(() => fn(...args), delay)
  }
}

describe('debounce', () => {
  beforeEach(() => { vi.useFakeTimers() })
  afterEach(() => { vi.useRealTimers() })

  test('延迟执行', () => {
    const fn = vi.fn()
    const debouncedFn = debounce(fn, 300)

    debouncedFn()
    expect(fn).not.toHaveBeenCalled()
    vi.advanceTimersByTime(300)
    expect(fn).toHaveBeenCalledOnce()
  })

  test('多次调用只执行最后一次', () => {
    const fn = vi.fn()
    const debouncedFn = debounce(fn, 300)

    debouncedFn('first')
    debouncedFn('second')
    debouncedFn('third')
    vi.advanceTimersByTime(300)

    expect(fn).toHaveBeenCalledOnce()
    expect(fn).toHaveBeenCalledWith('third')
  })
})
```

## 测试错误处理

### rejects 匹配器

```typescript
describe('错误处理', () => {
  test('Promise.reject 被正确捕获', async () => {
    await expect(Promise.reject(new Error('失败'))).rejects.toThrow('失败')
  })

  test('异步函数抛出错误', async () => {
    async function fail(): Promise<never> {
      throw new Error('异步错误')
    }
    await expect(fail()).rejects.toThrow('异步错误')
  })

  test('异步错误包含特定信息', async () => {
    async function validateAge(age: number): Promise<void> {
      return new Promise((resolve, reject) => {
        setTimeout(() => {
          if (age < 0) reject(new Error('年龄不能为负数'))
          else resolve()
        }, 100)
      })
    }

    await expect(validateAge(-1)).rejects.toThrow('年龄不能为负数')
  })
})
```

### try/catch 模式

```typescript
describe('try/catch', () => {
  test('捕获异步错误', async () => {
    try {
      await Promise.reject(new Error('失败'))
      expect(true).toBe(false) // 不应该到达这里
    } catch (error) {
      expect(error).toBeInstanceOf(Error)
      expect((error as Error).message).toBe('失败')
    }
  })
})
```

## 避免竞态条件

```typescript
// 问题：多个测试共享状态
let counter = 0
async function increment(): Promise<number> {
  await new Promise(r => setTimeout(r, 100))
  return ++counter
}

// 解决方案：隔离测试状态
describe('避免竞态', () => {
  beforeEach(() => { counter = 0 }) // 每个测试前重置

  test('第一个测试', async () => {
    expect(await increment()).toBe(1)
  })

  test('第二个测试', async () => {
    expect(await increment()).toBe(1) // 独立的
  })
})
```

## 完整示例：API 服务测试

```typescript
// src/services/userService.ts
export class UserService {
  constructor(private baseUrl: string) {}

  async getUserById(id: number): Promise<User> {
    const response = await fetch(`${this.baseUrl}/users/${id}`)
    if (!response.ok) throw new Error(`获取用户失败: ${response.status}`)
    return response.json()
  }
}
```

```typescript
// src/services/userService.test.ts
import { describe, test, expect, vi, beforeEach } from 'vitest'
import { UserService } from './userService'

const mockFetch = vi.fn()
vi.stubGlobal('fetch', mockFetch)

describe('UserService', () => {
  let service: UserService

  beforeEach(() => {
    service = new UserService('https://api.example.com')
    vi.clearAllMocks()
  })

  test('成功获取用户', async () => {
    const mockUser = { id: 1, name: '张三' }
    mockFetch.mockResolvedValue({
      ok: true,
      json: () => Promise.resolve(mockUser),
    })

    const user = await service.getUserById(1)
    expect(user).toEqual(mockUser)
    expect(mockFetch).toHaveBeenCalledWith('https://api.example.com/users/1')
  })

  test('用户不存在时抛出错误', async () => {
    mockFetch.mockResolvedValue({ ok: false, status: 404 })
    await expect(service.getUserById(999)).rejects.toThrow('获取用户失败: 404')
  })
})
```

## 常见误区

### 误区一：忘记处理异步

```typescript
// 错误：没有 await
test('异步', () => {
  const result = fetchUser(1) // result 是 Promise
  expect(result).toBeDefined() // Promise 当然是 defined
})

// 正确
test('异步', async () => {
  const result = await fetchUser(1)
  expect(result).toBeDefined()
})
```

### 误区二：忘记恢复定时器

```typescript
// 错误：影响后续测试
beforeEach(() => { vi.useFakeTimers() })

// 正确
afterEach(() => { vi.useRealTimers() })
```

## 工程建议

1. **优先使用 async/await**：比 .then() 更清晰
2. **每个异步测试都要有断言**：不要只调用不验证
3. **使用 fake timers**：避免真实等待
4. **隔离测试状态**：每个测试前重置共享变量
5. **测试错误路径**：异步错误处理同样重要

## 小结

- 测试 Promise 时使用 async/await 或 return Promise
- callback 测试使用 done 参数或转换为 Promise
- setTimeout/setInterval 使用 vi.useFakeTimers()
- 错误处理使用 rejects 匹配器或 try/catch
- 避免竞态条件：隔离测试状态

### 练习二

**答案**：
```typescript
beforeEach(() => { vi.useFakeTimers() })
afterEach(() => { vi.useRealTimers() })

test('只执行最后一次', () => {
  const fn = vi.fn()
  const debounced = debounce(fn, 300)

  debounced('last')
```
