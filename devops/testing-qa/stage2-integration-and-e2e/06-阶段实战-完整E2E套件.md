## Page Object 模式

```typescript
// e2e/pages/auth.page.ts
import { type Page, type Locator, expect } from '@playwright/test'

export class AuthPage {
  readonly email: Locator
  readonly password: Locator
  readonly name: Locator
  readonly submit: Locator
  readonly error: Locator

  constructor(private page: Page) {
    this.email = page.getByLabel('邮箱')
    this.password = page.getByLabel('密码')
    this.name = page.getByLabel('用户名')
    this.submit = page.getByRole('button', { name: /登录|注册/ })
    this.error = page.getByRole('alert')
  }

  async goto() { await this.page.goto('/login') }
  async register(email: string, password: string, name: string) {
    await this.page.getByRole('link', { name: /注册/ }).click()
    await this.name.fill(name)
    await this.email.fill(email)
    await this.password.fill(password)
    await this.submit.click()
  }
  async login(email: string, password: string) {
    await this.email.fill(email)
    await this.password.fill(password)
    await this.submit.click()
  }
  async expectError(msg: string | RegExp) { await expect(this.error).toContainText(msg) }
  async expectLoggedIn(name: string) {
    await expect(this.page.getByText(`欢迎, ${name}`)).toBeVisible()
  }
}
```

```typescript
// e2e/pages/todo.page.ts
import { type Page, type Locator, expect } from '@playwright/test'

export class TodoPage {
  readonly newTodo: Locator
  readonly addBtn: Locator
  readonly search: Locator

  constructor(private page: Page) {
    this.newTodo = page.getByPlaceholder('添加新任务..')
    this.addBtn = page.getByRole('button', { name: '添加' })
    this.search = page.getByPlaceholder('搜索任务...')
  }

  private item(t: string) { return this.page.getByRole('listitem').filter({ hasText: t }) }
  async goto() { await this.page.goto('/todos') }
  async addTodo(t: string) { await this.newTodo.fill(t); await this.addBtn.click() }
  async deleteTodo(t: string) {
    await this.item(t).getByRole('button', { name: '删除' }).click()
    await this.page.getByRole('button', { name: '确认删除' }).click()
  }
  async searchFor(kw: string) { await this.search.fill(kw) }
  async expectVisible(t: string) { await expect(this.item(t)).toBeVisible() }
  async expectNotVisible(t: string) { await expect(this.item(t)).not.toBeVisible() }
  async expectCount(n: number) { await expect(this.page.getByRole('listitem')).toHaveCount(n) }
}
```

## 自定义 Fixture

```typescript
// e2e/fixtures/index.ts
import { test as base } from '@playwright/test'
import { AuthPage } from '../pages/auth.page'
import { TodoPage } from '../pages/todo.page'
import { createTestUser } from '../utils/test-data'

export const test = base.extend<{ authPage: AuthPage; todoPage: TodoPage; loginAndGo: void }>({
  authPage: async ({ page }, use) => { await use(new AuthPage(page)) },
  todoPage: async ({ page }, use) => { await use(new TodoPage(page)) },
  loginAndGo: async ({ page }, use) => {
    const auth = new AuthPage(page)
    const user = createTestUser()
    await auth.goto()
    await auth.register(user.email, user.password, user.name)
    await auth.expectLoggedIn(user.name)
    await new TodoPage(page).goto()
    await use()
  },
})
export { expect } from '@playwright/test'
```

`loginAndGo` fixture 自动完成登录并跳转到 Todo 页，测试代码只需关注业务逻辑。

## 测试用例

```typescript
// e2e/tests/auth.spec.ts
import { test, expect } from '../fixtures'
import { createTestUser } from '../utils/test-data'

test.describe('用户认证', () => {
  test('注册后自动登录', async ({ authPage }) => {
    const u = createTestUser()
    await authPage.goto()
    await authPage.register(u.email, u.password, u.name)
    await authPage.expectLoggedIn(u.name)
  })

  test('错误密码登录失败', async ({ authPage }) => {
    const u = createTestUser()
    await authPage.goto()
    await authPage.register(u.email, u.password, u.name)
    await authPage.goto()
    await authPage.login(u.email, 'wrongpassword')
    await authPage.expectError(/密码错误/)
  })
})
```

### 练习二

问题：`waitForTimeout`（固定等待）和 `$$`（脆弱选择器）。

```typescript
test('搜索', async ({ todoPage }) => {
  await todoPage.goto()
  await todoPage.searchFor('买')
  await todoPage.expectVisible('买牛奶')
  await todoPage.expectCount(2)
})
```

去掉 `waitForTimeout` 改用自动等待，去掉 CSS 选择器改用 Page Object。
