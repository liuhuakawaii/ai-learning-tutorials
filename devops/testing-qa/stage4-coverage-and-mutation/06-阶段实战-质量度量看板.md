# 阶段实战：质量度量看板

## 目标

整合覆盖率、变异测试、基准测试三类数据，搭建统一的质量度量看板，一眼看到质量趋势。

## 数据收集

```typescript
// scripts/collect-metrics.ts
import { readFileSync, writeFileSync, existsSync, mkdirSync } from 'fs'
import { join } from 'path'

interface MetricsEntry {
  timestamp: string
  commit: string
  branch: string
  coverage: { statements: number; branches: number; functions: number; lines: number }
  mutationScore: number
  mutationDetails: { killed: number; survived: number; total: number }
  benchmarks: Record<string, { hz: number; mean: number; p99: number }>
}

function collectCoverage(): MetricsEntry['coverage'] {
  const path = join(process.cwd(), 'coverage', 'coverage-summary.json')
  if (!existsSync(path)) return { statements: 0, branches: 0, functions: 0, lines: 0 }
  const data = JSON.parse(readFileSync(path, 'utf-8'))
  return {
    statements: data.total.statements.pct,
    branches: data.total.branches.pct,
    functions: data.total.functions.pct,
    lines: data.total.lines.pct,
  }
}

function collectMutation() {
  const path = join(process.cwd(), 'reports', 'mutation.json')
  if (!existsSync(path)) return { killed: 0, survived: 0, total: 0, score: 0 }
  const data = JSON.parse(readFileSync(path, 'utf-8'))
  let killed = 0, survived = 0
  for (const file of Object.values(data.files) as any[]) {
    killed += file.killed; survived += file.survived
  }
  const total = killed + survived
  return { killed, survived, total, score: total > 0 ? Math.round((killed / total) * 100) : 0 }
}

function main() {
  const dir = join(process.cwd(), 'metrics')
  if (!existsSync(dir)) mkdirSync(dir, { recursive: true })
  const historyPath = join(dir, 'history.json')
  const history: MetricsEntry[] = existsSync(historyPath)
    ? JSON.parse(readFileSync(historyPath, 'utf-8'))
    : []

  const mutation = collectMutation()
  const entry: MetricsEntry = {
    timestamp: new Date().toISOString(),
    commit: process.env.GITHUB_SHA?.slice(0, 7) || 'local',
    branch: process.env.GITHUB_REF_NAME || 'local',
    coverage: collectCoverage(),
    mutationScore: mutation.score,
    mutationDetails: { killed: mutation.killed, survived: mutation.survived, total: mutation.total },
    benchmarks: {},
  }
  history.push(entry)
  writeFileSync(historyPath, JSON.stringify(history.slice(-90), null, 2))
  console.log(`📊 覆盖率: ${entry.coverage.statements}% | 变异: ${entry.mutationScore}%`)
}
main()
```

数据只保留最近 90 条，避免膨胀。

## 生成 HTML 看板

```typescript
// scripts/generate-dashboard.ts
import { readFileSync, writeFileSync, existsSync, mkdirSync } from 'fs'
import { join } from 'path'

function generateHTML(history: any[]): string {
  const latest = history[history.length - 1]
  const labels = history.map(h => h.timestamp.slice(0, 10))
  const stmtData = history.map(h => h.coverage.statements)
  const branchData = history.map(h => h.coverage.branches)
  const mutationData = history.map(h => h.mutationScore)

  return `<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <title>质量度量看板</title>
  <script src="https://cdn.jsdelivr.net/npm/chart.js@4"></script>
  <style>
    body{font-family:-apple-system,sans-serif;background:#0f172a;color:#e2e8f0;padding:24px}
    .grid{display:grid;grid-template-columns:repeat(auto-fit,minmax(220px,1fr));gap:16px;margin-bottom:24px}
    .card{background:#1e293b;border-radius:12px;padding:20px;border:1px solid #334155}
    .metric{font-size:2rem;font-weight:700}
    .good{color:#22c55e}.warn{color:#f59e0b}.bad{color:#ef4444}
    .box{background:#1e293b;border-radius:12px;padding:20px;border:1px solid #334155;margin-bottom:16px}
    canvas{max-height:280px}
  </style>
</head>
<body>
  <h1>质量度量看板</h1>
  <div class="grid">
    <div class="card"><h2>语句覆盖率</h2><div class="metric ${latest?.coverage.statements>=80?'good':latest?.coverage.statements>=60?'warn':'bad'}">${latest?.coverage.statements??'-'}%</div></div>
    <div class="card"><h2>分支覆盖率</h2><div class="metric ${latest?.coverage.branches>=75?'good':latest?.coverage.branches>=50?'warn':'bad'}">${latest?.coverage.branches??'-'}%</div></div>
    <div class="card"><h2>变异分数</h2><div class="metric ${latest?.mutationScore>=85?'good':latest?.mutationScore>=70?'warn':'bad'}">${latest?.mutationScore??'-'}%</div></div>
  </div>
  <div class="box"><h2>覆盖率趋势</h2><canvas id="covChart"></canvas></div>
  <div class="box"><h2>变异分数趋势</h2><canvas id="mutChart"></canvas></div>
  <script>
    const L=${JSON.stringify(labels)};
    const O={responsive:true,scales:{y:{min:0,max:100}}};
    new Chart('covChart',{type:'line',data:{labels:L,datasets:[
      {label:'语句',data:${JSON.stringify(stmtData)},borderColor:'#22c55e',tension:.3},
      {label:'分支',data:${JSON.stringify(branchData)},borderColor:'#3b82f6',tension:.3}
    ]},options:O});
    new Chart('mutChart',{type:'line',data:{labels:L,datasets:[
      {label:'变异分数',data:${JSON.stringify(mutationData)},borderColor:'#f59e0b',tension:.3}
    ]},options:O});
  </script>
</body></html>`
}

function main() {
  const p = join(process.cwd(), 'metrics', 'history.json')
  if (!existsSync(p)) { console.error('❌ 未找到 metrics/history.json'); process.exit(1) }
  const history = JSON.parse(readFileSync(p, 'utf-8'))
  const dir = join(process.cwd(), 'dashboard')
  if (!existsSync(dir)) mkdirSync(dir, { recursive: true })
  writeFileSync(join(dir, 'index.html'), generateHTML(history))
  console.log('✅ 看板已生成: dashboard/index.html')
}
main()
```

## CI 集成

```yaml
# .github/workflows/quality.yml
name: Quality Dashboard
on:
  push: { branches: [main] }

jobs:
  quality:
    runs-on: ubuntu-latest
    permissions: { contents: write, pages: write }
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20 }
      - run: npm ci
      - run: npx vitest run --coverage
        continue-on-error: true
      - run: npx tsx scripts/collect-metrics.ts
      - run: npx tsx scripts/generate-dashboard.ts
      - if: github.ref == 'refs/heads/main'
        uses: peaceiris/actions-gh-pages@v3
        with:
          github_token: ${{ secrets.GITHUB_TOKEN }}
          publish_dir: ./dashboard
```

### 练习一

```typescript
function collectTestStats() {
  const path = join(process.cwd(), 'test-results.json')
  if (!existsSync(path)) return { total: 0, passed: 0, failed: 0 }
  const data = JSON.parse(readFileSync(path, 'utf-8'))
  const tests: Array<{ status: string }> = []
  function extract(suite: any) {
    for (const t of suite.tests || []) tests.push({ status: t.result?.status || 'unknown' })
    for (const c of suite.suites || []) extract(c)
  }
  for (const s of data.testResults || []) extract(s)
  return {
    total: tests.length,
    passed: tests.filter(t => t.status === 'passed').length,
    failed: tests.filter(t => t.status === 'failed').length,
  }
}
```

### 练习二

```yaml
      - name: PR quality summary
        if: github.event_name == 'pull_request'
        uses: actions/github-script@v7
        with:
          script: |
            const fs = require('fs')
            const h = JSON.parse(fs.readFileSync('./metrics/history.json','utf-8'))
            const l = h[h.length-1], p = h.length>=2?h[h.length-2]:null
            function fmt(cur,base){
              if(!base) return cur+'%'
              const d=cur-base; return `${base}% → ${cur}% (${d>0?'+':''}${d.toFixed(1)}%)`
            }
            const body=['📊 **质量度量变化**',
              `- 语句覆盖率: ${fmt(l.coverage.statements,p?.coverage.statements)}`,
              `- 变异分数: ${fmt(l.mutationScore,p?.mutationScore)}`
            ].join('\n')
            await github.rest.issues.createComment({
              owner:context.repo.owner, repo:context.repo.repo,
              issue_number:context.issue.number, body
            })
```
