## Step 1：部署 Prometheus Stack

```bash
# 添加 Helm 仓库
helm repo add prometheus-community https://prometheus-community.github.io/helm-charts
helm repo update

# 创建 monitoring 命名空间
kubectl create namespace monitoring

# 安装 kube-prometheus-stack
helm install prometheus prometheus-community/kube-prometheus-stack \
  --namespace monitoring \
  --set prometheus.prometheusSpec.retention=15d \
  --set grafana.adminPassword=admin123 \
  --set grafana.service.type=NodePort \
  --set grafana.service.nodePort=30300
```

### 验证安装

```bash
kubectl get pods -n monitoring
# 应该看到：
# prometheus-kube-prometheus-operator
# prometheus-kube-prometheus-prometheus
# prometheus-grafana
# alertmanager-main
# kube-state-metrics
# node-exporter

# 访问 Grafana
kubectl port-forward svc/prometheus-grafana -n monitoring 3000:80
# 或通过 NodePort 访问 http://localhost:30300
```

## Step 2：配置自定义告警规则

```yaml
# custom-alerts.yaml
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: custom-alerts
  namespace: monitoring
  labels:
    release: prometheus
spec:
  groups:
    - name: pod-alerts
      rules:
        - alert: PodCrashLooping
          expr: |
            increase(kube_pod_container_status_restarts_total[1h]) > 5
          for: 10m
          labels:
            severity: warning
          annotations:
            summary: "Pod {{ $labels.pod }} 频繁重启"
            description: "过去 1 小时重启超过 5 次"

        - alert: PodNotReady
          expr: |
            kube_pod_status_ready{condition="false"} == 1
          for: 15m
          labels:
            severity: warning
          annotations:
            summary: "Pod {{ $labels.pod }} 长时间未就绪"

        - alert: ContainerOOMKilled
          expr: |
            kube_pod_container_status_last_terminated_reason{reason="OOMKilled"} == 1
          for: 5m
          labels:
            severity: warning
          annotations:
            summary: "容器 {{ $labels.container }} 被 OOM Kill"

    - name: node-alerts
      rules:
        - alert: NodeHighCPU
          expr: |
            100 - (avg by (instance) (rate(node_cpu_seconds_total{mode="idle"}[5m])) * 100) > 85
          for: 10m
          labels:
            severity: warning
          annotations:
            summary: "Node {{ $labels.instance }} CPU 使用率超过 85%"

        - alert: NodeHighMemory
          expr: |
            (1 - node_memory_MemAvailable_bytes / node_memory_MemTotal_bytes) * 100 > 85
          for: 10m
          labels:
            severity: warning
          annotations:
            summary: "Node {{ $labels.instance }} 内存使用率超过 85%"

        - alert: NodeDiskSpaceLow
          expr: |
            (node_filesystem_avail_bytes{mountpoint="/"} / node_filesystem_size_bytes{mountpoint="/"}) < 0.1
          for: 10m
          labels:
            severity: critical
          annotations:
            summary: "Node {{ $labels.instance }} 磁盘空间不足 10%"
```

## Step 3：配置 Alertmanager

```yaml
# alertmanager-config.yaml
apiVersion: v1
kind: Secret
metadata:
  name: alertmanager-config
  namespace: monitoring
type: Opaque
stringData:
  alertmanager.yaml: |
    global:
      resolve_timeout: 5m

    route:
      group_by: ['alertname', 'namespace']
      group_wait: 30s
      group_interval: 5m
      repeat_interval: 4h
      receiver: 'default'
      routes:
        - match:
            severity: critical
          receiver: 'critical'
          repeat_interval: 1h

    receivers:
      - name: 'default'
        webhook_configs:
          - url: 'http://notification-service:8080/webhook'
            send_resolved: true

      - name: 'critical'
        webhook_configs:
          - url: 'http://notification-service:8080/webhook/critical'
            send_resolved: true
```

## Step 4：部署 Loki 日志系统

```bash

  --namespace monitoring \
  --set grafana.enabled=false \
  --set promtail.enabled=true
```

```yaml
# loki-datasource.yaml

kind: ConfigMap
metadata:
  name: loki-datasource
  namespace: monitoring
  labels:
    grafana_datasource: "1"
data:
  loki-datasource.yaml: |
    apiVersion: 1
    datasources:
      - name: Loki
        type: loki
        access: proxy
        url: http://loki:3100
        isDefault: false
```

## Step 6：安全加固

```yaml
# 为 monitoring 命名空间配置 NetworkPolicy
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: monitoring-policy
  namespace: monitoring
spec:
  podSelector: {}
  policyTypes:
    - Ingress
    - Egress
  ingress:
    - from:
        - namespaceSelector:
            matchLabels:
              name: monitoring
    - from:
        - namespaceSelector:
            matchLabels:
              name: kube-system
  egress:
    - {}    # 允许所有出站（Prometheus 需要抓取所有命名空间的指标）
```

## 常见误区

**误区一："监控部署好就不管了"**

监控体系需要持续维护：更新 Dashboard、调整告警阈值、清理过期数据。

**误区二："告警越多越安全"**

过多的告警会导致告警疲劳，真正的问题被淹没。只告警需要人工介入的情况。

## 工程建议

1. **告警分级**：Critical（立即处理）、Warning（尽快处理）、Info（记录）
2. **告警收敛**：相同类型的告警合并通知
3. **定期审查告警规则**：删除无效告警，调整阈值
4. **Dashboard 分层**：概览 → 服务 → 实例
5. **日志保留策略**：热数据 7 天，冷数据 30 天

### 练习二

**答案**：

```bash
# 1. 创建会 OOM Kill 的 Pod
kubectl run oom-test --image=nginx --limits='memory=10Mi' \
  --command -- stress --vm 1 --vm-bytes 50M --vm-hang 0

# 2. 观察 Pod 状态
kubectl get pod oom-test -w
# 会看到 OOMKilled

# 3. 在 Prometheus 中查询
# kube_pod_container_status_last_terminated_reason{reason="OOMKilled"}

# 4. 在 Alertmanager 中查看告警
kubectl port-forward svc/alertmanager-main -n monitoring 9093:9093

# 5. 清理
kubectl delete pod oom-test
# 告警会在 for 时间后自动恢复
```

**要点**：
- 告警有 `for` 时间，不会立即触发
- 问题修复后告警自动恢复
- Alertmanager 的路由和接收器需要正确配置
