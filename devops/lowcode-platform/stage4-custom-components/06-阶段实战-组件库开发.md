## 组件开发示例：StatCard

### manifest.json

```json
{
  "name": "stat-card",
  "version": "1.0.0",
  "displayName": "统计卡片",
  "category": "数据展示",
  "icon": "dashboard",
  "props": {
    "title": { "type": "string", "label": "标题", "required": true },
    "value": { "type": "number", "label": "数值", "required": true },
    "prefix": { "type": "string", "label": "前缀", "default": "" },
    "suffix": { "type": "string", "label": "后缀", "default": "" },
    "precision": { "type": "number", "label": "小数位数", "default": 0 },
    "trend": {
      "type": "select",
      "options": ["up", "down", "flat"],
      "label": "趋势"
    },
    "trendValue": { "type": "string", "label": "趋势值" },
    "color": { "type": "color", "label": "颜色", "default": "#1890ff" }
  },
  "events": ["onClick"],
  "dataBinding": { "supported": true, "fields": ["value", "trend", "trendValue"] }
}
```

### 组件实现

```tsx
import React from 'react';
import { formatNumber } from '@mylib/core';
import './styles.css';

interface StatCardProps {
  title: string;
  value: number;
  prefix?: string;
  suffix?: string;
  precision?: number;
  trend?: 'up' | 'down' | 'flat';
  trendValue?: string;
  color?: string;
  onClick?: () => void;
}

export const StatCard: React.FC<StatCardProps> = ({
  title, value, prefix = '', suffix = '', precision = 0,
  trend, trendValue, color = '#1890ff', onClick,
}) => {
  const formattedValue = formatNumber(value, precision);

  return (
    <div className="stat-card" onClick={onClick} style={{ borderTopColor: color }}>
      <div className="stat-card__title">{title}</div>
      <div className="stat-card__value">
        {prefix && <span className="stat-card__prefix">{prefix}</span>}
        {formattedValue}
        {suffix && <span className="stat-card__suffix">{suffix}</span>}
      </div>
      {trend && trendValue && (
        <div className={`stat-card__trend stat-card__trend--${trend}`}>
          {trend === 'up' ? '↑' : trend === 'down' ? '↓' : '→'} {trendValue}
        </div>
      )}
    </div>
  );
};
```

### 练习二

```tsx
import { render, fireEvent } from '@testing-library/react';
import { StatCard } from './index';

describe('StatCard', () => {
  it('renders title and value', () => {
    const { getByText } = render(<StatCard title="GMV" value={1234567} />);
    expect(getByText('GMV')).toBeTruthy();
    expect(getByText('1234567')).toBeTruthy();
  });

  it('formats number with precision', () => {
    const { getByText } = render(<StatCard title="Rate" value={3.456} precision={2} />);
    expect(getByText('3.46')).toBeTruthy();
  });

  it('shows trend icon', () => {
    const { getByText } = render(<StatCard title="T" value={1} trend="up" trendValue="+10%" />);
    expect(getByText('↑ +10%')).toBeTruthy();
  });

  it('calls onClick', () => {
    const onClick = jest.fn();
    const { container } = render(<StatCard title="T" value={1} onClick={onClick} />);
    fireEvent.click(container.firstChild!);
    expect(onClick).toHaveBeenCalled();
  });
});
```

### 练习三

递增 minor 版本（1.0.0 → 1.1.0）。

理由：添加 `size` prop 是向后兼容的新功能——不设置 size 时行为和之前一样，所以不是 breaking change（major）。不是 bug 修复（patch），所以用 minor。
