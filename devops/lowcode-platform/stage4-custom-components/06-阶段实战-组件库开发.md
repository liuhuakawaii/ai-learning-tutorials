经过前面五节课的学习，你已掌握组件开发规范、图表集成、编辑器封装、第三方对接和组件市场管理。现在把知识综合起来，完成一个真正可用的组件库。

## 组件一：统计卡片（StatCard）

```typescript
import React from 'react';

interface StatCardProps {
  title: string; value: number | string;
  prefix?: string; suffix?: string;
  trend?: 'up' | 'down' | 'flat'; trendValue?: string;
  loading?: boolean; onClick?: () => void;
}

const trendColors = { up: '#52c41a', down: '#ff4d4f', flat: '#d9d9d9' };
const trendIcons = { up: '↑', down: '↓', flat: '→' };

const StatCard: React.FC<StatCardProps> = (rawProps) => {
  const props = { prefix: '', suffix: '', ...rawProps };
  if (props.loading) return <div className="stat-card stat-card--loading"><div className="skeleton" /></div>;

  return (
    <div className="stat-card" onClick={props.onClick} style={{ cursor: props.onClick ? 'pointer' : 'default' }}>
      <div className="stat-card__header"><span className="title">{props.title}</span></div>
      <div className="stat-card__value">
        {props.prefix && <span className="prefix">{props.prefix}</span>}
        <span className="number">{props.value}</span>
        {props.suffix && <span className="suffix">{props.suffix}</span>}
      </div>
      {props.trend && (
        <div className="stat-card__trend" style={{ color: trendColors[props.trend] }}>
          <span>{trendIcons[props.trend]}</span><span>{props.trendValue}</span>
        </div>
      )}
    </div>
  );
};
```

## 组件二：数据表格（DataTable）

```typescript
import React, { useState, useMemo } from 'react';

interface Column { key: string; title: string; width?: number; sortable?: boolean; render?: (value: unknown, row: Record<string, unknown>) => React.ReactNode; }
interface DataTableProps { columns: Column[]; data: Record<string, unknown>[]; pageSize?: number; rowKey?: string; onRowClick?: (row: Record<string, unknown>) => void; }

const DataTable: React.FC<DataTableProps> = (rawProps) => {
  const props = { pageSize: 10, rowKey: 'id', ...rawProps };
  const [page, setPage] = useState(1);
  const [sortKey, setSortKey] = useState<string | null>(null);
  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc');

  const sorted = useMemo(() => {
    if (!sortKey) return props.data;
    return [...props.data].sort((a, b) => {
      const cmp = String(a[sortKey]).localeCompare(String(b[sortKey]), 'zh-CN', { numeric: true });
      return sortDir === 'asc' ? cmp : -cmp;
    });
  }, [props.data, sortKey, sortDir]);

  const totalPages = Math.ceil(sorted.length / props.pageSize);
  const pageData = sorted.slice((page - 1) * props.pageSize, page * props.pageSize);

  return (
    <div className="data-table">
      <table>
        <thead><tr>{props.columns.map(col => (
          <th key={col.key} style={{ width: col.width }} onClick={() => { if (col.sortable) { setSortKey(col.key); setSortDir(d => d === 'asc' ? 'desc' : 'asc'); } }}>
            {col.title}{col.sortable && sortKey === col.key && (sortDir === 'asc' ? ' ↑' : ' ↓')}
          </th>
        ))}</tr></thead>
        <tbody>{pageData.map((row, i) => (
          <tr key={String(row[props.rowKey] ?? i)} onClick={() => props.onRowClick?.(row)}>
            {props.columns.map(col => <td key={col.key}>{col.render ? col.render(row[col.key], row) : String(row[col.key] ?? '')}</td>)}
          </tr>
        ))}</tbody>
      </table>
      {totalPages > 1 && <div className="pagination">
        <button disabled={page <= 1} onClick={() => setPage(p => p - 1)}>上一页</button>
        <span>{page}/{totalPages}</span>
        <button disabled={page >= totalPages} onClick={() => setPage(p => p + 1)}>下一页</button>
      </div>}
    </div>
  );
};
```

## 组件三：搜索栏（SearchBar）

```typescript
import React, { useState, useCallback } from 'react';

interface SearchBarProps {
  placeholder?: string; onSearch: (keyword: string) => void; debounceMs?: number;
  filters?: { key: string; label: string; options: { value: string; label: string }[] }[];
  onFilterChange?: (key: string, value: string) => void;
}

function debounce<T extends (...args: any[]) => void>(fn: T, ms: number): T {
  let timer: ReturnType<typeof setTimeout>;
  return ((...args: any[]) => { clearTimeout(timer); timer = setTimeout(() => fn(...args), ms); }) as unknown as T;
}

const SearchBar: React.FC<SearchBarProps> = (rawProps) => {
  const props = { placeholder: '搜索...', debounceMs: 300, ...rawProps };
  const [keyword, setKeyword] = useState('');
  const [activeFilters, setActiveFilters] = useState<Record<string, string>>({});
  const debouncedSearch = useCallback(debounce((kw: string) => props.onSearch(kw), props.debounceMs), [props.onSearch]);

  return (
    <div className="search-bar">
      <input value={keyword} onChange={e => { setKeyword(e.target.value); debouncedSearch(e.target.value); }} placeholder={props.placeholder} />
      {keyword && <button onClick={() => { setKeyword(''); props.onSearch(''); }}>清除</button>}
      {props.filters?.map(f => (
        <select key={f.key} value={activeFilters[f.key] || ''} onChange={e => { setActiveFilters(p => ({ ...p, [f.key]: e.target.value })); props.onFilterChange?.(f.key, e.target.value); }}>
          <option value="">{f.label}</option>
          {f.options.map(o => <option key={o.value} value={o.value}>{o.label}</option>)}
        </select>
      ))}
    </div>
  );
};
```

## 组件四到十：快速概览

**筛选面板（FilterPanel）**：接收 `FilterField[]`（select/date/number/text），输出 `Record<string, unknown>` 筛选条件，支持水平/垂直布局和折叠。

**操作栏（ActionBar）**：接收 `ActionItem[]`（primary/default/danger/link 类型），支持危险操作二次确认，展示已选中数量。

**时间线（Timeline）**：接收 `TimelineItem[]`（title/timestamp/status/icon），支持 left/right/alternate 模式和 pending 状态。

**看板卡片（KanbanCard）**：展示 title/description/priority/assignee/tags/dueDate，优先级用颜色区分。

**通知铃铛（NotificationBell）**：展示未读数量角标，下拉列表支持单条已读和全部已读，按时间倒序。

**用户头像（UserAvatar）**：支持图片/文字回退（取姓名首字母），small/medium/large 尺寸，在线状态圆点。

**状态徽章（StatusBadge）**：预设 success/warning/error/processing/default 配色，支持 dot 模式和 pulse 动画。

## 组件文档站点

类 Storybook 的在线预览：

```typescript
interface StoryConfig {
  name: string;
  component: React.FC<any>;
  defaultProps: Record<string, unknown>;
  propControls: Record<string, { type: string; options?: string[] }>;
}

const ComponentPreview: React.FC<{ story: StoryConfig }> = ({ story }) => {
  const [props, setProps] = useState(story.defaultProps);
  return (
    <div className="component-preview">
      <h2>{story.name}</h2>
      <div className="preview-area"><story.component {...props} /></div>
      <div className="controls">
        {Object.entries(story.propControls).map(([key, cfg]) => (
          <div key={key} className="control-row">
            <label>{key}</label>
            {cfg.type === 'boolean'
              ? <input type="checkbox" checked={!!props[key]} onChange={e => setProps(p => ({ ...p, [key]: e.target.checked }))} />
              : cfg.type === 'select'
              ? <select value={String(props[key])} onChange={e => setProps(p => ({ ...p, [key]: e.target.value }))}>{cfg.options?.map(o => <option key={o} value={o}>{o}</option>)}</select>
              : <input type="text" value={String(props[key] ?? '')} onChange={e => setProps(p => ({ ...p, [key]: e.target.value }))} />}
          </div>
        ))}
      </div>
    </div>
  );
};
```

## CI/CD 发布流程

```yaml
# .github/workflows/publish.yml
name: Publish Components
on:
  push:
    branches: [main]
    paths: ['packages/**']

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 18, registry-url: 'https://npm.mycompany.com' }
      - run: npm ci
      - run: npm run test --workspaces
      - run: npm run build --workspaces

  publish:
    needs: test
    if: contains(github.event.head_commit.message, 'release:')
    steps:
      - uses: actions/checkout@v4
      - run: npm ci && npm run build --workspaces
      - run: npx lerna publish from-package --yes
        env: { NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }} }
```

## 常见误区

1. **组件库不做类型导出**：.d.ts 没正确导出，使用者拿不到 Props 类型提示。
2. **所有组件打进一个包**：使用者只需 StatCard 却被迫安装整个库（含 ECharts、D3），打包体积膨胀。
3. **文档只有代码没有预览**：只有 API 说明没有在线预览，配置者无法直观了解效果。

## 工程建议

1. **用 Turborepo 加速构建**：缓存机制跳过未修改组件，构建时间从 2 分钟缩到 20 秒。
2. **Testing Library 写测试**：从用户视角写测试（「点击按钮后显示文本」），更健壮、重构更安全。
3. **质量门禁**：新组件必须通过 lint、覆盖率 > 80%、构建成功才能合并。

## 小结

- Monorepo 是组织组件库最佳实践，每个组件独立发包，共享构建工具。
- 10 个核心组件覆盖低代码平台 80% 常见 UI 场景。
- 组件文档站点让配置者在线预览和调试。
- CI/CD 自动化测试、构建和发布确保质量。

### 练习一

**思路**：预设状态映射颜色，dot 模式用 CSS 控制，pulse 用 CSS animation。

**答案**：
```typescript
import React from 'react';

interface StatusBadgeProps { status: string; text?: string; color?: string; dot?: boolean; pulse?: boolean; }

const presets: Record<string, string> = { success: '#52c41a', warning: '#faad14', error: '#ff4d4f', processing: '#1890ff', default: '#d9d9d9' };

const StatusBadge: React.FC<StatusBadgeProps> = (rawProps) => {
  const props = { text: rawProps.status, ...rawProps };
  const color = props.color || presets[props.status] || presets.default;
  const isPulsing = props.status === 'processing' || props.pulse;

  return (
    <span style={{
      display: 'inline-flex', alignItems: 'center', gap: 6, padding: '2px 8px', borderRadius: 4,
      fontSize: 12, color, backgroundColor: `${color}15`, border: `1px solid ${color}30`,
    }}>
      <span className={isPulsing ? 'dot--pulse' : ''} style={{
        width: 6, height: 6, borderRadius: '50%', backgroundColor: color, flexShrink: 0,
      }} />
      {!props.dot && <span>{props.text}</span>}
    </span>
  );
};
```

```css
.dot--pulse { animation: pulse 1.5s ease-in-out infinite; }
@keyframes pulse {
  0% { box-shadow: 0 0 0 0 currentColor; opacity: 1; }
  70% { box-shadow: 0 0 0 6px currentColor; opacity: 0.5; }
  100% { box-shadow: 0 0 0 0 currentColor; opacity: 1; }
}
```

**要点**：`${color}15` 做半透明背景免额外计算；pulse 用 currentColor 继承颜色。

### 练习二

**思路**：fs.readdir 遍历 packages，git diff 获取变更包，Promise.all 控制并发。

**答案**：
```typescript
import { execSync } from 'child_process';
import fs from 'fs';
import path from 'path';

interface BuildResult { name: string; status: 'success' | 'failed' | 'skipped'; duration: number; error?: string; }

async function buildAll(packagesDir: string, maxConcurrency = 4): Promise<BuildResult[]> {
  const allPkgs = fs.readdirSync(packagesDir).filter(d => fs.existsSync(path.join(packagesDir, d, 'package.json')));

  let changed = new Set(allPkgs);
  try {
    const diff = execSync('git diff --name-only HEAD~1', { encoding: 'utf-8' });
    changed = new Set(allPkgs.filter(p => diff.split('\n').some(f => f.startsWith(`packages/${p}/`))));
  } catch { /* git diff 失败全量构建 */ }

  const results: BuildResult[] = [];
  const queue = [...allPkgs];

  async function buildOne(pkg: string): Promise<BuildResult> {
    if (!changed.has(pkg)) return { name: pkg, status: 'skipped', duration: 0 };
    const start = Date.now();
    try {
      execSync('npm run build', { cwd: path.join(packagesDir, pkg), stdio: 'pipe', timeout: 60000 });
      return { name: pkg, status: 'success', duration: Date.now() - start };
    } catch (err: any) { return { name: pkg, status: 'failed', duration: Date.now() - start, error: err.message }; }
  }

  while (queue.length) {
    const batch = queue.splice(0, maxConcurrency);
    results.push(...await Promise.all(batch.map(buildOne)));
  }

  const s = results.filter(r => r.status === 'success').length;
  const f = results.filter(r => r.status === 'failed').length;
  const k = results.filter(r => r.status === 'skipped').length;
  console.log(`构建完成: ${s} 成功, ${f} 失败, ${k} 跳过`);
  if (f > 0) results.filter(r => r.status === 'failed').forEach(r => console.error(`  ✗ ${r.name}: ${r.error}`));
  return results;
}
```

**要点**：git diff 失败降级全量构建保 CI 兼容；Promise.all 控制并发防内存溢出；execSync 60 秒超时防卡住。
