# 多租户架构

## 场景引入

某低代码平台经过一年发展，已经服务了 50 多家企业客户。每家客户都要求自己的数据完全隔离，不能被其他企业看到。目前采用的是"一个客户一套部署"的模式，运维团队需要维护 50 多套独立的环境，每次升级都要逐一操作，人力成本巨大。更糟糕的是，服务器资源利用率极低——很多企业只有少量用户，却占用一整套独立的数据库和服务器。

多租户架构的核心目标是：在共享基础设施的同时，实现租户之间的数据隔离和资源隔离。本课将讲解三种主流的多租户模型，以及租户识别、行级安全、配额管理等关键实现。

## 学习目标

- 理解三种多租户模型（共享库共享表/共享库独立表/独立库）的优劣与适用场景
- 掌握租户识别机制（子域名/Header/JWT）和租户上下文传递
- 学会使用 PostgreSQL Row Level Security 实现行级数据隔离
- 理解资源配额管理和计量计费的基本实现

## 多租户模型对比

| 模式 | 隔离级别 | 成本 | 复杂度 | 适用场景 |
|------|----------|------|--------|----------|
| 共享库共享表 | 逻辑隔离 | 低 | 中 | 中小客户、SaaS 标准版 |
| 共享库独立表 | Schema 隔离 | 中 | 高 | 中型企业、定制化需求 |
| 独立库 | 物理隔离 | 高 | 低 | 大型企业、合规要求高 |

### 模式一：共享库共享表（推荐起步方案）

所有租户的数据存在同一张表中，通过 `tenant_id` 字段区分：

```sql
-- 所有业务表都包含 tenant_id
CREATE TABLE forms (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id   VARCHAR(64) NOT NULL,
    name        VARCHAR(255) NOT NULL,
    schema_json JSONB NOT NULL,
    created_by  VARCHAR(64) NOT NULL,
    created_at  TIMESTAMPTZ DEFAULT NOW(),
    updated_at  TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE form_data (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id   VARCHAR(64) NOT NULL,
    form_id     UUID NOT NULL REFERENCES forms(id),
    data_json   JSONB NOT NULL,
    created_by  VARCHAR(64) NOT NULL,
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- 关键索引：tenant_id 必须在所有查询条件中
CREATE INDEX idx_forms_tenant ON forms(tenant_id, created_at DESC);
CREATE INDEX idx_form_data_tenant_form ON form_data(tenant_id, form_id, created_at DESC);
```

### 模式二：独立 Schema

每个租户拥有独立的 Schema，表结构相同但数据物理隔离：

```sql
-- 创建租户 Schema
CREATE SCHEMA tenant_acme;
CREATE SCHEMA tenant_globex;

-- 在 Schema 下创建相同的表结构
CREATE TABLE tenant_acme.forms (LIKE public.forms_template INCLUDING ALL);
CREATE TABLE tenant_globex.forms (LIKE public.forms_template INCLUDING ALL);

-- 切换 Schema 查询（应用层动态设置）
SET search_path TO tenant_acme, public;
SELECT * FROM forms; -- 只看到 acme 的数据
```

## 租户识别机制

### 子域名识别

```typescript
// src/middleware/tenant.ts
import { Request, Response, NextFunction } from 'express';
import { pool } from '../db';

// 从子域名提取租户标识：acme.lowcode.example.com → acme
function extractTenantSlug(host: string): string | null {
  const parts = host.split('.');
  if (parts.length >= 3) {
    return parts[0];
  }
  return null;
}

export async function tenantMiddleware(
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> {
  // 优先从 JWT 中获取租户信息
  const jwtTenant = (req as any).user?.tenantId;
  if (jwtTenant) {
    (req as any).tenantId = jwtTenant;
    return next();
  }

  // 其次从子域名获取
  const slug = extractTenantSlug(req.hostname);
  if (!slug) {
    res.status(400).json({ error: '无法识别租户' });
    return;
  }

  const { rows } = await pool.query(
    'SELECT id, name, status, config FROM tenants WHERE slug = $1',
    [slug]
  );

  if (rows.length === 0 || rows[0].status !== 'active') {
    res.status(404).json({ error: '租户不存在或已停用' });
    return;
  }

  (req as any).tenant = rows[0];
  (req as any).tenantId = rows[0].id;
  next();
}
```

## PostgreSQL Row Level Security

RLS 是 PostgreSQL 原生支持的行级安全策略，在数据库层面强制执行数据隔离，即使应用代码有 Bug 也不会泄露数据。

```sql
-- 启用 RLS
ALTER TABLE forms ENABLE ROW LEVEL SECURITY;
ALTER TABLE form_data ENABLE ROW LEVEL SECURITY;

-- 创建策略：租户只能访问自己的数据
CREATE POLICY tenant_isolation_forms ON forms
  USING (tenant_id = current_setting('app.tenant_id'));

CREATE POLICY tenant_isolation_form_data ON form_data
  USING (tenant_id = current_setting('app.tenant_id'));

-- 应用层在每个连接/事务开始时设置租户上下文
-- SET LOCAL 仅在当前事务内有效，事务结束自动清除
```

```typescript
// src/db/tenantQuery.ts
import { pool } from './pool';
import { getTenantId } from '../context/tenant';

export async function tenantQuery(text: string, params?: any[]) {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    // 设置当前事务的租户 ID
    await client.query(`SET LOCAL app.tenant_id = '${getTenantId()}'`);
    const result = await client.query(text, params);
    await client.query('COMMIT');
    return result;
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}
```

## 资源配额管理

```typescript
// src/quota/manager.ts
import { redis } from '../redis';
import { pool } from '../db';

interface QuotaConfig {
  maxForms: number;         // 最大表单数
  maxSubmissions: number;   // 每月最大提交数
  maxStorageMB: number;     // 最大存储空间（MB）
  maxUsers: number;         // 最大用户数
  maxApiCalls: number;      // 每日 API 调用次数
}

const PLAN_QUOTAS: Record<string, QuotaConfig> = {
  free:       { maxForms: 5, maxSubmissions: 1000, maxStorageMB: 100, maxUsers: 5, maxApiCalls: 1000 },
  standard:   { maxForms: 50, maxSubmissions: 50000, maxStorageMB: 5000, maxUsers: 50, maxApiCalls: 50000 },
  enterprise: { maxForms: -1, maxSubmissions: -1, maxStorageMB: -1, maxUsers: -1, maxApiCalls: -1 },
};

export async function checkQuota(
  tenantId: string,
  plan: string,
  resource: keyof QuotaConfig
): Promise<{ allowed: boolean; current: number; limit: number }> {
  const quota = PLAN_QUOTAS[plan];
  const limit = quota[resource];

  // -1 表示无限制
  if (limit === -1) {
    return { allowed: true, current: 0, limit: -1 };
  }

  const current = await getCurrentUsage(tenantId, resource);
  return {
    allowed: current < limit,
    current,
    limit,
  };
}

async function getCurrentUsage(
  tenantId: string,
  resource: keyof QuotaConfig
): Promise<number> {
  const queries: Record<string, string> = {
    maxForms: `SELECT COUNT(*)::int AS cnt FROM forms WHERE tenant_id = $1`,
    maxSubmissions: `SELECT COUNT(*)::int AS cnt FROM form_data WHERE tenant_id = $1 AND created_at >= date_trunc('month', NOW())`,
    maxUsers: `SELECT COUNT(*)::int AS cnt FROM users WHERE tenant_id = $1`,
    maxApiCalls: `SELECT COUNT(*)::int AS cnt FROM api_logs WHERE tenant_id = $1 AND created_at >= date_trunc('day', NOW())`,
  };

  const query = queries[resource];
  if (!query) return 0;

  const { rows } = await pool.query(query, [tenantId]);
  return rows[0]?.cnt || 0;
}

// 配额检查中间件
export function quotaGuard(resource: keyof QuotaConfig) {
  return async (req: any, res: any, next: any) => {
    const { tenantId, tenant } = req;
    const result = await checkQuota(tenantId, tenant.config?.plan || 'free', resource);

    if (!result.allowed) {
      res.status(429).json({
        error: '已达到配额上限',
        resource,
        current: result.current,
        limit: result.limit,
      });
      return;
    }

    // 在响应头中返回配额信息
    res.set('X-Quota-Limit', String(result.limit));
    res.set('X-Quota-Remaining', String(result.limit - result.current));
    next();
  };
}
```

## 计量计费

```typescript
// src/billing/meter.ts
import { pool } from '../db';

interface UsageRecord {
  tenantId: string;
  metric: 'api_call' | 'storage' | 'submission' | 'user';
  quantity: number;
  timestamp: Date;
}

export async function recordUsage(record: UsageRecord): Promise<void> {
  await pool.query(
    `INSERT INTO usage_records (tenant_id, metric, quantity, recorded_at)
     VALUES ($1, $2, $3, $4)`,
    [record.tenantId, record.metric, record.quantity, record.timestamp]
  );
}

export async function getMonthlyUsage(tenantId: string) {
  const { rows } = await pool.query(
    `SELECT metric, SUM(quantity)::bigint AS total
     FROM usage_records
     WHERE tenant_id = $1 AND recorded_at >= date_trunc('month', NOW())
     GROUP BY metric`,
    [tenantId]
  );
  return rows;
}
```

## 租户管理后台

```typescript
// src/admin/routes.ts
import { Router } from 'express';
import { adminAuth } from '../middleware/adminAuth';

const router = Router();
router.use(adminAuth);

// 获取所有租户列表
router.get('/tenants', async (req, res) => {
  const { rows } = await pool.query(
    `SELECT t.*, COUNT(u.id) AS user_count
     FROM tenants t
     LEFT JOIN users u ON u.tenant_id = t.id
     GROUP BY t.id
     ORDER BY t.created_at DESC`
  );
  res.json(rows);
});

// 修改租户配额
router.put('/tenants/:id/quotas', async (req, res) => {
  const { quotas } = req.body;
  await pool.query(
    `UPDATE tenants SET config = config || $2 WHERE id = $1`,
    [req.params.id, JSON.stringify({ customQuotas: quotas })]
  );
  res.json({ success: true });
});

// 停用租户
router.post('/tenants/:id/suspend', async (req, res) => {
  await pool.query(
    `UPDATE tenants SET status = 'suspended', suspended_at = NOW() WHERE id = $1`,
    [req.params.id]
  );
  res.json({ success: true });
});

export default router;
```

## 常见误区

1. **忘记在所有查询中加 tenant_id 条件**：漏掉租户过滤会导致数据泄露。使用 RLS 在数据库层面兜底，比仅靠应用层代码更安全
2. **tenant_id 字段不建索引**：所有查询都带 tenant_id 条件，不建索引会导致全表扫描。tenant_id 应出现在每个复合索引的首位
3. **选择独立库模式但没有自动化运维**：独立库模式需要管理大量数据库实例，没有自动化工具会变成运维噩梦

## 工程建议

1. **起步用共享表模式**：大多数场景下共享表 + RLS 足够，只有明确的合规要求或性能瓶颈才需要升级到独立 Schema 或独立库
2. **租户标识贯穿全链路**：从 HTTP 请求到数据库查询，租户 ID 必须一致传递。使用 AsyncLocalStorage 或中间件统一注入，避免在每个接口手动传递
3. **配额管理前置化**：在写操作前检查配额，而非事后统计。使用 Redis 计数器实现实时配额检查，避免超量后再回滚

## 小结

- 三种多租户模型（共享表/独立 Schema/独立库）各有优劣，应根据客户规模和合规要求选择
- 租户识别可通过子域名、请求头或 JWT 实现，推荐子域名 + JWT 组合方案
- PostgreSQL RLS 在数据库层面强制数据隔离，是共享表模式的安全兜底
- 资源配额管理需要针对每个维度（表单数/提交数/API 调用/存储）设置上限
- 计量计费系统记录每项资源使用量，为按量付费提供数据支撑

## 参考答案

### 练习一

**思路**：中间件先从 req.user（JWT 解析后的用户信息）获取 tenantId，查询数据库获取租户详情，然后用 AsyncLocalStorage.run() 注入上下文。

**答案**：
```typescript
import { Request, Response, NextFunction } from 'express';
import { AsyncLocalStorage } from 'async_hooks';
import { pool } from '../db';

interface TenantCtx {
  tenantId: string;
  tenantName: string;
  plan: string;
}

export const tenantStore = new AsyncLocalStorage<TenantCtx>();

export async function tenantMiddleware(
  req: Request,
  res: Response,
  next: NextFunction
) {
  const tenantId = (req as any).user?.tenantId;
  if (!tenantId) {
    res.status(401).json({ error: '缺少租户信息' });
    return;
  }

  const { rows } = await pool.query(
    'SELECT id, name, status, config FROM tenants WHERE id = $1',
    [tenantId]
  );

  if (rows.length === 0) {
    res.status(404).json({ error: '租户不存在' });
    return;
  }

  if (rows[0].status !== 'active') {
    res.status(403).json({ error: '租户已停用' });
    return;
  }

  const tenant = rows[0];
  tenantStore.run(
    {
      tenantId: tenant.id,
      tenantName: tenant.name,
      plan: tenant.config?.plan || 'free',
    },
    () => next()
  );
}
```

**要点**：
- AsyncLocalStorage 确保每个请求的租户上下文独立，不会被并发请求污染
- 停用租户应返回 403 而非 404，语义更准确
- 租户信息应缓存（Redis TTL 5 分钟），避免每次请求都查库

### 练习二

**思路**：使用 Redis 的 INCR 命令实现计数器，键名包含年月（如 `quota:tenant123:submissions:202606`），设置 TTL 为 32 天自动跨月重置。

**答案**：
```typescript
import { redis } from '../redis';

function getMonthKey(tenantId: string): string {
  const now = new Date();
  const month = `${now.getFullYear()}${String(now.getMonth() + 1).padStart(2, '0')}`;
  return `quota:${tenantId}:submissions:${month}`;
}

export async function checkSubmissionQuota(
  tenantId: string,
  limit: number
): Promise<{ allowed: boolean; current: number; remaining: number }> {
  const key = getMonthKey(tenantId);
  const current = Number(await redis.get(key)) || 0;

  return {
    allowed: current < limit,
    current,
    remaining: Math.max(0, limit - current),
  };
}

export async function incrementSubmission(tenantId: string): Promise<number> {
  const key = getMonthKey(tenantId);
  const count = await redis.incr(key);

  // 首次设置时添加过期时间（32 天后自动删除，跨月自动重置）
  if (count === 1) {
    await redis.expire(key, 32 * 24 * 3600);
  }

  return count;
}

// 使用示例
async function handleSubmit(tenantId: string, formData: any) {
  const quota = await checkSubmissionQuota(tenantId, 1000);
  if (!quota.allowed) {
    throw new Error(`已达到本月提交上限（${quota.current}/${quota.current + quota.remaining}）`);
  }

  await saveFormData(tenantId, formData);
  await incrementSubmission(tenantId);
}
```

**要点**：
- Redis INCR 是原子操作，天然支持并发场景
- 键名包含年月，天然按月隔离，TTL 32 天确保跨月后旧键自动清理
- 首次写入时设置 TTL，后续 INCR 不需要重复设置
