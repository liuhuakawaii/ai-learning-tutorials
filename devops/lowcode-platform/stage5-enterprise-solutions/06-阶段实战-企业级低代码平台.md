```typescript
interface CreateTenantInput {

  slug: string;

    );

    // 2. 创建管理员用户
    const { rows: [admin] } = await client.query(
      `INSERT INTO users (tenant_id, email, name, role)
       VALUES ($1, $2, $3, 'admin')
       RETURNING id`,
      [tenant.id, input.adminEmail, `${input.name} 管理员`]
    );

    // 3. 初始化默认表单模板
    await client.query(
      `INSERT INTO forms (tenant_id, name, schema_json, created_by)
       VALUES ($1, '示例表单', $2, $3)`,
      [tenant.id, JSON.stringify(DEFAULT_FORM_SCHEMA), admin.id]
    );

    await client.query('COMMIT');

    // 4. 触发钩子（异步，不阻塞）
    hooks.emitAsync('tenant.created', tenant).catch(console.error);

    return tenant;
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}
```

## 安全加固

### 认证鉴权中间件栈

```typescript
// src/middleware/authStack.ts
import { verifyAccessToken } from '../auth/jwt';
import { checkQuota } from '../quota/manager';
import { logAudit } from '../audit/logger';

// JWT 认证
export function authenticate(req: any, res: any, next: any) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  if (!token) {
    res.status(401).json({ error: '未提供认证令牌' });
    return;
  }

  try {
    req.user = verifyAccessToken(token);
    req.tenantId = req.user.tenantId;
    next();
  } catch (err: any) {
    if (err.name === 'TokenExpiredError') {
      res.status(401).json({ error: '令牌已过期', code: 'TOKEN_EXPIRED' });
    } else {
      res.status(401).json({ error: '无效的认证令牌' });
    }
  }
}

// 角色鉴权
export function authorize(...roles: string[]) {
  return (req: any, res: any, next: any) => {
    if (!roles.includes(req.user?.role)) {
      logAudit({
        userId: req.user.userId,
        tenantId: req.tenantId,
        action: 'UNAUTHORIZED_ACCESS',
        resourceType: req.path,
        resourceId: 'N/A',
        ip: req.ip,
        userAgent: req.headers['user-agent'],
      });
      res.status(403).json({ error: '权限不足' });
      return;
    }
    next();
  };
}

// 审计日志（写操作）
export function audit(action: string, resourceType: string) {
  return (req: any, res: any, next: any) => {
    const originalJson = res.json.bind(res);
    res.json = (body: any) => {
      if (res.statusCode < 400) {
        logAudit({
          userId: req.user?.userId,
          tenantId: req.tenantId,
          action,
          resourceType,
          resourceId: req.params.id || body?.id || 'unknown',
          changes: { after: req.body },
          ip: req.ip,
          userAgent: req.headers['user-agent'],
        }).catch(console.error);
      }
      return originalJson(body);
    };
    next();
  };
}
```

### 数据加密服务

```typescript
// src/services/encryption.ts
import { createCipheriv, createDecipheriv, randomBytes } from 'crypto';

const KEY = Buffer.from(process.env.ENCRYPTION_KEY!, 'hex');
const ALGO = 'aes-256-gcm';

const SENSITIVE_FIELDS = new Set([
  'idCard', 'phone', 'bankAccount', 'medicalRecord', 'salary'
]);

export function encryptField(value: string): string {
  const iv = randomBytes(16);
  const cipher = createCipheriv(ALGO, KEY, iv);
  const enc = Buffer.concat([cipher.update(value, 'utf8'), cipher.final()]);
  const tag = cipher.getAuthTag();
  return Buffer.concat([iv, tag, enc]).toString('base64');
}

export function decryptField(value: string): string {
  const buf = Buffer.from(value, 'base64');
  const decipher = createDecipheriv(ALGO, KEY, buf.subarray(0, 16));
  decipher.setAuthTag(buf.subarray(16, 32));
  return decipher.update(buf.subarray(32)) + decipher.final('utf8');
}

export function encryptSensitiveData(data: Record<string, any>): Record<string, any> {
  const result = { ...data };
  for (const [key, value] of Object.entries(result)) {
    if (SENSITIVE_FIELDS.has(key) && typeof value === 'string') {
      result[key] = encryptField(value);
    }
  }
  return result;
}

export function decryptSensitiveData(data: Record<string, any>): Record<string, any> {
  const result = { ...data };
  for (const [key, value] of Object.entries(result)) {
    if (SENSITIVE_FIELDS.has(key) && typeof value === 'string') {
      try { result[key] = decryptField(value); } catch {}
    }
  }
  return result;
}
```

### 缓存层封装

```typescript
// src/cache/index.ts
import { redis } from '../redis';

const CACHE_CONFIG = {
  form: { ttl: 300, prefix: 'cache:form' },
  permission: { ttl: 600, prefix: 'cache:perm' },
  config: { ttl: 900, prefix: 'cache:cfg' },
} as const;

type CacheType = keyof typeof CACHE_CONFIG;

export async function getCached<T>(
  type: CacheType,
  key: string,
  fetcher: () => Promise<T>
): Promise<T> {
  const cfg = CACHE_CONFIG[type];
  const cacheKey = `${cfg.prefix}:${key}`;

  const cached = await redis.get(cacheKey);
  if (cached) return JSON.parse(cached);

  const data = await fetcher();
  await redis.setex(cacheKey, cfg.ttl, JSON.stringify(data));
  return data;
}

export async function invalidateCache(type: CacheType, key: string): Promise<void> {
  const cfg = CACHE_CONFIG[type];
  await redis.del(`${cfg.prefix}:${key}`);
}

export async function invalidateCachePattern(type: CacheType, pattern: string): Promise<void> {
  const cfg = CACHE_CONFIG[type];
  const keys = await redis.keys(`${cfg.prefix}:${pattern}`);
  if (keys.length > 0) await redis.del(...keys);
}
```

## 插件系统整合

```typescript
// src/plugin/sandbox.ts
import { Worker } from 'worker_threads';
import path from 'path';

export async function runInSandbox(
  pluginCode: string,
  context: { settings: Record<string, any>; data: any },
  timeoutMs: number = 5000
): Promise<any> {
  return new Promise((resolve, reject) => {
    const worker = new Worker(path.join(__dirname, 'sandbox-worker.js'), {
      workerData: { code: pluginCode, context },
    });

    const timer = setTimeout(() => {
      worker.terminate();
      reject(new Error('插件执行超时'));
    }, timeoutMs);

    worker.on('message', (result) => {
      clearTimeout(timer);
      resolve(result);
    });

    worker.on('error', (err) => {
      clearTimeout(timer);
      reject(err);
    });
  });
}
```

```typescript
import { Registry, Counter, Histogram, Gauge } from 'prom-client';

export const register = new Registry();

export const httpRequests = new Counter({
  name: 'http_requests_total',
  help: 'HTTP 请求总数',
  labelNames: ['method', 'path', 'status'],
  registers: [register],
});

export const httpRequestDuration = new Histogram({
  name: 'http_request_duration_seconds',
  help: 'HTTP 请求耗时',
  labelNames: ['method', 'path'],
  buckets: [0.01, 0.05, 0.1, 0.5, 1, 2, 5],
  registers: [register],
});

export const activeTenants = new Gauge({
  name: 'active_tenants_total',
  help: '活跃租户数',
  registers: [register],
});

export function metricsMiddleware(req: any, res: any, next: any) {
  const start = Date.now();
  res.on('finish', () => {
    const duration = (Date.now() - start) / 1000;
    const path = req.route?.path || req.path;
    httpRequests.inc({ method: req.method, path, status: res.statusCode });
    httpRequestDuration.observe({ method: req.method, path }, duration);
  });
  next();
}
```

### 故障恢复流程

```
1. 检测：Prometheus 告警 → Grafana 看板 → 确认故障类型
2. 止损：Kubernetes 自动重启 Pod / 手动回滚 Deployment
3. 恢复：
   - 数据库故障 → 从最近备份恢复 + WAL 日志回放
   - 服务故障 → 检查日志 → 修复 → 重新部署
   - 全站故障 → 切换到备用集群 / 灾备环境
4. 复盘：记录故障原因 → 制定改进措施 → 更新 Runbook
```

## 常见误区

1. **过度设计导致开发停滞**：一开始就追求完美的架构，结果半年都上不了线。应该先用最简方案跑通核心流程，再逐步演进
2. **只考虑正常流程，不考虑故障场景**：每个组件都可能出故障，必须为每个关键路径设计降级方案
3. **监控告警缺失或过多**：没有告警等于盲人摸象，告警太多等于没有告警。设置合理的阈值和分级策略

## 工程建议

1. **分阶段交付**：第一阶段实现核心表单和数据管理，第二阶段加入工作流和权限，第三阶段实现多租户和插件系统
2. **基础设施即代码**：所有 K8s 配置、Helm values、CI/CD pipeline 都应版本控制，确保环境可复现
3. **建立 Runbook**：为每种常见故障编写处理手册，包含检测方法、处理步骤和恢复验证，确保值班人员能快速响应

## 小结

- 企业级低代码平台需要整合多租户、安全、性能、可扩展性等多个子系统
- 架构设计应遵循"先跑通再优化"的原则，分阶段迭代交付
- 安全加固贯穿全链路：认证鉴权 → 数据加密 → 审计日志 → 安全响应头
- 监控告警是生产环境的"眼睛"，必须覆盖错误率、延迟、资源使用率等关键指标
- 容灾方案包括定期备份、故障检测、快速恢复和事后复盘四个环节

## 参考答案

### 练习一

**思路**：根据规模估算资源需求，选择合适的架构模式。1000 租户 × 500 用户 = 50 万用户，日均 100 万次 API ≈ 每秒 12 次（峰值按 5 倍算约 60 QPS），属于中等规模。

**答案**：
```
架构方案：

前端层：React SPA + CDN（静态资源走 CDN，API 走网关）
网关层：Nginx Ingress + Rate Limiting + SSL Termination
应用层：API Server × 3 副本（K8s Deployment），Pod 内运行 Node.js
数据层：
  - PostgreSQL 15（主从复制，读写分离）
  - Redis 7（缓存 + 会话 + 频率限制）
  - MinIO（文件存储，兼容 S3 协议）

多租户：共享库共享表 + PostgreSQL RLS
  - 1000 租户规模下共享表方案成本最低
  - RLS 在数据库层面保障隔离安全

关键设计决策：
1. 读写分离：读请求（表单渲染/数据查询）远多于写请求，读写比约 8:2
2. 缓存策略：表单配置缓存 5 分钟，用户权限缓存 10 分钟，减少 DB 压力
3. 文件存储：使用 MinIO 而非本地磁盘，支持水平扩展和数据持久化
```

**要点**：
- 规模估算是架构设计的基础，不同量级需要不同的架构方案
- 共享表方案在千租户级别完全可行，只有合规要求高时才考虑独立库
- 读写分离 + 缓存是应对中等规模最有效的优化手段

### 练习二

**思路**：使用数据库事务确保租户和管理员的原子性创建，通过 slug 唯一约束防止重名，最后触发异步钩子通知其他子系统。

**答案**：
```typescript
import { pool } from '../db';
import { logAudit } from '../audit/logger';
import { encryptField } from '../services/encryption';

interface CreateTenantDTO {
  name: string;
  slug: string;
  adminEmail: string;
  adminName: string;
  plan: 'free' | 'standard' | 'enterprise';
}

export async function createTenant(dto: CreateTenantDTO, operatorId: string) {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');

    // 检查 slug 唯一性
    const { rows: existing } = await client.query(
      'SELECT id FROM tenants WHERE slug = $1',
      [dto.slug]
    );
    if (existing.length > 0) {
      throw new Error('租户标识已存在');
    }

    // 创建租户
    const { rows: [tenant] } = await client.query(
      `INSERT INTO tenants (name, slug, status, config)
       VALUES ($1, $2, 'active', $3)
       RETURNING *`,
      [dto.name, dto.slug, JSON.stringify({ plan: dto.plan, createdBy: operatorId })]
    );

    // 创建管理员
    const { rows: [admin] } = await client.query(
      `INSERT INTO users (tenant_id, email, name, role, status)
       VALUES ($1, $2, $3, 'admin', 'active')
       RETURNING id, email, name`,
      [tenant.id, dto.adminEmail, dto.adminName]
    );

    // 初始化默认表单
    await client.query(
      `INSERT INTO forms (tenant_id, name, schema_json, created_by)
       VALUES ($1, '联系表单', $2, $3)`,
      [tenant.id, JSON.stringify({
        fields: [
          { type: 'input', name: 'name', label: '姓名', required: true },
          { type: 'input', name: 'phone', label: '手机号', required: true },
          { type: 'textarea', name: 'message', label: '留言', required: false },
        ]
      }), admin.id]
    );

    await client.query('COMMIT');

    // 审计日志（事务外，不阻塞）
    logAudit({
      userId: operatorId,
      tenantId: tenant.id,
      action: 'CREATE',
      resourceType: 'tenant',
      resourceId: tenant.id,
      changes: { after: { name: dto.name, slug: dto.slug, plan: dto.plan } },
      ip: '0.0.0.0',
      userAgent: 'system',
    }).catch(console.error);

    return { tenant, admin };
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}
```

**要点**：
- 事务确保租户、管理员、默认表单要么全部创建成功，要么全部回滚
- slug 唯一性在应用层检查（查询）和数据库层保障（唯一索引）双重防护
- 审计日志在事务提交后异步写入，避免影响接口响应时间
- 默认表单模板让新租户开箱即用，降低上手门槛
