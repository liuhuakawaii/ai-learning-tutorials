### Step 1：后端认证系统

```python
# backend/app/core/security.py
from datetime import datetime, timedelta
from jose import JWTError, jwt
from passlib.context import CryptContext
from app.core.config import settings

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

def hash_password(password: str) -> str:
    return pwd_context.hash(password)

def verify_password(plain: str, hashed: str) -> bool:
    return pwd_context.verify(plain, hashed)

def create_access_token(data: dict, expires_delta: timedelta | None = None) -> str:
    to_encode = data.copy()
    expire = datetime.utcnow() + (expires_delta or timedelta(hours=24))
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, settings.SECRET_KEY, algorithm="HS256")

def decode_token(token: str) -> dict | None:
    try:
        return jwt.decode(token, settings.SECRET_KEY, algorithms=["HS256"])
    except JWTError:
        return None
```

```python
# backend/app/api/v1/auth.py
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from app.core.database import get_db
from app.core.security import hash_password, verify_password, create_access_token
from app.models.user import User
from app.schemas.auth import RegisterRequest, LoginRequest, TokenResponse

router = APIRouter(prefix="/api/v1/auth", tags=["认证"])

@router.post("/register", response_model=TokenResponse, status_code=201)
async def register(request: RegisterRequest, db: AsyncSession = Depends(get_db)):
    # 检查邮箱是否已注册
    existing = await db.execute(select(User).where(User.email == request.email))
    if existing.scalar_one_or_none():
        raise HTTPException(400, "邮箱已注册")
    
    # 创建用户
    user = User(
        email=request.email,
        username=request.username,
        hashed_password=hash_password(request.password),
    )
    db.add(user)
    await db.flush()
    
    # 生成 token
    token = create_access_token({"sub": user.id})
    return TokenResponse(access_token=token, user_id=user.id)

@router.post("/login", response_model=TokenResponse)
async def login(request: LoginRequest, db: AsyncSession = Depends(get_db)):
    result = await db.execute(select(User).where(User.email == request.email))
    user = result.scalar_one_or_none()
    
    if not user or not verify_password(request.password, user.hashed_password):
        raise HTTPException(401, "邮箱或密码错误")
    
    token = create_access_token({"sub": user.id})
    return TokenResponse(access_token=token, user_id=user.id)
```

### Step 2：对话 API

```python
# backend/app/api/v1/chat.py
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, func
from app.core.database import get_db
from app.api.deps import get_current_user
from app.models.user import User
from app.models.session import ChatSession
from app.models.message import Message
from app.schemas.chat import (
    CreateSessionRequest, SessionResponse, SessionListResponse,
    SendMessageRequest, MessageResponse,
)
import uuid

router = APIRouter(prefix="/api/v1/chat", tags=["对话"])

@router.get("/sessions", response_model=SessionListResponse)
async def list_sessions(
    page: int = 1,
    size: int = 20,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    offset = (page - 1) * size
    
    # 查询总数
    count_result = await db.execute(
        select(func.count()).select_from(ChatSession).where(
            ChatSession.user_id == user.id
        )
    )
    total = count_result.scalar()
    
    # 查询列表
    result = await db.execute(
        select(ChatSession)
        .where(ChatSession.user_id == user.id)
        .order_by(ChatSession.updated_at.desc())
        .offset(offset)
        .limit(size)
    )
    sessions = result.scalars().all()
    
    return SessionListResponse(
        items=[SessionResponse.model_validate(s) for s in sessions],
        total=total,
        page=page,
        size=size,
    )

@router.post("/sessions", response_model=MessageResponse, status_code=201)
async def create_session(
    request: CreateSessionRequest,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    # 创建会话
    session = ChatSession(
        id=str(uuid.uuid4()),
        user_id=user.id,
        title=request.message[:50],  # 用第一条消息作为标题
    )
    db.add(session)
    
    # 保存用户消息
    user_msg = Message(
        id=str(uuid.uuid4()),
        session_id=session.id,
        role="user",
        content=request.message,
    )
    db.add(user_msg)
    
    # 生成模拟 AI 回复（后续阶段接入真实 LLM）
    ai_msg = Message(
        id=str(uuid.uuid4()),
        session_id=session.id,
        role="assistant",
        content=f"收到你的消息：「{request.message}」。这是模拟回复，后续阶段会接入真实 LLM。",
        model="mock",
        token_usage=0,
    )
    db.add(ai_msg)
    
    await db.flush()
    
    return MessageResponse.model_validate(ai_msg)

@router.post("/sessions/{session_id}/messages", response_model=MessageResponse)
async def send_message(
    session_id: str,
    request: SendMessageRequest,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    # 验证会话存在且属于当前用户
    result = await db.execute(
        select(ChatSession).where(
            ChatSession.id == session_id,
            ChatSession.user_id == user.id,
        )
    )
    session = result.scalar_one_or_none()
    if not session:
        raise HTTPException(404, "会话不存在")
    
    # 保存用户消息
    user_msg = Message(
        id=str(uuid.uuid4()),
        session_id=session_id,
        role="user",
        content=request.message,
    )
    db.add(user_msg)
    
    # 模拟 AI 回复
    ai_msg = Message(
        id=str(uuid.uuid4()),
        session_id=session_id,
        role="assistant",
        content=f"收到：「{request.message}」（模拟回复）",
        model="mock",
        token_usage=0,
    )
    db.add(ai_msg)
    
    await db.flush()
    return MessageResponse.model_validate(ai_msg)
```

<template>

      <div class="messages" ref="messagesRef">
        <div
          v-for="msg in chatStore.messages"
          :key="msg.id"
          :class="['message', msg.role]"
        >
          <n-avatar v-if="msg.role === 'assistant'" round size="small">AI</n-avatar>
          <div class="bubble">{{ msg.content }}</div>
          <n-avatar v-if="msg.role === 'user'" round size="small">U</n-avatar>
        </div>
      </div>

      <!-- 输入区域 -->
      <div class="input-area">
        <n-input
          v-model:value="inputText"
          type="textarea"
          placeholder="输入消息... (Ctrl+Enter 发送)"
          :rows="3"
          :disabled="chatStore.isSending"
          @keydown.ctrl.enter="handleSend"
        />
        <n-button
          type="primary"
          :loading="chatStore.isSending"
          :disabled="!inputText.trim()"
          @click="handleSend"
        >
          发送
        </n-button>
      </div>
    </div>
  </div>

<script setup lang="ts">
import { ref, onMounted, nextTick, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { useChatStore } from '@/stores/chat'
import { useApiError } from '@/composables/useApiError'

const route = useRoute()
const router = useRouter()
const chatStore = useChatStore()
const { handleError } = useApiError()

const inputText = ref('')
const messagesRef = ref<HTMLElement>()

onMounted(() => {
  chatStore.loadSessions()
  if (route.params.sessionId) {
    chatStore.loadSession(route.params.sessionId as string)
  }
})

// 消息列表变化时自动滚动到底部
watch(() => chatStore.messages.length, () => {
  nextTick(() => {
    if (messagesRef.value) {
      messagesRef.value.scrollTop = messagesRef.value.scrollHeight
    }
  })
})

async function createNewSession() {
  if (!inputText.value.trim()) return
  try {
    await chatStore.createSession(inputText.value)
    inputText.value = ''
    router.push(`/chat/${chatStore.currentSessionId}`)
  } catch (error) {
    handleError(error)
  }
}

async function handleSend() {
  if (!inputText.value.trim() || !chatStore.currentSessionId) return
  try {
    await chatStore.sendMessage(inputText.value)
    inputText.value = ''
  } catch (error) {
    handleError(error)
  }
}

function selectSession(id: string) {
  chatStore.loadSession(id)
  router.push(`/chat/${id}`)
}

function formatTime(dateStr: string) {
  return new Date(dateStr).toLocaleString('zh-CN')
}
</script>

## 常见错误

| 错误 | 原因 | 解决 |
|------|------|------|
| 前端登录后 token 不生效 | localStorage 存储失败 | 检查浏览器隐私设置 |
| 创建会话后列表不更新 | 忘记调用 `loadSessions()` | 在创建成功后刷新列表 |
| Docker 构建很慢 | 没用缓存 | 用 `.dockerignore` 排除 `node_modules` |
| 数据库迁移失败 | 模型导入不完整 | 确保 `alembic/env.py` 导入所有模型 |
