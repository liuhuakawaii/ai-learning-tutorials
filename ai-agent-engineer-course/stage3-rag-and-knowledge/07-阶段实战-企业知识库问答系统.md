    kb_id: str,
    file: UploadFile,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    # 1. 保存文件
    file_path = await save_upload_file(file)
    
    # 2. 创建文档记录
    document = Document(
        id=str(uuid.uuid4()),
        knowledge_base_id=kb_id,
        filename=file.filename,
        file_type=file.content_type,
        status="processing",
    )
    db.add(document)
    await db.flush()
    
    # 3. 异步处理（解析、切分、向量化）
    await task_queue.enqueue("process_document", {
        "document_id": document.id,
        "file_path": file_path,
    })
    
    return {"document_id": document.id, "status": "processing"}

### 2. 文档处理 Worker

```python
# backend/app/workers/document_worker.py
async def process_document(payload: dict):
    """文档处理 Worker"""
    document_id = payload["document_id"]
    file_path = payload["file_path"]
    
    # 1. 解析文档
    parser = DocumentParserFactory()
    parsed = await parser.parse(file_path)
    
    # 2. 切分文本
    splitter = RecursiveSplitter(chunk_size=500, chunk_overlap=50)
    chunks = splitter.split(parsed.content)
    
    # 3. 生成 Embedding 并存储
    embedding_service = EmbeddingService()
    vector_store = VectorStore(db, embedding_service)
    await vector_store.add_chunks(document_id, chunks)
    
    # 4. 建立 BM25 索引
    bm25_service.index(chunks)
    
    # 5. 更新文档状态
    document.status = "ready"
    document.chunk_count = len(chunks)
    await db.commit()
```

### 3. RAG 问答 API

```python
# backend/app/api/v1/chat.py
@router.post("/sessions/{session_id}/messages/stream")
async def stream_rag_message(
    session_id: str,
    request: SendMessageRequest,
    retrieval_pipeline: RetrievalPipeline = Depends(),
    query_optimizer: QueryOptimizer = Depends(),
):
    async def event_generator():
        # 1. 查询优化
        optimized_query = await query_optimizer.rewrite(request.message)
        
        # 2. 检索相关文档
        contexts = await retrieval_pipeline.retrieve(
            optimized_query, knowledge_base_id=request.knowledge_base_id
        )
        
        # 3. 构建带引用的 Prompt
        prompt = CITATION_PROMPT.format(
            contexts=format_contexts(contexts),
            question=request.message,
        )
        
        # 4. 流式生成回答
        full_answer = ""
        async for chunk in llm.stream_chat(
            messages=[{"role": "user", "content": prompt}],
            model="gpt-4o",
        ):
            full_answer += chunk.content
            yield sse_event({"type": "chunk", "content": chunk.content})
        
        # 5. 幻觉检测
        detector = HallucinationDetector(llm)
        hallucination = await detector.check(
            request.message, full_answer, [c["content"] for c in contexts]
        )
        
        # 6. 发送引用和置信度
        yield sse_event({
            "type": "metadata",
            "citations": extract_citations(full_answer),
            "sources": contexts,
            "hallucination": hallucination,
        })
        
        yield sse_event({"type": "done"})
    
    return StreamingResponse(event_generator(), media_type="text/event-stream")
```
