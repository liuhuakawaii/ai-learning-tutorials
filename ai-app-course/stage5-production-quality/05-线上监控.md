## 二、实时监控实现

### 2.1 指标收集器

```typescript
// lib/monitoring/collector.ts
import { prisma } from '../prisma'

export class MetricsCollector {
  // 记录请求指标
  async recordRequest(metrics: {
    userId: string
    duration: number
    tokens: number
    cost: number
    status: 'success' | 'error' | 'timeout'
    model: string
  }) {
    await prisma.metric.create({
      data: {
        type: 'request',
        timestamp: new Date(),
        value: metrics.duration,
        tags: {
          userId: metrics.userId,
          status: metrics.status,
          model: metrics.model,
        },
      },
    })

    // 记录 token 使用
    await prisma.metric.create({
      data: {
        type: 'tokens',
        timestamp: new Date(),
        value: metrics.tokens,
        tags: { userId: metrics.userId, model: metrics.model },
      },
    })

    // 记录成本
    await prisma.metric.create({
      data: {
        type: 'cost',
        timestamp: new Date(),
        value: metrics.cost,
        tags: { userId: metrics.userId, model: metrics.model },
      },
    })
  }

  // 查询指标
  async query(
    type: string,
    startTime: Date,
    endTime: Date,
    tags?: Record<string, string>
  ) {
    const where: any = {
      type,
      timestamp: { gte: startTime, lte: endTime },
    }

    if (tags) {
      where.tags = { contains: tags }
    }

    return prisma.metric.findMany({
      where,
      orderBy: { timestamp: 'asc' },
    })
  }

  // 聚合指标
  async aggregate(
    type: string,
    startTime: Date,
    endTime: Date,
    aggregation: 'avg' | 'sum' | 'min' | 'max' | 'count'
  ) {
    const result = await prisma.metric.aggregate({
      where: {
        type,
        timestamp: { gte: startTime, lte: endTime },
      },
      _avg: aggregation === 'avg' ? { value: true } : undefined,
      _sum: aggregation === 'sum' ? { value: true } : undefined,
      _min: aggregation === 'min' ? { value: true } : undefined,
      _max: aggregation === 'max' ? { value: true } : undefined,
      _count: aggregation === 'count' ? true : undefined,
    })

    return result
  }
}

export const metricsCollector = new MetricsCollector()
```

### 3.2 告警检查器

```typescript
export class AlertChecker {
  async check() {
    for (const rule of ALERT_RULES) {
      const triggered = await this.evaluateRule(rule)
      if (triggered) {
        await this.sendAlert(rule)
      }
    }
  }

  private async evaluateRule(rule: AlertRule): Promise<boolean> {
    const endTime = new Date()
    const startTime = new Date(endTime.getTime() - rule.duration * 1000)

    const value = await this.getMetricValue(rule.metric, startTime, endTime)

    switch (rule.condition) {
      case 'gt': return value > rule.threshold
      case 'lt': return value < rule.threshold
      case 'eq': return value === rule.threshold
    }
  }

  private async sendAlert(rule: AlertRule) {
    for (const channel of rule.channels) {
      switch (channel) {
        case 'slack':
          await this.sendSlackAlert(rule)
          break
        case 'email':
          await this.sendEmailAlert(rule)
          break
      }
    }
  }

  private async sendSlackAlert(rule: AlertRule) {
    await fetch(process.env.SLACK_WEBHOOK_URL!, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        text: `🚨 告警：${rule.name}\n指标：${rule.metric}\n阈值：${rule.threshold}`,
      }),
    })
  }
}
```
