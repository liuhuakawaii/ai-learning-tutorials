## 一、评估集回归测试

### 1.1 自动化评估脚本

```typescript
// scripts/run-eval.ts
import { runEval } from '@/lib/evals/runner'
import { saveEvalResult } from '@/lib/evals/storage'

async function main() {
  console.log('开始运行评估...')

  const result = await runEval()

  console.log(`评估完成：`)
  console.log(`  通过率：${((result.passed / result.totalCases) * 100).toFixed(1)}%`)
  console.log(`  相关性：${(result.scores.avgRelevancy * 100).toFixed(0)}%`)
  console.log(`  忠实度：${(result.scores.avgFaithfulness * 100).toFixed(0)}%`)
  console.log(`  完整性：${(result.scores.avgCompleteness * 100).toFixed(0)}%`)

  // 保存结果
  await saveEvalResult(result)

  // 检查是否达标
  const passed = result.passed / result.totalCases >= 0.9
  if (!passed) {
    console.error('❌ 评估未通过！')
    process.exit(1)
  }

  console.log('✅ 评估通过！')
}

main()
```

### 2.2 版本管理脚本

```typescript
// scripts/prompt-version.ts
import { promptManager } from '@/lib/prompt-manager'

async function main() {
  const action = process.argv[2]
  const name = process.argv[3]

  switch (action) {
    case 'list':
      const history = await promptManager.getHistory(name)
      console.log(`${name} 的版本历史：`)
      for (const v of history) {
        console.log(`  v${v.version} ${v.isActive ? '(当前)' : ''} - ${v.description || '无描述'}`)
      }
      break

    case 'activate':
      const version = parseInt(process.argv[4])
      await promptManager.activate(name, version)
      console.log(`已激活 ${name} v${version}`)
      break

    case 'rollback':
      const rolled = await promptManager.rollback(name)
      console.log(`已回滚到 ${name} v${rolled.version}`)
      break
  }
}

main()
```

export async function GET(req: NextRequest) {
  const period = req.nextUrl.searchParams.get('period') || '1h'
  const startTime = getStartTime(period)
  const endTime = new Date()

  const [requests, errors, latency, cost] = await Promise.all([
    metricsCollector.aggregate('request', startTime, endTime, 'count'),
    metricsCollector.aggregate('request_error', startTime, endTime, 'count'),
    metricsCollector.aggregate('request', startTime, endTime, 'avg'),
    metricsCollector.aggregate('cost', startTime, endTime, 'sum'),
  ])

  return Response.json({
    totalRequests: requests._count || 0,
    errorRate: (errors._count || 0) / (requests._count || 1),
    avgLatency: latency._avg?.value || 0,
    totalCost: cost._sum?.value || 0,
  })
}

## 四、发布检查清单

### 4.1 检查清单实现

```typescript
// lib/release/checklist.ts
export async function runReleaseChecks(): Promise<{
  passed: boolean
  checks: { name: string; status: string; details?: string }[]
}> {
  const checks = []

  // 1. 评估检查
  const evalResult = await getLatestEval()
  checks.push({
    name: '评估通过率 ≥ 90%',
    status: evalResult.passedRate >= 0.9 ? 'passed' : 'failed',
    details: `当前通过率：${(evalResult.passedRate * 100).toFixed(1)}%`,
  })

  // 2. 安全检查
  const securityCheck = await runSecurityChecks()
  checks.push({
    name: '安全测试通过',
    status: securityCheck.passed ? 'passed' : 'failed',
    details: securityCheck.details,
  })

  // 3. 性能检查
  const perfCheck = await runPerformanceChecks()
  checks.push({
    name: 'P95 延迟 < 5 秒',
    status: perfCheck.p95 < 5000 ? 'passed' : 'failed',
    details: `当前 P95：${perfCheck.p95}ms`,
  })

  // 4. 监控检查
  const monitoringCheck = await checkMonitoringSetup()
  checks.push({
    name: '监控告警已配置',
    status: monitoringCheck.configured ? 'passed' : 'failed',
  })

  const passed = checks.every(c => c.status === 'passed')

  return { passed, checks }
}
```

### 4.2 发布页面

```typescript
// app/(dashboard)/admin/release/page.tsx
'use client'

import { useState, useEffect } from 'react'

export default function ReleasePage() {
  const [checks, setChecks] = useState<any[]>([])
  const [allPassed, setAllPassed] = useState(false)

  useEffect(() => {
    runChecks()
  }, [])

  const runChecks = async () => {
    const res = await fetch('/api/release/check')
    const data = await res.json()
    setChecks(data.checks)
    setAllPassed(data.passed)
  }

  const handleRelease = async () => {
    if (!confirm('确定要发布吗？')) return

    await fetch('/api/release/publish', { method: 'POST' })
    alert('已开始灰度发布')
  }

  return (
    <div className="release-page">
      <h1>发布检查</h1>

      <div className="checks">
        {checks.map((check, i) => (
          <div key={i} className={`check ${check.status}`}>
            <span className="icon">{check.status === 'passed' ? '✅' : '❌'}</span>
            <span className="name">{check.name}</span>
            {check.details && <span className="details">{check.details}</span>}
          </div>
        ))}
      </div>

      <div className="actions">
        <button onClick={runChecks}>重新检查</button>
        <button
          onClick={handleRelease}
          disabled={!allPassed}
          className="release-btn"
        >
          {allPassed ? '开始灰度发布' : '检查未通过'}
        </button>
      </div>
    </div>
  )
}
```

// app/api/incidents/route.ts
export async function POST(req: NextRequest) {
  const body = await req.json()

  const incident = await prisma.incident.create({
    data: {
      title: body.title,
      severity: body.severity,
      description: body.description,
      rootCause: body.rootCause,
      impact: body.impact,
      retrospective: body.retrospective,
      actionItems: body.actionItems,
    },
  })

  return Response.json(incident)
}

export async function GET() {
  const incidents = await prisma.incident.findMany({
    orderBy: { discoveredAt: 'desc' },
    take: 20,
  })

  return Response.json(incidents)
}
