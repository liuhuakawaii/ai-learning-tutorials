## 二、核心模块实现

### 2.1 文档管理

```typescript
// lib/document-manager.ts
import fs from 'fs/promises'
import path from 'path'
import { parseDocument } from './parsers'
import { chunkDocument } from './chunker'
import { getEmbeddings } from './embedding'
import { vectorStore } from './vector-store'

interface DocumentRecord {
  id: string
  filename: string
  uploadedAt: Date
  chunkCount: number
  status: 'indexing' | 'ready' | 'error'
}

const documents = new Map<string, DocumentRecord>()

// 上传并索引文档
export async function uploadAndIndex(file: File): Promise<DocumentRecord> {
  const docId = `doc-${Date.now()}`

  // 保存文件
  const buffer = Buffer.from(await file.arrayBuffer())
  const uploadDir = path.join(process.cwd(), 'uploads')
  await fs.mkdir(uploadDir, { recursive: true })
  const filePath = path.join(uploadDir, file.name)
  await fs.writeFile(filePath, buffer)

  // 创建记录
  const record: DocumentRecord = {
    id: docId,
    filename: file.name,
    uploadedAt: new Date(),
    chunkCount: 0,
    status: 'indexing',
  }
  documents.set(docId, record)

  // 异步索引
  indexDocument(docId, filePath).catch(err => {
    console.error('索引失败:', err)
    record.status = 'error'
  })

  return record
}

// 索引文档
async function indexDocument(docId: string, filePath: string) {
  const record = documents.get(docId)!
  const filename = path.basename(filePath)
  const ext = path.extname(filename).slice(1)

  // 1. 解析文档
  const parsed = await parseDocument(filePath)

  // 2. 切分文档
  const chunks = chunkDocument(parsed.content, filename, ext)

  // 3. 生成 Embedding
  const embeddings = await getEmbeddings(chunks.map(c => c.content))

  // 4. 存储向量
  for (let i = 0; i < chunks.length; i++) {
    await vectorStore.add({
      id: `${docId}-chunk-${i}`,
      embedding: embeddings[i],
      content: chunks[i].content,
      metadata: {
        ...chunks[i].metadata,
        docId,
      },
    })
  }

  // 更新记录
  record.chunkCount = chunks.length
  record.status = 'ready'
}

// 删除文档
export async function deleteDocument(docId: string) {
  const record = documents.get(docId)
  if (!record) return

  // 删除向量
  await vectorStore.deleteByFilter((entry) => entry.metadata.docId === docId)

  // 删除文件
  const filePath = path.join(process.cwd(), 'uploads', record.filename)
  await fs.unlink(filePath).catch(() => {})

  // 删除记录
  documents.delete(docId)
}

// 重新索引
export async function reindexDocument(docId: string) {
  const record = documents.get(docId)
  if (!record) return

  const filePath = path.join(process.cwd(), 'uploads', record.filename)
  record.status = 'indexing'

  // 删除旧向量
  await vectorStore.deleteByFilter((entry) => entry.metadata.docId === docId)

  // 重新索引
  await indexDocument(docId, filePath)
}

// 获取文档列表
export function listDocuments(): DocumentRecord[] {
  return Array.from(documents.values())
}
```

### 2.2 RAG 问答

```typescript
// lib/rag.ts
import { retrieve } from './retrieval'
import { askWithCitations } from './rag-with-citations'

interface RAGResponse {
  answer: string
  citations: {
    id: number
    source: string
    location: string
    snippet: string
  }[]
  hasRelevantInfo: boolean
}

export async function askQuestion(
  query: string,
  options: {
    topK?: number
    filter?: Record<string, any>
  } = {}
): Promise<RAGResponse> {
  const { topK = 5, filter } = options

  // 1. 检索
  const searchResults = await retrieve(query, topK, filter)

  // 2. 检查是否有相关内容
  const hasRelevantInfo = searchResults.some(r => r.similarity > 0.7)

  if (!hasRelevantInfo) {
    return {
      answer: '根据现有资料，无法找到相关信息。请确认您的问题是否与已上传的文档相关。',
      citations: [],
      hasRelevantInfo: false,
    }
  }

  // 3. 生成回答
  const response = await askWithCitations(query, searchResults)

  return {
    ...response,
    hasRelevantInfo: true,
  }
}
```

// app/api/documents/route.ts
import { NextRequest } from 'next/server'
import { uploadAndIndex, listDocuments, deleteDocument, reindexDocument } from '@/lib/document-manager'

// 上传文档
export async function POST(req: NextRequest) {
  try {
    const formData = await req.formData()
    const file = formData.get('file') as File

    if (!file) {
      return Response.json({ error: '没有上传文件' }, { status: 400 })
    }

    const record = await uploadAndIndex(file)
    return Response.json({ success: true, document: record })

  } catch (error) {
    console.error('上传失败:', error)
    return Response.json({ error: '上传失败' }, { status: 500 })
  }
}

// 获取文档列表
export async function GET() {
  const documents = listDocuments()
  return Response.json({ documents })
}

// 删除文档
export async function DELETE(req: NextRequest) {
  const { docId } = await req.json()
  await deleteDocument(docId)
  return Response.json({ success: true })
}

### 4.1 文档上传组件

```typescript
// components/DocumentUpload.tsx
'use client'

import { useState, useRef } from 'react'

interface Props {
  onUploadComplete: () => void
}

export function DocumentUpload({ onUploadComplete }: Props) {
  const [isUploading, setIsUploading] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const fileInputRef = useRef<HTMLInputElement>(null)

  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    setIsUploading(true)
    setError(null)

    try {
      const formData = new FormData()
      formData.append('file', file)

      const response = await fetch('/api/documents', {
        method: 'POST',
        body: formData,
      })

      if (!response.ok) {
        const data = await response.json()
        throw new Error(data.error || '上传失败')
      }

      onUploadComplete()
      if (fileInputRef.current) {
        fileInputRef.current.value = ''
      }

    } catch (err) {
      setError(err instanceof Error ? err.message : '上传失败')
    } finally {
      setIsUploading(false)
    }
  }

  return (
    <div className="document-upload">
      <input
        ref={fileInputRef}
        type="file"
        accept=".md,.txt,.pdf"
        onChange={handleUpload}
        disabled={isUploading}
      />
      {isUploading && <span>上传并索引中...</span>}
      {error && <span className="error">{error}</span>}
    </div>
  )
}
```

// components/DocumentList.tsx
'use client'

interface Document {
  id: string
  filename: string
  uploadedAt: string
  chunkCount: number
  status: 'indexing' | 'ready' | 'error'
}

interface Props {
  documents: Document[]
  onDelete: (docId: string) => void
  onReindex: (docId: string) => void
}

export function DocumentList({ documents, onDelete, onReindex }: Props) {
  if (documents.length === 0) {
    return <div className="empty">还没有上传文档</div>
  }

  return (
    <div className="document-list">
      {documents.map(doc => (
        <div key={doc.id} className="document-item">
          <div className="doc-info">
            <span className="doc-name">{doc.filename}</span>
            <span className="doc-meta">
              {doc.chunkCount} 个片段 · {doc.status}
            </span>
          </div>
          <div className="doc-actions">
            <button onClick={() => onReindex(doc.id)} disabled={doc.status === 'indexing'}>
              重新索引
            </button>
            <button onClick={() => onDelete(doc.id)} className="delete">
              删除
            </button>
          </div>
        </div>
      ))}
    </div>
  )
}

### 4.3 问答界面

```typescript
// components/ChatInterface.tsx
'use client'

import { useState } from 'react'

interface Message {
  role: 'user' | 'assistant'
  content: string
  citations?: {
    id: number
    source: string
    location: string
    snippet: string
  }[]
}

interface Props {
  selectedDocId?: string
}

export function ChatInterface({ selectedDocId }: Props) {
  const [messages, setMessages] = useState<Message[]>([])
  const [input, setInput] = useState('')
  const [isLoading, setIsLoading] = useState(false)

  const handleSend = async () => {
    if (!input.trim() || isLoading) return

    const userMessage: Message = { role: 'user', content: input }
    setMessages(prev => [...prev, userMessage])
    setInput('')
    setIsLoading(true)

    try {
      const response = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          message: input,
          docId: selectedDocId,
        }),
      })

      const data = await response.json()

      const assistantMessage: Message = {
        role: 'assistant',
        content: data.answer,
        citations: data.citations,
      }
      setMessages(prev => [...prev, assistantMessage])

    } catch (error) {
      console.error('问答失败:', error)
    } finally {
      setIsLoading(false)
    }
  }

  return (
    <div className="chat-interface">
      <div className="messages">
        {messages.map((msg, i) => (
          <div key={i} className={`message ${msg.role}`}>
            <div className="content">{msg.content}</div>
            {msg.citations && msg.citations.length > 0 && (
              <div className="citations">
                <h4>参考来源</h4>
                {msg.citations.map(c => (
                  <div key={c.id} className="citation">
                    [{c.id}] {c.source}
                    {c.location && ` - ${c.location}`}
                  </div>
                ))}
              </div>
            )}
          </div>
        ))}
      </div>

      <div className="input-bar">
        <input
          value={input}
          onChange={e => setInput(e.target.value)}
          onKeyDown={e => e.key === 'Enter' && handleSend()}
          placeholder="输入问题..."
          disabled={isLoading}
        />
        <button onClick={handleSend} disabled={isLoading}>
          {isLoading ? '回答中...' : '发送'}
        </button>
      </div>
    </div>
  )
}
```

## 五、主页面

```typescript
// app/page.tsx
'use client'

import { useState, useEffect } from 'react'
import { DocumentUpload } from '@/components/DocumentUpload'
import { DocumentList } from '@/components/DocumentList'
import { ChatInterface } from '@/components/ChatInterface'

interface Document {
  id: string
  filename: string
  uploadedAt: string
  chunkCount: number
  status: string
}

export default function DocQAPage() {
  const [documents, setDocuments] = useState<Document[]>([])
  const [selectedDocId, setSelectedDocId] = useState<string | undefined>()

  const fetchDocuments = async () => {
    const response = await fetch('/api/documents')
    const data = await response.json()
    setDocuments(data.documents)
  }

  useEffect(() => {
    fetchDocuments()
  }, [])

  const handleDelete = async (docId: string) => {
    await fetch('/api/documents', {
      method: 'DELETE',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ docId }),
    })
    fetchDocuments()
    if (selectedDocId === docId) {
      setSelectedDocId(undefined)
    }
  }

  const handleReindex = async (docId: string) => {
    // 触发重新索引
    fetchDocuments()
  }

  return (
    <div className="doc-qa-page">
      <h1>文档问答系统</h1>

      <div className="layout">
        <div className="sidebar">
          <DocumentUpload onUploadComplete={fetchDocuments} />

          <DocumentList
            documents={documents}
            onDelete={handleDelete}
            onReindex={handleReindex}
          />

          <div className="doc-selector">
            <label>
              <input
                type="radio"
                checked={!selectedDocId}
                onChange={() => setSelectedDocId(undefined)}
              />
              所有文档
            </label>
            {documents.filter(d => d.status === 'ready').map(doc => (
              <label key={doc.id}>
                <input
                  type="radio"
                  checked={selectedDocId === doc.id}
                  onChange={() => setSelectedDocId(doc.id)}
                />
                {doc.filename}
              </label>
            ))}
          </div>
        </div>

        <div className="main">
          <ChatInterface selectedDocId={selectedDocId} />
        </div>
      </div>
    </div>
  )
}
```

## 小结

本课综合运用了阶段二的所有知识：

1. **文档解析**：从 Markdown/TXT 提取文本
