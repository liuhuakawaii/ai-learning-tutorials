### 2.3 编写爬虫 Spider

```python
# book_scraper/spiders/books_spider.py

import scrapy
from book_scraper.items import BookItem


class BooksSpider(scrapy.Spider):
    """books.toscrape.com 全站书籍爬虫

    爬取网站上所有书籍的基本信息和详情
    """
    name = "books"
    allowed_domains = ["books.toscrape.com"]
    start_urls = ["https://books.toscrape.com/catalogue/page-1.html"]

    # 评分文字到数字的映射
    RATING_MAP = {
        "One": 1,
        "Two": 2,
        "Three": 3,
        "Four": 4,
        "Five": 5,
    }

    def parse(self, response):
        """解析列表页，提取书籍基本信息和详情页链接"""
        # 遍历当前页的每本书
        for book in response.css("article.product_pod"):
            item = BookItem()

            # 提取书名（在 title 属性中）
            item["title"] = book.css("h3 a::attr(title)").get()

            # 提取价格
            item["price"] = book.css(".price_color::text").get()

            # 提取评分（从 class 名中解析）
            rating_class = book.css("p.star-rating::attr(class)").get()
            rating_text = rating_class.replace("star-rating ", "")
            item["rating"] = self.RATING_MAP.get(rating_text, 0)

            # 提取库存状态
            item["availability"] = book.css(".availability::text").getall()
            item["availability"] = " ".join(item["availability"]).strip()

            # 构造详情页 URL 并请求
            detail_url = book.css("h3 a::attr(href)").get()
            item["detail_url"] = response.urljoin(detail_url)

            # 请求详情页，通过 meta 传递已有数据
            yield response.follow(
                detail_url,
                callback=self.parse_detail,
                meta={"item": item}
            )

        # 处理分页：查找"下一页"按钮
        next_page = response.css("li.next a::attr(href)").get()
        if next_page:
            yield response.follow(next_page, callback=self.parse)

    def parse_detail(self, response):
        """解析详情页，补充更多信息"""
        item = response.meta["item"]

        # 提取封面图片 URL
        item["image_url"] = response.urljoin(
            response.css("#product_gallery img::attr(src)").get()
        )

        # 提取 UPC
        item["upc"] = response.css(
            "table.table-striped tr:nth-child(1) td::text"
        ).get()

        # 提取分类（面包屑导航中的第三个元素）
        breadcrumbs = response.css("ul.breadcrumb li a::text").getall()
        if len(breadcrumbs) >= 3:
            item["category"] = breadcrumbs[2].strip()
        else:
            item["category"] = "Unknown"

        # 提取简介
        desc = response.css("#product_description ~ p::text").get()
        item["description"] = desc if desc else "暂无简介"

        yield item
```

**生活类比：** 这个爬虫就像一个勤快的"采购员"。先在书架上扫描每一本书的标签（列表页），然后拿起感兴趣的书翻看版权页（详情页），把所有信息都记录在采购单上（Item）。

## 三、第二步：SQLite 存储 Pipeline

### 3.1 Pipeline 代码

```python
# book_scraper/pipelines.py

import sqlite3
import os


class SQLitePipeline:
    """将爬取数据存入 SQLite 数据库

    在爬虫启动时创建数据库和表，爬取过程中逐条插入数据
    """

    def __init__(self):
        self.db_path = os.path.join(os.path.dirname(__file__), "..", "books.db")
        self.connection = None
        self.cursor = None

    def open_spider(self, spider):
        """爬虫启动时调用：创建数据库连接和表"""
        self.connection = sqlite3.connect(self.db_path)
        self.cursor = self.connection.cursor()

        # 创建书籍表
        self.cursor.execute("""
            CREATE TABLE IF NOT EXISTS books (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL,
                price TEXT,
                rating INTEGER,
                availability TEXT,
                category TEXT,
                detail_url TEXT UNIQUE,
                image_url TEXT,
                upc TEXT,
                description TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)

        # 创建分类索引，加速查询
        self.cursor.execute("""
            CREATE INDEX IF NOT EXISTS idx_category ON books(category)
        """)

        # 创建评分索引
        self.cursor.execute("""
            CREATE INDEX IF NOT EXISTS idx_rating ON books(rating)
        """)

        self.connection.commit()
        spider.logger.info(f"✅ 数据库已就绪：{self.db_path}")

    def process_item(self, item, spider):
        """每爬取到一条数据时调用：插入数据库"""
        try:
            self.cursor.execute("""
                INSERT OR IGNORE INTO books
                (title, price, rating, availability, category,
                 detail_url, image_url, upc, description)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
            """, (
                item.get("title", ""),
                item.get("price", ""),
                item.get("rating", 0),
                item.get("availability", ""),
                item.get("category", ""),
                item.get("detail_url", ""),
                item.get("image_url", ""),
                item.get("upc", ""),
                item.get("description", ""),
            ))
            self.connection.commit()
        except sqlite3.Error as e:
            spider.logger.error(f"❌ 数据库插入失败：{e}")

        return item

    def close_spider(self, spider):
        """爬虫结束时调用：关闭数据库连接"""
        if self.connection:
            self.connection.close()
            spider.logger.info("✅ 数据库连接已关闭")
```

# api/app.py

import sqlite3
import os
from flask import Flask, jsonify, request, send_from_directory

app = Flask(__name__, static_folder="../frontend", static_url_path="")

# 数据库路径
DB_PATH = os.path.join(os.path.dirname(__file__), "..", "books.db")


def get_db():
    """获取数据库连接"""
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row  # 让查询结果像字典一样使用
    return conn


# ==================== 页面路由 ====================

@app.route("/")
def index():
    """提供前端页面"""
    return send_from_directory("../frontend", "index.html")


# ==================== API 路由 ====================

@app.route("/api/books", methods=["GET"])
def get_books():
    """获取书籍列表，支持分页、搜索、筛选、排序"""
    # 获取查询参数
    page = request.args.get("page", 1, type=int)
    per_page = request.args.get("per_page", 20, type=int)
    category = request.args.get("category", "")
    min_rating = request.args.get("min_rating", 0, type=int)
    search = request.args.get("search", "")
    sort_by = request.args.get("sort", "id")
    sort_order = request.args.get("order", "asc")

    # 安全检查：限制每页最多 100 条
    per_page = min(per_page, 100)

    # 安全检查：只允许按指定字段排序
    allowed_sort_fields = ["id", "title", "price", "rating"]
    if sort_by not in allowed_sort_fields:
        sort_by = "id"

    if sort_order not in ["asc", "desc"]:
        sort_order = "asc"

    conn = get_db()
    cursor = conn.cursor()

    # 构建查询条件
    conditions = []
    params = []

    if category:
        conditions.append("category = ?")
        params.append(category)

    if min_rating > 0:
        conditions.append("rating >= ?")
        params.append(min_rating)

    if search:
        conditions.append("title LIKE ?")
        params.append(f"%{search}%")

    where_clause = ""
    if conditions:
        where_clause = "WHERE " + " AND ".join(conditions)

    # 查询总数
    count_sql = f"SELECT COUNT(*) FROM books {where_clause}"
    cursor.execute(count_sql, params)
    total = cursor.fetchone()[0]

    # 查询数据（带分页）
    offset = (page - 1) * per_page
    data_sql = f"""
        SELECT id, title, price, rating, availability,
               category, detail_url, image_url, upc, description
        FROM books {where_clause}
        ORDER BY {sort_by} {sort_order}
        LIMIT ? OFFSET ?
    """
    cursor.execute(data_sql, params + [per_page, offset])
    rows = cursor.fetchall()

    # 转换为字典列表
    books = [dict(row) for row in rows]

    conn.close()

    return jsonify({
        "data": books,
        "total": total,
        "page": page,
        "per_page": per_page,
        "total_pages": (total + per_page - 1) // per_page
    })


@app.route("/api/books/<int:book_id>", methods=["GET"])

    conn.close()

    return jsonify({
        "data": {
            "total_books": total_books,
            "avg_rating": round(avg_rating, 2),
            "total_categories": len(categories),
            "categories": categories[:10],  # 只返回前 10 个分类
            "rating_distribution": rating_dist,
            "price_range": {
                "min": round(price_row[0], 2) if price_row[0] else 0,
                "max": round(price_row[1], 2) if price_row[1] else 0,
                "avg": round(price_row[2], 2) if price_row[2] else 0,
            }
        }
    })


@app.route("/api/categories", methods=["GET"])
def get_categories():
    """获取所有分类列表"""
    conn = get_db()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT category, COUNT(*) as count
        FROM books
        GROUP BY category
        ORDER BY count DESC
    """)
    categories = [{"name": row[0], "count": row[1]} for row in cursor.fetchall()]

    conn.close()
    return jsonify({"data": categories})

const state = {
    currentPage: 1,

    search: "",
    category: "",

    sortOrder: "asc",

/**
 * 获取书籍列表
 */
async function fetchBooks() {
    const params = new URLSearchParams({
        page: state.currentPage,
        per_page: state.perPage,
        search: state.search,
        category: state.category,
        min_rating: state.minRating,
        sort: state.sortBy,
        order: state.sortOrder,
    });

    showLoading(true);

    try {
        const response = await fetch(`/api/books?${params}`);
        const result = await response.json();
        renderBooks(result.data);
        renderPagination(result.page, result.total_pages, result.total);
    } catch (error) {
        console.error("获取书籍列表失败:", error);
        renderEmpty("加载失败，请检查 API 服务是否启动");
    } finally {
        showLoading(false);
    }
}

/**
 * 渲染分页控件
 */
function renderPagination(currentPage, totalPages, total) {
    const container = document.getElementById("pagination");

    if (totalPages <= 1) {
        container.innerHTML = `<p style="color:#95a5a6">共 ${total} 条记录</p>`;
        return;
    }

    let buttons = [];

    // 上一页按钮
    buttons.push(`
        <button ${currentPage === 1 ? "disabled" : ""}
                onclick="goToPage(${currentPage - 1})">
            ← 上一页
        </button>
    `);

    // 页码按钮（最多显示 7 个）
    let startPage = Math.max(1, currentPage - 3);
    let endPage = Math.min(totalPages, startPage + 6);
    startPage = Math.max(1, endPage - 6);

    if (startPage > 1) {
        buttons.push(`<button onclick="goToPage(1)">1</button>`);
        if (startPage > 2) buttons.push(`<button disabled>...</button>`);
    }

    for (let i = startPage; i <= endPage; i++) {
        buttons.push(`
            <button class="${i === currentPage ? "active" : ""}"
                    onclick="goToPage(${i})">
                ${i}
            </button>
        `);
    }

    if (endPage < totalPages) {
        if (endPage < totalPages - 1) buttons.push(`<button disabled>...</button>`);
        buttons.push(`<button onclick="goToPage(${totalPages})">${totalPages}</button>`);
    }

    // 下一页按钮
    buttons.push(`
        <button ${currentPage === totalPages ? "disabled" : ""}
                onclick="goToPage(${currentPage + 1})">
            下一页 →
        </button>
    `);

    container.innerHTML = buttons.join("") +
        `<p style="color:#95a5a6; margin-left:16px; line-height:38px;">
            共 ${total} 条记录
        </p>`;
}

## 动手练习

### 练习 1：扩展 API 接口

```
任务：为 Flask API 添加以下功能

1. 添加 /api/books/random 端点，随机返回一本书
2. 添加价格筛选参数 min_price 和 max_price
3. 添加 /api/search 端点，支持按简介内容搜索

提示：
- 使用 SQL 的 RANDOM() 函数实现随机查询
- 价格需要先去掉英镑符号 £ 再比较
- 搜索简介时使用 LIKE '%关键词%'
```

### 练习 2：美化前端页面

```
任务：增强前端展示功能

1. 添加一个"查看详情"弹窗（点击书籍卡片后显示完整信息）
2. 添加评分分布的柱状图
3. 添加"加载更多"按钮替代分页
4. 添加暗色主题切换功能

提示：
- 弹窗可以用 CSS 的 position: fixed 实现
- 柱状图可以用纯 CSS div 或 Chart.js 库
- "加载更多"需要维护一个数据累加数组
- 暗色主题可以通过 CSS 变量 + class 切换实现
```

### 练习 3：增量更新

```
任务：实现数据增量更新

1. 修改爬虫，让它在运行前检查数据库中已有的 URL
2. 只爬取数据库中不存在的新书
3. 更新已有书籍的价格和库存信息
4. 在 API 中添加一个 /api/last-update 端点显示最后更新时间

提示：
- 在 open_spider 中查询数据库获取已有 URL 集合
- 在 parse 方法中先检查 URL 是否已存在
- 使用 UPDATE 语句更新已有记录
- 在 books 表中添加 updated_at 字段
```
