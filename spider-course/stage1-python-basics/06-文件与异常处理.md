## 一、文件读写基础

### 1.1 为什么要学文件操作？

爬虫抓到的数据要保存到文件里，配置信息要从文件里读取。文件操作是爬虫开发的基本功。

```
┌──────────────────────────────────────────────┐
│              文件操作流程                      │
│                                              │
│  读取文件：                                    │
│  ┌──────┐    ┌──────────┐    ┌────────────┐  │
│  │ 文件 │──→ │ open()   │──→ │ 读取内容    │  │
│  └──────┘    └──────────┘    └────────────┘  │
│                                              │
│  写入文件：                                    │
│  ┌────────────┐    ┌──────────┐    ┌──────┐  │
│  │ 准备数据   │──→ │ open()   │──→ │ 文件 │  │
│  └────────────┘    └──────────┘    └──────┘  │
└──────────────────────────────────────────────┘
```

### 1.2 读取文件

```python
# 方式一：传统方式（需要手动关闭文件）
f = open("test.txt", "r", encoding="utf-8")
content = f.read()
print(content)
f.close()  # 必须关闭！否则资源泄露

# 方式二：with 语句（推荐！自动关闭文件）
with open("test.txt", "r", encoding="utf-8") as f:
    content = f.read()
    print(content)
# 离开 with 块后，文件自动关闭
```

### 1.3 三种读取方式

```python
# 假设 test.txt 内容如下：
# 第一行
# 第二行
# 第三行

# 方式1：read() - 一次性读取全部内容
with open("test.txt", "r", encoding="utf-8") as f:
    content = f.read()
    print(content)
    # 第一行
    # 第二行
    # 第三行

# 方式2：readline() - 每次读取一行
with open("test.txt", "r", encoding="utf-8") as f:
    line1 = f.readline()  # "第一行\n"
    line2 = f.readline()  # "第二行\n"
    line3 = f.readline()  # "第三行\n"
    print(line1.strip())  # strip() 去掉末尾换行符

# 方式3：readlines() - 读取所有行，返回列表
with open("test.txt", "r", encoding="utf-8") as f:
    lines = f.readlines()
    print(lines)  # ['第一行\n', '第二行\n', '第三行\n']

# 方式4（最常用）：直接遍历文件对象
with open("test.txt", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())
```

```
┌──────────────────────────────────────────────┐
│           读取方式对比                         │
│                                              │
│  read()      → 一次读全部，适合小文件          │
│  readline()  → 一次读一行，适合逐行处理        │
│  readlines() → 返回列表，适合需要索引的场景    │
│  for line in f → 逐行遍历，最省内存            │
│                                              │
│  大文件推荐：for line in f                     │
│  小文件推荐：read()                           │
└──────────────────────────────────────────────┘
```

### 1.4 写入文件

```python
# 写入模式 "w"：覆盖写入（文件不存在则创建）
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("第一行\n")
    f.write("第二行\n")

# 追加模式 "a"：在文件末尾追加（文件不存在则创建）
with open("output.txt", "a", encoding="utf-8") as f:
    f.write("追加的一行\n")

# writelines()：写入多行
lines = ["苹果\n", "香蕉\n", "橙子\n"]
with open("fruits.txt", "w", encoding="utf-8") as f:
    f.writelines(lines)
```

## 二、路径处理

### 2.1 os.path 模块

```python
import os

# 获取当前工作目录
cwd = os.getcwd()
print(f"当前目录：{cwd}")

# 路径拼接（跨平台兼容）
path = os.path.join("folder", "subfolder", "file.txt")
print(path)  # folder/subfolder/file.txt（Linux/Mac）
             # folder\subfolder\file.txt（Windows）

# 获取文件名和扩展名
file_path = "/home/user/data/report.csv"
print(os.path.basename(file_path))  # report.csv
print(os.path.dirname(file_path))   # /home/user/data
print(os.path.splitext(file_path))  # ('/home/user/data/report', '.csv')

# 检查路径
print(os.path.exists("/some/path"))     # 是否存在
print(os.path.isfile("/some/path"))     # 是否是文件
print(os.path.isdir("/some/path"))      # 是否是目录

# 获取文件大小
size = os.path.getsize("test.txt")
print(f"文件大小：{size} 字节")
```

### 2.2 pathlib 模块（更现代的方式）

```python
from pathlib import Path

# 创建路径对象
p = Path("data") / "reports" / "2024" / "summary.txt"
print(p)  # data/reports/2024/summary.txt

# 常用操作
p = Path("test.txt")
print(p.exists())      # 是否存在
print(p.is_file())     # 是否是文件
print(p.name)          # 文件名：test.txt
print(p.stem)          # 不含扩展名：test
print(p.suffix)        # 扩展名：.txt
print(p.parent)        # 父目录

# 遍历目录
current_dir = Path(".")
for item in current_dir.iterdir():
    if item.is_file():
        print(f"文件：{item.name}")
    elif item.is_dir():
        print(f"目录：{item.name}")

# 查找文件
for txt_file in current_dir.glob("*.txt"):
    print(f"找到：{txt_file}")

# 递归查找（包括子目录）
for py_file in current_dir.rglob("*.py"):
    print(f"找到：{py_file}")
```

## 三、with 语句详解

### 3.1 上下文管理器

`with` 语句确保资源被正确释放，即使发生异常也不怕。

```javascript
// JavaScript 中你可能会这样做：
let connection;
try {
    connection = getConnection();
    // 使用 connection
} finally {
    if (connection) {
        connection.close();  // 确保关闭
    }
}
```

```python
# Python 的 with 语句更优雅
with open("test.txt", "r", encoding="utf-8") as f:
    content = f.read()
    # 即使这里发生异常，文件也会被自动关闭
# 离开 with 块，文件自动关闭
```

### 3.2 为什么 with 比 try/finally 好？

```python
# ❌ 不推荐：手动管理资源
f = open("test.txt", "r", encoding="utf-8")
try:
    content = f.read()
    # 如果这里出错，close() 仍然会执行
finally:
    f.close()

# ✅ 推荐：使用 with
with open("test.txt", "r", encoding="utf-8") as f:
    content = f.read()
    # 自动关闭，代码更简洁
```

```
┌──────────────────────────────────────────────┐
│          with 语句执行流程                    │
│                                              │
│  with open(...) as f:                        │
│      │                                       │
│      ▼                                       │
│  ┌──────────────┐                            │
│  │ 调用 __enter__│ ← 打开文件，返回文件对象   │
│  └──────────────┘                            │
│      │                                       │
│      ▼                                       │
│  ┌──────────────┐                            │
│  │ 执行 with 块 │ ← 你的代码                 │
│  │ 中的代码     │                            │
│  └──────────────┘                            │
│      │                                       │
│      ▼                                       │
│  ┌──────────────┐                            │
│  │ 调用 __exit__│ ← 关闭文件，释放资源        │
│  │ （即使异常） │                            │
│  └──────────────┘                            │
└──────────────────────────────────────────────┘
```

## 四、CSV 和 JSON 文件处理

### 4.1 CSV 文件读写

CSV（逗号分隔值）是最常见的数据交换格式之一：

```python
import csv

# ===== 读取 CSV =====
# 假设 data.csv 内容：
# 姓名,年龄,城市
# 张三,25,北京
# 李四,30,上海
# 王五,28,深圳

# 方式一：基本读取
with open("data.csv", "r", encoding="utf-8") as f:
    reader = csv.reader(f)
    header = next(reader)  # 读取表头
    print(f"表头：{header}")
    for row in reader:
        print(row)  # 每行是一个列表
        # ['张三', '25', '北京']

# 方式二：字典读取（推荐！列名更清晰）
with open("data.csv", "r", encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(f"{row['姓名']}，{row['年龄']}岁，住在{row['城市']}")
```

```python
import csv

# ===== 写入 CSV =====
# 方式一：基本写入
with open("output.csv", "w", encoding="utf-8", newline="") as f:
    writer = csv.writer(f)
    writer.writerow(["姓名", "年龄", "城市"])  # 写表头
    writer.writerow(["张三", 25, "北京"])
    writer.writerow(["李四", 30, "上海"])

# 方式二：字典写入
data = [
    {"姓名": "张三", "年龄": 25, "城市": "北京"},
    {"姓名": "李四", "年龄": 30, "城市": "上海"},
    {"姓名": "王五", "年龄": 28, "城市": "深圳"},
]

with open("output.csv", "w", encoding="utf-8", newline="") as f:
    fieldnames = ["姓名", "年龄", "城市"]
    writer = csv.DictWriter(f, fieldnames=fieldnames)
    writer.writeheader()  # 写表头
    writer.writerows(data)  # 写多行数据
```

### 4.2 JSON 文件读写

JSON 是 Web 开发中最常用的数据格式，爬虫中经常用到：

```python
import json

# ===== 读取 JSON 文件 =====
with open("data.json", "r", encoding="utf-8") as f:
    data = json.load(f)  # 从文件读取
    print(data)

# JSON 字符串 → Python 对象
json_str = '{"name": "张三", "age": 25, "skills": ["Python", "爬虫"]}'
data = json.loads(json_str)
print(data["name"])     # 张三
print(data["skills"])   # ['Python', '爬虫']
```

```python
import json

# ===== 写入 JSON 文件 =====
data = {
    "users": [
        {"name": "张三", "age": 25, "city": "北京"},
        {"name": "李四", "age": 30, "city": "上海"},
    ],
    "total": 2
}

# 写入文件
with open("output.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)
    # ensure_ascii=False: 保证中文正常显示
    # indent=2: 格式化缩进，方便阅读

# Python 对象 → JSON 字符串
json_str = json.dumps(data, ensure_ascii=False, indent=2)
print(json_str)
```

## 五、异常处理

### 5.1 为什么要处理异常？

爬虫运行时会遇到各种意外：网络超时、文件不存在、数据格式错误……如果不处理异常，程序直接崩溃。

```
┌──────────────────────────────────────────────┐
│           爬虫常见异常场景                     │
│                                              │
│  网络请求 → 超时、连接失败、404               │
│  文件操作 → 文件不存在、权限不足              │
│  数据解析 → 格式错误、字段缺失                │
│  数据转换 → 类型不匹配、值无效                │
│                                              │
│  不处理：程序崩溃，爬虫中断                   │
│  处理了：记录错误，继续爬下一个               │
└──────────────────────────────────────────────┘
```

### 5.2 try/except 基础

```python
# JavaScript 的 try/catch
# try {
#     const data = JSON.parse(input);
# } catch (error) {
#     console.log("解析错误：" + error.message);
# }

# Python 的 try/except
try:
    data = json.loads("这不是JSON")
except json.JSONDecodeError as e:
    print(f"JSON 解析错误：{e}")

# 捕获多种异常
try:
    result = 10 / 0
except ZeroDivisionError:
    print("不能除以零！")
except TypeError:
    print("类型错误！")

# 捕获所有异常（不推荐过度使用）
try:
    # 某些可能出错的操作
    pass
except Exception as e:
    print(f"发生了错误：{e}")
```

### 5.3 try/except/else/finally

```python
def read_json_file(filepath):
    """读取 JSON 文件的完整示例"""
    try:
        # 尝试执行的代码
        with open(filepath, "r", encoding="utf-8") as f:
            data = json.load(f)
    except FileNotFoundError:
        # 文件不存在
        print(f"错误：找不到文件 {filepath}")
        return None
    except json.JSONDecodeError as e:
        # JSON 格式错误
        print(f"错误：JSON 格式不正确 - {e}")
        return None
    except PermissionError:
        # 权限不足
        print(f"错误：没有权限读取文件 {filepath}")
        return None
    else:
        # 没有异常时执行
        print(f"成功读取文件 {filepath}")
        return data
    finally:
        # 无论如何都会执行（清理资源）
        print("读取操作完成")

# 测试
result = read_json_file("config.json")
```

```
┌──────────────────────────────────────────────┐
│        try/except/else/finally 流程           │
│                                              │

│      │                                       │

│  except: 捕获并处理异常                       │
```

### 5.4 常见异常类型

```python
# FileNotFoundError: 文件不存在
try:
    with open("不存在的文件.txt", "r") as f:
        content = f.read()
except FileNotFoundError:
    print("文件不存在！")

# ValueError: 值不合法
try:
    num = int("abc")
except ValueError:
    print("无法转换为整数！")

# KeyError: 字典键不存在
data = {"name": "张三"}
try:
    age = data["age"]
except KeyError:
    print("键 'age' 不存在！")

# TypeError: 类型不匹配
try:
    result = "hello" + 123
except TypeError:
    print("字符串和数字不能相加！")

# IndexError: 索引越界
lst = [1, 2, 3]
try:
    value = lst[10]
except IndexError:
    print("索引超出范围！")
```

## 六、自定义异常

### 6.1 为什么要自定义异常？

内置异常有时候不够精确，自定义异常能让错误信息更有意义：

```python
# 自定义异常类
class SpiderError(Exception):
    """爬虫异常基类"""
    pass

class NetworkError(SpiderError):
    """网络请求异常"""
    pass

class ParseError(SpiderError):
    """数据解析异常"""
    pass

class DataError(SpiderError):
    """数据异常"""
    pass

# 使用自定义异常
def fetch_page(url):
    if not url.startswith("http"):
        raise NetworkError(f"无效的URL：{url}")
    # 模拟网络请求
    return "<html>...</html>"

def parse_data(html):
    if not html:
        raise ParseError("HTML内容为空")
    return {"title": "示例"}

# 使用
try:
    html = fetch_page("ftp://invalid")
    data = parse_data(html)
except NetworkError as e:
    print(f"网络错误：{e}")
except ParseError as e:
    print(f"解析错误：{e}")
except SpiderError as e:
    print(f"爬虫错误：{e}")
```

### 6.2 raise 主动抛出异常

```python
def validate_age(age):
    """验证年龄"""
    if not isinstance(age, int):
        raise TypeError(f"年龄必须是整数，收到：{type(age)}")
    if age < 0 or age > 150:
        raise ValueError(f"年龄必须在 0-150 之间，收到：{age}")
    return True

# 使用
try:
    validate_age("abc")
except TypeError as e:
    print(e)  # 年龄必须是整数，收到：<class 'str'>

try:
    validate_age(200)
except ValueError as e:
    print(e)  # 年龄必须在 0-150 之间，收到：200
```

## 七、爬虫中的异常处理最佳实践

### 7.1 文件操作异常处理

```python
import json
from pathlib import Path

def safe_save_json(data, filepath):
    """安全保存 JSON 数据"""
    try:
        # 确保目录存在
        Path(filepath).parent.mkdir(parents=True, exist_ok=True)

        # 先写入临时文件，成功后再重命名（防止写入中断导致文件损坏）
        temp_path = filepath + ".tmp"
        with open(temp_path, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

        # 重命名（原子操作）
        import os
        os.replace(temp_path, filepath)
        return True

    except PermissionError:
        print(f"错误：没有权限写入 {filepath}")
        return False
    except OSError as e:
        print(f"错误：文件操作失败 - {e}")
        return False

def safe_load_json(filepath):
    """安全加载 JSON 数据"""
    if not Path(filepath).exists():
        print(f"文件不存在：{filepath}")
        return None

    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return json.load(f)
    except json.JSONDecodeError as e:
        print(f"JSON 格式错误：{e}")
        return None
    except Exception as e:
        print(f"读取失败：{e}")
        return None
```

### 7.2 批量处理中的异常处理

```python
import os
import json

def process_files(directory):
    """批量处理文件，记录错误但不中断"""
    results = {"success": [], "failed": []}

    for filename in os.listdir(directory):
        filepath = os.path.join(directory, filename)

        try:
            # 尝试处理每个文件
            with open(filepath, "r", encoding="utf-8") as f:
                data = json.load(f)
            results["success"].append(filename)

        except json.JSONDecodeError:
            results["failed"].append({
                "file": filename,
                "error": "JSON 格式错误"
            })
        except PermissionError:
            results["failed"].append({
                "file": filename,
                "error": "权限不足"
            })
        except Exception as e:
            results["failed"].append({
                "file": filename,
                "error": str(e)
            })

    return results
```

```python
from pathlib import Path

CONFIG_FILE = "spider_config.json"

    "max_retries": 3,
    "timeout": 30,
    "headers": {
        "User-Agent": "Mozilla/5.0"
    },
    "save_path": "./data"
}

def load_config():
    """加载配置，不存在则创建默认配置"""

        with open(CONFIG_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    else:

    """保存配置"""
```

**练习3：文件整理脚本**

写一个脚本，统计指定目录下各类型文件的数量：

```python
import os
from pathlib import Path

def count_files_by_extension(directory):
    """统计目录下各类型文件数量"""
    stats = {}

    for item in Path(directory).rglob("*"):
        if item.is_file():
            ext = item.suffix.lower() or "无扩展名"
            stats[ext] = stats.get(ext, 0) + 1

    return stats

def print_stats(stats, directory):
    """打印统计结果"""
    print(f"\n目录 {directory} 的文件统计：")
    print("-" * 30)

    # 按数量排序
    sorted_stats = sorted(stats.items(), key=lambda x: x[1], reverse=True)

    total = 0
    for ext, count in sorted_stats:
        print(f"  {ext:15} {count:>5} 个")
        total += count

    print("-" * 30)
    print(f"  {'总计':15} {total:>5} 个")

# 使用
if __name__ == "__main__":
    target = input("请输入目录路径（直接回车为当前目录）：").strip() or "."
    if Path(target).is_dir():
        stats = count_files_by_extension(target)
        print_stats(stats, target)
    else:
        print("目录不存在！")
```
