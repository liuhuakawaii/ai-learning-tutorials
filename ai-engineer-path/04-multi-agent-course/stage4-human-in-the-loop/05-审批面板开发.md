## 一、审批面板设计

```python
import streamlit as st
import requests

def approval_dashboard():
    """审批面板"""
    st.title("Agent 审批中心")
    
    # 获取待审批任务
    pending_tasks = get_pending_tasks()
    
    # 统计
    col1, col2, col3 = st.columns(3)
    col1.metric("待审批", len([t for t in pending_tasks if t["status"] == "pending"]))
    col2.metric("今日已审批", get_today_approved_count())
    col3.metric("平均审批时间", get_avg_approval_time())
    
    # 待审批列表
    st.header("待审批任务")
    
    for task in pending_tasks:
        with st.expander(f"🔴 {task['title']}", expanded=False):
            st.write(f"**任务**: {task['task']}")
            st.write(f"**提出者**: {task['agent']}")
            st.write(f"**方案**: {task['proposed_action']}")
            st.write(f"**风险等级**: {task['risk_level']}")
            st.write(f"**创建时间**: {task['created_at']}")
            
            # 审批操作
            col1, col2, col3 = st.columns(3)
            
            with col1:
                if st.button("✅ 批准", key=f"approve_{task['id']}"):
                    approve_task(task['id'])
                    st.success("已批准")
                    st.rerun()
            
            with col2:
                if st.button("❌ 拒绝", key=f"reject_{task['id']}"):
                    reject_task(task['id'], "不符合要求")
                    st.error("已拒绝")
                    st.rerun()
            
            with col3:
                if st.button("💬 修改", key=f"modify_{task['id']}"):
                    st.session_state[f"modifying_{task['id']}"] = True
            
            # 修改界面
            if st.session_state.get(f"modifying_{task['id']}"):
                modification = st.text_area("修改建议", key=f"mod_{task['id']}")
                if st.button("提交修改", key=f"submit_mod_{task['id']}"):
                    modify_task(task['id'], modification)
                    st.info("已提交修改建议")
                    st.rerun()
```

## 二、审批详情页

```python
def task_detail_page(task_id: str):
    """任务详情页"""
    task = get_task_detail(task_id)
    
    st.title(f"任务详情: {task_id}")
    
    # 基本信息
    st.header("基本信息")
    st.json(task)
    
    # 执行历史
    st.header("执行历史")
    for step in task["execution_history"]:
        st.write(f"- {step['timestamp']}: {step['action']}")
    
    # 相关上下文
    st.header("相关上下文")
    st.write(task["context"])
    
    # 审批操作
    st.header("审批操作")
    
    approval_comment = st.text_area("审批意见")
    
    col1, col2 = st.columns(2)
    with col1:
        if st.button("批准"):
            approve_task(task_id, approval_comment)
            st.success("已批准")
    with col2:
        if st.button("拒绝"):
            reject_task(task_id, approval_comment)
            st.error("已拒绝")
```

```python
def approval_history_page():
    """审批历史页"""
    st.title("审批历史")
    
    # 筛选
    date_range = st.date_input("日期范围", [])
    status_filter = st.selectbox("状态", ["全部", "已批准", "已拒绝"])
    
    # 获取历史
    history = get_approval_history(date_range, status_filter)
    
    # 展示
    for record in history:
        status_icon = "✅" if record["status"] == "approved" else "❌"
        st.write(f"{status_icon} {record['task_title']} - {record['approved_at']}")
```

## 四、通知集成

```python
class ApprovalNotification:
    """审批通知"""
    
    def __init__(self):
        self.channels = {
            "email": EmailChannel(),
            "slack": SlackChannel(),
            "webhook": WebhookChannel()
        }
    
    def notify_pending_approval(self, task: dict):
        """通知待审批"""
        message = f"新任务需要审批: {task['title']}"
        
        for channel in self.channels.values():
            channel.send(message)
```
