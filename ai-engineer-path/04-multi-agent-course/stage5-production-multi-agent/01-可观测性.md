class MultiAgentTracer:
    """多 Agent 追踪器"""
    
    def __init__(self):
        self.traces = {}
    
    def start_trace(self, trace_id: str, task: str):
        """开始追踪"""
        self.traces[trace_id] = {
            "id": trace_id,
            "task": task,
            "spans": [],
            "start_time": datetime.now().isoformat()
        }
    
    def add_span(self, trace_id: str, agent_name: str, action: str, result: str):
        """添加 Span"""
        self.traces[trace_id]["spans"].append({
            "agent": agent_name,
            "action": action,
            "result": result[:100],
            "timestamp": datetime.now().isoformat()
        })
    
    def end_trace(self, trace_id: str):
        """结束追踪"""
        self.traces[trace_id]["end_time"] = datetime.now().isoformat()

## 二、Agent 日志

```python
class AgentLogger:
    """Agent 日志"""
    
    def __init__(self, agent_name: str):
        self.agent_name = agent_name
        self.logger = structlog.get_logger()
    
    def log_execution(self, task: str, result: str, duration: float):
        """记录执行日志"""
        self.logger.info(
            "agent_execution",
            agent=self.agent_name,
            task=task[:100],
            result=result[:100],
            duration=duration
        )
    
    def log_error(self, task: str, error: str):
        """记录错误日志"""
        self.logger.error(
            "agent_error",
            agent=self.agent_name,
            task=task[:100],
            error=error
        )
```

## 三、多 Agent 指标

```python
class MultiAgentMetrics:
    """多 Agent 指标"""
    
    def __init__(self):
        self.agent_metrics = {}
    
    def record_agent_execution(self, agent_name: str, duration: float, success: bool):
        """记录 Agent 执行指标"""
        if agent_name not in self.agent_metrics:
            self.agent_metrics[agent_name] = {
                "total_executions": 0,
                "success_count": 0,
                "total_duration": 0
            }
        
        metrics = self.agent_metrics[agent_name]
        metrics["total_executions"] += 1
        metrics["total_duration"] += duration
        if success:
            metrics["success_count"] += 1
    
    def get_agent_summary(self, agent_name: str) -> dict:
        """获取 Agent 摘要"""
        metrics = self.agent_metrics.get(agent_name, {})
        return {
            "total_executions": metrics.get("total_executions", 0),
            "success_rate": metrics.get("success_count", 0) / metrics.get("total_executions", 1),
            "avg_duration": metrics.get("total_duration", 0) / metrics.get("total_executions", 1)
        }
```

def multi_agent_dashboard():

    # 执行追踪

    traces = get_recent_traces()
