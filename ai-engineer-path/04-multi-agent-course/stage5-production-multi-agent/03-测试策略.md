# 03 测试策略——多 Agent 系统的单元测试 / 集成测试 / 端到端测试

> 多 Agent 系统的测试比单 Agent 更复杂。分层测试是关键。

## 学习目标

- 掌握多 Agent 系统的测试策略
- 理解单元测试、集成测试、端到端测试
- 学会设计和实现多 Agent 测试

---

## 一、测试金字塔

```
多 Agent 测试金字塔：

        /\
       /  \  端到端测试（少量）
      /    \
     /──────\  集成测试（适量）
    /        \
   /──────────\  单元测试（大量）
```

---

## 二、单元测试

```python
import pytest

class TestAgent:
    """Agent 单元测试"""
    
    def test_researcher_agent(self):
        """测试研究 Agent"""
        agent = ResearchAgent("test", MockLLM())
        result = agent.execute("测试任务")
        assert result is not None
        assert len(result) > 0
    
    def test_analyst_agent(self):
        """测试分析 Agent"""
        agent = AnalystAgent("test", MockLLM())
        result = agent.execute("测试数据")
        assert "分析" in result
    
    def test_writer_agent(self):
        """测试写作 Agent"""
        agent = WriterAgent("test", MockLLM())
        result = agent.execute("测试主题")
        assert len(result) > 100
```

## 三、集成测试

```python
class TestMultiAgentIntegration:
    """多 Agent 集成测试"""
    
    def test_sequential_pipeline(self):
        """测试顺序流水线"""
        pipeline = AgentPipeline()
        pipeline.add_step("research", ResearchAgent("research", MockLLM()))
        pipeline.add_step("analysis", AnalystAgent("analysis", MockLLM()))
        pipeline.add_step("writing", WriterAgent("writing", MockLLM()))
        
        result = pipeline.run("测试任务")
        assert result["result"] is not None
    
    def test_supervisor_orchestration(self):
        """测试 Supervisor 编排"""
        supervisor = SupervisorOrchestrator(MockLLM(), {
            "researcher": ResearchAgent("researcher", MockLLM()),
            "analyst": AnalystAgent("analyst", MockLLM())
        })
        
        result = supervisor.run("测试任务")
        assert result["answer"] is not None
```

## 四、端到端测试

class TestEndToEnd:
    """端到端测试"""
    
    def test_full_workflow(self):
        """测试完整工作流"""
        app = build_multi_agent_app()
        
        result = app.invoke({
            "task": "分析 AI 行业趋势",
            "messages": []
        })
        
        assert result["final_output"] is not None
        assert len(result["final_output"]) > 200
    
    def test_human_in_the_loop(self):
        """测试人机协作"""
        app = build_approval_workflow()
        
        # 运行到审批点
        config = {"configurable": {"thread_id": "test"}}
        result = app.invoke({"task": "测试", "approval_status": "pending"}, config)
        
        # 模拟审批
        app.update_state(config, {"approval_status": "approved"})
        result = app.invoke(None, config)
        
        assert result["status"] == "completed"
```

```
本课核心要点：

3. 集成测试验证 Agent 间的协作
4. 端到端测试验证完整工作流
