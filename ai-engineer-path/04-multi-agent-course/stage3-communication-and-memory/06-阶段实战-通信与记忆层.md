## 二、核心实现

```python
class MultiAgentCommunicationLayer:
    """多 Agent 通信层"""
    
    def __init__(self, llm):
        self.llm = llm
        self.message_broker = MessageBroker()
        self.blackboard = Blackboard()
        self.workspace = Workspace("./workspace")
        self.short_term_memory = ShortTermMemory()
        self.long_term_memory = LongTermMemory()
        self.conflict_resolver = ConflictResolver(llm)
    
    def send_message(self, from_agent: str, to_agent: str, content: str):
        """发送消息"""
        self.message_broker.send(to_agent, {
            "from": from_agent,
            "content": content,
            "timestamp": datetime.now().isoformat()
        })
    
    def write_to_blackboard(self, key: str, value: any, author: str):
        """写入黑板"""
        self.blackboard.write(key, value, author)
    
    def store_memory(self, content: str, metadata: dict = None):
        """存储长期记忆"""
        self.long_term_memory.store(content, metadata)
    
    def retrieve_memory(self, query: str, top_k: int = 5) -> list:
        """检索记忆"""
        return self.long_term_memory.retrieve(query, top_k)
    
    def resolve_conflict(self, task: str, results: list) -> dict:
        """解决冲突"""
        return self.conflict_resolver.resolve_by_llm(task, results)
```

## 三、Agent 集成

```class CommunicatingAgent:
    """支持通信的 Agent"""
    
    def __init__(self, name: str, llm, comm_layer: MultiAgentCommunicationLayer):
        self.name = name
        self.llm = llm
        self.comm = comm_layer
    
    def execute(self, task: str) -> str:
        # 检索相关记忆
        memories = self.comm.retrieve_memory(task)
        
        # 读取黑板数据
        context = self.comm.blackboard.read("context")
        
        # 执行任务
        prompt = f"""你是 {self.name}。

任务：{task}

相关记忆：{json.dumps(memories, ensure_ascii=False)}
上下文：{context}

请完成任务。"""
        
        response = self.llm.invoke(prompt)
        result = response.content
        
        # 存储记忆
        self.comm.store_memory(f"任务：{task}\n结果：{result}", {"agent": self.name})
        
        # 写入黑板
        self.comm.write_to_blackboard(f"{self.name}_result", result, self.name)
        
        return result
```

## 四、运行示例

```python
def main():
    llm = ChatOpenAI(model="gpt-4o")
    comm = MultiAgentCommunicationLayer(llm)
    
    agents = {
        "researcher": CommunicatingAgent("researcher", llm, comm),
        "analyst": CommunicatingAgent("analyst", llm, comm),
        "writer": CommunicatingAgent("writer", llm, comm)
    }
    
    # 执行任务
    task = "分析 2024 年 AI 行业趋势"
    
    research = agents["researcher"].execute(task)
    analysis = agents["analyst"].execute(f"分析：{research}")
    report = agents["writer"].execute(f"撰写报告：{analysis}")
    
    print(report)

if __name__ == "__main__":
    main()
```
