## 1. 后端：DAG 执行引擎

先实现核心的拓扑排序和执行引擎：

```python
# backend/app/workflow/engine.py
import logging
from dataclasses import dataclass, field
from enum import Enum

logger = logging.getLogger(__name__)


class NodeStatus(str, Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    SKIPPED = "skipped"


@dataclass
class NodeExecution:
    node_id: str
    node_type: str
    status: NodeStatus = NodeStatus.PENDING
    input_data: dict = field(default_factory=dict)
    output_data: dict = field(default_factory=dict)
    error: str | None = None


class WorkflowEngine:
    def __init__(self):
        self.node_handlers: dict[str, callable] = {}

    def register_handler(self, node_type: str, handler: callable):
        self.node_handlers[node_type] = handler

    def _topological_sort(self, nodes: list[dict], edges: list[dict]) -> list[str]:
        """拓扑排序：确定节点执行顺序"""
        adj: dict[str, list[str]] = {n["id"]: [] for n in nodes}
        in_degree: dict[str, int] = {n["id"]: 0 for n in nodes}

        for edge in edges:
            source = edge.get("source", "")
            target = edge.get("target", "")
            if source in adj and target in in_degree:
                adj[source].append(target)
                in_degree[target] += 1

        queue = [nid for nid, deg in in_degree.items() if deg == 0]
        order = []

        while queue:
            node_id = queue.pop(0)
            order.append(node_id)
            for neighbor in adj.get(node_id, []):
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    queue.append(neighbor)

        if len(order) != len(nodes):
            raise ValueError("工作流存在循环依赖，无法执行")

        return order

    async def execute(self, workflow_data: dict) -> dict:
        nodes = workflow_data.get("nodes", [])
        edges = workflow_data.get("edges", [])
        variables = workflow_data.get("variables", {})

        if not nodes:
            return {"status": "empty", "executions": []}

        try:
            execution_order = self._topological_sort(nodes, edges)
        except ValueError as e:
            return {"status": "error", "error": str(e), "executions": []}

        node_map = {n["id"]: n for n in nodes}
        executions: list[NodeExecution] = []
        context: dict = {**variables}

        for node_id in execution_order:
            node = node_map.get(node_id)
            if not node:
                continue

            node_type = node.get("type", "unknown")
            exec_result = NodeExecution(
                node_id=node_id,
                node_type=node_type,
                status=NodeStatus.RUNNING,
                input_data=dict(context),
            )

            handler = self.node_handlers.get(node_type)
            if handler:
                try:
                    output = await handler(node, context)
                    exec_result.output_data = output
                    exec_result.status = NodeStatus.COMPLETED
                    context.update(output)
                except Exception as e:
                    exec_result.status = NodeStatus.FAILED
                    exec_result.error = str(e)
            else:
                exec_result.status = NodeStatus.SKIPPED

            executions.append(exec_result)

        all_ok = all(
            e.status in (NodeStatus.COMPLETED, NodeStatus.SKIPPED)
            for e in executions
        )

        return {
            "status": "completed" if all_ok else "failed",
            "executions": [
                {
                    "node_id": e.node_id,
                    "node_type": e.node_type,
                    "status": e.status.value,
                    "output_data": e.output_data,
                    "error": e.error,
                }
                for e in executions
            ],
        }
```

## 2. 后端：节点处理器注册

```python
# backend/app/workflow/handlers.py
from app.services.llm_service import LLMService


async def llm_node_handler(node: dict, context: dict) -> dict:
    """LLM 节点：调用大模型生成内容"""
    config = node.get("config", {})
    prompt_template = config.get("prompt", "")

    # 用上下文变量替换模板中的 {{variable}}
    prompt = prompt_template
    for key, value in context.items():
        prompt = prompt.replace(f"{{{{{key}}}}}", str(value))

    llm = LLMService()
    response = await llm.chat(
        messages=[{"role": "user", "content": prompt}],
        model=config.get("model"),
    )
    return {"output": response["content"]}


async def condition_node_handler(node: dict, context: dict) -> dict:
    """条件节点：根据表达式选择分支"""
    config = node.get("config", {})
    expression = config.get("expression", "true")

    # 安全的表达式求值
    try:
        result = eval(expression, {"__builtins__": {}}, context)
        return {"condition_result": bool(result)}
    except Exception:
        return {"condition_result": False}


async def tool_node_handler(node: dict, context: dict) -> dict:
    """工具节点：调用指定工具"""
    config = node.get("config", {})
    tool_name = config.get("tool_name", "")
    tool_input = config.get("input", {})

    # 替换输入中的变量引用
    resolved_input = {}
    for key, value in tool_input.items():
        if isinstance(value, str) and value.startswith("{{"):
            var_name = value.strip("{}").strip()
            resolved_input[key] = context.get(var_name, value)
        else:
            resolved_input[key] = value

    return {
        "tool_name": tool_name,
        "tool_output": f"工具 {tool_name} 执行完成（模拟）",
    }


async def approval_node_handler(node: dict, context: dict) -> dict:
    """审批节点：暂停等待人工审批"""
    config = node.get("config", {})
    return {
        "approval_status": "approved",
        "approver": "system",
        "message": config.get("message", "已自动审批"),
    }


# 注册所有处理器
def get_all_handlers() -> dict[str, callable]:
    return {
        "llm": llm_node_handler,
        "condition": condition_node_handler,
        "tool": tool_node_handler,
        "approval": approval_node_handler,
    }
```

## 3. 后端：工作流 API

```python
# backend/app/api/v1/workflows.py
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from app.core.database import get_db
from app.api.deps import get_current_user
from app.models.user import User
from app.models import Workflow
from app.workflow.engine import WorkflowEngine
from app.workflow.handlers import get_all_handlers
from pydantic import BaseModel
import uuid

router = APIRouter()

# 初始化引擎并注册处理器
engine = WorkflowEngine()
for node_type, handler in get_all_handlers().items():
    engine.register_handler(node_type, handler)


class WorkflowCreateRequest(BaseModel):
    name: str
    description: str = ""
    nodes: list = []
    edges: list = []
    variables: dict = {}


@router.post("/", status_code=201)
async def create_workflow(
    request: WorkflowCreateRequest,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    workflow = Workflow(
        id=str(uuid.uuid4()),
        name=request.name,
        description=request.description,
        nodes=request.nodes,
        edges=request.edges,
        variables=request.variables,
        created_by=user.id,
    )
    db.add(workflow)
    await db.flush()
    return workflow


@router.post("/{workflow_id}/execute")
async def execute_workflow(
    workflow_id: str,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    from sqlalchemy import select
    result = await db.execute(select(Workflow).where(Workflow.id == workflow_id))
    workflow = result.scalar_one_or_none()
    if not workflow:
        raise HTTPException(404, "工作流不存在")

    workflow_data = {
        "nodes": workflow.nodes,
        "edges": workflow.edges,
        "variables": workflow.variables,
    }

    execution_result = await engine.execute(workflow_data)

    return {
        "workflow_id": workflow.id,
        "execution_id": str(uuid.uuid4()),
        **execution_result,
    }
```

function addNode(type: string) {
  const id = `node_${++nodeIdCounter}`
  nodes.value.push({
    id,
    type,
    x: 100 + Math.random() * 300,
    y: 100 + Math.random() * 200,
    config: { label: nodeTypeLabels[type], prompt: '', model: 'gpt-4o-mini', expression: '', tool_name: '' },

function selectNode(node: any) {
  selectedNode.value = node
  showConfig.value = true

function getNodeCenter(nodeId: string) {
  const node = nodes.value.find((n) => n.id === nodeId)
  return node ? { x: node.x + 80, y: node.y + 30 } : { x: 0, y: 0 }
}

function startDrag(event: MouseEvent, node: any) {
  const startX = event.clientX - node.x
  const startY = event.clientY - node.y
  const onMove = (e: MouseEvent) => { node.x = e.clientX - startX; node.y = e.clientY - startY }
  const onUp = () => { document.removeEventListener('mousemove', onMove); document.removeEventListener('mouseup', onUp) }
  document.addEventListener('mousemove', onMove)
  document.addEventListener('mouseup', onUp)
}
