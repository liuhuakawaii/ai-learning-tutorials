## 二、完整实现

```python
class VoiceAssistant:
    """语音助手"""
    
    def __init__(self):
        self.asr = whisper.load_model("base")
        self.llm = OpenAI()
        self.conversation = []
    
    def process(self, audio_path: str) -> str:
        """处理语音"""
        # ASR
        text = self.asr.transcribe(audio_path)["text"]
        
        # LLM
        self.conversation.append({"role": "user", "content": text})
        response = self.llm.chat.completions.create(
            model="gpt-4o",
            messages=self.conversation
        )
        reply = response.choices[0].message.content
        self.conversation.append({"role": "assistant", "content": reply})
        
        # TTS
        asyncio.run(self._synthesize(reply))
        
        return reply
    
    async def _synthesize(self, text: str):
        """语音合成"""
        await edge_tts.Communicate(text, "zh-CN-XiaoxiaoNeural").save("reply.mp3")
```

## 三、Web 界面

```python
import gradio as gr

def voice_chat(audio):
    assistant = VoiceAssistant()
    reply = assistant.process(audio)
    return "reply.mp3", reply

demo = gr.Interface(
    fn=voice_chat,
    inputs=gr.Audio(sources="microphone", type="filepath"),
    outputs=[gr.Audio(type="filepath"), gr.Textbox()]
)

demo.launch()
```
