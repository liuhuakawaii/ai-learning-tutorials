  │          │          │          │        │
  ▼          ▼          ▼          ▼        ▼

## 二、核心实现

```python
class UniversalAssistant:
    """全能 AI 助手"""
    
    def __init__(self):
        self.gateway = InputGateway()
        self.agent = MultimodalAgent()
        self.rag = MultimodalRAG()
    
    def process(self, inputs: dict) -> dict:
        """处理输入"""
        # 输入处理
        processed = {}
        for key, value in inputs.items():
            processed[key] = self.gateway.process(value)
        
        # 检索增强
        if "text" in processed:
            rag_results = self.rag.ask(processed["text"]["content"])
            processed["rag"] = rag_results
        
        # Agent 处理
        result = self.agent.process(processed)
        
        return result
```

## 三、使用示例

```python
assistant = UniversalAssistant()

# 文本问答
result = assistant.process({"text": "什么是 AI？"})

# 图片分析
result = assistant.process({"image": "photo.jpg"})

# 语音对话
result = assistant.process({"audio": "question.mp3"})

# 文档问答
result = assistant.process({"document": "report.pdf", "text": "总结这份报告"})
```

## 四、Web 界面

```python
import gradio as gr

def assistant_interface(text, image, audio, document):
    assistant = UniversalAssistant()
    
    inputs = {}
    if text:
        inputs["text"] = text
    if image:
        inputs["image"] = image
    if audio:
        inputs["audio"] = audio
    if document:
        inputs["document"] = document
    
    result = assistant.process(inputs)
    return result["text_output"], result.get("audio_output")

demo = gr.Interface(
    fn=assistant_interface,
    inputs=[
        gr.Textbox(label="文本"),
        gr.Image(type="filepath", label="图片"),
        gr.Audio(type="filepath", label="语音"),
        gr.File(label="文档")
    ],
    outputs=[
        gr.Textbox(label="回答"),
        gr.Audio(label="语音回答")
    ]
)

demo.launch()
```
