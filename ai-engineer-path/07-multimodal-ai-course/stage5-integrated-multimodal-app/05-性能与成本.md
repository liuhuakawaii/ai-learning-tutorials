## 一、成本计算

```
多模态 API 成本：

GPT-4o 图片处理：
- 每张图片约 85-170 tokens
- 取决于图片分辨率

GPT-4o 音频处理：
- 每分钟约 25-50 tokens
- 取决于音频质量

总成本 = 文本 tokens + 图片 tokens + 音频 tokens
```

## 二、Token 计算

```python
def calculate_multimodal_tokens(text: str, images: list, audio_duration: float) -> dict:
    """计算多模态 Token"""
    text_tokens = count_tokens(text)
    image_tokens = len(images) * 170  # 每张图片约 170 tokens
    audio_tokens = audio_duration * 50  # 每分钟约 50 tokens
    
    return {
        "text": text_tokens,
        "images": image_tokens,
        "audio": audio_tokens,
        "total": text_tokens + image_tokens + audio_tokens
    }
```

## 四、缓存策略

```python
class MultimodalCache:
    """多模态缓存"""
    
    def __init__(self):
        self.cache = {}
    
    def get(self, key: str):
        """获取缓存"""
        return self.cache.get(key)
    
    def set(self, key: str, value, ttl: int = 3600):
        """设置缓存"""
        self.cache[key] = {
            "value": value,
            "expires": time.time() + ttl
        }
    
    def get_or_compute(self, key: str, compute_fn):
        """获取或计算"""
        cached = self.get(key)
        if cached and cached["expires"] > time.time():
            return cached["value"]
        
        value = compute_fn()
        self.set(key, value)
        return value
```

## 六、监控

```python
class PerformanceMonitor:
    """性能监控"""
    
    def __init__(self):
        self.metrics = {
            "requests": 0,
            "tokens": 0,
            "latency": [],
            "cost": 0
        }
    
    def record(self, tokens: int, latency: float, cost: float):
        """记录指标"""
        self.metrics["requests"] += 1
        self.metrics["tokens"] += tokens
        self.metrics["latency"].append(latency)
        self.metrics["cost"] += cost
    
    def get_summary(self) -> dict:
        """获取摘要"""
        return {
            "total_requests": self.metrics["requests"],
            "avg_latency": sum(self.metrics["latency"]) / len(self.metrics["latency"]),
            "total_cost": self.metrics["cost"]
        }
```
