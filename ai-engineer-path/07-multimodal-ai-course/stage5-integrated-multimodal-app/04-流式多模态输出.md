## 一、流式文本输出

```python
async def stream_text_response(question: str):
    """流式文本输出"""
    stream = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": question}],
        stream=True
    )
    
    for chunk in stream:
        if chunk.choices[0].delta.content:
            yield chunk.choices[0].delta.content
```

## 二、流式语音输出

```python
async def stream_audio_response(text: str):
    """流式语音输出"""
    communicate = edge_tts.Communicate(text, "zh-CN-XiaoxiaoNeural")
    
    async for chunk in communicate.stream():
        if chunk["type"] == "audio":
            yield chunk["data"]
```

## 三、图文混排

```python
def render_multimodal_response(text: str, images: list) -> str:
    """渲染图文混排"""
    # Markdown 格式
    content = text
    
    for i, image in enumerate(images):
        content += f"\n\n![图片{i+1}]({image})"
    
    return content
```

## 五、流式渲染

```javascript
// 流式渲染
const response = await fetch('/api/stream', {
  method: 'POST',
  body: JSON.stringify({ question })
});

const reader = response.body.getReader();
const decoder = new TextDecoder();

while (true) {
  const { done, value } = await reader.read();
  if (done) break;
  
  const text = decoder.decode(value);
  appendToChat(text);
}
```

## 六、Gradio 界面

```python
import gradio as gr

def multimodal_chat(text, image, audio):
    """多模态聊天"""
    agent = MultimodalAgent()
    
    result = agent.process({
        "text": text,
        "image": image,
        "audio": audio
    })
    
    return result["text_output"], result.get("audio_output")

demo = gr.Interface(
    fn=multimodal_chat,
    inputs=[
        gr.Textbox(),
        gr.Image(type="filepath"),
        gr.Audio(type="filepath")
    ],
    outputs=[
        gr.Textbox(),
        gr.Audio()
    ]
)

demo.launch()
```
