## 四、语音处理

```python
class AudioProcessor:
    """语音处理器"""
    
    def __init__(self):
        self.asr = whisper.load_model("base")
    
    def process(self, audio_input) -> dict:
        """处理语音"""
        # 语音识别
        text = self.asr.transcribe(audio_input)["text"]
        
        return {
            "type": "audio",
            "transcript": text,
            "metadata": self._extract_metadata(audio_input)
        }
```

## 五、文档处理

```python
class DocumentProcessor:
    """文档处理器"""
    
    def process(self, doc_input) -> dict:
        """处理文档"""
        # 解析文档
        elements = parse_document(doc_input)
        
        # 提取内容
        text = "\n".join([e.text for e in elements])
        tables = [e.text for e in elements if e.category == "Table"]
        images = [e.metadata.image_path for e in elements if e.category == "Image"]
        
        return {
            "type": "document",
            "text": text,
            "tables": tables,
            "images": images
        }
```

## 六、统一网关

```python
class InputGateway:
    """输入网关"""
    
    def __init__(self):
        self.processors = {
            "text": TextProcessor(),
            "image": ImageProcessor(),
            "audio": AudioProcessor(),
            "document": DocumentProcessor()
        }
    
    def process(self, input_data) -> dict:
        """处理输入"""
        # 识别类型
        input_type = self._identify_type(input_data)
        
        # 选择处理器
        processor = self.processors.get(input_type)
        if not processor:
            raise ValueError(f"不支持的输入类型：{input_type}")
        
        # 处理
        return processor.process(input_data)
    
    def _identify_type(self, input_data) -> str:
        """识别输入类型"""
        if isinstance(input_data, str):
            if input_data.endswith((".pdf", ".docx", ".doc")):
                return "document"
            elif input_data.endswith((".jpg", ".png", ".gif")):
                return "image"
            elif input_data.endswith((".mp3", ".wav", ".m4a")):
                return "audio"
            else:
                return "text"
        return "unknown"
```
