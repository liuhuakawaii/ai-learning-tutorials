## 二、核心实现

```python
class MultimodalKnowledgeBase:
    """多模态知识库"""
    
    def __init__(self):
        self.index = MultimodalIndex()
        self.rag = MultimodalRAG()
    
    def add_documents(self, doc_paths: list):
        """添加文档"""
        for path in doc_paths:
            self.rag.add_document(path)
    
    def query(self, question: str) -> dict:
        """查询"""
        return self.rag.ask(question)
    
    def search_images(self, query: str) -> list:
        """搜索图片"""
        return self.index.search(query, query_type="text", top_k=5)
    
    def search_texts(self, query: str) -> list:
        """搜索文本"""
        return self.index.search(query, query_type="text", top_k=5)
```

---

## 三、使用示例

```python
kb = MultimodalKnowledgeBase()

# 添加文档
kb.add_documents(["product_catalog.pdf", "user_manual.pdf"])

# 查询
result = kb.query("这款产品的安装步骤是什么？")
print(result["answer"])

# 搜索图片
images = kb.search_images("产品外观")
```

## 四、Web 界面

```python
import gradio as gr

def knowledge_base_interface(file, question):
    kb = MultimodalKnowledgeBase()
    kb.add_documents([file.name])
    result = kb.query(question)
    return result["answer"]

demo = gr.Interface(
    fn=knowledge_base_interface,
    inputs=[gr.File(), gr.Textbox()],
    outputs=gr.Textbox()
)

demo.launch()
```
