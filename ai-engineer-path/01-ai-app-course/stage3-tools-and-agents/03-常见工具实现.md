## 一、搜索工具

### 1.1 知识库搜索

```typescript
// lib/tools/search.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'
import { retrieve } from '../retrieval'

const SearchSchema = z.object({
  query: z.string().min(1).describe('搜索查询，包含关键词'),
  max_results: z.number().int().min(1).max(20).default(5).describe('最大返回结果数'),
  document_id: z.string().optional().describe('限定在某个文档中搜索'),
})

toolRegistry.register({
  name: 'search_knowledge_base',
  description: '在知识库中搜索与查询相关的文档片段。当用户询问基于已上传文档的问题时使用。',
  schema: SearchSchema,
  handler: async (args) => {
    const { query, max_results, document_id } = args

    const filter = document_id ? { docId: document_id } : undefined
    const results = await retrieve(query, max_results, filter)

    return {
      success: true,
      data: {
        results: results.map(r => ({
          content: r.content,
          source: r.metadata.source,
          similarity: r.similarity,
        })),
        total: results.length,
      },
    }
  },
})
```

### 1.2 网页搜索（调用外部 API）

```typescript
// lib/tools/web-search.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'

const WebSearchSchema = z.object({
  query: z.string().min(1).describe('搜索查询'),
  num_results: z.number().int().min(1).max(10).default(3).describe('返回结果数'),
})

toolRegistry.register({
  name: 'web_search',
  description: '在互联网上搜索信息。当需要获取实时信息或模型不知道的知识时使用。',
  schema: WebSearchSchema,
  handler: async (args) => {
    const { query, num_results } = args

    try {
      // 使用搜索 API（如 SerpAPI、Bing API 等）
      const response = await fetch(
        `https://api.search.com/search?q=${encodeURIComponent(query)}&limit=${num_results}`,
        {
          headers: {
            'Authorization': `Bearer ${process.env.SEARCH_API_KEY}`,
          },
        }
      )

      if (!response.ok) {
        throw new Error('搜索服务不可用')
      }

      const data = await response.json()

      return {
        success: true,
        data: {
          results: data.results.map((r: any) => ({
            title: r.title,
            url: r.url,
            snippet: r.snippet,
          })),
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `搜索失败: ${error}`,
      }
    }
  },
})
```

## 二、数据库查询工具

### 2.1 查询用户数据

```typescript
// lib/tools/database.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'
import { pool } from '../db'

const QueryUserSchema = z.object({
  user_id: z.string().describe('用户 ID'),
  fields: z.array(z.enum(['name', 'email', 'orders', 'preferences']))
    .describe('要查询的字段'),
})

toolRegistry.register({
  name: 'query_user_info',
  description: '查询用户信息。当需要获取用户的订单、偏好等信息时使用。',
  schema: QueryUserSchema,
  handler: async (args) => {
    const { user_id, fields } = args

    try {
      // 构建查询
      const selectFields = fields.join(', ')
      const result = await pool.query(
        `SELECT ${selectFields} FROM users WHERE id = $1`,
        [user_id]
      )

      if (result.rows.length === 0) {
        return {
          success: false,
          error: '用户不存在',
        }
      }

      return {
        success: true,
        data: result.rows[0],
      }
    } catch (error) {
      return {
        success: false,
        error: `查询失败: ${error}`,
      }
    }
  },
})
```

### 2.2 查询订单

```typescript
const QueryOrdersSchema = z.object({
  user_id: z.string().describe('用户 ID'),
  status: z.enum(['all', 'pending', 'completed', 'cancelled']).default('all'),
  limit: z.number().int().min(1).max(50).default(10),
})

toolRegistry.register({
  name: 'query_orders',
  description: '查询用户的订单列表。当用户询问订单状态、历史订单时使用。',
  schema: QueryOrdersSchema,
  handler: async (args) => {
    const { user_id, status, limit } = args

    try {
      let query = 'SELECT * FROM orders WHERE user_id = $1'
      const params: any[] = [user_id]

      if (status !== 'all') {
        query += ' AND status = $2'
        params.push(status)
      }

      query += ' ORDER BY created_at DESC LIMIT $' + (params.length + 1)
      params.push(limit)

      const result = await pool.query(query, params)

      return {
        success: true,
        data: {
          orders: result.rows,
          total: result.rows.length,
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `查询订单失败: ${error}`,
      }
    }
  },
})
```

## 三、文件读写工具

### 3.1 读取文件

```typescript
// lib/tools/file.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'
import fs from 'fs/promises'
import path from 'path'

const ReadFileSchema = z.object({
  path: z.string().describe('文件路径（相对于项目根目录）'),
  encoding: z.enum(['utf-8', 'base64']).default('utf-8'),
})

toolRegistry.register({
  name: 'read_file',
  description: '读取文件内容。当需要查看配置文件、日志文件等时使用。',
  schema: ReadFileSchema,
  handler: async (args) => {
    const { path: filePath, encoding } = args

    // 安全检查：限制只能访问特定目录
    const allowedDirs = ['uploads', 'data', 'logs']
    const normalizedPath = path.normalize(filePath)

    if (!allowedDirs.some(dir => normalizedPath.startsWith(dir))) {
      return {
        success: false,
        error: '不允许访问该路径',
      }
    }

    try {
      const content = await fs.readFile(normalizedPath, encoding)
      const stats = await fs.stat(normalizedPath)

      return {
        success: true,
        data: {
          content,
          size: stats.size,
          modified: stats.mtime,
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `读取文件失败: ${error}`,
      }
    }
  },
})
```

### 3.2 写入文件

```typescript
const WriteFileSchema = z.object({
  path: z.string().describe('文件路径'),
  content: z.string().describe('文件内容'),
  create_dirs: z.boolean().default(true).describe('是否自动创建目录'),
})

toolRegistry.register({
  name: 'write_file',
  description: '写入文件。当需要保存生成的内容、创建报告等时使用。',
  schema: WriteFileSchema,
  handler: async (args) => {
    const { path: filePath, content, create_dirs } = args

    // 安全检查
    const allowedDirs = ['uploads', 'output', 'reports']
    const normalizedPath = path.normalize(filePath)

    if (!allowedDirs.some(dir => normalizedPath.startsWith(dir))) {
      return {
        success: false,
        error: '不允许写入该路径',
      }
    }

    try {
      if (create_dirs) {
        await fs.mkdir(path.dirname(normalizedPath), { recursive: true })
      }

      await fs.writeFile(normalizedPath, content, 'utf-8')

      return {
        success: true,
        data: {
          path: normalizedPath,
          size: content.length,
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `写入文件失败: ${error}`,
      }
    }
  },
})
```

## 四、保存笔记工具

```typescript
// lib/tools/notes.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'
import { pool } from '../db'

const SaveNoteSchema = z.object({
  title: z.string().min(1).describe('笔记标题'),
  content: z.string().min(1).describe('笔记内容'),
  tags: z.array(z.string()).default([]).describe('标签'),
  source_message_id: z.string().optional().describe('来源消息 ID'),
})

toolRegistry.register({
  name: 'save_note',
  description: '保存笔记。当用户要求保存某段内容、记录重要信息时使用。',
  schema: SaveNoteSchema,
  handler: async (args) => {
    const { title, content, tags, source_message_id } = args

    try {
      const result = await pool.query(
        `INSERT INTO notes (title, content, tags, source_message_id, created_at)
         VALUES ($1, $2, $3, $4, NOW())
         RETURNING id, created_at`,
        [title, content, JSON.stringify(tags), source_message_id]
      )

      return {
        success: true,
        data: {
          note_id: result.rows[0].id,
          created_at: result.rows[0].created_at,
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `保存笔记失败: ${error}`,
      }
    }
  },
})

// 查询笔记
const QueryNotesSchema = z.object({
  query: z.string().optional().describe('搜索关键词'),
  tags: z.array(z.string()).optional().describe('按标签过滤'),
  limit: z.number().int().min(1).max(50).default(10),
})

toolRegistry.register({
  name: 'query_notes',
  description: '查询笔记。当用户询问之前保存的笔记时使用。',
  schema: QueryNotesSchema,
  handler: async (args) => {
    const { query, tags, limit } = args

    try {
      let sql = 'SELECT * FROM notes WHERE 1=1'
      const params: any[] = []
      let paramIndex = 1

      if (query) {
        sql += ` AND (title ILIKE $${paramIndex} OR content ILIKE $${paramIndex})`
        params.push(`%${query}%`)
        paramIndex++
      }

      if (tags && tags.length > 0) {
        sql += ` AND tags ?| $${paramIndex}`
        params.push(tags)
        paramIndex++
      }

      sql += ` ORDER BY created_at DESC LIMIT $${paramIndex}`
      params.push(limit)

      const result = await pool.query(sql, params)

      return {
        success: true,
        data: {
          notes: result.rows,
          total: result.rows.length,
        },
      }
    } catch (error) {
      return {
        success: false,
        error: `查询笔记失败: ${error}`,
      }
    }
  },
})
```

## 五、工具的权限边界

### 5.1 权限控制

```
┌─────────────────────────────────────────────────────────────────┐
│                    工具权限控制                                   │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  读取权限                                                        │
│  ├── ✅ 读取自己的笔记                                          │
│  ├── ✅ 读取公开的知识库                                        │
│  ├── ❌ 读取其他用户的数据                                      │
│  └── ❌ 读取系统配置文件                                        │
│                                                                 │
│  写入权限                                                        │
│  ├── ✅ 保存笔记                                                │
│  ├── ✅ 创建待办                                                │
│  ├── ❌ 修改其他用户的数据                                      │
│  └── ❌ 删除系统文件                                            │
│                                                                 │
│  调用权限                                                        │
│  ├── ✅ 发送邮件（需要确认）                                    │
│  ├── ✅ 创建日历事件（需要确认）                                │
│  ├── ❌ 删除数据（需要二次确认）                                │
│  └── ❌ 支付操作（需要验证）                                    │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

### 5.2 实现权限检查

```typescript
function checkPermission(
  userId: string,
  toolName: string,
  args: any
): { allowed: boolean; reason?: string } {
  // 读取权限
  if (toolName === 'read_file') {
    // 只能读取自己的文件
    if (!args.path.startsWith(`uploads/${userId}/`)) {
      return { allowed: false, reason: '只能读取自己的文件' }
    }
  }

  // 写入权限
  if (toolName === 'write_file') {
    if (!args.path.startsWith(`output/${userId}/`)) {
      return { allowed: false, reason: '只能写入自己的目录' }
    }
  }

  // 数据库权限
  if (toolName === 'query_user_info') {
    if (args.user_id !== userId) {
      return { allowed: false, reason: '只能查询自己的信息' }
    }
  }

  return { allowed: true }
}
```
