## 二、研究助手工具

### 2.1 搜索工具

```typescript
// lib/tools/search.ts
import { z } from 'zod'
import { toolRegistry } from '../tool-registry'

const SearchSchema = z.object({
  query: z.string().describe('搜索查询'),
  sources: z.array(z.enum(['knowledge_base', 'web', 'notes'])).default(['knowledge_base']),
  max_results: z.number().int().default(5),
})

toolRegistry.register({
  name: 'search',
  description: '在知识库、网页或笔记中搜索信息',
  schema: SearchSchema,
  handler: async (args) => {
    const { query, sources, max_results } = args
    const results: any[] = []

    for (const source of sources) {
      if (source === 'knowledge_base') {
        // 搜索知识库
        const kbResults = await searchKnowledgeBase(query, max_results)
        results.push(...kbResults.map(r => ({ ...r, source: 'knowledge_base' })))
      } else if (source === 'notes') {
        // 搜索笔记
        const noteResults = await searchNotes(query, max_results)
        results.push(...noteResults.map(r => ({ ...r, source: 'notes' })))
      }
    }

    return {
      success: true,
      data: { results: results.slice(0, max_results) },
    }
  },
})
```

### 2.2 摘要工具

```typescript
const SummarizeSchema = z.object({
  content: z.string().describe('要总结的内容'),
  max_length: z.number().int().default(200).describe('摘要最大长度'),
  style: z.enum(['bullet', 'paragraph', 'outline']).default('paragraph'),
})

toolRegistry.register({
  name: 'summarize',
  description: '对内容进行摘要总结',
  schema: SummarizeSchema,
  handler: async (args) => {
    const { content, max_length, style } = args

    // 调用模型生成摘要
    const response = await openai.responses.create({
      model: 'gpt-5.5',
      input: [
        {
          role: 'system',
          content: `请对以下内容进行摘要，格式为 ${style}，长度不超过 ${max_length} 字。`,
        },
        { role: 'user', content },
      ],
      temperature: 0.3,
    })

    return {
      success: true,
      data: { summary: response.output_text },
    }
  },
})
```

### 2.3 保存报告工具（需要确认）

```typescript
const SaveReportSchema = z.object({
  title: z.string().describe('报告标题'),
  content: z.string().describe('报告内容'),
  format: z.enum(['markdown', 'txt']).default('markdown'),
})

toolRegistry.register({
  name: 'save_report',
  description: '保存研究报告到文件',
  schema: SaveReportSchema,
  requiresConfirmation: true,
  riskLevel: 'low',
  confirmationMessage: (args) => `确定要保存报告"${args.title}"吗？`,
  handler: async (args) => {
    const { title, content, format } = args
    const filename = `${title.replace(/\s+/g, '-')}.${format}`
    const filePath = path.join(process.cwd(), 'data', 'research', filename)

    await fs.mkdir(path.dirname(filePath), { recursive: true })
    await fs.writeFile(filePath, content, 'utf-8')

    return {
      success: true,
      data: { path: filePath, filename },
    }
  },
})
```

## 三、研究 Agent

```typescript
// lib/research-agent.ts
import { Agent } from './agent'
import { toolRegistry } from './tool-registry'

export class ResearchAgent extends Agent {
  constructor(topic: string) {
    super(`研究主题：${topic}`)
  }

  async research(): Promise<ResearchResult> {
    // 第一步：制定研究计划
    this.addMessage({
      role: 'user',
      content: `请为以下主题制定研究计划，列出 3-5 个需要调研的子主题：
      
主题：${this.state.task}

然后逐一搜索资料、整理摘要，最后生成完整的研究报告。`,
    })

    // 运行 Agent
    await this.run()

    // 提取研究结果
    return this.extractResult()
  }

  private extractResult(): ResearchResult {
    const steps = this.state.steps
    const searchResults = steps
      .filter(s => s.toolCall?.name === 'search')
      .map(s => s.toolResult?.data?.results || [])
      .flat()

    const summaries = steps
      .filter(s => s.toolCall?.name === 'summarize')
      .map(s => s.toolResult?.data?.summary || '')

    return {
      topic: this.state.task,
      searchResults,
      summaries,
      report: this.state.result || '',
      steps: steps.length,
    }
  }
}

interface ResearchResult {
  topic: string
  searchResults: any[]
  summaries: string[]
  report: string
  steps: number
}
```

## 四、API Routes

### 4.1 研究任务 API

```typescript
// app/api/research/route.ts
import { NextRequest } from 'next/server'
import { ResearchAgent } from '@/lib/research-agent'
import { saveAgentState } from '@/lib/agent-store'

export async function POST(req: NextRequest) {
  const { topic } = await req.json()

  // 创建研究 Agent
  const agent = new ResearchAgent(topic)

  // 保存初始状态
  await saveAgentState(agent)

  // 开始研究（异步）
  agent.research().then(async (result) => {
    await saveAgentState(agent)
  }).catch(async (error) => {
    console.error('研究失败:', error)
    await saveAgentState(agent)
  })

  return Response.json({
    agentId: agent.getState().id,
    status: agent.getStatus(),
  })
}

// 查询研究状态
export async function GET(req: NextRequest) {
  const agentId = req.nextUrl.searchParams.get('id')
  if (!agentId) {
    return Response.json({ error: 'Missing agent ID' }, { status: 400 })
  }

  const agent = await loadAgentState(agentId)
  if (!agent) {
    return Response.json({ error: 'Agent not found' }, { status: 404 })
  }

  return Response.json({
    status: agent.getStatus(),
    state: agent.getState(),
  })
}
```

```typescript
// components/ResearchForm.tsx
'use client'

import { useState } from 'react'

interface Props {
  onStartResearch: (topic: string) => void
  isLoading: boolean
}

export function ResearchForm({ onStartResearch, isLoading }: Props) {
  const [topic, setTopic] = useState('')

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    if (topic.trim()) {
      onStartResearch(topic)
    }
  }

  return (
    <form onSubmit={handleSubmit} className="research-form">
      <h2>AI 研究助手</h2>
      <p>输入研究主题，AI 会自动搜索资料、整理摘要并生成报告。</p>

      <input
        type="text"
        value={topic}
        onChange={e => setTopic(e.target.value)}
        placeholder="输入研究主题..."
        disabled={isLoading}
      />

      <button type="submit" disabled={isLoading || !topic.trim()}>
        {isLoading ? '研究中...' : '开始研究'}
      </button>
    </form>
  )
}
```

### 5.2 研究状态展示

```typescript
// components/ResearchStatus.tsx
'use client'

interface AgentStep {
  id: string
  toolCall?: { name: string; args: any }
  toolResult?: any
  status: string
}

interface Props {
  status: string
  steps: AgentStep[]
  result?: string
}

export function ResearchStatus({ status, steps, result }: Props) {
  const stepLabels: Record<string, string> = {
    search: '搜索资料',
    summarize: '整理摘要',
    save_report: '保存报告',
  }

  return (
    <div className="research-status">
      <div className={`status ${status}`}>
        {status === 'running' ? '正在研究...' :
         status === 'done' ? '研究完成' :
         status === 'failed' ? '研究失败' : '准备中'}
      </div>

      <div className="steps">
        {steps.map((step, i) => (
          <div key={step.id} className={`step ${step.status}`}>
            <span className="step-number">{i + 1}</span>
            <span className="step-name">
              {step.toolCall ? (stepLabels[step.toolCall.name] || step.toolCall.name) : '...'}
            </span>
            <span className="step-status">
              {step.status === 'done' ? '✓' :
               step.status === 'running' ? '...' :
               step.status === 'failed' ? '✗' : '○'}
            </span>
          </div>
        ))}
      </div>

      {result && (
        <div className="result">
          <h3>研究报告</h3>
          <div className="report-content">
            {result.split('\n').map((line, i) => (
              <p key={i}>{line}</p>
            ))}
          </div>
        </div>
      )}
    </div>
  )
}
```

## 六、主页面

```typescript
// app/page.tsx
'use client'

import { useState, useEffect } from 'react'
import { ResearchForm } from '@/components/ResearchForm'
import { ResearchStatus } from '@/components/ResearchStatus'
import { ConfirmationDialog } from '@/components/ConfirmationDialog'

export default function ResearchPage() {
  const [agentId, setAgentId] = useState<string | null>(null)
  const [agentState, setAgentState] = useState<any>(null)
  const [confirmation, setConfirmation] = useState<any>(null)

  // 开始研究
  const handleStartResearch = async (topic: string) => {
    const response = await fetch('/api/research', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ topic }),
    })

    const data = await response.json()
    setAgentId(data.agentId)
  }

  // 轮询状态
  useEffect(() => {
    if (!agentId) return

    const interval = setInterval(async () => {
      const response = await fetch(`/api/research?id=${agentId}`)
      const data = await response.json()
      setAgentState(data.state)

      if (data.state.status === 'done' || data.state.status === 'failed') {
        clearInterval(interval)
      }
    }, 2000)

    return () => clearInterval(interval)
  }, [agentId])

  // 监听确认请求
  useEffect(() => {
    const eventSource = new EventSource('/api/confirmation/events')
    eventSource.onmessage = (event) => {
      setConfirmation(JSON.parse(event.data))
    }
    return () => eventSource.close()
  }, [])

  return (
    <div className="research-page">
      <ResearchForm
        onStartResearch={handleStartResearch}
        isLoading={agentState?.status === 'running'}
      />

      {agentState && (
        <ResearchStatus
          status={agentState.status}
          steps={agentState.steps}
          result={agentState.result}
        />
      )}

      {confirmation && (
        <ConfirmationDialog
          request={confirmation}
          onConfirm={async () => {
            await fetch('/api/confirmation/respond', {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify({ id: confirmation.id, confirmed: true }),
            })
            setConfirmation(null)
          }}
          onCancel={async () => {
            await fetch('/api/confirmation/respond', {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify({ id: confirmation.id, confirmed: false }),
            })
            setConfirmation(null)
          }}
        />
      )}
    </div>
  )
}
```

## 小结

本课综合运用了阶段三的所有知识：

1. **工具调用**：搜索、摘要、保存等工具
2. **多步骤任务**：计划-执行-观察循环
3. **Agent 状态机**：pending、running、done、failed
4. **人工确认**：高风险操作需要用户确认
5. **错误处理**：工具失败不中断任务
