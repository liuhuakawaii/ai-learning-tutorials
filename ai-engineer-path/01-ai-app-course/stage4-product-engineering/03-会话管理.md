```typescript
// lib/conversation.ts
export async function generateTitle(firstMessage: string): Promise<string> {
  // 从第一条消息提取标题
  if (firstMessage.length <= 50) {
    return firstMessage
  }

  // 使用模型生成标题
  const response = await openai.responses.create({
    model: 'gpt-5.5',
    input: [
      {
        role: 'system',
        content: '请用 5-10 个字概括以下内容，作为对话标题。',
      },
      { role: 'user', content: firstMessage },
    ],
    max_output_tokens: 20,
    temperature: 0,
  })

  return response.output_text || firstMessage.slice(0, 50)
}
```

## 二、会话操作 API

### 2.1 创建会话

```typescript
// app/api/conversations/route.ts
import { NextRequest } from 'next/server'
import { getCurrentUser } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { checkQuota } from '@/lib/quota'

export async function POST(req: NextRequest) {
  const user = await getCurrentUser()
  if (!user) {
    return Response.json({ error: '未登录' }, { status: 401 })
  }

  // 检查配额
  const quota = await checkQuota(user.id, 'conversation')
  if (!quota.allowed) {
    return Response.json({ error: quota.reason }, { status: 429 })
  }

  const body = await req.json()

  const conversation = await prisma.conversation.create({
    data: {
      userId: user.id,
      title: body.title || '新对话',
      model: body.model || 'gpt-5.5',
      systemPrompt: body.systemPrompt,
    },
  })

  return Response.json(conversation)
}
```

### 2.2 获取会话列表

```typescript
export async function GET(req: NextRequest) {
  const user = await getCurrentUser()
  if (!user) {
    return Response.json({ error: '未登录' }, { status: 401 })
  }

  const searchParams = req.nextUrl.searchParams
  const archived = searchParams.get('archived') === 'true'
  const search = searchParams.get('search')
  const page = parseInt(searchParams.get('page') || '1')
  const limit = parseInt(searchParams.get('limit') || '20')

  const where: any = {
    userId: user.id,
    isArchived: archived,
  }

  if (search) {
    where.OR = [
      { title: { contains: search, mode: 'insensitive' } },
      { input: { some: { content: { contains: search, mode: 'insensitive' } } } },
    ]
  }

  const [conversations, total] = await Promise.all([
    prisma.conversation.findMany({
      where,
      orderBy: [
        { isPinned: 'desc' },
        { updatedAt: 'desc' },
      ],
      skip: (page - 1) * limit,
      take: limit,
      include: {
        _count: { select: { input: true } },
      },
    }),
    prisma.conversation.count({ where }),
  ])

  return Response.json({
    conversations,
    total,
    page,
    totalPages: Math.ceil(total / limit),
  })
}
```

### 2.3 更新会话

```typescript
// app/api/conversations/[id]/route.ts
export async function PATCH(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  const user = await getCurrentUser()
  if (!user) {
    return Response.json({ error: '未登录' }, { status: 401 })
  }

  // 检查权限
  const conversation = await prisma.conversation.findUnique({
    where: { id: params.id },
  })

  if (!conversation || conversation.userId !== user.id) {
    return Response.json({ error: '无权限' }, { status: 403 })
  }

  const body = await req.json()

  const updated = await prisma.conversation.update({
    where: { id: params.id },
    data: {
      title: body.title,
      isArchived: body.isArchived,
      isPinned: body.isPinned,
      systemPrompt: body.systemPrompt,
    },
  })

  return Response.json(updated)
}
```

## 三、会话搜索

### 3.1 全文搜索

```typescript
// lib/search.ts
export async function searchConversations(
  userId: string,
  query: string,
  options: {
    includeMessages?: boolean
    limit?: number
  } = {}
) {
  const { includeMessages = true, limit = 20 } = options

  const where: any = {
    userId,
    OR: [
      { title: { contains: query, mode: 'insensitive' } },
    ],
  }

  if (includeMessages) {
    where.OR.push({
      input: {
        some: {
          content: { contains: query, mode: 'insensitive' },
        },
      },
    })
  }

  const results = await prisma.conversation.findMany({
    where,
    orderBy: { updatedAt: 'desc' },
    take: limit,
    include: {
      input: includeMessages ? {
        where: {
          content: { contains: query, mode: 'insensitive' },
        },
        take: 3,
      } : false,
    },
  })

  return results
}
```

### 3.2 搜索 API

```typescript
// app/api/search/route.ts
export async function GET(req: NextRequest) {
  const user = await getCurrentUser()
  if (!user) {
    return Response.json({ error: '未登录' }, { status: 401 })
  }

  const query = req.nextUrl.searchParams.get('q')
  if (!query) {
    return Response.json({ error: '缺少搜索关键词' }, { status: 400 })
  }

  const results = await searchConversations(user.id, query)

  return Response.json({ results })
}
```

## 四、前端组件

### 4.1 会话列表组件

```typescript
// components/ConversationList.tsx
'use client'

import { useState, useEffect } from 'react'

interface Conversation {
  id: string
  title: string
  isPinned: boolean
  isArchived: boolean
  messageCount: number
  updatedAt: string
}

interface Props {
  onSelect: (id: string) => void
  selectedId?: string
}

export function ConversationList({ onSelect, selectedId }: Props) {
  const [conversations, setConversations] = useState<Conversation[]>([])
  const [search, setSearch] = useState('')
  const [showArchived, setShowArchived] = useState(false)

  useEffect(() => {
    fetchConversations()
  }, [search, showArchived])

  const fetchConversations = async () => {
    const params = new URLSearchParams()
    if (search) params.set('search', search)
    if (showArchived) params.set('archived', 'true')

    const response = await fetch(`/api/conversations?${params}`)
    const data = await response.json()
    setConversations(data.conversations)
  }

  const handlePin = async (id: string, isPinned: boolean) => {
    await fetch(`/api/conversations/${id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ isPinned: !isPinned }),
    })
    fetchConversations()
  }

  const handleArchive = async (id: string, isArchived: boolean) => {
    await fetch(`/api/conversations/${id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ isArchived: !isArchived }),
    })
    fetchConversations()
  }

  const handleDelete = async (id: string) => {
    if (!confirm('确定要删除这个对话吗？')) return

    await fetch(`/api/conversations/${id}`, { method: 'DELETE' })
    fetchConversations()
  }

  return (
    <div className="conversation-list">
      <div className="search-bar">
        <input
          type="text"
          placeholder="搜索对话..."
          value={search}
          onChange={e => setSearch(e.target.value)}
        />
      </div>

      <div className="filter-bar">
        <button
          className={!showArchived ? 'active' : ''}
          onClick={() => setShowArchived(false)}
        >
          进行中
        </button>
        <button
          className={showArchived ? 'active' : ''}
          onClick={() => setShowArchived(true)}
        >
          已归档
        </button>
      </div>

      <div className="list">
        {conversations.map(conv => (
          <div
            key={conv.id}
            className={`conversation-item ${conv.id === selectedId ? 'selected' : ''}`}
            onClick={() => onSelect(conv.id)}
          >
            <div className="conv-title">
              {conv.isPinned && <span className="pin">📌</span>}
              {conv.title}
            </div>
            <div className="conv-meta">
              {conv.messageCount} 条消息
            </div>
            <div className="conv-actions">
              <button onClick={(e) => { e.stopPropagation(); handlePin(conv.id, conv.isPinned) }}>
                {conv.isPinned ? '取消置顶' : '置顶'}
              </button>
              <button onClick={(e) => { e.stopPropagation(); handleArchive(conv.id, conv.isArchived) }}>
                {conv.isArchived ? '取消归档' : '归档'}
              </button>
              <button onClick={(e) => { e.stopPropagation(); handleDelete(conv.id) }}>
                删除
              </button>
            </div>
          </div>
        ))}
      </div>
    </div>
  )
}
```

## 五、会话导出

```typescript
// lib/export.ts
export async function exportConversation(
  conversationId: string,
  format: 'markdown' | 'json' | 'txt'
): Promise<string> {
  const conversation = await prisma.conversation.findUnique({
    where: { id: conversationId },
    include: {
      input: {
        orderBy: { createdAt: 'asc' },
      },
    },
  })

  if (!conversation) throw new Error('对话不存在')

  switch (format) {
    case 'markdown':
      return exportAsMarkdown(conversation)
    case 'json':
      return JSON.stringify(conversation, null, 2)
    case 'txt':
      return exportAsText(conversation)
  }
}

function exportAsMarkdown(conv: any): string {
  let md = `# ${conv.title}\n\n`
  md += `创建时间：${conv.createdAt}\n\n---\n\n`

  for (const msg of conv.messages) {
    const role = msg.role === 'user' ? '👤 用户' : '🤖 AI'
    md += `### ${role}\n\n${msg.content}\n\n---\n\n`
  }

  return md
}
```
