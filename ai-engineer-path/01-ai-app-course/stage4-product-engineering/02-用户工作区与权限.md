## 一、用户认证

### 1.1 使用 NextAuth.js

```typescript
// app/api/auth/[...nextauth]/route.ts
import NextAuth from 'next-auth'
import GithubProvider from 'next-auth/providers/github'
import GoogleProvider from 'next-auth/providers/google'
import { PrismaAdapter } from '@auth/prisma-adapter'
import { prisma } from '@/lib/prisma'

export const authOptions = {
  adapter: PrismaAdapter(prisma),
  providers: [
    GithubProvider({
      clientId: process.env.GITHUB_ID!,
      clientSecret: process.env.GITHUB_SECRET!,
    }),
    GoogleProvider({
      clientId: process.env.GOOGLE_ID!,
      clientSecret: process.env.GOOGLE_SECRET!,
    }),
  ],
  callbacks: {
    async session({ session, user }) {
      session.user.id = user.id
      return session
    },
  },
}

export default NextAuth(authOptions)
```

### 1.2 获取当前用户

```typescript
// lib/auth.ts
import { getServerSession } from 'next-auth'
import { authOptions } from '@/app/api/auth/[...nextauth]/route'

export async function getCurrentUser() {
  const session = await getServerSession(authOptions)
  return session?.user
}

// 在 API Route 中使用
export async function GET(req: Request) {
  const user = await getCurrentUser()
  if (!user) {
    return Response.json({ error: '未登录' }, { status: 401 })
  }
  // ...
}
```

## 二、工作区隔离

### 2.1 工作区概念

```
┌─────────────────────────────────────────────────────────────────┐
│                    工作区隔离                                     │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  用户 A 的工作区                                                 │
│  ├── 对话 1                                                     │
│  ├── 对话 2                                                     │
│  ├── 知识库 1                                                   │
│  └── 知识库 2                                                   │
│                                                                 │
│  用户 B 的工作区                                                 │
│  ├── 对话 1                                                     │
│  ├── 对话 2                                                     │
│  └── 知识库 1                                                   │
│                                                                 │
│  用户 A 无法看到用户 B 的数据                                    │
│  用户 B 无法看到用户 A 的数据                                    │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

### 2.2 数据隔离实现

```typescript
// lib/data-access.ts
import { prisma } from './prisma'
import { getCurrentUser } from './auth'

// 获取用户的对话列表
export async function getUserConversations() {
  const user = await getCurrentUser()
  if (!user) throw new Error('未登录')

  return prisma.conversation.findMany({
    where: { userId: user.id },
    orderBy: { updatedAt: 'desc' },
  })
}

// 获取用户的知识库
export async function getUserKnowledgeBases() {
  const user = await getCurrentUser()
  if (!user) throw new Error('未登录')

  return prisma.knowledgeBase.findMany({
    where: { userId: user.id },
    orderBy: { createdAt: 'desc' },
  })
}

// 检查用户是否有权限访问某个对话
export async function canAccessConversation(
  userId: string,
  conversationId: string
): Promise<boolean> {
  const conversation = await prisma.conversation.findUnique({
    where: { id: conversationId },
  })

  return conversation?.userId === userId
}
```

### 2.3 中间件权限检查

```typescript
// lib/middleware/auth.ts
import { NextRequest, NextResponse } from 'next/server'
import { getCurrentUser } from '@/lib/auth'

export function withAuth(handler: Function) {
  return async (req: NextRequest, ...args: any[]) => {
    const user = await getCurrentUser()

    if (!user) {
      return NextResponse.json(
        { error: '请先登录' },
        { status: 401 }
      )
    }

    return handler(req, { user }, ...args)
  }
}

// 使用
export const GET = withAuth(async (req: NextRequest, { user }) => {
  const conversations = await prisma.conversation.findMany({
    where: { userId: user.id },
  })
  return NextResponse.json(conversations)
})
```

## 三、知识库权限

### 3.1 权限级别

```typescript
enum PermissionLevel {
  OWNER = 'owner',     // 所有者：完全控制
  EDITOR = 'editor',   // 编辑者：可以修改内容
  VIEWER = 'viewer',   // 查看者：只能查看
}

interface KnowledgeBasePermission {
  id: string
  knowledgeBaseId: string
  userId: string
  level: PermissionLevel
  createdAt: Date
}
```

### 3.2 权限检查

```typescript
// lib/permissions.ts
import { prisma } from './prisma'

export async function checkKnowledgeBasePermission(
  userId: string,
  knowledgeBaseId: string,
  requiredLevel: PermissionLevel
): Promise<boolean> {
  // 检查是否是所有者
  const knowledgeBase = await prisma.knowledgeBase.findUnique({
    where: { id: knowledgeBaseId },
  })

  if (!knowledgeBase) return false
  if (knowledgeBase.userId === userId) return true  // 所有者

  // 检查是否有权限
  const permission = await prisma.knowledgeBasePermission.findUnique({
    where: {
      knowledgeBaseId_userId: {
        knowledgeBaseId,
        userId,
      },
    },
  })

  if (!permission) return false

  // 检查权限级别
  const levelHierarchy = {
    [PermissionLevel.OWNER]: 3,
    [PermissionLevel.EDITOR]: 2,
    [PermissionLevel.VIEWER]: 1,
  }

  return levelHierarchy[permission.level] >= levelHierarchy[requiredLevel]
}

// 中间件
export function withKnowledgeBasePermission(requiredLevel: PermissionLevel) {
  return (handler: Function) => {
    return async (req: NextRequest, context: any) => {
      const user = await getCurrentUser()
      if (!user) {
        return NextResponse.json({ error: '未登录' }, { status: 401 })
      }

      const knowledgeBaseId = context.params?.id
      if (!knowledgeBaseId) {
        return NextResponse.json({ error: '缺少知识库 ID' }, { status: 400 })
      }

      const hasPermission = await checkKnowledgeBasePermission(
        user.id,
        knowledgeBaseId,
        requiredLevel
      )

      if (!hasPermission) {
        return NextResponse.json({ error: '没有权限' }, { status: 403 })
      }

      return handler(req, context)
    }
  }
}

// 使用
export const DELETE = withKnowledgeBasePermission(PermissionLevel.OWNER)(
  async (req: NextRequest, context: any) => {
    // 删除知识库逻辑
  }
)
```

## 四、配额限制

### 4.1 配额设计

```typescript
interface UserQuota {
  userId: string
  plan: 'free' | 'pro' | 'enterprise'
  limits: {
    conversationsPerMonth: number    // 每月对话数
    messagesPerDay: number           // 每日消息数
    tokensPerMonth: number           // 每月 token 数
    knowledgeBases: number           // 知识库数量
    documentsPerBase: number         // 每个知识库的文档数
    storageMB: number                // 存储空间（MB）
  }
  usage: {
    conversationsThisMonth: number
    messagesToday: number
    tokensThisMonth: number
    knowledgeBasesCount: number
    storageUsedMB: number
  }
}

const PLAN_LIMITS: Record<string, UserQuota['limits']> = {
  free: {
    conversationsPerMonth: 50,
    messagesPerDay: 100,
    tokensPerMonth: 100000,
    knowledgeBases: 3,
    documentsPerBase: 10,
    storageMB: 100,
  },
  pro: {
    conversationsPerMonth: 500,
    messagesPerDay: 1000,
    tokensPerMonth: 1000000,
    knowledgeBases: 20,
    documentsPerBase: 100,
    storageMB: 1000,
  },
  enterprise: {
    conversationsPerMonth: -1,  // 无限制
    messagesPerDay: -1,
    tokensPerMonth: -1,
    knowledgeBases: -1,
    documentsPerBase: -1,
    storageMB: -1,
  },
}
```

### 4.2 配额检查

```typescript
// lib/quota.ts
import { prisma } from './prisma'

export async function checkQuota(
  userId: string,
  action: 'message' | 'conversation' | 'knowledge_base' | 'document'
): Promise<{ allowed: boolean; reason?: string }> {
  const user = await prisma.user.findUnique({
    where: { id: userId },
    include: { usage: true },
  })

  if (!user) return { allowed: false, reason: '用户不存在' }

  const limits = PLAN_LIMITS[user.plan]
  const usage = user.usage

  switch (action) {
    case 'message':
      if (limits.messagesPerDay !== -1 && usage.messagesToday >= limits.messagesPerDay) {
        return { allowed: false, reason: '今日消息数已达上限' }
      }
      if (limits.tokensPerMonth !== -1 && usage.tokensThisMonth >= limits.tokensPerMonth) {
        return { allowed: false, reason: '本月 token 数已达上限' }
      }
      break

    case 'conversation':
      if (limits.conversationsPerMonth !== -1 && usage.conversationsThisMonth >= limits.conversationsPerMonth) {
        return { allowed: false, reason: '本月对话数已达上限' }
      }
      break

    case 'knowledge_base':
      if (limits.knowledgeBases !== -1 && usage.knowledgeBasesCount >= limits.knowledgeBases) {
        return { allowed: false, reason: '知识库数量已达上限' }
      }
      break
  }

  return { allowed: true }
}
```

### 4.3 配额中间件

```typescript
export function withQuota(action: string) {
  return (handler: Function) => {
    return async (req: NextRequest, context: any) => {
      const user = await getCurrentUser()
      if (!user) {
        return NextResponse.json({ error: '未登录' }, { status: 401 })
      }

      const quotaCheck = await checkQuota(user.id, action as any)
      if (!quotaCheck.allowed) {
        return NextResponse.json(
          { error: quotaCheck.reason },
          { status: 429 }
        )
      }

      return handler(req, context)
    }
  }
}
```

## 五、前端权限控制

```typescript
// components/QuotaWarning.tsx
'use client'

interface Props {
  plan: string
  usage: {
    tokensThisMonth: number
    tokenLimit: number
  }
}

export function QuotaWarning({ plan, usage }: Props) {
  const percentage = (usage.tokensThisMonth / usage.tokenLimit) * 100

  if (percentage < 80) return null

  return (
    <div className="quota-warning">
      {percentage >= 100 ? (
        <p>您本月的 token 配额已用完，请升级套餐或等待下月重置。</p>
      ) : (
        <p>您本月的 token 配额已使用 {percentage.toFixed(0)}%，请注意控制用量。</p>
      )}
      {plan === 'free' && (
        <a href="/settings/billing">升级到 Pro</a>
      )}
    </div>
  )
}
```
