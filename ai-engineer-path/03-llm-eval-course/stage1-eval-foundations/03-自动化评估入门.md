## 一、从零构建 LLM-as-Judge

### 1.1 最简实现

```python
from openai import OpenAI
from dataclasses import dataclass

client = OpenAI()

@dataclass
class EvalResult:
    score: int          # 1-5
    reasoning: str      # 评估理由
    passed: bool        # 是否通过阈值

def judge_answer(
    question: str,
    answer: str,
    reference: str = None,
    threshold: int = 3
) -> EvalResult:
    """基础版 LLM-as-Judge"""
    
    ref_section = ""
    if reference:
        ref_section = f"\n参考答案：{reference}"
    
    prompt = f"""你是一个严格的 AI 回答质量评估员。

问题：{question}
模型回答：{answer}{ref_section}

请从以下维度评估回答质量：
1. 准确性：信息是否正确
2. 完整性：是否覆盖了问题的关键方面
3. 相关性：是否切题，没有跑题

评分标准：
1 分：完全错误、不相关、或拒绝回答
2 分：有部分正确信息，但有重大错误或遗漏
3 分：基本正确，有小的不准确或不完整
4 分：正确且完整，表述清晰
5 分：优秀，准确、完整、表述专业

请严格按以下 JSON 格式输出，不要输出其他内容：
{{"score": <1-5>, "reasoning": "<简短评估理由>"}}"""

    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
        temperature=0,
        response_format={"type": "json_object"}
    )
    
    result = response.choices[0].message.content
    import json
    data = json.loads(result)
    
    return EvalResult(
        score=data["score"],
        reasoning=data["reasoning"],
        passed=data["score"] >= threshold
    )
```

## 二、评分 Prompt 的设计技巧

### 2.1 关键设计原则

```
评分 Prompt 的四要素：

1. 角色定义
   "你是一个严格的 AI 回答质量评估员"
   → 让模型进入"评估模式"而不是"聊天模式"

2. 评分维度
   明确告诉模型从哪些角度评估
   → 减少评估的随机性

3. 评分标准
   每个分数对应什么质量水平
   → 让评分有据可依

4. 输出格式
   JSON 格式，包含 score 和 reasoning
   → 方便程序解析
```

### 2.2 高级评分 Prompt 模板

```python
def build_judge_prompt(
    question: str,
    answer: str,
    context: str = None,
    reference: str = None,
    criteria: list[str] = None
) -> str:
    """构建高质量的评估 Prompt"""
    
    sections = [f"## 问题\n{question}", f"## 模型回答\n{answer}"]
    
    if context:
        sections.insert(1, f"## 参考资料\n{context}")
    if reference:
        sections.append(f"## 参考答案\n{reference}")
    
    criteria_text = ""
    if criteria:
        criteria_text = "\n".join(f"- {c}" for c in criteria)
    else:
        criteria_text = """- 准确性：信息是否正确，是否有幻觉
- 完整性：是否覆盖了问题的关键方面
- 相关性：是否切题，没有无关信息
- 清晰度：表述是否清楚、结构是否合理"""
    
    return f"""你是一个专业的 AI 回答质量评估员。请严格按照评分标准进行评估。

{chr(10).join(sections)}

## 评估维度
{criteria_text}

## 评分标准
| 分数 | 含义 | 判断标准 |
|------|------|----------|
| 1 | 差 | 完全错误、不相关、或拒绝回答 |
| 2 | 较差 | 有部分正确信息，但有重大错误 |
| 3 | 一般 | 基本正确，有明显不足 |
| 4 | 良好 | 正确且完整，表述清晰 |
| 5 | 优秀 | 准确、完整、表述专业、有额外价值 |

## 输出要求
请以 JSON 格式输出评估结果：
```json
{{
  "score": 1-5,
  "reasoning": "简短评估理由（50字以内）",
  "strengths": ["优点1", "优点2"],
  "weaknesses": ["不足1", "不足2"]
}}
```"""
```

### 2.3 避免常见陷阱

```python
# 陷阱 1：评分标准模糊
# 差：
"请评估回答的质量"
# 好：
"请从准确性、完整性、相关性三个维度评估，每个维度 1-5 分"

# 陷阱 2：没有参考信息
# 差：只给问题和回答，让模型"猜"对不对
# 好：提供参考答案或参考资料，让模型有判断依据

# 陷阱 3：没有要求输出理由
# 差：只要求输出分数
# 好：要求输出分数 + 理由（便于调试和校准）

# 陷阱 4：温度设得太高
# 差：temperature=0.7（评估结果不稳定）
# 好：temperature=0（评估结果可复现）
```

## 三、多维度评估实现

### 3.1 并行评估多个维度

```python
import asyncio
from openai import AsyncOpenAI

async_client = AsyncOpenAI()

async def evaluate_dimension(
    question: str,
    answer: str,
    dimension: str,
    criteria: str
) -> dict:
    """评估单个维度"""
    prompt = f"""评估以下回答在"{dimension}"维度的表现。

问题：{question}
回答：{answer}

评估标准：{criteria}

请输出 JSON：{{"score": 1-5, "reasoning": "理由"}}"""

    response = await async_client.chat.completions.create(
        model="gpt-4o-mini",  # 用便宜的模型做评估
        messages=[{"role": "user", "content": prompt}],
        temperature=0,
        response_format={"type": "json_object"}
    )
    
    return json.loads(response.choices[0].message.content)


async def multi_dimension_eval(question: str, answer: str) -> dict:
    """多维度并行评估"""
    dimensions = {
        "accuracy": "回答中的事实信息是否正确？是否有编造的内容？",
        "completeness": "回答是否覆盖了问题的所有关键方面？",
        "relevancy": "回答是否切题？是否有无关信息？",
        "clarity": "回答的表述是否清晰？结构是否合理？",
    }
    
    tasks = [
        evaluate_dimension(question, answer, dim, criteria)
        for dim, criteria in dimensions.items()
    ]
    
    results = await asyncio.gather(*tasks)
    
    return {
        dim: result
        for dim, result in zip(dimensions.keys(), results)
    }
```

### 3.2 综合评分计算

```python
def compute_weighted_score(eval_results: dict) -> float:
    """计算加权综合评分"""
    weights = {
        "accuracy": 0.40,
        "completeness": 0.25,
        "relevancy": 0.20,
        "clarity": 0.15,
    }
    
    total = sum(
        eval_results[dim]["score"] * weights[dim]
        for dim in weights
    )
    
    return round(total, 2)

# 使用示例
results = asyncio.run(multi_dimension_eval(
    question="什么是 RAG？",
    answer="RAG（Retrieval-Augmented Generation）是一种结合检索和生成的技术..."
))

overall = compute_weighted_score(results)
print(f"综合评分：{overall}/5.0")
print(f"各维度：{json.dumps(results, indent=2, ensure_ascii=False)}")

import pandas as pd
import numpy as np

def batch_evaluate(dataset: list[dict]) -> pd.DataFrame:
    """批量评估并生成统计报告"""
    results = []
    
    for case in dataset:
        eval_result = judge_answer(
            question=case["question"],
            answer=case["answer"],
            reference=case.get("reference")
        )
        results.append({
            "id": case["id"],
            "question": case["question"][:50],
            "score": eval_result.score,
            "passed": eval_result.passed,
            "reasoning": eval_result.reasoning,
        })
    
    df = pd.DataFrame(results)
    return df


def generate_report(df: pd.DataFrame) -> str:
    """生成评估报告"""
    report = f"""
# 评估报告

## 总体统计
- 总用例数：{len(df)}
- 通过率：{df['passed'].mean():.1%}
- 平均分：{df['score'].mean():.2f}/5.0
- 标准差：{df['score'].std():.2f}

## 分数分布
{df['score'].value_counts().sort_index().to_string()}

## 低分案例（需重点关注）
{df[df['score'] <= 2][['id', 'question', 'score', 'reasoning']].to_string()}

## 高分案例（优秀表现）
{df[df['score'] >= 4][['id', 'question', 'score', 'reasoning']].to_string()}
"""
    return report
```

## 五、评估器的校准与优化

### 5.1 人工校准流程

```python
def calibrate_judge(
    dataset: list[dict],
    human_scores: list[int],
    sample_size: int = 50
) -> dict:
    """校准评估器，对比人工评分"""
    
    # 随机采样
    indices = np.random.choice(len(dataset), min(sample_size, len(dataset)), replace=False)
    
    judge_scores = []
    for idx in indices:
        result = judge_answer(
            question=dataset[idx]["question"],
            answer=dataset[idx]["answer"],
            reference=dataset[idx].get("reference")
        )
        judge_scores.append(result.score)
    
    human_sample = [human_scores[i] for i in indices]
    
    # 计算一致性
    agreement = sum(1 for j, h in zip(judge_scores, human_sample) if abs(j - h) <= 1) / len(judge_scores)
    
    # 计算相关性
    correlation = np.corrcoef(judge_scores, human_sample)[0, 1]
    
    return {
        "agreement_rate": agreement,  # ±1 分内的一致率
        "correlation": correlation,   # 相关系数
        "avg_diff": np.mean(np.array(judge_scores) - np.array(human_sample)),  # 平均分差
        "judge_scores": judge_scores,
        "human_scores": human_sample,
    }
```

### 5.2 常见校准问题

```
问题 1：评估器偏高/偏低
  现象：judge 平均分比人工高/低 1 分以上
  解决：调整评分标准中的分数描述，或加偏移量

问题 2：评估器区分度不够
  现象：所有回答都得 3-4 分，没有 1 分和 5 分
  解决：在 Prompt 中加入更多极端案例的描述

问题 3：对特定类型回答偏差大
  现象：对简短回答系统性偏低，对冗长回答系统性偏高
  解决：在评分标准中明确"简洁但完整"的要求

问题 4：评估结果不稳定
  现象：同一回答多次评估分数不同
  解决：确保 temperature=0，使用 JSON 格式输出
```

## 六、完整评估 Pipeline

```python
class EvalPipeline:
    """完整的评估 Pipeline"""
    
    def __init__(self, dataset_path: str):
        with open(dataset_path, "r", encoding="utf-8") as f:
            self.dataset = json.load(f)
        self.results = []
    
    def run(self, get_answer_fn) -> pd.DataFrame:
        """运行评估
        
        Args:
            get_answer_fn: 接受 question 返回 answer 的函数
                           （可以是你的 RAG/Agent 系统）
        """
        for case in self.dataset:
            # 1. 获取模型回答
            answer = get_answer_fn(case["question"])
            
            # 2. 评估
            eval_result = judge_answer(
                question=case["question"],
                answer=answer,
                reference=case.get("reference")
            )
            
            # 3. 记录结果
            self.results.append({
                "id": case["id"],
                "question": case["question"],
                "answer": answer,
                "reference": case.get("reference"),
                "score": eval_result.score,
                "passed": eval_result.passed,
                "reasoning": eval_result.reasoning,
            })
        
        return pd.DataFrame(self.results)
    
    def report(self) -> str:
        """生成报告"""
        df = pd.DataFrame(self.results)
        return generate_report(df)
    
    def save(self, path: str):
        """保存结果"""
        df = pd.DataFrame(self.results)
        df.to_csv(path, index=False, encoding="utf-8-sig")
        print(f"评估结果已保存到 {path}")

# 使用示例
def my_rag_system(question: str) -> str:
    """你的 RAG 系统"""
    # 这里替换为你的实际系统

    return response.choices[0].message.content

# 运行评估

## 小结

```
本课核心要点：

1. LLM-as-Judge 的核心是设计好评分 Prompt（角色+维度+标准+格式）
2. 多维度并行评估可以提高效率和覆盖面
3. 评估器需要人工校准，确保与人工评分一致
4. 使用 temperature=0 和 JSON 格式输出确保稳定性
