## 一、A/B 测试设计

### 1.1 实验框架

```python
class ABTest:
    """A/B 测试框架"""
    
    def __init__(self, name: str, variants: list[dict]):
        self.name = name
        self.variants = variants  # [{"name": "A", "config": {...}}, ...]
        self.results = {v["name"]: [] for v in variants}
    
    def assign_variant(self, user_id: str) -> str:
        """分配变体（确定性）"""
        hash_val = hash(f"{self.name}_{user_id}") % len(self.variants)
        return self.variants[hash_val]["name"]
    
    def record_result(self, variant: str, metrics: dict):
        """记录结果"""
        self.results[variant].append(metrics)
    
    def analyze(self) -> dict:
        """分析结果"""
        analysis = {}
        
        for variant_name, results in self.results.items():
            if not results:
                continue
            
            df = pd.DataFrame(results)
            analysis[variant_name] = {
                "count": len(df),
                "avg_quality": df["quality"].mean(),
                "avg_latency": df["latency"].mean(),
                "success_rate": df["success"].mean(),
            }
        
        return analysis
    
    def check_significance(self, metric: str, confidence: float = 0.95) -> dict:
        """检查统计显著性"""
        from scipy import stats
        
        variant_a = [r[metric] for r in self.results[self.variants[0]["name"]]]
        variant_b = [r[metric] for r in self.results[self.variants[1]["name"]]]
        
        t_stat, p_value = stats.ttest_ind(variant_a, variant_b)
        
        return {
            "metric": metric,
            "t_statistic": t_stat,
            "p_value": p_value,
            "significant": p_value < (1 - confidence),
            "variant_a_mean": np.mean(variant_a),
            "variant_b_mean": np.mean(variant_b),
            "improvement": (np.mean(variant_b) - np.mean(variant_a)) / np.mean(variant_a) * 100
        }
```

def run_prompt_ab_test(test_name: str, prompts: list[str], dataset: list[dict]):
    """运行 Prompt A/B 测试"""
    
    # 创建变体
    variants = [{"name": f"prompt_{i}", "config": {"prompt": p}} for i, p in enumerate(prompts)]
    
    test = ABTest(test_name, variants)
    
    # 运行测试
    for item in dataset:
        for variant in variants:

            result = run_with_prompt(item["question"], variant["config"]["prompt"])
            
            # 评估
            quality = evaluate_quality(item["question"], result["answer"])
            
            # 记录结果

                "quality": quality,
                "latency": result["latency"],

## 二、实验类型

```
A/B 测试类型：

1. Prompt 对比
   - 不同的 System Prompt
   - 不同的指令表述
   - 不同的 Few-shot 示例

2. 模型对比
   - GPT-4o vs Claude
   - GPT-4o vs GPT-4o-mini
   - 不同版本的同一模型

3. 参数对比
   - temperature: 0 vs 0.7
   - max_tokens: 500 vs 1000
   - top_k: 3 vs 5 (RAG)

4. 策略对比
   - 不同的检索策略
   - 不同的重排序方法
   - 不同的上下文组装方式
```

## 三、样本量计算

```python
def calculate_sample_size(
    baseline_rate: float,
    minimum_detectable_effect: float,
    alpha: float = 0.05,
    power: float = 0.8
) -> int:
    """计算所需样本量"""
    from scipy import stats
    
    z_alpha = stats.norm.ppf(1 - alpha / 2)
    z_beta = stats.norm.ppf(power)
    
    p1 = baseline_rate
    p2 = baseline_rate * (1 + minimum_detectable_effect)
    
    n = (
        (z_alpha * np.sqrt(2 * p1 * (1 - p1)) +
         z_beta * np.sqrt(p1 * (1 - p1) + p2 * (1 - p2))) ** 2 /
        (p2 - p1) ** 2
    )
    
    return int(np.ceil(n))

# 示例：检测 5% 的质量提升
sample_size = calculate_sample_size(
    baseline_rate=0.8,
    minimum_detectable_effect=0.05
)
print(f"每组需要 {sample_size} 个样本")
```

## 四、实验报告

```python
def generate_ab_report(test: ABTest) -> str:
    """生成 A/B 测试报告"""
    
    analysis = test.analyze()
    
    report = f"# A/B 测试报告：{test.name}\n\n"
    
    report += "## 变体对比\n\n"
    report += "| 变体 | 样本数 | 平均质量 | 平均延迟 | 成功率 |\n"
    report += "|------|--------|----------|----------|--------|\n"
    
    for name, stats in analysis.items():
        report += f"| {name} | {stats['count']} | {stats['avg_quality']:.3f} | {stats['avg_latency']:.2f}s | {stats['success_rate']:.1%} |\n"
    
    # 显著性检验
    if len(analysis) == 2:
        sig = test.check_significance("quality")
        report += f"\n## 统计显著性\n\n"
        report += f"- p-value: {sig['p_value']:.4f}\n"
        report += f"- 显著: {'是' if sig['significant'] else '否'}\n"
        report += f"- 改进: {sig['improvement']:.1f}%\n"
    
    return report
```
