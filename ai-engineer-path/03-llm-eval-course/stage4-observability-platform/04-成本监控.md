## 一、成本监控架构

```python
class CostMonitor:
    """成本监控器"""
    
    def __init__(self, daily_limit: float, monthly_limit: float):
        self.daily_limit = daily_limit
        self.monthly_limit = monthly_limit
        self.daily_cost = 0.0
        self.monthly_cost = 0.0
        self.records = []
        
        # 模型定价
        self.pricing = {
            "gpt-4o": {"input": 2.50, "output": 10.00},
            "gpt-4o-mini": {"input": 0.15, "output": 0.60},
            "claude-3-5-sonnet": {"input": 3.00, "output": 15.00},
        }
    
    def record_usage(
        self,
        model: str,
        input_tokens: int,
        output_tokens: int,
        operation: str,
        user_id: str = None
    ):
        """记录使用量"""
        
        # 计算成本
        pricing = self.pricing.get(model, self.pricing["gpt-4o-mini"])
        cost = (
            input_tokens * pricing["input"] / 1_000_000 +
            output_tokens * pricing["output"] / 1_000_000
        )
        
        # 记录
        self.daily_cost += cost
        self.monthly_cost += cost
        self.records.append({
            "timestamp": datetime.now().isoformat(),
            "model": model,
            "input_tokens": input_tokens,
            "output_tokens": output_tokens,
            "cost": cost,
            "operation": operation,
            "user_id": user_id
        })
        
        # 检查预警
        self._check_alerts()
        
        return cost
    
    def _check_alerts(self):
        """检查预警"""
        # 日预算预警
        if self.daily_cost > self.daily_limit * 0.8:
            self._send_alert("daily", self.daily_cost, self.daily_limit)
        
        # 月预算预警
        if self.monthly_cost > self.monthly_limit * 0.8:
            self._send_alert("monthly", self.monthly_cost, self.monthly_limit)
    
    def _send_alert(self, period: str, current: float, limit: float):
        """发送预警"""
        print(f"⚠️ 成本预警 [{period}]：${current:.2f} / ${limit:.2f} ({current/limit:.0%})")
    
    def get_report(self) -> dict:
        """获取成本报告"""
        return {
            "daily_cost": self.daily_cost,
            "monthly_cost": self.monthly_cost,
            "daily_budget_remaining": self.daily_limit - self.daily_cost,
            "monthly_budget_remaining": self.monthly_limit - self.monthly_cost,
            "total_requests": len(self.records),
            "avg_cost_per_request": sum(r["cost"] for r in self.records) / len(self.records) if self.records else 0,
            "by_model": self._group_by_model(),
            "by_operation": self._group_by_operation()
        }
    
    def _group_by_model(self) -> dict:
        """按模型分组"""
        groups = {}
        for r in self.records:
            model = r["model"]
            if model not in groups:
                groups[model] = {"requests": 0, "cost": 0.0}
            groups[model]["requests"] += 1
            groups[model]["cost"] += r["cost"]
        return groups
    
    def _group_by_operation(self) -> dict:
        """按操作分组"""
        groups = {}
        for r in self.records:
            op = r["operation"]
            if op not in groups:
                groups[op] = {"requests": 0, "cost": 0.0}
            groups[op]["requests"] += 1
            groups[op]["cost"] += r["cost"]
        return groups
```

## 二、成本优化策略

```
成本优化策略：

1. 模型选择
   简单任务 → gpt-4o-mini（成本低 16 倍）
   复杂任务 → gpt-4o
   关键任务 → claude-3-5-sonnet

2. Prompt 优化
   精简 System Prompt
   减少冗余上下文
   使用更简洁的指令

3. 缓存策略
   相同问题缓存结果
   相似问题复用结果

4. 批处理
   合并多个小请求
   减少 API 调用次数

5. 流式输出
   减少首 Token 延迟
   提升用户体验

6. 限流
   按用户限制调用频率
   防止滥用
```

## 三、预算控制

```python
class BudgetController:
    """预算控制器"""
    
    def __init__(self, cost_monitor: CostMonitor):
        self.cost_monitor = cost_monitor
    
    def check_budget(self, user_id: str, operation: str) -> bool:
        """检查是否超出预算"""
        
        # 检查总预算
        if self.cost_monitor.daily_cost >= self.cost_monitor.daily_limit:
            raise BudgetExceededError("Daily budget exceeded")
        
        # 检查用户预算（可选）
        user_cost = self._get_user_daily_cost(user_id)
        user_limit = self._get_user_daily_limit(user_id)
        
        if user_cost >= user_limit:
            raise UserBudgetExceededError(f"User {user_id} daily budget exceeded")
        
        return True
    
    def _get_user_daily_cost(self, user_id: str) -> float:
        """获取用户当日成本"""
        today = datetime.now().date()
        return sum(
            r["cost"] for r in self.cost_monitor.records
            if r["user_id"] == user_id
            and datetime.fromisoformat(r["timestamp"]).date() == today
        )
    
    def _get_user_daily_limit(self, user_id: str) -> float:
        """获取用户每日限额"""
        # 可以从配置或数据库获取
        return 10.0  # 默认 $10/天
```

## 四、成本报告

```python
def generate_cost_report(monitor: CostMonitor) -> str:
    """生成成本报告"""
    report_data = monitor.get_report()
    
    report = f"""# AI 应用成本报告

## 预算使用情况
- 日预算：${monitor.daily_limit:.2f}
- 日消耗：${report_data['daily_cost']:.2f}
- 日剩余：${report_data['daily_budget_remaining']:.2f}
- 月预算：${monitor.monthly_limit:.2f}
- 月消耗：${report_data['monthly_cost']:.2f}
- 月剩余：${report_data['monthly_budget_remaining']:.2f}

## 请求统计
- 总请求数：{report_data['total_requests']}
- 平均单次成本：${report_data['avg_cost_per_request']:.4f}

## 按模型分布
"""
    for model, stats in report_data['by_model'].items():
        report += f"- {model}: {stats['requests']} 次, ${stats['cost']:.2f}\n"
    
    report += "\n## 按操作分布\n"
    for op, stats in report_data['by_operation'].items():
        report += f"- {op}: {stats['requests']} 次, ${stats['cost']:.2f}\n"
    
    return report
```
