## 一、端到端评估的设计

### 1.1 评估维度

```python
class E2EEvaluator:
    """端到端评估器"""
    
    def __init__(self, client):
        self.client = client
    
    def evaluate(
        self,
        question: str,
        answer: str,
        contexts: list[str],
        latency: float,
        token_count: int
    ) -> dict:
        """端到端评估"""
        
        # 1. 回答质量（综合评估）
        quality_prompt = f"""从用户视角评估以下回答的整体质量。

问题：{question}
回答：{answer}

评估标准：
- 是否解决了用户的问题？
- 回答是否准确、完整、易懂？
- 如果你是用户，满意吗？

请以 JSON 格式输出：
{{"user_satisfaction": <1-5>, "solves_problem": <true/false>, "reasoning": "理由"}}"""

        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[{"role": "user", "content": quality_prompt}],
            temperature=0,
            response_format={"type": "json_object"}
        )
        quality = json.loads(response.choices[0].message.content)
        
        # 2. 效率评分
        efficiency_score = 1.0
        if latency > 5.0:
            efficiency_score *= 0.5
        if token_count > 10000:
            efficiency_score *= 0.8
        
        # 3. 综合评分
        overall = (
            quality["user_satisfaction"] / 5.0 * 0.6 +
            (1.0 if quality["solves_problem"] else 0.3) * 0.3 +
            efficiency_score * 0.1
        )
        
        return {
            "quality": quality,
            "efficiency": {
                "latency": latency,
                "token_count": token_count,
                "score": efficiency_score
            },
            "overall_score": overall
        }
```

class FeedbackCollector:
    """用户反馈收集器"""
    
    def __init__(self):
        self.feedback_data = []
    
    def add_feedback(
        self,
        question: str,
        answer: str,
        rating: int,        # 1-5
        feedback_type: str,  # "positive" / "negative" / "neutral"
        comment: str = ""
    ):
        """添加用户反馈"""
        self.feedback_data.append({
            "question": question,
            "answer": answer,
            "rating": rating,
            "feedback_type": feedback_type,
            "comment": comment,
            "timestamp": datetime.now().isoformat()
        })
    
    def analyze_feedback(self) -> dict:
        """分析反馈数据"""
        if not self.feedback_data:
            return {}
        
        df = pd.DataFrame(self.feedback_data)

## 三、综合评估报告

```python
def generate_e2e_report(
    eval_results: list[dict],
    feedback_stats: dict
) -> str:
    """生成端到端评估报告"""
    
    report = "# 端到端评估报告\n\n"
    
    # 质量统计
    avg_satisfaction = sum(r["quality"]["user_satisfaction"] for r in eval_results) / len(eval_results)
    solve_rate = sum(1 for r in eval_results if r["quality"]["solves_problem"]) / len(eval_results)
    
    report += "## 质量指标\n\n"
    report += f"- 平均用户满意度：{avg_satisfaction:.2f}/5.0\n"
    report += f"- 问题解决率：{solve_rate:.1%}\n"
    
    # 效率统计
    avg_latency = sum(r["efficiency"]["latency"] for r in eval_results) / len(eval_results)
    avg_tokens = sum(r["efficiency"]["token_count"] for r in eval_results) / len(eval_results)
    
    report += "\n## 效率指标\n\n"
    report += f"- 平均延迟：{avg_latency:.2f}s\n"
    report += f"- 平均 Token 消耗：{avg_tokens:.0f}\n"
    
    # 用户反馈
    if feedback_stats:
        report += "\n## 用户反馈\n\n"
        report += f"- 反馈总数：{feedback_stats['total_feedback']}\n"
        report += f"- 平均评分：{feedback_stats['avg_rating']:.2f}\n"
        report += f"- 好评率：{feedback_stats['positive_rate']:.1%}\n"
    
    return report
```

本课核心要点：
