## 二、数据收集

```python
def collect_domain_data(domain: str, sources: list) -> list:
    """收集领域数据"""
    data = []
    
    for source in sources:
        if source["type"] == "api":
            # 从 API 获取
            api_data = fetch_from_api(source["url"])
            data.extend(api_data)
        elif source["type"] == "file":
            # 从文件读取
            file_data = read_from_file(source["path"])
            data.extend(file_data)
        elif source["type"] == "crawl":
            # 爬取数据
            crawl_data = crawl_from_web(source["url"])
            data.extend(crawl_data)
    
    return data
```

## 三、数据构造

```python
def construct_training_data(raw_data: list, domain: str, client) -> list:
    """构造训练数据"""
    training_data = []
    
    for item in raw_data:
        # 用 LLM 生成指令
        prompt = f"""基于以下{domain}领域的原始数据，生成一条微调训练数据。

原始数据：
{json.dumps(item, ensure_ascii=False)}

请生成：
1. instruction（指令）
2. input（输入，可选）
3. output（期望输出）

请以 JSON 格式输出。"""
        
        response = client.chat.completions.create(
            model="gpt-4o",
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"}
        )
        
        training_data.append(json.loads(response.choices[0].message.content))
    
    return training_data
```

## 四、完整流程

```python
def build_domain_dataset(domain: str, sources: list, output_path: str):
    """构建领域数据集"""
    
    # 1. 收集数据
    raw_data = collect_domain_data(domain, sources)
    print(f"收集到 {len(raw_data)} 条原始数据")
    
    # 2. 清洗数据
    cleaned_data = clean_dataset(raw_data)
    print(f"清洗后 {len(cleaned_data)} 条数据")
    
    # 3. 构造训练数据
    training_data = construct_training_data(cleaned_data, domain, client)
    print(f"生成 {len(training_data)} 条训练数据")
    
    # 4. 去重
    unique_data = deduplicate(training_data)
    print(f"去重后 {len(unique_data)} 条数据")
    
    # 5. 质量检查
    quality = check_data_quality(unique_data)
    print(f"质量分数：{quality['quality_score']:.2f}")
    
    # 6. 保存
    with open(output_path, "w") as f:
        json.dump(unique_data, f, ensure_ascii=False, indent=2)
    
    print(f"数据集已保存到 {output_path}")
```
