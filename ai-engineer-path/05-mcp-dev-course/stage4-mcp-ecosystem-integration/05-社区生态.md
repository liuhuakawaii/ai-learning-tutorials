## 二、使用开源 Server

```bash
# 安装
npm install @modelcontextprotocol/server-filesystem

# 使用
npx @modelcontextprotocol/server-filesystem /path/to/directory
```

```python
# Python 使用
from mcp import MCPClient
from mcp.transport.stdio import StdioServerTransport

# 连接到文件系统 Server
transport = StdioServerTransport("npx", ["@modelcontextprotocol/server-filesystem", "/tmp"])
client = MCPClient(transport)
await client.initialize()

# 使用工具
tools = await client.list_tools()
result = await client.call_tool("read_file", {"path": "/tmp/test.txt"})
```

## 三、贡献 MCP Server

```python
# 1. 创建项目
# my-mcp-server/
# ├── src/
# │   └── server.py
# ├── pyproject.toml
# └── README.md

# 2. 实现 Server
from mcp.server import Server
from mcp.types import Tool, TextContent

server = Server("my-server")

@server.tool()
async def my_tool(param: str) -> list[TextContent]:
    """我的工具"""
    return [TextContent(type="text", text=f"结果：{param}")]

# 3. 发布
# pip install build
# python -m build
# twine upload dist/*
```
