## QuotesSpider

```python
# spiders/quotes_spider.py
import scrapy
from datetime import datetime
from news_aggregator.items import NewsItem

class QuotesSpider(scrapy.Spider):
    name = 'quotes'
    allowed_domains = ['quotes.toscrape.com']
    start_urls = ['http://quotes.toscrape.com/']

    def parse(self, response):
        for quote in response.css('div.quote'):
            text = quote.css('span.text::text').get('')
            author = quote.css('small.author::text').get('')
            tags = quote.css('a.tag::text').getall()

            item = NewsItem()
            item['title'] = text.strip('"\u201c\u201d')
            item['author'] = author
            item['source'] = 'quotes.toscrape.com'
            item['url'] = response.url
            item['date'] = datetime.now().strftime('%Y-%m-%d')
            item['content'] = text
            item['tags'] = tags
            item['crawled_at'] = datetime.now().isoformat()
            yield item

        next_page = response.css('li.next a::attr(href)').get()
        if next_page:
            yield response.follow(next_page, callback=self.parse)
```

## BooksSpider

```python
# spiders/books_spider.py
import scrapy
from datetime import datetime
from news_aggregator.items import NewsItem

class BooksSpider(scrapy.Spider):
    name = 'books'
    allowed_domains = ['books.toscrape.com']
    start_urls = ['http://books.toscrape.com/']

    def parse(self, response):
        for book in response.css('article.product_pod'):
            detail_url = book.css('h3 a::attr(href)').get()
            if detail_url:
                yield response.follow(
                    detail_url,
                    callback=self.parse_detail,
                    meta={'list_url': response.url}
                )
        next_page = response.css('li.next a::attr(href)').get()
        if next_page:
            yield response.follow(next_page, callback=self.parse)

    def parse_detail(self, response):
        title = response.css('h1::text').get('')
        content = response.css('#product_description ~ p::text').get('暂无描述')
        tags = response.css('ul.breadcrumb li a::text').getall()
        tags = [t.strip() for t in tags if t.strip() and t.strip() != 'Home']

        item = NewsItem()
        item['title'] = title
        item['author'] = '未知作者'
        item['source'] = 'books.toscrape.com'
        item['url'] = response.url
        item['date'] = datetime.now().strftime('%Y-%m-%d')
        item['content'] = f'{title} - {content[:200]}'
        item['tags'] = tags
        item['crawled_at'] = datetime.now().isoformat()
        yield item
```

## Pipeline 清洗

```python
# pipelines.py
import re
from scrapy.exceptions import DropItem

class CleanDataPipeline:
    def process_item(self, item, spider):
        for field in ['title', 'author', 'source', 'content']:
            if item.get(field):
                item[field] = item[field].strip()
        if item.get('content'):
            item['content'] = re.sub(r'<[^>]+>', '', item['content'])
        if not isinstance(item.get('tags'), list):
            item['tags'] = []
        return item

class ValidateItemPipeline:
    def process_item(self, item, spider):
        if not item.get('title') or not item.get('source'):
            raise DropItem(f'缺少必填字段: {item}')
        if len(item['title']) < 2:
            raise DropItem(f'标题太短: {item["title"]}')
        return item

class DuplicateFilterPipeline:
    def __init__(self):
        self.seen_urls = set()

    def process_item(self, item, spider):
        url = item.get('url')
        if url in self.seen_urls:
            raise DropItem(f'重复数据: {url}')
        self.seen_urls.add(url)
        return item
```

Pipeline 按优先级执行：清洗(100) → 验证(200) → 去重(300)。

## 踩坑提醒

**先写 Spider 再想 Item**：边写边加字段，最后 Item 一团糟。先设计 Item。

**所有数据源共用一个 Spider**：不同网站页面结构完全不同，一个 Spider 一个数据源。

**开发时不开 HTTP 缓存**：每次调试都重新请求，既慢又容易被限流。
