## 完整实现

```python
"""
文件整理工具
功能：按文件类型整理杂乱的文件夹
"""

import os
import shutil
from pathlib import Path
from datetime import datetime

FILE_CATEGORIES = {
    "文档": [".txt", ".pdf", ".doc", ".docx", ".xls", ".xlsx",
             ".ppt", ".pptx", ".md", ".csv", ".rtf"],
    "图片": [".jpg", ".jpeg", ".png", ".gif", ".bmp",
             ".svg", ".webp", ".ico"],
    "音频": [".mp3", ".wav", ".flac", ".aac", ".ogg"],
    "视频": [".mp4", ".avi", ".mkv", ".mov", ".wmv", ".flv"],
    "压缩包": [".zip", ".rar", ".7z", ".tar", ".gz"],
    "程序": [".exe", ".msi", ".dmg", ".app", ".apk"],
    "代码": [".py", ".js", ".html", ".css", ".java",
             ".cpp", ".c", ".go", ".rs", ".json", ".xml"],
}


def get_category(extension):
    extension = extension.lower()
    for category, extensions in FILE_CATEGORIES.items():
        if extension in extensions:
            return category
    return "其他"


def scan_files(directory):
    files = []
    for item in Path(directory).iterdir():
        if item.is_file() and not item.name.startswith("."):
            files.append(item)
    return files


def resolve_conflict(source, destination_folder):
    dest = destination_folder / source.name
    if not dest.exists():
        return dest
    stem, suffix = source.stem, source.suffix
    counter = 1
    while dest.exists():
        dest = destination_folder / f"{stem}_{counter}{suffix}"
        counter += 1
    return dest


def move_file(source, destination):
    try:
        shutil.move(str(source), str(destination))
        return True, None
    except PermissionError:
        return False, "权限不足"
    except OSError as e:
        return False, str(e)


def organize_files(target_dir):
    target_path = Path(target_dir)
    if not target_path.exists():
        print(f"错误：目录 {target_dir} 不存在！")
        return None

    files = scan_files(target_path)
    print(f"扫描到 {len(files)} 个文件")

    if not files:
        print("没有需要整理的文件")
        return None

    results = {
        "total": len(files), "success": 0, "failed": 0,
        "details": {}, "errors": [],
    }

    for file_path in files:
        extension = file_path.suffix
        category = get_category(extension) if extension else "其他"
        category_folder = target_path / category
        category_folder.mkdir(exist_ok=True)

        dest_path = resolve_conflict(file_path, category_folder)
        success, error = move_file(file_path, dest_path)

        if success:
            results["success"] += 1
            results["details"][category] = results["details"].get(category, 0) + 1
            print(f"  ✓ {file_path.name} → {category}/")
        else:
            results["failed"] += 1
            results["errors"].append({"file": file_path.name, "error": error})
            print(f"  ✗ {file_path.name} - {error}")

    return results


def generate_report(results, target_dir):
    report_path = Path(target_dir) / "整理报告.txt"
    with open(report_path, "w", encoding="utf-8") as f:
        f.write("=" * 40 + "\n")
        f.write("        文件整理报告\n")
        f.write("=" * 40 + "\n\n")
        f.write(f"整理时间：{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
        f.write(f"整理目录：{target_dir}\n\n")
        f.write(f"文件总数：{results['total']}\n")
        f.write(f"成功整理：{results['success']}\n")
        f.write(f"整理失败：{results['failed']}\n\n")
        if results["details"]:
            f.write("分类统计\n")
            for category, count in sorted(results["details"].items()):
                f.write(f"  {category}：{count} 个文件\n")
        if results["errors"]:
            f.write("\n错误记录\n")
            for err in results["errors"]:
                f.write(f"  {err['file']}：{err['error']}\n")
    print(f"\n报告已生成：{report_path}")


def main():
    print("=" * 40)
    print("    文件整理工具")
    print("=" * 40)

    target = input("请输入要整理的目录路径（直接回车为当前目录）：").strip()
    if not target:
        target = "."

    print(f"\n准备整理目录：{os.path.abspath(target)}")
    confirm = input("确认开始整理？(y/n): ").strip().lower()
    if confirm != "y":
        print("已取消操作")
        return

    print("\n开始整理...\n")
    results = organize_files(target)

    if results:
        generate_report(results, target)
        print(f"\n整理完成！成功：{results['success']}，失败：{results['failed']}")


if __name__ == "__main__":
    main()
```

## 关键设计决策

**`resolve_conflict` 处理重名**：同名文件自动加序号 `file_1.txt`、`file_2.txt`，不覆盖。

**`move_file` 返回元组**：`(success, error)` 比直接抛异常更适合批量操作——一个文件失败不影响其他文件。

**报告生成独立函数**：方便后续扩展为 HTML、JSON 等格式。

## 踩坑提醒

**路径拼接用字符串**：`path + "/" + filename` 在 Windows 上出问题。用 `Path(base_dir) / filename`。

**文件操作不做异常处理**：`shutil.move()` 可能因为权限不足、磁盘已满失败。每个文件操作都应被 try/except 包裹。

**直接在真实目录测试**：可能误删重要文件。先创建测试目录。

```python
from pathlib import Path

def format_size(size_bytes):
    for unit in ["B", "KB", "MB", "GB"]:
        if size_bytes < 1024:
            return f"{size_bytes:.1f} {unit}"
        size_bytes /= 1024
    return f"{size_bytes:.1f} TB"

def get_folder_size(folder_path):
    total = 0
    for item in Path(folder_path).rglob("*"):
        if item.is_file():
            total += item.stat().st_size
    return total
```

```python
from datetime import datetime

def filter_by_date(files, days=None, before=None, after=None):
    filtered = []
    now = datetime.now()
    for f in files:
        mtime = datetime.fromtimestamp(f.stat().st_mtime)
        if days and (now - mtime).days > days:
            continue
        if before and mtime > before:
            continue
        if after and mtime < after:
            continue
        filtered.append(f)
    return filtered
```
