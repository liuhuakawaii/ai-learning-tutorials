```python
"""
文件整理工具 v1.0 - 基础版本
功能：按文件类型整理杂乱的文件夹
"""

import os
import shutil
from pathlib import Path
from datetime import datetime

# ===== 全局配置 =====

# 文件分类规则：类型名 -> 扩展名列表
FILE_CATEGORIES = {
    "文档": [".txt", ".pdf", ".doc", ".docx", ".xls", ".xlsx",
             ".ppt", ".pptx", ".md", ".csv", ".rtf"],
    "图片": [".jpg", ".jpeg", ".png", ".gif", ".bmp",
             ".svg", ".webp", ".ico"],
    "音频": [".mp3", ".wav", ".flac", ".aac", ".ogg",
             ".wma", ".m4a"],
    "视频": [".mp4", ".avi", ".mkv", ".mov", ".wmv",
             ".flv", ".webm"],
    "压缩包": [".zip", ".rar", ".7z", ".tar", ".gz",
               ".bz2"],
    "程序": [".exe", ".msi", ".dmg", ".app", ".apk"],
    "代码": [".py", ".js", ".html", ".css", ".java",
             ".cpp", ".c", ".go", ".rs", ".json", ".xml"],
}


def get_category(extension):
    """
    根据文件扩展名获取分类

    参数：
        extension: 文件扩展名，如 ".pdf"
    返回：
        分类名称，如 "文档"；未知类型返回 "其他"
    """
    extension = extension.lower()  # 统一转小写
    for category, extensions in FILE_CATEGORIES.items():
        if extension in extensions:
            return category
    return "其他"


def scan_files(directory):
    """
    扫描目录中的所有文件（不包括子目录中的文件）

    参数：
        directory: 要扫描的目录路径
    返回：
        文件路径列表
    """
    files = []
    dir_path = Path(directory)

    for item in dir_path.iterdir():
        # 只处理文件，跳过目录
        if item.is_file():
            # 跳过隐藏文件（以 . 开头）
            if not item.name.startswith("."):
                files.append(item)

    return files


def create_category_folder(base_path, category):
    """
    创建分类文件夹

    参数：
        base_path: 基础路径
        category: 分类名称
    返回：
        文件夹路径
    """
    folder_path = Path(base_path) / category
    folder_path.mkdir(exist_ok=True)  # exist_ok=True 表示文件夹存在时不报错
    return folder_path


def resolve_conflict(source, destination_folder):
    """
    处理文件名冲突：如果目标文件夹中已有同名文件，自动重命名

    参数：
        source: 源文件路径
        destination_folder: 目标文件夹路径
    返回：
        不冲突的目标路径
    """
    dest = destination_folder / source.name

    if not dest.exists():
        return dest

    # 文件已存在，添加序号
    stem = source.stem       # 文件名（不含扩展名）
    suffix = source.suffix   # 扩展名
    counter = 1

    while dest.exists():
        new_name = f"{stem}_{counter}{suffix}"
        dest = destination_folder / new_name
        counter += 1

    return dest


def move_file(source, destination):
    """
    移动文件，带异常处理

    参数：
        source: 源文件路径
        destination: 目标文件路径
    返回：
        (是否成功, 错误信息)
    """
    try:
        shutil.move(str(source), str(destination))
        return True, None
    except PermissionError:
        return False, "权限不足"
    except OSError as e:
        return False, str(e)


def organize_files(target_dir):
    """
    主函数：整理指定目录中的文件

    参数：
        target_dir: 要整理的目录路径
    返回：
        整理结果统计字典
    """
    target_path = Path(target_dir)

    # 检查目录是否存在
    if not target_path.exists():
        print(f"错误：目录 {target_dir} 不存在！")
        return None

    if not target_path.is_dir():
        print(f"错误：{target_dir} 不是目录！")
        return None

    # 扫描文件
    files = scan_files(target_path)
    print(f"扫描到 {len(files)} 个文件")

    if not files:
        print("没有需要整理的文件")
        return None

    # 统计结果
    results = {
        "total": len(files),
        "success": 0,
        "failed": 0,
        "skipped": 0,
        "details": {},  # 按分类统计
        "errors": [],   # 错误记录
    }

    # 遍历并整理每个文件
    for file_path in files:
        # 获取文件扩展名
        extension = file_path.suffix

        # 没有扩展名的文件归到"其他"
        if not extension:
            category = "其他"
        else:
            category = get_category(extension)

        # 创建分类文件夹
        category_folder = create_category_folder(target_path, category)

        # 处理重名冲突
        dest_path = resolve_conflict(file_path, category_folder)

        # 移动文件
        success, error = move_file(file_path, dest_path)

        if success:
            results["success"] += 1
            # 更新分类统计
            if category not in results["details"]:
                results["details"][category] = 0
            results["details"][category] += 1
            print(f"  ✓ {file_path.name} → {category}/")
        else:
            results["failed"] += 1
            results["errors"].append({
                "file": file_path.name,
                "error": error
            })
            print(f"  ✗ {file_path.name} - {error}")

    return results


def generate_report(results, target_dir):
    """
    生成整理报告

    参数：
        results: 整理结果统计
        target_dir: 目标目录
    """
    report_path = Path(target_dir) / "整理报告.txt"

    with open(report_path, "w", encoding="utf-8") as f:
        f.write("=" * 40 + "\n")
        f.write("        文件整理报告\n")
        f.write("=" * 40 + "\n\n")

        f.write(f"整理时间：{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
        f.write(f"整理目录：{target_dir}\n\n")

        f.write("─" * 40 + "\n")
        f.write("统计摘要\n")
        f.write("─" * 40 + "\n")
        f.write(f"  文件总数：{results['total']}\n")
        f.write(f"  成功整理：{results['success']}\n")
        f.write(f"  整理失败：{results['failed']}\n\n")

        if results["details"]:
            f.write("─" * 40 + "\n")
            f.write("分类统计\n")
            f.write("─" * 40 + "\n")
            for category, count in sorted(results["details"].items()):
                f.write(f"  {category}：{count} 个文件\n")
            f.write("\n")

        if results["errors"]:
            f.write("─" * 40 + "\n")
            f.write("错误记录\n")
            f.write("─" * 40 + "\n")
            for err in results["errors"]:
                f.write(f"  {err['file']}：{err['error']}\n")

        f.write("\n" + "=" * 40 + "\n")

    print(f"\n报告已生成：{report_path}")


def main():
    """主入口函数"""
    print("=" * 40)
    print("    文件整理工具 v1.0")
    print("=" * 40)
    print()

    # 获取用户输入
    target = input("请输入要整理的目录路径（直接回车为当前目录）：").strip()
    if not target:
        target = "."

    print(f"\n准备整理目录：{os.path.abspath(target)}")
    confirm = input("确认开始整理？(y/n): ").strip().lower()

    if confirm != "y":
        print("已取消操作")
        return

    print("\n开始整理...\n")

    # 执行整理
    results = organize_files(target)

    if results:
        # 生成报告
        generate_report(results, target)

        # 打印摘要
        print("\n" + "=" * 40)
        print("整理完成！")
        print(f"  成功：{results['success']} 个文件")
        print(f"  失败：{results['failed']} 个文件")
        print("=" * 40)


# 程序入口
if __name__ == "__main__":
    main()
```

```python
# ===== get_category 函数解析 =====

def get_category(extension):
    """
    工作原理：
    1. 遍历 FILE_CATEGORIES 字典
    2. 检查扩展名是否在某个分类的列表中
    3. 找到就返回分类名，找不到返回"其他"
    """
    extension = extension.lower()  # .PDF -> .pdf

    for category, extensions in FILE_CATEGORIES.items():
        # category = "文档"
        # extensions = [".txt", ".pdf", ".doc", ...]
        if extension in extensions:
            return category

    return "其他"  # 没有匹配的分类
```

```python
# ===== resolve_conflict 函数解析 =====

def resolve_conflict(source, destination_folder):
    """
    处理重名文件的逻辑：
    1. 如果目标不存在同名文件，直接返回目标路径
    2. 如果存在，在文件名后加序号：file_1.txt, file_2.txt, ...
    """
    dest = destination_folder / source.name

    # 第一次检查：没有冲突
    if not dest.exists():
        return dest

    # 有冲突，开始添加序号
    stem = source.stem       # "report"
    suffix = source.suffix   # ".pdf"
    counter = 1

    while dest.exists():
        new_name = f"{stem}_{counter}{suffix}"  # "report_1.pdf"
        dest = destination_folder / new_name
        counter += 1

    return dest

# 示例：
# 源文件: report.pdf
# 目标已有: report.pdf
# 结果: report_1.pdf
# 如果 report_1.pdf 也存在: report_2.pdf
```

```python
    for category, extensions in FILE_CATEGORIES.items():
        if extension in extensions:
            return category
    return "其他"


def scan_files(directory, recursive=False):
    """
    扫描文件

    参数：
        directory: 目录路径
        recursive: 是否递归扫描子目录
    """
    files = []
    dir_path = Path(directory)

    if recursive:
        # 递归扫描
        for item in dir_path.rglob("*"):
            if item.is_file() and not item.name.startswith("."):
                # 跳过已在分类文件夹中的文件
                if item.parent.name not in FILE_CATEGORIES and \
                   item.parent.name != "其他":
                    files.append(item)
    else:
        # 只扫描当前目录
        for item in dir_path.iterdir():
            if item.is_file() and not item.name.startswith("."):
                files.append(item)

    return files


def preview_organization(files):
    """
    预览整理结果（不实际移动）

    参数：
        files: 文件列表
    返回：
        分类结果字典
    """
    preview = {}

    for file_path in files:
        ext = file_path.suffix
        category = get_category(ext) if ext else "其他"

        if category not in preview:
            preview[category] = []
        preview[category].append(file_path.name)

    return preview


def organize_files(target_dir, recursive=False, dry_run=False):
    """
    整理文件

    参数：
        target_dir: 目标目录
        recursive: 是否递归整理
        dry_run: 预览模式，不实际移动
    """
    target_path = Path(target_dir)

    if not target_path.exists():
        logger.error(f"目录不存在：{target_dir}")
        return None

    # 扫描文件
    files = scan_files(target_path, recursive)
    logger.info(f"扫描到 {len(files)} 个文件")

    if not files:
        logger.info("没有需要整理的文件")
        return None

    # 预览模式
    if dry_run:
        preview = preview_organization(files)
        print("\n预览整理结果：")
        print("─" * 40)
        for category, filenames in sorted(preview.items()):
            print(f"\n{category}/ ({len(filenames)} 个文件)")
            for name in filenames:
                print(f"  ├── {name}")
        return None

    # 实际整理
    results = {
        "total": len(files),
        "success": 0,
        "failed": 0,
        "details": {},
        "errors": [],
        "start_time": datetime.now(),
    }

    for file_path in files:
        ext = file_path.suffix
        category = get_category(ext) if ext else "其他"

        # 创建目标文件夹
        dest_folder = target_path / category
        dest_folder.mkdir(exist_ok=True)

        # 处理重名
        dest = dest_folder / file_path.name
        if dest.exists():
            stem = file_path.stem
            suffix = file_path.suffix
            counter = 1
            while dest.exists():
                dest = dest_folder / f"{stem}_{counter}{suffix}"
                counter += 1

        # 移动文件
        try:
            shutil.move(str(file_path), str(dest))
            results["success"] += 1
            results["details"][category] = results["details"].get(category, 0) + 1
            logger.info(f"移动：{file_path.name} → {category}/")
        except Exception as e:
            results["failed"] += 1
            results["errors"].append({"file": file_path.name, "error": str(e)})
            logger.error(f"失败：{file_path.name} - {e}")

    results["end_time"] = datetime.now()
    results["duration"] = (results["end_time"] - results["start_time"]).total_seconds()

    return results


def generate_report(results, target_dir):
    """生成整理报告"""
    if not results:
        return

    report_path = Path(target_dir) / "整理报告.txt"

    with open(report_path, "w", encoding="utf-8") as f:
        f.write("=" * 50 + "\n")
        f.write("            文件整理报告\n")
        f.write("=" * 50 + "\n\n")

        f.write(f"整理时间：{results['start_time'].strftime('%Y-%m-%d %H:%M:%S')}\n")
        f.write(f"完成时间：{results['end_time'].strftime('%Y-%m-%d %H:%M:%S')}\n")
        f.write(f"耗时：{results['duration']:.2f} 秒\n\n")

        f.write("─" * 50 + "\n")
        f.write("统计摘要\n")
        f.write("─" * 50 + "\n")
        f.write(f"  文件总数：{results['total']}\n")
        f.write(f"  成功整理：{results['success']}\n")
        f.write(f"  整理失败：{results['failed']}\n")
        f.write(f"  成功率：{results['success']/results['total']*100:.1f}%\n\n")

        if results["details"]:
            f.write("─" * 50 + "\n")
            f.write("分类统计\n")
            f.write("─" * 50 + "\n")
            for cat, count in sorted(results["details"].items(),
                                      key=lambda x: x[1], reverse=True):
                bar = "█" * count

            f.write("─" * 50 + "\n")
            for err in results["errors"]:

        f.write("\n" + "=" * 50 + "\n")

    logger.info(f"报告已生成：{report_path}")
```

```python
import json

def save_operation_log(operations, log_file="operation_log.json"):
    """保存操作日志，用于撤销"""
    log_data = []
    for source, dest in operations:
        log_data.append({
            "source": str(source),
            "dest": str(dest),
            "timestamp": datetime.now().isoformat()
        })

    with open(log_file, "w", encoding="utf-8") as f:
        json.dump(log_data, f, ensure_ascii=False, indent=2)

def undo_operations(log_file="operation_log.json"):
    """撤销操作"""
    with open(log_file, "r", encoding="utf-8") as f:
        log_data = json.load(f)

    # 反向执行
    for op in reversed(log_data):
        source = Path(op["dest"])
        dest = Path(op["source"])
        if source.exists():
            shutil.move(str(source), str(dest))
            print(f"撤销：{source.name} → {dest.parent.name}/")
```

```python
from datetime import datetime, timedelta

def filter_by_date(files, days=None, before=None, after=None):
    """
    按修改时间筛选文件

    参数：
        files: 文件列表
        days: 最近N天内修改的文件
        before: 在此日期之前修改的文件
        after: 在此日期之后修改的文件
    """
    filtered = []
    now = datetime.now()

    for f in files:
        mtime = datetime.fromtimestamp(f.stat().st_mtime)

        if days and (now - mtime).days > days:
            continue
        if before and mtime > before:
            continue
        if after and mtime < after:
            continue

        filtered.append(f)

    return filtered
```

## 常见误区

- **"拿到需求就开始写代码"**：很多人一上来就打开编辑器写代码，写到一半发现需求没搞清楚，又要推倒重来。正确做法是先做需求分析——列出功能清单、设计函数结构、画出流程图，然后再动手编码。花 20 分钟做设计，能省 2 小时的返工。
- **"文件操作不需要异常处理"**：`shutil.move()` 可能因为权限不足、磁盘已满、文件被占用而失败；`os.listdir()` 可能因为目录不存在而报错。文件操作是爬虫中最容易出问题的环节，每一个文件操作都应该被 `try/except` 包裹。
- **"路径拼接用字符串拼接就行"**：`path + "/" + filename` 在 Windows 上会出问题（Windows 用 `\` 不是 `/`）。用 `os.path.join()` 或 `pathlib.Path / "filename"` 能自动处理跨平台路径分隔符。
- **"测试环境不重要，直接在真实目录上测试"**：在真实的下载目录上测试整理工具，可能误删重要文件或打乱目录结构。一定要先创建测试目录，验证功能正确后再用在真实场景。

---

## 工程建议

- **先做需求分析，再做技术设计**：拿到项目后，先列出功能清单和非功能需求（性能、安全性、可扩展性），再设计函数结构和数据流。编码只是最后一步，前面的思考过程才是决定项目质量的关键。
- **模块化设计，函数职责单一**：每个函数只做一件事——扫描目录的只管扫描，分类的只管分类，移动文件的只管移动，生成报告的只管生成报告。这样每个函数都能独立测试和复用，修改某一部分不会影响其他部分。
- **做好边界条件处理**：空目录怎么办？没有扩展名的文件怎么分类？目标文件夹已有同名文件怎么办？权限不足怎么办？这些边界条件不处理，程序在真实环境中一定会崩溃。在写核心逻辑之前，先想清楚所有可能出错的场景。
- **养成写日志的习惯**：用 `logging` 模块记录程序运行过程，比 `print()` 更专业。日志可以输出到文件，方便事后排查问题；可以设置不同级别（INFO/WARNING/ERROR），区分正常信息和异常信息。

```python
            category_folder = target_path / category
            if category_folder.exists():
                size = get_folder_size(category_folder)
                f.write(f"  {category:8} {count} 个文件  {format_size(size)}\n")
            else:
                f.write(f"  {category:8} {count} 个文件\n")

    print(f"报告已生成：{report_path}")

# 测试 format_size 函数
print(format_size(1024))         # 1.0 KB
print(format_size(1048576))      # 1.0 MB
print(format_size(1073741824))   # 1.0 GB
```

**要点**：
- `Path.rglob("*")` 递归遍历所有文件和子目录
- `item.stat().st_size` 返回文件大小（字节）
- `format_size` 用循环逐步除以 1024 来自动选择合适的单位
- `.1f` 表示保留一位小数

### 练习2

**思路**：使用 `datetime` 模块获取文件的修改时间（`st_mtime`），与用户指定的时间范围进行比较。

**答案**：

```python
from datetime import datetime, timedelta
from pathlib import Path

def filter_by_date(files, days=None, before=None, after=None):
    """
    按修改时间筛选文件

    参数：
        files: 文件路径列表
        days: 只保留最近 N 天内修改的文件
        before: 只保留在此日期之前修改的文件（datetime 对象）
        after: 只保留在此日期之后修改的文件（datetime 对象）
    返回：
        符合条件的文件列表
    """
    filtered = []
    now = datetime.now()

    for f in files:
        mtime = datetime.fromtimestamp(f.stat().st_mtime)

        if days is not None and (now - mtime).days > days:
            continue
        if before is not None and mtime > before:
            continue
        if after is not None and mtime < after:
            continue

        filtered.append(f)

    return filtered

# 使用示例：
from pathlib import Path

# 获取所有文件
all_files = list(Path(".").iterdir())

# 只保留最近 7 天修改的文件
recent_files = filter_by_date(all_files, days=7)
print(f"最近 7 天的文件：{len(recent_files)} 个")

# 只保留 2024 年 1 月 1 日之后修改的文件
cutoff = datetime(2024, 1, 1)
new_files = filter_by_date(all_files, after=cutoff)
print(f"2024 年之后的文件：{len(new_files)} 个")

# 结合 organize_files 使用
# files = scan_files(target_path)
# files = filter_by_date(files, days=30)  # 只整理最近 30 天的文件
# for file_path in files:
#     ...（后续整理逻辑）
```

**要点**：
- `f.stat().st_mtime` 返回文件最后修改时间的时间戳（Unix 时间戳）
- `datetime.fromtimestamp()` 把时间戳转为 `datetime` 对象，方便比较
- `(now - mtime).days` 计算两个日期之间相差的天数
- 多个筛选条件之间是"且"的关系（同时满足才保留）

### 练习3

**思路**：使用 `tkinter` 创建图形界面，包含目录选择、预览和执行三个核心功能。将之前写好的 `organize_files` 函数作为后端逻辑，GUI 只负责交互。

**答案**：

```python
import tkinter as tk
from tkinter import filedialog, messagebox, scrolledtext
import os
import shutil
from pathlib import Path
from datetime import datetime

# 复用之前的 FILE_CATEGORIES 和核心函数
FILE_CATEGORIES = {
    "文档": [".txt", ".pdf", ".doc", ".docx", ".xls", ".xlsx",
             ".ppt", ".pptx", ".md", ".csv", ".rtf"],
    "图片": [".jpg", ".jpeg", ".png", ".gif", ".bmp",
             ".svg", ".webp", ".ico"],
    "音频": [".mp3", ".wav", ".flac", ".aac", ".ogg",
             ".wma", ".m4a"],
    "视频": [".mp4", ".avi", ".mkv", ".mov", ".wmv",
             ".flv", ".webm"],
    "压缩包": [".zip", ".rar", ".7z", ".tar", ".gz", ".bz2"],
    "程序": [".exe", ".msi", ".dmg", ".app", ".apk"],
    "代码": [".py", ".js", ".html", ".css", ".java",
             ".cpp", ".c", ".go", ".rs", ".json", ".xml"],
}

def get_category(extension):
    extension = extension.lower()
    for category, extensions in FILE_CATEGORIES.items():
        if extension in extensions:
            return category
    return "其他"


class FileOrganizerGUI:
    def __init__(self):
        self.root = tk.Tk()
        self.root.title("文件整理工具")
        self.root.geometry("500x400")

        # 顶部：目录选择
        top_frame = tk.Frame(self.root)
        top_frame.pack(fill=tk.X, padx=10, pady=5)

        self.dir_label = tk.Label(top_frame, text="未选择目录", anchor="w")
        self.dir_label.pack(side=tk.LEFT, fill=tk.X, expand=True)

        self.select_btn = tk.Button(
            top_frame, text="选择目录", command=self.select_directory
        )
        self.select_btn.pack(side=tk.RIGHT)

        # 中部：日志输出
        self.log_area = scrolledtext.ScrolledText(self.root, height=15)
        self.log_area.pack(fill=tk.BOTH, expand=True, padx=10, pady=5)

        # 底部：操作按钮
        btn_frame = tk.Frame(self.root)
        btn_frame.pack(fill=tk.X, padx=10, pady=5)

        self.preview_btn = tk.Button(
            btn_frame, text="预览", command=self.preview, state=tk.DISABLED
        )
        self.preview_btn.pack(side=tk.LEFT, padx=5)

        self.organize_btn = tk.Button(
            btn_frame, text="开始整理", command=self.organize, state=tk.DISABLED
        )
        self.organize_btn.pack(side=tk.LEFT, padx=5)

        self.selected_dir = None

    def select_directory(self):
        self.selected_dir = filedialog.askdirectory()
        if self.selected_dir:
            self.dir_label.config(text=self.selected_dir)
            self.preview_btn.config(state=tk.NORMAL)
            self.organize_btn.config(state=tk.NORMAL)
            self.log(f"已选择目录：{self.selected_dir}")

    def log(self, message):
        self.log_area.insert(tk.END, message + "\n")
        self.log_area.see(tk.END)

    def preview(self):
        if not self.selected_dir:
            return
        self.log_area.delete("1.0", tk.END)
        self.log("=== 预览模式 ===")

        files = [f for f in Path(self.selected_dir).iterdir()
                 if f.is_file() and not f.name.startswith(".")]

        categories = {}
        for f in files:
            cat = get_category(f.suffix) if f.suffix else "其他"
            categories.setdefault(cat, []).append(f.name)

        for cat, names in sorted(categories.items()):
            self.log(f"\n{cat}/ ({len(names)} 个)")
            for name in names:
                self.log(f"  ├── {name}")

        self.log(f"\n共 {len(files)} 个文件")

    def organize(self):
        if not self.selected_dir:
            return

        confirm = messagebox.askyesno("确认", "确定要开始整理文件吗？")
        if not confirm:
            return

        self.log("\n=== 开始整理 ===")
        target = Path(self.selected_dir)
        files = [f for f in target.iterdir()
                 if f.is_file() and not f.name.startswith(".")]

        success, failed = 0, 0
        for f in files:
            cat = get_category(f.suffix) if f.suffix else "其他"
            dest_folder = target / cat
            dest_folder.mkdir(exist_ok=True)
            dest = dest_folder / f.name

            try:
                shutil.move(str(f), str(dest))
                self.log(f"  ✓ {f.name} → {cat}/")
                success += 1
            except Exception as e:
                self.log(f"  ✗ {f.name} - {e}")
                failed += 1

        self.log(f"\n整理完成！成功 {success} 个，失败 {failed} 个")
        messagebox.showinfo("完成", f"成功：{success}\n失败：{failed}")

    def run(self):
        self.root.mainloop()


if __name__ == "__main__":
    app = FileOrganizerGUI()
    app.run()
```

**要点**：
- `tkinter` 是 Python 内置的 GUI 库，不需要额外安装
- `filedialog.askdirectory()` 弹出目录选择对话框
- `scrolledtext.ScrolledText` 是带滚动条的文本框，适合显示日志
- `messagebox.askyesno()` 弹出确认对话框，返回 `True`/`False`
- GUI 程序的核心逻辑和命令行版本一样，只是输入输出方式变了
