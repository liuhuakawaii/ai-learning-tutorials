```python
# book_scraper/spiders/books_spider.py
import scrapy
from book_scraper.items import BookItem

class BooksSpider(scrapy.Spider):
    name = "books"
    allowed_domains = ["books.toscrape.com"]
    start_urls = ["https://books.toscrape.com/catalogue/page-1.html"]
    RATING_MAP = {"One": 1, "Two": 2, "Three": 3, "Four": 4, "Five": 5}

    def parse(self, response):
        for book in response.css("article.product_pod"):
            item = BookItem()
            item["title"] = book.css("h3 a::attr(title)").get()
            item["price"] = book.css(".price_color::text").get()
            rating_class = book.css("p.star-rating::attr(class)").get()
            rating_text = rating_class.replace("star-rating ", "")
            item["rating"] = self.RATING_MAP.get(rating_text, 0)
            item["availability"] = " ".join(book.css(".availability::text").getall()).strip()

            detail_url = book.css("h3 a::attr(href)").get()
            item["detail_url"] = response.urljoin(detail_url)
            yield response.follow(detail_url, callback=self.parse_detail, meta={"item": item})

        next_page = response.css("li.next a::attr(href)").get()
        if next_page:
            yield response.follow(next_page, callback=self.parse)

    def parse_detail(self, response):
        item = response.meta["item"]
        item["image_url"] = response.urljoin(response.css("#product_gallery img::attr(src)").get())
        item["upc"] = response.css("table.table-striped tr:nth-child(1) td::text").get()
        breadcrumbs = response.css("ul.breadcrumb li a::text").getall()
        item["category"] = breadcrumbs[2].strip() if len(breadcrumbs) >= 3 else "Unknown"
        desc = response.css("#product_description ~ p::text").get()
        item["description"] = desc if desc else "暂无简介"
        yield item
```

## SQLite Pipeline

```python
# book_scraper/pipelines.py
import sqlite3, os

class SQLitePipeline:
    def __init__(self):
        self.db_path = os.path.join(os.path.dirname(__file__), "..", "books.db")

    def open_spider(self, spider):
        self.conn = sqlite3.connect(self.db_path)
        self.cursor = self.conn.cursor()
        self.cursor.execute("""
            CREATE TABLE IF NOT EXISTS books (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL, price TEXT, rating INTEGER,
                availability TEXT, category TEXT,
                detail_url TEXT UNIQUE, image_url TEXT,
                upc TEXT, description TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        self.cursor.execute("CREATE INDEX IF NOT EXISTS idx_category ON books(category)")
        self.cursor.execute("CREATE INDEX IF NOT EXISTS idx_rating ON books(rating)")
        self.conn.commit()

    def process_item(self, item, spider):
        try:
            self.cursor.execute("""
                INSERT OR IGNORE INTO books
                (title, price, rating, availability, category, detail_url, image_url, upc, description)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
            """, (
                item.get("title", ""), item.get("price", ""), item.get("rating", 0),
                item.get("availability", ""), item.get("category", ""),
                item.get("detail_url", ""), item.get("image_url", ""),
                item.get("upc", ""), item.get("description", ""),
            ))
            self.conn.commit()
        except sqlite3.Error as e:
            spider.logger.error(f"数据库插入失败: {e}")
        return item

    def close_spider(self, spider):
        if self.conn:
            self.conn.close()
```

## Flask API

```python
# api/app.py
import sqlite3, os
from flask import Flask, jsonify, request, send_from_directory

app = Flask(__name__, static_folder="../frontend", static_url_path="")
DB_PATH = os.path.join(os.path.dirname(__file__), "..", "books.db")

def get_db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

@app.route("/")
def index():
    return send_from_directory("../frontend", "index.html")

@app.route("/api/books", methods=["GET"])
def get_books():
    page = request.args.get("page", 1, type=int)
    per_page = min(request.args.get("per_page", 20, type=int), 100)
    category = request.args.get("category", "")
    min_rating = request.args.get("min_rating", 0, type=int)
    search = request.args.get("search", "")
    sort_by = request.args.get("sort", "id")
    if sort_by not in ["id", "title", "price", "rating"]:
        sort_by = "id"

    conditions, params = [], []
    if category:
        conditions.append("category = ?"); params.append(category)
    if min_rating > 0:
        conditions.append("rating >= ?"); params.append(min_rating)
    if search:
        conditions.append("title LIKE ?"); params.append(f"%{search}%")

    where = ("WHERE " + " AND ".join(conditions)) if conditions else ""

    conn = get_db()
    cursor = conn.cursor()
    cursor.execute(f"SELECT COUNT(*) FROM books {where}", params)
    total = cursor.fetchone()[0]

    offset = (page - 1) * per_page
    cursor.execute(f"SELECT * FROM books {where} ORDER BY {sort_by} LIMIT ? OFFSET ?",
                   params + [per_page, offset])
    books = [dict(row) for row in cursor.fetchall()]
    conn.close()

    return jsonify({
        "data": books, "total": total, "page": page,
        "per_page": per_page,
        "total_pages": (total + per_page - 1) // per_page
    })

@app.route("/api/stats", methods=["GET"])
def get_stats():
    conn = get_db()
    cursor = conn.cursor()
    cursor.execute("SELECT COUNT(*) FROM books")
    total = cursor.fetchone()[0]
    cursor.execute("SELECT AVG(rating) FROM books WHERE rating > 0")
    avg = cursor.fetchone()[0] or 0
    cursor.execute("SELECT category, COUNT(*) FROM books GROUP BY category ORDER BY COUNT(*) DESC")
    cats = [{"name": r[0], "count": r[1]} for r in cursor.fetchall()]
    conn.close()
    return jsonify({"data": {"total_books": total, "avg_rating": round(avg, 2), "categories": cats[:10]}})

if __name__ == "__main__":
    app.run(debug=True, port=5000)
```

```javascript
// frontend/app.js 核心逻辑
const state = { currentPage: 1, search: "", category: "", minRating: 0, sortBy: "id" };

async function fetchBooks() {
    const params = new URLSearchParams({
        page: state.currentPage, per_page: 20,
        search: state.search, category: state.category,
        min_rating: state.minRating, sort: state.sortBy,
    });
    const res = await fetch(`/api/books?${params}`);
    const result = await res.json();
    renderBooks(result.data);
    renderPagination(result.page, result.total_pages, result.total);
}

function renderBooks(books) {
    document.getElementById("bookGrid").innerHTML = books.map(book => `
        <div class="book-card">
            <img class="book-image" src="${book.image_url}" onerror="this.src='placeholder.png'">
            <div class="book-info">
                <div class="book-title">${book.title}</div>
                <span class="book-category">${book.category}</span>
                <div class="book-meta">
                    <span class="book-price">${book.price}</span>
                    <span class="book-rating">${"⭐".repeat(book.rating)}</span>
                </div>
            </div>
        </div>
    `).join("");
}
```
