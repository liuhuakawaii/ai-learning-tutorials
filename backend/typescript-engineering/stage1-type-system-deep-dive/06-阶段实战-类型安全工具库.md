# 06 - 阶段实战：类型安全工具库

## 场景引入

你的团队有多个项目，每个项目都有一些重复的工具函数：pick 选取对象属性、omit 排除属性、deep merge 深度合并、get 安全访问嵌套属性。这些函数要么没有类型安全，要么每个项目各写一份。现在要构建一个统一的类型安全工具库。

## 学习目标

- 综合运用泛型、条件类型、映射类型、infer 等阶段一所学知识
- 实现 `pick`、`omit`、`merge`、`deepMerge`、`get`、`set` 等常用工具函数
- 理解工具函数设计中类型安全与实用性的权衡
- 为工具函数编写完整的类型测试

## 一、pick 函数

从对象中选取指定的属性：

```typescript
function pick<T extends object, K extends keyof T>(
  obj: T,
  ...keys: K[]
): Pick<T, K> {
  const result = {} as Pick<T, K>
  for (const key of keys) {
    result[key] = obj[key]
  }
  return result
}

interface User {
  id: string
  name: string
  email: string
  age: number
  active: boolean
}

declare const user: User

const nameAndEmail = pick(user, "name", "email")
// 类型是 { name: string; email: string }
// pick(user, "name", "unknown") // 编译报错
```

## 二、omit 函数

排除指定的属性：

```typescript
function omit<T extends object, K extends keyof T>(
  obj: T,
  ...keys: K[]
): Omit<T, K> {
  const keysSet = new Set<string | number | symbol>(keys)
  const result = {} as Omit<T, K>
  for (const key of Object.keys(obj) as Array<keyof T>) {
    if (!keysSet.has(key)) {
      ;(result as any)[key] = obj[key]
    }
  }
  return result
}

const userWithoutEmail = omit(user, "email", "age")
// 类型是 { id: string; name: string; active: boolean }
```

## 三、merge 函数

浅合并两个对象：

```typescript
type Merged<A, B> = {
  [K in keyof A | keyof B]: K extends keyof B
    ? B[K]
    : K extends keyof A
      ? A[K]
      : never
}

function merge<A extends object, B extends object>(
  a: A,
  b: B
): Merged<A, B> {
  return { ...a, ...b } as Merged<A, B>
}

const defaults = { host: "localhost", port: 3000, debug: false }
const overrides = { port: 8080, debug: true }

const config = merge(defaults, overrides)
// { host: string; port: number; debug: boolean }
```

## 四、deepMerge 函数

深度合并两个对象：

```typescript
type DeepMerge<A, B> = A extends object
  ? B extends object
    ? {
        [K in keyof A | keyof B]: K extends keyof B
          ? K extends keyof A
            ? DeepMerge<A[K], B[K]>
            : B[K]
          : K extends keyof A
            ? A[K]
            : never
      }
    : B
  : B

function isPlainObject(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value) &&
    Object.getPrototypeOf(value) === Object.prototype
}

function deepMerge<A extends object, B extends object>(a: A, b: B): DeepMerge<A, B> {
  const result: Record<string, unknown> = { ...a }
  for (const key of Object.keys(b)) {
    const aVal = (a as Record<string, unknown>)[key]
    const bVal = (b as Record<string, unknown>)[key]
    if (isPlainObject(aVal) && isPlainObject(bVal)) {
      result[key] = deepMerge(aVal, bVal)
    } else {
      result[key] = bVal
    }
  }
  return result as DeepMerge<A, B>
}

const baseConfig = {
  server: { host: "localhost", port: 3000 },
  logging: { level: "info" as const, format: "json" },
}

const userConfig = {
  server: { port: 8080 },
  logging: { level: "debug" as const },
}

const finalConfig = deepMerge(baseConfig, userConfig)
// { server: { host: string; port: number }; logging: { level: "debug"; format: string } }
```

## 五、get 函数

类型安全的嵌套属性访问：

```typescript
type GetByPath<T, P extends string> =
  P extends `${infer Key}.${infer Rest}`
    ? Key extends keyof T
      ? GetByPath<T[Key], Rest>
      : never
    : P extends keyof T
      ? T[P]
      : never

function get<T extends object, P extends string>(obj: T, path: P): GetByPath<T, P> {
  const keys = path.split(".")
  let result: any = obj
  for (const key of keys) {
    if (result == null) return undefined as GetByPath<T, P>
    result = result[key]
  }
  return result
}

interface AppState {
  user: {
    profile: { name: string; address: { city: string; zip: string } }
    settings: { theme: "light" | "dark" }
  }
}

declare const state: AppState

const city = get(state, "user.profile.address.city")  // string
const theme = get(state, "user.settings.theme")        // "light" | "dark"
// get(state, "user.unknown") // 编译报错
```

## 六、set 函数

```typescript
function set<T extends object, P extends string>(
  obj: T, path: P, value: GetByPath<T, P>
): void {
  const keys = path.split(".")
  let current: any = obj
  for (let i = 0; i < keys.length - 1; i++) {
    current = current[keys[i]]
    if (current == null) return
  }
  current[keys[keys.length - 1]] = value
}

set(state, "user.profile.name", "李四")
set(state, "user.settings.theme", "dark")
// set(state, "user.settings.theme", "blue") // 编译报错
```

## 七、类型测试

使用 `expect-type` 库验证类型推断：

```typescript
import { expectTypeOf } from "expect-type"

const picked = pick(user, "name", "email")
expectTypeOf(picked).toEqualTypeOf<{ name: string; email: string }>()

const omitted = omit(user, "email")
expectTypeOf(omitted).toEqualTypeOf<{ id: string; name: string; age: number; active: boolean }>()

const merged = merge({ a: 1 }, { b: "hello" })
expectTypeOf(merged).toEqualTypeOf<{ a: number; b: string }>()
```

## 常见误区

1. **get 函数的路径类型过于宽松**：`path` 参数类型是 `string` 时失去类型安全，必须约束合法路径。
2. **deepMerge 不处理数组**：数组不是普通对象，应该用后者覆盖而不是递归合并。
3. **忽略 null 和 undefined**：`get` 在路径中断时应返回 `undefined`，而不是抛出错误。
4. **过度追求类型精度**：有些场景用 `string` 比字面量联合类型更实用。

## 工程建议

1. **工具函数的类型和实现放在同一文件**：便于维护和理解。
2. **为每个工具函数编写类型测试**：类型测试和运行时测试同样重要。
3. **导出类型定义**：让使用者可以在自己的类型系统中引用。
4. **用 JSDoc 注释使用示例**：在 IDE 中自动显示。

## 小结

这个阶段的实战把类型推断、泛型、条件类型、映射类型等知识综合运用到一个实用的工具库中。类型体操的最终目的是让调用方获得零成本的类型安全。

## 参考答案

### 练习一

**思路**：用映射类型遍历对象的键，将值类型替换为 fn 的返回类型。

**答案**：

```typescript
function mapValues<T extends Record<string, unknown>, R>(
  obj: T,
  fn: (value: T[keyof T], key: keyof T) => R
): { [K in keyof T]: R } {
  const result = {} as { [K in keyof T]: R }
  for (const key of Object.keys(obj) as Array<keyof T>) {
    ;(result as any)[key] = fn(obj[key], key)
  }
  return result
}

const scores = { alice: 85, bob: 92, charlie: 78 }
const grades = mapValues(scores, (score) => {
  if (score >= 90) return "A"
  if (score >= 80) return "B"
  return "C"
})
// { alice: string; bob: string; charlie: string }
```

**要点**：`{ [K in keyof T]: R }` 遍历 T 的所有键，将值类型替换为 R。

### 练习二

**思路**：约束 T[K] 为 string | number | symbol，使其可以作为 Record 的键。

**答案**：

```typescript
function groupBy<T extends Record<string, unknown>, K extends keyof T>(
  items: T[],
  key: K
): Record<string, T[]> {
  const result: Record<string, T[]> = {}
  for (const item of items) {
    const groupKey = String(item[key])
    if (!result[groupKey]) result[groupKey] = []
    result[groupKey].push(item)
  }
  return result
}

interface Task {
  id: string
  status: "todo" | "in-progress" | "done"
}

declare const tasks: Task[]
const byStatus = groupBy(tasks, "status")
// Record<string, Task[]>
```

**要点**：运行时用 `String()` 转换键以确保对象键的正确性。

### 练习三

**思路**：用泛型元组表示函数链，每个函数的输出类型作为下一个函数的输入类型。

**答案**：

```typescript
function pipe<A, B>(ab: (a: A) => B): (a: A) => B
function pipe<A, B, C>(ab: (a: A) => B, bc: (b: B) => C): (a: A) => C
function pipe<A, B, C, D>(ab: (a: A) => B, bc: (b: B) => C, cd: (c: C) => D): (a: A) => D
function pipe(...fns: Function[]) {
  return (input: unknown) => fns.reduce((result, fn) => fn(result), input)
}

const processNumber = pipe(
  (x: number) => x * 2,
  (x) => x + 10,
  (x) => `结果是 ${x}`
)

const result = processNumber(5)
// 类型是 string
// 运行时: 5 → 10 → 20 → "结果是 20"
```

**要点**：函数重载让 TypeScript 自动推断中间步骤的类型。每个重载签名定义了不同长度的函数链。
