```typescript
// src/db/repositories/order.repository.ts
import { eq } from 'drizzle-orm'
import { ResultAsync, ok, err } from 'neverthrow'
import { Database } from '../client'
import { orders, orderItems } from '../schema'
import { DatabaseError, NotFoundError } from '../../errors'
import type { NewOrder, Order } from '../schema'

export class OrderRepository {
  constructor(private db: Database) {}

  findById(id: string): ResultAsync<Order, NotFoundError | DatabaseError> {
    return ResultAsync.fromPromise(
      this.db.query.orders.findFirst({
        where: eq(orders.id, id),
        with: { items: true },
      }),
      (e) => new DatabaseError('Failed to query order', { cause: e, orderId: id })
    ).andThen((order) =>
      order ? ok(order) : err(new NotFoundError('Order', id))
    )
  }

  create(data: {
    userId: string
    items: Array<{ productId: string; quantity: number; unitPrice: number }>
    shippingAddress: string
  }): ResultAsync<Order, DatabaseError> {
    return ResultAsync.fromPromise(
      this.db.transaction(async (tx) => {
        const totalAmount = data.items.reduce(
          (sum, item) => sum + item.quantity * item.unitPrice, 0
        )
        const [order] = await tx.insert(orders).values({
          userId: data.userId,
          totalAmount,
          shippingAddress: data.shippingAddress,
        }).returning()

        await tx.insert(orderItems).values(
          data.items.map((item) => ({
            orderId: order.id,
            productId: item.productId,
            quantity: item.quantity,
            unitPrice: item.unitPrice,
          }))
        )

        return order
      }),
      (e) => new DatabaseError('Failed to create order', { cause: e })
    )
  }

  updateStatus(
    id: string,
    status: 'pending' | 'paid' | 'shipped' | 'delivered' | 'cancelled'
  ): ResultAsync<Order, NotFoundError | DatabaseError> {
    return ResultAsync.fromPromise(
      this.db.update(orders)
        .set({ status, updatedAt: new Date() })
        .where(eq(orders.id, id))
        .returning()
        .then(([order]) => order),
      (e) => new DatabaseError('Failed to update order status', { cause: e, orderId: id })
    ).andThen((order) =>
      order ? ok(order) : err(new NotFoundError('Order', id))
    )
  }
}
```

```typescript
import { initTRPC } from '@trpc/server'
import crypto from 'crypto'
import { Context } from '../server'
import { requestContext } from '../logging'

export const t = initTRPC.context<Context>().create()

export const loggingMiddleware = t.middleware(async ({ path, type, next, ctx }) => {
  const traceId = crypto.randomUUID()
  
  return requestContext.run({ traceId }, async () => {
    const start = performance.now()
    ctx.logger.info({ path, type, traceId }, 'Request started')
    
    const result = await next()
    
    const duration = performance.now() - start
    ctx.logger.info({ path, type, duration: `${duration.toFixed(2)}ms` }, 'Request completed')
    
    return result
  })
})
```

## 五、业务逻辑与 Router

```typescript
// src/services/order.service.ts
import { ResultAsync } from 'neverthrow'
import { OrderRepository } from '../db/repositories/order.repository'
import { NotFoundError, DatabaseError, ValidationError, ForbiddenError } from '../errors'

export class OrderService {
  constructor(private orderRepo: OrderRepository) {}

  getOrder(id: string) {
    return this.orderRepo.findById(id)
  }

  createOrder(input: {
    userId: string
    items: Array<{ productId: string; quantity: number; unitPrice: number }>
    shippingAddress: string
  }): ResultAsync<any, ValidationError | DatabaseError> {
    if (input.items.length === 0) {
      return ResultAsync.fromPromise(
        Promise.reject(new ValidationError('At least one item required', { items: 'empty' })),
        (e) => e as ValidationError
      )
    }

    return this.orderRepo.create(input)
  }

  cancelOrder(
    orderId: string,
    userId: string
  ): ResultAsync<any, NotFoundError | ForbiddenError | DatabaseError> {
    return this.orderRepo.findById(orderId).andThen((order) => {
      if (order.userId !== userId) {
        return ResultAsync.fromPromise(
          Promise.reject(new ForbiddenError('cancel this order')),
          (e) => e as ForbiddenError
        )
      }
      if (order.status !== 'pending') {
        return ResultAsync.fromPromise(
          Promise.reject(new ValidationError('Only pending orders can be cancelled', {
            status: order.status,
          })),
          (e) => e as ValidationError
        )
      }
      return this.orderRepo.updateStatus(orderId, 'cancelled')
    })
  }
}
```

```typescript
// src/routers/order.router.ts
import { z } from 'zod'
import { t, loggingMiddleware } from '../middleware/logging'
import { protectedProcedure } from '../middleware/auth'
import { AppError } from '../errors'
import { TRPCError } from '@trpc/server'

const baseProcedure = t.procedure.use(loggingMiddleware)

export function createOrderRouter(orderService: OrderService) {
  return t.router({
    getById: baseProcedure
      .input(z.object({ id: z.string().uuid() }))
      .query(async ({ input }) => {
        const result = await orderService.getOrder(input.id)
        return result.match(
          (order) => order,
          (error) => {
            if (error.code === 'NOT_FOUND') {
              throw new TRPCError({ code: 'NOT_FOUND', message: error.message })
            }
            throw new TRPCError({ code: 'INTERNAL_SERVER_ERROR', message: 'Database error' })
          }
        )
      }),

    create: protectedProcedure
      .use(loggingMiddleware)
      .input(z.object({
        items: z.array(z.object({
          productId: z.string().uuid(),
          quantity: z.number().int().positive(),
          unitPrice: z.number().int().positive(),
        })).min(1).max(50),
        shippingAddress: z.string().min(10).max(500),
      }))
      .mutation(async ({ input, ctx }) => {
        const result = await orderService.createOrder({
          userId: ctx.userId,
          items: input.items,
          shippingAddress: input.shippingAddress,
        })

        return result.match(
          (order) => order,
          (error) => {
            if (error instanceof AppError) {
              throw new TRPCError({
                code: error.code === 'VALIDATION_ERROR' ? 'BAD_REQUEST' : 'INTERNAL_SERVER_ERROR',
                message: error.message,
              })
            }
            throw new TRPCError({ code: 'INTERNAL_SERVER_ERROR', message: 'Unknown error' })
          }
        )
      }),

    cancel: protectedProcedure
      .use(loggingMiddleware)
      .input(z.object({ id: z.string().uuid() }))
      .mutation(async ({ input, ctx }) => {
        const result = await orderService.cancelOrder(input.id, ctx.userId)

        return result.match(
          (order) => order,
          (error) => {
            if (error.code === 'NOT_FOUND') throw new TRPCError({ code: 'NOT_FOUND', message: error.message })
            if (error.code === 'FORBIDDEN') throw new TRPCError({ code: 'FORBIDDEN', message: error.message })
            throw new TRPCError({ code: 'BAD_REQUEST', message: error.message })
          }
        )
      }),
  })
}
```

## 六、服务器启动与优雅关闭

```typescript
// src/server.ts
import { createHTTPHandler } from '@trpc/server/adapters/standalone'
import http from 'http'
import { loadConfig } from './config'
import { createLogger } from './logging'
import { createDatabase } from './db/client'
import { OrderRepository } from './db/repositories/order.repository'
import { OrderService } from './services/order.service'
import { createOrderRouter } from './routers/order.router'
import { createAppRouter } from './routers'

export interface Context {
  logger: ReturnType<typeof createLogger>
  userId?: string
  req?: http.IncomingMessage
}

async function main() {
  const config = loadConfig()
  const logger = createLogger(config)
  const db = createDatabase(config.database)

  const orderRepo = new OrderRepository(db)
  const orderService = new OrderService(orderRepo)

  const appRouter = createAppRouter(orderService)

  const handler = createHTTPHandler({
    router: appRouter,
    createContext({ req }): Context {
      return { logger, req }
    },
  })

  const server = http.createServer((req, res) => {
    // 健康检查端点，不走 tRPC
    if (req.url === '/health') {
      res.writeHead(200, { 'Content-Type': 'application/json' })
      res.end(JSON.stringify({ status: 'ok', timestamp: new Date().toISOString() }))
      return
    }
    handler(req, res)
  })

  // 优雅关闭
  const shutdown = async (signal: string) => {
    logger.info({ signal }, 'Received shutdown signal, starting graceful shutdown')
    
    // 停止接受新连接
    server.close(async () => {
      logger.info('HTTP server closed')
      
      try {
        // 关闭数据库连接
        await db.$client.end()
        logger.info('Database connections closed')
        
        logger.info('Graceful shutdown completed')
        process.exit(0)
      } catch (error) {
        logger.fatal({ err: error }, 'Error during shutdown')
        process.exit(1)
      }
    })

    // 超时强制退出
    setTimeout(() => {
      logger.fatal('Shutdown timeout exceeded, forcing exit')
      process.exit(1)
    }, 30000)
  }

  process.on('SIGTERM', () => shutdown('SIGTERM'))
  process.on('SIGINT', () => shutdown('SIGINT'))

  server.listen(config.port, () => {
    logger.info({ port: config.port }, 'Server started')
  })
}

main().catch((error) => {
  console.fatal({ err: error }, 'Failed to start server')
  process.exit(1)
})
```

## 常见误区

1. **Repository 层直接抛异常**：Repository 应返回 Result，让 Service 层决定如何处理错误
2. **tRPC Procedure 里写业务逻辑**：Procedure 是薄薄的一层，只负责输入校验、调用 Service、错误映射
3. **日志打印整个请求体**：大请求体（如文件上传）会撑爆日志存储，应该只打印关键字段
4. **Docker 用 root 用户运行**：生产镜像必须用非 root 用户，避免容器逃逸风险

## 工程建议

1. **错误边界在 tRPC 中间件中统一收口**：所有 AppError 子类在中间件中映射为 TRPCError，避免在每个 Procedure 中重复处理
2. **健康检查端点不走 tRPC**：`/health` 直接用 HTTP 处理，避免不必要的中间件开销
3. **数据库连接池大小设为 CPU 核心数 × 2 + 磁盘数**：这是 PostgreSQL 的经验公式，避免连接过多导致数据库过载
4. **优雅关闭设置超时**：30 秒内必须完成清理，否则强制退出，避免被编排系统（K8s）强制杀死

### 练习一

**答案**：

```typescript
// src/db/schema.ts 新增
export const users = pgTable('users', {
  id: uuid('id').primaryKey().defaultRandom(),
  email: varchar('email', { length: 255 }).notNull().unique(),
  passwordHash: varchar('password_hash', { length: 255 }).notNull(),
  name: varchar('name', { length: 100 }).notNull(),
  createdAt: timestamp('created_at').notNull().defaultNow(),
})

// src/services/user.service.ts
import { ResultAsync, ok, err } from 'neverthrow'
import { ValidationError, NotFoundError } from '../errors'

export class UserService {
  constructor(private userRepo: UserRepository) {}

  register(input: { email: string; password: string; name: string }) {
    return this.userRepo.findByEmail(input.email).andThen((existing) => {
      if (existing) {
        return err(new ValidationError('Email already registered', { email: 'taken' }))
      }
      const passwordHash = hashPassword(input.password)
      return this.userRepo.create({ ...input, passwordHash })
    })
  }

  login(email: string, password: string) {
    return this.userRepo.findByEmail(email).andThen((user) => {
      if (!user || !verifyPassword(password, user.passwordHash)) {
        return err(new ValidationError('Invalid credentials', { email: 'invalid' }))
      }
      return ok({ id: user.id, email: user.email, name: user.name })
    })
  }
}
```

**要点**：
- 注册时先检查邮箱是否已存在，再创建用户
- 密码只存储 hash，不存储明文
- 登录成功返回不含密码的用户信息

```typescript
const rateLimitStore = new Map<string, { count: number; resetAt: number }>()

export const rateLimitMiddleware = t.middleware(async ({ ctx, next }) => {
  const userId = ctx.userId || 'anonymous'
  const now = Date.now()
  const windowMs = 60_000 // 1 分钟
  const maxRequests = 100

  let entry = rateLimitStore.get(userId)

  if (!entry || now > entry.resetAt) {
    entry = { count: 0, resetAt: now + windowMs }
    rateLimitStore.set(userId, entry)
  }

  entry.count++

  if (entry.count > maxRequests) {
    throw new TRPCError({
      code: 'TOO_MANY_REQUESTS',
      message: `Rate limit exceeded. Try again in ${Math.ceil((entry.resetAt - now) / 1000)}s`,
    })
  }

  return next()
})
```

**要点**：
- 每个用户独立计数，匿名用户共用一个计数器
- 窗口过期后自动重置，不需要定时清理
- 生产环境应使用 Redis 替代内存 Map，支持多实例共享

### 练习三

**思路**：用 Drizzle 的 better-sqlite3 驱动创建测试数据库，在 beforeAll 中初始化，每个测试后清理数据。

**答案**：
```typescript
import { describe, it, expect, beforeAll, afterEach } from 'vitest'
import { drizzle } from 'drizzle-orm/better-sqlite3'
import Database from 'better-sqlite3'
import { OrderRepository } from '../db/repositories/order.repository'
import { OrderService } from '../services/order.service'
import * as schema from '../db/schema'

describe('OrderService', () => {
  let orderService: OrderService
  let db: ReturnType<typeof drizzle>

  beforeAll(() => {
    const sqlite = new Database(':memory:')
    db = drizzle(sqlite, { schema })
    // 创建表结构
    sqlite.exec(`CREATE TABLE orders (
      id TEXT PRIMARY KEY,
      user_id TEXT NOT NULL,
      status TEXT NOT NULL DEFAULT 'pending',
      total_amount INTEGER NOT NULL,
      shipping_address TEXT NOT NULL,
      created_at TEXT NOT NULL,
      updated_at TEXT NOT NULL
    )`)
    sqlite.exec(`CREATE TABLE order_items (
      id TEXT PRIMARY KEY,
      order_id TEXT NOT NULL,
      product_id TEXT NOT NULL,
      quantity INTEGER NOT NULL,
      unit_price INTEGER NOT NULL
    )`)

    const orderRepo = new OrderRepository(db)
    orderService = new OrderService(orderRepo)
  })

  afterEach(() => {
    db.delete(schema.orders).run()
    db.delete(schema.orderItems).run()
  })

  it('should create an order', async () => {
    const result = await orderService.createOrder({
      userId: 'user-1',
      items: [{ productId: 'prod-1', quantity: 2, unitPrice: 1000 }],
      shippingAddress: '123 Main St',
    })

    expect(result.isOk()).toBe(true)
    if (result.isOk()) {
      expect(result.value.userId).toBe('user-1')
      expect(result.value.totalAmount).toBe(2000)
    }
  })

  it('should reject empty items', async () => {
    const result = await orderService.createOrder({
      userId: 'user-1',
      items: [],
      shippingAddress: '123 Main St',
    })

    expect(result.isErr()).toBe(true)
  })

  it('should cancel a pending order', async () => {
    const createResult = await orderService.createOrder({
      userId: 'user-1',
      items: [{ productId: 'prod-1', quantity: 1, unitPrice: 1000 }],
      shippingAddress: '123 Main St',
    })

    if (createResult.isOk()) {
      const cancelResult = await orderService.cancelOrder(createResult.value.id, 'user-1')
      expect(cancelResult.isOk()).toBe(true)
    }
  })
})
```

**要点**：
- SQLite 内存数据库不需要外部依赖，测试速度快
- 每个测试后清理数据，保证测试隔离性
- 用 `result.isOk()` / `result.isErr()` 断言 Result 类型
