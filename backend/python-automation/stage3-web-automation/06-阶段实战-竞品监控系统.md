# 阶段实战：竞品监控系统

## 场景引入

你的产品团队需要实时关注竞品网站的变化——价格调整、功能更新、文案修改。手动检查效率低且容易遗漏。你需要构建一个自动化系统：定时抓取竞品页面，检测内容变化，发现变更时自动发送邮件通知，并记录监控历史。

本项目综合运用本阶段所有技能：HTTP 请求、网页解析、邮件发送、配置管理。

## 学习目标

- 综合运用 requests、BeautifulSoup、smtplib 构建完整系统
- 实现基于 diff 的内容变更检测
- 设计配置驱动的监控任务
- 实现定时调度和历史记录

## 分步实现

### 第一步：配置加载

```python
import os, yaml
from dataclasses import dataclass, field

@dataclass
class Target:
    name: str; url: str; selector: str
    check_type: str = "text"; enabled: bool = True

@dataclass
class NotifyConfig:
    smtp_host: str; smtp_port: int; username: str; password: str; recipients: list[str]

@dataclass
class MonitorConfig:
    interval_minutes: int = 60; history_days: int = 30
    targets: list[Target] = field(default_factory=list)
    notification: NotifyConfig = None

def load_config(path: str = "config.yaml") -> MonitorConfig:
    with open(path, encoding="utf-8") as f:
        raw = yaml.safe_load(f)
    m = raw.get("monitor", {})
    targets = [Target(**t) for t in raw.get("targets", [])]
    n = raw.get("notification", {})
    pw = n.get("password", "")
    if pw.startswith("${") and pw.endswith("}"):
        pw = os.environ.get(pw[2:-1], "")
    return MonitorConfig(
        interval_minutes=m.get("interval_minutes", 60),
        history_days=m.get("history_days", 30), targets=targets,
        notification=NotifyConfig(n["smtp_host"], n["smtp_port"], n["username"], pw, n.get("recipients", [])),
    )
```

### 第二步：页面抓取

```python
import requests
from bs4 import BeautifulSoup

HEADERS = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/120.0.0.0"}

def fetch_content(url: str, selector: str, check_type: str = "text") -> str | None:
    try:
        resp = requests.get(url, headers=HEADERS, timeout=15)
        resp.raise_for_status()
        resp.encoding = resp.apparent_encoding
        el = BeautifulSoup(resp.text, "lxml").select_one(selector)
        return str(el) if el and check_type == "html" else (el.get_text("\n", strip=True) if el else None)
    except requests.RequestException as e:
        print(f"抓取失败 {url}: {e}")
        return None
```

### 第三步：变更检测

```python
import difflib
from pathlib import Path
from datetime import datetime

class ChangeDetector:
    def __init__(self, history_dir: str = "history"):
        self.dir = Path(history_dir)

    def get_latest(self, name: str) -> str | None:
        d = self.dir / name.replace(" ", "_").replace("/", "_")
        snaps = sorted(d.glob("*.txt"), reverse=True) if d.exists() else []
        return snaps[0].read_text(encoding="utf-8") if snaps else None

    def save(self, name: str, content: str):
        d = self.dir / name.replace(" ", "_").replace("/", "_")
        d.mkdir(parents=True, exist_ok=True)
        (d / f"{datetime.now():%Y-%m-%d_%H-%M}.txt").write_text(content, encoding="utf-8")

    def detect(self, name: str, new: str) -> str | None:
        old = self.get_latest(name)
        self.save(name, new)
        if not old or old == new:
            return None
        return "".join(difflib.unified_diff(old.splitlines(True), new.splitlines(True), fromfile="变更前", tofile="变更后"))

    def cleanup(self, name: str, keep_days: int = 30):
        d = self.dir / name.replace(" ", "_").replace("/", "_")
        cutoff = datetime.now().timestamp() - keep_days * 86400
        if d.exists():
            for f in d.glob("*.txt"):
                if f.stat().st_mtime < cutoff:
                    f.unlink()
```

### 第四步：邮件通知

```python
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from datetime import datetime
from jinja2 import Template

ALERT_TPL = Template("""<h2>竞品监控变更通知</h2>
<table border="1" cellpadding="8">
<tr><th>目标</th><td>{{ name }}</td></tr>
<tr><th>URL</th><td><a href="{{ url }}">{{ url }}</a></td></tr>
<tr><th>时间</th><td>{{ time }}</td></tr></table>
<h3>变更内容</h3><pre style="background:#f5f5f5;padding:15px;">{{ diff }}</pre>""")

class Notifier:
    def __init__(self, cfg): self.cfg = cfg

    def send_alert(self, name: str, url: str, diff: str) -> bool:
        html = ALERT_TPL.render(name=name, url=url, time=f"{datetime.now():%Y-%m-%d %H:%M}", diff=diff)
        msg = MIMEMultipart()
        msg["From"] = self.cfg.username
        msg["To"] = ", ".join(self.cfg.recipients)
        msg["Subject"] = f"[竞品监控] {name} 发生变更"
        msg.attach(MIMEText(html, "html", "utf-8"))
        try:
            with smtplib.SMTP_SSL(self.cfg.smtp_host, self.cfg.smtp_port) as s:
                s.login(self.cfg.username, self.cfg.password)
                s.sendmail(self.cfg.username, self.cfg.recipients, msg.as_string())
            return True
        except Exception as e:
            print(f"邮件失败: {e}")
            return False
```

### 第五步：主程序整合

```python
import time, logging

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[logging.FileHandler("monitor.log", encoding="utf-8"), logging.StreamHandler()])
logger = logging.getLogger(__name__)

class CompetitorMonitor:
    def __init__(self, config_path="config.yaml"):
        self.config = load_config(config_path)
        self.detector = ChangeDetector()
        self.notifier = Notifier(self.config.notification)

    def check(self, t: Target) -> bool:
        content = fetch_content(t.url, t.selector, t.check_type)
        if not content:
            return False
        diff = self.detector.detect(t.name, content)
        if diff:
            logger.info(f"检测到变更: {t.name}")
            self.notifier.send_alert(t.name, t.url, diff)
            return True
        return False

    def run_once(self):
        stats = {"checked": 0, "changed": 0, "errors": 0}
        for t in self.config.targets:
            if not t.enabled:
                continue
            stats["checked"] += 1
            try:
                if self.check(t): stats["changed"] += 1
            except Exception as e:
                logger.error(f"失败 {t.name}: {e}"); stats["errors"] += 1
        logger.info(f"本轮: 检查 {stats['checked']}，变更 {stats['changed']}，错误 {stats['errors']}")

    def run_loop(self):
        interval = self.config.interval_minutes * 60
        logger.info(f"启动监控，间隔 {self.config.interval_minutes} 分钟")
        while True:
            try:
                self.run_once()
                for t in self.config.targets:
                    self.detector.cleanup(t.name, self.config.history_days)
            except KeyboardInterrupt:
                logger.info("监控已停止"); break
            except Exception as e:
                logger.error(f"异常: {e}")
            time.sleep(interval)

if __name__ == "__main__":
    CompetitorMonitor().run_loop()
```

## 常见误区

### 1. 不清理历史快照

```python
# ❌ 历史文件无限增长，磁盘占满
# ✅ 定期清理过期快照
detector.cleanup(name, keep_days=30)
```

### 2. 选择器太精确

```python
# ❌ 微小变化就失效
selector = "div.container > div:nth-child(3) > div > h2"
# ✅ 稳定的语义化选择器
selector = "div.pricing-table"
```

### 3. 不记录运行状态

```python
# ❌ 崩溃了不知道发生了什么
# ✅ 文件和控制台双输出日志
logging.basicConfig(handlers=[logging.FileHandler("monitor.log"), logging.StreamHandler()])
```

## 工程建议

1. **配置驱动**：监控目标在 YAML 中管理，不修改代码
2. **快照留档**：每次抓取保存快照，便于回溯
3. **diff 通知**：只通知变化部分，不发完整页面
4. **优雅降级**：单个目标失败不影响其他目标
5. **环境变量存密码**：敏感信息不写入配置文件

## 参考答案

### 练习一

**思路**：提取数值型价格，比较新旧值计算变化幅度。

**答案**：

```python
import re

def extract_price(text: str) -> float | None:
    m = re.search(r"[\d,]+\.?\d*", text.replace(",", ""))
    return float(m.group()) if m else None

def check_price(name, url, selector, detector, notifier):
    content = fetch_content(url, selector, "text")
    if not content:
        return None
    new = extract_price(content)
    if new is None:
        return None
    old_text = detector.get_latest(name)
    detector.save(name, content)
    if not old_text:
        return None
    old = extract_price(old_text)
    if not old or old == new:
        return None
    pct = (new - old) / old * 100
    diff = f"{'降价' if pct < 0 else '涨价'}: ¥{old:.2f} → ¥{new:.2f} ({pct:+.1f}%)"
    notifier.send_alert(name, url, diff)
    return {"old": old, "new": new, "change_pct": pct}
```

**要点**：正则提取价格先去除逗号；百分比用 `:+.1f` 格式化正负号。

### 练习二

**思路**：读取历史快照目录，比较最新两个快照判断变更状态。

**答案**：

```python
from pathlib import Path
from flask import Flask, render_template_string

app = Flask(__name__)
HISTORY = Path("history")

HTML = """<h1>竞品监控仪表盘</h1>
<table border="1" cellpadding="8"><tr><th>目标</th><th>最后检查</th><th>状态</th></tr>
{% for t in targets %}<tr><td>{{ t.name }}</td><td>{{ t.last }}</td>
<td style="color:{{ 'green' if t.status=='无变更' else 'red' }}">{{ t.status }}</td></tr>
{% endfor %}</table>"""

def get_targets():
    result = []
    for d in HISTORY.iterdir():
        if not d.is_dir(): continue
        snaps = sorted(d.glob("*.txt"), reverse=True)
        name = d.name.replace("_", " ")
        if len(snaps) >= 2:
            status = "有变更" if snaps[1].read_text() != snaps[0].read_text() else "无变更"
        else:
            status = "首次检查"
        result.append({"name": name, "last": snaps[0].stem.replace("_", " ") if snaps else "无", "status": status})
    return result

@app.route("/")
def dashboard():
    return render_template_string(HTML, targets=get_targets())
```

**要点**：比较最新两个快照判断是否有变更；Flask 内置服务器足够监控场景使用。
