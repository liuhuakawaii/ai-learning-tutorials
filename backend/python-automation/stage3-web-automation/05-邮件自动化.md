# 邮件自动化

## 场景引入

你的团队每周五需要发送周报邮件，包含工作汇总、下周计划和关键指标。手动编写和发送耗时且容易遗漏收件人。你需要一个脚本自动拉取数据、渲染模板、添加附件并批量发送。Python 的 `smtplib` 和 `imaplib` 负责邮件收发，`Jinja2` 负责模板渲染。

## 学习目标

- 掌握 `smtplib` 发送纯文本、HTML 和带附件的邮件
- 能用 `imaplib` 接收和解析邮件
- 能使用 Jinja2 模板生成个性化内容
- 能实现批量发送和限流控制

## smtplib 发送邮件

### 纯文本邮件

```python
import smtplib
from email.mime.text import MIMEText

def send_text_email(host, port, username, password, to_addrs, subject, body):
    msg = MIMEText(body, "plain", "utf-8")
    msg["From"] = username
    msg["To"] = ", ".join(to_addrs)
    msg["Subject"] = subject
    with smtplib.SMTP_SSL(host, port) as server:
        server.login(username, password)
        server.sendmail(username, to_addrs, msg.as_string())
```

### 带附件的邮件

```python
from email.mime.base import MIMEBase
from email import encoders
from pathlib import Path

def send_with_attachment(host, port, username, password, to_addrs, subject, body, files):
    msg = MIMEMultipart()
    msg["From"] = username
    msg["To"] = ", ".join(to_addrs)
    msg["Subject"] = subject
    msg.attach(MIMEText(body, "plain", "utf-8"))
    for filepath in files:
        part = MIMEBase("application", "octet-stream")
        with open(filepath, "rb") as f:
            part.set_payload(f.read())
        encoders.encode_base64(part)
        part.add_header("Content-Disposition", "attachment", filename=Path(filepath).name)
        msg.attach(part)
    with smtplib.SMTP_SSL(host, port) as server:
        server.login(username, password)
        server.sendmail(username, to_addrs, msg.as_string())
```

## imaplib 接收邮件

```python
import imaplib
import email
from email.header import decode_header

def fetch_emails(host, username, password, folder="INBOX", limit=10):
    with imaplib.IMAP4_SSL(host) as mail:
        mail.login(username, password)
        mail.select(folder)
        _, ids = mail.search(None, "ALL")
        result = []
        for msg_id in ids[0].split()[-limit:]:
            _, data = mail.fetch(msg_id, "(RFC822)")
            msg = email.message_from_bytes(data[0][1])
            subject = decode_header(msg["Subject"])[0][0]
            if isinstance(subject, bytes):
                subject = subject.decode()
            from_addr = email.utils.parseaddr(msg["From"])[1]
            body = ""
            if msg.is_multipart():
                for part in msg.walk():
                    if part.get_content_type() == "text/plain":
                        body = part.get_payload(decode=True).decode()
                        break
            else:
                body = msg.get_payload(decode=True).decode()
            result.append({"subject": subject, "from": from_addr, "body": body[:500]})
    return result
```

## 批量发送与限流

```python
import time

def send_bulk(host, port, username, password, recipients, subject_tpl, body_tpl, interval=2.0):
    results = {"success": [], "failed": []}
    with smtplib.SMTP_SSL(host, port) as server:
        server.login(username, password)
        for r in recipients:
            try:
                msg = MIMEMultipart()
                msg["From"] = username
                msg["To"] = r["email"]
                msg["Subject"] = subject_tpl.format(**r)
                msg.attach(MIMEText(body_tpl.format(**r), "html", "utf-8"))
                server.sendmail(username, [r["email"]], msg.as_string())
                results["success"].append(r["email"])
            except Exception as e:
                results["failed"].append({"email": r["email"], "error": str(e)})
            time.sleep(interval)
    return results
```

`interval` 控制每封邮件间隔，避免触发 SMTP 限流。

## 完整示例：自动化周报发送系统

```python
import smtplib
import time
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from email.mime.base import MIMEBase
from email import encoders
from pathlib import Path
from jinja2 import Template
from dataclasses import dataclass, field

TEMPLATE = Template("""
<h2>{{ team }} 周报 — {{ week }}</h2>
<h3>本周完成</h3><ul>{% for i in completed %}<li>{{ i }}</li>{% endfor %}</ul>
<h3>下周计划</h3><ul>{% for i in planned %}<li>{{ i }}</li>{% endfor %}</ul>
{% if metrics %}<h3>关键指标</h3><table border="1" cellpadding="6">
{% for k, v in metrics.items() %}<tr><td>{{ k }}</td><td>{{ v }}</td></tr>{% endfor %}
</table>{% endif %}
<p style="color:#aaa;font-size:11px">此邮件由自动化脚本生成</p>
""")

@dataclass
class EmailConfig:
    smtp_host: str; smtp_port: int; username: str; password: str

@dataclass
class WeeklyReport:
    team: str; week: str
    completed: list[str]; planned: list[str]
    metrics: dict = field(default_factory=dict)
    attachments: list[str] = field(default_factory=list)

class WeeklyReportSender:
    def __init__(self, config: EmailConfig):
        self.config = config

    def send(self, report: WeeklyReport, to_addrs: list[str], interval: float = 2.0) -> dict:
        html = TEMPLATE.render(team=report.team, week=report.week,
                               completed=report.completed, planned=report.planned,
                               metrics=report.metrics)
        results = {"success": [], "failed": []}
        with smtplib.SMTP_SSL(self.config.smtp_host, self.config.smtp_port) as server:
            server.login(self.config.username, self.config.password)
            for addr in to_addrs:
                msg = MIMEMultipart()
                msg["From"] = self.config.username
                msg["To"] = addr
                msg["Subject"] = f"{report.team} 周报 — {report.week}"
                msg.attach(MIMEText(html, "html", "utf-8"))
                for fp in report.attachments:
                    part = MIMEBase("application", "octet-stream")
                    with open(fp, "rb") as f:
                        part.set_payload(f.read())
                    encoders.encode_base64(part)
                    part.add_header("Content-Disposition", "attachment", filename=Path(fp).name)
                    msg.attach(part)
                try:
                    server.sendmail(self.config.username, [addr], msg.as_string())
                    results["success"].append(addr)
                except Exception as e:
                    results["failed"].append({"email": addr, "error": str(e)})
                time.sleep(interval)
        return results

if __name__ == "__main__":
    config = EmailConfig("smtp.example.com", 465, "weekly@example.com", "your-password")
    report = WeeklyReport("后端组", "2025 第 3 周",
        ["完成认证重构", "修复支付超时"], ["开发通知系统", "接入监控告警"],
        {"API 可用性": "99.95%", "平均响应": "120ms"},
        ["reports/week3_data.xlsx"])
    result = WeeklyReportSender(config).send(report, ["leader@example.com", "team@example.com"])
    print(f"成功 {len(result['success'])}，失败 {len(result['failed'])}")
```

## 常见误区

### 1. 密码硬编码

```python
# ❌ 暴露在代码仓库
password = "my-email-password"
# ✅ 从环境变量读取
import os
password = os.environ["EMAIL_PASSWORD"]
```

### 2. 批量发送无间隔

```python
# ❌ 连续发送触发限流
for addr in recipients:
    server.sendmail(...)
# ✅ 加间隔
time.sleep(2)
```

### 3. 忘记处理中文编码

```python
# ❌ 主题乱码
msg["Subject"] = "周报"
# ✅ 自动编码
msg = MIMEText(body, "html", "utf-8")
```

## 工程建议

1. **使用应用专用密码**：Gmail 等需要生成应用专用密码
2. **HTML 邮件用内联样式**：邮件客户端 CSS 支持有限
3. **附件控制在 25MB 以内**：大文件用云链接替代
4. **记录发送日志**：追踪每封邮件状态

## 参考答案

### 练习一

**思路**：imaplib 搜索未读邮件，匹配关键词后 smtplib 回复。

**答案**：

```python
import imaplib, smtplib, email
from email.mime.text import MIMEText
from email.header import decode_header

def auto_reply(imap_host, smtp_host, smtp_port, user, password, keywords, reply_body):
    replied = []
    with imaplib.IMAP4_SSL(imap_host) as mail:
        mail.login(user, password)
        mail.select("INBOX")
        _, ids = mail.search(None, "UNSEEN")
        for msg_id in ids[0].split():
            _, data = mail.fetch(msg_id, "(RFC822)")
            msg = email.message_from_bytes(data[0][1])
            subject_raw = decode_header(msg["Subject"])[0][0]
            subject = subject_raw.decode() if isinstance(subject_raw, bytes) else subject_raw
            body = ""
            if msg.is_multipart():
                for part in msg.walk():
                    if part.get_content_type() == "text/plain":
                        body = part.get_payload(decode=True).decode()
                        break
            else:
                body = msg.get_payload(decode=True).decode()
            if any(kw.lower() in f"{subject} {body}".lower() for kw in keywords):
                from_addr = email.utils.parseaddr(msg["From"])[1]
                reply = MIMEText(reply_body, "plain", "utf-8")
                reply["From"] = user
                reply["To"] = from_addr
                reply["Subject"] = f"Re: {subject}"
                reply["In-Reply-To"] = msg["Message-ID"]
                with smtplib.SMTP_SSL(smtp_host, smtp_port) as server:
                    server.login(user, password)
                    server.sendmail(user, [from_addr], reply.as_string())
                replied.append(from_addr)
    return replied
```

**要点**：`In-Reply-To` 确保回复关联原始邮件。

### 练习二

**思路**：用 Jinja2 FileSystemLoader 从目录加载模板。

**答案**：

```python
from jinja2 import Environment, FileSystemLoader
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

class EmailTemplateEngine:
    def __init__(self, template_dir: str = "templates"):
        self.env = Environment(loader=FileSystemLoader(template_dir), autoescape=True)

    def render(self, template_name: str, **context) -> MIMEMultipart:
        tpl = self.env.get_template(template_name)
        html = tpl.render(**context)
        msg = MIMEMultipart("alternative")
        msg.attach(MIMEText(html, "html", "utf-8"))
        try:
            text_tpl = self.env.get_template(template_name.replace(".html", ".txt"))
            msg.attach(MIMEText(text_tpl.render(**context), "plain", "utf-8"))
        except Exception:
            pass
        return msg
```

**要点**：`autoescape=True` 防止 XSS；尝试加载同名 `.txt` 作为纯文本版本。
