# 阶段实战：批量文件重命名工具

## 要解决的问题

前五课分别学了脚本思维、环境管理、命令行参数、日志、配置文件。现在把它们综合起来，构建一个真正有用的工具。

批量文件重命名是经典场景：摄影爱好者要把 `IMG_20240115_143022.jpg` 改成 `旅行_2024-01-15_001.jpg`；运营要把素材按日期归档。每个场景规则不同，但核心逻辑一样：遍历文件、匹配规则、执行重命名。

## 日志配置

```python
# src/file_renamer/logger.py
import logging
import sys
from pathlib import Path
from logging.handlers import RotatingFileHandler

def setup_logger(name: str = "file-renamer", log_dir: str = "logs") -> logging.Logger:
    logger = logging.getLogger(name)
    logger.setLevel(logging.DEBUG)
    if logger.handlers:
        return logger

    detailed = logging.Formatter(
        "%(asctime)s [%(levelname)-8s] %(funcName)s:%(lineno)d - %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
    )
    simple = logging.Formatter("%(asctime)s [%(levelname)-8s] %(message)s", datefmt="%H:%M:%S")

    console = logging.StreamHandler(sys.stdout)
    console.setLevel(logging.INFO)
    console.setFormatter(simple)
    logger.addHandler(console)

    log_path = Path(log_dir)
    log_path.mkdir(exist_ok=True)
    fh = RotatingFileHandler(log_path / f"{name}.log", maxBytes=5*1024*1024,
                              backupCount=3, encoding="utf-8")
    fh.setLevel(logging.DEBUG)
    fh.setFormatter(detailed)
    logger.addHandler(fh)

    return logger
```

控制台只显示 INFO 以上，文件记录 DEBUG 细节。`RotatingFileHandler` 防止日志文件无限增长。

## 配置管理

```python
# src/file_renamer/config.py
import json
from copy import deepcopy
from pathlib import Path

DEFAULT_CONFIG = {
    "directory": ".", "mode": "regex", "dry_run": True,
    "recursive": False, "file_filter": "*",
    "regex": {"pattern": "", "replacement": ""},
    "sequence": {"prefix": "", "start": 1, "step": 1, "pad_width": 3, "suffix": ""},
    "date_prefix": {"format": "%Y-%m-%d", "use_modified_time": True},
    "logging": {"level": "INFO", "dir": "logs"},
}

def deep_merge(base: dict, override: dict) -> dict:
    result = deepcopy(base)
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = deepcopy(value)
    return result

def load_config(config_path: str | None = None, preset: str | None = None) -> dict:
    config = deepcopy(DEFAULT_CONFIG)
    if preset:
        preset_path = Path("config/presets") / f"{preset}.json"
        if preset_path.exists():
            config = deep_merge(config, json.loads(preset_path.read_text(encoding="utf-8")))
    if config_path and Path(config_path).exists():
        config = deep_merge(config, json.loads(Path(config_path).read_text(encoding="utf-8")))
    return config
```

`deep_merge` 让用户只需配置差异项，其余走默认值。预设文件可以针对不同场景（照片、文档）提供不同默认配置。

## 核心重命名逻辑

```python
# src/file_renamer/renamer.py
import re, logging
from datetime import datetime
from pathlib import Path

logger = logging.getLogger("file-renamer")

class RenameAction:
    def __init__(self, source: Path, target: Path):
        self.source, self.target = source, target
    def __repr__(self) -> str:
        return f"{self.source.name} → {self.target.name}"

def collect_files(directory: str, pattern: str = "*", recursive: bool = False) -> list[Path]:
    target_dir = Path(directory)
    if not target_dir.exists():
        raise FileNotFoundError(f"目录不存在: {directory}")
    files = list(target_dir.rglob(pattern) if recursive else target_dir.glob(pattern))
    files = [f for f in files if f.is_file() and not f.name.startswith(".")]
    logger.info(f"找到 {len(files)} 个文件")
    return sorted(files)

def generate_regex_names(files: list[Path], pattern: str, replacement: str) -> list[RenameAction]:
    regex = re.compile(pattern)
    actions = [RenameAction(f, f.parent / regex.sub(replacement, f.name))
               for f in files if regex.search(f.name)]
    logger.info(f"正则匹配: {len(actions)}/{len(files)} 个文件")
    return actions

def generate_sequence_names(files: list[Path], prefix: str = "",
                            start: int = 1, pad_width: int = 3) -> list[RenameAction]:
    actions = [RenameAction(f, f.parent / f"{prefix}{str(start+i).zfill(pad_width)}{f.suffix}")
               for i, f in enumerate(files)]
    return actions

def generate_date_prefix_names(files: list[Path], date_format: str = "%Y-%m-%d") -> list[RenameAction]:
    actions = []
    for f in files:
        date_str = datetime.fromtimestamp(f.stat().st_mtime).strftime(date_format)
        new_name = f"{date_str}_{f.name}"
        if new_name != f.name:
            actions.append(RenameAction(f, f.parent / new_name))
    return actions

def execute_rename(actions: list[RenameAction], dry_run: bool = True) -> dict:
    success, failed = 0, 0
    for action in actions:
        if action.target.exists():
            logger.warning(f"跳过(目标已存在): {action.target.name}")
            failed += 1; continue
        if dry_run:
            logger.info(f"[预览] {action}")
        else:
            try:
                action.source.rename(action.target); success += 1
                logger.info(f"[执行] {action}")
            except OSError as e:
                logger.error(f"失败 {action.source.name}: {e}"); failed += 1
    return {"success": success, "failed": failed, "total": len(actions)}
```

三种模式覆盖常见场景：正则替换处理复杂规则，序号填充处理批量编号，日期前缀处理归档。每种模式返回 `RenameAction` 列表，`execute_rename` 统一处理执行或预览。

## CLI 入口

```python
# src/file_renamer/cli.py
import argparse, logging
from .logger import setup_logger
from .config import load_config
from .renamer import (collect_files, generate_regex_names,
                      generate_sequence_names, generate_date_prefix_names,
                      execute_rename)

def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description="批量文件重命名工具")
    parser.add_argument("directory", nargs=".", default=".", help="目标目录")
    parser.add_argument("--mode", choices=["regex", "sequence", "date-prefix"], default="regex")
    parser.add_argument("--config", help="配置文件路径")
    parser.add_argument("--preset", help="使用预设 (photo/document)")
    parser.add_argument("--execute", action="store_true", help="执行重命名（默认只预览）")
    parser.add_argument("--recursive", "-r", action="store_true")
    parser.add_argument("--pattern", help="文件匹配模式 (glob)")

    regex_g = parser.add_argument_group("正则模式")
    regex_g.add_argument("--regex-pattern", help="正则表达式")
    regex_g.add_argument("--replacement", help="替换字符串")

    seq_g = parser.add_argument_group("序号模式")
    seq_g.add_argument("--prefix", default="", help="序号前缀")
    seq_g.add_argument("--start", type=int, default=1)
    seq_g.add_argument("--pad-width", type=int, default=3)

    parser.add_argument("--date-format", default="%Y-%m-%d")
    parser.add_argument("--log-level", default="INFO", choices=["DEBUG", "INFO", "WARNING", "ERROR"])
    return parser

def main(args=None):
    parsed = build_parser().parse_args(args)
    config = load_config(parsed.config, parsed.preset)
    config.update({"directory": parsed.directory, "mode": parsed.mode, "dry_run": not parsed.execute})

    logger = setup_logger(log_dir=config["logging"]["dir"],
                          console_level=getattr(logging, parsed.log_level))
    logger.info(f"模式: {config['mode']}, 预览: {config['dry_run']}")

    files = collect_files(config["directory"], config["file_filter"], parsed.recursive)
    if not files:
        logger.warning("没有找到匹配的文件"); return

    if config["mode"] == "regex":
        actions = generate_regex_names(files, parsed.regex_pattern or config["regex"]["pattern"],
                                       parsed.replacement or config["regex"]["replacement"])
    elif config["mode"] == "sequence":
        actions = generate_sequence_names(files, parsed.prefix, parsed.start, parsed.pad_width)
    else:
        actions = generate_date_prefix_names(files, parsed.date_format)

    if not actions:
        logger.info("没有需要重命名的文件"); return

    print(f"\n{'='*50}\n预览 ({len(actions)} 个操作):")
    for action in actions[:20]:
        print(f"  {action}")
    if len(actions) > 20:
        print(f"  ... 还有 {len(actions) - 20} 个操作")

    if not config["dry_run"]:
        if input(f"\n确认执行? (y/n): ").strip().lower() != "y":
            logger.info("用户取消"); return

    result = execute_rename(actions, dry_run=config["dry_run"])
    if config["dry_run"]:
        print(f"\n预览完成: {result['total']} 个文件将被重命名，使用 --execute 执行")
```

预览模式是默认行为，用户必须显式传 `--execute`。这是文件操作工具的安全规范。

## 使用方式

```bash
# 正则模式：空格替换为下划线
python -m file_renamer ./docs --mode regex --regex-pattern " " --replacement "_"

# 序号模式：添加前缀和编号
python -m file_renamer ./docs --mode sequence --prefix report_ --pad-width 3

# 日期前缀模式
python -m file_renamer ./photos --mode date-prefix --date-format "%Y%m%d"

# 清理临时文件
python -m file_renamer ./project --mode regex --regex-pattern "\.tmp$" -r --execute
```

## 常见陷阱

**不做预览直接执行**：批量重命名不可逆，必须先预览。

**不检查目标文件是否已存在**：直接覆盖导致数据丢失。`execute_rename` 中已处理。

**正则写错导致所有文件同名**：如 `--regex-pattern ".*" --replacement "file"` 会把所有文件改名为 `file`。在预览阶段就能发现。

## 练习

### 练习一：添加撤销功能

执行重命名时将操作记录保存到 `rename_history.json`，实现 `--undo` 命令回滚上一次操作。

### 练习二：冲突检测

在执行前检查：目标文件名是否已被占用、多个源文件是否指向同一目标、正则替换后文件名是否为空。

## 参考答案

### 练习一

```python
import json
from datetime import datetime
from pathlib import Path

HISTORY_FILE = "rename_history.json"

def save_history(actions: list):
    record = {"timestamp": datetime.now().isoformat(),
              "operations": [{"source": str(a.source), "target": str(a.target)} for a in actions]}
    history = json.loads(Path(HISTORY_FILE).read_text(encoding="utf-8")) if Path(HISTORY_FILE).exists() else []
    history.append(record)
    Path(HISTORY_FILE).write_text(json.dumps(history, ensure_ascii=False, indent=2), encoding="utf-8")

def undo_last() -> int:
    path = Path(HISTORY_FILE)
    if not path.exists():
        print("没有历史记录"); return 0
    history = json.loads(path.read_text(encoding="utf-8"))
    if not history:
        return 0
    last = history.pop()
    undone = 0
    for op in reversed(last["operations"]):
        src, tgt = Path(op["target"]), Path(op["source"])
        if src.exists():
            src.rename(tgt); undone += 1
    path.write_text(json.dumps(history, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"撤销完成: {undone} 个操作")
    return undone
```

历史记录用列表存储多次操作，反向遍历避免文件名冲突。

### 练习二

```python
def check_conflicts(actions: list) -> list[str]:
    errors, targets = [], {}
    for action in actions:
        tgt = action.target
        if not tgt.name:
            errors.append(f"文件名为空: {action.source.name}"); continue
        if tgt.exists() and tgt not in [a.source for a in actions]:
            errors.append(f"目标已存在: {tgt.name}"); continue
        tgt_str = str(tgt)
        if tgt_str in targets:
            errors.append(f"冲突: {action.source.name} 和 {targets[tgt_str]} → {tgt.name}")
        else:
            targets[tgt_str] = action.source.name
    return errors
```

冲突检测要在执行前完成，不能边执行边检测。
