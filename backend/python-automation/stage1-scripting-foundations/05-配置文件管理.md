# 配置文件管理

## 场景引入

你写了一个数据同步脚本，里面硬编码了数据库地址、API 密钥、同步频率。部署到服务器后，需要改数据库地址——你打开代码改了第 15 行。过两天 API 密钥过期了——又改第 18 行。同事要在另一台服务器上跑这个脚本，他又复制了一份代码改了第 15 行。

现在你们有三份几乎一样但又不完全一样的代码。噩梦开始了。

配置和代码分离是工程化的基本原则。本课教你用 Python 管理配置：YAML/JSON/TOML 文件读写、环境变量、多环境配置合并策略。

## 学习目标

- 掌握 YAML、JSON、TOML 三种配置格式的读写
- 学会使用环境变量和 python-dotenv
- 理解配置合并策略：默认值 → 配置文件 → 环境变量 → 命令行参数

### 一、配置文件格式对比

| 格式 | 可读性 | 注释支持 | 生态 |
|------|--------|---------|------|
| JSON | 一般 | 不支持 | 最广泛 |
| YAML | 好 | 支持 | 广泛 |
| TOML | 好 | 支持 | Python 社区首选 |

#### JSON 配置

```python
import json
from pathlib import Path

def load_json_config(filepath: str) -> dict:
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"配置文件不存在: {filepath}")
    return json.loads(path.read_text(encoding="utf-8"))

def save_json_config(config: dict, filepath: str):
    path = Path(filepath)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(config, ensure_ascii=False, indent=2), encoding="utf-8")

# config.json 示例：
# {
#   "database": {"host": "localhost", "port": 5432, "name": "mydb"},
#   "api": {"timeout": 30, "retry_count": 3}
# }
```

import sys
if sys.version_info >= (3, 11):
    import tomllib
else:
    import tomli as tomllib

from pathlib import Path

def load_toml_config(filepath: str) -> dict:
    path = Path(filepath)

### 二、环境变量

环境变量适合存储敏感信息和环境相关配置。用 `python-dotenv` 从 `.env` 文件自动加载：

```bash
pip install python-dotenv
```

```python
import os
from dotenv import load_dotenv

load_dotenv()  # 加载 .env 文件（DB_HOST, API_KEY 等）

def get_config() -> dict:
    return {
        "db_host": os.getenv("DB_HOST", "localhost"),
        "db_port": int(os.getenv("DB_PORT", "5432")),
        "api_key": os.getenv("API_KEY", ""),
        "debug": os.getenv("DEBUG", "false").lower() == "true",
    }
```

**重要**：`.env` 包含敏感信息，绝对不能提交到 Git。在 `.gitignore` 中添加 `.env`。

### 三、配置合并策略

生产级脚本的配置来自多个来源，优先级从低到高：

```
默认值（代码中）→ 配置文件 → 环境变量 → 命令行参数
```

核心是 `deep_merge` 函数——深度合并两个字典，override 中的值覆盖 base 中的值：

```python
import json, os
from copy import deepcopy
from pathlib import Path

def deep_merge(base: dict, override: dict) -> dict:
    result = deepcopy(base)
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = deepcopy(value)
    return result

def load_config(config_path: str | None = None) -> dict:
    config = {"database": {"host": "localhost", "port": 5432}, "logging": {"level": "INFO"}}

    if config_path and Path(config_path).exists():
        config = deep_merge(config, json.loads(Path(config_path).read_text(encoding="utf-8")))

    env_overrides = {}
    if os.getenv("APP_DB_HOST"):
        env_overrides.setdefault("database", {})["host"] = os.getenv("APP_DB_HOST")
    if os.getenv("APP_DB_PORT"):
        env_overrides.setdefault("database", {})["port"] = int(os.getenv("APP_DB_PORT"))

    return deep_merge(config, env_overrides)
```

### 四、完整示例：多环境配置管理

```python
import json
import os
from copy import deepcopy
from pathlib import Path

class ConfigManager:
    """配置管理器，支持多环境和多层合并。"""

    def __init__(self, app_name: str, config_dir: str = "config"):
        self.app_name = app_name
        self.config_dir = Path(config_dir)
        self._config: dict = {}
        self._loaded = False

    def load(self, env: str | None = None) -> "ConfigManager":
        if env is None:
            env = os.getenv(f"{self.app_name.upper()}_ENV", "dev")

        # 1. 基础配置
        self._config = self._load_file(self.config_dir / "base.json")

        # 2. 环境配置覆盖
        env_config = self._load_file(self.config_dir / f"{env}.json")
        self._config = self._deep_merge(self._config, env_config)

        # 3. 本地覆盖（不提交到 Git）
        local_config = self._load_file(self.config_dir / "local.json")
        self._config = self._deep_merge(self._config, local_config)

        # 4. 环境变量覆盖
        env_prefix = f"{self.app_name.upper()}_"
        env_overrides = {}
        for key, value in os.environ.items():
            if key.startswith(env_prefix):
                parts = key[len(env_prefix):].lower().split("__")
                current = env_overrides
                for part in parts[:-1]:
                    current = current.setdefault(part, {})
                current[parts[-1]] = value
        self._config = self._deep_merge(self._config, env_overrides)

        self._loaded = True
        return self

    def get(self, key: str, default=None):
        """获取配置值，支持点号路径如 'database.host'。"""
        if not self._loaded:
            raise RuntimeError("配置未加载，请先调用 load()")
        keys = key.split(".")
        value = self._config
        for k in keys:
            if isinstance(value, dict) and k in value:
                value = value[k]
            else:
                return default
        return value

    def _load_file(self, path: Path) -> dict:
        if not path.exists():
            return {}
        return json.loads(path.read_text(encoding="utf-8"))

    def _deep_merge(self, base: dict, override: dict) -> dict:
        result = deepcopy(base)
        for key, value in override.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                result[key] = self._deep_merge(result[key], value)
            else:
                result[key] = deepcopy(value)
        return result

# 使用示例
if __name__ == "__main__":
    # config/base.json: {"database": {"host": "localhost", "port": 5432}}
    # config/prod.json: {"database": {"host": "prod-db.example.com"}}

    config = ConfigManager("myscript")
    config.load("prod")
    print(f"数据库: {config.get('database.host')}:{config.get('database.port')}")
    print(f"不存在: {config.get('not.exist', '默认值')}")
```

环境变量命名规则：`{APP名}_{层级}__{层级}__{键}`，双下划线表示嵌套。

```bash
export MYSCRIPT_DATABASE__HOST=override-db.example.com
```

## 常见误区

**误区一：硬编码配置** — 改配置就要改代码，改代码就要重新部署。

**误区二：把密钥提交到 Git** — `.env` 文件应该用 `.gitignore` 排除。

**误区三：只支持一种配置来源** — 好的工具应支持文件、环境变量、命令行参数多种来源。

**误区四：不做类型转换** — 环境变量永远是字符串，读取数字要 `int()`，布尔值要手动判断。

## 工程建议

1. 敏感信息用环境变量，不要写在配置文件中
2. 实现 dry-run 模式，让用户预览配置值
3. 配置要有默认值，用户只覆盖需要修改的部分
4. 提供 `--config` 参数允许指定配置文件路径

## 小结

- 配置和代码分离是工程化的基本原则
- JSON 兼容性最好，TOML 可读性最好
- 环境变量适合存储敏感信息，python-dotenv 简化 .env 使用
- 配置合并优先级：默认值 < 配置文件 < 环境变量 < 命令行参数

### 练习一

**答案**：

```python
import json
from copy import deepcopy
from pathlib import Path

try:
    import yaml
    HAS_YAML = True
except ImportError:
    HAS_YAML = False

def deep_merge(base: dict, override: dict) -> dict:
    result = deepcopy(base)
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = deepcopy(value)
    return result

def load_config(filepath: str) -> dict:
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"配置文件不存在: {filepath}")
    content = path.read_text(encoding="utf-8")
    ext = path.suffix.lower()

    if ext in (".json",):
        return json.loads(content)
    elif ext in (".yaml", ".yml"):
        if not HAS_YAML:
            raise ImportError("需要安装 pyyaml: pip install pyyaml")
        return yaml.safe_load(content)
    else:
        raise ValueError(f"不支持的格式: {ext}")

def load_and_merge(filepaths: list[str]) -> dict:
    result = {}
    for filepath in filepaths:
        result = deep_merge(result, load_config(filepath))
    return result
```

**要点**：`deep_merge` 确保嵌套字典被正确合并而非直接覆盖。

### 练习二

**答案**：

```python
from typing import Any

SCHEMA = {
    "database": {"type": dict, "required": True, "schema": {
        "host": {"type": str, "required": True},
        "port": {"type": int, "required": True, "min": 1, "max": 65535},
    }},
    "log_level": {"type": str, "choices": ["DEBUG", "INFO", "WARNING", "ERROR"]},
}

def validate_config(config: dict, schema: dict) -> list[str]:
    errors = []
    for key, rules in schema.items():
        if "schema" in rules:
            if key not in config:
                if rules.get("required"): errors.append(f"缺少: {key}")
                continue
            errors.extend(validate_config(config[key], rules["schema"]))
        elif "type" in rules:
            if key not in config:
                if rules.get("required"): errors.append(f"缺少: {key}")
                continue
            if not isinstance(config[key], rules["type"]):
                errors.append(f"{key}: 类型错误"); continue
            if "min" in rules and config[key] < rules["min"]:
                errors.append(f"{key}: 小于 {rules['min']}")
            if "choices" in rules and config[key] not in rules["choices"]:
                errors.append(f"{key}: 不在 {rules['choices']} 内")
    return errors
```

**要点**：返回错误列表而非直接抛出异常，一次性报告所有错误。
