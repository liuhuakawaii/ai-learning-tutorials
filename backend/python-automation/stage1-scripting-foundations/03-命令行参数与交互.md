# 命令行参数与交互

## 场景引入

你写了一个批量图片压缩脚本，同事想用。你告诉他："打开 `compress.py`，把第 15 行的路径改成你的目录，第 18 行的质量改成 80，然后运行。" 改代码来使用工具？显然不是正确的方式。

一个专业的命令行工具应该让用户通过参数控制行为，而不是修改源代码。本课教你用 Python 构建用户友好的 CLI——从 `sys.argv` 到 `argparse`，再到 `click` 框架。

## 学习目标

- 掌握 argparse 的基础和高级用法
- 了解 click 框架的核心概念
- 学会用 rich 美化终端输出
- 理解三种方案的适用场景

### 一、sys.argv：最原始的方式

```python
#!/usr/bin/env python3
import sys

def main():
    if len(sys.argv) < 3:
        print(f"用法: {sys.argv[0]} <源目录> <目标目录>")
        sys.exit(1)

    source, target = sys.argv[1], sys.argv[2]
    print(f"从 {source} 复制到 {target}")

if __name__ == "__main__":
    main()
```

问题：没有参数说明、没有类型检查、没有可选参数。超过 2 个参数就不够用了。

### 二、argparse：标准库方案

#### 基础用法

```python
#!/usr/bin/env python3
import argparse
from pathlib import Path

def compress_image(input_path: str, quality: int, output_dir: str | None = None):
    source = Path(input_path)
    if not source.exists():
        raise FileNotFoundError(f"文件不存在: {input_path}")
    target_dir = Path(output_dir) if output_dir else source.parent / "compressed"
    target_dir.mkdir(exist_ok=True)
    print(f"压缩 {source.name} → 质量={quality}, 输出={target_dir / source.name}")

def main():
    parser = argparse.ArgumentParser(
        description="批量压缩图片",
        epilog="示例: python compress.py ./photos -q 80 -o ./compressed"
    )
    parser.add_argument("input", help="输入文件或目录路径")
    parser.add_argument("-q", "--quality", type=int, default=85, help="压缩质量 (1-100)")
    parser.add_argument("-o", "--output", default=None, help="输出目录")
    parser.add_argument("-v", "--verbose", action="store_true", help="显示详细输出")

    args = parser.parse_args()
    if args.verbose:
        print(f"输入: {args.input}, 质量: {args.quality}")
    compress_image(args.input, args.quality, args.output)

if __name__ == "__main__":
    main()
```

#### 子命令

类似 `git add`、`git commit` 的组织方式：

```python
#!/usr/bin/env python3
import argparse
from pathlib import Path

def cmd_list(args):
    for f in sorted(Path(args.directory).iterdir()):
        size = f.stat().st_size if f.is_file() else "-"
        print(f"  {f.name:40s} {size}")

def cmd_clean(args):
    target = Path(args.directory)
    for ext in args.extensions.split(","):
        for f in target.glob(f"*.{ext.strip()}"):
            if args.dry_run:
                print(f"[预览] 删除: {f.name}")
            else:
                f.unlink()
                print(f"已删除: {f.name}")

def cmd_rename(args):
    target = Path(args.directory)
    for f in target.iterdir():
        if f.is_file() and args.find in f.name:
            new_name = f.name.replace(args.find, args.replace)
            if args.dry_run:
                print(f"[预览] {f.name} → {new_name}")
            else:
                f.rename(f.parent / new_name)

def main():
    parser = argparse.ArgumentParser(description="文件管理工具")
    sub = parser.add_subparsers(dest="command", help="可用命令")

    lp = sub.add_parser("list", help="列出文件")
    lp.add_argument("directory", nargs=".", default=".")

    cp = sub.add_parser("clean", help="清理临时文件")
    cp.add_argument("directory", nargs=".", default=".")
    cp.add_argument("-e", "--extensions", default="tmp,log,bak")
    cp.add_argument("--dry-run", action="store_true")

    rp = sub.add_parser("rename", help="批量重命名")
    rp.add_argument("directory", nargs=".", default=".")
    rp.add_argument("--find", required=True)
    rp.add_argument("--replace", required=True)
    rp.add_argument("--dry-run", action="store_true")

    args = parser.parse_args()
    if not args.command:
        parser.print_help()
        return
    {"list": cmd_list, "clean": cmd_clean, "rename": cmd_rename}[args.command](args)

if __name__ == "__main__":
    main()
```

### 三、click：第三方框架

Click 用装饰器构建 CLI，代码更简洁：

```bash
pip install click
```

```python
#!/usr/bin/env python3
import click
from pathlib import Path

@click.group()
@click.version_option(version="1.0.0")
def cli():
    """文件管理工具。"""
    pass

@cli.command()
@click.argument("directory", default=".", type=click.Path(exists=True))
@click.option("--ext", "-e", multiple=True, help="只显示指定扩展名")
def ls(directory, ext):
    """列出目录中的文件。"""
    for f in sorted(Path(directory).iterdir()):
        if ext and f.suffix.lstrip(".") not in ext:
            continue
        size = f"{f.stat().st_size:>10,}" if f.is_file() else "     <DIR>"
        click.echo(f"{size}  {f.name}")

@cli.command()
@click.argument("directory", default=".", type=click.Path(exists=True))
@click.option("--find", "-f", required=True, help="查找字符串")
@click.option("--replace", "-r", required=True, help="替换字符串")
@click.option("--dry-run", is_flag=True, help="预览模式")
def rename(directory, find, replace, dry_run):
    """批量重命名文件。"""
    count = 0
    for f in Path(directory).iterdir():
        if f.is_file() and find in f.name:
            new_name = f.name.replace(find, replace)
            if not dry_run:
                f.rename(f.parent / new_name)
            count += 1
    click.secho(f"{'预览' if dry_run else '完成'}: {count} 个文件", fg="green")

if __name__ == "__main__":
    cli()
```

Click 优势：装饰器语法、自动 `--help`、类型校验、进度条、颜色输出。

### 五、input() 交互式输入

```python
def confirm_prompt(message: str, default: bool = False) -> bool:
    suffix = " [Y/n] " if default else " [y/N] "
    choice = input(message + suffix).strip().lower()
    if not choice:
        return default
    return choice in ("y", "yes")

if __name__ == "__main__":
    if confirm_prompt("开始清理临时文件?", default=False):
        print("开始清理...")
```

## 常见误区

**误区一：不校验参数** — 用户可能传入不存在的路径或负数。用 argparse 的 `type` 或 click 的类型系统做校验。

**误区二：帮助信息太简陋** — `--help` 是用户了解工具的第一个入口，为每个参数写清楚说明。

**误区三：把所有参数做成可选的** — 核心输入用位置参数，让用户知道最少要传什么。

## 工程建议

1. 简单脚本用 argparse，复杂工具用 click
2. 用 rich 提升可读性（表格、进度条、颜色）
3. 始终提供 `--help` 和 `--dry-run`
4. 交互确认用 `click.confirm()`，比 `input()` 更简洁

## 小结

- `sys.argv` 适合 2-3 个参数的简单脚本，`argparse` 是标准库的正式方案
- `click` 用装饰器构建 CLI，适合复杂工具
- `rich` 让终端输出更美观
- 永远校验用户输入，永远提供 `--help`

### 练习一

**答案**：

```python
import argparse
import json
from datetime import datetime
from pathlib import Path

NOTES_FILE = Path("notes.json")

def load_notes() -> list:
    if NOTES_FILE.exists():
        return json.loads(NOTES_FILE.read_text(encoding="utf-8"))
    return []

def save_notes(notes: list):
    NOTES_FILE.write_text(json.dumps(notes, ensure_ascii=False, indent=2), encoding="utf-8")

def cmd_add(args):
    notes = load_notes()
    notes.append({"content": args.content, "tags": args.tags.split(",") if args.tags else [],
                   "created_at": datetime.now().isoformat()})
    save_notes(notes)
    print(f"已添加笔记")

def cmd_list(args):
    for i, note in enumerate(load_notes(), 1):
        tags = ", ".join(note.get("tags", []))
        print(f"  {i}. [{tags}] {note['content'][:60]}")

def cmd_search(args):
    results = [n for n in load_notes() if args.keyword.lower() in n["content"].lower()]
    print(f"找到 {len(results)} 条结果")

def main():
    parser = argparse.ArgumentParser(description="笔记管理工具")
    sub = parser.add_subparsers(dest="command")

    add_p = sub.add_parser("add", help="添加笔记")
    add_p.add_argument("content", help="笔记内容")
    add_p.add_argument("-t", "--tags", help="标签，逗号分隔")

    sub.add_parser("list", help="列出所有笔记")

    sp = sub.add_parser("search", help="搜索笔记")
    sp.add_argument("keyword", help="搜索关键词")

    args = parser.parse_args()
    if not args.command:
        parser.print_help(); return
    {"add": cmd_add, "list": cmd_list, "search": cmd_search}[args.command](args)

if __name__ == "__main__":
    main()
```

**要点**：子命令函数统一放在字典中映射调用，避免 if-elif 链。

### 练习二

**答案**：

```python
import click
import socket

def parse_ports(value: str) -> list[int]:
    ports = []
    for part in value.split(","):
        part = part.strip()
        if "-" in part:
            start, end = part.split("-", 1)
            ports.extend(range(int(start), int(end) + 1))
        else:
            ports.append(int(part))
    return sorted(set(ports))

@click.command()
@click.option("--host", "-h", required=True, help="目标主机")
@click.option("--ports", "-p", required=True,
              callback=lambda ctx, param, value: parse_ports(value),
              help="端口列表，如 80,443 或 80-90")
@click.option("--timeout", "-t", default=1.0, type=float)
def scan(host, ports, timeout):
    """扫描指定主机的开放端口。"""
    for port in ports:
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(timeout)
            if sock.connect_ex((host, port)) == 0:
                click.secho(f"  端口 {port}: 开放", fg="green")
            sock.close()
        except socket.error:
            pass

if __name__ == "__main__":
    scan()
```

**要点**：click 的 `callback` 参数在解析后执行自定义逻辑，把参数解析和业务分离。
