# 阶段实战：服务器巡检脚本

## 场景引入

你负责维护 5 台服务器，每天需要人工登录检查 CPU、内存、磁盘使用率，确认关键服务是否在运行，检查端口是否正常。这套流程每天花 30 分钟，而且容易遗漏。现在你已经学完了进程管理、定时任务、系统监控、数据库操作和网络工具，是时候把这些技能整合起来，构建一个完整的自动化巡检系统。

## 学习目标

- 综合运用 psutil、socket、subprocess 构建巡检系统
- 实现多项检查指标的采集和评估
- 生成格式化的 HTML 巡检报告
- 设计告警通知机制

## 巡检系统设计

```
巡检系统架构:
┌──────────────┐
│  巡检调度器   │ ← 定时触发（APScheduler/cron）

├──────────────┤
│  告警通知器   │ ← 控制台 / 文件 / 邮件
└──────────────┘
```

## 第一步：系统指标检查

```python
import psutil, socket, platform
from dataclasses import dataclass
from datetime import datetime
from enum import Enum

class Status(Enum):
    OK = "正常"
    WARNING = "警告"
    CRITICAL = "严重"

@dataclass
class CheckResult:
    name: str
    status: Status
    value: str
    detail: str = ""

def check_cpu(warn=70, crit=90) -> CheckResult:
    percent = psutil.cpu_percent(interval=1)
    status = Status.CRITICAL if percent >= crit else Status.WARNING if percent >= warn else Status.OK
    return CheckResult("CPU 使用率", status, f"{percent}%", f"核心数: {psutil.cpu_count()}")

def check_memory(warn=75, crit=90) -> CheckResult:
    mem = psutil.virtual_memory()
    status = Status.CRITICAL if mem.percent >= crit else Status.WARNING if mem.percent >= warn else Status.OK
    return CheckResult("内存使用率", status, f"{mem.percent}%",
                       f"已用 {mem.used/1024**3:.1f}GB / 总计 {mem.total/1024**3:.1f}GB")

def check_disk(warn=80, crit=95) -> list[CheckResult]:
    results = []
    for part in psutil.disk_partitions():
        try:
            usage = psutil.disk_usage(part.mountpoint)
            status = Status.CRITICAL if usage.percent >= crit else Status.WARNING if usage.percent >= warn else Status.OK
            results.append(CheckResult(f"磁盘 {part.mountpoint}", status,
                                       f"{usage.percent}%",
                                       f"已用 {usage.used/1024**3:.1f}GB / 总计 {usage.total/1024**3:.1f}GB"))
        except PermissionError:
            continue
    return results
```

## 第二步：进程和端口检查

```python
def check_process(name: str) -> CheckResult:
    found = False
    pid = None
    for proc in psutil.process_iter(["pid", "name"]):
        if name.lower() in proc.info["name"].lower():
            found, pid = True, proc.info["pid"]
            break
    return CheckResult(f"进程 {name}",
                       Status.OK if found else Status.CRITICAL,
                       f"运行中 (PID={pid})" if found else "未运行")

def check_port(host: str, port: int, timeout=3) -> CheckResult:
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(timeout)
    try:
        sock.connect((host, port))
        return CheckResult(f"端口 {host}:{port}", Status.OK, "可达")
    except (socket.timeout, ConnectionRefusedError, OSError):
        return CheckResult(f"端口 {host}:{port}", Status.CRITICAL, "不可达")
    finally:
        sock.close()

def run_all_checks() -> list[CheckResult]:
    results = [check_cpu(), check_memory()]
    results.extend(check_disk())
    for proc in ["nginx", "mysql", "redis-server"]:
        results.append(check_process(proc))
    for host, port in [("127.0.0.1", 80), ("127.0.0.1", 3306), ("127.0.0.1", 6379)]:
        results.append(check_port(host, port))
    return results
```

## 第三步：HTML 报告生成

```python
def generate_html_report(results: list[CheckResult], hostname: str = "") -> str:
    if not hostname:
        hostname = socket.gethostname()
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    colors = {Status.OK: ("#d4edda", "#155724"), Status.WARNING: ("#fff3cd", "#856404"),
              Status.CRITICAL: ("#f8d7da", "#721c24")}
    icons = {Status.OK: "✓", Status.WARNING: "⚠", Status.CRITICAL: "✗"}

    has_crit = any(r.status == Status.CRITICAL for r in results)
    has_warn = any(r.status == Status.WARNING for r in results)
    overall = ("严重", "#f8d7da") if has_crit else ("有警告", "#fff3cd") if has_warn else ("正常", "#d4edda")

    rows = ""
    for r in results:
        bg, fg = colors[r.status]
        rows += f'<tr><td>{r.name}</td><td style="background:{bg};color:{fg};text-align:center">{icons[r.status]}</td><td>{r.value}</td><td>{r.detail}</td></tr>'

    return f"""<!DOCTYPE html>
<html lang="zh-CN"><head><meta charset="UTF-8"><title>服务器巡检报告</title>
<style>
body{{font-family:-apple-system,sans-serif;margin:20px;background:#f5f5f5}}
```

## 第四步：告警和存储

```python
import os, logging

logging.basicConfig(filename="patrol.log", level=logging.INFO,
                    format="%(asctime)s [%(levelname)s] %(message)s")

def notify_alerts(results: list[CheckResult]):
    alerts = [r for r in results if r.status in (Status.WARNING, Status.CRITICAL)]
    if not alerts:
        logging.info("巡检完成，所有指标正常")
        return
    for r in alerts:
        msg = f"[{r.status.value}] {r.name}: {r.value} ({r.detail})"
        logging.warning(msg)
        print(f"🚨 {msg}")

def save_report(html: str, output_dir="reports") -> str:
    os.makedirs(output_dir, exist_ok=True)
    filename = f"patrol_{datetime.now().strftime('%Y%m%d_%H%M%S')}.html"
    path = os.path.join(output_dir, filename)
    with open(path, "w", encoding="utf-8") as f:
        f.write(html)
    return path
```

## 第五步：完整入口

```python
import argparse

def main():
    parser = argparse.ArgumentParser(description="服务器巡检脚本")
    parser.add_argument("--output", default="reports", help="报告输出目录")
    parser.add_argument("--no-alert", action="store_true", help="不输出告警")
    args = parser.parse_args()

    print(f"[{datetime.now().strftime('%H:%M:%S')}] 开始巡检...")
    results = run_all_checks()

    ok = sum(1 for r in results if r.status == Status.OK)
    warn = sum(1 for r in results if r.status == Status.WARNING)
    crit = sum(1 for r in results if r.status == Status.CRITICAL)
    print(f"检查完成: {ok} 正常, {warn} 警告, {crit} 严重")

    if not args.no_alert:
        notify_alerts(results)

    html = generate_html_report(results)
    report_path = save_report(html, args.output)
    print(f"报告已保存: {report_path}")

if __name__ == "__main__":
    main()
```

## 常见误区

1. **检查项硬编码**：进程名、端口号、阈值写死在代码里，应使用配置文件管理。
2. **不做异常隔离**：一个检查项抛异常导致整个巡检中断，每项应独立 try/except。
3. **报告只存内存**：巡检结果不持久化，重启后无法回溯，应写入文件。
4. **告警没有去重**：每次都发送全部告警造成疲劳，应标记已告警项。

## 工程建议

1. **配置与代码分离**：把检查项、阈值、目标主机写入 YAML 配置文件。
2. **报告保留策略**：保留最近 30 天，旧报告归档或删除。
3. **定时执行**：用 APScheduler 或 cron 每天自动执行。
4. **渐进增强**：先跑通核心检查，再加告警、报告美化、远程巡检。

## 小结

本节综合运用了 psutil 系统监控、socket 网络探测、subprocess 进程管理和 HTML 报告生成，构建了一个完整的服务器巡检系统。这个脚本可以直接用于日常运维，也可以作为更复杂监控系统的起点。

## 参考答案

### 练习一

**思路**：用 `PyYAML` 解析配置文件，将硬编码的检查项替换为配置驱动。

**答案**：

```yaml
# config.yaml
thresholds:
  cpu: {warning: 70, critical: 90}
  memory: {warning: 75, critical: 90}
  disk: {warning: 80, critical: 95}
processes: [nginx, mysql, redis-server]
ports:
  - {host: "127.0.0.1", port: 80}
  - {host: "127.0.0.1", port: 3306}
```

```python
import yaml

def load_config(path="config.yaml"):
    with open(path, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)

def run_checks_from_config(config):
    results = []
    t = config.get("thresholds", {})
    results.append(check_cpu(t.get("cpu", {}).get("warning", 70),
                             t.get("cpu", {}).get("critical", 90)))
    results.append(check_memory(t.get("memory", {}).get("warning", 75),
                                t.get("memory", {}).get("critical", 90)))
    for proc in config.get("processes", []):
        results.append(check_process(proc))
    for p in config.get("ports", []):
        results.append(check_port(p["host"], p["port"]))
    return results
```

**要点**：`yaml.safe_load` 比 `yaml.load` 安全，不会执行任意 Python 对象。

### 练习二

**思路**：用 `paramiko` SSH 连接远程服务器，执行巡检命令，汇总结果。

**答案**：

```python
import paramiko, json

def remote_check(host, user, key_path, config_path="config.yaml"):
    ssh = paramiko.SSHClient()
    ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())
    ssh.connect(host, username=user, key_filename=key_path)
    sftp = ssh.open_sftp()
    sftp.put(config_path, "/tmp/patrol_config.yaml")
    sftp.close()
    stdin, stdout, stderr = ssh.exec_command(
        "python3 /tmp/patrol.py --config /tmp/patrol_config.yaml --json", timeout=30)
    exit_code = stdout.channel.recv_exit_status()
    output = stdout.read().decode()
    ssh.close()
    if exit_code == 0:
        return json.loads(output)
    return {"error": output}

def multi_host_patrol(hosts: list[dict]) -> dict:
    all_results = {}
    for h in hosts:
        try:
            all_results[h["host"]] = remote_check(
                h["host"], h["user"], h["key_path"])
        except Exception as e:
            all_results[h["host"]] = {"error": str(e)}
    return all_results
```

**要点**：`paramiko` 支持密钥认证；SFTP 上传配置保证远程使用一致配置。
