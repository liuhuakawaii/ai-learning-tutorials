# 进程管理

## 真实场景

你负责一批服务器，每天需要登录每台机器执行健康检查、收集日志、重启异常服务。手动 SSH 一台台操作效率低且容易遗漏。你需要脚本能批量执行系统命令、管理后台进程、监控进程状态。

Python 的 `subprocess` 和 `psutil` 是完成这些任务的核心工具。

## 三种进程执行方式

`os.system` 最原始，无法捕获输出、无法设置超时、存在 shell 注入风险。生产环境用 `subprocess`。

`subprocess.run` 是推荐方式——同步执行、捕获输出、支持超时：

```python
import subprocess

result = subprocess.run(
    ["ping", "-c", "3", "127.0.0.1"],
    capture_output=True, text=True, timeout=10
)
print("输出:", result.stdout)
print("返回码:", result.returncode)

# shell=True 执行管道命令（注意安全风险）
result = subprocess.run(
    "ps aux | grep python | grep -v grep",
    shell=True, capture_output=True, text=True
)
```

`subprocess.Popen` 提供非阻塞控制，适合需要与进程交互、实时读取输出的场景：

```python
proc = subprocess.Popen(
    ["ping", "-c", "10", "127.0.0.1"],
    stdout=subprocess.PIPE, text=True
)
print(f"PID: {proc.pid}, 运行中: {proc.poll() is None}")

for line in proc.stdout:
    print(f"收到: {line.strip()}")
proc.wait()
print(f"退出码: {proc.returncode}")
```

选择标准：大多数场景只需 `subprocess.run`。需要实时读取输出或非阻塞控制时用 `Popen`。

## PIPE 与进程间通信

```python
# 通过 PIPE 向进程发送输入
proc = subprocess.Popen(
    ["python", "-c", "name = input('输入: '); print(f'你好, {name}!')"],
    stdin=subprocess.PIPE, stdout=subprocess.PIPE,
    stderr=subprocess.PIPE, text=True
)
stdout, stderr = proc.communicate(input="Alice\n")
print(f"输出: {stdout.strip()}")

# 多进程管道串联：等效于 cmd1 | cmd2
proc1 = subprocess.Popen(["echo", "hello world python"], stdout=subprocess.PIPE)
proc2 = subprocess.Popen(["grep", "python"], stdin=proc1.stdout,
                         stdout=subprocess.PIPE, text=True)
proc1.stdout.close()
output = proc2.communicate()[0]
print(f"管道结果: {output.strip()}")
```

## 后台任务管理

```python
import subprocess, signal, time

class BackgroundTask:
    def __init__(self, name: str, cmd: list[str]):
        self.name = name
        self.cmd = cmd
        self.process: subprocess.Popen | None = None

    def start(self):
        self.process = subprocess.Popen(
            self.cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True
        )
        print(f"[{self.name}] 启动 PID={self.process.pid}")

    def is_running(self) -> bool:
        return self.process is not None and self.process.poll() is None

    def stop(self, timeout: int = 5):
        if not self.is_running():
            return
        self.process.send_signal(signal.SIGTERM)
        try:
            self.process.wait(timeout=timeout)
        except subprocess.TimeoutExpired:
            self.process.kill()
            self.process.wait()

task = BackgroundTask("日志监控", ["tail", "-f", "/var/log/syslog"])
task.start()
time.sleep(2)
print(f"运行中: {task.is_running()}")
task.stop()
```

先发 SIGTERM 优雅终止，超时后 SIGTERM 无效才 kill。给进程清理资源的机会。

## psutil 进程监控

```python
import psutil

def find_processes(name: str) -> list[dict]:
    results = []
    for proc in psutil.process_iter(["pid", "name", "cmdline"]):
        if name.lower() in proc.info["name"].lower():
            results.append(proc.info)
    return results

for p in find_processes("python"):
    print(f"PID={p['pid']} 命令={p['cmdline']}")
```

## 完整示例：批量执行远程命令

```python
import subprocess
import concurrent.futures
from dataclasses import dataclass

@dataclass
class HostResult:
    host: str
    command: str
    returncode: int = -1
    stdout: str = ""
    stderr: str = ""
    success: bool = False

def run_remote_command(host: str, command: str, user: str = "root",
                       timeout: int = 30) -> HostResult:
    ssh_cmd = ["ssh", "-o", "StrictHostKeyChecking=no",
               "-o", "ConnectTimeout=10", f"{user}@{host}", command]
    result = HostResult(host=host, command=command)
    try:
        proc = subprocess.run(ssh_cmd, capture_output=True, text=True, timeout=timeout)
        result.returncode = proc.returncode
        result.stdout = proc.stdout
        result.stderr = proc.stderr
        result.success = proc.returncode == 0
    except subprocess.TimeoutExpired:
        result.stderr = f"命令超时 ({timeout}s)"
    except FileNotFoundError:
        result.stderr = "ssh 命令未找到"
    return result

def batch_execute(hosts: list[str], command: str, max_workers: int = 5) -> list[HostResult]:
    results = []
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = {executor.submit(run_remote_command, h, command): h for h in hosts}
        for future in concurrent.futures.as_completed(futures):
            result = future.result()
            status = "✓" if result.success else "✗"
            print(f"  {status} {result.host}: {result.stdout.strip()[:80]}")
            results.append(result)
    return results

if __name__ == "__main__":
    hosts = ["192.168.1.10", "192.168.1.11", "192.168.1.12"]
    results = batch_execute(hosts, "df -h / | tail -1")
    print(f"完成: {sum(1 for r in results if r.success)}/{len(hosts)} 成功")
```

`ThreadPoolExecutor` 控制并发数，`as_completed` 按完成顺序处理结果。

## 踩坑提醒

**用 `shell=True` 拼接用户输入**：存在命令注入风险，应使用参数列表。

**不等待子进程**：`Popen` 不调用 `wait()` 导致僵尸进程。

**忽略超时**：网络命令不设超时，脚本可能永久挂起。

## 练习

### 练习一：进程超时管理器

编写 `run_with_timeout(cmd, timeout)` 函数，超时时自动终止进程，返回 `(success, output, timed_out)` 元组。

### 练习二：进程存活监控

每 5 秒检查指定进程是否在运行，消失则自动重启并记录日志。

---

## 参考答案

### 练习一

```python
import subprocess

def run_with_timeout(cmd: list[str], timeout: int = 10) -> tuple[bool, str, bool]:
    try:
        result = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout)
        return (result.returncode == 0, result.stdout, False)
    except subprocess.TimeoutExpired:
        return (False, "", True)
    except Exception as e:
        return (False, str(e), False)
```

`TimeoutExpired` 被捕获后进程已被自动终止，无需手动 kill。

### 练习二

```python
import psutil, subprocess, time, logging

logging.basicConfig(filename="monitor.log", level=logging.INFO,
                    format="%(asctime)s %(message)s")

def is_running(name: str) -> bool:
    for proc in psutil.process_iter(["name"]):
        if name.lower() in proc.info["name"].lower():
            return True
    return False

def monitor_loop(service: str, interval: int = 5):
    while True:
        if not is_running(service):
            logging.warning(f"{service} 消失，正在重启")
            try:
                subprocess.run(["systemctl", "restart", service], timeout=30)
                logging.info(f"{service} 重启成功")
            except Exception as e:
                logging.error(f"重启失败: {e}")
        time.sleep(interval)
```

`systemctl restart` 比直接启动二进制更规范；用 `logging` 而非 `print`。
