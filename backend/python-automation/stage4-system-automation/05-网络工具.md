# 网络工具

## 场景引入

你的团队需要定期检查服务器集群中各台机器的端口是否正常开放、DNS 解析是否正确、网络连通性是否良好。手动用 `ping` 和 `telnet` 一台台检查效率太低。你需要用 Python 编写网络探测工具，自动化完成这些检查。`socket` 模块是 Python 网络编程的基础。

## 学习目标

- 掌握 `socket` 模块的 TCP/UDP 通信基础
- 能实现端口扫描器
- 能进行 DNS 查询和网络连通性检测
- 理解网络超时和异常处理

## socket 编程基础

### TCP 客户端

```python
import socket

def tcp_connect(host: str, port: int, timeout: float = 5) -> dict:
    result = {"host": host, "port": port, "open": False, "error": ""}
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(timeout)
    try:
        sock.connect((host, port))
        result["open"] = True
    except socket.timeout:
        result["error"] = "连接超时"
    except ConnectionRefusedError:
        result["error"] = "连接被拒绝"
    except socket.gaierror:
        result["error"] = "DNS 解析失败"
    finally:
        sock.close()
    return result

r = tcp_connect("example.com", 80)
print(f"{r['host']}:{r['port']} - {'开放' if r['open'] else r['error']}")
```

### TCP 服务端

```python
import socket, threading

def handle_client(client):
    try:
        client.sendall(b"Hello!\n")
        data = client.recv(1024)
        client.sendall(f"Echo: {data.decode().strip()}\n".encode())
    finally:
        client.close()

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(("0.0.0.0", 9999))
server.listen(5)
while True:
    client, addr = server.accept()
    threading.Thread(target=handle_client, args=(client,)).start()
```

## 端口扫描器

```python
import socket, concurrent.futures
from dataclasses import dataclass

SERVICES = {21: "FTP", 22: "SSH", 80: "HTTP", 443: "HTTPS",
            3306: "MySQL", 5432: "PostgreSQL", 6379: "Redis", 8080: "HTTP-Alt"}

@dataclass
class PortResult:
    port: int
    open: bool
    service: str = ""

def scan_port(host, port, timeout=1.0):
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(timeout)
    try:
        sock.connect((host, port))
        return PortResult(port=port, open=True, service=SERVICES.get(port, "未知"))
    except (socket.timeout, ConnectionRefusedError, OSError):
        return PortResult(port=port, open=False)
    finally:
        sock.close()

def scan_ports(host, ports, max_workers=100):
    results = []
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = {executor.submit(scan_port, host, p): p for p in ports}
        for f in concurrent.futures.as_completed(futures):
            results.append(f.result())
    return sorted(results, key=lambda r: r.port)

if __name__ == "__main__":
    results = scan_ports("127.0.0.1", [22, 80, 443, 3306, 6379, 8080])
    for r in results:
        if r.open:
            print(f"  {r.port}/tcp 开放 ({r.service})")
```

## DNS 查询

import socket

def dns_lookup(hostname):
    try:
        info = socket.getaddrinfo(hostname, None)
        return list(set(addr[4][0] for addr in info))
    except socket.gaierror:
        return []

def reverse_dns(ip):
    try: return socket.gethostbyaddr(ip)[0]
    except socket.herror: return "无反向记录"

for host in ["google.com", "github.com", "baidu.com"]:
    print(f"{host}: {', '.join(dns_lookup(host)) or '解析失败'}")

## 网络连通性检测

```python
import subprocess, platform, socket
from dataclasses import dataclass

@dataclass
class ConnectivityResult:
    host: str
    ping_ok: bool
    dns_ok: bool
    resolved_ip: str
    tcp_80_ok: bool
    tcp_443_ok: bool

def ping(host, count=3, timeout=5):
    param = "-n" if platform.system() == "Windows" else "-c"
    cmd = ["ping", param, str(count), "-w" if platform.system() == "Windows" else "-W",
           str(timeout), host]
    try:
        return subprocess.run(cmd, capture_output=True, timeout=timeout + 5).returncode == 0
    except Exception:
        return False

def check_connectivity(host):
    ping_ok = ping(host)
    resolved_ip = ""
    try:
        resolved_ip = socket.gethostbyname(host)
    except socket.gaierror:
        pass

    tcp_80 = tcp_443 = False
    for port in [80, 443]:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(3)
        try:
            sock.connect((resolved_ip, port))
            if port == 80: tcp_80 = True
            else: tcp_443 = True
        except Exception:
            pass
        finally:
            sock.close()

    return ConnectivityResult(host, ping_ok, bool(resolved_ip),
                              resolved_ip, tcp_80, tcp_443)

if __name__ == "__main__":
    for host in ["google.com", "github.com"]:
        r = check_connectivity(host)
        print(f"{r.host} ({r.resolved_ip}): Ping={'✓' if r.ping_ok else '✗'} "
              f"80={'✓' if r.tcp_80_ok else '✗'} 443={'✓' if r.tcp_443_ok else '✗'}")
```

## 完整示例：网络设备发现和端口扫描工具

```python
import socket, subprocess, platform, ipaddress, concurrent.futures
from dataclasses import dataclass, field

@dataclass
class DeviceInfo:
    ip: str
    hostname: str = ""
    alive: bool = False
    open_ports: list = field(default_factory=list)

def ping_host(ip, timeout=1):
    param = "-n" if platform.system() == "Windows" else "-c"
    cmd = ["ping", param, "1", "-w" if platform.system() == "Windows" else "-W",
           str(timeout), str(ip)]
    try:
        return subprocess.run(cmd, capture_output=True, timeout=timeout + 2).returncode == 0
    except Exception:
        return False

def scan_host_ports(ip, ports, timeout=0.5):
    open_ports = []
    for port in ports:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(timeout)
        try:
            if sock.connect_ex((ip, port)) == 0:
                open_ports.append(port)
        except Exception:
            pass
        finally:
            sock.close()
    return open_ports

def discover_network(subnet, ports=None, max_workers=50):
    if ports is None:
        ports = [22, 80, 443, 3389, 8080]
    network = ipaddress.ip_network(subnet, strict=False)
    print(f"扫描 {subnet} ({network.num_addresses} 个地址)...")

    alive_ips = []
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as ex:
        futures = {ex.submit(ping_host, str(ip)): str(ip) for ip in network.hosts()}
        for f in concurrent.futures.as_completed(futures):
            ip = futures[f]
            if f.result():
                alive_ips.append(ip)
                print(f"  发现: {ip}")

    print(f"\n活跃 {len(alive_ips)} 台，扫描端口...")
    devices = []
    for ip in alive_ips:
        dev = DeviceInfo(ip=ip, alive=True)
        try:
            dev.hostname = socket.gethostbyaddr(ip)[0]
        except socket.herror:
            pass
        dev.open_ports = scan_host_ports(ip, ports)
        devices.append(dev)
    return devices

if __name__ == "__main__":
    devices = discover_network("192.168.1.0/24", [22, 80, 443, 3306])
    for d in devices:
        host = f" ({d.hostname})" if d.hostname else ""
        ports = ", ".join(str(p) for p in d.open_ports) or "无"
        print(f"{d.ip}{host}: {ports}")
```

## 常见误区

1. **不设超时**：`socket.connect` 默认阻塞数分钟，始终调用 `settimeout()`。
2. **忘记关闭 socket**：大量 socket 不关闭耗尽文件描述符，用 `with` 或 `try/finally`。
3. **扫描速度过快**：同时数千连接触发防火墙规则，控制并发数。
4. **DNS 解析阻塞**：`gethostbyname` 同步阻塞，批量解析放线程池。
5. **忽略平台差异**：`ping` 参数在 Windows/Linux 不同，需判断 `platform.system()`。

## 工程建议

1. **用 `connect_ex` 代替 `connect`**：端口扫描时返回错误码而非抛异常，性能更好。
2. **合理设置超时**：局域网 0.5-1 秒，公网 3-5 秒。
3. **两阶段扫描**：先 ping 发现活跃主机，再扫描端口，避免无用扫描。
4. **大规模用 `nmap`**：Python 脚本适合定制化需求，大规模用 `nmap` 或 `masscan`。

## 参考答案

### 练习一

**思路**：用 `ThreadPoolExecutor` 并行执行 `connect_ex` 检测。

**答案**：

```python
import socket, concurrent.futures

def check_port(host, port, timeout=3):
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(timeout)
    try:
        return sock.connect_ex((host, port)) == 0
    finally:
        sock.close()

def check_service_ports(services: dict[str, tuple[str, int]]) -> dict[str, bool]:
    results = {}
    with concurrent.futures.ThreadPoolExecutor(max_workers=20) as ex:
        futures = {ex.submit(check_port, h, p): name
                   for name, (h, p) in services.items()}
        for f in concurrent.futures.as_completed(futures):
            results[futures[f]] = f.result()
    return results

services = {"Web": ("example.com", 80), "DB": ("127.0.0.1", 3306),
            "Redis": ("127.0.0.1", 6379)}
for name, ok in check_service_ports(services).items():
    print(f"  {name}: {'✓' if ok else '✗'}")
```

**要点**：`connect_ex` 返回 0 表示连接成功，比 `try/except` 更高效。

### 练习二

**思路**：通过 socket 发送原始 HTTP GET 请求，解析响应第一行获取状态码。

**答案**：

```python
import socket
from urllib.parse import urlparse

def http_health_check(url, timeout=5):
    parsed = urlparse(url)
    host = parsed.hostname
    port = parsed.port or 80
    path = parsed.path or "/"
    result = {"url": url, "status_code": None, "ok": False}

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(timeout)
    try:
        sock.connect((host, port))
        req = f"GET {path} HTTP/1.1\r\nHost: {host}\r\nConnection: close\r\n\r\n"
        sock.sendall(req.encode())
        resp = b""
        while True:
            data = sock.recv(4096)
            if not data: break
            resp += data
        status_line = resp.split(b"\r\n")[0].decode()
        parts = status_line.split(" ")
        if len(parts) >= 2:
            result["status_code"] = int(parts[1])
            result["ok"] = result["status_code"] == 200
    except Exception:
        pass
    finally:
        sock.close()
    return result

r = http_health_check("http://example.com")
print(f"状态码: {r['status_code']}, 健康: {'✓' if r['ok'] else '✗'}")
```

**要点**：HTTP 请求遵循 `\r\n` 分隔格式；此示例仅支持 HTTP，HTTPS 需 `ssl.wrap_socket`。
