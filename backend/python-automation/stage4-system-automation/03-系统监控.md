# 系统监控

## 场景引入

你的服务器偶尔会因为内存泄漏或磁盘满导致服务中断，但你总是在故障发生后才收到投诉。你需要一个监控脚本，持续采集 CPU、内存、磁盘、网络等关键指标，在异常时主动告警。`psutil` 是 Python 生态中最成熟的系统监控库，跨平台支持 Linux、macOS 和 Windows。

## 学习目标

- 掌握 `psutil` 采集 CPU、内存、磁盘、网络等系统指标
- 理解监控指标的含义和正常范围
- 能设计阈值告警机制
- 能将监控数据持久化存储

## CPU 监控

```python
import psutil

cpu_percent = psutil.cpu_percent(interval=1)
print(f"CPU 使用率: {cpu_percent}%")

per_cpu = psutil.cpu_percent(interval=1, percpu=True)
for i, p in enumerate(per_cpu):
    print(f"  核心 {i}: {p}%")

freq = psutil.cpu_freq()
print(f"频率: {freq.current}MHz" if freq else "无法获取频率")

load = psutil.getloadavg()
print(f"负载 (1/5/15分钟): {load[0]:.2f} / {load[1]:.2f} / {load[2]:.2f}")
```

`cpu_percent(interval=1)` 阻塞 1 秒采样。不传 interval 时返回自上次调用以来的使用率。

## 进程监控

```python
import psutil

def get_top_processes(n=5, by="cpu"):
    key = "cpu_percent" if by == "cpu" else "memory_percent"
    procs = []
    for proc in psutil.process_iter(["pid", "name", "cpu_percent", "memory_percent"]):
        try:
            procs.append(proc.info)
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            continue
    return sorted(procs, key=lambda p: p[key] or 0, reverse=True)[:n]

for p in get_top_processes(3, "memory"):
    print(f"PID={p['pid']} {p['name']}: MEM={p['memory_percent']:.1f}%")
```

## 监控数据采集器

```python
import psutil, json
from dataclasses import dataclass, asdict
from datetime import datetime

@dataclass
class SystemMetrics:
    timestamp: str
    cpu_percent: float
    memory_percent: float
    disk_percent: float
    net_bytes_sent: int
    net_bytes_recv: int
    process_count: int

def collect_metrics() -> SystemMetrics:
    return SystemMetrics(
        timestamp=datetime.now().isoformat(),
        cpu_percent=psutil.cpu_percent(interval=1),
        memory_percent=psutil.virtual_memory().percent,
        disk_percent=psutil.disk_usage("/").percent,
        net_bytes_sent=psutil.net_io_counters().bytes_sent,
        net_bytes_recv=psutil.net_io_counters().bytes_recv,
        process_count=len(psutil.pids())
    )

def save_metrics(metrics: SystemMetrics, filepath="metrics.jsonl"):
    with open(filepath, "a", encoding="utf-8") as f:
        f.write(json.dumps(asdict(metrics), ensure_ascii=False) + "\n")
```

## 阈值告警

```python
from dataclasses import dataclass

@dataclass
class AlertRule:
    metric: str
    threshold: float
    message: str

class AlertManager:
    def __init__(self, rules: list[AlertRule]):
        self.rules = rules
        self.alerted = set()

    def check(self, metrics: SystemMetrics) -> list[str]:
        alerts = []
        values = {"cpu_percent": metrics.cpu_percent,
                  "memory_percent": metrics.memory_percent,
                  "disk_percent": metrics.disk_percent}
        for rule in self.rules:
            value = values.get(rule.metric, 0)
            key = f"{rule.metric}:{rule.threshold}"
            if value >= rule.threshold and key not in self.alerted:
                alerts.append(f"[告警] {rule.message}: {value:.1f}% (阈值 {rule.threshold}%)")
                self.alerted.add(key)
            elif value < rule.threshold * 0.9:
                self.alerted.discard(key)
        return alerts

rules = [
    AlertRule("cpu_percent", 80, "CPU 使用率过高"),
    AlertRule("memory_percent", 85, "内存使用率过高"),
    AlertRule("disk_percent", 90, "磁盘使用率过高"),
]
alert_mgr = AlertManager(rules)
```

## 监控数据存储

```python
import sqlite3

class MetricsStore:
    def __init__(self, db_path="metrics.db"):
        self.conn = sqlite3.connect(db_path)
        self.conn.execute("""CREATE TABLE IF NOT EXISTS metrics (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            timestamp TEXT, cpu REAL, memory REAL, disk REAL
        )""")
        self.conn.commit()

    def save(self, m: SystemMetrics):
        self.conn.execute("INSERT INTO metrics (timestamp,cpu,memory,disk) VALUES (?,?,?,?)",
                          (m.timestamp, m.cpu_percent, m.memory_percent, m.disk_percent))
        self.conn.commit()

    def query_recent(self, hours=24):
        cursor = self.conn.execute(
            "SELECT * FROM metrics WHERE timestamp > datetime('now', ?) ORDER BY timestamp",
            (f"-{hours} hours",))
        cols = [d[0] for d in cursor.description]
        return [dict(zip(cols, row)) for row in cursor.fetchall()]
```

## 完整示例：系统资源监控 + 告警脚本

```python
import psutil, sqlite3, logging, time
from dataclasses import dataclass
from datetime import datetime

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")
logger = logging.getLogger(__name__)

THRESHOLDS = {"cpu": 80, "memory": 85, "disk": 90}

@dataclass
class MetricPoint:
    timestamp: str
    cpu_percent: float
    memory_percent: float
    disk_percent: float

def collect() -> MetricPoint:
    return MetricPoint(
        timestamp=datetime.now().isoformat(),
        cpu_percent=psutil.cpu_percent(interval=1),
        memory_percent=psutil.virtual_memory().percent,
        disk_percent=psutil.disk_usage("/").percent
    )

def check_alerts(p: MetricPoint) -> list[str]:
    alerts = []
    if p.cpu_percent >= THRESHOLDS["cpu"]:
        alerts.append(f"CPU {p.cpu_percent}% >= {THRESHOLDS['cpu']}%")
    if p.memory_percent >= THRESHOLDS["memory"]:
        alerts.append(f"内存 {p.memory_percent}% >= {THRESHOLDS['memory']}%")
    if p.disk_percent >= THRESHOLDS["disk"]:
        alerts.append(f"磁盘 {p.disk_percent}% >= {THRESHOLDS['disk']}%")
    return alerts

def monitor_loop(interval=60, db_path="monitor.db"):
    conn = sqlite3.connect(db_path)
    conn.execute("CREATE TABLE IF NOT EXISTS metrics (id INTEGER PRIMARY KEY AUTOINCREMENT, timestamp TEXT, cpu REAL, memory REAL, disk REAL)")
    conn.commit()
    logger.info(f"监控启动: 间隔 {interval}s, 阈值 {THRESHOLDS}")
    try:
        while True:
            p = collect()
            conn.execute("INSERT INTO metrics (timestamp,cpu,memory,disk) VALUES (?,?,?,?)",
                         (p.timestamp, p.cpu_percent, p.memory_percent, p.disk_percent))
            conn.commit()
            logger.info(f"CPU={p.cpu_percent}% MEM={p.memory_percent}% DISK={p.disk_percent}%")
            for alert in check_alerts(p):
                logger.warning(f"告警: {alert}")
            time.sleep(interval)
    except KeyboardInterrupt:
        logger.info("监控已停止")
    finally:
        conn.close()

if __name__ == "__main__":
    monitor_loop(interval=10)
```

## 常见误区

1. **`cpu_percent(interval=1)` 频繁调用**：每次阻塞 1 秒，监控多项指标时总间隔远超预期。
2. **只看百分比不看绝对值**：90% 在 64GB 机器上还有 6.4GB，4GB 机器上只剩 400MB。
3. **告警风暴**：每分钟检查每次都触发，应加冷却机制。
4. **监控脚本本身消耗资源**：频繁采集和写磁盘成为瓶颈，生产环境 30-60 秒间隔足够。

## 工程建议

1. **用 JSONL 格式存储**：每行一条，追加写入，方便流式处理。
2. **监控数据定期清理**：保留 30 天，旧数据归档或删除。
3. **告警去重**：用集合记录已触发告警，恢复后清除标记。
4. **大规模用 Prometheus + Grafana**：自建脚本适合小规模，大规模用专业监控栈。

### 练习一

**思路**：两次 `net_io_counters` 差值除以时间间隔得到速率。

**答案**：

```python
import psutil, time

def monitor_network(interval=10, threshold_kbps=10240):
    prev = psutil.net_io_counters()
    while True:
        time.sleep(interval)
        curr = psutil.net_io_counters()
        send_rate = (curr.bytes_sent - prev.bytes_sent) / interval / 1024
        recv_rate = (curr.bytes_recv - prev.bytes_recv) / interval / 1024
        print(f"发送: {send_rate:.1f}KB/s | 接收: {recv_rate:.1f}KB/s")
        if send_rate > threshold_kbps:
            print(f"  告警: 发送 {send_rate:.1f}KB/s 超阈值")
        if recv_rate > threshold_kbps:
            print(f"  告警: 接收 {recv_rate:.1f}KB/s 超阈值")
        prev = curr
```

**要点**：必须保存上一次采样值，用差值计算速率。

### 练习二

**思路**：用 `os.walk` 递归遍历，`os.stat` 获取大小和修改时间。

**答案**：

```python
import os
from datetime import datetime, timedelta

def find_large_old_files(directory, days=30, min_size_mb=100):
    cutoff = datetime.now() - timedelta(days=days)
    candidates = []
    for root, _, files in os.walk(directory):
        for name in files:
            path = os.path.join(root, name)
            try:
                stat = os.stat(path)
                size_mb = stat.st_size / 1024 / 1024
                mtime = datetime.fromtimestamp(stat.st_mtime)
                if size_mb >= min_size_mb and mtime < cutoff:
                    candidates.append({"path": path, "size_mb": round(size_mb, 1),
                                       "days_old": (datetime.now() - mtime).days})
            except (OSError, PermissionError):
                continue
    return sorted(candidates, key=lambda x: x["size_mb"], reverse=True)
```

**要点**：`try/except` 处理权限不足的文件；返回列表按大小降序排列。
