# 文件系统操作

## 场景引入

你有一个"下载"文件夹，里面堆了几百个文件——PDF、Excel、图片、安装包——全都混在一起。你希望能写一个脚本，自动按文件类型分门别类整理好。这个需求涉及三个核心能力：遍历文件系统、操作文件和目录、监控文件变化。Python 提供了 `pathlib`、`shutil` 和 `watchdog` 三个工具来分别解决。

## 学习目标

- 掌握 `pathlib.Path` 对象的核心 API
- 理解 `pathlib` 相比 `os.path` 的优势
- 能使用 `shutil` 进行文件复制、移动和压缩
- 能使用 `watchdog` 监控文件系统变化
- 能编写自动整理文件夹的脚本

## pathlib：现代文件路径操作

### 创建 Path 对象与属性

```python
from pathlib import Path

cwd = Path.cwd()                        # 当前目录
home = Path.home()                      # 用户主目录
config = Path("config") / "app.yaml"    # 用 / 拼接路径

p = Path("reports/2024/Q1.xlsx")
print(p.name)       # Q1.xlsx
print(p.stem)       # Q1
print(p.suffix)     # .xlsx
print(p.parent)     # reports/2024
p.exists()          # 是否存在
p.is_file()         # 是否是文件
p.is_dir()          # 是否是目录
```

`Path` 用 `/` 拼接路径，比 `os.path.join()` 更直观，自动处理平台分隔符。

### 遍历目录

```python
from pathlib import Path

root = Path("D:/data")

for item in root.iterdir():         # 遍历直接子项
    print(item.name)

for py in root.rglob("*.py"):       # 递归找所有 Python 文件
    print(py)

for md in root.glob("*/*.md"):      # 匹配特定深度
    print(md)
```

`glob()` 匹配当前目录，`rglob()` 递归匹配所有子目录。

### 文件读写与目录操作

```python
from pathlib import Path

p = Path("notes.txt")
p.write_text("你好世界", encoding="utf-8")
content = p.read_text(encoding="utf-8")
p.write_bytes(b"\x89PNG...")

Path("output/reports").mkdir(parents=True, exist_ok=True)
p.unlink(missing_ok=True)
p.rename("new_name.txt")
```

### os.path 对比 pathlib

```python
import os
from pathlib import Path

os.path.join("data", "report.csv")   # vs Path("data") / "report.csv"
os.path.splitext("r.csv")[1]         # vs Path("r.csv").suffix
os.path.exists("data")               # vs Path("data").exists()
os.path.getsize("r.csv")             # vs Path("r.csv").stat().st_size
```

`pathlib` 代码更短、可读性更高，支持链式调用。新项目统一使用 `pathlib`。

## shutil：高级文件操作

```python
import shutil

# 复制（保留元数据）
shutil.copy2("source.txt", "backup/source.txt")

# 复制整个目录树
shutil.copytree("project", "project_backup", dirs_exist_ok=True)

# 移动
shutil.move("old.txt", "archive/old.txt")

# 压缩与解压
shutil.make_archive("backup/2024", "zip", "data/reports")
shutil.unpack_archive("backup/2024.zip", "restore/")

# 磁盘信息
usage = shutil.disk_usage("/")
print(f"可用: {usage.free / (1024**3):.1f} GB")
```

`copy2` 保留文件元数据（时间戳、权限），适合备份。`dirs_exist_ok=True` 允许目标目录已存在。

## watchdog：文件系统监控

```python
import time
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class MyHandler(FileSystemEventHandler):
    def on_created(self, event):
        if not event.is_directory:
            print(f"新文件: {event.src_path}")

    def on_modified(self, event):
        if not event.is_directory:
            print(f"修改: {event.src_path}")

    def on_deleted(self, event):
        print(f"删除: {event.src_path}")

observer = Observer()
observer.schedule(MyHandler(), path="D:/downloads", recursive=False)
observer.start()

try:
    while True:
        time.sleep(1)
except KeyboardInterrupt:
    observer.stop()
observer.join()
```

`watchdog` 是跨平台文件监控库，通过事件回调通知文件变化，比轮询高效得多。

## 完整示例：自动整理下载文件夹

```python
import shutil
from pathlib import Path

CATEGORY_MAP = {
    "图片": {".jpg", ".jpeg", ".png", ".gif", ".bmp", ".webp", ".svg"},
    "文档": {".pdf", ".doc", ".docx", ".xls", ".xlsx", ".ppt", ".pptx", ".txt", ".md"},
    "视频": {".mp4", ".avi", ".mkv", ".mov", ".wmv", ".flv"},
    "音频": {".mp3", ".wav", ".flac", ".aac", ".ogg"},
    "压缩包": {".zip", ".rar", ".7z", ".tar", ".gz"},
    "安装包": {".exe", ".msi", ".dmg", ".deb", ".apk"},
    "代码": {".py", ".js", ".ts", ".html", ".css", ".java"},
    "数据": {".csv", ".json", ".xml", ".yaml", ".sql", ".db"},
}

def get_category(suffix: str) -> str:
    suffix = suffix.lower()
    for cat, exts in CATEGORY_MAP.items():
        if suffix in exts:
            return cat
    return "其他"

def organize_folder(source_dir: str) -> dict:
    source = Path(source_dir)
    result = {}

    for item in source.iterdir():
        if item.is_dir() or item.name.startswith("."):
            continue

        category = get_category(item.suffix)
        target_dir = source / category
        target_dir.mkdir(exist_ok=True)

        target = target_dir / item.name
        if target.exists():
            stem, suffix = item.stem, item.suffix
            counter = 1
            while target.exists():
                target = target_dir / f"{stem}_{counter}{suffix}"
                counter += 1

        shutil.move(str(item), str(target))
        result.setdefault(category, []).append(target.name)

    return result

if __name__ == "__main__":
    downloads = Path.home() / "Downloads"
    report = organize_folder(downloads)
    for cat, files in sorted(report.items()):
        print(f"\n{cat} ({len(files)} 个文件)")
        for f in files[:3]:
            print(f"  {f}")
```

核心逻辑：遍历下载文件夹 → 按扩展名分类 → 创建子目录 → 移动文件，同名文件自动加序号。

## 常见误区

### 1. 用字符串拼接路径

```python
# ❌ 硬编码分隔符，跨平台出问题
path = base_dir + "\\" + filename
# ✅ 用 Path
path = Path(base_dir) / filename
```

### 2. 忘记处理文件不存在

```python
# ❌ 文件不存在直接崩溃
content = Path("config.txt").read_text()
# ✅ 先检查
if Path("config.txt").exists():
    content = Path("config.txt").read_text()
```

### 3. watchdog 回调中执行耗时操作

```python
# ❌ 阻塞事件队列
def on_created(self, event):
    heavy_processing(event.src_path)
# ✅ 放入队列，单独线程处理
def on_created(self, event):
    task_queue.put(event.src_path)
```

## 工程建议

1. **新项目统一用 `pathlib`**：比 `os.path` 更 Pythonic
2. **文件操作前先检查**：用 `exists()`、`is_file()` 避免意外错误
3. **大文件复制用 `shutil.copy2`**：保留元数据，适合备份
4. **监控文件变化用 `watchdog`**：比轮询高效得多
5. **压缩格式选择**：小文件用 zip（兼容性好），大量文件用 tar.gz（压缩率高）

## 参考答案

### 练习一

**思路**：递归遍历目录，用 `Counter` 统计扩展名。

**答案**：

```python
from pathlib import Path
from collections import Counter
import sys

def count_extensions(directory: str) -> None:
    root = Path(directory)
    counter = Counter()
    for f in root.rglob("*"):
        if f.is_file():
            counter[f.suffix.lower() or "(无扩展名)"] += 1
    for ext, count in counter.most_common():
        print(f"  {ext}: {count}")

if __name__ == "__main__":
    count_extensions(sys.argv[1] if len(sys.argv) > 1 else ".")
```

**要点**：`Counter.most_common()` 自动按数量降序；无扩展名文件需特殊处理。

### 练习二

**思路**：用 `watchdog` 监控创建事件，按日期创建目录并复制。

**答案**：

```python
import shutil
import time
from datetime import date
from pathlib import Path
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class BackupHandler(FileSystemEventHandler):
    def __init__(self, backup_root: str):
        self.backup_root = Path(backup_root)

    def on_created(self, event):
        if event.is_directory:
            return
        src = Path(event.src_path)
        dest_dir = self.backup_root / date.today().isoformat()
        dest_dir.mkdir(parents=True, exist_ok=True)
        shutil.copy2(str(src), str(dest_dir / src.name))
        print(f"已备份: {src.name}")

observer = Observer()
observer.schedule(BackupHandler("D:/backup"), path="D:/watch", recursive=False)
observer.start()
try:
    while True:
        time.sleep(1)
except KeyboardInterrupt:
    observer.stop()
observer.join()
```

**要点**：`date.today().isoformat()` 生成 `YYYY-MM-DD` 格式目录名。

### 练习三

**思路**：先按文件大小分组（大小不同肯定不重复），再计算 MD5 哈希。

**答案**：

```python
import hashlib
from pathlib import Path
from collections import defaultdict

def file_hash(path: Path) -> str:
    h = hashlib.md5()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            h.update(chunk)
    return h.hexdigest()

def find_duplicates(directory: str) -> list[list[Path]]:
    root = Path(directory)
    size_map = defaultdict(list)
    for f in root.rglob("*"):
        if f.is_file():
            size_map[f.stat().st_size].append(f)

    duplicates = []
    for files in size_map.values():
        if len(files) < 2:
            continue
        hash_map = defaultdict(list)
        for f in files:
            hash_map[file_hash(f)].append(f)
        for group in hash_map.values():
            if len(group) > 1:
                duplicates.append(group)
    return duplicates
```

**要点**：先按大小初筛减少哈希计算量；分块读取大文件避免内存溢出。
