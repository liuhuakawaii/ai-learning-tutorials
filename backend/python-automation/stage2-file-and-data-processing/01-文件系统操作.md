# 文件系统操作

## 从一个真实需求开始

你有一个下载文件夹，几百个文件混在一起——PDF、Excel、图片、安装包。你想写个脚本按文件类型自动归类。这涉及三个核心能力：遍历文件系统、操作文件和目录、监控文件变化。Python 的 `pathlib`、`shutil` 和 `watchdog` 分别解决这三个问题。

## pathlib：现代路径操作

`pathlib.Path` 用 `/` 拼接路径，比 `os.path.join()` 更直观，自动处理平台分隔符：

```python
from pathlib import Path

cwd = Path.cwd()
home = Path.home()
config = Path("config") / "app.yaml"

p = Path("reports/2024/Q1.xlsx")
print(p.name)       # Q1.xlsx
print(p.stem)       # Q1
print(p.suffix)     # .xlsx
print(p.parent)     # reports/2024
p.exists()          # 是否存在
p.is_file()         # 是否是文件
```

遍历目录：

```python
root = Path("D:/data")

for item in root.iterdir():         # 遍历直接子项
    print(item.name)

for py in root.rglob("*.py"):       # 递归找所有 Python 文件
    print(py)
```

`glob()` 匹配当前目录，`rglob()` 递归匹配所有子目录。

文件读写和目录操作：

```python
p = Path("notes.txt")
p.write_text("你好世界", encoding="utf-8")
content = p.read_text(encoding="utf-8")

Path("output/reports").mkdir(parents=True, exist_ok=True)
p.unlink(missing_ok=True)
```

## shutil：高级文件操作

```python
import shutil

# 复制（保留元数据：时间戳、权限）
shutil.copy2("source.txt", "backup/source.txt")

# 复制整个目录树
shutil.copytree("project", "project_backup", dirs_exist_ok=True)

# 移动
shutil.move("old.txt", "archive/old.txt")

# 压缩与解压
shutil.make_archive("backup/2024", "zip", "data/reports")
shutil.unpack_archive("backup/2024.zip", "restore/")
```

`copy2` 保留文件元数据，适合备份。`dirs_exist_ok=True` 允许目标目录已存在。

## watchdog：文件系统监控

`watchdog` 是跨平台文件监控库，通过事件回调通知文件变化，比轮询高效：

```python
import time
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class MyHandler(FileSystemEventHandler):
    def on_created(self, event):
        if not event.is_directory:
            print(f"新文件: {event.src_path}")

    def on_modified(self, event):
        if not event.is_directory:
            print(f"修改: {event.src_path}")

observer = Observer()
observer.schedule(MyHandler(), path="D:/downloads", recursive=False)
observer.start()

try:
    while True:
        time.sleep(1)
except KeyboardInterrupt:
    observer.stop()
observer.join()
```

注意：回调中不要执行耗时操作，会阻塞事件队列。应该把任务放入队列，单独线程处理。

## 完整示例：自动整理下载文件夹

```python
import shutil
from pathlib import Path

CATEGORY_MAP = {
    "图片": {".jpg", ".jpeg", ".png", ".gif", ".webp", ".svg"},
    "文档": {".pdf", ".doc", ".docx", ".xls", ".xlsx", ".pptx", ".txt", ".md"},
    "视频": {".mp4", ".avi", ".mkv", ".mov", ".wmv"},
    "音频": {".mp3", ".wav", ".flac", ".aac"},
    "压缩包": {".zip", ".rar", ".7z", ".tar", ".gz"},
    "安装包": {".exe", ".msi", ".dmg", ".deb"},
    "代码": {".py", ".js", ".ts", ".html", ".css", ".java"},
}

def get_category(suffix: str) -> str:
    suffix = suffix.lower()
    for cat, exts in CATEGORY_MAP.items():
        if suffix in exts:
            return cat
    return "其他"

def organize_folder(source_dir: str) -> dict:
    source = Path(source_dir)
    result = {}

    for item in source.iterdir():
        if item.is_dir() or item.name.startswith("."):
            continue

        category = get_category(item.suffix)
        target_dir = source / category
        target_dir.mkdir(exist_ok=True)

        target = target_dir / item.name
        if target.exists():
            stem, suffix = item.stem, item.suffix
            counter = 1
            while target.exists():
                target = target_dir / f"{stem}_{counter}{suffix}"
                counter += 1

        shutil.move(str(item), str(target))
        result.setdefault(category, []).append(target.name)

    return result

if __name__ == "__main__":
    downloads = Path.home() / "Downloads"
    report = organize_folder(downloads)
    for cat, files in sorted(report.items()):
        print(f"\n{cat} ({len(files)} 个文件)")
        for f in files[:3]:
            print(f"  {f}")
```

核心逻辑：遍历 → 按扩展名分类 → 创建子目录 → 移动文件。同名文件自动加序号。

## 踩坑提醒

**用字符串拼接路径**：`base_dir + "\\" + filename` 在跨平台时出问题。用 `Path(base_dir) / filename`。

**watchdog 回调中执行耗时操作**：阻塞事件队列，后续事件丢失。放入队列单独处理。

**大文件复制用 `shutil.copy2`**：保留元数据，适合备份场景。

```python
from pathlib import Path
from collections import Counter

def count_extensions(directory: str) -> None:
    counter = Counter()
    for f in Path(directory).rglob("*"):
        if f.is_file():
            counter[f.suffix.lower() or "(无扩展名)"] += 1
    for ext, count in counter.most_common():
        print(f"  {ext}: {count}")

count_extensions(".")
```

### 练习二

```python
import shutil
import time
from datetime import date
from pathlib import Path
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class BackupHandler(FileSystemEventHandler):
    def __init__(self, backup_root: str):
        self.backup_root = Path(backup_root)

    def on_created(self, event):
        if event.is_directory:
            return
        src = Path(event.src_path)
        dest_dir = self.backup_root / date.today().isoformat()
        dest_dir.mkdir(parents=True, exist_ok=True)
        shutil.copy2(str(src), str(dest_dir / src.name))
        print(f"已备份: {src.name}")

observer = Observer()
observer.schedule(BackupHandler("D:/backup"), path="D:/watch", recursive=False)
observer.start()
try:
    while True:
        time.sleep(1)
except KeyboardInterrupt:
    observer.stop()
observer.join()
```

### 练习三

```python
import hashlib
from pathlib import Path
from collections import defaultdict

def file_hash(path: Path) -> str:
    h = hashlib.md5()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            h.update(chunk)
    return h.hexdigest()

def find_duplicates(directory: str) -> list[list[Path]]:
    size_map = defaultdict(list)
    for f in Path(directory).rglob("*"):
        if f.is_file():
            size_map[f.stat().st_size].append(f)

    duplicates = []
    for files in size_map.values():
        if len(files) < 2:
            continue
        hash_map = defaultdict(list)
        for f in files:
            hash_map[file_hash(f)].append(f)
        for group in hash_map.values():
            if len(group) > 1:
                duplicates.append(group)
    return duplicates
```

先按大小初筛减少哈希计算量；分块读取大文件避免内存溢出。
