# 图片批量处理

## 场景引入

你在运营一个电商店铺，供应商发来几百张产品图片，但尺寸不统一、文件太大、没有水印。你需要：统一尺寸为 800x800、压缩到 200KB 以内、加上店铺 Logo 水印。手动处理一张图要几分钟，批量处理用 Python 几行代码就能搞定。

## 学习目标

- 掌握 `Pillow` 的基础图片操作（打开、保存、格式转换）
- 能进行图片缩放、裁剪和旋转
- 能批量压缩图片到指定大小
- 能批量给图片加水印
- 能编写完整的图片批量处理脚本

## 图片缩放

```python
from PIL import Image

img = Image.open("large_photo.jpg")

# 直接指定尺寸（可能变形）
resized = img.resize((800, 600))

# 等比缩放到指定最大尺寸
def scale_image(img: Image.Image, max_w: int, max_h: int) -> Image.Image:
    ratio = min(max_w / img.width, max_h / img.height)
    if ratio >= 1:
        return img.copy()
    return img.resize((int(img.width * ratio), int(img.height * ratio)), Image.LANCZOS)

# 缩略图（直接修改原对象，只缩小不放大）
thumb = img.copy()
thumb.thumbnail((200, 200), Image.LANCZOS)
```

`Image.LANCZOS` 是高质量缩放算法，适合缩小图片。

## 图片裁剪

```python
from PIL import Image

img = Image.open("photo.jpg")

# 裁剪指定区域 (左, 上, 右, 下)
cropped = img.crop((100, 50, 500, 400))

# 居中裁剪为正方形
def center_crop_square(img: Image.Image) -> Image.Image:
    size = min(img.width, img.height)
    left, top = (img.width - size) // 2, (img.height - size) // 2
    return img.crop((left, top, left + size, top + size))

# 智能裁剪：先等比缩放使短边匹配，再居中裁剪
def crop_and_resize(img: Image.Image, target: tuple[int, int]) -> Image.Image:
    tw, th = target
    if img.width / img.height > tw / th:
        new_h, new_w = th, int(th * img.width / img.height)
    else:
        new_w, new_h = tw, int(tw * img.height / img.width)
    img = img.resize((new_w, new_h), Image.LANCZOS)
    left, top = (new_w - tw) // 2, (new_h - th) // 2
    return img.crop((left, top, left + tw, top + th))
```

## 批量压缩图片

```python
from PIL import Image
from pathlib import Path

def compress_image(input_path: str, output_path: str, max_kb: int = 200) -> None:
    img = Image.open(input_path)
    if img.mode == "RGBA":
        img = img.convert("RGB")
    low, high = 10, 95
    while low <= high:
        quality = (low + high) // 2
        img.save(output_path, "JPEG", quality=quality, optimize=True)
        if Path(output_path).stat().st_size / 1024 > max_kb:
            high = quality - 1
        else:
            low = quality + 1

def batch_compress(input_dir: str, output_dir: str, max_kb: int = 200) -> None:
    source, dest = Path(input_dir), Path(output_dir)
    dest.mkdir(parents=True, exist_ok=True)
    for f in source.glob("*"):
        if f.suffix.lower() not in {".jpg", ".jpeg", ".png", ".bmp", ".webp"}:
            continue
        out = dest / f"{f.stem}.jpg"
        compress_image(str(f), str(out), max_kb)
        print(f"{f.name}: {f.stat().st_size/1024:.0f}KB -> {out.stat().st_size/1024:.0f}KB")
```

二分法查找最佳质量：从 10-95 二分搜索，找到刚好满足大小限制的最高质量。

## 批量加水印

```python
from PIL import Image

def add_logo_watermark(img_path: str, logo_path: str, output_path: str,
                       position: str = "bottom-right", opacity: float = 0.5) -> None:
    img = Image.open(img_path).convert("RGBA")
    logo = Image.open(logo_path).convert("RGBA")
    lw = img.width // 5
    logo = logo.resize((lw, int(logo.height * lw / logo.width)), Image.LANCZOS)
    logo.putalpha(logo.getchannel("A").point(lambda x: int(x * opacity)))
    margin = 20
    pos_map = {
        "bottom-right": (img.width - lw - margin, img.height - logo.height - margin),
        "bottom-left": (margin, img.height - logo.height - margin),
    }
    img.paste(logo, pos_map.get(position, pos_map["bottom-right"]), logo)
    img.convert("RGB").save(output_path, "JPEG", quality=90)
```

## 完整示例：批量处理产品图片

```python
from PIL import Image, ImageDraw, ImageFont
from pathlib import Path
import io

class ImageProcessor:
    def __init__(self, target_size: tuple[int, int] = (800, 800), max_kb: int = 200):
        self.target_size = target_size
        self.max_kb = max_kb

    def crop_to_size(self, img: Image.Image) -> Image.Image:
        tw, th = self.target_size
        if img.width / img.height > tw / th:
            new_h, new_w = th, int(th * img.width / img.height)
        else:
            new_w, new_h = tw, int(tw * img.height / img.width)
        img = img.resize((new_w, new_h), Image.LANCZOS)
        left, top = (new_w - tw) // 2, (new_h - th) // 2
        return img.crop((left, top, left + tw, top + th))

    def add_watermark(self, img: Image.Image, text: str) -> Image.Image:
        if img.mode != "RGBA":
            img = img.convert("RGBA")
        overlay = Image.new("RGBA", img.size, (0, 0, 0, 0))
        draw = ImageDraw.Draw(overlay)
        try:
            font = ImageFont.truetype("arial.ttf", 24)
        except OSError:
            font = ImageFont.load_default()
        bbox = draw.textbbox((0, 0), text, font=font)
        x = img.width - (bbox[2] - bbox[0]) - 20
        draw.text((x, img.height - 60), text, fill=(255, 255, 255, 100), font=font)
        return Image.alpha_composite(img, overlay).convert("RGB")

    def compress_to_limit(self, img: Image.Image) -> bytes:
        quality = 90
        while quality > 10:
            buf = io.BytesIO()
            img.save(buf, "JPEG", quality=quality, optimize=True)
            if buf.tell() / 1024 <= self.max_kb:
                return buf.getvalue()
            quality -= 5
        buf = io.BytesIO()
        img.save(buf, "JPEG", quality=10, optimize=True)
        return buf.getvalue()

    def process_image(self, input_path: str, output_path: str, watermark: str = None) -> dict:
        img = Image.open(input_path)
        original_kb = Path(input_path).stat().st_size / 1024
        img = self.crop_to_size(img)
        if watermark:
            img = self.add_watermark(img, watermark)
        result = self.compress_to_limit(img)
        with open(output_path, "wb") as f:
            f.write(result)
        return {"original_kb": original_kb, "output_kb": len(result) / 1024}

    def batch_process(self, input_dir: str, output_dir: str, watermark: str = None) -> None:
        source, dest = Path(input_dir), Path(output_dir)
        dest.mkdir(parents=True, exist_ok=True)
        exts = {".jpg", ".jpeg", ".png", ".bmp", ".webp"}
        for f in sorted(source.iterdir()):
            if f.suffix.lower() not in exts:
                continue
            try:
                info = self.process_image(str(f), str(dest / f"{f.stem}.jpg"), watermark)
                print(f"✓ {f.name}: {info['original_kb']:.0f}KB -> {info['output_kb']:.0f}KB")
            except Exception as e:
                print(f"✗ {f.name}: {e}")

if __name__ == "__main__":
    processor = ImageProcessor(target_size=(800, 800), max_kb=200)
    processor.batch_process("D:/photos/original", "D:/photos/processed", "MyShop.com")
```

## 常见误区

### 1. RGBA 图片直接保存为 JPEG

```python
# ❌ JPEG 不支持透明通道，会报错
Image.open("logo.png").save("logo.jpg")
# ✅ 先转为 RGB
Image.open("logo.png").convert("RGB").save("logo.jpg")
```

### 2. 缩放时用错算法

```python
# ❌ 默认算法效果差
img.resize((800, 600))
# ✅ 用 LANCZOS
img.resize((800, 600), Image.LANCZOS)
```

### 3. thumbnail 直接修改原对象

```python
# ❌ 原图被覆盖
img = Image.open("photo.jpg")
img.thumbnail((200, 200))
# ✅ 先 copy
thumb = img.copy()
thumb.thumbnail((200, 200))
```

### 4. 同时打开大量图片导致内存溢出

```python
# ❌ 一次加载所有图片
images = [Image.open(f) for f in file_list]
# ✅ 逐个处理
for f in file_list:
    img = Image.open(f)
    process(img)
    img.close()
```

## 工程建议

1. **缩放用 `LANCZOS` 算法**：质量最高，适合缩小场景
2. **RGBA 转 RGB 再存 JPEG**：避免透明通道错误
3. **批量处理时逐个打开和关闭**：避免内存溢出
4. **压缩用二分法找最佳质量**：比固定质量更精确
5. **水印用 RGBA 透明层合成**：比直接绘制更灵活

## 参考答案

### 练习一

**思路**：遍历 PNG 文件，用 Pillow 打开后保存为 WebP，比较文件大小。

**答案**：

```python
from PIL import Image
from pathlib import Path

def convert_png_to_webp(input_dir: str, output_dir: str = None) -> None:
    source = Path(input_dir)
    dest = Path(output_dir) if output_dir else source / "webp_output"
    dest.mkdir(parents=True, exist_ok=True)
    for png in source.glob("*.png"):
        webp = dest / f"{png.stem}.webp"
        Image.open(png).save(str(webp), "WEBP", quality=85)
        orig_kb = png.stat().st_size / 1024
        conv_kb = webp.stat().st_size / 1024
        print(f"{png.name}: {orig_kb:.0f}KB -> {conv_kb:.0f}KB (节省 {(1-conv_kb/orig_kb)*100:.1f}%)")
```

**要点**：WebP 通常比 PNG 小 30%-50%；`quality=85` 是质量和大小的良好平衡。

### 练习二

**思路**：计算网格行列数，创建空白画布，逐个粘贴图片。

**答案**：

```python
from PIL import Image
from pathlib import Path
import math

def create_collage(image_paths: list[str], output: str,
                   cols: int = 2, cell: tuple[int, int] = (400, 400), gap: int = 10) -> None:
    rows = math.ceil(len(image_paths) / cols)
    cw, ch = cell
    collage = Image.new("RGB", (cols * cw + (cols + 1) * gap, rows * ch + (rows + 1) * gap), (255, 255, 255))
    for idx, path in enumerate(image_paths):
        r, c = idx // cols, idx % cols
        img = Image.open(path).convert("RGB").resize(cell, Image.LANCZOS)
        collage.paste(img, (gap + c * (cw + gap), gap + r * (ch + gap)))
    collage.save(output, quality=90)
```

**要点**：`math.ceil(n / cols)` 计算行数；每个格子位置通过行列索引计算。

### 练习三

**思路**：用 Pillow 的 `getexif()` 提取 EXIF 数据。

**答案**：

```python
from PIL import Image
from PIL.ExTags import TAGS
from pathlib import Path
import csv

def batch_extract_exif(input_dir: str, output_csv: str) -> None:
    all_data = []
    for f in sorted(Path(input_dir).glob("*.jpg")):
        exif = Image.open(f).getexif()
        row = {"文件名": f.name}
        for tag_id, value in exif.items():
            tag = TAGS.get(tag_id, tag_id)
            row[tag] = value.decode("utf-8", errors="replace") if isinstance(value, bytes) else value
        all_data.append(row)
    if not all_data:
        return
    keys = set()
    for d in all_data:
        keys.update(d.keys())
    with open(output_csv, "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.DictWriter(f, fieldnames=sorted(keys))
        writer.writeheader()
        writer.writerows(all_data)
    print(f"已保存 {len(all_data)} 条记录")
```

**要点**：`getexif()` 返回 EXIF 字典；`TAGS` 将标签 ID 转为可读名称；不同图片 EXIF 字段不同，需动态收集所有键。
