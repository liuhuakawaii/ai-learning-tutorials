```python
# src/autotool/config.py
import yaml
from pathlib import Path

DEFAULT_PATH = Path(__file__).parent.parent.parent / "config" / "default.yaml"

def load_config(path: str = None) -> dict:
    default = _load_yaml(DEFAULT_PATH)
    if path: default = _deep_merge(default, _load_yaml(Path(path)))
    return default

def _load_yaml(p: Path) -> dict:
    if not p.exists(): return {}
    with open(p, encoding="utf-8") as f: return yaml.safe_load(f) or {}

def _deep_merge(base: dict, over: dict) -> dict:
    r = base.copy()
    for k, v in over.items():
        if k in r and isinstance(r[k], dict) and isinstance(v, dict):
            r[k] = _deep_merge(r[k], v)
        else: r[k] = v
    return r
```

配置文件 `config/default.yaml` 定义默认阈值（disk: 80, memory: 85）和输出目录。

## 文件处理模块

```python
# src/autotool/file_ops/batch_rename.py
import re, hashlib
from pathlib import Path
from collections import defaultdict

def batch_rename(directory: Path, pattern: str, replacement: str,
                 dry_run: bool = True) -> list:
    results = []
    for fp in sorted(directory.iterdir()):
        if not fp.is_file(): continue
        new_name = re.sub(pattern, replacement, fp.name)
        if new_name == fp.name: continue
        results.append({"original": fp.name, "renamed": new_name,
                        "status": "preview" if dry_run else "done"})
        if not dry_run: fp.rename(fp.parent / new_name)
    return results

def file_hash(fp: Path) -> str:
    m = hashlib.md5()
    with open(fp, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""): m.update(chunk)
    return m.hexdigest()

def find_duplicates(directory: Path) -> dict:
    size_groups = defaultdict(list)
    for fp in directory.rglob("*"):
        if fp.is_file(): size_groups[fp.stat().st_size].append(fp)
    dupes = {}
    for size, files in size_groups.items():
        if len(files) < 2: continue
        hash_groups = defaultdict(list)
        for f in files: hash_groups[file_hash(f)].append(f)
        for h, group in hash_groups.items():
            if len(group) > 1: dupes[h] = group
    return dupes
```

## Web 监控与系统巡检

```python
# src/autotool/web_monitor/checker.py
import requests, time, psutil, socket
from dataclasses import dataclass

@dataclass
class CheckResult:
    url: str; status_code: int; response_time: float; is_up: bool; error: str = None

def check_url(url: str, timeout: int = 30) -> CheckResult:
    try:
        start = time.time()
        r = requests.get(url, timeout=timeout)
        return CheckResult(url, r.status_code, time.time() - start, r.status_code == 200)
    except requests.RequestException as e:
        return CheckResult(url, 0, 0, False, str(e))

@dataclass
class HealthReport:
    hostname: str; cpu_percent: float; memory_percent: float
    disk_info: list; warnings: list

def collect_health(thresholds: dict = None) -> HealthReport:
    t = thresholds or {"disk": 80, "memory": 85, "cpu": 90}
    warnings = []
    cpu = psutil.cpu_percent(interval=1)
    if cpu > t["cpu"]: warnings.append(f"CPU 过高: {cpu}%")
    mem = psutil.virtual_memory()
    if mem.percent > t["memory"]: warnings.append(f"内存过高: {mem.percent}%")
    disks = []
    for p in psutil.disk_partitions():
        try:
            u = psutil.disk_usage(p.mountpoint)
            disks.append({"device": p.device, "mount": p.mountpoint, "percent": u.percent})
            if u.percent > t["disk"]: warnings.append(f"磁盘 {p.mountpoint} 过高: {u.percent}%")
        except PermissionError: continue
    return HealthReport(socket.gethostname(), cpu, mem.percent, disks, warnings)
```

# src/autotool/report/excel.py
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill
from pathlib import Path

def generate_health_report(report, output: Path):
    wb = Workbook(); ws = wb.active
    hdr = PatternFill(start_color="4472C4", fill_type="solid")
    ws.cell(row=1, column=1, value="系统巡检报告").font = Font(bold=True, size=16)
    ws.cell(row=2, column=1, value=f"主机: {report.hostname}")
    for col, h in enumerate(["指标", "当前值", "状态"], 1):
        c = ws.cell(row=4, column=col, value=h)
        c.font = Font(bold=True, color="FFFFFF"); c.fill = hdr
    warn = PatternFill(start_color="FFC7CE", fill_type="solid")
    ok = PatternFill(start_color="C6EFCE", fill_type="solid")
    for r, (name, val, is_warn) in enumerate([
        ("CPU", f"{report.cpu_percent}%", report.cpu_percent > 90),
        ("内存", f"{report.memory_percent}%", report.memory_percent > 85),
    ], 5):
        ws.cell(row=r, column=1, value=name); ws.cell(row=r, column=2, value=val)
        c = ws.cell(row=r, column=3, value="告警" if is_warn else "正常")
        c.fill = warn if is_warn else ok
    wb.save(output)

## 常见误区

1. **模块紧耦合**：文件处理直接调用报表，应通过数据传递
2. **不写测试就重构**：没测试的重构是赌博
3. **配置硬编码**：阈值路径写死在代码里
4. **忽略错误处理**：外部调用不做异常处理
5. **CLI 输出不友好**：没进度提示、没颜色区分

## 工程建议

- 每个模块保持独立，通过参数和返回值传递数据
- click Group 组织子命令，保持 CLI 结构清晰
- 配置支持默认值覆盖，用户只需配置差异项
- 测试用 mock 隔离外部依赖

### 练习一

**思路**：独立函数实现，click 子命令注册。`find_large_files` 用 `rglob` 扫描后排序取 Top N，`clean_empty_dirs` 自底向上遍历删除空目录。

```python
from pathlib import Path

def find_large_files(directory: Path, n: int = 10) -> list:
    files = [(fp, fp.stat().st_size) for fp in directory.rglob("*") if fp.is_file()]
    return sorted(files, key=lambda x: x[1], reverse=True)[:n]

def clean_empty_dirs(directory: Path, dry_run: bool = True) -> list:
    removed = []
    for dp in sorted(directory.rglob("*"), reverse=True):
        if dp.is_dir() and not any(dp.iterdir()):
            removed.append(dp)
            if not dry_run: dp.rmdir()
    return removed

# cli.py: @file.command("large") / @file.command("clean-empty") 注册子命令
```

### 练习二

**思路**：策略模式支持多渠道，时间戳实现冷却。`should_alert` 检查距上次告警是否超过冷却时间。

```python
import time, requests, smtplib
from email.mime.text import MIMEText

class AlertManager:
    def __init__(self, config: dict):
        self.channels = config.get("channels", [])
        self.cooldown = config.get("cooldown", 3600)
        self._history = {}

    def should_alert(self, key: str) -> bool:
        return (time.time() - self._history.get(key, 0)) >= self.cooldown

    def send(self, title: str, msg: str):
        key = f"{title}:{msg[:50]}"
        if not self.should_alert(key): return
        for ch in self.channels:
            if ch["type"] == "email":
                m = MIMEText(msg, "plain", "utf-8")
                m["Subject"], m["From"], m["To"] = f"[告警] {title}", ch["from"], ch["to"]
                with smtplib.SMTP(ch["host"], ch["port"]) as s:
                    s.starttls(); s.login(ch["user"], ch["pass"]); s.send_message(m)
            elif ch["type"] == "webhook":
                requests.post(ch["url"], json={"text": f"[告警] {title}\n{msg}"}, timeout=10)
        self._history[key] = time.time()

# 测试：用 time.sleep 验证冷却逻辑，用 mock 验证 webhook 调用
```

**要点**：冷却用时间戳去重，渠道通过配置扩展，测试用 mock 隔离外部调用。
