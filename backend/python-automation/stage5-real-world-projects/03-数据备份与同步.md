# 数据备份与同步

## 场景引入

项目代码和文档分散在多个目录，每周手动复制到移动硬盘。上周硬盘坏了，发现最近一次备份是三周前的——手动备份太麻烦总忘记。而且每次全量复制几百 GB 要一个多小时，根本不可能每天执行。

我们需要一个自动化的增量备份工具：**只复制变化的文件**，几分钟完成，带完整性验证。

## 学习目标

- 掌握文件复制和增量备份的实现原理
- 理解 rsync 算法的核心思想
- 学会使用 Python 实现压缩备份
- 了解云存储（S3）的备份方案

## 文件复制备份

import shutil
from pathlib import Path
from datetime import datetime


def copy_directory(src: Path, dst: Path):
    """递归复制目录，排除常见无关目录"""
    ignore = shutil.ignore_patterns("__pycache__", "*.pyc", ".git", "node_modules")
    shutil.copytree(src, dst, ignore=ignore, dirs_exist_ok=True)


def backup_with_timestamp(source: Path, backup_root: Path) -> Path:
    ts = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_dir = backup_root / f"backup_{ts}"
    copy_directory(source, backup_dir)
    return backup_dir

## 增量备份（Hash 对比）

核心思想：通过文件哈希判断是否变化，只复制变化的文件。

```python
import hashlib, json
from pathlib import Path
from typing import Dict


def file_hash(filepath: Path) -> str:
    md5 = hashlib.md5()
    with open(filepath, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            md5.update(chunk)
    return md5.hexdigest()


def scan_directory(source: Path) -> Dict[str, dict]:
    """扫描目录，返回所有文件的路径和哈希"""
    index = {}
    for fp in source.rglob("*"):
        if fp.is_file():
            rel = str(fp.relative_to(source))
            index[rel] = {"hash": file_hash(fp), "size": fp.stat().st_size}
    return index


def diff_index(old: dict, new: dict):
    """对比新旧索引，返回变化的文件"""
    old_files, new_files = set(old.keys()), set(new.keys())
    added = new_files - old_files
    deleted = old_files - new_files
    modified = {p for p in old_files & new_files if old[p]["hash"] != new[p]["hash"]}
    return added, deleted, modified


class IncrementalBackup:
    def __init__(self, source: Path, backup_dir: Path):
        self.source = source
        self.backup_dir = backup_dir
        self.index_path = backup_dir / ".backup_index.json"

    def _load_index(self) -> dict:
        if self.index_path.exists():
            return json.loads(self.index_path.read_text(encoding="utf-8"))
        return {}

    def _save_index(self, index: dict):
        self.index_path.parent.mkdir(parents=True, exist_ok=True)
        self.index_path.write_text(json.dumps(index, indent=2), encoding="utf-8")

    def run(self) -> dict:
        print(f"扫描: {self.source}")
        current = scan_directory(self.source)
        old = self._load_index()
        added, deleted, modified = diff_index(old, current)

        if not added and not deleted and not modified:
            print("无变化，跳过"); return {"added": 0, "deleted": 0, "modified": 0}

        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_path = self.backup_dir / f"inc_{ts}"
        backup_path.mkdir(parents=True, exist_ok=True)

        import shutil
        for rel in added | modified:
            src, dst = self.source / rel, backup_path / rel
            dst.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(src, dst)

        if deleted:
            (backup_path / ".deleted.txt").write_text("\n".join(deleted), encoding="utf-8")

        self._save_index(current)
        result = {"added": len(added), "modified": len(modified), "deleted": len(deleted)}
        print(f"完成: 新增 {result['added']}, 修改 {result['modified']}, 删除 {result['deleted']}")
        return result
```

## rsync 原理

rsync 的核心是**滚动校验和**算法：将文件分块，通过弱校验和快速定位匹配，强校验和确认，只传输差异部分。

```python
import hashlib

BLOCK_SIZE = 4096

def weak_hash(data: bytes) -> int:
    """Adler-32 弱校验和"""
    a, b = 1, 0
    for byte in data: a = (a + byte) % 65521; b = (b + a) % 65521
    return (b << 16) | a

def strong_hash(data: bytes) -> str:
    return hashlib.md5(data).hexdigest()

def calculate_checksums(data: bytes):
    return [(weak_hash(data[i:i+BLOCK_SIZE]), strong_hash(data[i:i+BLOCK_SIZE]), i)
            for i in range(0, len(data), BLOCK_SIZE)]

def find_matches(source: bytes, checksums: list):
    matches = {}
    for i in range(0, len(source), BLOCK_SIZE):
        block = source[i:i+BLOCK_SIZE]
        w = weak_hash(block)
        for c_weak, c_strong, c_offset in checksums:
            if w == c_weak and strong_hash(block) == c_strong:
                matches[c_offset] = i; break
    return matches
```

## 压缩备份

```python
import tarfile
from pathlib import Path


def create_tar_backup(source: Path, output: Path) -> Path:
    ts = datetime.now().strftime("%Y%m%d_%H%M%S")
    path = output / f"backup_{ts}.tar.gz"
    with tarfile.open(path, "w:gz") as tar:
        tar.add(source, arcname=source.name)

    original = sum(f.stat().st_size for f in source.rglob("*") if f.is_file())
    compressed = path.stat().st_size
    ratio = (1 - compressed / original) * 100 if original > 0 else 0
    print(f"压缩: {original/1024/1024:.1f}MB → {compressed/1024/1024:.1f}MB ({ratio:.1f}%)")
    return path


def extract_backup(archive: Path, target: Path):
    with tarfile.open(archive, "r:gz") as tar:
        tar.extractall(path=target)
    print(f"解压完成: {target}")
```

## 云存储备份（S3）

```python
import boto3, hashlib
from pathlib import Path

class S3Backup:
    def __init__(self, bucket: str, prefix: str = "backups/"):
        self.s3, self.bucket, self.prefix = boto3.client("s3"), bucket, prefix

    def upload_file(self, local: Path, s3_key: str = None):
        key = s3_key or self.prefix + local.name
        self.s3.upload_file(str(local), self.bucket, key)
        print(f"✓ 上传: {local.name}")

    def upload_directory(self, directory: Path):
        count = 0
        for fp in directory.rglob("*"):
            if fp.is_file():
                self.upload_file(fp, self.prefix + str(fp.relative_to(directory)))
                count += 1
        print(f"共上传 {count} 个文件")

    def verify(self, local: Path, s3_key: str) -> bool:
        local_md5 = hashlib.md5(local.read_bytes()).hexdigest()
        resp = self.s3.head_object(Bucket=self.bucket, Key=s3_key)
        return local_md5 == resp["ETag"].strip('"')
```

## 备份验证

```python
import filecmp
from pathlib import Path


def verify_backup(source: Path, backup: Path) -> dict:
    src_files = {f.relative_to(source) for f in source.rglob("*") if f.is_file()}
    bak_files = {f.relative_to(backup) for f in backup.rglob("*") if f.is_file()}

    missing = src_files - bak_files
    mismatched = [
        str(f) for f in src_files & bak_files
        if not filecmp.cmp(source / f, backup / f, shallow=False)
    ]

    result = {
        "source_count": len(src_files),
        "backup_count": len(bak_files),
        "missing": sorted(str(f) for f in missing),
        "mismatched": mismatched,
        "is_valid": len(missing) == 0 and len(mismatched) == 0,
    }
    print(f"验证: {'✓ 通过' if result['is_valid'] else '✗ 失败'}")
    return result
```

## 常见误区

1. **只做全量不做增量**：全量耗时长占空间大，日常应增量为主
2. **备份后不验证**：备份文件可能损坏，必须定期验证可恢复性
3. **备份与源数据同位置**：机器故障时一起丢失
4. **保留无限版本**：不清理旧备份，磁盘撑满导致新备份失败
5. **忽略权限和软链接**：备份丢失文件权限信息

## 工程建议

- 采用"3-2-1"策略：3 份数据、2 种介质、1 份异地
- 每天增量 + 每周全量，备份文件名含时间戳
- 敏感数据备份加密，编写自动验证脚本

### 练习一

**思路**：增加配置文件读取、邮件通知和清理逻辑。

```python
import shutil, smtplib, json
from email.mime.text import MIMEText
from pathlib import Path
from datetime import datetime

class EnhancedBackup:
    def __init__(self, source: Path, backup_dir: Path, config: dict):
        self.source, self.backup_dir, self.config = source, backup_dir, config

    def should_exclude(self, rel_path: str) -> bool:
        for p in self.config.get("exclude", []):
            if rel_path.startswith(p) or rel_path.endswith(p): return True
        return False

    def run(self):
        from incremental_backup import scan_directory, diff_index
        current = {k: v for k, v in scan_directory(self.source).items()
                   if not self.should_exclude(k)}
        index_path = self.backup_dir / ".index.json"
        old = json.loads(index_path.read_text()) if index_path.exists() else {}
        added, deleted, modified = diff_index(old, current)
        if not added and not deleted and not modified: return
        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_path = self.backup_dir / f"inc_{ts}"
        backup_path.mkdir(parents=True, exist_ok=True)
        for rel in added | modified:
            dst = backup_path / rel
            dst.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(self.source / rel, dst)
        index_path.write_text(json.dumps(current))
        self.cleanup()
        self.notify({"added": len(added), "modified": len(modified)})

    def cleanup(self):
        keep = self.config.get("keep_backups", 10)
        backups = sorted(self.backup_dir.glob("inc_*"))
        for old in backups[:-keep]: shutil.rmtree(old)

    def notify(self, result: dict):
        cfg = self.config.get("email")
        if not cfg: return
        msg = MIMEText(f"备份: +{result['added']} ~{result['modified']}")
        msg["Subject"], msg["From"], msg["To"] = "备份完成", cfg["from"], cfg["to"]
        with smtplib.SMTP(cfg["host"], cfg["port"]) as s:
            s.starttls(); s.login(cfg["user"], cfg["password"]); s.send_message(msg)
```

### 练习二

**思路**：解压到临时目录，逐文件 MD5 对比，生成报告。

```python
import hashlib, tempfile, tarfile
from pathlib import Path

def file_hash(fp: Path) -> str:
    m = hashlib.md5()
    with open(fp, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""): m.update(chunk)
    return m.hexdigest()

def verify(archive: Path, source: Path) -> dict:
    with tempfile.TemporaryDirectory() as tmp:
        with tarfile.open(archive, "r:gz") as tar: tar.extractall(tmp)
        restored = next(Path(tmp).iterdir())
        src = {f.relative_to(source): f for f in source.rglob("*") if f.is_file()}
        rst = {f.relative_to(restored): f for f in restored.rglob("*") if f.is_file()}
        missing = set(src) - set(rst)
        matched, mismatched = 0, []
        for rel in set(src) & set(rst):
            if file_hash(src[rel]) == file_hash(rst[rel]): matched += 1
            else: mismatched.append(str(rel))
    ok = len(missing) == 0 and len(mismatched) == 0
    print(f"验证: {'✓' if ok else '✗'} 匹配{matched} 缺失{len(missing)} 不一致{len(mismatched)}")
    return {"matched": matched, "missing": sorted(map(str, missing)),
            "mismatched": mismatched, "is_valid": ok}
```

**要点**：临时目录自动清理，逐文件 MD5 比文件名更可靠，返回结构化报告。
