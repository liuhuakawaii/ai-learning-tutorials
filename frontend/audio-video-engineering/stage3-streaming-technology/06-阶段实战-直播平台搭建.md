# 第六课：阶段实战——直播平台搭建

## 场景引入

经过前五课的学习，你已经掌握了 HLS/DASH 流媒体协议、视频播放器开发、CDN 分发、推流技术和低延迟方案。现在是时候将这些知识串联起来，搭建一个**完整的迷你直播平台**。

这个平台需要支持：主播推流、服务端转码、多码率自适应播放、实时弹幕互动、观众人数统计。虽然规模远不及 B 站或抖音，但核心架构是一致的。

本课将从零开始，逐步搭建这个平台的每个模块。

## 学习目标

完成本课学习后，你将能够：

1. 设计直播平台的整体架构
2. 实现推流端（OBS 或 WebRTC）
3. 搭建 Node.js 媒体服务器
4. 开发自定义播放器前端
5. 添加实时互动功能（弹幕、观众数）
6. 处理流断开重连等异常情况

### 技术栈选择

| 层级 | 技术 | 说明 |
|------|------|------|
| 推流端 | Web Media API / FFmpeg | 浏览器推流或 OBS |
| 媒体服务器 | node-media-server | RTMP 接收 + HLS 输出 |
| 信令服务 | Socket.IO | 推流状态、观众计数 |
| 弹幕服务 | WebSocket | 实时消息广播 |
| 播放端 | hls.js + 自定义 UI | 自适应播放 + 互动 |
| 存储 | 文件系统 / 对象存储 | 录制回放 |

### 媒体服务器

```javascript
// server/media-server.js
const NodeMediaServer = require('node-media-server');
const config = require('./config');
const fs = require('fs');
const path = require('path');

class LiveMediaServer {
  constructor() {
    this.streams = new Map();  // 活跃流信息
    this.nms = null;
  }

  start() {
    const nmsConfig = {
      rtmp: config.rtmp,
      http: config.http,
      trans: {
        ffmpeg: config.transcode.ffmpeg,
        tasks: config.transcode.profiles.map(profile => ({
          app: 'live',
          hls: true,
          hlsFlags: '[hls_time=2:hls_list_size=10:hls_flags=delete_segments]',
          hlsKeep: false,
          vc: 'libx264',
          vcParam: [
            ['-vf', `scale=${profile.size}`],
            ['-b:v', profile.bitrate],
            ['-g', '60'],
            ['-keyint_min', '60'],
            ['-sc_threshold', '0'],
            ['-preset', 'veryfast'],
            ['-tune', 'zerolatency'],
          ],
          ac: 'aac',
          acParam: ['-b:a', '128k', '-ar', '44100'],
        })),
      },
    };

    this.nms = new NodeMediaServer(nmsConfig);

    this.nms.on('prePublish', (id, StreamPath, args) => {
      this.handlePrePublish(id, StreamPath, args);
    });

    this.nms.on('postPublish', (id, StreamPath, args) => {
      this.handlePostPublish(id, StreamPath, args);
    });

    this.nms.on('donePublish', (id, StreamPath, args) => {
      this.handleDonePublish(id, StreamPath, args);
    });

    this.nms.run();
    console.log('[媒体服务器] 已启动');
    console.log(`[RTMP] rtmp://localhost:${config.rtmp.port}/live/<stream_key>`);
    console.log(`[HLS]  http://localhost:${config.http.port}/live/<stream_key>/index.m3u8`);
  }

  handlePrePublish(id, StreamPath, args) {
    const streamKey = StreamPath.split('/').pop();

    // 简单的密钥验证
    if (!streamKey || streamKey.length < 8) {
      const session = this.nms.getSession(id);
      session.reject();
      console.log(`[推流拒绝] 无效的流密钥: ${streamKey}`);
      return;
    }

    console.log(`[推流开始] 密钥: ${streamKey}`);
  }

  handlePostPublish(id, StreamPath, args) {
    const streamKey = StreamPath.split('/').pop();

    this.streams.set(streamKey, {
      id,
      startTime: Date.now(),
      viewerCount: 0,
      status: 'live',
    });

    // 确保 HLS 目录存在
    const hlsDir = path.join(config.http.mediaroot, 'live', streamKey);
    fs.mkdirSync(hlsDir, { recursive: true });

    console.log(`[直播中] ${streamKey}`);

    // 通知信令服务器
    if (this.onStreamStart) {
      this.onStreamStart(streamKey);
    }
  }

  handleDonePublish(id, StreamPath, args) {
    const streamKey = StreamPath.split('/').pop();

    if (this.streams.has(streamKey)) {
      const stream = this.streams.get(streamKey);
      stream.status = 'ended';
      stream.endTime = Date.now();
      stream.duration = (stream.endTime - stream.startTime) / 1000;
    }

    console.log(`[推流结束] ${streamKey}, 时长: ${this.streams.get(streamKey)?.duration?.toFixed(1)}s`);

    if (this.onStreamEnd) {
      this.onStreamEnd(streamKey);
    }
  }

  getStreamInfo(streamKey) {
    return this.streams.get(streamKey) || null;
  }

  getActiveStreams() {
    const active = [];
    this.streams.forEach((info, key) => {
      if (info.status === 'live') {
        active.push({ streamKey: key, ...info });
      }
    });
    return active;
  }
}

module.exports = LiveMediaServer;
```

### 信令与弹幕服务

```javascript
// server/signaling.js
const express = require('express');
const http = require('http');
const { Server } = require('socket.io');
const path = require('path');
const config = require('./config');

class SignalingServer {
  constructor(mediaServer) {
    this.mediaServer = mediaServer;
    this.app = express();
    this.server = http.createServer(this.app);
    this.io = new Server(this.server, {
      cors: { origin: '*' },
    });

    this.chatHistory = new Map();  // 每个流的弹幕历史
    this.viewerSockets = new Map();  // 观众 socket 映射

    this.setupRoutes();
    this.setupSocketHandlers();
    this.setupMediaServerCallbacks();
  }

  setupRoutes() {
    // 静态文件
    this.app.use('/live', express.static(path.join(__dirname, '..', 'media', 'live')));
    this.app.use(express.static(path.join(__dirname, '..', 'web')));

    // API：获取活跃流列表
    this.app.get('/api/streams', (req, res) => {
      res.json(this.mediaServer.getActiveStreams());
    });

    // API：获取流信息
    this.app.get('/api/streams/:streamKey', (req, res) => {
      const info = this.mediaServer.getStreamInfo(req.params.streamKey);
      if (!info) {
        return res.status(404).json({ error: '流不存在' });
      }
      res.json(info);
    });
  }

  setupSocketHandlers() {
    this.io.on('connection', (socket) => {
      console.log(`[Socket] 连接: ${socket.id}`);

      // 加入直播间
      socket.on('join-stream', (streamKey) => {
        socket.join(`stream:${streamKey}`);
        socket.currentStream = streamKey;

        if (!this.viewerSockets.has(streamKey)) {
          this.viewerSockets.set(streamKey, new Set());
        }
        this.viewerSockets.get(streamKey).add(socket.id);

        // 更新观众计数
        const viewerCount = this.viewerSockets.get(streamKey).size;
        this.io.to(`stream:${streamKey}`).emit('viewer-count', viewerCount);

        // 发送历史弹幕
        const history = this.chatHistory.get(streamKey) || [];
        socket.emit('chat-history', history.slice(-50));

        // 更新流信息中的观众数
        const streamInfo = this.mediaServer.getStreamInfo(streamKey);
        if (streamInfo) streamInfo.viewerCount = viewerCount;

        console.log(`[观众] ${socket.id} 加入 ${streamKey}, 当前 ${viewerCount} 人`);
      });

      // 离开直播间
      socket.on('leave-stream', (streamKey) => {
        this.handleLeaveStream(socket, streamKey);
      });

      // 发送弹幕
      socket.on('chat-message', (data) => {
        const { streamKey, message, nickname } = data;

        if (!streamKey || !message || message.length > 200) return;

        const chatMsg = {
          id: Date.now().toString(36) + Math.random().toString(36).slice(2, 6),
          nickname: nickname || '匿名用户',
          message: message.trim(),
          timestamp: Date.now(),
          color: this.getRandomColor(),
        };

        // 保存到历史
        if (!this.chatHistory.has(streamKey)) {
          this.chatHistory.set(streamKey, []);
        }
        const history = this.chatHistory.get(streamKey);
        history.push(chatMsg);
        if (history.length > 500) history.shift();

        // 广播给直播间所有人
        this.io.to(`stream:${streamKey}`).emit('chat-message', chatMsg);
      });

      // 断开连接
      socket.on('disconnect', () => {
        if (socket.currentStream) {
          this.handleLeaveStream(socket, socket.currentStream);
        }
      });
    });
  }

  handleLeaveStream(socket, streamKey) {
    const viewers = this.viewerSockets.get(streamKey);
    if (viewers) {
      viewers.delete(socket.id);
      const viewerCount = viewers.size;
      this.io.to(`stream:${streamKey}`).emit('viewer-count', viewerCount);

      const streamInfo = this.mediaServer.getStreamInfo(streamKey);
      if (streamInfo) streamInfo.viewerCount = viewerCount;

      if (viewers.size === 0) {
        this.viewerSockets.delete(streamKey);
      }
    }
    socket.leave(`stream:${streamKey}`);
    socket.currentStream = null;
  }

  setupMediaServerCallbacks() {
    this.mediaServer.onStreamStart = (streamKey) => {
      this.io.emit('stream-start', { streamKey });
    };

    this.mediaServer.onStreamEnd = (streamKey) => {
      this.io.to(`stream:${streamKey}`).emit('stream-end', { streamKey });
      this.chatHistory.delete(streamKey);
      this.viewerSockets.delete(streamKey);
    };
  }

  getRandomColor() {
    const colors = ['#ff6b6b', '#ffd93d', '#6bcb77', '#4d96ff', '#ff6bff', '#ff9f43'];
    return colors[Math.floor(Math.random() * colors.length)];
  }

  start() {
    this.server.listen(config.signaling.port, () => {
      console.log(`[信令服务] 运行在 http://localhost:${config.signaling.port}`);
    });
  }
}

module.exports = SignalingServer;
```

### 播放器模块

```javascript
// web/player.js
class LivePlayer {
  constructor(videoElement) {
    this.video = videoElement;
    this.hls = null;
    this.currentLevel = -1;
  }

  load(streamUrl) {
    if (this.hls) {
      this.hls.destroy();
    }

    if (Hls.isSupported()) {
      this.hls = new Hls({
        lowLatencyMode: true,
        maxBufferLength: 3,
        backBufferLength: 30,
        startLevel: -1,
      });

      this.hls.loadSource(streamUrl);
      this.hls.attachMedia(this.video);

      this.hls.on(Hls.Events.MANIFEST_PARSED, (_, data) => {
        console.log(`[播放器] 可用质量: ${data.levels.length} 个`);
        this.video.play().catch(() => {});
      });

      this.hls.on(Hls.Events.LEVEL_SWITCHED, (_, data) => {
        const level = this.hls.levels[data.level];
        console.log(`[播放器] 切换到: ${level.height}p`);
      });

      this.hls.on(Hls.Events.ERROR, (_, data) => {
        if (data.fatal) {
          if (data.type === Hls.ErrorTypes.NETWORK_ERROR) {
            console.warn('[播放器] 网络错误，尝试恢复...');
            setTimeout(() => this.hls.startLoad(), 3000);
          } else if (data.type === Hls.ErrorTypes.MEDIA_ERROR) {
            this.hls.recoverMediaError();
          }
        }
      });

      return true;
    } else if (this.video.canPlayType('application/vnd.apple.mpegurl')) {
      this.video.src = streamUrl;
      return true;
    }

    return false;
  }

  getQualityLevels() {
    if (!this.hls) return [];
    return this.hls.levels.map((level, index) => ({
      index,
      height: level.height,
      bitrate: level.bitrate,
      label: `${level.height}p`,

    this.hls.currentLevel = -1;

      this.video.play().catch(() => {});
    } else {

    return this.video.muted;

      document.exitFullscreen();
    } else {

    for (let i = 0; i < this.video.buffered.length; i++) {
      if (this.video.buffered.start(i) <= currentTime &&
          this.video.buffered.end(i) > currentTime) {
        return this.video.buffered.end(i) - currentTime;
```

```javascript
      this.streamKey = null;
    }
  }

    const message = this.input.value.trim();

    this.socket.emit('chat-message', {

      message,
      nickname: this.nicknameInput.value.trim() || '匿名用户',
```

## 常见误区

### 误区一：不需要处理流断开重连

推流端网络波动会导致流中断。服务端需要检测流断开事件，播放端需要实现自动重连逻辑（hls.js 的 `startLoad` 会自动重试）。

### 误区二：弹幕不需要频率限制

没有频率限制的弹幕系统会被恶意用户刷屏。需要在服务端实现：每用户每秒最多 3 条、消息长度上限 200 字符、重复消息过滤。

### 误区三：HLS 分片目录不需要清理

`hls_list_size=10` 只控制播放列表中保留的分片数量，但旧的 TS 文件可能仍然在磁盘上。必须使用 `delete_segments` 标志或定时任务清理。

### 误区四：观众计数可以用数据库

观众计数是高频更新（每次进出都变化），不适合写数据库。使用内存中的 Set 计数 + Socket.IO 广播即可。

## 工程建议

1. **监控推流健康**：每隔 5 秒检查推流的码率、帧率、丢帧率，异常时告警
2. **使用 GPU 转码**：生产环境用 NVENC 替代 libx264，单卡可同时转码 20+ 路
3. **弹幕消息压缩**：高频弹幕场景下使用 protobuf 替代 JSON，减少传输大小
4. **流密钥使用 UUID**：避免可预测的密钥被猜测
5. **HLS 路径加时间戳**：`/live/{streamKey}/{timestamp}/playlist.m3u8` 避免缓存冲突
6. **优雅停播**：推流结束后保持 HLS 分片可访问一段时间，让观众看完缓冲中的内容

## 小结

本课综合运用了前五课的知识，搭建了一个完整的迷你直播平台：

- **媒体服务器**：node-media-server 接收 RTMP 推流，输出 HLS 多码率

**思路**：监听推流开始/结束事件，在推流期间将 HLS 分片复制到录制目录，结束后生成完整的播放列表。

**答案**：

```javascript
// server/recorder.js
const fs = require('fs');
const path = require('path');
const chokidar = require('chokidar');

class LiveRecorder {
  constructor(mediaRoot, recordRoot) {
    this.mediaRoot = mediaRoot;
    this.recordRoot = recordRoot;
    this.recordings = new Map();
  }

  startRecording(streamKey) {
    const hlsDir = path.join(this.mediaRoot, 'live', streamKey);
    const recordDir = path.join(this.recordRoot, streamKey);
    fs.mkdirSync(recordDir, { recursive: true });

    const state = {
      segments: [],
      startTime: Date.now(),
      watcher: null,
    };

    // 等待 HLS 目录创建后再监控
    const checkAndWatch = () => {
      if (fs.existsSync(hlsDir)) {
        const watcher = chokidar.watch(path.join(hlsDir, '*.ts'), {
          ignoreInitial: true,
        });

        watcher.on('add', (filePath) => {
          const filename = path.basename(filePath);
          const dest = path.join(recordDir, filename);

          setTimeout(() => {
            try {
              if (fs.existsSync(filePath)) {
                fs.copyFileSync(filePath, dest);
                state.segments.push({ filename, time: Date.now() });
                console.log(`[录制] ${streamKey}: ${filename}`);
              }
            } catch (err) {
              console.error(`[录制] 复制失败: ${filename}`, err.message);
            }
          }, 500);
        });

        state.watcher = watcher;
      } else {
        setTimeout(checkAndWatch, 1000);
      }
    };

    checkAndWatch();
    this.recordings.set(streamKey, state);
    console.log(`[录制] 开始: ${streamKey}`);
  }

  stopRecording(streamKey) {
    const state = this.recordings.get(streamKey);
    if (!state) return null;

    if (state.watcher) state.watcher.close();

    const recordDir = path.join(this.recordRoot, streamKey);
    const duration = (Date.now() - state.startTime) / 1000;

    // 生成播放列表
    let playlist = '#EXTM3U\n#EXT-X-VERSION:3\n';
    playlist += '#EXT-X-TARGETDURATION:2\n';
    playlist += '#EXT-X-MEDIA-SEQUENCE:0\n';
    state.segments.forEach(seg => {
      playlist += `#EXTINF:2.000,\n${seg.filename}\n`;
    });
    playlist += '#EXT-X-ENDLIST\n';

    fs.writeFileSync(path.join(recordDir, 'playlist.m3u8'), playlist);

    this.recordings.delete(streamKey);
    console.log(`[录制] 结束: ${streamKey}, ${state.segments.length} 分片, ${duration.toFixed(1)}s`);

    return {
      streamKey,
      segmentCount: state.segments.length,
      duration,
      playlistUrl: `/recordings/${streamKey}/playlist.m3u8`,
    };
  }
}

module.exports = LiveRecorder;
```

**要点**：
- 使用 chokidar 监控分片目录，实时复制新分片
- 延迟 500ms 复制确保分片已完全写入
- 生成带 `#EXT-X-ENDLIST` 的播放列表支持完整回放

```javascript
    this.container = container;
    this.onSelect = onSelect;
    this.createElement();
  }

  createElement() {
    this.panel = document.createElement('div');
    this.panel.className = 'emoji-panel';
    this.panel.style.cssText = `
      display: none;
      position: absolute;
      bottom: 100%;
      left: 0;
      background: #2a2a2a;
      border: 1px solid #444;
      border-radius: 8px;
      padding: 8px;
      width: 280px;
      flex-wrap: wrap;
      gap: 4px;
    `;

    for (const [key, emoji] of Object.entries(EMOJI_MAP)) {
      const btn = document.createElement('button');
      btn.textContent = emoji;
      btn.title = key;
      btn.style.cssText = `
        background: none;
        border: none;
        font-size: 20px;
        cursor: pointer;
        padding: 4px;
        border-radius: 4px;
      `;
      btn.addEventListener('click', () => {
        this.onSelect(key);
        this.hide();
      });
      btn.addEventListener('mouseenter', () => {
        btn.style.background = 'rgba(255,255,255,0.1)';
      });
      btn.addEventListener('mouseleave', () => {
        btn.style.background = 'none';
      });
      this.panel.appendChild(btn);
    }

    this.container.appendChild(this.panel);
  }

  toggle() {
    this.panel.style.display = this.panel.style.display === 'none' ? 'flex' : 'none';
  }

  hide() {
    this.panel.style.display = 'none';
  }
}
```
