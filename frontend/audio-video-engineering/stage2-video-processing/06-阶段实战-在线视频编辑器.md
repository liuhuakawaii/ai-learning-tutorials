# 第六课：阶段实战 —— 在线视频编辑器

## 场景引入

前五课我们分别学习了编解码原理、FFmpeg.wasm、视频滤镜、视频合成、字幕水印。现在是时候把这些技术整合起来，构建一个浏览器端的在线视频编辑器。

这个编辑器不需要像 Premiere 那样功能齐全，但要覆盖视频编辑的核心流程：导入素材、在时间线上排列片段、添加滤镜和字幕、实时预览、最终导出。通过这个实战项目，你将理解音视频工程中各模块如何协作。

### 核心模块

```javascript
class VideoEditor {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');

    // 核心模块
    this.timeline = new Timeline();
    this.assetManager = new AssetManager();
    this.previewRenderer = new PreviewRenderer(canvas);
    this.subtitleManager = new SubtitleManager();
    this.filterManager = new FilterManager();
    this.exporter = new VideoExporter(canvas);

    // 状态管理
    this.isPlaying = false;
    this.currentTime = 0;
    this.undoStack = [];
    this.redoStack = [];
  }

  // 导入素材
  async importAsset(file) {
    const asset = await this.assetManager.add(file);
    this.saveState();
    return asset;
  }

  // 添加片段到时间线
  addClipToTimeline(assetId, trackId, startTime) {
    const asset = this.assetManager.get(assetId);
    if (!asset) return;

    const clip = this.timeline.addClip(trackId, {
      sourceUrl: asset.url,
      startTime,
      duration: asset.duration,
    });

    this.saveState();
    return clip;
  }

  // 播放控制
  play() {
    this.isPlaying = true;
    this.lastFrameTime = performance.now();
    this.renderLoop();
  }

  pause() {
    this.isPlaying = false;
  }

  seek(time) {
    this.currentTime = Math.max(0, Math.min(time, this.timeline.duration));
    this.previewRenderer.render(this.timeline, this.currentTime);
  }

  // 渲染循环
  renderLoop() {
    if (!this.isPlaying) return;

    const now = performance.now();
    const delta = (now - this.lastFrameTime) / 1000;
    this.lastFrameTime = now;

    this.currentTime += delta;
    if (this.currentTime >= this.timeline.duration) {
      this.currentTime = 0;
      this.isPlaying = false;
      return;
    }

    this.previewRenderer.render(this.timeline, this.currentTime);
    this.updateTimelineUI();

    requestAnimationFrame(() => this.renderLoop());
  }

  // 撤销/重做
  saveState() {
    this.undoStack.push(this.timeline.serialize());
    this.redoStack = [];
  }

  undo() {
    if (this.undoStack.length === 0) return;
    this.redoStack.push(this.timeline.serialize());
    const state = this.undoStack.pop();
    this.timeline.deserialize(state);
  }

  redo() {
    if (this.redoStack.length === 0) return;
    this.undoStack.push(this.timeline.serialize());
    const state = this.redoStack.pop();
    this.timeline.deserialize(state);
  }

  // 导出
  async exportVideo(options = {}) {
    return this.exporter.exportTimeline(this.timeline, options);
  }
}
```

### AssetManager

```javascript
class AssetManager {
  constructor() {
    this.assets = new Map();
    this.videoElements = new Map();
  }

  async add(file) {
    const id = `asset_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
    const url = URL.createObjectURL(file);

    let duration = 0;
    let type = 'unknown';

    if (file.type.startsWith('video/')) {
      type = 'video';
      duration = await this.getVideoDuration(url);
      this.videoElements.set(id, await this.createVideoElement(url));
    } else if (file.type.startsWith('image/')) {
      type = 'image';
      duration = 5; // 图片默认 5 秒
    } else if (file.type.startsWith('audio/')) {
      type = 'audio';
      duration = await this.getAudioDuration(url);
    }

    const asset = {
      id,
      name: file.name,
      type,
      file,
      url,
      duration,
      thumbnail: type === 'video' ? await this.generateThumbnail(url) : null,
    };

    this.assets.set(id, asset);
    return asset;
  }

  get(id) {
    return this.assets.get(id);
  }

  getVideoElement(id) {
    return this.videoElements.get(id);
  }

  getVideoDuration(url) {
    return new Promise((resolve) => {
      const video = document.createElement('video');
      video.preload = 'metadata';
      video.onloadedmetadata = () => resolve(video.duration);
      video.src = url;
    });
  }

  getAudioDuration(url) {
    return new Promise((resolve) => {
      const audio = new Audio();
      audio.preload = 'metadata';
      audio.onloadedmetadata = () => resolve(audio.duration);
      audio.src = url;
    });
  }

  createVideoElement(url) {
    return new Promise((resolve) => {
      const video = document.createElement('video');
      video.muted = true;
      video.preload = 'auto';
      video.onloadeddata = () => resolve(video);
      video.src = url;
    });
  }

  async generateThumbnail(url) {
    return new Promise((resolve) => {
      const video = document.createElement('video');
      video.currentTime = 1;
      video.onseeked = () => {
        const canvas = document.createElement('canvas');
        canvas.width = 160;
        canvas.height = 90;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(video, 0, 0, 160, 90);
        resolve(canvas.toDataURL('image/jpeg', 0.7));
      };
      video.src = url;
    });
  }

  dispose() {
    for (const asset of this.assets.values()) {
      URL.revokeObjectURL(asset.url);
    }
    this.assets.clear();
    this.videoElements.clear();
  }
}
```

## 时间线实现

### 可交互的时间线

```javascript
class TimelineUI {
  constructor(container, timeline) {
    this.container = container;
    this.timeline = timeline;
    this.pixelsPerSecond = 50; // 时间线缩放
    this.scrollLeft = 0;
    this.dragState = null;

    this.render();
    this.bindEvents();
  }

  render() {
    this.container.innerHTML = `
      <div class="timeline-header">
        <div class="time-ruler" id="time-ruler"></div>
      </div>
      <div class="timeline-tracks" id="timeline-tracks"></div>
      <div class="timeline-playhead" id="playhead"></div>
    `;

    this.renderTimeRuler();
    this.renderTracks();
  }

  renderTimeRuler() {
    const ruler = this.container.querySelector('#time-ruler');
    const totalWidth = this.timeline.duration * this.pixelsPerSecond;

    let html = '';
    for (let t = 0; t <= this.timeline.duration; t++) {
      const x = t * this.pixelsPerSecond;
      const isMajor = t % 5 === 0;
      html += `
        <div class="ruler-mark ${isMajor ? 'major' : 'minor'}"
             style="left: ${x}px">
          ${isMajor ? this.formatTime(t) : ''}
        </div>
      `;
    }
    ruler.innerHTML = html;
    ruler.style.width = `${totalWidth}px`;
  }

  renderTracks() {
    const tracksContainer = this.container.querySelector('#timeline-tracks');
    let html = '';

    for (const track of this.timeline.tracks) {
      html += `<div class="track" data-track-id="${track.id}">`;
      html += `<div class="track-label">${track.type}</div>`;
      html += `<div class="track-clips">`;

      for (const clip of track.clips) {
        const left = clip.startTime * this.pixelsPerSecond;
        const width = clip.duration * this.pixelsPerSecond;
        html += `
          <div class="clip" data-clip-id="${clip.id}"
               style="left: ${left}px; width: ${width}px">
            <span class="clip-name">${clip.sourceUrl.split('/').pop()}</span>
          </div>
        `;
      }

      html += `</div></div>`;
    }

    tracksContainer.innerHTML = html;
  }

  bindEvents() {
    const tracks = this.container.querySelector('#timeline-tracks');

    // 片段拖拽
    tracks.addEventListener('mousedown', (e) => {
      const clipEl = e.target.closest('.clip');
      if (!clipEl) return;

      const clipId = clipEl.dataset.clipId;
      const rect = clipEl.getBoundingClientRect();
      const offsetX = e.clientX - rect.left;

      this.dragState = {
        clipId,
        offsetX,
        startX: e.clientX,
        originalLeft: parseFloat(clipEl.style.left),
      };

      clipEl.classList.add('dragging');
    });

    document.addEventListener('mousemove', (e) => {
      if (!this.dragState) return;

      const dx = e.clientX - this.dragState.startX;
      const newLeft = this.dragState.originalLeft + dx;
      const newTime = Math.max(0, newLeft / this.pixelsPerSecond);

      // 更新时间线数据
      const clip = this.findClip(this.dragState.clipId);
      if (clip) {
        clip.startTime = newTime;
      }

      // 更新 UI
      const clipEl = this.container.querySelector(
        `.clip[data-clip-id="${this.dragState.clipId}"]`
      );
      if (clipEl) {
        clipEl.style.left = `${newLeft}px`;
      }
    });

    document.addEventListener('mouseup', () => {
      if (this.dragState) {
        const clipEl = this.container.querySelector(
          `.clip[data-clip-id="${this.dragState.clipId}"]`
        );
        if (clipEl) {
          clipEl.classList.remove('dragging');
        }
        this.dragState = null;
      }
    });

    // 时间线点击跳转
    this.container.querySelector('#timeline-tracks').addEventListener('click', (e) => {
      if (e.target.closest('.clip')) return;
      const rect = e.currentTarget.getBoundingClientRect();
      const x = e.clientX - rect.left + this.scrollLeft;
      const time = x / this.pixelsPerSecond;
      this.seekTo(time);
    });
  }

  findClip(clipId) {
    for (const track of this.timeline.tracks) {
      const clip = track.clips.find(c => c.id === clipId);
      if (clip) return clip;
    }
    return null;
  }

  seekTo(time) {
    this.currentTime = time;
    this.updatePlayhead();
  }

  updatePlayhead() {
    const playhead = this.container.querySelector('#playhead');
    const x = this.currentTime * this.pixelsPerSecond;
    playhead.style.left = `${x}px`;
  }

  formatTime(seconds) {
    const m = Math.floor(seconds / 60);
    const s = Math.floor(seconds % 60);
    return `${m}:${s.toString().padStart(2, '0')}`;
  }
}
```

### PreviewRenderer

```javascript
class PreviewRenderer {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
    this.filterRenderer = null;
    this.subtitleRenderer = null;
    this.watermarkRenderer = null;
  }

  render(timeline, currentTime) {
    const ctx = this.ctx;
    ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    // 渲染视频轨道
    for (const track of timeline.tracks) {
      if (track.type !== 'video') continue;

      const activeClip = this.getActiveClip(track, currentTime);
      if (!activeClip) continue;

      const videoElement = this.getVideoForClip(activeClip);
      if (!videoElement) continue;

      // 设置视频播放位置
      const clipTime = currentTime - activeClip.startTime + activeClip.trimStart;
      videoElement.currentTime = clipTime;

      // 绘制视频帧
      ctx.drawImage(videoElement, 0, 0, this.canvas.width, this.canvas.height);
    }

    // 应用滤镜
    if (this.filterRenderer) {
      this.filterRenderer.apply(this.canvas);
    }

    // 渲染字幕
    if (this.subtitleRenderer) {
      this.subtitleRenderer.render(ctx, currentTime);
    }

    // 渲染水印
    if (this.watermarkRenderer) {
      this.watermarkRenderer.render(ctx);
    }
  }

  getActiveClip(track, time) {
    return track.clips.find(
      clip => time >= clip.startTime && time < clip.startTime + clip.duration
    );
  }

  getVideoForClip(clip) {
    // 从 AssetManager 获取对应的 video 元素
    return clip._videoElement || null;
  }
}
```

## 导出功能

### 完整的导出流程

```javascript
class TimelineExporter {
  constructor(canvas) {
    this.canvas = canvas;
    this.stream = canvas.captureStream(30);
    this.recorder = null;
    this.chunks = [];
  }

  async exportTimeline(timeline, options = {}) {
    const fps = options.fps || 30;
    const bitrate = options.bitrate || 5000000;
    const mimeType = this.getSupportedMimeType();

    this.recorder = new MediaRecorder(this.stream, {
      mimeType,
      videoBitsPerSecond: bitrate,
    });

    this.chunks = [];
    this.recorder.ondataavailable = (e) => {
      if (e.data.size > 0) this.chunks.push(e.data);
    };

    return new Promise((resolve, reject) => {
      this.recorder.onstop = () => {
        const blob = new Blob(this.chunks, { type: mimeType });
        resolve(blob);
      };

      this.recorder.onerror = reject;
      this.recorder.start(100);

      // 启动渲染循环
      this.startExportLoop(timeline, fps);
    });
  }

  startExportLoop(timeline, fps) {
    const frameDuration = 1000 / fps;
    const totalFrames = Math.ceil(timeline.duration * fps);
    let frame = 0;

    const renderNextFrame = () => {
      if (frame >= totalFrames) {
        this.recorder.stop();
        return;
      }

      const currentTime = frame / fps;

      // 更新所有视频元素的播放位置
      for (const track of timeline.tracks) {
        if (track.type !== 'video') continue;
        const clip = track.clips.find(
          c => currentTime >= c.startTime && currentTime < c.startTime + c.duration
        );
        if (clip && clip._videoElement) {
          clip._videoElement.currentTime = currentTime - clip.startTime + clip.trimStart;
        }
      }

      // 渲染当前帧
      this.renderFrame(timeline, currentTime);

      frame++;
      setTimeout(renderNextFrame, frameDuration);
    };

    renderNextFrame();
  }

  renderFrame(timeline, currentTime) {
    const ctx = this.canvas.getContext('2d');
    ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    for (const track of timeline.tracks) {
      if (track.type !== 'video') continue;
      const clip = track.clips.find(
        c => currentTime >= c.startTime && currentTime < c.startTime + c.duration
      );
      if (clip && clip._videoElement) {
        ctx.drawImage(clip._videoElement, 0, 0, this.canvas.width, this.canvas.height);
      }
    }
  }

  getSupportedMimeType() {
    const types = [
      'video/webm;codecs=vp9,opus',
      'video/webm;codecs=vp8,opus',
      'video/webm',
    ];
    return types.find(t => MediaRecorder.isTypeSupported(t)) || '';
  }

  download(blob, filename = 'export.webm') {
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    a.click();
    URL.revokeObjectURL(url);
  }
}
```

## 常见误区

### 误区一：视频元素不会导致内存泄漏

每创建一个 `<video>` 元素并设置 `src`，浏览器都会分配解码器资源。如果不及时释放，会导致内存快速增长。

```javascript
function cleanupVideoElement(video) {
  video.pause();
  video.removeAttribute('src');
  video.load(); // 释放解码器资源
  video.remove();
}
```

### 误区二：可以用 setInterval 代替 requestAnimationFrame

`setInterval` 的时间精度不够，且在页面不可见时仍会执行，浪费资源。`requestAnimationFrame` 与浏览器刷新率同步，且在页面不可见时自动暂停。

### 误区三：导出时可以同时预览

导出时需要以最大速度渲染帧（不等待真实时间流逝），这会严重影响 UI 流畅度。建议导出时禁用预览，或使用独立的 Worker 线程。

## 工程建议

1. **使用 Web Worker 处理重计算**：FFmpeg 操作、滤镜计算等放到 Worker 线程
2. **及时释放视频资源**：不再使用的 `<video>` 元素调用 `load()` 释放解码器
3. **使用离屏 Canvas 缓存**：静态内容（如字幕背景、水印）缓存到离屏 Canvas
4. **实现分段导出**：长视频分段处理，避免单次操作耗时过长
5. **IndexedDB 存储项目状态**：自动保存编辑状态，防止意外丢失

## 参考答案

### 练习一

**思路**：使用命令模式记录每个操作的逆操作，或使用状态快照保存完整的时间线状态。命令模式更省内存但实现复杂，状态快照简单但内存占用大。这里使用状态快照 + 限制步数的方式。

**答案**：

```javascript
class UndoRedoManager {
  constructor(maxSteps = 50) {
    this.undoStack = [];
    this.redoStack = [];
    this.maxSteps = maxSteps;
  }

  saveState(state) {
    this.undoStack.push(JSON.stringify(state));
    if (this.undoStack.length > this.maxSteps) {
      this.undoStack.shift();
    }
    this.redoStack = [];
  }

  undo(currentState) {
    if (this.undoStack.length === 0) return null;
    this.redoStack.push(JSON.stringify(currentState));
    return JSON.parse(this.undoStack.pop());
  }

  redo(currentState) {
    if (this.redoStack.length === 0) return null;
    this.undoStack.push(JSON.stringify(currentState));
    return JSON.parse(this.redoStack.pop());
  }

  canUndo() {
    return this.undoStack.length > 0;
  }

  canRedo() {
    return this.redoStack.length > 0;
  }

  clear() {
    this.undoStack = [];
    this.redoStack = [];
  }
}

class EditorWithUndo {
  constructor(canvas) {
    this.editor = new VideoEditor(canvas);
    this.undoManager = new UndoRedoManager(50);
    this.bindShortcuts();
  }

  saveState() {
    const state = {
      tracks: this.editor.timeline.tracks.map(track => ({
        id: track.id,
        type: track.type,
        clips: track.clips.map(clip => ({ ...clip })),
      })),
      duration: this.editor.timeline.duration,
    };
    this.undoManager.saveState(state);
  }

  undo() {
    const currentState = {
      tracks: this.editor.timeline.tracks.map(track => ({
        id: track.id,
        type: track.type,
        clips: track.clips.map(clip => ({ ...clip })),
      })),
      duration: this.editor.timeline.duration,
    };

    const prevState = this.undoManager.undo(currentState);
    if (prevState) {
      this.restoreState(prevState);
    }
  }

  redo() {
    const currentState = {
      tracks: this.editor.timeline.tracks.map(track => ({
        id: track.id,
        type: track.type,
        clips: track.clips.map(clip => ({ ...clip })),
      })),
      duration: this.editor.timeline.duration,
    };

    const nextState = this.undoManager.redo(currentState);
    if (nextState) {
      this.restoreState(nextState);
    }
  }

  restoreState(state) {
    this.editor.timeline.tracks = state.tracks;
    this.editor.timeline.duration = state.duration;
    // 重新绑定视频元素引用
    for (const track of this.editor.timeline.tracks) {
      for (const clip of track.clips) {
        clip._videoElement = this.editor.assetManager.getVideoElement(clip.assetId);
      }
    }
    this.editor.timelineUI.render();
  }

  bindShortcuts() {
    document.addEventListener('keydown', (e) => {
      if (e.ctrlKey || e.metaKey) {
        if (e.key === 'z' && !e.shiftKey) {
          e.preventDefault();
          this.undo();
        } else if ((e.key === 'z' && e.shiftKey) || e.key === 'y') {
          e.preventDefault();
          this.redo();
        }
      }
    });
  }
}
```

**要点**：
- 使用 `JSON.stringify` 序列化状态快照，简单可靠
- 限制最大撤销步数（50 步），避免内存无限增长
- 恢复状态时需要重新绑定视频元素引用
- 快捷键需要区分 `Ctrl+Z`（撤销）和 `Ctrl+Shift+Z`（重做）

### 练习二

**思路**：扩展 Timeline 的 track 结构，支持多个同类型轨道。Canvas 渲染时按轨道顺序叠加绘制，实现画中画效果。

**答案**：

```javascript
class MultiTrackTimeline extends Timeline {
  constructor() {
    super();
    this.trackCounter = 0;
  }

  addVideoTrack(name) {
    this.trackCounter++;
    const track = {
      id: `video_track_${this.trackCounter}`,
      type: 'video',
      name: name || `视频 ${this.trackCounter}`,
      clips: [],
      muted: false,
      volume: 1.0,
      visible: true,
      opacity: 1.0,
      zIndex: this.trackCounter,
    };
    this.tracks.push(track);
    return track;
  }

  addAudioTrack(name) {
    this.trackCounter++;
    const track = {
      id: `audio_track_${this.trackCounter}`,
      type: 'audio',
      name: name || `音频 ${this.trackCounter}`,
      clips: [],
      muted: false,
      volume: 1.0,
    };
    this.tracks.push(track);
    return track;
  }

  removeTrack(trackId) {
    this.tracks = this.tracks.filter(t => t.id !== trackId);
    this.updateDuration();
  }

  getVideoTracks() {
    return this.tracks.filter(t => t.type === 'video');
  }

  getAudioTracks() {
    return this.tracks.filter(t => t.type === 'audio');
  }
}

class MultiTrackRenderer {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
  }

  render(timeline, currentTime) {
    const ctx = this.ctx;
    ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    // 按 zIndex 排序渲染视频轨道
    const videoTracks = timeline.getVideoTracks()
      .sort((a, b) => a.zIndex - b.zIndex);

    for (const track of videoTracks) {
      if (!track.visible) continue;

      const clip = track.clips.find(
        c => currentTime >= c.startTime && currentTime < c.startTime + c.duration
      );

      if (clip && clip._videoElement) {
        ctx.save();
        ctx.globalAlpha = track.opacity;

        if (clip.transform) {
          // 支持缩放和位移
          const { x, y, width, height } = clip.transform;
          ctx.drawImage(clip._videoElement, x, y, width, height);
        } else {
          ctx.drawImage(clip._videoElement, 0, 0, this.canvas.width, this.canvas.height);
        }

        ctx.restore();
      }
    }
  }
}

class AudioTrackMixer {
  constructor() {
    this.audioContext = new AudioContext();
    this.masterGain = this.audioContext.createGain();
    this.masterGain.connect(this.audioContext.destination);
    this.trackGains = new Map();
  }

  addTrack(trackId) {
    const gainNode = this.audioContext.createGain();
    gainNode.connect(this.masterGain);
    this.trackGains.set(trackId, gainNode);
    return gainNode;
  }

  setTrackVolume(trackId, volume) {
    const gain = this.trackGains.get(trackId);
    if (gain) {
      gain.gain.setValueAtTime(volume, this.audioContext.currentTime);
    }
  }

  muteTrack(trackId) {
    this.setTrackVolume(trackId, 0);
  }

  unmuteTrack(trackId, volume = 1) {
    this.setTrackVolume(trackId, volume);
  }

  connectAudioElement(trackId, audioElement) {
    const gain = this.trackGains.get(trackId);
    if (!gain) return;

    const source = this.audioContext.createMediaElementSource(audioElement);
    source.connect(gain);
  }
}

// 使用示例
const timeline = new MultiTrackTimeline();
const videoTrack1 = timeline.addVideoTrack('主视频');
const videoTrack2 = timeline.addVideoTrack('画中画');
const audioTrack = timeline.addAudioTrack('背景音乐');

// 画中画片段设置变换参数
timeline.addClip(videoTrack2.id, {
  sourceUrl: 'pip.mp4',
  startTime: 5,
  duration: 10,
  transform: {
    x: timeline.duration * 0.6,
    y: 20,
    width: 320,
    height: 180,
  },
});
```

**要点**：
- 多轨道通过 `zIndex` 控制渲染顺序，实现图层叠加
- 每个轨道独立控制可见性、透明度和音量
- 画中画通过 `transform` 参数设置位置和大小
- 音频混合使用 Web Audio API 的 GainNode 控制每轨音量
