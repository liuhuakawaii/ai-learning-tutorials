# 第四课：视频合成与剪辑

## 场景引入

想象你在用浏览器做一个简单的视频编辑器：用户导入几段视频，拖拽排列到时间线上，添加过渡效果，再配上背景音乐，最后导出为一个完整的视频文件。这就是视频合成与剪辑的核心场景。

这个需求看似简单，但涉及多个技术难点：多路视频如何同步播放、时间线如何建模、过渡效果如何实现、音视频如何混合、最终如何录制导出。本课将逐一拆解这些问题。

## 学习目标

1. 理解时间线模型的设计：轨道、片段、过渡
2. 掌握 Canvas 合成多路视频源的方法
3. 学会实现常见的过渡效果（交叉淡化、擦除、溶解）
4. 理解 Web Audio API 的音频混合能力
5. 掌握使用 MediaRecorder + Canvas Stream 导出视频的方案

## 时间线模型

视频编辑器的核心数据结构是时间线（Timeline）。时间线由多个轨道（Track）组成，每个轨道包含多个片段（Clip），片段之间可以有过渡效果（Transition）。

### 数据结构设计

```javascript
class Timeline {
  constructor() {
    this.tracks = [];
    this.duration = 0;      // 时间线总时长（秒）
    this.currentTime = 0;   // 当前播放位置
  }

  addTrack(type = 'video') {
    const track = {
      id: this.generateId(),
      type,                  // 'video' | 'audio'
      clips: [],
      muted: false,
      volume: 1.0,
    };
    this.tracks.push(track);
    return track;
  }

  addClip(trackId, clipData) {
    const track = this.tracks.find(t => t.id === trackId);
    if (!track) return null;

    const clip = {
      id: this.generateId(),
      sourceUrl: clipData.sourceUrl,
      startTime: clipData.startTime,     // 在时间线上的起始时间
      duration: clipData.duration,        // 片段时长
      trimStart: clipData.trimStart || 0, // 源视频的裁剪起点
      trimEnd: clipData.trimEnd || 0,     // 源视频的裁剪终点
      filters: [],                        // 应用的滤镜列表
      volume: 1.0,
      speed: 1.0,
    };

    track.clips.push(clip);
    track.clips.sort((a, b) => a.startTime - b.startTime);
    this.updateDuration();
    return clip;
  }

  updateDuration() {
    this.duration = this.tracks.reduce((max, track) => {
      const trackEnd = track.clips.reduce((end, clip) => {
        return Math.max(end, clip.startTime + clip.duration);
      }, 0);
      return Math.max(max, trackEnd);
    }, 0);
  }

  generateId() {
    return `clip_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
  }
}
```

### 片段排列与排序

```javascript
function getClipsAtTime(track, time) {
  return track.clips.filter(clip => {
    return time >= clip.startTime && time < clip.startTime + clip.duration;
  });
}

function getActiveClipAtTime(track, time) {
  // 返回在指定时间点应该播放的片段（取最后一个覆盖该时间的片段）
  const candidates = getClipsAtTime(track, time);
  return candidates.length > 0 ? candidates[candidates.length - 1] : null;
}
```

## Canvas 合成多路视频

使用 Canvas 的 `drawImage` 方法可以将多个视频源绘制到同一个画布上，实现图层叠加合成。

### 基础合成

```javascript
class VideoCompositor {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
    this.layers = [];
  }

  addLayer(videoElement, options = {}) {
    const layer = {
      video: videoElement,
      x: options.x || 0,
      y: options.y || 0,
      width: options.width || this.canvas.width,
      height: options.height || this.canvas.height,
      opacity: options.opacity ?? 1.0,
      blendMode: options.blendMode || 'source-over',
    };
    this.layers.push(layer);
    return layer;
  }

  removeLayer(index) {
    this.layers.splice(index, 1);
  }

  render() {
    const ctx = this.ctx;
    ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    for (const layer of this.layers) {
      if (layer.video.paused || layer.video.ended) continue;

      ctx.save();
      ctx.globalAlpha = layer.opacity;
      ctx.globalCompositeOperation = layer.blendMode;
      ctx.drawImage(
        layer.video,
        layer.x, layer.y,
        layer.width, layer.height
      );
      ctx.restore();
    }
  }
}
```

## 动画过渡效果

过渡效果是两个片段衔接时的视觉动画。常见的过渡类型包括交叉淡化、擦除和溶解。

### 交叉淡化（Crossfade）

```javascript
function crossfade(ctx, fromVideo, toVideo, progress, width, height) {
  // progress: 0.0 → 1.0

  // 淡出当前片段
  ctx.save();
  ctx.globalAlpha = 1 - progress;
  ctx.drawImage(fromVideo, 0, 0, width, height);
  ctx.restore();

  // 淡入下一个片段
  ctx.save();
  ctx.globalAlpha = progress;
  ctx.drawImage(toVideo, 0, 0, width, height);
  ctx.restore();
}
```

### 擦除（Wipe）

```javascript
function wipe(ctx, fromVideo, toVideo, progress, width, height) {
  // 从左到右擦除
  const wipeX = width * progress;

  // 绘制当前片段（左侧保留部分）
  ctx.save();
  ctx.beginPath();
  ctx.rect(0, 0, wipeX, height);
  ctx.clip();
  ctx.drawImage(fromVideo, 0, 0, width, height);
  ctx.restore();

  // 绘制下一个片段（右侧新部分）
  ctx.save();
  ctx.beginPath();
  ctx.rect(wipeX, 0, width - wipeX, height);
  ctx.clip();
  ctx.drawImage(toVideo, 0, 0, width, height);
  ctx.restore();

  // 擦除线
  ctx.strokeStyle = '#ffffff';
  ctx.lineWidth = 3;
  ctx.beginPath();
  ctx.moveTo(wipeX, 0);
  ctx.lineTo(wipeX, height);
  ctx.stroke();
}
```

### 溶解（Dissolve）

```javascript
function dissolve(ctx, fromVideo, toVideo, progress, width, height) {
  // 使用随机像素遮罩实现溶解效果
  ctx.drawImage(fromVideo, 0, 0, width, height);
  const fromData = ctx.getImageData(0, 0, width, height);

  ctx.drawImage(toVideo, 0, 0, width, height);
  const toData = ctx.getImageData(0, 0, width, height);

  const result = ctx.createImageData(width, height);
  const threshold = progress * 255;

  for (let i = 0; i < fromData.data.length; i += 4) {
    // 使用简单的阈值判断
    const noise = Math.random() * 255;
    if (noise < threshold) {
      result.data[i] = toData.data[i];
      result.data[i + 1] = toData.data[i + 1];
      result.data[i + 2] = toData.data[i + 2];
      result.data[i + 3] = toData.data[i + 3];
    } else {
      result.data[i] = fromData.data[i];
      result.data[i + 1] = fromData.data[i + 1];
      result.data[i + 2] = fromData.data[i + 2];
      result.data[i + 3] = fromData.data[i + 3];
    }
  }

  ctx.putImageData(result, 0, 0);
}
```

## 音频混合

使用 Web Audio API 可以将多路音频混合在一起，并控制每路的音量和声道。

```javascript
class AudioMixer {
  constructor() {
    this.audioContext = new AudioContext();
    this.masterGain = this.audioContext.createGain();
    this.masterGain.connect(this.audioContext.destination);
    this.tracks = new Map();
  }

  addTrack(trackId, audioElement) {
    const source = this.audioContext.createMediaElementSource(audioElement);
    const gainNode = this.audioContext.createGain();
    const panNode = this.audioContext.createStereoPanner();

    source.connect(gainNode);
    gainNode.connect(panNode);
    panNode.connect(this.masterGain);

    this.tracks.set(trackId, {
      source,
      gainNode,
      panNode,
      audioElement,
    });
  }

  setVolume(trackId, volume) {
    const track = this.tracks.get(trackId);
    if (track) {
      track.gainNode.gain.setValueAtTime(volume, this.audioContext.currentTime);
    }
  }

  setPan(trackId, pan) {
    // pan: -1.0（左）到 1.0（右）
    const track = this.tracks.get(trackId);
    if (track) {
      track.panNode.pan.setValueAtTime(pan, this.audioContext.currentTime);
    }
  }

  setMasterVolume(volume) {
    this.masterGain.gain.setValueAtTime(volume, this.audioContext.currentTime);
  }

  // 添加淡入淡出效果
  fadeTrack(trackId, fromVolume, toVolume, duration) {
    const track = this.tracks.get(trackId);
    if (!track) return;

    const now = this.audioContext.currentTime;
    track.gainNode.gain.setValueAtTime(fromVolume, now);
    track.gainNode.gain.linearRampToValueAtTime(toVolume, now + duration);
  }
}
```

## 导出方案：MediaRecorder + Canvas Stream

使用 `canvas.captureStream()` 获取 Canvas 的视频流，配合 MediaRecorder API 录制并导出为视频文件。

```javascript
class VideoExporter {
  constructor(canvas, audioContext) {
    this.canvas = canvas;
    this.audioContext = audioContext;
    this.chunks = [];
    this.recorder = null;
  }

  setupRecorder(options = {}) {
    const canvasStream = this.canvas.captureStream(options.fps || 30);

    // 如果有音频上下文，混合音频轨道
    if (this.audioContext) {
      const audioDestination = this.audioContext.createMediaStreamDestination();
      // 需要将音频连接到这个 destination
      const audioStream = audioDestination.stream;
      audioStream.getAudioTracks().forEach(track => {
        canvasStream.addTrack(track);
      });
    }

    const mimeType = this.getSupportedMimeType();
    this.recorder = new MediaRecorder(canvasStream, {
      mimeType,
      videoBitsPerSecond: options.bitrate || 5000000,
    });

    this.recorder.ondataavailable = (e) => {
      if (e.data.size > 0) {
        this.chunks.push(e.data);
      }
    };

    return this.recorder;
  }

  getSupportedMimeType() {
    const types = [
      'video/webm;codecs=vp9,opus',
      'video/webm;codecs=vp8,opus',
      'video/webm',
      'video/mp4',
    ];
    return types.find(type => MediaRecorder.isTypeSupported(type)) || '';
  }

  start() {
    this.chunks = [];
    this.recorder.start(100); // 每 100ms 生成一个数据块
  }

  stop() {
    return new Promise((resolve) => {
      this.recorder.onstop = () => {
        const blob = new Blob(this.chunks, { type: this.recorder.mimeType });
        resolve(blob);
      };
      this.recorder.stop();
    });
  }

  download(blob, filename = 'export.webm') {
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    a.click();
    URL.revokeObjectURL(url);
  }
}
```

### 完整的渲染-导出流程

```javascript
async function exportTimeline(timeline, compositor, exporter, fps = 30) {
  exporter.setupRecorder({ fps });
  exporter.start();

  const frameDuration = 1000 / fps;
  const totalFrames = Math.ceil(timeline.duration * fps);

  for (let frame = 0; frame < totalFrames; frame++) {
    const currentTime = frame / fps;

    // 更新每个轨道的视频播放位置
    for (const track of timeline.tracks) {
      if (track.type !== 'video') continue;

      const activeClip = getActiveClipAtTime(track, currentTime);
      if (activeClip) {
        const clipTime = currentTime - activeClip.startTime + activeClip.trimStart;
        // 设置视频的播放时间（关键帧定位）
        // 注意：实际实现中需要预加载所有视频元素
      }
    }

    // 渲染当前帧
    compositor.render();

    // 等待下一帧
    await new Promise(resolve => setTimeout(resolve, frameDuration));
  }

  const blob = await exporter.stop();
  exporter.download(blob, 'timeline-export.webm');
  return blob;
}
```

## 常见误区

### 误区一：音视频同步很简单

音视频同步是视频编辑中最棘手的问题之一。主要原因：

1. `requestAnimationFrame` 的时间精度有限（约 16ms）
2. `<video>` 元素的 `currentTime` 更新不精确
3. Canvas 渲染和音频播放之间存在延迟差异

解决方案是使用 `requestVideoFrameCallback`（见工程建议）。

### 误区二：可以用一个 Canvas 同时预览和导出

预览时需要降低分辨率以保证流畅度，导出时需要原始分辨率。建议使用两个独立的 Canvas，或者在导出时动态调整分辨率。

### 误区三：MediaRecorder 支持所有格式

MediaRecorder 的格式支持取决于浏览器。Chrome 支持 WebM（VP8/VP9），Safari 支持 MP4。需要检测支持的 MIME 类型并做降级处理。

---

## 工程建议

1. **使用 `requestVideoFrameCallback`**：这个 API 提供精确到帧的回调时机，比 `requestAnimationFrame` 更适合视频同步

```javascript
function renderLoop(video, callback) {
  const loop = () => {
    video.requestVideoFrameCallback((now, metadata) => {
      callback(metadata);
      loop();
    });
  };
  loop();
}
```

2. **预加载所有视频元素**：在开始渲染前，确保所有视频都已加载完成，避免渲染时出现黑帧
3. **使用离屏 Canvas 缓存**：将不需要每帧更新的内容（如背景图、文字标题）缓存到离屏 Canvas
4. **导出时使用 Web Worker**：将 Canvas 渲染和编码放到 Worker 线程，避免阻塞主线程
5. **分段导出大文件**：对于长视频，分段录制并拼接，避免内存溢出

## 小结

本课讲解了视频合成与剪辑的核心技术：

- 时间线模型由轨道、片段和过渡组成，是编辑器的数据基础

- 过渡效果通过在两个片段之间插值实现，常见类型有交叉淡化、擦除和溶解

## 参考答案

### 练习一

**思路**：使用 Timeline 管理片段数据，Canvas 渲染合成，MediaRecorder 导出。过渡期间同时绘制两个视频并调节透明度。

**答案**：

```javascript
class SimpleVideoMerger {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
    this.timeline = new Timeline();
    this.videoTrack = this.timeline.addTrack('video');
    this.transitionDuration = 1.0; // 默认过渡 1 秒
  }

  async loadVideos(file1, file2) {
    const video1 = await this.loadVideo(file1);
    const video2 = await this.loadVideo(file2);

    const clip1Duration = video1.duration;
    const clip2Duration = video2.duration;

    this.timeline.addClip(this.videoTrack.id, {
      sourceUrl: URL.createObjectURL(file1),
      startTime: 0,
      duration: clip1Duration,
    });

    // 第二个片段的起始时间 = 第一个片段时长 - 过渡时长
    this.timeline.addClip(this.videoTrack.id, {
      sourceUrl: URL.createObjectURL(file2),
      startTime: clip1Duration - this.transitionDuration,
      duration: clip2Duration,
    });

    return { video1, video2 };
  }

  loadVideo(file) {
    return new Promise((resolve) => {
      const video = document.createElement('video');
      video.src = URL.createObjectURL(file);
      video.muted = true;
      video.addEventListener('loadedmetadata', () => resolve(video));
    });
  }

  renderFrame(video1, video2, currentTime) {
    const { width, height } = this.canvas;
    const clip1End = video1.duration;
    const transitionStart = clip1End - this.transitionDuration;

    if (currentTime < transitionStart) {
      // 只有第一个视频
      this.ctx.drawImage(video1, 0, 0, width, height);
    } else if (currentTime < clip1End) {
      // 过渡阶段：交叉淡化
      const progress = (currentTime - transitionStart) / this.transitionDuration;
      this.ctx.globalAlpha = 1 - progress;
      this.ctx.drawImage(video1, 0, 0, width, height);
      this.ctx.globalAlpha = progress;
      this.ctx.drawImage(video2, 0, 0, width, height);
      this.ctx.globalAlpha = 1;
    } else {
      // 只有第二个视频
      this.ctx.drawImage(video2, 0, 0, width, height);
    }
  }

  async exportVideo(video1, video2) {
    const stream = this.canvas.captureStream(30);
    const recorder = new MediaRecorder(stream, {
      mimeType: 'video/webm;codecs=vp9',
      videoBitsPerSecond: 5000000,
    });

    const chunks = [];
    recorder.ondataavailable = (e) => chunks.push(e.data);

    return new Promise((resolve) => {
      recorder.onstop = () => {
        resolve(new Blob(chunks, { type: 'video/webm' }));
      };

      recorder.start(100);

      // 同步播放两个视频
      video1.currentTime = 0;
      video2.currentTime = 0;
      video1.play();

      const totalDuration = this.timeline.duration;
      const startTime = performance.now();

      const renderLoop = () => {
        const elapsed = (performance.now() - startTime) / 1000;
        if (elapsed >= totalDuration) {
          recorder.stop();
          video1.pause();
          video2.pause();
          return;
        }

        // 更新视频播放位置
        if (elapsed < video1.duration) {
          video1.currentTime = elapsed;
        }
        if (elapsed >= video1.duration - this.transitionDuration) {
          const clip2Time = elapsed - (video1.duration - this.transitionDuration);
          video2.currentTime = clip2Time;
          if (video2.paused) video2.play();
        }

        this.renderFrame(video1, video2, elapsed);
        requestAnimationFrame(renderLoop);
      };

      renderLoop();
    });
  }
}
```

**要点**：
- 第二个片段的起始时间减去过渡时长，实现重叠
- 过渡阶段同时绘制两个视频，通过 `globalAlpha` 控制透明度
- 导出时需要同步更新视频的播放位置

### 练习二

**思路**：将过渡效果封装为独立函数，通过配置选择不同的过渡类型。

**答案**：

```javascript
const TRANSITIONS = {
  crossfade(ctx, fromVideo, toVideo, progress, width, height) {
    ctx.globalAlpha = 1 - progress;
    ctx.drawImage(fromVideo, 0, 0, width, height);
    ctx.globalAlpha = progress;
    ctx.drawImage(toVideo, 0, 0, width, height);
    ctx.globalAlpha = 1;
  },

  wipe(ctx, fromVideo, toVideo, progress, width, height) {
    const wipeX = width * progress;

    ctx.save();
    ctx.beginPath();
    ctx.rect(0, 0, wipeX, height);
    ctx.clip();
    ctx.drawImage(fromVideo, 0, 0, width, height);
    ctx.restore();

    ctx.save();
    ctx.beginPath();
    ctx.rect(wipeX, 0, width - wipeX, height);
    ctx.clip();
    ctx.drawImage(toVideo, 0, 0, width, height);
    ctx.restore();
  },

  wipeDown(ctx, fromVideo, toVideo, progress, width, height) {
    const wipeY = height * progress;

    ctx.save();
    ctx.beginPath();
    ctx.rect(0, 0, width, wipeY);
    ctx.clip();
    ctx.drawImage(fromVideo, 0, 0, width, height);
    ctx.restore();

    ctx.save();
    ctx.beginPath();
    ctx.rect(0, wipeY, width, height - wipeY);
    ctx.clip();
    ctx.drawImage(toVideo, 0, 0, width, height);
    ctx.restore();
  },
};

class TransitionRenderer {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
    this.transitionType = 'crossfade';
    this.transitionDuration = 1.0;
  }

  setTransition(type, duration) {
    if (!TRANSITIONS[type]) {
      throw new Error(`未知过渡类型: ${type}`);
    }
    this.transitionType = type;
    this.transitionDuration = duration;
  }

  render(fromVideo, toVideo, progress) {
    const { width, height } = this.canvas;
    const transitionFn = TRANSITIONS[this.transitionType];
    transitionFn(this.ctx, fromVideo, toVideo, progress, width, height);
  }
}

// 使用示例
const renderer = new TransitionRenderer(canvas);
renderer.setTransition('wipe', 1.5);

// 在过渡期间每帧调用
function onTransitionFrame(fromVideo, toVideo, progress) {
  renderer.render(fromVideo, toVideo, progress);
}
```

- `setTransition` 方法支持动态切换过渡类型和时长
- Wipe 方向可以通过修改裁剪区域实现（水平/垂直/对角线）
