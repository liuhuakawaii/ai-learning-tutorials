# 第四课：版权保护

## 场景引入

Netflix 在全球拥有超过 2 亿付费用户，每年投入数百亿美元购买和制作内容。如果这些内容被盗版传播，Netflix 的商业模式将瞬间崩塌。因此，Netflix 采用了多层版权保护体系：内容在传输和存储时全程加密，播放时通过 DRM 系统解密，每个用户的播放会话都有唯一的密钥，甚至在视频画面中嵌入不可见的数字水印，可以追溯泄露源头。

版权保护不是"加个密码"那么简单。它是一个涉及加密、密钥管理、硬件安全、法律合规的系统工程。本课将深入讲解 Web 端版权保护的核心技术。

## 学习目标

完成本课学习后，你将能够：

1. 理解 DRM 系统的工作原理（Widevine、FairPlay、PlayReady）
2. 使用 EME API 实现加密视频播放
3. 设计加密方案（AES-128 加密、密钥轮换）
4. 了解数字水印技术和防盗录方案
5. 在保护版权和用户体验之间找到平衡

```javascript
async function detectDRMSupport() {
  const configs = [
    { name: 'Widevine', keySystem: 'com.widevine.alpha' },
    { name: 'FairPlay', keySystem: 'com.apple.fps.1_0' },
    { name: 'PlayReady', keySystem: 'com.microsoft.playready' },
  ];

  const results = [];

  for (const config of configs) {
    try {
      const access = await navigator.requestMediaKeySystemAccess(
        config.keySystem,
        [{
          initDataTypes: ['cenc'],
          videoCapabilities: [{ contentType: 'video/mp4; codecs="avc1.42E01E"' }],
          audioCapabilities: [{ contentType: 'audio/mp4; codecs="mp4a.40.2"' }]
        }]
      );

      results.push({
        name: config.name,
        keySystem: config.keySystem,
        supported: true,
        configuration: access.getConfiguration()
      });
    } catch (e) {
      results.push({
        name: config.name,
        keySystem: config.keySystem,
        supported: false
      });
    }
  }

  return results;
```

## EME API

### Encrypted Media Extensions

EME 是 W3C 标准，提供了 Web 端与 DRM 系统交互的统一接口：

```javascript
// EME 播放流程
class DRMPlayer {
  constructor(videoElement) {
    this.video = videoElement;
    this.keySession = null;
    this.mediaKeys = null;
  }

  async initialize(keySystem = 'com.widevine.alpha') {
    // 第一步：请求媒体密钥系统访问权限
    const config = [{
      initDataTypes: ['cenc', 'keyids'],
      videoCapabilities: [
        { contentType: 'video/mp4; codecs="avc1.42E01E"' },
        { contentType: 'video/mp4; codecs="hev1.1.6.L93.B0"' }
      ],
      audioCapabilities: [
        { contentType: 'audio/mp4; codecs="mp4a.40.2"' }
      ]
    }];

    try {
      const keySystemAccess = await navigator.requestMediaKeySystemAccess(
        keySystem, config
      );

      // 第二步：创建媒体密钥对象
      this.mediaKeys = await keySystemAccess.createMediaKeys();

      // 第三步：将媒体密钥绑定到 video 元素
      await this.video.setMediaKeys(this.mediaKeys);

      console.log('DRM 初始化完成');
      return true;
    } catch (error) {
      console.error('DRM 初始化失败:', error);
      return false;
    }
  }

  // 处理加密事件
  async handleEncryptedEvent(event) {
    console.log('收到加密事件:', event.initDataType);

    // 创建密钥会话
    this.keySession = this.mediaKeys.createSession();

    // 监听密钥消息
    this.keySession.addEventListener('message', async (e) => {
      await this.handleKeyMessage(e);
    });

    // 监听密钥状态变化
    this.keySession.addEventListener('keystatuseschange', (e) => {
      this.handleKeyStatusChange(e);
    });

    // 生成许可证请求
    await this.keySession.generateRequest(
      event.initDataType,
      event initData
    );
  }

  // 处理密钥消息（请求许可证）
  async handleKeyMessage(event) {
    const licenseServerUrl = 'https://license.example.com/drm/widevine';

    try {
      const response = await fetch(licenseServerUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/octet-stream' },
        body: event.message
      });

      const license = await response.arrayBuffer();

      // 更新密钥会话（传入许可证响应）
      await this.keySession.update(license);
      console.log('许可证更新成功');
    } catch (error) {
      console.error('许可证请求失败:', error);
    }
  }

  // 处理密钥状态变化
  handleKeyStatusChange(event) {
    this.keySession.keyStatuses.forEach((status, keyId) => {
      switch (status) {
        case 'usable':
          console.log('密钥可用，可以播放');
          break;
        case 'expired':
          console.warn('密钥已过期');
          this.handleExpiredKey();
          break;
        case 'output-restricted':
          console.warn('输出受限（可能连接了录制设备）');
          break;
        case 'internal-error':
          console.error('内部错误');
          break;
      }
    });
  }

  handleExpiredKey() {
    // 重新请求许可证
    if (this.keySession) {
      this.keySession.close();
      this.keySession = null;
    }
    // 触发重新初始化
    this.video.load();
  }
}
```

```javascript
// 解析 PSSH（Protection System Specific Header）数据
function parsePSSH(initData) {
  const view = new DataView(initData);
  const boxes = [];
  let offset = 0;

  while (offset < initData.byteLength) {
    const boxSize = view.getUint32(offset);
    const boxType = String.fromCharCode(
      view.getUint8(offset + 4),
      view.getUint8(offset + 5),
      view.getUint8(offset + 6),
      view.getUint8(offset + 7)
    );

    if (boxType === 'pssh') {
      const version = view.getUint8(offset + 8);
      const systemId = initData.slice(offset + 12, offset + 28);

      boxes.push({
        type: boxType,
        version,
        systemId: Array.from(new Uint8Array(systemId))
          .map(b => b.toString(16).padStart(2, '0'))
          .join(''),
        size: boxSize
      });
    }

    offset += boxSize;
  }

  return boxes;
}
```

### AES-128 加密

```javascript
// AES-128-CBC 加密示例
async function encryptSegment(segmentData, key, iv) {
  const cryptoKey = await crypto.subtle.importKey(
    'raw',
    key,
    { name: 'AES-CBC' },
    false,
    ['encrypt']
  );

  const encrypted = await crypto.subtle.encrypt(
    { name: 'AES-CBC', iv },
    cryptoKey,
    segmentData
  );

  return new Uint8Array(encrypted);
}

// AES-128-CBC 解密
async function decryptSegment(encryptedData, key, iv) {
  const cryptoKey = await crypto.subtle.importKey(
    'raw',
    key,
    { name: 'AES-CBC' },
    false,
    ['decrypt']
  );

  const decrypted = await crypto.subtle.decrypt(
    { name: 'AES-CBC', iv },
    cryptoKey,
    encryptedData
  );

  return new Uint8Array(decrypted);
}

// 生成随机密钥和 IV
function generateEncryptionParams() {
  return {
    key: crypto.getRandomValues(new Uint8Array(16)),  // 128 位密钥
    iv: crypto.getRandomValues(new Uint8Array(16))     // 128 位 IV
  };
}
```

### 密钥轮换

密钥轮换（Key Rotation）是定期更换加密密钥的安全策略：

```javascript
// 密钥轮换管理器
class KeyRotationManager {
  constructor(config) {
    this.rotationInterval = config.rotationInterval || 300; // 默认 5 分钟
    this.keys = new Map();
    this.currentKeyId = 0;
    this.licenseServerUrl = config.licenseServerUrl;
  }

  // 获取当前密钥
  async getCurrentKey() {
    const keyId = `key_${this.currentKeyId}`;

    if (!this.keys.has(keyId)) {
      const keyData = await this.requestKey(keyId);
      this.keys.set(keyId, keyData);
    }

    return this.keys.get(keyId);
  }

  // 从许可证服务器获取密钥
  async requestKey(keyId) {
    const response = await fetch(`${this.licenseServerUrl}/key/${keyId}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        contentId: 'movie_123',
        keyId,
        timestamp: Date.now()
      })
    });

    const data = await response.json();

    return {
      id: keyId,
      key: new Uint8Array(data.key),
      iv: new Uint8Array(data.iv),
      expiresAt: Date.now() + this.rotationInterval * 1000
    };
  }

  // 检查是否需要轮换
  shouldRotate() {
    const currentKey = this.keys.get(`key_${this.currentKeyId}`);
    if (!currentKey) return true;
    return Date.now() >= currentKey.expiresAt;
  }

  // 执行密钥轮换
  async rotate() {
    this.currentKeyId++;
    console.log(`密钥轮换: key_${this.currentKeyId - 1} → key_${this.currentKeyId}`);
    return this.getCurrentKey();
  }

  // 为分片加密
  async encryptSegment(segmentData, segmentIndex) {
    if (this.shouldRotate()) {
      await this.rotate();
    }

    const keyData = await this.getCurrentKey();
    return encryptSegment(segmentData, keyData.key, keyData.iv);
  }
}
```

## 数字水印

### 不可见水印

数字水印在不影响观看体验的前提下，将标识信息嵌入视频中：

```javascript
// 基于 LSB（最低有效位）的简单水印
class LSBWatermark {
  constructor() {
    this.watermarkLength = 64; // 水印长度（比特）
  }

  // 嵌入水印
  embed(imageData, watermarkBits) {
    const data = new Uint8ClampedArray(imageData.data);

    for (let i = 0; i < watermarkBits.length; i++) {
      const pixelIndex = i * 4; // 每个像素的 R 通道
      const bit = parseInt(watermarkBits[i]);

      // 修改最低有效位
      data[pixelIndex] = (data[pixelIndex] & 0xFE) | bit;
    }

    return new ImageData(data, imageData.width, imageData.height);
  }

  // 提取水印
  extract(imageData, length) {
    const data = imageData.data;
    const bits = [];

    for (let i = 0; i < length; i++) {
      const pixelIndex = i * 4;
      bits.push(data[pixelIndex] & 1);
    }

    return bits.join('');
  }
}

// 基于频域的水印（更鲁棒）
class FrequencyWatermark {
  // 将水印嵌入 DCT 系数的中频区域
  // 这种方法对压缩、缩放等操作有更好的抵抗力

  embed(imageData, watermark) {
    // 将图像分块（8x8）
    const blocks = this.splitIntoBlocks(imageData);

    for (let i = 0; i < watermark.length && i < blocks.length; i++) {
      const block = blocks[i];
      // 对每个块做 DCT 变换
      const dct = this.dct2D(block);
      // 在中频系数中嵌入水印
      this.embedInDCT(dct, watermark[i]);
      // 逆 DCT 变换
      blocks[i] = this.idct2D(dct);
    }

    return this.mergeBlocks(blocks, imageData.width, imageData.height);
  }

  splitIntoBlocks(imageData) {
    // 简化：将图像分成 8x8 的块
    const blocks = [];
    const { width, height, data } = imageData;

    for (let y = 0; y < height; y += 8) {
      for (let x = 0; x < width; x += 8) {
        const block = new Float64Array(64);
        for (let by = 0; by < 8; by++) {
          for (let bx = 0; bx < 8; bx++) {
            const idx = ((y + by) * width + (x + bx)) * 4;
            block[by * 8 + bx] = data[idx]; // 灰度
          }
        }
        blocks.push(block);
      }
    }

    return blocks;
  }

  dct2D(block) {
    // 简化的 2D DCT
    const result = new Float64Array(64);
    for (let u = 0; u < 8; u++) {
      for (let v = 0; v < 8; v++) {
        let sum = 0;
        for (let x = 0; x < 8; x++) {
          for (let y = 0; y < 8; y++) {
            sum += block[x * 8 + y] *
              Math.cos((2 * x + 1) * u * Math.PI / 16) *
              Math.cos((2 * y + 1) * v * Math.PI / 16);
          }
        }
        result[u * 8 + v] = sum;
      }
    }
    return result;
  }

  idct2D(dct) {
    const result = new Float64Array(64);
    for (let x = 0; x < 8; x++) {
      for (let y = 0; y < 8; y++) {
        let sum = 0;
        for (let u = 0; u < 8; u++) {
          for (let v = 0; v < 8; v++) {
            const cu = u === 0 ? 1 / Math.SQRT2 : 1;
            const cv = v === 0 ? 1 / Math.SQRT2 : 1;
            sum += cu * cv * dct[u * 8 + v] *
              Math.cos((2 * x + 1) * u * Math.PI / 16) *
              Math.cos((2 * y + 1) * v * Math.PI / 16);
          }
        }
        result[x * 8 + y] = sum / 4;
      }
    }
    return result;
  }

  embedInDCT(dct, bit) {
    // 在中频系数 (4,3) 和 (3,4) 嵌入水印
    const threshold = 10;
    if (bit === '1') {
```

### 屏幕捕获检测

```javascript
// 检测屏幕录制行为
class ScreenCaptureDetector {
  constructor() {
    this.detectionMethods = [];
    this.init();
  }

  init() {
    // 方法 1：检测 Canvas 指纹污染
    this.checkCanvasFingerprint();

    // 方法 2：监测性能指标异常
    this.monitorPerformance();

    // 方法 3：检测屏幕共享 API
    this.detectScreenCapture();
  }

  // 检测 Canvas 是否被录制软件劫持
  checkCanvasFingerprint() {
    const canvas = document.createElement('canvas');
    const ctx = canvas.getContext('2d');

    // 绘制已知内容
    ctx.fillStyle = 'rgb(1,2,3)';
    ctx.fillRect(0, 0, 1, 1);

    const pixel = ctx.getImageData(0, 0, 1, 1).data;

    // 检查像素值是否被篡改
    if (pixel[0] !== 1 || pixel[1] !== 2 || pixel[2] !== 3) {
      console.warn('Canvas 指纹异常，可能存在屏幕录制');
      this.onDetection('canvas_pollution');
    }
  }

  // 监测性能异常
  monitorPerformance() {
    let frameCount = 0;
    let lastTime = performance.now();

    const check = () => {
      frameCount++;
      const now = performance.now();

      if (now - lastTime >= 1000) {
        const fps = frameCount;

        // 录制软件通常会导致帧率下降
        if (fps < 15 && fps > 0) {
          console.warn(`帧率异常低: ${fps}，可能存在屏幕录制`);
          this.onDetection('low_fps');
        }

        frameCount = 0;
        lastTime = now;
      }

      requestAnimationFrame(check);
    };

    requestAnimationFrame(check);
  }

  // 检测屏幕捕获 API
  detectScreenCapture() {
    // 检查是否有活跃的屏幕共享
    if (navigator.mediaDevices && navigator.mediaDevices.getDisplayMedia) {
      // 监听页面可见性变化
      document.addEventListener('visibilitychange', () => {
        if (document.hidden) {
          // 页面隐藏时可能是录制软件在工作
          this.suspiciousActivity++;
        }
      });
    }
  }

  onDetection(method) {
    console.warn(`检测到可疑的屏幕录制行为: ${method}`);
    // 可以选择：暂停播放、降低画质、显示警告
  }
}
```

## 常见误区

### 误区一：DRM 让视频不可能被盗

DRM 只是提高盗版成本，不能完全阻止。任何在用户设备上播放的内容，理论上都可以通过模拟器、调试工具或硬件捕获卡录制。DRM 的真正作用是让普通用户无法轻易获取原始内容，同时为法律追责提供技术依据。

### 误区二：客户端保护足够安全

禁用右键、添加水印叠加层等客户端保护手段可以被轻易绕过。真正的安全依赖服务端加密和 DRM 硬件安全。客户端保护只是"防君子不防小人"的第一道防线。

### 误区三：加密会影响播放性能

现代设备的硬件解密能力非常强，AES-128 解密的性能开销几乎可以忽略不计。DRM 解密通常在硬件安全区域（TEE）中完成，不会影响主处理器的性能。真正影响性能的是过度的客户端保护措施。

### 误区四：一个 DRM 系统就够了

不同平台支持不同的 DRM 系统。要覆盖所有主流浏览器和设备，至少需要 Widevine（Chrome/Android）、FairPlay（Apple）和 PlayReady（Windows/Edge）三套系统。单一 DRM 方案会丢失大量用户。

## 小结

本课深入讲解了 Web 端版权保护的核心技术：

- **DRM 系统**：Widevine、FairPlay、PlayReady 三大系统覆盖不同平台
- **EME API**：W3C 标准的加密媒体扩展，提供统一的 DRM 交互接口
- **加密方案**：AES-128 加密结合密钥轮换，保护内容安全
- **数字水印**：不可见的水印技术，用于追溯泄露源头
- **防盗录**：多层防护措施，提高盗版成本

版权保护的核心原则是**纵深防御**——不依赖单一技术，而是通过多层防护提高攻击成本。同时需要在安全性和用户体验之间找到平衡，过度保护会损害合法用户的体验。

### 练习一

**思路**：使用 `navigator.requestMediaKeySystemAccess` 获取 DRM 访问权限，监听 `encrypted` 事件，通过许可证服务器获取密钥。

**答案**：

```javascript
class WidevinePlayer {
  constructor(videoElement, licenseServerUrl) {
    this.video = videoElement;
    this.licenseServerUrl = licenseServerUrl;
    this.mediaKeys = null;
    this.session = null;
  }

  async init() {
    const config = [{
      initDataTypes: ['cenc', 'keyids'],
      videoCapabilities: [
        { contentType: 'video/mp4; codecs="avc1.42E01E"' }
      ],
      audioCapabilities: [
        { contentType: 'audio/mp4; codecs="mp4a.40.2"' }
      ]
    }];

    try {
      const access = await navigator.requestMediaKeySystemAccess(
        'com.widevine.alpha', config
      );
      this.mediaKeys = await access.createMediaKeys();
      await this.video.setMediaKeys(this.mediaKeys);
      this.video.addEventListener('encrypted', (e) => this.onEncrypted(e));
      console.log('Widevine 初始化成功');
    } catch (err) {
      console.error('Widevine 初始化失败:', err);
      throw err;
    }
  }

  async onEncrypted(event) {
    this.session = this.mediaKeys.createSession();
    this.session.addEventListener('message', (e) => this.onMessage(e));
    this.session.addEventListener('keystatuseschange', () => this.onKeyStatusChange());
    await this.session.generateRequest(event.initDataType, event.initData);
  }

  async onMessage(event) {
    const resp = await fetch(this.licenseServerUrl, {
      method: 'POST',
      body: event.message
    });
    await this.session.update(await resp.arrayBuffer());
  }

  onKeyStatusChange() {
    for (const [keyId, status] of this.session.keyStatuses) {
      if (status === 'usable') {
        console.log('密钥就绪，可以播放');
        this.video.play();
      } else if (status === 'expired') {
        console.warn('密钥过期');
      }
    }
  }
}

// 使用示例
const player = new WidevinePlayer(
  document.getElementById('video'),
  'https://license.example.com/widevine'
);
await player.init();
player.video.src = 'encrypted-stream.mpd';
```

**要点**：
- `encrypted` 事件由浏览器自动触发，当检测到加密内容时
- 许可证请求的消息格式由 DRM 系统决定，不要尝试手动解析
- 密钥状态需要持续监控，处理过期和受限等情况

### 练习二

**思路**：使用 Canvas 的像素操作，在每个视频帧的最低有效位（LSB）嵌入水印数据。

**答案**：

```javascript
class VideoWatermark {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d', { willReadFrequently: true });
  }

  // 将字符串转为二进制位数组
  textToBits(text) {
    const bits = [];
    for (let i = 0; i < text.length; i++) {
      const byte = text.charCodeAt(i);
      for (let bit = 7; bit >= 0; bit--) {
        bits.push((byte >> bit) & 1);
      }
    }
    return bits;
  }

  bitsToText(bits) {
    let text = '';
    for (let i = 0; i < bits.length; i += 8) {
      let byte = 0;
      for (let bit = 0; bit < 8; bit++) {
        byte = (byte << 1) | (bits[i + bit] || 0);
      }
      text += String.fromCharCode(byte);
    }
    return text;
  }

  // 嵌入水印到当前帧
  embed(userId) {
    const imageData = this.ctx.getImageData(0, 0, this.canvas.width, this.canvas.height);
    const data = imageData.data;
    const bits = this.textToBits(userId);

    // 在每帧的左上角区域嵌入水印
    for (let i = 0; i < bits.length; i++) {
      const pixelIndex = i * 4;
      if (pixelIndex >= data.length) break;

      // 修改 R 通道的最低有效位
      data[pixelIndex] = (data[pixelIndex] & 0xFE) | bits[i];
    }

    this.ctx.putImageData(imageData, 0, 0);
  }

  // 从帧中提取水印
  extract(bitLength) {
    const imageData = this.ctx.getImageData(0, 0, this.canvas.width, this.canvas.height);
    const data = imageData.data;
    const bits = [];

    for (let i = 0; i < bitLength; i++) {
      const pixelIndex = i * 4;
      if (pixelIndex >= data.length) break;
      bits.push(data[pixelIndex] & 1);
    }

    return this.bitsToText(bits);
  }

  // 实时处理视频帧
  processVideo(videoElement, userId) {
    const draw = () => {
      this.ctx.drawImage(videoElement, 0, 0, this.canvas.width, this.canvas.height);
      this.embed(userId);
      requestAnimationFrame(draw);
    };
    draw();
  }
}

// 使用示例
const canvas = document.getElementById('canvas');
const wm = new VideoWatermark(canvas);
wm.processVideo(document.getElementById('video'), 'user_12345');

// 验证水印
setTimeout(() => {
  const extracted = wm.extract(40); // 5 个字符 = 40 位
  console.log('提取的水印:', extracted); // "user_"
}, 2000);
```

**要点**：
- LSB 水印对视觉质量影响极小（每个像素最多变化 1）
- 水印区域越大，可嵌入的信息越多，但对画质的影响也越大
- 实际应用中需要考虑视频压缩对水印的破坏
