# 第六课：阶段实战——完整视频平台

## 场景引入

经过五个阶段的学习，你已经掌握了音视频的全部核心技术：编解码、流媒体、实时通信、质量优化、版权保护、跨平台方案。现在，是时候将这些知识整合为一个完整的生产级视频平台。

一个真实的视频平台涉及完整的内容生命周期：用户上传视频 → 后端转码为多种分辨率 → 存储到对象存储 → 通过 CDN 分发 → 前端自适应播放 → 质量监控与分析。每个环节都有大量的工程细节和权衡决策。

本课将带你从零构建一个简化但完整的视频平台，涵盖上传、转码、存储、分发、播放、监控全链路。

## 上传模块

### 分片上传

大文件必须分片上传，支持断点续传和并行上传：

```javascript
// 分片上传 SDK
class ChunkUploader {
  constructor(config) {
    this.chunkSize = config.chunkSize || 5 * 1024 * 1024; // 5MB
    this.maxConcurrent = config.maxConcurrent || 3;
    this.uploadUrl = config.uploadUrl;
    this.retryLimit = config.retryLimit || 3;
  }

  async upload(file, options = {}) {
    const chunks = this.createChunks(file);
    const uploadId = await this.initUpload(file);

    // 并行上传分片
    const results = await this.uploadChunks(chunks, uploadId);

    // 合并分片
    const videoId = await this.completeUpload(uploadId, results);

    return videoId;
  }

  createChunks(file) {
    const chunks = [];
    let start = 0;
    let index = 0;

    while (start < file.size) {
      const end = Math.min(start + this.chunkSize, file.size);
      chunks.push({
        index,
        blob: file.slice(start, end),
        start,
        end,
        size: end - start
      });
      start = end;
      index++;
    }

    return chunks;
  }

  async initUpload(file) {
    const response = await fetch(`${this.uploadUrl}/init`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        filename: file.name,
        fileSize: file.size,
        mimeType: file.type,
        chunkSize: this.chunkSize,
        chunkCount: Math.ceil(file.size / this.chunkSize)
      })
    });

    const data = await response.json();
    return data.uploadId;
  }

  async uploadChunks(chunks, uploadId) {
    const results = [];
    const queue = [...chunks];
    const inProgress = new Map();

    while (queue.length > 0 || inProgress.size > 0) {
      // 填充并发槽
      while (inProgress.size < this.maxConcurrent && queue.length > 0) {
        const chunk = queue.shift();
        const promise = this.uploadChunk(chunk, uploadId)
          .then(result => {
            inProgress.delete(chunk.index);
            results.push(result);
          })
          .catch(error => {
            console.error(`分片 ${chunk.index} 上传失败:`, error);
            // 重试
            if (chunk.retryCount < this.retryLimit) {
              chunk.retryCount = (chunk.retryCount || 0) + 1;
              queue.push(chunk);
            }
            inProgress.delete(chunk.index);
          });
        inProgress.set(chunk.index, promise);
      }

      // 等待至少一个完成
      if (inProgress.size > 0) {
        await Promise.race(inProgress.values());
      }
    }

    return results.sort((a, b) => a.index - b.index);
  }

  async uploadChunk(chunk, uploadId) {
    const formData = new FormData();
    formData.append('chunk', chunk.blob);
    formData.append('index', chunk.index);

    let retries = 0;
    while (retries <= this.retryLimit) {
      try {
        const response = await fetch(`${this.uploadUrl}/${uploadId}/chunk`, {
          method: 'POST',
          body: formData
        });

        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        return await response.json();
      } catch (error) {
        retries++;
        if (retries > this.retryLimit) throw error;
        await this.delay(1000 * retries); // 指数退避
      }
    }
  }

  async completeUpload(uploadId, chunks) {
    const response = await fetch(`${this.uploadUrl}/${uploadId}/complete`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        chunks: chunks.map(c => ({ index: c.index, etag: c.etag }))
      })
    });

    const data = await response.json();
    return data.videoId;
  }

  delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

### 上传进度追踪

```javascript
// 带进度追踪的上传
class UploadProgressTracker {
  constructor() {
    this.chunks = new Map();
    this.startTime = Date.now();
  }

  trackChunk(index, totalChunks) {
    this.chunks.set(index, {
      status: 'pending',
      progress: 0,
      startTime: null,
      endTime: null
    });
  }

  updateProgress(index, loaded, total) {
    const chunk = this.chunks.get(index);
    if (!chunk) return;

    chunk.progress = loaded / total;
    chunk.status = 'uploading';
    if (!chunk.startTime) chunk.startTime = Date.now();

    return this.getOverallProgress();
  }

  completeChunk(index) {
    const chunk = this.chunks.get(index);
    if (chunk) {
      chunk.status = 'completed';
      chunk.progress = 1;
      chunk.endTime = Date.now();
    }
  }

  getOverallProgress() {
    let totalProgress = 0;
    let completedCount = 0;

    for (const chunk of this.chunks.values()) {
      totalProgress += chunk.progress;
      if (chunk.status === 'completed') completedCount++;
    }

    const overall = totalProgress / this.chunks.size;
    const elapsed = (Date.now() - this.startTime) / 1000;
    const speed = (completedCount * this.chunkSize) / elapsed;

    return {
      overall: Math.round(overall * 100),
      completedChunks: completedCount,
      totalChunks: this.chunks.size,
      speed: this.formatSpeed(speed),
      eta: this.estimateETA(overall, speed)
    };
  }

  formatSpeed(bytesPerSec) {
    if (bytesPerSec > 1024 * 1024) {
      return `${(bytesPerSec / 1024 / 1024).toFixed(1)} MB/s`;
    }
    return `${(bytesPerSec / 1024).toFixed(0)} KB/s`;
  }

  estimateETA(progress, speed) {
    if (progress === 0 || speed === 0) return '计算中...';
    const remaining = (1 - progress) * this.totalSize / speed;
    return `${Math.ceil(remaining)} 秒`;
  }
}
```

### FFmpeg 转码管线

```javascript
// 转码服务核心
const { exec } = require('child_process');
const path = require('path');

class TranscodeService {
  constructor(config) {
    this.inputDir = config.inputDir;
    this.outputDir = config.outputDir;
    this.ffmpegPath = config.ffmpegPath || 'ffmpeg';

    // 转码配置
    this.renditions = [
      { name: '1080p', width: 1920, height: 1080, bitrate: '5800k', audioBitrate: '192k' },
      { name: '720p',  width: 1280, height: 720,  bitrate: '2800k', audioBitrate: '128k' },
      { name: '480p',  width: 854,  height: 480,  bitrate: '1400k', audioBitrate: '128k' },
      { name: '360p',  width: 640,  height: 360,  bitrate: '800k',  audioBitrate: '96k' },
    ];
  }

  async transcode(videoId, inputFile) {
    const outputDir = path.join(this.outputDir, videoId);
    const results = [];

    // 1. 获取源视频信息
    const videoInfo = await this.getVideoInfo(inputFile);
    console.log('源视频信息:', videoInfo);

    // 2. 选择需要生成的清晰度（不超过源分辨率）
    const targetRenditions = this.renditions.filter(r =>
      r.height <= videoInfo.height
    );

    // 3. 并行转码各清晰度
    const tasks = targetRenditions.map(rendition =>
      this.transcodeToRendition(inputFile, outputDir, rendition, videoInfo)
    );

    const renditionResults = await Promise.all(tasks);
    results.push(...renditionResults);

    // 4. 生成 HLS 分片
    await this.generateHLS(videoId, outputDir, targetRenditions);

    // 5. 生成缩略图
    await this.generateThumbnails(inputFile, outputDir);

    return {
      videoId,
      renditions: results,
      hlsPlaylist: `${outputDir}/master.m3u8`
    };
  }

  async getVideoInfo(inputFile) {
    return new Promise((resolve, reject) => {
      const cmd = `${this.ffmpegPath} -i "${inputFile}" -print_format json -show_streams -show_format`;
      exec(cmd, (error, stdout) => {
        if (error) return reject(error);

        const info = JSON.parse(stdout);
        const videoStream = info.streams.find(s => s.codec_type === 'video');
        const audioStream = info.streams.find(s => s.codec_type === 'audio');

        resolve({
          width: videoStream.width,
          height: videoStream.height,
          duration: parseFloat(info.format.duration),
          bitrate: parseInt(info.format.bit_rate),
          fps: eval(videoStream.r_frame_rate),
          videoCodec: videoStream.codec_name,
          audioCodec: audioStream?.codec_name
        });
      });
    });
  }

  async transcodeToRendition(inputFile, outputDir, rendition, sourceInfo) {
    const outputFile = path.join(outputDir, `${rendition.name}.mp4`);

    const cmd = [
      this.ffmpegPath,
      '-i', `"${inputFile}"`,
      '-vf', `scale=${rendition.width}:${rendition.height}`,
      '-c:v', 'libx264',
      '-preset', 'medium',
      '-b:v', rendition.bitrate,
      '-maxrate', rendition.bitrate,
      '-bufsize', `${parseInt(rendition.bitrate) * 2}k`,
      '-c:a', 'aac',
      '-b:a', rendition.audioBitrate,
      '-movflags', '+faststart',
      '-y',
      `"${outputFile}"`
    ].join(' ');

    return new Promise((resolve, reject) => {
      exec(cmd, (error) => {
        if (error) return reject(error);
        resolve({
          name: rendition.name,
          path: outputFile,
          width: rendition.width,
          height: rendition.height,
          bitrate: rendition.bitrate
        });
      });
    });
  }

  async generateHLS(videoId, outputDir, renditions) {
    // 生成主播放列表（Master Playlist）
    let masterPlaylist = '#EXTM3U\n#EXT-X-VERSION:3\n\n';

    for (const rendition of renditions) {
      const bandwidth = parseInt(rendition.bitrate) * 1000;
      masterPlaylist += `#EXT-X-STREAM-INF:BANDWIDTH=${bandwidth},RESOLUTION=${rendition.width}x${rendition.height}\n`;
      masterPlaylist += `${rendition.name}/playlist.m3u8\n\n`;
    }

    const fs = require('fs');
    fs.writeFileSync(path.join(outputDir, 'master.m3u8'), masterPlaylist);

    // 为每个清晰度生成分片
    for (const rendition of renditions) {
      const renditionDir = path.join(outputDir, rendition.name);
      fs.mkdirSync(renditionDir, { recursive: true });

      const cmd = [
        this.ffmpegPath,
        '-i', `"${path.join(outputDir, `${rendition.name}.mp4`)}"`,
        '-c', 'copy',
        '-hls_time', '6',
        '-hls_list_size', '0',
        '-hls_segment_filename', `"${renditionDir}/segment_%03d.ts"`,
        `"${renditionDir}/playlist.m3u8"`
      ].join(' ');

      await new Promise((resolve, reject) => {
        exec(cmd, (error) => error ? reject(error) : resolve());
      });
    }
  }

  async generateThumbnails(inputFile, outputDir) {
    const cmd = [
      this.ffmpegPath,
      '-i', `"${inputFile}"`,
      '-vf', 'fps=1/10,scale=320:180',
      '-q:v', '5',
      `"${outputDir}/thumb_%03d.jpg"`
    ].join(' ');

    return new Promise((resolve, reject) => {
      exec(cmd, (error) => error ? reject(error) : resolve());
    });
  }
}
```

```javascript
// 基于 hls.js 的自适应播放器
import Hls from 'hls.js';

class AdaptivePlayer {
  constructor(videoElement, config = {}) {
    this.video = videoElement;
    this.hls = null;
    this.config = {
      enableWorker: true,
      lowLatencyMode: false,
      maxBufferLength: 30,
      maxMaxBufferLength: 600,
      startLevel: -1, // 自动选择起始码率
      ...config
    };

    this.metrics = {
      bufferHealth: 0,
      currentLevel: -1,
      droppedFrames: 0,
      loadLatency: 0,
      bitrateHistory: []
    };
  }

  load(sourceUrl) {
    if (Hls.isSupported()) {
      this.hls = new Hls(this.config);

      this.hls.loadSource(sourceUrl);
      this.hls.attachMedia(this.video);

      this.hls.on(Hls.Events.MANIFEST_PARSED, (event, data) => {
        console.log('可用码率:', data.levels.map(l =>
          `${l.width}x${l.height} @ ${l.bitrate / 1000}kbps`
        ));
        this.video.play();
      });

      this.hls.on(Hls.Events.LEVEL_SWITCHED, (event, data) => {
        const level = this.hls.levels[data.level];
        this.metrics.currentLevel = data.level;
        this.metrics.bitrateHistory.push({
          time: Date.now(),
          level: data.level,
          bitrate: level.bitrate
        });
      });

      this.setupMetricsCollection();
    } else if (this.video.canPlayType('application/vnd.apple.mpegurl')) {
      // Safari 原生支持 HLS
      this.video.src = sourceUrl;
    }
  }

  setupMetricsCollection() {
    setInterval(() => {
      if (!this.hls) return;

      // 缓冲健康度
      const buffered = this.video.buffered;
      if (buffered.length > 0) {
        this.metrics.bufferHealth = buffered.end(buffered.length - 1) - this.video.currentTime;
      }

      // 丢帧统计
      this.metrics.droppedFrames = this.video.webkitDroppedFrameCount || 0;
    }, 1000);
  }

  // 手动切换码率
  setLevel(levelIndex) {
    if (this.hls) {
      this.hls.currentLevel = levelIndex;
    }
  }

  // 获取当前播放质量
  getQuality() {
    if (!this.hls) return null;

    const level = this.hls.levels[this.hls.currentLevel];
    return {
      width: level?.width,
      height: level?.height,
      bitrate: level?.bitrate,
      bufferHealth: this.metrics.bufferHealth
    };
  }

  destroy() {
    if (this.hls) {
      this.hls.destroy();
      this.hls = null;
    }
  }
}
```

### QoS 指标采集

```javascript
// 播放质量监控 SDK
class QualityMonitor {
  constructor(player) {
    this.player = player;
    this.sessionId = this.generateSessionId();
    this.events = [];
    this.startTime = Date.now();

    this.setupEventListeners();
  }

  setupEventListeners() {
    const video = this.player.video;

    video.addEventListener('play', () => this.logEvent('play'));
    video.addEventListener('pause', () => this.logEvent('pause'));
    video.addEventListener('waiting', () => this.logEvent('buffer_start'));
    video.addEventListener('playing', () => this.logEvent('buffer_end'));
    video.addEventListener('error', () => this.logEvent('error', {
      code: video.error?.code,
      message: video.error?.message
    }));

    // 定期采集指标
    setInterval(() => this.collectMetrics(), 5000);
  }

  collectMetrics() {
    const video = this.player.video;
    const quality = this.player.getQuality();

    const metrics = {
      timestamp: Date.now(),
      currentTime: video.currentTime,
      duration: video.duration,
      paused: video.paused,
      muted: video.muted,
      volume: video.volume,

      // 缓冲指标
      bufferHealth: quality?.bufferHealth || 0,
      bufferedRanges: this.getBufferedRanges(),

      // 质量指标
      currentResolution: quality ? `${quality.width}x${quality.height}` : 'unknown',
      currentBitrate: quality?.bitrate || 0,

      // 性能指标
      droppedFrames: video.webkitDroppedFrameCount || 0,
      totalFrames: video.webkitDecodedFrameCount || 0,

      // 网络指标
      downloadSpeed: this.estimateDownloadSpeed(),
    };

    this.events.push({ type: 'metrics', data: metrics });
  }

  getBufferedRanges() {
    const ranges = [];
    const buffered = this.player.video.buffered;
    for (let i = 0; i < buffered.length; i++) {
      ranges.push({
        start: buffered.start(i),
        end: buffered.end(i)
      });
    }
    return ranges;
  }

  estimateDownloadSpeed() {
    // 基于最近的分片下载速度估算
    if (this.player.hls) {
      const stats = this.player.hls.stats;
      return stats.total / (stats.loading.end - stats.loading.start) * 1000;
    }
    return 0;
  }

  logEvent(type, data = {}) {
    this.events.push({
      type,
      timestamp: Date.now(),
      data
    });
  }

  // 生成质量报告
  generateReport() {
    const metrics = this.events.filter(e => e.type === 'metrics');
    if (metrics.length === 0) return null;

    const bufferEvents = this.events.filter(e =>
      e.type === 'buffer_start' || e.type === 'buffer_end'
    );

    const totalBufferTime = this.calculateTotalBufferTime(bufferEvents);
    const sessionDuration = (Date.now() - this.startTime) / 1000;

    return {
      sessionId: this.sessionId,
      duration: sessionDuration,
      averageBitrate: this.calculateAverageBitrate(metrics),
      rebufferCount: bufferEvents.length / 2,
      rebufferRatio: totalBufferTime / sessionDuration,
      resolutionChanges: this.countResolutionChanges(metrics),
      errorCount: this.events.filter(e => e.type === 'error').length,
      qualityScore: this.calculateQualityScore(metrics)
    };
  }

  calculateAverageBitrate(metrics) {
    const bitrates = metrics
      .map(m => m.data.currentBitrate)
      .filter(b => b > 0);
    return bitrates.length > 0
      ? bitrates.reduce((a, b) => a + b, 0) / bitrates.length
      : 0;
  }

  calculateTotalBufferTime(bufferEvents) {
    let total = 0;
    for (let i = 0; i < bufferEvents.length - 1; i += 2) {
      if (bufferEvents[i].type === 'buffer_start' &&
          bufferEvents[i + 1]?.type === 'buffer_end') {
        total += bufferEvents[i + 1].timestamp - bufferEvents[i].timestamp;
      }
    }
    return total / 1000;
  }

  countResolutionChanges(metrics) {
    let changes = 0;
    for (let i = 1; i < metrics.length; i++) {
      if (metrics[i].data.currentResolution !== metrics[i - 1].data.currentResolution) {
        changes++;
      }
    }
    return changes;
  }

  calculateQualityScore(metrics) {
    // 综合质量评分（0-100）
    const avgBitrate = this.calculateAverageBitrate(metrics);
    const bitrateScore = Math.min(avgBitrate / 5000000, 1) * 40;

    const bufferHealth = metrics[metrics.length - 1].data.bufferHealth;
    const bufferScore = Math.min(bufferHealth / 10, 1) * 30;

    const dropRate = metrics[metrics.length - 1].data.droppedFrames /
      (metrics[metrics.length - 1].data.totalFrames || 1);
    const dropScore = (1 - Math.min(dropRate * 100, 1)) * 30;

    return Math.round(bitrateScore + bufferScore + dropScore);
  }

  generateSessionId() {
    return `session_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }
}
```

```javascript
  },

  // 2. 编码优化
  encodingOptimization: {

    codecMigration: {

    },

    perTitleEncoding: true,

    // 移除低使用率的清晰度
    removeUnusedRenditions: {
      threshold: '使用率 < 1% 的清晰度可以移除',

    }
  },

    tieredCaching: true,

    prefetch: {
      trigger: '新视频上线后',
      scope: '前 24 小时的预测热门内容'
    },

    // 回源流量优化
    originShield: '启用 CDN 源站保护，减少回源请求'
  }

// 成本计算工具
function estimateMonthlyCost(config) {
  const { storageGB, transferGB, transcodingHours, requests } = config;

  const costs = {
    // 存储成本（按月）
    storage: storageGB * 0.023,  // $0.023/GB/月

    // 流量成本
    transfer: transferGB * 0.09,  // $0.09/GB

    // 转码成本
    transcoding: transcodingHours * 0.03,  // $0.03/分钟

    // 请求成本
    requests: requests / 10000 * 0.004,  // $0.004/万次
  };

  costs.total = Object.values(costs).reduce((a, b) => a + b, 0);
  return costs;
}
```

## 常见误区

### 误区一：忽略成本优化

视频平台的存储和流量成本会随着用户增长快速膨胀。一个百万用户的平台，每月的 CDN 流量费用可能高达数万美元。从项目初期就应该考虑分层存储、编码优化、CDN 策略等成本控制措施。

### 误区二：追求极致画质

用户在手机上观看 720p 和 1080p 的差异很小，但码率可能相差一倍。盲目追求高画质会导致带宽成本翻倍，而用户体验提升有限。应该基于真实用户数据做决策，而非工程师的主观判断。

### 误区三：所有功能都需要自研

播放器、转码服务、推荐算法——这些复杂功能不一定需要从零开始。开源方案（hls.js、FFmpeg）和云服务（AWS MediaConvert、阿里云视频点播）可以大幅降低开发成本。自研的核心价值在于差异化竞争力，而非重复造轮子。

### 误区四：上线后再考虑监控

没有监控的视频平台就像没有仪表盘的汽车——出了问题你都不知道。从第一个版本开始就应该埋入质量监控，包括播放成功率、缓冲率、错误率、码率分布等核心指标。

### 练习一

**思路**：使用用户-视频交互矩阵计算相似度，结合视频标签做内容推荐，最终混合两种推荐结果。

**答案**：

```javascript
class RecommendationEngine {
  constructor() {
    this.userHistory = new Map();  // userId -> [{videoId, watchTime, rating}]
    this.videoTags = new Map();    // videoId -> [tag1, tag2, ...]
    this.videoStats = new Map();   // videoId -> {views, avgWatchTime, completionRate}
  }

  // 记录用户行为
  recordInteraction(userId, videoId, watchTime, rating = null) {
    if (!this.userHistory.has(userId)) {
      this.userHistory.set(userId, []);
    }
    this.userHistory.get(userId).push({ videoId, watchTime, rating });
  }

  // 协同过滤推荐
  collaborativeFilter(userId, limit = 10) {
    const userVideos = new Set(
      (this.userHistory.get(userId) || []).map(h => h.videoId)
    );

    // 找到相似用户
    const similarities = [];
    for (const [otherId, otherHistory] of this.userHistory) {
      if (otherId === userId) continue;
      const sim = this.userSimilarity(userId, otherId);
      if (sim > 0.3) similarities.push({ userId: otherId, similarity: sim });
    }

    // 按相似度排序
    similarities.sort((a, b) => b.similarity - a.similarity);
    const topSimilar = similarities.slice(0, 20);

    // 从相似用户中收集推荐
    const candidates = new Map();
    for (const { userId: similarUser, similarity } of topSimilar) {
      for (const history of this.userHistory.get(similarUser)) {
        if (userVideos.has(history.videoId)) continue;

        const score = similarity * (history.rating || 3);
        candidates.set(
          history.videoId,
          (candidates.get(history.videoId) || 0) + score
        );
      }
    }

    return Array.from(candidates.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, limit)
      .map(([videoId, score]) => ({ videoId, score, method: 'collaborative' }));
  }

  // 计算用户相似度（余弦相似度）
  userSimilarity(userId1, userId2) {
    const history1 = this.userHistory.get(userId1);
    const history2 = this.userHistory.get(userId2);

    const videos1 = new Map(history1.map(h => [h.videoId, h.rating || 3]));
    const videos2 = new Map(history2.map(h => [h.videoId, h.rating || 3]));

    // 找到共同评分的视频
    const commonVideos = [...videos1.keys()].filter(v => videos2.has(v));
    if (commonVideos.length === 0) return 0;

    // 计算余弦相似度
    let dotProduct = 0, norm1 = 0, norm2 = 0;
    for (const videoId of commonVideos) {
      const r1 = videos1.get(videoId);
      const r2 = videos2.get(videoId);
      dotProduct += r1 * r2;
      norm1 += r1 * r1;
      norm2 += r2 * r2;
    }

    return dotProduct / (Math.sqrt(norm1) * Math.sqrt(norm2));
  }

  // 基于内容的推荐
  contentBased(userId, limit = 10) {
    const userVideos = this.userHistory.get(userId) || [];
    if (userVideos.length === 0) return this.getPopularVideos(limit);

    // 计算用户偏好标签
    const tagScores = new Map();
    for (const { videoId, rating } of userVideos) {
      const tags = this.videoTags.get(videoId) || [];
      const weight = rating || 3;
      for (const tag of tags) {
        tagScores.set(tag, (tagScores.get(tag) || 0) + weight);
      }
    }

    // 归一化标签权重
    const maxScore = Math.max(...tagScores.values());
    for (const [tag, score] of tagScores) {
      tagScores.set(tag, score / maxScore);
    }

    // 计算候选视频与用户偏好的匹配度
    const watchedIds = new Set(userVideos.map(h => h.videoId));
    const candidates = [];

    for (const [videoId, tags] of this.videoTags) {
      if (watchedIds.has(videoId)) continue;

      let matchScore = 0;
      for (const tag of tags) {
        matchScore += tagScores.get(tag) || 0;
      }

      if (matchScore > 0) {
        candidates.push({ videoId, score: matchScore, method: 'content' });
      }
    }

    return candidates
      .sort((a, b) => b.score - a.score)
      .slice(0, limit);
  }

  // 混合推荐
  recommend(userId, limit = 10) {
    const collaborative = this.collaborativeFilter(userId, limit);
    const content = this.contentBased(userId, limit);

    // 混合权重
    const scores = new Map();
    for (const item of collaborative) {
      scores.set(item.videoId, item.score * 0.6);
    }
    for (const item of content) {
      const current = scores.get(item.videoId) || 0;
      scores.set(item.videoId, current + item.score * 0.4);
    }

    return Array.from(scores.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, limit)
      .map(([videoId, score]) => ({ videoId, score }));
  }

  // 热门视频（冷启动回退）
  getPopularVideos(limit) {
    const sorted = Array.from(this.videoStats.entries())
      .sort((a, b) => b[1].views - a[1].views)
      .slice(0, limit);

    return sorted.map(([videoId]) => ({ videoId, score: 1, method: 'popular' }));
  }
}
```

**要点**：
- 协同过滤适合有足够用户行为数据的场景
- 基于内容的推荐适合冷启动阶段（新用户或新视频）
- 实际生产中需要考虑实时性、多样性和新颖性

### 练习二

**思路**：将用户随机分配到实验组，记录各组的质量指标，使用统计检验判断差异是否显著。

**答案**：

```javascript
class ABTestManager {
  constructor() {
    this.experiments = new Map();
    this.userAssignments = new Map();
  }

  // 创建实验
  createExperiment(config) {
    const experiment = {
      id: config.id,
      name: config.name,
      variants: config.variants,  // [{id: 'control', name: '默认'}, {id: 'treatment', name: '新方案'}]
      trafficSplit: config.trafficSplit || [50, 50],  // 流量分配比例
      metrics: config.metrics || ['rebufferRatio', 'averageBitrate', 'qualityScore'],
      startDate: Date.now(),
      endDate: config.endDate || Date.now() + 7 * 24 * 60 * 60 * 1000,
      results: new Map()
    };

    // 初始化每个变体的结果收集
    for (const variant of experiment.variants) {
      experiment.results.set(variant.id, {
        participants: 0,
        metrics: {},
        samples: []
      });
    }

    this.experiments.set(config.id, experiment);
    return experiment;
  }

  // 用户分组
  assignUser(userId, experimentId) {
    const key = `${userId}_${experimentId}`;
    if (this.userAssignments.has(key)) {
      return this.userAssignments.get(key);
    }

    const experiment = this.experiments.get(experimentId);
    if (!experiment) return null;

    // 基于用户 ID 哈希的确定性分组
    const hash = this.hashCode(userId + experimentId);
    const bucket = Math.abs(hash) % 100;

    let cumulative = 0;
    for (let i = 0; i < experiment.variants.length; i++) {
      cumulative += experiment.trafficSplit[i];
      if (bucket < cumulative) {
        const variant = experiment.variants[i];
        this.userAssignments.set(key, variant.id);

        const result = experiment.results.get(variant.id);
        result.participants++;

        return variant.id;
      }
    }

    return experiment.variants[0].id;
  }

  // 记录指标
  recordMetric(userId, experimentId, metricName, value) {
    const key = `${userId}_${experimentId}`;
    const variantId = this.userAssignments.get(key);
    if (!variantId) return;

    const experiment = this.experiments.get(experimentId);
    const result = experiment.results.get(variantId);

    result.samples.push({
      userId,
      metric: metricName,
      value,
      timestamp: Date.now()
    });
  }

  // 分析结果
  analyze(experimentId) {
    const experiment = this.experiments.get(experimentId);
    if (!experiment) return null;

    const analysis = {};

    for (const metric of experiment.metrics) {
      const variantStats = {};

      for (const variant of experiment.variants) {
        const result = experiment.results.get(variant.id);
        const values = result.samples
          .filter(s => s.metric === metric)
          .map(s => s.value);

        if (values.length === 0) continue;

        const mean = values.reduce((a, b) => a + b, 0) / values.length;
        const variance = values.reduce((sum, v) => sum + (v - mean) ** 2, 0) / values.length;
        const stdError = Math.sqrt(variance / values.length);

        variantStats[variant.id] = {
          mean: mean.toFixed(4),
          stdError: stdError.toFixed(4),
          sampleSize: values.length,
          confidenceInterval: [
            (mean - 1.96 * stdError).toFixed(4),
            (mean + 1.96 * stdError).toFixed(4)
          ]
        };
      }

      // 计算统计显著性（简化版 t 检验）
      const control = variantStats['control'];
      const treatment = variantStats['treatment'];

      if (control && treatment) {
        const diff = treatment.mean - control.mean;
        const pooledSE = Math.sqrt(
          control.stdError ** 2 + treatment.stdError ** 2
        );
        const tStat = diff / pooledSE;
        const isSignificant = Math.abs(tStat) > 1.96; // 95% 置信度

        analysis[metric] = {
          control: control.mean,
          treatment: treatment.mean,
          difference: diff.toFixed(4),
          percentChange: ((diff / control.mean) * 100).toFixed(2) + '%',
          tStatistic: tStat.toFixed(4),
          isSignificant,
          winner: isSignificant
            ? (diff > 0 ? 'treatment' : 'control')
            : 'no clear winner'
        };
      }
    }

    return {
      experimentId,
      name: experiment.name,
      duration: Date.now() - experiment.startDate,
      analysis,
      recommendation: this.generateRecommendation(analysis)
    };
  }

  generateRecommendation(analysis) {
    const significantImprovements = Object.entries(analysis)
      .filter(([_, result]) => result.isSignificant && result.winner === 'treatment');

    if (significantImprovements.length === 0) {
      return '实验结果无显著差异，建议继续收集数据或调整实验方案';
    }

    const improvedMetrics = significantImprovements.map(([metric]) => metric);
    return `实验方案在 ${improvedMetrics.join(', ')} 指标上显著优于对照组，建议全量上线`;
  }

  hashCode(str) {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      hash = ((hash << 5) - hash) + str.charCodeAt(i);
      hash |= 0;
    }
    return hash;
  }
}
```
