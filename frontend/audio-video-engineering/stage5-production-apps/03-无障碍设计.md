# 第三课：无障碍设计

## 场景引入

BBC 曾做过一项调查：英国约有 1200 万人存在听力障碍，200 万人存在视力障碍。当 BBC iPlayer 推出无障碍功能后，字幕使用量远超预期——不仅听障用户使用，大量普通用户也在嘈杂环境、安静场所或学习外语时依赖字幕。

无障碍设计不是"为少数人做的额外工作"，而是提升所有用户体验的基础能力。当你的视频平台缺少字幕，你失去的不只是听障用户，还有在地铁上看视频的通勤者、在安静办公室学习的白领、以及所有非母语观众。

本课将系统讲解音视频无障碍设计的核心规范和实现方案。

## 学习目标

完成本课学习后，你将能够：

1. 理解字幕规范（封闭字幕、开放字幕、音频描述）
2. 使用 WebVTT 和 track 元素实现无障碍字幕
3. 实现手语视频窗口和音频描述功能
4. 设计符合 WCAG 标准的键盘导航方案
5. 将无障碍设计融入开发流程，而非事后补救

### 动态字幕生成

```javascript
// 运行时动态生成 WebVTT
class DynamicCaptionGenerator {
  constructor(videoElement) {
    this.video = videoElement;
    this.captions = [];
    this.track = null;
    this.cueIndex = 0;
  }

  // 添加字幕
  addCaption(startTime, endTime, text, speaker = '') {
    const vttText = speaker ? `<v ${speaker}>${text}` : text;
    const cue = new VTTCue(startTime, endTime, vttText);

    // 设置字幕定位
    cue.line = -3;  // 距离底部 3 行
    cue.align = 'center';

    if (this.track) {
      this.track.addCue(cue);
    }

    this.captions.push({ startTime, endTime, text, speaker });
  }

  // 从实时语音识别结果生成字幕
  processSpeechResult(result) {
    const startTime = result.startTime / 1000;
    const endTime = result.endTime / 1000;
    const text = result.transcript;
    const isFinal = result.isFinal;

    if (isFinal) {
      this.addCaption(startTime, endTime, text, result.speaker);
    } else {
      // 临时结果：更新最后一行字幕
      this.updatePendingCaption(startTime, endTime, text);
    }
  }

  updatePendingCaption(startTime, endTime, text) {
    if (this.track && this.track.cues.length > 0) {
      const lastCue = this.track.cues[this.track.cues.length - 1];
      lastCue.text = text;
      lastCue.endTime = endTime;
    }
  }

  // 创建字幕轨道
  createTrack(language = 'zh') {
    this.track = this.video.addTextTrack('subtitles', '实时字幕', language);
    this.track.mode = 'showing';
    return this.track;
  }
}
```

```javascript
// 手语视频画中画
class SignLanguageWindow {
  constructor(mainVideo) {
    this.mainVideo = mainVideo;
    this.signVideo = null;
    this.container = null;
    this.isDragging = false;
    this.position = { x: 10, y: 10 };
    this.size = { width: 200, height: 150 };
  }

  // 创建手语视频窗口
  create(signLanguageSrc) {
    // 创建容器
    this.container = document.createElement('div');
    this.container.id = 'sign-language-window';
    Object.assign(this.container.style, {
      position: 'absolute',
      bottom: '10px',
      right: '10px',
      width: `${this.size.width}px`,
      height: `${this.size.height}px`,
      border: '2px solid #fff',
      borderRadius: '8px',
      overflow: 'hidden',
      cursor: 'move',
      zIndex: '100',
      boxShadow: '0 4px 12px rgba(0,0,0,0.5)'
    });

    // 创建手语视频元素
    this.signVideo = document.createElement('video');
    this.signVideo.src = signLanguageSrc;
    this.signVideo.autoplay = true;
    this.signVideo.muted = true;
    this.signVideo.playsInline = true;
    Object.assign(this.signVideo.style, {
      width: '100%',
      height: '100%',
      objectFit: 'cover'
    });

    this.container.appendChild(this.signVideo);

    // 同步播放
    this.setupSync();

    // 拖拽功能
    this.setupDrag();

    // 控制按钮
    this.createControls();

    return this.container;
  }

  setupSync() {
    // 主视频播放/暂停时同步手语视频
    this.mainVideo.addEventListener('play', () => this.signVideo.play());
    this.mainVideo.addEventListener('pause', () => this.signVideo.pause());
    this.mainVideo.addEventListener('seeked', () => {
      this.signVideo.currentTime = this.mainVideo.currentTime;
    });

    // 处理时间偏差
    this.mainVideo.addEventListener('timeupdate', () => {
      const drift = Math.abs(this.mainVideo.currentTime - this.signVideo.currentTime);
      if (drift > 0.3) {
        this.signVideo.currentTime = this.mainVideo.currentTime;
      }
    });
  }

  setupDrag() {
    let startX, startY, startLeft, startTop;

    this.container.addEventListener('mousedown', (e) => {
      this.isDragging = true;
      startX = e.clientX;
      startY = e.clientY;
      startLeft = this.container.offsetLeft;
      startTop = this.container.offsetTop;
      e.preventDefault();
    });

    document.addEventListener('mousemove', (e) => {
      if (!this.isDragging) return;
      const dx = e.clientX - startX;
      const dy = e.clientY - startY;
      this.container.style.left = `${startLeft + dx}px`;
      this.container.style.top = `${startTop + dy}px`;
      this.container.style.right = 'auto';
      this.container.style.bottom = 'auto';
    });

    document.addEventListener('mouseup', () => {
      this.isDragging = false;
    });
  }

  createControls() {
    const controls = document.createElement('div');
    Object.assign(controls.style, {
      position: 'absolute',
      bottom: '5px',
      left: '5px',
      display: 'flex',
      gap: '5px'
    });

    // 缩放按钮
    const resizeBtn = document.createElement('button');
    resizeBtn.textContent = '↔';
    resizeBtn.title = '调整大小';
    resizeBtn.addEventListener('click', () => this.toggleSize());
    controls.appendChild(resizeBtn);

    // 关闭按钮
    const closeBtn = document.createElement('button');
    closeBtn.textContent = '×';
    closeBtn.title = '关闭手语窗口';
    closeBtn.addEventListener('click', () => this.hide());
    controls.appendChild(closeBtn);

    this.container.appendChild(controls);
  }

  toggleSize() {
    const isSmall = this.size.width === 200;
    this.size = isSmall
      ? { width: 320, height: 240 }
      : { width: 200, height: 150 };

    this.container.style.width = `${this.size.width}px`;
    this.container.style.height = `${this.size.height}px`;
  }

  show() {
    if (this.container) this.container.style.display = 'block';
  }

  hide() {
    if (this.container) this.container.style.display = 'none';
  }
}
```

## 音频描述

### 音频描述实现

```javascript
// 音频描述管理器
class AudioDescriptionManager {
  constructor(videoElement) {
    this.video = videoElement;
    this.descriptionTrack = null;
    this.synth = window.speechSynthesis;
    this.isEnabled = false;
    this.volume = 0.8;
  }

  // 从 WebVTT 加载音频描述
  async loadDescriptions(vttUrl) {
    const response = await fetch(vttUrl);
    const text = await response.text();
    this.parseVTT(text);
  }

  parseVTT(vttText) {
    const lines = vttText.split('\n');
    const cues = [];
    let currentCue = null;

    for (const line of lines) {
      if (line.includes('-->')) {
        const [start, end] = line.split('-->').map(t => this.parseTime(t.trim()));
        currentCue = { start, end, text: '' };
      } else if (currentCue && line.trim()) {
        currentCue.text += (currentCue.text ? ' ' : '') + line.trim();
      } else if (currentCue && !line.trim()) {
        cues.push(currentCue);
        currentCue = null;
      }
    }

    this.descriptionCues = cues;
  }

  parseTime(timeStr) {
    const parts = timeStr.split(':');
    const seconds = parseFloat(parts[2]);
    return parseInt(parts[0]) * 3600 + parseInt(parts[1]) * 60 + seconds;
  }

  // 启用音频描述
  enable() {
    this.isEnabled = true;
    this.video.addEventListener('timeupdate', () => this.checkAndSpeak());
  }

  disable() {
    this.isEnabled = false;
    this.synth.cancel();
  }

  checkAndSpeak() {
    if (!this.isEnabled || !this.descriptionCues) return;

    const currentTime = this.video.currentTime;

    for (const cue of this.descriptionCues) {
      if (Math.abs(currentTime - cue.start) < 0.1) {
        this.speak(cue.text);
        break;
      }
    }
  }

  speak(text) {
    this.synth.cancel();

    const utterance = new SpeechSynthesisUtterance(text);
    utterance.lang = 'zh-CN';
    utterance.volume = this.volume;
    utterance.rate = 1.1; // 稍快的语速，减少对主音频的干扰

    this.synth.speak(utterance);
  }

  // 调整描述音量
  setVolume(vol) {
    this.volume = Math.max(0, Math.min(1, vol));
  }
}
```

## 键盘导航

### 焦点管理

```javascript
// 视频播放器键盘控制
class AccessibleVideoControls {
  constructor(videoElement, controlsContainer) {
    this.video = videoElement;
    this.container = controlsContainer;
    this.shortcuts = new Map();

    this.setupShortcuts();
    this.setupFocusManagement();
  }

  setupShortcuts() {
    this.shortcuts.set('Space', { action: 'togglePlay', label: '播放/暂停' });
    this.shortcuts.set('ArrowLeft', { action: 'seekBackward', label: '后退 5 秒' });
    this.shortcuts.set('ArrowRight', { action: 'seekForward', label: '前进 5 秒' });
    this.shortcuts.set('ArrowUp', { action: 'volumeUp', label: '增大音量' });
    this.shortcuts.set('ArrowDown', { action: 'volumeDown', label: '减小音量' });
    this.shortcuts.set('KeyM', { action: 'toggleMute', label: '静音' });
    this.shortcuts.set('KeyF', { action: 'toggleFullscreen', label: '全屏' });
    this.shortcuts.set('KeyC', { action: 'toggleCaptions', label: '字幕' });
    this.shortcuts.set('Home', { action: 'seekToStart', label: '跳到开头' });
    this.shortcuts.set('End', { action: 'seekToEnd', label: '跳到结尾' });
    this.shortcuts.set('Period', { action: 'nextFrame', label: '下一帧' });
    this.shortcuts.set('Comma', { action: 'prevFrame', label: '上一帧' });

    document.addEventListener('keydown', (e) => this.handleKeydown(e));
  }

  handleKeydown(e) {
    const shortcut = this.shortcuts.get(e.code);
    if (!shortcut) return;

    // 防止与输入框冲突
    if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') return;

    e.preventDefault();
    this.executeAction(shortcut.action);

    // 无障碍反馈
    this.announceAction(shortcut.label);
  }

  executeAction(action) {
    switch (action) {
      case 'togglePlay':
        this.video.paused ? this.video.play() : this.video.pause();
        break;
      case 'seekBackward':
        this.video.currentTime = Math.max(0, this.video.currentTime - 5);
        break;
      case 'seekForward':
        this.video.currentTime = Math.min(this.video.duration, this.video.currentTime + 5);
        break;
      case 'volumeUp':
        this.video.volume = Math.min(1, this.video.volume + 0.1);
        break;
      case 'volumeDown':
        this.video.volume = Math.max(0, this.video.volume - 0.1);
        break;
      case 'toggleMute':
        this.video.muted = !this.video.muted;
        break;
      case 'toggleFullscreen':
        if (document.fullscreenElement) {
          document.exitFullscreen();
        } else {
          this.video.requestFullscreen();
        }
        break;
      case 'toggleCaptions':
        this.toggleCaptions();
        break;
      case 'seekToStart':
        this.video.currentTime = 0;
        break;
      case 'seekToEnd':
        this.video.currentTime = this.video.duration;
        break;
      case 'nextFrame':
        this.video.currentTime += 1 / 30;
        this.video.pause();
        break;
      case 'prevFrame':
        this.video.currentTime -= 1 / 30;
        this.video.pause();
        break;
    }
  }

  toggleCaptions() {
    const tracks = this.video.textTracks;
    let hasShowing = false;

    for (let i = 0; i < tracks.length; i++) {
      if (tracks[i].kind === 'subtitles' && tracks[i].mode === 'showing') {
        hasShowing = true;
        tracks[i].mode = 'disabled';
      }
    }

    if (!hasShowing) {
      for (let i = 0; i < tracks.length; i++) {
        if (tracks[i].kind === 'subtitles') {
          tracks[i].mode = 'showing';
          break;
        }
      }
    }
  }

  // 屏幕阅读器反馈
  announceAction(message) {
    let announcer = document.getElementById('sr-announcer');
    if (!announcer) {
      announcer = document.createElement('div');
      announcer.id = 'sr-announcer';
      announcer.setAttribute('aria-live', 'polite');
      announcer.setAttribute('aria-atomic', 'true');
      announcer.style.cssText = 'position:absolute;left:-9999px;width:1px;height:1px;';
      document.body.appendChild(announcer);
    }
    announcer.textContent = message;
  }

  setupFocusManagement() {
    // 确保控件可聚焦
    this.container.setAttribute('tabindex', '0');
    this.container.setAttribute('role', 'application');
    this.container.setAttribute('aria-label', '视频播放器');
  }
}
```

### 合规性检测

```javascript
// 自动检测 WCAG 合规性
function checkVideoAccessibility(videoElement) {
  const issues = [];

  // 检查是否有字幕轨道
  const tracks = videoElement.textTracks;
  let hasCaptions = false;
  let hasSubtitles = false;

  for (let i = 0; i < tracks.length; i++) {
    if (tracks[i].kind === 'captions') hasCaptions = true;
    if (tracks[i].kind === 'subtitles') hasSubtitles = true;
  }

  if (!hasCaptions && !hasSubtitles) {
    issues.push({
      level: 'A',
      criterion: '1.2.2',
      message: '视频缺少字幕轨道，听障用户无法获取音频信息'
    });
  }

  // 检查 video 元素的无障碍属性
  if (!videoElement.hasAttribute('aria-label') && !videoElement.hasAttribute('title')) {
    issues.push({
      level: 'A',
      criterion: '1.2.1',
      message: 'video 元素缺少 aria-label 或 title 属性'
    });
  }

  // 检查是否有键盘可访问的控件
  const controls = videoElement.querySelectorAll('button, [role="button"]');
  controls.forEach((ctrl, index) => {
    if (!ctrl.hasAttribute('aria-label') && !ctrl.textContent.trim()) {
      issues.push({
        level: 'A',
        criterion: '4.1.2',
        message: `控件 ${index + 1} 缺少可访问的名称`
      });
    }
  });

  return {
    compliant: issues.length === 0,
    issues,
    summary: issues.length === 0
      ? '视频播放器符合 WCAG 2.1 AA 标准'
      : `发现 ${issues.length} 个无障碍问题`
  };
}
```

## 常见误区

### 误区一：自动生成的字幕足够用了

语音识别技术虽然进步很大，但仍存在明显局限：专业术语识别率低、多人对话区分困难、背景噪声干扰严重。自动生成的字幕需要人工校对才能达到无障碍标准。WCAG 2.1 AA 级要求字幕准确率达到 99% 以上。

### 误区二：字幕只是给听障用户用的

字幕的实际使用场景远超预期。在嘈杂的公交车上、安静的图书馆里、非母语学习场景中，字幕都是必需品。Netflix 的数据显示，超过 80% 的用户在某些场景下会使用字幕。

### 误区三：无障碍是上线前的最后一步

无障碍设计如果留到最后才做，成本会成倍增加。字幕需要内容团队配合、键盘导航需要重构交互逻辑、颜色对比需要重新设计视觉方案。正确做法是在项目初期就将无障碍纳入设计规范。

### 误区四：加了 ARIA 属性就无障碍了

ARIA 属性是补充手段，不是替代方案。语义化 HTML（如 `<button>` 而非 `<div onclick>`）永远是首选。ARIA 属性使用不当反而会制造新的无障碍问题。

## 小结

本课系统讲解了音视频无障碍设计的核心内容：

- **字幕规范**：封闭字幕、开放字幕、音频描述各有用途，需要根据场景选择
- **WebVTT 实现**：使用 `<track>` 元素和 TextTrack API 实现多语言字幕
- **手语视频**：画中画窗口为听障用户提供手语翻译
- **音频描述**：通过语音合成描述视觉内容，帮助视障用户
- **键盘导航**：完整的键盘控制方案，确保无鼠标可用
- **WCAG 合规**：遵循 1.2 时间媒体指南，从项目初期融入无障碍设计

无障碍设计的核心原则是**包容性**——好的无障碍设计让所有用户受益，而非仅仅满足合规要求。

## 参考答案

### 练习一

**思路**：使用 `<track>` 元素添加多语言字幕，通过 TextTrack API 控制显示，使用 CSS `::cue` 伪元素自定义样式。

**答案**：

```html
<!DOCTYPE html>
<html lang="zh">
<head>
  <meta charset="UTF-8">
  <title>无障碍视频播放器</title>
  <style>
    .player-wrapper { position: relative; max-width: 800px; margin: 0 auto; }
    video { width: 100%; }

    /* 字幕样式 */
    video::cue {
      background-color: rgba(0, 0, 0, 0.85);
      color: #ffffff;
      font-size: 20px;
      font-family: 'Microsoft YaHei', sans-serif;
    }

    /* 说话人标识 */
    video::cue(v[voice="旁白"]) { color: #7ec8e3; }
    video::cue(v[voice="角色A"]) { color: #98fb98; }

    /* 字幕控制面板 */
    .caption-controls {
      padding: 10px;
      background: #f5f5f5;
      display: flex;
      gap: 10px;
      align-items: center;
      flex-wrap: wrap;
    }
    .caption-controls label { font-weight: bold; }
    .caption-controls select, .caption-controls input {
      padding: 5px;
      border: 1px solid #ccc;
      border-radius: 4px;
    }
  </style>
</head>
<body>
  <div class="player-wrapper">
    <video id="video" controls crossorigin="anonymous">
      <source src="demo.mp4" type="video/mp4">
      <track kind="subtitles" src="subs/zh.vtt" srclang="zh" label="中文" default>
      <track kind="subtitles" src="subs/en.vtt" srclang="en" label="English">
      <track kind="captions" src="caps/zh-cc.vtt" srclang="zh" label="中文（无障碍）">
    </video>

    <div class="caption-controls">
      <label for="lang-select">字幕语言：</label>
      <select id="lang-select">
        <option value="off">关闭</option>
      </select>

      <label for="font-size">字号：</label>
      <input id="font-size" type="range" min="14" max="32" value="20">

      <label for="bg-opacity">背景透明度：</label>
      <input id="bg-opacity" type="range" min="0" max="100" value="85">
    </div>
  </div>

  <script>
    const video = document.getElementById('video');
    const langSelect = document.getElementById('lang-select');
    const fontSizeInput = document.getElementById('font-size');
    const bgOpacityInput = document.getElementById('bg-opacity');

    // 初始化字幕选项
    function initTrackOptions() {
      const tracks = video.textTracks;
      for (let i = 0; i < tracks.length; i++) {
        const option = document.createElement('option');
        option.value = i;
        option.textContent = tracks[i].label + (tracks[i].kind === 'captions' ? ' (CC)' : '');
        langSelect.appendChild(option);
      }
    }

    // 切换字幕
    langSelect.addEventListener('change', () => {
      const selectedIndex = parseInt(langSelect.value);

      for (let i = 0; i < video.textTracks.length; i++) {
        video.textTracks[i].mode = i === selectedIndex ? 'showing' : 'disabled';
      }
    });

    // 调整字号
    fontSizeInput.addEventListener('input', () => {
      const size = fontSizeInput.value;
      document.documentElement.style.setProperty('--cue-font-size', `${size}px`);
      // 更新所有活跃轨道的样式
      const style = document.createElement('style');
      style.textContent = `video::cue { font-size: ${size}px !important; }`;
      document.head.appendChild(style);
    });

    // 调整背景透明度
    bgOpacityInput.addEventListener('input', () => {
      const opacity = bgOpacityInput.value / 100;
      const style = document.createElement('style');
      style.textContent = `video::cue { background-color: rgba(0,0,0,${opacity}) !important; }`;
      document.head.appendChild(style);
    });

    // 快捷键 C 切换字幕
    document.addEventListener('keydown', (e) => {
      if (e.code === 'KeyC' && e.target.tagName !== 'INPUT') {
        const currentIndex = parseInt(langSelect.value);
        const nextIndex = (currentIndex + 1) % langSelect.options.length;
        langSelect.value = nextIndex;
        langSelect.dispatchEvent(new Event('change'));
      }
    });

    initTrackOptions();
  </script>
</body>
</html>
```

**要点**：
- `<track kind="captions">` 包含环境音描述，`kind="subtitles"` 只有对白
- `::cue` 伪元素可以自定义字幕样式，支持按说话人着色
- 快捷键 C 循环切换字幕语言，方便用户操作

### 练习二

**思路**：使用 `tabindex` 让播放器容器可聚焦，监听 `keydown` 事件实现所有控制功能，使用 `aria-live` 区域向屏幕阅读器反馈操作结果。

**答案**：

```html
<!DOCTYPE html>
<html lang="zh">
<head>
  <meta charset="UTF-8">
  <title>键盘无障碍播放器</title>
  <style>
    .player {

    </div>

  </div>

    <h3>键盘快捷键</h3>
    <table>
      <tr><td><kbd>空格</kbd></td><td>播放 / 暂停</td></tr>
      <tr><td><kbd>←</kbd> <kbd>→</kbd></td><td>后退 / 前进 10 秒</td></tr>
      <tr><td><kbd>↑</kbd> <kbd>↓</kbd></td><td>增大 / 减小音量</td></tr>
      <tr><td><kbd>M</kbd></td><td>静音</td></tr>
      <tr><td><kbd>F</kbd></td><td>全屏</td></tr>

      <tr><td><kbd>Home</kbd> <kbd>End</kbd></td><td>跳到开头 / 结尾</td></tr>
      <tr><td><kbd>H</kbd></td><td>显示 / 隐藏快捷键帮助</td></tr>
    </table>
  </div>

  <script>
    const video = document.getElementById('video');
    const feedback = document.getElementById('sr-feedback');
    const btnPlay = document.getElementById('btn-play');
    const btnMute = document.getElementById('btn-mute');

    function announce(msg) {
      feedback.textContent = '';
      requestAnimationFrame(() => { feedback.textContent = msg; });
    }

    function formatTime(seconds) {
      const m = Math.floor(seconds / 60);
      const s = Math.floor(seconds % 60);
      return `${m} 分 ${s} 秒`;
    }

    document.getElementById('player').addEventListener('keydown', (e) => {
      switch (e.code) {
        case 'Space':
          e.preventDefault();
          video.paused ? video.play() : video.pause();
          announce(video.paused ? '已暂停' : '正在播放');
          break;
        case 'ArrowLeft':
          e.preventDefault();
          video.currentTime = Math.max(0, video.currentTime - 10);
          announce(`后退到 ${formatTime(video.currentTime)}`);
          break;
        case 'ArrowRight':
          e.preventDefault();
          video.currentTime = Math.min(video.duration, video.currentTime + 10);
          announce(`前进到 ${formatTime(video.currentTime)}`);
          break;
        case 'ArrowUp':
          e.preventDefault();
          video.volume = Math.min(1, video.volume + 0.1);
          announce(`音量 ${Math.round(video.volume * 100)}%`);
          break;
        case 'ArrowDown':
          e.preventDefault();
          video.volume = Math.max(0, video.volume - 0.1);
          announce(`音量 ${Math.round(video.volume * 100)}%`);
          break;
        case 'KeyM':
          video.muted = !video.muted;
          announce(video.muted ? '已静音' : '已取消静音');
          break;
        case 'KeyF':
          document.fullscreenElement
            ? document.exitFullscreen()
            : video.requestFullscreen();
          announce(document.fullscreenElement ? '进入全屏' : '退出全屏');
          break;
        case 'KeyC':
          const tracks = video.textTracks;
          let showing = false;
          for (let i = 0; i < tracks.length; i++) {
            if (tracks[i].mode === 'showing') { showing = true; tracks[i].mode = 'disabled'; }
          }
          if (!showing && tracks.length > 0) tracks[0].mode = 'showing';
          announce(showing ? '字幕已关闭' : '字幕已开启');
          break;
        case 'Home':
          video.currentTime = 0;
          announce('跳到开头');
          break;
        case 'End':
          video.currentTime = video.duration;
          announce('跳到结尾');
          break;
        case 'KeyH':
          const help = document.getElementById('help');
          help.style.display = help.style.display === 'none' ? 'block' : 'none';
          break;
      }
    });

    // 同步按钮状态
    video.addEventListener('play', () => { btnPlay.textContent = '⏸ 暂停'; btnPlay.ariaLabel = '暂停'; });
    video.addEventListener('pause', () => { btnPlay.textContent = '▶ 播放'; btnPlay.ariaLabel = '播放'; });
    btnPlay.addEventListener('click', () => video.paused ? video.play() : video.pause());
    btnMute.addEventListener('click', () => { video.muted = !video.muted; });
  </script>
</body>
</html>
```

**要点**：
- 使用 `aria-live="polite"` 区域实时向屏幕阅读器反馈操作结果
- 每个按钮都有 `aria-label`，确保屏幕阅读器可以正确朗读
- 焦点样式（蓝色边框）确保键盘用户能看到当前焦点位置
- `H` 键可以显示/隐藏快捷键帮助，降低学习成本
