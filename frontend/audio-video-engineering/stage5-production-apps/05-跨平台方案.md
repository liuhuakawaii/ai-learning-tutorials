# 第五课：跨平台方案

## 场景引入

一个短视频团队需要在 iOS、Android 和 Web 三个平台上实现相同的视频播放和录制功能。原生开发意味着三套代码、三个团队、三倍的维护成本。React Native 和 Flutter 等跨平台框架承诺"一套代码，多端运行"，但音视频功能恰恰是跨平台差异最大的领域——iOS 的硬件编码器和 Android 的完全不同，Web 端的 MediaStream API 在原生环境中根本不存在。

如何在跨平台框架中正确处理音视频功能？如何在代码复用和平台特性之间找到平衡？本课将系统讲解跨平台音视频开发的核心方案。

## 学习目标

完成本课学习后，你将能够：

1. 在 React Native 中实现视频播放和录制
2. 在 Flutter 中使用 video_player 和 camera 插件
3. 设计跨平台的音视频抽象层
4. 处理不同平台的编解码器差异
5. 实现平台特有的功能（画中画、后台播放）

```javascript
import React, { useRef, useState } from 'react';
import { View, StyleSheet, TouchableOpacity, Text } from 'react-native';
import Video from 'react-native-video';

const VideoPlayer = ({ source }) => {
  const videoRef = useRef(null);
  const [paused, setPaused] = useState(false);
  const [currentTime, setCurrentTime] = useState(0);
  const [duration, setDuration] = useState(0);
  const [resizeMode, setResizeMode] = useState('contain');

    setDuration(data.duration);
    console.log('视频加载完成:', {
      duration: data.duration,
      width: data.naturalSize.width,
      height: data.naturalSize.height
    });
  };

  const onProgress = (data) => {
    setCurrentTime(data.currentTime);
  };

  const onEnd = () => {
    setPaused(true);
    setCurrentTime(0);
    videoRef.current?.seek(0);
  };

  const onError = (error) => {
    console.error('视频播放错误:', error);
  };

  const togglePlay = () => setPaused(!paused);

  const seekTo = (seconds) => {
    videoRef.current?.seek(seconds);
  };

  return (
    <View style={styles.container}>
      <Video
        ref={videoRef}
        source={{ uri: source }}
        style={styles.video}
        paused={paused}
        resizeMode={resizeMode}
        repeat={false}
        playInBackground={false}
        playWhenInactive={false}
        ignoreSilentSwitch="ignore"
        onLoad={onLoad}
        onProgress={onProgress}
        onEnd={onEnd}
        onError={onError}
        progressUpdateInterval={1000}
      />

      <View style={styles.controls}>
        <TouchableOpacity onPress={togglePlay}>
          <Text style={styles.controlText}>{paused ? '▶' : '⏸'}</Text>
        </TouchableOpacity>

        <Text style={styles.timeText}>
          {formatTime(currentTime)} / {formatTime(duration)}
        </Text>

        <TouchableOpacity onPress={() => setResizeMode(
          resizeMode === 'contain' ? 'cover' : 'contain'
        )}>
          <Text style={styles.controlText}>📐</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
};

const formatTime = (seconds) => {
  const m = Math.floor(seconds / 60);
  const s = Math.floor(seconds % 60);
  return `${m}:${s.toString().padStart(2, '0')}`;
};

const styles = StyleSheet.create({

    backgroundColor: 'rgba(0,0,0,0.7)'
```

### expo-av

```javascript
// 使用 expo-av 录制和播放音频
import { Audio } from 'expo-av';
import { useState, useEffect } from 'react';

const AudioRecorder = () => {
  const [recording, setRecording] = useState(null);
  const [sound, setSound] = useState(null);
  const [isRecording, setIsRecording] = useState(false);
  const [recordings, setRecordings] = useState([]);

  useEffect(() => {
    return () => {
      if (sound) sound.unloadAsync();
    };
  }, [sound]);

  const startRecording = async () => {
    try {
      const permission = await Audio.requestPermissionsAsync();
      if (!permission.granted) {
        console.warn('录音权限被拒绝');
        return;
      }

      await Audio.setAudioModeAsync({
        allowsRecordingIOS: true,
        playsInSilentModeIOS: true,
      });

      const { recording } = await Audio.Recording.createAsync(
        Audio.RecordingOptionsPresets.HIGH_QUALITY
      );

      setRecording(recording);
      setIsRecording(true);
    } catch (error) {
      console.error('开始录音失败:', error);
    }
  };

  const stopRecording = async () => {
    if (!recording) return;

    setIsRecording(false);
    await recording.stopAndUnloadAsync();
    await Audio.setAudioModeAsync({ allowsRecordingIOS: false });

    const uri = recording.getURI();
    setRecordings(prev => [...prev, { uri, duration: 0 }]);
    setRecording(null);
  };

  const playRecording = async (uri) => {
    if (sound) await sound.unloadAsync();

    const { sound: newSound } = await Audio.Sound.createAsync(
      { uri },
      { shouldPlay: true }
    );
    setSound(newSound);
  };

  return { startRecording, stopRecording, playRecording, isRecording, recordings };
};
```

```dart
      return const Center(child: CircularProgressIndicator());

      backgroundColor: Colors.black,
      body: Center(
        child: AspectRatio(
          aspectRatio: _controller.value.aspectRatio,
          child: Stack(
            alignment: Alignment.center,
            children: [
              VideoPlayer(_controller),
              _buildControls(),
            ],
          ),
        ),
      ),
    );
  }

    return GestureDetector(
      onTap: () {
        setState(() {
          _controller.value.isPlaying
              ? _controller.pause()
              : _controller.play();
        });
      },
      child: Container(
        color: Colors.transparent,
        child: Center(
          child: Icon(
            _controller.value.isPlaying ? Icons.pause : Icons.play_arrow,
            color: Colors.white,
            size: 64,
          ),
        ),
      ),
    );
  }

  @override
  void dispose() {
    _controller.removeListener(_onProgress);
    _controller.dispose();
    super.dispose();
  }
}
```

```typescript
interface VideoProcessor {
  // 压缩视频
  compress(inputPath: string, options: CompressOptions): Promise<string>;

  // 裁剪视频
  trim(inputPath: string, startTime: number, endTime: number): Promise<string>;

  addWatermark(inputPath: string, watermark: WatermarkConfig): Promise<string>;

  fps?: number;

  height: number;

  fps: number;

class WebVideoProcessor implements VideoProcessor {
  async compress(inputPath: string, options: CompressOptions): Promise<string> {
    const { createFFmpeg, fetchFile } = await import('@ffmpeg/ffmpeg');
    const ffmpeg = createFFmpeg({ log: true });
    await ffmpeg.load();

    ffmpeg.FS('writeFile', 'input.mp4', await fetchFile(inputPath));

    const resolutionMap: Record<string, string> = {
      '720p': '1280:720',
      '1080p': '1920:1080',
      '480p': '854:480'
    };

    const resolution = resolutionMap[options.resolution || '720p'];

    await ffmpeg.run(
      '-i', 'input.mp4',
      '-vf', `scale=${resolution}`,
      '-c:v', 'libx264',
      '-preset', 'medium',
      '-crf', options.quality === 'high' ? '18' : options.quality === 'medium' ? '23' : '28',
      'output.mp4'
    );

    const data = ffmpeg.FS('readFile', 'output.mp4');
    const blob = new Blob([data.buffer], { type: 'video/mp4' });

  }
```

### 画中画支持

```javascript
// 跨平台画中画实现
class PictureInPictureManager {
  constructor(platform) {
    this.platform = platform;
    this.isSupported = false;
    this.isActive = false;
  }

  async checkSupport() {
    switch (this.platform) {
      case 'web':
        this.isSupported = 'pictureInPictureEnabled' in document;
        break;
      case 'ios':
        // AVPictureInPictureController.isPictureInPictureSupported
        this.isSupported = await NativeModules.PiPManager.isSupported();
        break;
      case 'android':
        // PictureInPictureParams.Builder
        this.isSupported = await NativeModules.PiPManager.isSupported();
        break;
    }
    return this.isSupported;
  }

  async enter(videoElement) {
    if (!this.isSupported) {
      console.warn('当前平台不支持画中画');
      return false;
    }

    try {
      switch (this.platform) {
        case 'web':
          await videoElement.requestPictureInPicture();
          break;
        case 'ios':
        case 'android':
          await NativeModules.PiPManager.enter();
          break;
      }
      this.isActive = true;
      return true;
    } catch (error) {
      console.error('进入画中画失败:', error);
      return false;
    }
  }

  async exit() {
    switch (this.platform) {
      case 'web':
        await document.exitPictureInPicture();
        break;
      case 'ios':
      case 'android':
        await NativeModules.PiPManager.exit();
        break;
    }
    this.isActive = false;
  }
}
```

## 常见误区

### 误区一：跨平台意味着行为完全一致

不同平台的音视频能力差异巨大。iOS 的硬件编码质量通常优于 Android 的低端设备，Web 端的编码能力最弱。UI 交互也应遵循各平台的设计规范——iOS 用户期望滑动调音量，Android 用户习惯点击按钮。"一致"指的是功能逻辑一致，而非界面和行为完全相同。

### 误区二：一套代码可以覆盖所有场景

对于简单的视频播放，跨平台方案确实可以大幅复用代码。但对于高级功能（硬件编码、DRM、画中画），每个平台都需要原生实现。正确的做法是抽象层定义接口，平台层实现细节。

### 误区三：跨平台框架的性能可以媲美原生

跨平台框架在 UI 层的性能已经非常接近原生，但音视频处理涉及大量底层硬件交互，性能差距仍然存在。对于性能敏感的场景（4K 视频编辑、实时滤镜），原生实现仍然是更好的选择。

### 误区四：Web 端可以和原生端共用所有代码

Web 端的音视频 API 与原生端有本质区别。MediaStream、WebAudio、MediaRecorder 等 API 在原生端没有对应物。业务逻辑可以共享，但音视频采集和处理层必须分别实现。

## 小结

本课系统讲解了跨平台音视频开发的核心方案：

- **React Native**：通过 react-native-video 和 expo-av 实现视频播放和音频录制
- **Flutter**：使用 video_player 和 camera 插件快速集成音视频功能
- **原生桥接**：通过 Native Modules 和 Platform Channel 调用平台特有能力
- **统一抽象**：设计跨平台接口层，隔离平台差异
- **平台差异**：编解码器支持、画中画、后台播放等功能各平台实现不同

跨平台开发的核心原则是**抽象共享逻辑，平台实现细节**——业务逻辑尽量复用，音视频处理层根据平台特性分别实现。

## 练习

### 练习一：构建跨平台视频播放器

使用 React Native 或 Flutter 实现一个跨平台视频播放器，支持播放控制、进度条、全屏切换，并在 iOS 和 Android 上测试。

### 练习二：实现画中画功能

在跨平台框架中实现画中画（PiP）功能，处理各平台的差异（Web 使用 PIP API，iOS/Android 使用原生模块）。

## 参考答案

### 练习一

**思路**：使用 react-native-video 封装一个带完整控件的播放器组件，通过 Platform API 处理平台差异。

**答案**：

```javascript
import React, { useRef, useState, useEffect } from 'react';
import {
  View, StyleSheet, TouchableOpacity, Text,
  Platform, Dimensions, StatusBar
} from 'react-native';
import Video from 'react-native-video';
import Orientation from 'react-native-orientation-locker';

const UniversalVideoPlayer = ({ source, title }) => {
  const videoRef = useRef(null);
  const [state, setState] = useState({
    paused: true,
    currentTime: 0,
    duration: 0,
    isFullscreen: false,
    isLoading: true,
    error: null
  });

  const onLoad = (data) => {
    setState(prev => ({
      ...prev,
      duration: data.duration,
      isLoading: false
    }));
  };

  const onProgress = (data) => {
    setState(prev => ({ ...prev, currentTime: data.currentTime }));
  };

  const onError = (error) => {
    setState(prev => ({ ...prev, error: error.error.message, isLoading: false }));
  };

  const togglePlay = () => {
    setState(prev => ({ ...prev, paused: !prev.paused }));
  };

  const seekTo = (seconds) => {
    videoRef.current?.seek(Math.max(0, Math.min(seconds, state.duration)));
  };

  const toggleFullscreen = () => {
    if (state.isFullscreen) {
      Orientation.lockToPortrait();
      StatusBar.setHidden(false);
    } else {
      Orientation.lockToLandscape();
      StatusBar.setHidden(true);
    }
    setState(prev => ({ ...prev, isFullscreen: !prev.isFullscreen }));
  };

  const skipForward = () => seekTo(state.currentTime + 10);
  const skipBackward = () => seekTo(state.currentTime - 10);

  const formatTime = (seconds) => {
    const m = Math.floor(seconds / 60);
    const s = Math.floor(seconds % 60);
    return `${m}:${s.toString().padStart(2, '0')}`;
  };

  const progress = state.duration > 0
    ? state.currentTime / state.duration
    : 0;

  return (
    <View style={[
      styles.container,
      state.isFullscreen && styles.fullscreenContainer
    ]}>
      <Video
        ref={videoRef}
        source={{ uri: source }}
        style={styles.video}
        paused={state.paused}
        resizeMode="contain"
        onLoad={onLoad}
        onProgress={onProgress}
        onError={onError}
        progressUpdateInterval={500}
        playInBackground={Platform.OS === 'ios'}
      />

      {/* 标题栏 */}
      {state.isFullscreen && (
        <View style={styles.titleBar}>
          <Text style={styles.titleText}>{title}</Text>
        </View>
      )}

      {/* 加载指示器 */}
      {state.isLoading && (
        <View style={styles.loadingOverlay}>
          <Text style={styles.loadingText}>加载中...</Text>
        </View>
      )}

      {/* 错误提示 */}
      {state.error && (
        <View style={styles.errorOverlay}>
          <Text style={styles.errorText}>播放失败: {state.error}</Text>
        </View>
      )}

      {/* 控制栏 */}
      <View style={styles.controls}>
        <TouchableOpacity onPress={skipBackward} style={styles.controlButton}>
          <Text style={styles.controlText}>⏪</Text>
        </TouchableOpacity>

        <TouchableOpacity onPress={togglePlay} style={styles.playButton}>
          <Text style={styles.playText}>{state.paused ? '▶' : '⏸'}</Text>
        </TouchableOpacity>

        <TouchableOpacity onPress={skipForward} style={styles.controlButton}>
          <Text style={styles.controlText}>⏩</Text>
        </TouchableOpacity>
      </View>

      {/* 进度条 */}
      <View style={styles.progressBar}>
        <Text style={styles.timeText}>{formatTime(state.currentTime)}</Text>
        <View style={styles.track}>
          <View style={[styles.progress, { width: `${progress * 100}%` }]} />
          <TouchableOpacity
            style={[styles.thumb, { left: `${progress * 100}%` }]}
            onPress={() => {}}
          />
        </View>
        <Text style={styles.timeText}>{formatTime(state.duration)}</Text>
        <TouchableOpacity onPress={toggleFullscreen}>
          <Text style={styles.controlText}>
            {state.isFullscreen ? '⊡' : '⛶'}
          </Text>
        </TouchableOpacity>
      </View>
    </View>
  );
};

const styles = StyleSheet.create({
  container: { backgroundColor: '#000', aspectRatio: 16 / 9 },
  fullscreenContainer: {
    position: 'absolute', top: 0, left: 0,
    width: Dimensions.get('window').width,
    height: Dimensions.get('window').height,
    zIndex: 999
  },
  video: { flex: 1 },
  titleBar: {
    position: 'absolute', top: 10, left: 10, right: 10
  },
  titleText: { color: '#fff', fontSize: 16, fontWeight: 'bold' },
  loadingOverlay: {
    ...StyleSheet.absoluteFillObject,
    justifyContent: 'center', alignItems: 'center'
  },
  loadingText: { color: '#fff', fontSize: 16 },
  errorOverlay: {
    ...StyleSheet.absoluteFillObject,
    justifyContent: 'center', alignItems: 'center',
    backgroundColor: 'rgba(0,0,0,0.7)'
  },
  errorText: { color: '#ff4444', fontSize: 14 },
  controls: {
    position: 'absolute', bottom: 40, left: 0, right: 0,
    flexDirection: 'row', justifyContent: 'center',
    alignItems: 'center', gap: 30
  },
  controlButton: { padding: 10 },
  playButton: { padding: 15 },
  controlText: { color: '#fff', fontSize: 24 },
  playText: { color: '#fff', fontSize: 32 },
  progressBar: {
    position: 'absolute', bottom: 10, left: 10, right: 10,
    flexDirection: 'row', alignItems: 'center', gap: 8
  },
  timeText: { color: '#fff', fontSize: 12, minWidth: 35 },
  track: { flex: 1, height: 4, backgroundColor: '#555', borderRadius: 2 },
  progress: { height: '100%', backgroundColor: '#e74c3c', borderRadius: 2 },
  thumb: {
    position: 'absolute', top: -6, width: 16, height: 16,
    backgroundColor: '#e74c3c', borderRadius: 8
  }
});

export default UniversalVideoPlayer;
```

**要点**：
- 使用 `react-native-video` 处理核心播放功能
- 通过 `react-native-orientation-locker` 处理屏幕旋转
- 平台差异通过 `Platform.OS` 判断，iOS 允许后台播放
- 全屏模式需要隐藏状态栏并锁定横屏

### 练习二

**思路**：Web 端使用 `requestPictureInPicture` API，原生端通过桥接调用系统的 PiP 功能。

**答案**：

```javascript
// 跨平台画中画管理器
import { Platform, NativeModules } from 'react-native';

const { PiPModule } = NativeModules;

class CrossPlatformPiP {
  static async isSupported() {
    if (Platform.OS === 'web') {
      return document.pictureInPictureEnabled;
    }
    return PiPModule?.isSupported() ?? false;
  }

  static async enter(videoRef) {
    try {
      if (Platform.OS === 'web') {
        await videoRef.current.requestPictureInPicture();
      } else if (Platform.OS === 'ios') {
        await PiPModule.startPiP();
      } else if (Platform.OS === 'android') {
        // Android 需要在 AndroidManifest 中配置
        // android:supportsPictureInPicture="true"
        await PiPModule.enterPiP({
          width: 16,
          height: 9,
          autoEnter: true
        });
      }
      return true;
    } catch (error) {
      console.error('进入画中画失败:', error);
      return false;
    }
  }

  static async exit() {
    try {
      if (Platform.OS === 'web') {
        await document.exitPictureInPicture();
      } else {
        await PiPModule.exitPiP();
      }
      return true;
    } catch (error) {
      console.error('退出画中画失败:', error);
      return false;
    }
  }

  static onPiPChange(callback) {
    if (Platform.OS === 'web') {
      document.addEventListener('enterpictureinpicture', () => callback(true));
      document.addEventListener('leavepictureinpicture', () => callback(false));
    } else {
      // 原生端通过事件订阅
      PiPModule?.addListener('onPiPChanged', (event) => {
        callback(event.isActive);
      });
    }
  }
}

// 使用示例
const PiPToggle = ({ videoRef }) => {
  const [isPiP, setIsPiP] = useState(false);

  useEffect(() => {
    CrossPlatformPiP.onPiPChange(setIsPiP);
  }, []);

  const togglePiP = async () => {
    if (isPiP) {
      await CrossPlatformPiP.exit();
    } else {
      await CrossPlatformPiP.enter(videoRef);
    }
  };

  return (
    <TouchableOpacity onPress={togglePiP}>
      <Text>{isPiP ? '退出画中画' : '画中画'}</Text>
    </TouchableOpacity>
  );
};
```

**要点**：
- Android 需要在 `AndroidManifest.xml` 中声明 `supportsPictureInPicture`
- iOS 的 PiP 需要在 `Info.plist` 中配置 `UIBackgroundModes`
- Web 端的 PiP 需要用户手势触发，不能自动进入
