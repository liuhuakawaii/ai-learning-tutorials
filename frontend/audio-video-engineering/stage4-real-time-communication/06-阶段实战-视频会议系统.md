# 阶段实战：视频会议系统

## 场景引入

前面五节课我们学习了 WebRTC 的核心概念、点对点通信、多方会议架构、实时音视频处理和数据通道。现在是时候把它们组合起来，构建一个完整的视频会议系统。

我们要实现的是一个"Mini Zoom"：支持多人视频通话、屏幕共享、实时聊天、参与者管理。这个项目会用到 SFU 架构，因为我们需要支持多人场景。

## 学习目标

1. 设计视频会议系统的整体架构
2. 实现基于 WebSocket 的信令服务器
3. 实现 SFU 客户端逻辑
4. 集成屏幕共享和聊天功能
5. 处理参与者生命周期管理

## 信令服务器实现

### 服务器核心

```javascript
// signaling-server.js
const WebSocket = require('ws');

class SignalingServer {
  constructor(port) {
    this.wss = new WebSocket.Server({ port });
    this.rooms = new Map();
    this.clients = new Map(); // ws -> { userId, roomId }

    this.wss.on('connection', (ws) => this.handleConnection(ws));
    console.log(`信令服务器运行在端口 ${port}`);
  }

  handleConnection(ws) {
    ws.on('message', (data) => {
      try {
        const message = JSON.parse(data);
        this.handleMessage(ws, message);
      } catch (e) {
        ws.send(JSON.stringify({ type: 'error', message: '消息格式错误' }));
      }
    });

    ws.on('close', () => {
      const client = this.clients.get(ws);
      if (client) {
        this.handleLeave(client.roomId, client.userId);
        this.clients.delete(ws);
      }
    });
  }

  handleMessage(ws, message) {
    switch (message.type) {
      case 'join':
        this.handleJoin(ws, message.roomId, message.userId, message.displayName);
        break;
      case 'leave':
        const client = this.clients.get(ws);
        if (client) this.handleLeave(client.roomId, client.userId);
        break;
      case 'offer':
      case 'answer':
      case 'candidate':
        this.forwardToRoom(ws, message);
        break;
      case 'chat':
        this.broadcastToRoom(ws, message);
        break;
      case 'screen-share-started':
      case 'screen-share-stopped':
        this.broadcastToRoom(ws, message);
        break;
    }
  }

  handleJoin(ws, roomId, userId, displayName) {
    if (!this.rooms.has(roomId)) {
      this.rooms.set(roomId, new Map());
    }

    const room = this.rooms.get(roomId);

    // 通知新用户已有参与者
    const participants = [];
    for (const [id, info] of room) {
      participants.push({ userId: id, displayName: info.displayName });
    }

    ws.send(JSON.stringify({
      type: 'room-joined',
      participants,
      roomId
    }));

    // 添加到房间
    room.set(userId, { ws, displayName });
    this.clients.set(ws, { userId, roomId });

    // 通知其他人
    this.broadcastToRoom(ws, {
      type: 'participant-joined',
      userId,
      displayName
    }, userId);
  }

  handleLeave(roomId, userId) {
    const room = this.rooms.get(roomId);
    if (!room) return;

    room.delete(userId);

    if (room.size === 0) {
      this.rooms.delete(roomId);
    } else {
      this.broadcastToRoom(null, {
        type: 'participant-left',
        userId
      }, userId, roomId);
    }
  }

  forwardToRoom(ws, message) {
    const client = this.clients.get(ws);
    if (!client) return;

    const room = this.rooms.get(client.roomId);
    if (!room) return;

    // 转发给目标用户
    const target = room.get(message.targetId);
    if (target && target.ws.readyState === WebSocket.OPEN) {
      target.ws.send(JSON.stringify({
        ...message,
        senderId: client.userId
      }));
    }
  }

  broadcastToRoom(ws, message, excludeUserId, roomId) {
    const client = ws ? this.clients.get(ws) : null;
    const rid = roomId || client?.roomId;
    if (!rid) return;

    const room = this.rooms.get(rid);
    if (!room) return;

    for (const [userId, info] of room) {
      if (userId !== excludeUserId && info.ws.readyState === WebSocket.OPEN) {
        info.ws.send(JSON.stringify(message));
      }
    }
  }
}

new SignalingServer(8080);
```

## 客户端实现

### 主应用类

```javascript
// conference-app.js
class ConferenceApp {
  constructor(signalingUrl) {
    this.signalingUrl = signalingUrl;
    this.ws = null;
    this.userId = 'user-' + Math.random().toString(36).substr(2, 8);
    this.displayName = '';
    this.roomId = null;

    this.localStream = null;
    this.screenStream = null;
    this.isScreenSharing = false;

    this.participants = new Map(); // userId -> { pc, stream, displayName }
    this.chatMessages = [];

    this.pcConfig = {
      iceServers: [
        { urls: 'stun:stun.l.google.com:19302' }
      ]
    };
  }

  // === 连接管理 ===

  connect() {
    return new Promise((resolve, reject) => {
      this.ws = new WebSocket(this.signalingUrl);

      this.ws.onopen = () => {
        console.log('信令连接已建立');
        resolve();
      };

      this.ws.onmessage = (event) => {
        const message = JSON.parse(event.data);
        this.handleSignalingMessage(message);
      };

      this.ws.onerror = (error) => {
        console.error('信令连接错误:', error);
        reject(error);
      };

      this.ws.onclose = () => {
        console.log('信令连接已断开');
        this.handleDisconnect();
      };
    });
  }

  // === 房间管理 ===

  async joinRoom(roomId, displayName) {
    this.roomId = roomId;
    this.displayName = displayName;

    // 获取本地媒体
    this.localStream = await navigator.mediaDevices.getUserMedia({
      video: { width: { ideal: 1280 }, height: { ideal: 720 } },
      audio: { echoCancellation: true, noiseSuppression: true }
    });

    // 显示本地视频
    this.showLocalVideo();

    // 发送加入请求
    this.send({
      type: 'join',
      roomId,
      userId: this.userId,
      displayName
    });
  }

  leaveRoom() {
    // 停止所有轨道
    this.localStream?.getTracks().forEach(t => t.stop());
    this.screenStream?.getTracks().forEach(t => t.stop());

    // 关闭所有 PeerConnection
    for (const [userId, participant] of this.participants) {
      participant.pc?.close();
    }
    this.participants.clear();

    // 发送离开消息
    this.send({ type: 'leave' });

    this.roomId = null;
    this.onParticipantsUpdate?.([]);
  }

  // === 信令消息处理 ===

  handleSignalingMessage(message) {
    switch (message.type) {
      case 'room-joined':
        this.onRoomJoined(message);
        break;
      case 'participant-joined':
        this.onParticipantJoined(message);
        break;
      case 'participant-left':
        this.onParticipantLeft(message);
        break;
      case 'offer':
        this.handleOffer(message);
        break;
      case 'answer':
        this.handleAnswer(message);
        break;
      case 'candidate':
        this.handleCandidate(message);
        break;
      case 'chat':
        this.handleChatMessage(message);
        break;
      case 'screen-share-started':
        this.onScreenShareStarted?.(message.userId);
        break;
      case 'screen-share-stopped':
        this.onScreenShareStopped?.(message.userId);
        break;
    }
  }

  async onRoomJoined(message) {
    console.log(`加入房间 ${message.roomId}，已有 ${message.participants.length} 人`);

    // 为每个已有参与者创建 PeerConnection
    for (const participant of message.participants) {
      await this.createPeerConnection(participant.userId, participant.displayName, true);
    }
  }

  async onParticipantJoined(message) {
    console.log(`${message.displayName} 加入了会议`);
    // 等待对方发起 offer
    this.participants.set(message.userId, {
      pc: null,
      stream: null,
      displayName: message.displayName
    });
    this.updateParticipantsUI();
  }

  onParticipantLeft(message) {
    console.log(`${message.userId} 离开了会议`);
    const participant = this.participants.get(message.userId);
    if (participant) {
      participant.pc?.close();
      this.participants.delete(message.userId);
      this.removeRemoteVideo(message.userId);
      this.updateParticipantsUI();
    }
  }

  // === PeerConnection 管理 ===

  async createPeerConnection(remoteUserId, displayName, isInitiator) {
    const pc = new RTCPeerConnection(this.pcConfig);

    // 添加本地轨道
    this.localStream.getTracks().forEach(track => {
      pc.addTrack(track, this.localStream);
    });

    // ICE 候选
    pc.onicecandidate = (event) => {
      if (event.candidate) {
        this.send({
          type: 'candidate',
          targetId: remoteUserId,
          candidate: event.candidate.toJSON()
        });
      }
    };

    // 收到远程轨道
    pc.ontrack = (event) => {
      const participant = this.participants.get(remoteUserId);
      if (participant) {
        participant.stream = event.streams[0];
        this.showRemoteVideo(remoteUserId, event.streams[0], displayName);
      }
    };

    // 连接状态
    pc.onconnectionstatechange = () => {
      console.log(`与 ${remoteUserId} 的连接状态:`, pc.connectionState);
      if (pc.connectionState === 'failed') {
        pc.restartIce();
      }
    };

    this.participants.set(remoteUserId, { pc, stream: null, displayName });

    // 如果是发起方，创建 Offer
    if (isInitiator) {
      const offer = await pc.createOffer();
      await pc.setLocalDescription(offer);
      this.send({
        type: 'offer',
        targetId: remoteUserId,
        sdp: offer.sdp
      });
    }

    return pc;
  }

  async handleOffer(message) {
    let participant = this.participants.get(message.senderId);

    if (!participant || !participant.pc) {
      await this.createPeerConnection(
        message.senderId,
        message.displayName || message.senderId,
        false
      );
      participant = this.participants.get(message.senderId);
    }

    const pc = participant.pc;
    await pc.setRemoteDescription({ type: 'offer', sdp: message.sdp });
    const answer = await pc.createAnswer();
    await pc.setLocalDescription(answer);

    this.send({
      type: 'answer',
      targetId: message.senderId,
      sdp: answer.sdp
    });
  }

  async handleAnswer(message) {
    const participant = this.participants.get(message.senderId);
    if (participant?.pc) {
      await participant.pc.setRemoteDescription({
        type: 'answer',
        sdp: message.sdp
      });
    }
  }

  async handleCandidate(message) {
    const participant = this.participants.get(message.senderId);
    if (participant?.pc) {
      await participant.pc.addIceCandidate(
        new RTCIceCandidate(message.candidate)
      );
    }
  }

  // === 媒体控制 ===

  toggleMute() {
    const audioTrack = this.localStream.getAudioTracks()[0];
    if (audioTrack) {
      audioTrack.enabled = !audioTrack.enabled;
      return !audioTrack.enabled;
    }
    return false;
  }

  toggleCamera() {
    const videoTrack = this.localStream.getVideoTracks()[0];
    if (videoTrack) {
      videoTrack.enabled = !videoTrack.enabled;
      return !videoTrack.enabled;
    }
    return false;
  }

  async startScreenShare() {
    try {
      this.screenStream = await navigator.mediaDevices.getDisplayMedia({
        video: { cursor: 'always' },
        audio: false
      });

      const screenTrack = this.screenStream.getVideoTracks()[0];

      screenTrack.onended = () => {
        this.stopScreenShare();
      };

      // 替换所有 PeerConnection 的视频轨道
      for (const [userId, participant] of this.participants) {
        if (participant.pc) {
          const sender = participant.pc.getSenders().find(s => s.track?.kind === 'video');
          await sender.replaceTrack(screenTrack);
        }
      }

      // 更新本地预览
      document.getElementById('localVideo').srcObject = this.screenStream;

      this.isScreenSharing = true;
      this.send({ type: 'screen-share-started' });
      this.onScreenShareStateChanged?.(true);

      return true;
    } catch (error) {
      if (error.name !== 'NotAllowedError') {
        console.error('屏幕共享失败:', error);
      }
      return false;
    }
  }

  async stopScreenShare() {
    if (!this.isScreenSharing) return;

    this.screenStream?.getTracks().forEach(t => t.stop());

    const cameraTrack = this.localStream.getVideoTracks()[0];

    for (const [userId, participant] of this.participants) {
      if (participant.pc) {
        const sender = participant.pc.getSenders().find(s => s.track?.kind === 'video');
        await sender.replaceTrack(cameraTrack);
      }
    }

    document.getElementById('localVideo').srcObject = this.localStream;

    this.isScreenSharing = false;
    this.screenStream = null;
    this.send({ type: 'screen-share-stopped' });
    this.onScreenShareStateChanged?.(false);
  }

  // === 聊天 ===

  sendChatMessage(text) {
    const message = {
      type: 'chat',
      userId: this.userId,
      displayName: this.displayName,
      text,
      timestamp: Date.now()
    };
    this.send(message);
    this.chatMessages.push(message);
    this.onChatMessage?.(message);
  }

  handleChatMessage(message) {
    this.chatMessages.push(message);
    this.onChatMessage?.(message);
  }

  // === UI 辅助 ===

  showLocalVideo() {
    const video = document.getElementById('localVideo');
    video.srcObject = this.localStream;
    video.muted = true;
    video.play();
  }

  showRemoteVideo(userId, stream, displayName) {
    let container = document.getElementById(`remote-${userId}`);

    if (!container) {
      container = document.createElement('div');
      container.id = `remote-${userId}`;
      container.className = 'video-container remote';
      container.innerHTML = `
        <video autoplay playsinline></video>
        <span class="name-label">${displayName || userId}</span>
      `;
      document.getElementById('videoGrid').appendChild(container);
    }

    container.querySelector('video').srcObject = stream;
    this.updateParticipantsUI();
  }

  removeRemoteVideo(userId) {
    const container = document.getElementById(`remote-${userId}`);
    if (container) container.remove();
  }

  updateParticipantsUI() {
    const list = Array.from(this.participants.entries()).map(([id, p]) => ({
      userId: id,
      displayName: p.displayName
    }));
    this.onParticipantsUpdate?.(list);
  }

  // === 工具方法 ===

  send(message) {
    if (this.ws?.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify(message));
    }
  }

  handleDisconnect() {
    for (const [userId, participant] of this.participants) {
      participant.pc?.close();
    }
    this.participants.clear();
    this.onDisconnected?.();
  }

  destroy() {
    this.leaveRoom();
    this.ws?.close();
  }
}
```

## 常见误区

### 误区一：不处理参与者断开

最常见的 bug：某个参与者关闭浏览器或网络断开，其他人看不到任何变化。必须在信令层检测连接断开，并通知所有参与者更新 UI 和清理 PeerConnection。

### 误区二：ICE 重启没有重新发送 Offer

当 ICE 连接失败时，`restartIce()` 只会重新收集候选。但如果你的信令已经断开，新的候选无法发送给对方。需要确保信令通道在 ICE 重启期间保持可用。

### 误区三：屏幕共享时没有同步状态

用户 A 开始屏幕共享后，如果直接关闭标签页，其他人不会收到 `screen-share-stopped` 事件。需要在 `beforeunload` 中主动通知：

```javascript
window.addEventListener('beforeunload', () => {
  if (app.isScreenSharing) {
    app.send({ type: 'screen-share-stopped' });
  }
  app.send({ type: 'leave' });
});
```

### 误区四：聊天消息没有持久化

DataChannel 的消息只在连接存在时有效。如果用户刷新页面，所有聊天记录都会丢失。需要在服务器端持久化消息，或者在新用户加入时从服务器拉取历史记录。

## 工程建议

1. **实现重连逻辑**。WebSocket 可能断开，需要自动重连：

```javascript
class ReconnectingWebSocket {
  constructor(url) {
    this.url = url;
    this.ws = null;
    this.reconnectDelay = 1000;
    this.maxReconnectDelay = 30000;
    this.connect();
  }

  connect() {
    this.ws = new WebSocket(this.url);

    this.ws.onopen = () => {
      this.reconnectDelay = 1000;
      this.onopen?.();
    };

    this.ws.onclose = () => {
      setTimeout(() => {
        this.reconnectDelay = Math.min(
          this.reconnectDelay * 2,
          this.maxReconnectDelay
        );
        this.connect();
      }, this.reconnectDelay);
    };
  }

  send(data) {
    if (this.ws.readyState === WebSocket.OPEN) {
      this.ws.send(data);
    }
  }
}
```

2. **添加连接质量监控**。定期采集 WebRTC 统计数据，显示给用户：

```javascript
async function getCallStats(pc) {
  const stats = await pc.getStats();
  const result = { video: {}, audio: {}, connection: {} };

  stats.forEach(report => {
    if (report.type === 'inbound-rtp' && report.kind === 'video') {
      result.video.inbound = {
        fps: report.framesPerSecond,
        width: report.frameWidth,
        height: report.frameHeight,
        packetsLost: report.packetsLost,
        jitter: report.jitter
      };
    }
    if (report.type === 'candidate-pair' && report.state === 'succeeded') {
      result.connection = {
        rtt: report.currentRoundTripTime,
        availableBandwidth: report.availableOutgoingBitrate
      };
    }
  });

  return result;
}
```

```javascript
async function adaptiveQuality(pc) {
  const stats = await pc.getStats();
  stats.forEach(report => {
    if (report.type === 'candidate-pair' && report.availableOutgoingBitrate) {
      const bandwidth = report.availableOutgoingBitrate;
      const videoSender = pc.getSenders().find(s => s.track?.kind === 'video');

      if (bandwidth < 300000) {
        // 带宽极低，关闭视频
        videoSender.track.enabled = false;
      } else if (bandwidth < 800000) {
        // 带宽低，降低分辨率
        videoSender.track.applyConstraints({ width: 320, height: 240 });
      }
    }
  });
}
```

## 小结

这个视频会议系统涵盖了 WebRTC 多方通信的核心功能：

- **信令层**：WebSocket 实现房间管理和消息路由
- **媒体层**：PeerConnection 管理音视频传输
- **数据层**：DataChannel 实现聊天消息
- **UI 层**：视频网格、聊天面板、控制按钮

每个模块都有独立的职责，通过事件回调解耦。生产环境中还需要考虑 SFU 服务器（如 mediasoup、Janus）、TURN 服务器部署、HTTPS/WSS 安全传输等。

### 练习一

**思路**：使用 MediaRecorder 录制本地和远程流。对于多方会议，可以将所有远程流合并到一个 Canvas 上录制，或者录制发言人画面。

**答案**：

```javascript
class MeetingRecorder {
  constructor() {
    this.recorder = null;
    this.chunks = [];
    this.isRecording = false;
    this.canvas = document.createElement('canvas');
    this.ctx = this.canvas.getContext('2d');
    this.compositeStream = null;
  }

  async startRecording(participants, localStream) {
    this.canvas.width = 1280;
    this.canvas.height = 720;

    // 创建合成流
    const videos = [];

    // 本地视频
    const localVideo = document.createElement('video');
    localVideo.srcObject = localStream;
    localVideo.muted = true;
    localVideo.play();
    videos.push({ video: localVideo, label: '我' });

    // 远程视频
    for (const [userId, participant] of participants) {
      if (participant.stream) {
        const remoteVideo = document.createElement('video');
        remoteVideo.srcObject = participant.stream;
        remoteVideo.play();
        videos.push({ video: remoteVideo, label: participant.displayName });
      }
    }

    // 合成绘制
    const drawFrame = () => {
      if (!this.isRecording) return;

      this.ctx.fillStyle = '#1a1a2e';
      this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);

      const cols = Math.ceil(Math.sqrt(videos.length));
      const rows = Math.ceil(videos.length / cols);
      const cellWidth = this.canvas.width / cols;
      const cellHeight = this.canvas.height / rows;

      videos.forEach((item, index) => {
        const col = index % cols;
        const row = Math.floor(index / cols);
        const x = col * cellWidth;
        const y = row * cellHeight;

        // 绘制视频
        this.ctx.drawImage(item.video, x + 2, y + 2, cellWidth - 4, cellHeight - 4);

        // 绘制标签
        this.ctx.fillStyle = 'rgba(0,0,0,0.6)';
        this.ctx.fillRect(x + 4, y + cellHeight - 30, 100, 24);
        this.ctx.fillStyle = '#fff';
        this.ctx.font = '14px sans-serif';
        this.ctx.fillText(item.label, x + 10, y + cellHeight - 12);
      });

      requestAnimationFrame(drawFrame);
    };

    drawFrame();

    // 获取合成流
    this.compositeStream = this.canvas.captureStream(30);

    // 添加音频（从本地流）
    const audioTrack = localStream.getAudioTracks()[0];
    if (audioTrack) {
      this.compositeStream.addTrack(audioTrack);
    }

    // 开始录制
    this.recorder = new MediaRecorder(this.compositeStream, {
      mimeType: 'video/webm;codecs=vp9,opus',
      videoBitsPerSecond: 2500000
    });

    this.chunks = [];
    this.recorder.ondataavailable = (e) => {
      if (e.data.size > 0) this.chunks.push(e.data);
    };

    this.recorder.start(1000);
    this.isRecording = true;

    this.onRecordingStateChange?.(true);
  }

  stopRecording() {
    return new Promise((resolve) => {
      if (!this.recorder || !this.isRecording) {
        resolve(null);
        return;
      }

      this.recorder.onstop = () => {
        const blob = new Blob(this.chunks, { type: 'video/webm' });
        const url = URL.createObjectURL(blob);
        this.chunks = [];
        this.isRecording = false;
        this.onRecordingStateChange?.(false);
        resolve({ blob, url });
      };

      this.recorder.stop();
      this.compositeStream?.getTracks().forEach(t => t.stop());
    });
  }

  downloadRecording(url, filename = 'meeting-recording.webm') {
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    a.click();
  }
}

// 集成到会议应用
const recorder = new MeetingRecorder();

recorder.onRecordingStateChange = (isRecording) => {
  document.getElementById('recordBtn').classList.toggle('active', isRecording);
  document.getElementById('recordBtn').textContent = isRecording ? '⏹️' : '⏺️';
};

document.getElementById('recordBtn').onclick = async () => {
  if (recorder.isRecording) {
    const result = await recorder.stopRecording();
    if (result) {
      recorder.downloadRecording(result.url);
    }
  } else {
    await recorder.startRecording(app.participants, app.localStream);
  }
};
```

**要点**：
- Canvas 合成可以自定义布局
- 音频从本地流获取，包含所有参与者的声音
- MediaRecorder 的 timeslice 参数控制数据回调频率

### 练习二

**思路**：使用之前课程中的 VirtualBackgroundProcessor，在会议系统中集成。关键是管理好处理管线的生命周期，以及在切换背景时不影响正在进行的通话。

**答案**：

```javascript
class BackgroundManager {
  constructor() {
    this.processor = null;
    this.originalTrack = null;
    this.processedTrack = null;
    this.currentMode = 'none'; // 'none' | 'blur' | 'image'
    this.isInitialized = false;
  }

  async init(videoTrack) {
    this.originalTrack = videoTrack;

    // 动态加载 BodyPix
    const bodyPix = await import(
      'https://cdn.jsdelivr.net/npm/@tensorflow-models/body-pix'
    );
    this.net = await bodyPix.load({
      architecture: 'MobileNetV1',
      outputStride: 16,
      multiplier: 0.75,
      quantBytes: 2
    });

    this.canvas = document.createElement('canvas');
    this.ctx = this.canvas.getContext('2d', { willReadFrequently: true });
    this.isInitialized = true;
  }

  async setBlurBackground(pc) {
    if (!this.isInitialized) return;
    this.currentMode = 'blur';
    await this.startProcessing(pc);
  }

  async setImageBackground(pc, imageUrl) {
    if (!this.isInitialized) return;

    const img = new Image();
    img.crossOrigin = 'anonymous';
    await new Promise((resolve, reject) => {
      img.onload = resolve;
      img.onerror = reject;
      img.src = imageUrl;
    });

    this.bgImage = img;
    this.currentMode = 'image';
    await this.startProcessing(pc);
  }

  async removeBackground(pc) {
    this.currentMode = 'none';
    this.stopProcessing();

    const sender = pc.getSenders().find(s => s.track?.kind === 'video');
    await sender.replaceTrack(this.originalTrack);
  }

  async startProcessing(pc) {
    this.stopProcessing();

    const video = document.createElement('video');
    video.srcObject = new MediaStream([this.originalTrack]);
    video.muted = true;
    await video.play();

    const settings = this.originalTrack.getSettings();
    this.canvas.width = settings.width || 640;
    this.canvas.height = settings.height || 480;

    this.isProcessing = true;

    const processLoop = async () => {
      if (!this.isProcessing) return;

      const segmentation = await this.net.segmentPerson(video, {
        internalResolution: 'medium',
        segmentationThreshold: 0.7
      });

      this.ctx.drawImage(video, 0, 0, this.canvas.width, this.canvas.height);
      const fgData = this.ctx.getImageData(0, 0, this.canvas.width, this.canvas.height);

      if (this.currentMode === 'blur') {
        this.ctx.filter = 'blur(12px)';
        this.ctx.drawImage(video, 0, 0, this.canvas.width, this.canvas.height);
        this.ctx.filter = 'none';
        const bgData = this.ctx.getImageData(0, 0, this.canvas.width, this.canvas.height);

        for (let i = 0; i < segmentation.data.length; i++) {
          if (segmentation.data[i] === 0) {
            const idx = i * 4;
            fgData.data[idx] = bgData.data[idx];
            fgData.data[idx + 1] = bgData.data[idx + 1];
            fgData.data[idx + 2] = bgData.data[idx + 2];
          }
        }
      } else if (this.currentMode === 'image' && this.bgImage) {
        this.ctx.drawImage(this.bgImage, 0, 0, this.canvas.width, this.canvas.height);
        const bgData = this.ctx.getImageData(0, 0, this.canvas.width, this.canvas.height);

        for (let i = 0; i < segmentation.data.length; i++) {
          if (segmentation.data[i] === 0) {
            const idx = i * 4;
            fgData.data[idx] = bgData.data[idx];
            fgData.data[idx + 1] = bgData.data[idx + 1];
            fgData.data[idx + 2] = bgData.data[idx + 2];
          }
        }
      }

      this.ctx.putImageData(fgData, 0, 0);
      requestAnimationFrame(processLoop);
    };

    processLoop();

    // 生成处理后的轨道
    this.processedTrack = this.canvas.captureStream(30).getVideoTracks()[0];

    // 替换发送轨道
    const sender = pc.getSenders().find(s => s.track?.kind === 'video');
    await sender.replaceTrack(this.processedTrack);
  }

  stopProcessing() {
    this.isProcessing = false;
    if (this.processedTrack) {
      this.processedTrack.stop();
      this.processedTrack = null;
    }
  }

  destroy() {
    this.stopProcessing();
    this.net?.dispose();
  }
}

// 集成到设置面板
const bgManager = new BackgroundManager();

document.getElementById('bgBlur').onclick = async () => {
```
