# 点对点通信

## 场景引入

上一课我们理解了 WebRTC 的架构。现在来实际做一件事：实现一个 1 对 1 的视频通话。

这听起来简单，但涉及的问题比想象的多：如何获取摄像头和麦克风？如何让用户选择设备？SDP 协商的每一步到底在做什么？如何处理带宽变化？如何在通话中共享屏幕？

这一课我们从零开始，逐个击破这些问题，最终完成一个可用的 1 对 1 视频通话功能。

## 学习目标

1. 掌握 `getUserMedia` 的约束配置和设备选择
2. 理解 `RTCPeerConnection` 的生命周期和事件
3. 掌握 SDP 协商的完整流程和常见操作
4. 了解带宽控制和编解码器偏好设置
5. 实现屏幕共享功能

## getUserMedia：获取本地媒体

### 基础约束

```javascript
// 最简单的调用
const stream = await navigator.mediaDevices.getUserMedia({
  video: true,
  audio: true
});
```

但实际项目中，你需要控制分辨率、帧率、降噪等参数：

```javascript
const stream = await navigator.mediaDevices.getUserMedia({
  video: {
    width: { ideal: 1280 },
    height: { ideal: 720 },
    frameRate: { ideal: 30 },
    facingMode: 'user'  // 前置摄像头
  },
  audio: {
    echoCancellation: true,       // 回声消除
    noiseSuppression: true,       // 噪声抑制
    autoGainControl: true,        // 自动增益
    sampleRate: 48000             // 采样率
  }
});
```

约束中的 `ideal` 和 `exact` 区别很大：

```javascript
// ideal: 尽量满足，不满足也能用
{ width: { ideal: 1920 } }

// exact: 必须满足，否则报错
{ width: { exact: 1920 } }

// min/max: 范围约束
{ width: { min: 640, max: 1920 } }
```

### 设备枚举与选择

用户可能有多个摄像头和麦克风，你需要提供设备选择界面：

```javascript
// 列出所有设备
async function listDevices() {
  // 先请求一次权限，否则设备标签为空
  await navigator.mediaDevices.getUserMedia({ video: true, audio: true });

  const devices = await navigator.mediaDevices.enumerateDevices();
  const cameras = devices.filter(d => d.kind === 'videoinput');
  const microphones = devices.filter(d => d.kind === 'audioinput');
  const speakers = devices.filter(d => d.kind === 'audiooutput');

  return { cameras, microphones, speakers };
}

// 使用指定设备
async function getStreamWithDevice(cameraId, micId) {
  return navigator.mediaDevices.getUserMedia({
    video: { deviceId: { exact: cameraId } },
    audio: { deviceId: { exact: micId } }
  });
}
```

设备切换时，需要替换 PeerConnection 中的轨道：

```javascript
async function switchCamera(pc, newCameraId) {
  const newStream = await navigator.mediaDevices.getUserMedia({
    video: { deviceId: { exact: newCameraId } }
  });
  const newTrack = newStream.getVideoTracks()[0];

  // 找到当前的发送器
  const sender = pc.getSenders().find(s => s.track?.kind === 'video');

  // 替换轨道（无需重新协商 SDP）
  await sender.replaceTrack(newTrack);

  // 停止旧轨道
  const oldTrack = sender.track;
  if (oldTrack) oldTrack.stop();
}
```

### 错误处理

getUserMedia 可能因为多种原因失败：

```javascript
async function safeGetMedia(constraints) {
  try {
    return await navigator.mediaDevices.getUserMedia(constraints);
  } catch (error) {
    switch (error.name) {
      case 'NotAllowedError':
        throw new Error('用户拒绝了摄像头/麦克风权限');

        throw new Error('设备被其他应用占用');
      case 'OverconstrainedError':

    }
  }
}
```

## SDP 协商详解

### 完整协商流程

```javascript
async function initiateCall(pc, stream) {
  // 1. 添加本地轨道
  stream.getTracks().forEach(track => pc.addTrack(track, stream));

  // 2. 创建 Offer
  const offer = await pc.createOffer({
    offerToReceiveAudio: true,
    offerToReceiveVideo: true
  });

  // 3. 设置本地描述（触发 ICE 收集）
  await pc.setLocalDescription(offer);

  // 4. 发送 Offer
  sendToSignalingServer({ type: 'offer', sdp: pc.localDescription.sdp });
}

async function handleOffer(pc, offerSdp, localStream) {
  // 1. 设置远程描述
  await pc.setRemoteDescription({ type: 'offer', sdp: offerSdp });

  // 2. 添加本地轨道
  localStream.getTracks().forEach(track => pc.addTrack(track, localStream));

  // 3. 创建 Answer
  const answer = await pc.createAnswer();

  // 4. 设置本地描述
  await pc.setLocalDescription(answer);

  // 5. 发送 Answer
  sendToSignalingServer({ type: 'answer', sdp: pc.localDescription.sdp });
}

async function handleAnswer(pc, answerSdp) {
  await pc.setRemoteDescription({ type: 'answer', sdp: answerSdp });
}

async function handleCandidate(pc, candidateData) {
  await pc.addIceCandidate(new RTCIceCandidate(candidateData));
}
```

### SDP 修改：控制编解码器偏好

```javascript
// 优先使用 H.264（如果支持）
function preferCodec(sdp, codecName) {
  const sections = sdp.split('\r\nm=');
  const videoSection = sections.find(s => s.startsWith('video'));
  if (!videoSection) return sdp;

  const lines = videoSection.split('\r\n');
  const payloadTypes = [];

  // 找到目标编解码器的 payload type
  for (const line of lines) {
    if (line.startsWith('a=rtpmap:') && line.includes(codecName)) {
      const pt = line.split(':')[1].split(' ')[0];
      payloadTypes.push(pt);
    }
  }

  if (payloadTypes.length === 0) return sdp;

  // 重排 m= 行中的 payload type
  const mLine = lines.find(l => l.startsWith('m=video'));
  const parts = mLine.split(' ');
  const otherPts = parts.slice(3).filter(pt => !payloadTypes.includes(pt));
  parts.splice(3, parts.length - 3, ...payloadTypes, ...otherPts);
  lines[lines.indexOf(mLine)] = parts.join(' ');

  sections[sections.indexOf(videoSection)] = lines.join('\r\n');
  return sections.join('\r\nm=');
}

// 使用方式
const offer = await pc.createOffer();
const modifiedSdp = preferCodec(offer.sdp, 'H264');
await pc.setLocalDescription({ type: 'offer', sdp: modifiedSdp });
```

更现代的方式是使用 `setCodecPreferences`：

```javascript
const transceiver = pc.getTransceivers().find(t => t.receiver.track.kind === 'video');
const codecs = RTCRtpSender.getCapabilities('video').codecs;

// 优先 H.264
const h264Codecs = codecs.filter(c => c.mimeType.includes('H264'));
const otherCodecs = codecs.filter(c => !c.mimeType.includes('H264'));
transceiver.setCodecPreferences([...h264Codecs, ...otherCodecs]);
```

### 带宽控制

```javascript
// 限制视频发送码率
async function setMaxBitrate(pc, maxBitrateKbps) {
  const sender = pc.getSenders().find(s => s.track?.kind === 'video');
  if (!sender) return;

  const params = sender.getParameters();
  if (!params.encodings || params.encodings.length === 0) {
    params.encodings = [{}];
  }
  params.encodings[0].maxBitrate = maxBitrateKbps * 1000;
  await sender.setParameters(params);
}

// 根据网络状况动态调整
async function adaptiveBitrate(pc) {
  let lastBytesSent = 0;

  setInterval(async () => {
    const stats = await pc.getStats();
    stats.forEach(report => {
      if (report.type === 'outbound-rtp' && report.kind === 'video') {
        const bytesSent = report.bytesSent;
        const bitrate = (bytesSent - lastBytesSent) * 8 / 3; // 3秒间隔
        lastBytesSent = bytesSent;

        console.log(`当前码率: ${(bitrate / 1000).toFixed(0)} kbps`);

        // 根据码率调整
        if (bitrate < 300000) {
          setMaxBitrate(pc, 300); // 低带宽，降低码率
        } else if (bitrate > 2000000) {
          setMaxBitrate(pc, 2500); // 高带宽，提高码率
        }
      }
    });
  }, 3000);
}
```

## 完整的 1 对 1 通话实现

```javascript
class OneOnOneCall {
  constructor(signalingUrl) {
    this.ws = new WebSocket(signalingUrl);
    this.pc = null;
    this.localStream = null;

    this.ws.onmessage = (event) => {
      const message = JSON.parse(event.data);
      this.handleSignalingMessage(message);
    };
  }

  async startCall(constraints = { video: true, audio: true }) {
    this.localStream = await navigator.mediaDevices.getUserMedia(constraints);
    this.showLocalVideo();

    this.pc = this.createPeerConnection();
    this.localStream.getTracks().forEach(track => {
      this.pc.addTrack(track, this.localStream);
    });

    const offer = await this.pc.createOffer();
    await this.pc.setLocalDescription(offer);
    this.send({ type: 'offer', sdp: offer.sdp });
  }

  createPeerConnection() {
    const pc = new RTCPeerConnection({
      iceServers: [{ urls: 'stun:stun.l.google.com:19302' }]
    });

    pc.onicecandidate = (e) => {
      if (e.candidate) {
        this.send({ type: 'candidate', candidate: e.candidate.toJSON() });
      }
    };

    pc.ontrack = (e) => {
      this.showRemoteVideo(e.streams[0]);
    };

    pc.onconnectionstatechange = () => {
      if (pc.connectionState === 'failed') {
        console.error('连接失败');
        this.restartIce();
      }
    };

    return pc;
  }

  async handleSignalingMessage(message) {
    switch (message.type) {
      case 'offer':
        if (!this.localStream) {
          this.localStream = await navigator.mediaDevices.getUserMedia({
            video: true, audio: true
          });
          this.showLocalVideo();
        }
        this.pc = this.createPeerConnection();
        this.localStream.getTracks().forEach(track => {
          this.pc.addTrack(track, this.localStream);
        });

        await this.pc.setRemoteDescription({ type: 'offer', sdp: message.sdp });
        const answer = await this.pc.createAnswer();
        await this.pc.setLocalDescription(answer);
        this.send({ type: 'answer', sdp: answer.sdp });
        break;

      case 'answer':
        await this.pc.setRemoteDescription({ type: 'answer', sdp: message.sdp });
        break;

      case 'candidate':
        await this.pc.addIceCandidate(new RTCIceCandidate(message.candidate));
        break;
    }
  }

  restartIce() {
    this.pc.restartIce();
  }

  endCall() {
    this.localStream?.getTracks().forEach(track => track.stop());
    this.pc?.close();
    this.pc = null;
  }

  send(message) {
    this.ws.send(JSON.stringify(message));
  }

  showLocalVideo() {
    document.getElementById('localVideo').srcObject = this.localStream;
  }

  showRemoteVideo(stream) {
    document.getElementById('remoteVideo').srcObject = stream;
  }
}
```

## 常见误区

### 误区一：ICE 收集是同步的

`setLocalDescription` 返回后，ICE 收集可能还在进行中。`onicecandidate` 事件会持续触发。不要在 `setLocalDescription` 之后立即判断"收集完成"。正确做法是监听 `onicecandidate` 事件，直到收到 `null` candidate 或 `iceGatheringState` 变为 `complete`。

### 误区二：addTrack 后需要重新协商

在 Unified Plan（现代浏览器的 SDP 语义）中，`addTrack` 不会立即触发重新协商。只有在调用 `createOffer`/`createAnswer` 时才会生成新的 SDP。`replaceTrack` 甚至完全不需要重新协商。

### 误区三：远程视频流需要手动创建

`ontrack` 事件的 `event.streams[0]` 已经是完整的媒体流，直接赋值给 `video.srcObject` 即可。不需要手动创建 MediaStream 并添加轨道。

### 误区四：屏幕共享会自动停止

用户点击浏览器的"停止共享"按钮会触发 `track.onended` 事件，但开发者主动停止共享需要手动处理轨道。如果只是 `replaceTrack` 回摄像头，旧的屏幕共享轨道需要 `stop()` 释放资源。

## 工程建议

1. **使用 Trickle ICE 加快连接速度**。不要等待所有候选收集完毕再发送，而是每收集到一个就立即通过信令发送。这样对方可以立即开始连通性检查，大幅缩短连接建立时间。

2. **处理设备热插拔**。用户可能在通话中途插拔耳机、摄像头：

```javascript
navigator.mediaDevices.ondevicechange = async () => {
  const devices = await navigator.mediaDevices.enumerateDevices();
  updateDeviceList(devices);
};
```

3. **实现优雅降级**。如果 720p 获取失败，自动降级到 480p：

```javascript
async function getStreamWithFallback() {
  const resolutions = [
    { width: 1280, height: 720 },
    { width: 640, height: 480 },
    { width: 320, height: 240 }
  ];

  for (const res of resolutions) {
    try {
      return await navigator.mediaDevices.getUserMedia({
        video: { width: { exact: res.width }, height: { exact: res.height } },
        audio: true
      });
    } catch (e) {
      console.warn(`${res.width}x${res.height} 不可用，尝试更低分辨率`);
    }
  }
  throw new Error('无法获取任何分辨率的视频流');
}
```

4. **监控通话质量**。定期采集 `getStats()` 数据，关注丢包率、抖动、RTT：


## 小结

点对点通信的核心流程：

1. **获取媒体**：`getUserMedia` 获取本地音视频流
2. **建立连接**：创建 `RTCPeerConnection`，添加轨道
3. **SDP 协商**：offer → setLocal → send → receive → setRemote → answer
4. **ICE 连通**：收集候选、交换、检查，找到可用路径
5. **媒体传输**：加密的音视频流在两端之间传输

每一步都有对应的错误处理和性能优化空间。理解这个流程后，你就能实现稳定的 1 对 1 视频通话。

## 参考答案

### 练习一

**思路**：使用 Node.js + ws 库实现信令服务器，客户端使用原生 WebRTC API。关键是在信令服务器上实现房间匹配，让两个用户配对。

**答案**：

信令服务器（Node.js）：

```javascript
const WebSocket = require('ws');
const wss = new WebSocket.Server({ port: 8080 });

const rooms = new Map();

wss.on('connection', (ws) => {
  let currentRoom = null;
  let userId = Math.random().toString(36).substr(2, 9);

  ws.on('message', (data) => {
    const message = JSON.parse(data);

    switch (message.type) {
      case 'join':
        currentRoom = message.roomId;
        if (!rooms.has(currentRoom)) {
          rooms.set(currentRoom, new Set());
        }
        rooms.get(currentRoom).add(ws);

        // 通知房间内其他人
        broadcast(currentRoom, { type: 'peer-joined', userId }, ws);
        break;

      case 'offer':
      case 'answer':
      case 'candidate':
        // 转发给房间内其他人
        broadcast(currentRoom, message, ws);
        break;
    }
  });

  ws.on('close', () => {
    if (currentRoom && rooms.has(currentRoom)) {
      rooms.get(currentRoom).delete(ws);
      broadcast(currentRoom, { type: 'peer-left', userId });
      if (rooms.get(currentRoom).size === 0) {
        rooms.delete(currentRoom);
      }
    }
  });

  function broadcast(roomId, message, exclude) {
    const room = rooms.get(roomId);
    if (room) {
      room.forEach(client => {
        if (client !== exclude && client.readyState === WebSocket.OPEN) {
          client.send(JSON.stringify(message));
        }
      });
    }
  }
});
```

客户端核心代码：

```javascript
class VideoCall {
  constructor(signalingUrl, roomId) {
    this.roomId = roomId;
    this.ws = new WebSocket(signalingUrl);
    this.pc = null;
    this.localStream = null;
    this.isMuted = false;
    this.isCameraOff = false;

    this.ws.onopen = () => {
      this.ws.send(JSON.stringify({ type: 'join', roomId }));
    };

    this.ws.onmessage = (event) => {
      this.handleMessage(JSON.parse(event.data));
    };
  }

  async init() {
    this.localStream = await navigator.mediaDevices.getUserMedia({
      video: true, audio: true
    });
    document.getElementById('localVideo').srcObject = this.localStream;
    this.setupPeerConnection();
  }

  setupPeerConnection() {
    this.pc = new RTCPeerConnection({
      iceServers: [{ urls: 'stun:stun.l.google.com:19302' }]
    });

    this.localStream.getTracks().forEach(track => {
      this.pc.addTrack(track, this.localStream);
    });

    this.pc.onicecandidate = (e) => {
      if (e.candidate) {
        this.ws.send(JSON.stringify({
          type: 'candidate', candidate: e.candidate.toJSON()
        }));
      }
    };

    this.pc.ontrack = (e) => {
      document.getElementById('remoteVideo').srcObject = e.streams[0];
    };

    this.pc.onconnectionstatechange = () => {
      if (this.pc.connectionState === 'failed') {
        this.pc.restartIce();
      }
    };
  }

  async handleMessage(msg) {
    switch (msg.type) {
      case 'peer-joined':
        const offer = await this.pc.createOffer();
        await this.pc.setLocalDescription(offer);
        this.ws.send(JSON.stringify({ type: 'offer', sdp: offer.sdp }));
        break;

      case 'offer':
        await this.pc.setRemoteDescription({ type: 'offer', sdp: msg.sdp });
        const answer = await this.pc.createAnswer();
        await this.pc.setLocalDescription(answer);
        this.ws.send(JSON.stringify({ type: 'answer', sdp: answer.sdp }));
        break;

      case 'answer':
        await this.pc.setRemoteDescription({ type: 'answer', sdp: msg.sdp });
        break;

      case 'candidate':
        await this.pc.addIceCandidate(new RTCIceCandidate(msg.candidate));
        break;

      case 'peer-left':
        document.getElementById('remoteVideo').srcObject = null;
        break;
    }
  }

  toggleMute() {
    const audioTrack = this.localStream.getAudioTracks()[0];
    audioTrack.enabled = !audioTrack.enabled;
    this.isMuted = !audioTrack.enabled;
  }

  toggleCamera() {
    const videoTrack = this.localStream.getVideoTracks()[0];
    videoTrack.enabled = !videoTrack.enabled;
    this.isCameraOff = !videoTrack.enabled;
  }

  hangup() {
    this.localStream.getTracks().forEach(t => t.stop());
    this.pc.close();
    this.ws.close();
  }
}
```

**要点**：
- 信令服务器只负责转发消息，不处理媒体逻辑
- 第二个加入房间的用户触发 offer 流程
- 设备切换用 `replaceTrack` 不需要重新协商

### 练习二

**思路**：在通话中用 `getDisplayMedia` 获取屏幕流，替换视频发送器的轨道。监听 `track.onended` 恢复摄像头。

**答案**：

```javascript
class VideoCallWithScreenShare extends VideoCall {
  constructor(signalingUrl, roomId) {
    super(signalingUrl, roomId);
    this.screenTrack = null;
    this.isScreenSharing = false;
  }

  async startScreenShare() {
    try {
      const screenStream = await navigator.mediaDevices.getDisplayMedia({
        video: { cursor: 'always' },
        audio: false
      });

      this.screenTrack = screenStream.getVideoTracks()[0];
      this.isScreenSharing = true;

      this.screenTrack.onended = () => {
        this.stopScreenShare();
      };

      const videoSender = this.pc.getSenders().find(s => s.track?.kind === 'video');
      await videoSender.replaceTrack(this.screenTrack);

      document.getElementById('localVideo').srcObject = screenStream;
      document.getElementById('shareBtn').textContent = '停止共享';
    } catch (error) {
      if (error.name !== 'NotAllowedError') {
        console.error('屏幕共享失败:', error);
      }
    }
  }

  async stopScreenShare() {
    if (!this.isScreenSharing) return;

    this.screenTrack.stop();
    this.isScreenSharing = false;

    const cameraStream = await navigator.mediaDevices.getUserMedia({ video: true });
    const cameraTrack = cameraStream.getVideoTracks()[0];

    const videoSender = this.pc.getSenders().find(s => s.track?.kind === 'video');
    await videoSender.replaceTrack(cameraTrack);

    document.getElementById('localVideo').srcObject = this.localStream;
    document.getElementById('shareBtn').textContent = '共享屏幕';
  }
}
```

**要点**：
- `getDisplayMedia` 必须由用户手势触发（点击按钮）
- `track.onended` 在用户点击浏览器的"停止共享"时触发
- 共享屏幕时本地预览也要更新，让用户知道自己在共享什么
