# 第六课：阶段实战——在线录音机

## 场景引入

经过前五课的学习，你已经掌握了音视频基础概念、Web Audio API、Canvas 帧操作、MediaRecorder API 和文件格式处理。现在，是时候将这些知识整合起来，构建一个完整的在线录音机应用。

这个项目将综合运用以下能力：

- **MediaRecorder**：录制麦克风音频
- **Web Audio API**：实时音频可视化（波形图）
- **Canvas**：绘制音频波形
- **Blob/File API**：生成可下载的音频文件
- **格式处理**：支持多种输出格式

完成本课后，你将拥有一个功能完整、可以直接使用的在线录音机。

    const state = {
      mediaRecorder: null,
      audioContext: null,
      analyser: null,
      stream: null,
      chunks: [],
      recordings: [],
      startTime: 0,
      elapsed: 0,
      timerInterval: null,
      animationId: null,
      isRecording: false,
      isPaused: false
    };

    // ===== DOM 引用 =====
    const canvas = document.getElementById('visualizer');
    const ctx = canvas.getContext('2d');
    const timerEl = document.getElementById('timer');
    const statusEl = document.getElementById('status');
    const recordBtn = document.getElementById('recordBtn');
    const pauseBtn = document.getElementById('pauseBtn');
    const stopBtn = document.getElementById('stopBtn');
    const formatSelect = document.getElementById('formatSelect');
    const noiseSuppressionCb = document.getElementById('noiseSuppression');
    const echoCancellationCb = document.getElementById('echoCancellation');
    const recordingsList = document.getElementById('recordingsList');

    // ===== Canvas 尺寸设置 =====
    function resizeCanvas() {
      const rect = canvas.parentElement.getBoundingClientRect();
      canvas.width = rect.width * window.devicePixelRatio;
      canvas.height = rect.height * window.devicePixelRatio;
      ctx.scale(window.devicePixelRatio, window.devicePixelRatio);

    // ===== 工具函数 =====
    function formatTime(ms) {
      const totalSeconds = Math.floor(ms / 1000);
      const minutes = Math.floor(totalSeconds / 60);
      const seconds = totalSeconds % 60;
      const tenths = Math.floor((ms % 1000) / 100);
      return `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}.${tenths}`;
    }

    function getSupportedMimeType() {
      const preference = formatSelect.value;
      const typeMap = {
        'webm-opus': 'audio/webm;codecs=opus',
        'webm-vorbis': 'audio/webm;codecs=vorbis',
        'mp4': 'audio/mp4;codecs=mp4a',
      };

      if (preference !== 'auto') {
        const type = typeMap[preference];
        if (MediaRecorder.isTypeSupported(type)) return type;
      }

      // 自动选择
      const candidates = [
        'audio/webm;codecs=opus',
        'audio/webm;codecs=vorbis',
        'audio/mp4;codecs=mp4a',
        'audio/ogg;codecs=opus',
        'audio/wav'
      ];
      return candidates.find(t => MediaRecorder.isTypeSupported(t));
    }

    function getFileExtension(mimeType) {
      if (mimeType.includes('webm')) return '.webm';
      if (mimeType.includes('mp4')) return '.m4a';
      if (mimeType.includes('ogg')) return '.ogg';
      if (mimeType.includes('wav')) return '.wav';
      return '.audio';
    }

    // ===== 可视化 =====
    function drawVisualization() {
      if (!state.analyser) return;

      state.animationId = requestAnimationFrame(drawVisualization);

      const bufferLength = state.analyser.frequencyBinCount;
      const dataArray = new Uint8Array(bufferLength);
      state.analyser.getByteTimeDomainData(dataArray);

      const width = canvas.width / window.devicePixelRatio;
      const height = canvas.height / window.devicePixelRatio;

      ctx.fillStyle = '#0f3460';
      ctx.fillRect(0, 0, width, height);

      ctx.lineWidth = 2;
      ctx.strokeStyle = state.isRecording && !state.isPaused ? '#e94560' : '#533483';
      ctx.beginPath();

      const sliceWidth = width / bufferLength;
      let x = 0;

      for (let i = 0; i < bufferLength; i++) {
        const v = dataArray[i] / 128.0;
        const y = (v * height) / 2;

        if (i === 0) {
          ctx.moveTo(x, y);
        } else {
          ctx.lineTo(x, y);
        }
        x += sliceWidth;
      }

      ctx.lineTo(width, height / 2);
      ctx.stroke();

      // 绘制中心线
      ctx.strokeStyle = 'rgba(255,255,255,0.1)';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(0, height / 2);
      ctx.lineTo(width, height / 2);
      ctx.stroke();
    }

    function startVisualization() {
      if (state.animationId) cancelAnimationFrame(state.animationId);
      drawVisualization();
    }

    function stopVisualization() {
      if (state.animationId) {
        cancelAnimationFrame(state.animationId);
        state.animationId = null;
      }
      // 绘制静音状态
      const width = canvas.width / window.devicePixelRatio;
      const height = canvas.height / window.devicePixelRatio;
      ctx.fillStyle = '#0f3460';
      ctx.fillRect(0, 0, width, height);
      ctx.strokeStyle = '#533483';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(0, height / 2);
      ctx.lineTo(width, height / 2);
      ctx.stroke();
    }

    // ===== 计时器 =====
    function startTimer() {
      state.startTime = Date.now() - state.elapsed;
      state.timerInterval = setInterval(() => {
        state.elapsed = Date.now() - state.startTime;
        timerEl.textContent = formatTime(state.elapsed);
      }, 100);
    }

    function stopTimer() {
      clearInterval(state.timerInterval);
      state.timerInterval = null;
    }

    // ===== 录音控制 =====
    async function startRecording() {
      try {
        const constraints = {
          audio: {
            noiseSuppression: noiseSuppressionCb.checked,
            echoCancellation: echoCancellationCb.checked,
            autoGainControl: true
          }
        };

        state.stream = await navigator.mediaDevices.getUserMedia(constraints);

        // 创建 AudioContext 用于可视化
        state.audioContext = new AudioContext();
        if (state.audioContext.state === 'suspended') {
          await state.audioContext.resume();
        }
        state.analyser = state.audioContext.createAnalyser();
        state.analyser.fftSize = 2048;

        const source = state.audioContext.createMediaStreamSource(state.stream);
        source.connect(state.analyser);

        // 创建 MediaRecorder
        const mimeType = getSupportedMimeType();
        state.mediaRecorder = new MediaRecorder(state.stream, { mimeType });
        state.chunks = [];

        state.mediaRecorder.ondataavailable = (e) => {
          if (e.data.size > 0) state.chunks.push(e.data);
        };

        state.mediaRecorder.onstop = () => {
          const blob = new Blob(state.chunks, { type: mimeType });
          addRecording(blob, mimeType, state.elapsed);
          cleanup();
        };

        state.mediaRecorder.onerror = (e) => {
          console.error('录制错误:', e.error);
          statusEl.textContent = `录制错误: ${e.error.message}`;
          cleanup();
        };

        // 开始录制
        state.mediaRecorder.start(100);
        state.isRecording = true;
        state.isPaused = false;

        startTimer();
        startVisualization();
        updateUI();
        statusEl.textContent = '录音中...';

      } catch (err) {
        handleRecordingError(err);
      }
    }

    function pauseRecording() {
      if (!state.mediaRecorder || state.mediaRecorder.state !== 'recording') return;
      state.mediaRecorder.pause();
      state.isPaused = true;
      stopTimer();
      updateUI();
      statusEl.textContent = '已暂停';
    }

    function resumeRecording() {
      if (!state.mediaRecorder || state.mediaRecorder.state !== 'paused') return;
      state.mediaRecorder.resume();
      state.isPaused = false;
      startTimer();
      updateUI();
      statusEl.textContent = '录音中...';
    }

    function stopRecording() {
      if (!state.mediaRecorder) return;
      state.mediaRecorder.stop();
      state.isRecording = false;
      state.isPaused = false;
      stopTimer();
      stopVisualization();
      updateUI();
      statusEl.textContent = '录音完成';
    }

    function cleanup() {
      if (state.stream) {
        state.stream.getTracks().forEach(track => track.stop());
        state.stream = null;
      }
      if (state.audioContext && state.audioContext.state !== 'closed') {
        state.audioContext.close();
        state.audioContext = null;
      }
      state.analyser = null;
      state.mediaRecorder = null;
    }

    function handleRecordingError(err) {
      let message = '录音失败';
      if (err.name === 'NotAllowedError') {
        message = '麦克风权限被拒绝，请在浏览器设置中允许';
      } else if (err.name === 'NotFoundError') {
        message = '未找到麦克风设备';
      } else if (err.name === 'NotReadableError') {
        message = '麦克风被其他应用占用';
      } else {
        message = `录音失败: ${err.message}`;
      }
      statusEl.textContent = message;
      cleanup();
    }

    // ===== UI 更新 =====
    function updateUI() {
      recordBtn.disabled = state.isRecording;
      recordBtn.classList.toggle('recording', state.isRecording);
      pauseBtn.disabled = !state.isRecording;
      stopBtn.disabled = !state.isRecording;

      if (state.isPaused) {
        pauseBtn.textContent = '▶';
        pauseBtn.title = '继续';
      } else {
        pauseBtn.textContent = '❚❚';
        pauseBtn.title = '暂停';
      }
    }

    // ===== 录音列表管理 =====
    function addRecording(blob, mimeType, duration) {
      const url = URL.createObjectURL(blob);
      const recording = {
        id: Date.now(),
        name: `录音 ${state.recordings.length + 1}`,
        blob,
        url,
        mimeType,
        duration,
        size: blob.size,
        createdAt: new Date()
      };
      state.recordings.push(recording);
      renderRecordingsList();
    }

    function deleteRecording(id) {
      const index = state.recordings.findIndex(r => r.id === id);
      if (index === -1) return;

      const recording = state.recordings[index];
      URL.revokeObjectURL(recording.url);
      state.recordings.splice(index, 1);
      renderRecordingsList();
    }

    function downloadRecording(id) {
      const recording = state.recordings.find(r => r.id === id);
      if (!recording) return;

      const a = document.createElement('a');
      a.href = recording.url;
      const ext = getFileExtension(recording.mimeType);
      a.download = `${recording.name}${ext}`;
      a.click();
    }

    function renderRecordingsList() {
      if (state.recordings.length === 0) {
        recordingsList.innerHTML = '<div class="empty-state">暂无录音</div>';
        return;
      }

      recordingsList.innerHTML = state.recordings.map(rec => {
        const sizeKB = (rec.size / 1024).toFixed(1);
        const durationStr = formatTime(rec.duration);
        const timeStr = rec.createdAt.toLocaleTimeString();

        return `
          <div class="recording-item">
            <div class="info">
              <div class="name">${rec.name}</div>
              <div class="meta">${durationStr} · ${sizeKB} KB · ${timeStr}</div>
            </div>
            <audio controls src="${rec.url}"></audio>
            <div class="actions">
              <button onclick="downloadRecording(${rec.id})">下载</button>
              <button class="delete" onclick="deleteRecording(${rec.id})">删除</button>
            </div>
          </div>
        `;
      }).join('');
    }

    // ===== 事件绑定 =====
    recordBtn.addEventListener('click', () => {
      if (state.isRecording) return;
      state.elapsed = 0;
      timerEl.textContent = '00:00.0';
      startRecording();

    pauseBtn.addEventListener('click', () => {

        resumeRecording();
      } else {
        pauseRecording();
      }
    });

    stopBtn.addEventListener('click', stopRecording);

    // 初始化静音波形
    stopVisualization();

### AudioContext 与 MediaRecorder 的协作

```javascript
// 关键：AudioContext 用于可视化，MediaRecorder 用于录制
// 两者共享同一个 MediaStream，互不干扰

const stream = await navigator.mediaDevices.getUserMedia({ audio: true });

// 路径一：MediaRecorder 录制
const recorder = new MediaRecorder(stream);
recorder.start();

// 路径二：Web Audio 可视化
const audioCtx = new AudioContext();
const source = audioCtx.createMediaStreamSource(stream);
const analyser = audioCtx.createAnalyser();
source.connect(analyser);
// 注意：不要连接到 destination，否则会回放麦克风声音
```

## 功能扩展

### 添加录音降噪

Web Audio API 提供了 `MediaStreamAudioSourceNode`，可以在录制前对音频进行处理：

```javascript
async function startRecordingWithNoiseSuppression() {
  const stream = await navigator.mediaDevices.getUserMedia({
    audio: {
      noiseSuppression: true,    // 浏览器内置降噪
      echoCancellation: true,    // 回声消除
      autoGainControl: true      // 自动增益
    }
  });

  // 如果需要更精细的降噪，可以添加高通滤波器
  const audioCtx = new AudioContext();
  const source = audioCtx.createMediaStreamSource(stream);

  // 高通滤波器：去除低频噪音
  const highpass = audioCtx.createBiquadFilter();
  highpass.type = 'highpass';
  highpass.frequency.setValueAtTime(80, audioCtx.currentTime); // 80Hz 以下切除

  // 低通滤波器：去除高频噪音
  const lowpass = audioCtx.createBiquadFilter();
  lowpass.type = 'lowpass';
  lowpass.frequency.setValueAtTime(8000, audioCtx.currentTime); // 8kHz 以上切除

  source.connect(highpass);
  highpass.connect(lowpass);
  // lowpass 用于可视化，stream 用于录制
}
```

### 添加录音计时器显示

```javascript
// 在已有计时器基础上，添加倒计时功能
class RecordingTimer {
  constructor(element) {
    this.element = element;
    this.startTime = 0;
    this.maxDuration = 0; // 0 表示无限制
    this.interval = null;
    this.onMaxReached = null;
  }

  start(maxDuration = 0) {
    this.startTime = Date.now();
    this.maxDuration = maxDuration;
    this.interval = setInterval(() => this.update(), 100);
  }

  update() {
    const elapsed = Date.now() - this.startTime;
    this.element.textContent = this.format(elapsed);

    if (this.maxDuration > 0 && elapsed >= this.maxDuration) {
      this.stop();
      if (this.onMaxReached) this.onMaxReached();
    }
  }

  format(ms) {
    const s = Math.floor(ms / 1000);
    const m = Math.floor(s / 60);
    return `${String(m).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}.${Math.floor((ms % 1000) / 100)}`;
  }

  stop() {
    clearInterval(this.interval);
    this.interval = null;
  }
}
```

## 常见误区

### 误区一：不处理麦克风权限被撤销的情况

用户可能在录音过程中手动撤销麦克风权限。代码需要监听 MediaStreamTrack 的 `ended` 事件来处理这种情况：

```javascript
stream.getAudioTracks()[0].addEventListener('ended', () => {
  if (state.isRecording) {
    statusEl.textContent = '麦克风权限被撤销，录音已停止';
    stopRecording();
  }
});
```

### 误区二：AudioContext 不关闭导致资源泄漏

每个 AudioContext 都会占用系统音频资源。录音结束后必须关闭：

```javascript
function cleanup() {
  // ... 其他清理
  if (state.audioContext && state.audioContext.state !== 'closed') {
    state.audioContext.close();
    state.audioContext = null;
  }
}
```

### 误区三：忘记停止 MediaStream 的轨道

MediaRecorder 停止后，MediaStream 的轨道仍然活跃，麦克风指示灯仍然亮着：

```javascript
// MediaRecorder 停止后，必须手动停止轨道
recorder.onstop = () => {
  stream.getTracks().forEach(track => track.stop());
};
```

### 误区四：Blob URL 未释放导致内存泄漏

每次创建 Blob URL 都会占用内存。如果不再需要，必须调用 `URL.revokeObjectURL()` 释放：

```javascript
// 不好：不断创建新的 Blob URL 但从不释放
const url = URL.createObjectURL(blob);

// 好：用完即释放
const url = URL.createObjectURL(blob);
a.click();
setTimeout(() => URL.revokeObjectURL(url), 1000);
```

### 练习一

**思路**：在 AudioContext 中插入一个 BiquadFilterNode，通过开关控制是否启用。

**答案**：

```javascript
// 在 startRecording 函数中，修改可视化部分的代码
async function startRecordingWithFilter() {
  const stream = await navigator.mediaDevices.getUserMedia({
    audio: {
      noiseSuppression: noiseSuppressionCb.checked,
      echoCancellation: echoCancellationCb.checked,
      autoGainControl: true
    }
  });

  state.audioContext = new AudioContext();
  if (state.audioContext.state === 'suspended') {
    await state.audioContext.resume();
  }

  const source = state.audioContext.createMediaStreamSource(stream);

  // 创建高通滤波器
  state.highpassFilter = state.audioContext.createBiquadFilter();
  state.highpassFilter.type = 'highpass';
  state.highpassFilter.frequency.setValueAtTime(100, state.audioContext.currentTime);

  state.analyser = state.audioContext.createAnalyser();
  state.analyser.fftSize = 2048;

  // 连接：source → filter → analyser
  source.connect(state.highpassFilter);
  state.highpassFilter.connect(state.analyser);

  // 控制滤波器开关
  window.toggleFilter = (enabled) => {
    if (enabled) {
      source.disconnect();
      source.connect(state.highpassFilter);
      state.highpassFilter.connect(state.analyser);
    } else {
      source.disconnect();
      state.highpassFilter.disconnect();
      source.connect(state.analyser);
    }
  };

  // MediaRecorder 使用原始 stream（不受滤波器影响）
  // 如需录制经过滤波的音频，需要使用 MediaStreamDestination
  const destination = state.audioContext.createMediaStreamDestination();
  state.highpassFilter.connect(destination);

  const filteredStream = destination.stream;
  state.mediaRecorder = new MediaRecorder(filteredStream, { mimeType: getSupportedMimeType() });

  // ... 其余逻辑不变
}
```

**要点**：
- 噪声抑制通过 `getUserMedia` 的 constraints 控制，是浏览器原生能力
- 高通滤波器通过 Web Audio API 实现，可以去除低频噪音
- 注意区分"可视化用的流"和"录制用的流"

### 练习二

**思路**：增强计时器逻辑，添加倒计时和样式变化。

**答案**：

```javascript
class EnhancedTimer {
  constructor(timerElement, maxMinutes = 60) {
    this.element = timerElement;
    this.maxDuration = maxMinutes * 60 * 1000;
    this.warningThreshold = 10000; // 剩余 10 秒警告
    this.startTime = 0;
    this.elapsed = 0;
    this.interval = null;
    this.onMaxReached = null;
    this.isWarning = false;
  }

  start() {
    this.startTime = Date.now() - this.elapsed;
    this.isWarning = false;
    this.element.style.color = '#e94560';
    this.interval = setInterval(() => this.update(), 50); // 50ms 精度
  }

  update() {
    this.elapsed = Date.now() - this.startTime;
    const remaining = this.maxDuration - this.elapsed;

    // 显示当前时间（毫秒精度）
    this.element.textContent = this.formatWithMs(this.elapsed);

    // 接近限制时警告
    if (remaining <= this.warningThreshold && remaining > 0) {
      if (!this.isWarning) {
        this.isWarning = true;
        this.element.classList.add('warning');
      }
      // 闪烁效果
      this.element.style.opacity = Math.sin(Date.now() / 100) > 0 ? '1' : '0.3';
    }

    // 到达限制
    if (this.elapsed >= this.maxDuration) {
      this.stop();
      this.element.style.opacity = '1';
      if (this.onMaxReached) this.onMaxReached();
    }
  }

  formatWithMs(ms) {
    const totalSeconds = Math.floor(ms / 1000);
    const minutes = Math.floor(totalSeconds / 60);
    const seconds = totalSeconds % 60;
    const msDigit = Math.floor((ms % 1000) / 10);
    return `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}.${String(msDigit).padStart(2, '0')}`;
  }

  stop() {
    clearInterval(this.interval);
    this.interval = null;
  }

  reset() {
    this.stop();
    this.elapsed = 0;
    this.isWarning = false;
    this.element.style.opacity = '1';
    this.element.style.color = '#e94560';
    this.element.textContent = '00:00.00';
  }
}

// 使用
const timer = new EnhancedTimer(timerEl, 60); // 最大 60 分钟
timer.onMaxReached = () => {
  stopRecording();
  statusEl.textContent = '已达到最大录音时长';
};
```

**要点**：
- 毫秒精度使用 50ms 的更新间隔，比 100ms 更流畅
- 使用 `Math.sin` 实现平滑的闪烁效果
- 警告状态需要在录音停止时重置
