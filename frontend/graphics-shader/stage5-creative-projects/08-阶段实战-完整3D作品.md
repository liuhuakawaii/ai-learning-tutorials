### 地球渲染系统

```typescript
class GlobeRenderSystem implements System {
  private scene: THREE.Scene

  constructor(scene: THREE.Scene) {
    this.scene = scene
  }

  update(world: World, dt: number): void {
    const globes = world.getStore<GlobeComponent>('globe')
    for (const entity of globes.entities()) {
      const globe = globes.get(entity)!
      if (globe.autoRotate) {
        globe.mesh.rotation.y += globe.rotationSpeed * dt
        globe.atmosphereMesh.rotation.y = globe.mesh.rotation.y
      }
    }
  }

  async buildGlobe(world: World, resources: ResourceManager): Promise<Entity> {
    const entity = world.createEntity()

    // 球体几何
    const geometry = new THREE.SphereGeometry(5, 128, 128)
    const [dayTex, nightTex, bumpTex, specTex] = await Promise.all([
      resources.load<THREE.Texture>('texture', '/textures/earth-day.jpg'),
      resources.load<THREE.Texture>('texture', '/textures/earth-night.jpg'),
      resources.load<THREE.Texture>('texture', '/textures/earth-bump.jpg'),
      resources.load<THREE.Texture>('texture', '/textures/earth-specular.jpg'),
    ])

    // PBR 材质
    const material = new THREE.MeshStandardMaterial({
      map: dayTex,
      roughnessMap: specTex,
      roughness: 0.7,
      metalness: 0.1,
      bumpMap: bumpTex,
      bumpScale: 0.05,
    })

    const mesh = new THREE.Mesh(geometry, material)
    mesh.castShadow = true
    mesh.receiveShadow = true

    // 大气层 Shader
    const atmosphereMesh = this.createAtmosphere(5)

    this.scene.add(mesh)
    this.scene.add(atmosphereMesh)

    const store = world.getStore<GlobeComponent>('globe')
    store.add(entity, {
      radius: 5,
      segments: 128,
      mesh,
      atmosphereMesh,
      rotationSpeed: 0.02,
      autoRotate: true,
    })

    return entity
  }

  private createAtmosphere(radius: number): THREE.Mesh {
    const geometry = new THREE.SphereGeometry(radius * 1.15, 64, 64)
    const material = new THREE.ShaderMaterial({
      vertexShader: `
        varying vec3 vNormal;
        varying vec3 vPosition;
        void main() {
          vNormal = normalize(normalMatrix * normal);
          vPosition = (modelViewMatrix * vec4(position, 1.0)).xyz;
          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
        }
      `,
      fragmentShader: `
        varying vec3 vNormal;
        varying vec3 vPosition;
        void main() {
          vec3 viewDir = normalize(-vPosition);
          float rim = 1.0 - max(0.0, dot(viewDir, vNormal));
          rim = pow(rim, 3.0);
          vec3 color = mix(vec3(0.3, 0.6, 1.0), vec3(0.1, 0.3, 0.8), rim);
          gl_FragColor = vec4(color, rim * 0.6);
        }
      `,
      transparent: true,
      side: THREE.BackSide,
      blending: THREE.AdditiveBlending,
      depthWrite: false,
    })
    return new THREE.Mesh(geometry, material)
  }
}
```

```typescript
    return data

    geojson: any,
    countryData: CountryData[]
  ): Promise<DataPointComponent[]> {
    return new Promise((resolve) => {
      this.worker.onmessage = (e) => {
        resolve(e.data)
      }
      this.worker.postMessage({
        type: 'process-geo-data',
        geojson,
        countryData,
      })
    })
  }
```

### DataProcessor Worker

```typescript
// DataProcessor.worker.ts
self.onmessage = (e: MessageEvent) => {
  const { type, geojson, countryData } = e.data

  if (type === 'process-geo-data') {
    const dataMap = new Map(countryData.map((d: any) => [d.code, d]))
    const maxGdp = Math.max(...countryData.map((d: any) => d.gdp))
    const minGdp = Math.min(...countryData.map((d: any) => d.gdp))

    const result = countryData.map((country: any) => {
      const normalizedValue = (country.gdp - minGdp) / (maxGdp - minGdp)
      return {
        lat: country.lat,
        lon: country.lon,
        value: normalizedValue,
        label: country.name,
        country: country.code,
        color: hslColor(0.66 * (1 - normalizedValue), 0.9, 0.5),
      }
    })

    self.postMessage(result)
  }
}

function hslColor(h: number, s: number, l: number): { r: number; g: number; b: number } {
  const c = (1 - Math.abs(2 * l - 1)) * s
  const x = c * (1 - Math.abs((h * 6) % 2 - 1))
  const m = l - c / 2
  let r = 0, g = 0, b = 0
  if (h < 1/6) { r = c; g = x; }
  else if (h < 2/6) { r = x; g = c; }
  else if (h < 3/6) { g = c; b = x; }
  else if (h < 4/6) { g = x; b = c; }
  else if (h < 5/6) { r = x; b = c; }
  else { r = c; b = x; }
  return { r: r + m, g: g + m, b: b + m }
}
```

### 柱状图系统

```typescript
class DataVisualizationSystem implements System {
  private scene: THREE.Scene
  private barGroup: THREE.Group

  constructor(scene: THREE.Scene) {
    this.scene = scene
    this.barGroup = new THREE.Group()
    this.scene.add(this.barGroup)
  }

  update(world: World, dt: number): void {
    const dataPoints = world.getStore<DataPointComponent>('dataPoint')
    // 动画：柱体高度从 0 增长到目标值
    for (const entity of dataPoints.entities()) {
      const dp = dataPoints.get(entity)
      if (dp?.barMesh) {
        const targetScale = dp.value * 2
        const currentScale = dp.barMesh.scale.y
        if (Math.abs(currentScale - targetScale) > 0.001) {
          dp.barMesh.scale.y += (targetScale - currentScale) * dt * 3
        }
      }
    }
  }

  createBars(world: World, dataPoints: DataPointComponent[]): void {
    const globeRadius = 5

    dataPoints.forEach((dp, i) => {
      const entity = world.createEntity()

      const barHeight = dp.value * 2
      const geometry = new THREE.CylinderGeometry(0.03, 0.03, barHeight, 8)
      geometry.translate(0, barHeight / 2, 0)

      const material = new THREE.MeshStandardMaterial({
        color: new THREE.Color(dp.color.r, dp.color.g, dp.color.b),
        emissive: new THREE.Color(dp.color.r, dp.color.g, dp.color.b),
        emissiveIntensity: 0.3,
      })

      const bar = new THREE.Mesh(geometry, material)

      // 定位到地球表面
      const pos = latLonToVector3(dp.lat, dp.lon, globeRadius)
      bar.position.copy(pos)
      bar.lookAt(new THREE.Vector3(0, 0, 0))
      bar.rotateX(Math.PI / 2)
      bar.scale.y = 0 // 初始高度为 0，用于动画

      this.barGroup.add(bar)
      dp.barMesh = bar

      const store = world.getStore<DataPointComponent>('dataPoint')
      store.add(entity, { ...dp, barMesh: bar })

      const interactables = world.getStore<InteractableComponent>('interactable')
      interactables.add(entity, {
        type: 'bar',
        entityId: entity,
        hoverEffect: true,
        clickAction: 'showDetail',
      })
    })
  }
}
```

### 热力图系统

```typescript
class HeatmapSystem implements System {
  private scene: THREE.Scene

  constructor(scene: THREE.Scene) {
    this.scene = scene
  }

  createHeatmapLayer(
    world: World,
    data: Float32Array,
    width: number,
    height: number,
    radius: number
  ): Entity {
    const entity = world.createEntity()

    // 创建数据纹理
    const texture = new THREE.DataTexture(
      this.generateHeatmapColors(data, width, height),
      width,
      height,
      THREE.RGBAFormat
    )
    texture.needsUpdate = true

    // 热力图球面材质
    const geometry = new THREE.SphereGeometry(radius * 1.002, 128, 128)
    const material = new THREE.ShaderMaterial({
      uniforms: {
        u_heatmap: { value: texture },
        u_opacity: { value: 0.6 },
      },
      vertexShader: `
        varying vec2 v_uv;
        void main() {
          v_uv = uv;
          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
        }
      `,
      fragmentShader: `
        uniform sampler2D u_heatmap;
        uniform float u_opacity;
        varying vec2 v_uv;

        vec3 colorMap(float t) {
          vec3 cold = vec3(0.0, 0.0, 1.0);
          vec3 warm = vec3(1.0, 1.0, 0.0);
          vec3 hot = vec3(1.0, 0.0, 0.0);
          if (t < 0.5) return mix(cold, warm, t * 2.0);
          return mix(warm, hot, (t - 0.5) * 2.0);
        }

        void main() {
          vec4 heat = texture2D(u_heatmap, v_uv);
          float intensity = heat.r;
          vec3 color = colorMap(intensity);
          gl_FragColor = vec4(color, intensity * u_opacity);
        }
      `,
      transparent: true,
      depthWrite: false,
    })

    const mesh = new THREE.Mesh(geometry, material)
    this.scene.add(mesh)

    const store = world.getStore<HeatmapLayerComponent>('heatmap')
    store.add(entity, { texture, data, width, height, mesh, opacity: 0.6 })

    return entity
  }

  private generateHeatmapColors(data: Float32Array, w: number, h: number): Uint8Array {
    const pixels = new Uint8Array(w * h * 4)
    for (let i = 0; i < w * h; i++) {
      const value = data[i]
      const idx = i * 4
      pixels[idx] = Math.floor(value * 255)
      pixels[idx + 1] = 0
      pixels[idx + 2] = Math.floor((1 - value) * 255)
      pixels[idx + 3] = Math.floor(value * 200)
    }
    return pixels
  }
}
```

```typescript
class InteractionSystem implements System {
  private raycaster = new THREE.Raycaster()
  private mouse = new THREE.Vector2()
  private camera: THREE.Camera
  private tooltip: HTMLDivElement
  private hoveredEntity: number | null = null

  constructor(camera: THREE.Camera) {
    this.camera = camera
    this.tooltip = this.createTooltip()
    window.addEventListener('mousemove', this.onMouseMove.bind(this))
    window.addEventListener('click', this.onClick.bind(this))
  }

  private createTooltip(): HTMLDivElement {
    const el = document.createElement('div')
    Object.assign(el.style, {
      position: 'absolute', padding: '8px 16px',
      background: 'rgba(0,0,0,0.85)', color: '#fff',
      borderRadius: '6px', fontSize: '13px',
      pointerEvents: 'none', display: 'none',
      backdropFilter: 'blur(4px)',
      border: '1px solid rgba(255,255,255,0.1)',
    })
    document.body.appendChild(el)
    return el
  }

  private onMouseMove(e: MouseEvent) {
    this.mouse.x = (e.clientX / window.innerWidth) * 2 - 1
    this.mouse.y = -(e.clientY / window.innerHeight) * 2 + 1
  }

  private onClick(e: MouseEvent) {
    if (this.hoveredEntity !== null) {
      // 触发选中事件
    }
  }

  update(world: World, dt: number): void {
    this.raycaster.setFromCamera(this.mouse, this.camera)
    const interactables = world.getStore<InteractableComponent>('interactable')
    const dataPoints = world.getStore<DataPointComponent>('dataPoint')

    let found = false
    for (const entity of interactables.entities()) {
      const interactable = interactables.get(entity)!
      const dp = dataPoints.get(entity)
      if (!dp?.barMesh) continue

      const intersects = this.raycaster.intersectObject(dp.barMesh)
      if (intersects.length > 0) {
        this.hoveredEntity = entity
        this.showTooltip(intersects[0], dp)
        dp.barMesh.material.emissiveIntensity = 0.8
        found = true
      } else {
        dp.barMesh.material.emissiveIntensity = 0.3
      }
    }

    if (!found) {
      this.hoveredEntity = null
      this.tooltip.style.display = 'none'
    }
  }

  private showTooltip(intersect: THREE.Intersection, dp: DataPointComponent) {
    const screen = intersect.point.clone().project(this.camera)
    const x = (screen.x * 0.5 + 0.5) * window.innerWidth
    const y = (-screen.y * 0.5 + 0.5) * window.innerHeight
    this.tooltip.style.display = 'block'
    this.tooltip.style.left = x + 15 + 'px'
    this.tooltip.style.top = y + 15 + 'px'
    this.tooltip.innerHTML = `
      <strong>${dp.label}</strong><br/>
      GDP: $${(dp.value * 100).toFixed(1)}T<br/>
      人口: ${(dp.value * 1400).toFixed(0)}M
    `
  }
}
```

## 音频背景与后处理

```typescript
class AudioSystem implements System {
  private audio: AudioContext | null = null
  private analyser: AnalyserNode | null = null
  private source: AudioBufferSourceNode | null = null
  uniforms: Record<string, THREE.IUniform> = {}

  async loadAndPlay(url: string): Promise<void> {
    this.audio = new AudioContext()
    this.analyser = this.audio.createAnalyser()
    this.analyser.fftSize = 256

    const response = await fetch(url)
    const buffer = await response.arrayBuffer()
    const audioBuffer = await this.audio.decodeAudioData(buffer)

    this.source = this.audio.createBufferSource()
    this.source.buffer = audioBuffer
    this.source.loop = true
    this.source.connect(this.analyser)
    this.analyser.connect(this.audio.destination)
    this.source.start()

    this.uniforms.u_audioLevel = { value: 0 }
  }

  update(world: World, dt: number): void {
    if (!this.analyser) return
    const data = new Uint8Array(this.analyser.frequencyBinCount)
    this.analyser.getByteFrequencyData(data)
    const avg = data.reduce((a, b) => a + b, 0) / data.length / 255
    this.uniforms.u_audioLevel.value = avg
  }
}

class PostProcessSystem implements System {
  private composer: EffectComposer

  constructor(renderer: THREE.WebGLRenderer, scene: THREE.Scene, camera: THREE.Camera) {
    this.composer = new EffectComposer(renderer)

    const renderPass = new RenderPass(scene, camera)
    this.composer.addPass(renderPass)

    const bloomPass = new UnrealBloomPass(
      new THREE.Vector2(window.innerWidth, window.innerHeight),
      0.5, 0.4, 0.85
    )
    this.composer.addPass(bloomPass)

    const toneMappingPass = new ShaderPass(ToneMappingShader)
    this.composer.addPass(toneMappingPass)
  }

  update(world: World, dt: number): void {
    this.composer.render()
  }

  resize(width: number, height: number): void {
    this.composer.setSize(width, height)
  }
}
```

## 组装应用

```typescript
class GlobeVisualizationApp {
  private app: Application
  private globeSystem!: GlobeRenderSystem
  private dataVizSystem!: DataVisualizationSystem
  private interactionSystem!: InteractionSystem
  private audioSystem!: AudioSystem
  private postProcessSystem!: PostProcessSystem
  private dataLoader = new DataLoader()

  constructor(container: HTMLElement) {
    this.app = new Application(container)
    this.setupSystems()
    this.setupLighting()
  }

  private setupSystems(): void {
    const { scene, camera, world } = this.app

    this.globeSystem = new GlobeRenderSystem(scene)
    this.dataVizSystem = new DataVisualizationSystem(scene)
    this.interactionSystem = new InteractionSystem(camera)
    this.audioSystem = new AudioSystem()
    this.postProcessSystem = new PostProcessSystem(
      this.app.renderer, scene, camera
    )

    world.addSystem(this.globeSystem)
    world.addSystem(this.dataVizSystem)
    world.addSystem(this.interactionSystem)
    world.addSystem(this.audioSystem)
    world.addSystem(this.postProcessSystem)
  }

  private setupLighting(): void {
    const { scene } = this.app
    scene.add(new THREE.AmbientLight(0xffffff, 0.3))
    const sun = new THREE.DirectionalLight(0xffffff, 1.2)
    sun.position.set(10, 5, 10)
    sun.castShadow = true
    scene.add(sun)
  }

  async init(): Promise<void> {
    const { world, resources } = this.app

    // 阶段 1: 加载地球
    await this.globeSystem.buildGlobe(world, resources)

    // 阶段 2: 加载数据
    const [geojson, countryData] = await Promise.all([
      this.dataLoader.loadGeoJSON('/data/countries.geojson'),
      this.dataLoader.loadCountryData('/data/gdp-population.json'),
    ])

    // 阶段 3: 处理数据（Worker）
    const dataPoints = await this.dataLoader.processGeoData(geojson, countryData)
    this.dataVizSystem.createBars(world, dataPoints)

    // 阶段 4: 背景音乐
    await this.audioSystem.loadAndPlay('/audio/ambient.mp3')

    // 阶段 5: 启动
    this.app.start()
  }

  dispose(): void {
    this.app.dispose()
  }
}
```

## 常见误区

1. **所有系统都在同一个 update 循环中执行**：不同系统的执行频率需求不同——渲染需要 60fps，但数据处理和音频分析可以 30fps 更新。应该允许系统配置不同的更新频率。

2. **大数据一次性全部添加到场景**：上千个数据点的柱状图应该用 InstancedMesh 渲染，或者根据视口裁剪只渲染可见区域的数据点。

3. **忽略移动端的触摸与鼠标事件冲突**：在移动设备上，touch 和 mouse 事件可能同时触发。应该用特性检测决定监听哪类事件，避免重复处理。

4. **后处理效果在低端设备上拖垮帧率**：Bloom、ToneMapping 等后处理会额外增加多遍全屏渲染。在移动端应该提供关闭后处理的选项，或者自动检测帧率并降级。

## 工程建议

1. **资源清单驱动加载**：将所有需要加载的资源定义在 AssetManifest 中，ResourceManager 根据清单批量加载，避免散落在各处的加载调用。

2. **实现完整的生命周期管理**：init → load → start → pause → resume → dispose，每个阶段都有明确的资源分配和释放逻辑，避免内存泄漏。

3. **数据驱动的可视化配置**：颜色映射、柱体高度比例、热力图半径等参数应该通过配置对象传入，而非硬编码。这样同一个框架可以复用到不同数据集。

4. **自动化验证脚本**：`check.cjs` 应该验证项目结构完整性（必要文件是否存在）、代码质量（无 TypeScript 错误）、基本功能（场景是否能在 headless 浏览器中渲染）。

### 练习二

**思路**：飞行线的核心是用 CatmullRomCurve3 生成从起点到终点的弧线路径——先将两国的经纬度转为球面 3D 坐标，然后在两端点之间插入一个高于球面的中间控制点，形成抛物线弧。线的粗细和颜色通过 LineMaterial（或自定义 Shader）映射贸易量。

**答案**：

```typescript
import * as THREE from 'three'

interface TradeRoute {
  from: { lat: number; lon: number; name: string }
  to: { lat: number; lon: number; name: string }
  volume: number
}

function latLonToVec3(lat: number, lon: number, r: number): THREE.Vector3 {
  const phi = (90 - lat) * Math.PI / 180
  const theta = (lon + 180) * Math.PI / 180
  return new THREE.Vector3(
    -r * Math.sin(phi) * Math.cos(theta),
    r * Math.cos(phi),
    r * Math.sin(phi) * Math.sin(theta)
  )
}

function createFlightLine(route: TradeRoute, globeGroup: THREE.Group, maxVolume: number) {
  const start = latLonToVec3(route.from.lat, route.from.lon, 1.01)
  const end = latLonToVec3(route.to.lat, route.to.lon, 1.01)

  const mid = start.clone().add(end).multiplyScalar(0.5)
  const dist = start.distanceTo(end)
  mid.normalize().multiplyScalar(1 + dist * 0.3)

  const curve = new THREE.CatmullRomCurve3([start, mid, end])
  const points = curve.getPoints(50)

  const lineGeo = new THREE.BufferGeometry().setFromPoints(points)

  const normalizedVolume = route.volume / maxVolume
  const lineMat = new THREE.LineBasicMaterial({
    color: new THREE.Color().setHSL(0.6 - normalizedVolume * 0.4, 0.9, 0.5 + normalizedVolume * 0.3),
    transparent: true,
    opacity: 0.3 + normalizedVolume * 0.5,
    linewidth: 1
  })

  const line = new THREE.Line(lineGeo, lineMat)
  globeGroup.add(line)

  const dotGeo = new THREE.SphereGeometry(0.015, 8, 8)
  const dotMat = new THREE.MeshBasicMaterial({ color: 0x00ffff })
  const dot = new THREE.Mesh(dotGeo, dotMat)
  globeGroup.add(dot)

  return { line, dot, curve }
}

  let t = 0
  function update() {
    requestAnimationFrame(update)
    t = (t + 0.002) % 1
    animatables.forEach(({ dot, curve }, i) => {
      const offset = (t + i * 0.1) % 1
      const pos = curve.getPoint(offset)
      dot.position.copy(pos)
    })
  }
  update()
}
```

### 练习三

**思路**：数据过滤功能需要一个 UI 面板（lil-gui）暴露过滤条件（大洲、GDP 范围、人口范围），过滤时遍历所有柱体 Mesh，根据 userData 中的数据判断是否显示，显示/隐藏用动画（scale 从 0 到 1 的 lerp）实现平滑过渡。

**答案**：

```typescript
import * as THREE from 'three'
import GUI from 'lil-gui'

interface FilterState {
  continent: string
  gdpMin: number
  gdpMax: number
  popMin: number
  popMax: number
}

interface AnimatedBar {
  mesh: THREE.Mesh
  targetScale: number
  currentScale: number
}

function createFilterPanel(barMeshes: THREE.Mesh[], data: any[]) {
  const state: FilterState = {
    continent: '全部',
    gdpMin: 0, gdpMax: 30,
    popMin: 0, popMax: 2e9
  }

  const animatedBars: AnimatedBar[] = barMeshes.map(mesh => ({
    mesh,
    targetScale: 1,
    currentScale: 1
  }))

  const continents = ['全部', ...new Set(data.map((d: any) => d.continent))]

  const gui = new GUI({ title: '数据过滤' })
  gui.add(state, 'continent', continents).name('大洲').onChange(applyFilter)
  gui.add(state, 'gdpMin', 0, 30, 0.1).name('GDP 最小值 (万亿)').onChange(applyFilter)
  gui.add(state, 'gdpMax', 0, 30, 0.1).name('GDP 最大值 (万亿)').onChange(applyFilter)
  gui.add(state, 'popMin', 0, 2e9, 1e7).name('人口最小值').onChange(applyFilter)
  gui.add(state, 'popMax', 0, 2e9, 1e7).name('人口最大值').onChange(applyFilter)

  function applyFilter() {
    animatedBars.forEach(ab => {
      const d = ab.mesh.userData
      const continentMatch = state.continent === '全部' || d.continent === state.continent
      const gdpMatch = d.gdp >= state.gdpMin && d.gdp <= state.gdpMax
      const popMatch = d.population >= state.popMin && d.population <= state.popMax
      ab.targetScale = (continentMatch && gdpMatch && popMatch) ? 1 : 0
    })
  }

  function update(dt: number) {
    animatedBars.forEach(ab => {
      ab.currentScale += (ab.targetScale - ab.currentScale) * Math.min(dt * 8, 1)
      if (ab.currentScale < 0.01) {
        ab.mesh.visible = false
      } else {
        ab.mesh.visible = true
        ab.mesh.scale.set(1, ab.currentScale, 1)
      }
    })
  }

  return { update, gui }
}
```

**要点**：
- 过滤逻辑是纯数据判断，通过 `userData` 读取柱体绑定的数据
- 动画用指数插值（`currentScale += (target - current) * speed`）实现平滑的出现/消失效果
- `visible = false` 在 scale < 0.01 时设置，避免渲染不可见的物体浪费 GPU
- lil-gui 的 `onChange` 回调即时触发过滤，用户拖动滑块时实时看到效果

**要点**：
- `manualChunks` 将 Three.js 拆为独立 chunk（~600KB），浏览器缓存后后续访问不再重复下载
- `Cache-Control: immutable` 配合 content hash 文件名，资源更新时自动失效
- OG 标签让社交平台（微信、Twitter）在分享链接时显示预览图和描述
- `drop_console: true` 在生产构建中移除所有 console.log，减少包体积和信息泄露
