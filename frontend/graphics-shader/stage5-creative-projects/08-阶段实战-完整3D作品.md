### 音频分析

```typescript
class AudioAnalyzer {
  private analyser: AnalyserNode
  private frequencyData: Uint8Array
  private timeDomainData: Uint8Array

  constructor(audioContext: AudioContext) {
    this.analyser = audioContext.createAnalyser()
    this.analyser.fftSize = 2048
    this.frequencyData = new Uint8Array(this.analyser.frequencyBinCount)
    this.timeDomainData = new Uint8Array(this.analyser.frequencyBinCount)
  }

  connect(source: AudioNode) {
    source.connect(this.analyser)
  }

  update() {
    this.analyser.getByteFrequencyData(this.frequencyData)
    this.analyser.getByteTimeDomainData(this.timeDomainData)
  }

  // 获取频段能量
  getBandEnergy(startHz: number, endHz: number): number {
    const sampleRate = 44100
    const binSize = sampleRate / this.analyser.fftSize
    const startBin = Math.floor(startHz / binSize)
    const endBin = Math.floor(endHz / binSize)

    let sum = 0
    for (let i = startBin; i < endBin; i++) {
      sum += this.frequencyData[i]
    }
    return sum / (endBin - startBin) / 255
  }

  get bass() { return this.getBandEnergy(20, 250) }
  get mid() { return this.getBandEnergy(250, 4000) }
  get high() { return this.getBandEnergy(4000, 20000) }
  get overall() { return this.getBandEnergy(20, 20000) }
}
```

### 3D 场景

```typescript
class MusicVisualizer {
  private scene: THREE.Scene
  private camera: THREE.PerspectiveCamera
  private renderer: THREE.WebGLRenderer
  private particles: GPUParticleSystem
  private analyzer: AudioAnalyzer
  private bloomPass: UnrealBloomPass

  constructor(container: HTMLElement) {
    this.scene = new THREE.Scene()
    this.camera = new THREE.PerspectiveCamera(60, container.clientWidth/container.clientHeight, 0.1, 100)
    this.camera.position.set(0, 2, 5)

    this.renderer = new THREE.WebGLRenderer({ antialias: true })
    this.renderer.setSize(container.clientWidth, container.clientHeight)
    container.appendChild(this.renderer.domElement)

    // 粒子系统
    this.particles = new GPUParticleSystem(100000)
    this.scene.add(this.particles.getObject())

    // 后处理
    const composer = new EffectComposer(this.renderer)
    composer.addPass(new RenderPass(this.scene, this.camera))
    this.bloomPass = new UnrealBloomPass(
      new THREE.Vector2(container.clientWidth, container.clientHeight),
      0.5, 0.4, 0.85
    )
    composer.addPass(this.bloomPass)
    composer.addPass(new OutputPass())

    this.animate()
  }

  private animate() {
    requestAnimationFrame(() => this.animate())

    this.analyzer.update()

    // 音频驱动视觉
    const bass = this.analyzer.bass
    const mid = this.analyzer.mid
    const high = this.analyzer.high

    // 粒子大小随低频脉动
    this.particles.material.uniforms.u_bass.value = bass

    // Bloom 强度随整体能量变化
    this.bloomPass.strength = 0.3 + bass * 0.7

    // 相机轻微晃动
    this.camera.position.x = Math.sin(Date.now() * 0.001) * mid * 0.5
    this.camera.position.y = 2 + Math.sin(Date.now() * 0.0015) * high * 0.3

    this.particles.update(1/60)
    this.composer.render()
  }
}
```

## 你可能踩的坑

**坑一：音频上下文需要用户交互**

浏览器要求用户交互后才能创建 AudioContext。在点击事件中初始化。

**坑二：频谱数据更新频率**

`getByteFrequencyData` 每帧调用就够了。不需要更频繁。

**坑三：视觉效果太花哨**

音频可视化的核心是"音乐的视觉表达"，不是"越花越好"。效果要和音乐情绪匹配。

```typescript
class BeatDetector {
  private lastBass = 0
  private threshold = 0.6
  private cooldown = 0

  detect(bass: number): boolean {
    if (this.cooldown > 0) {
      this.cooldown--
      return false
    }

    if (bass > this.threshold && bass > this.lastBass * 1.2) {
      this.cooldown = 10 // 10 帧冷却
      this.lastBass = bass
      return true
    }

    this.lastBass = bass
    return false
  }
}
```
