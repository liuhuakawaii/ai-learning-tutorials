# 发布与部署

## 场景引入

你精心构建的 3D 作品在本地开发机上运行完美，但部署到线上后问题接踵而来：手机用户反馈页面白屏，低端设备帧率只有个位数，搜索引擎完全抓取不到内容，首屏加载耗时 30 秒。本课将系统学习 3D Web 应用从开发到上线的全流程——响应式适配、移动端优化、CDN 分发、Web Worker 计算、渐进式加载、SEO 优化、性能预算和 CI/CD 集成。

## 响应式 3D：Canvas 适配

### 基础适配方案

```
Canvas 适配三要素：

1. CSS 尺寸  → canvas 占据的页面空间
2. 分辨率    → canvas 内部像素数（devicePixelRatio）
3. 宽高比    → camera.aspect 必须同步更新

  ┌─────────────────────────┐
  │       viewport          │
  │  ┌─────────────────┐    │
  │  │     canvas       │    │
  │  │  (width × height) │    │
  │  │                  │    │
  │  └─────────────────┘    │
  │         ↕               │
  │  CSS: 100vw × 100vh     │
  │  GL:   × pixelRatio     │
  └─────────────────────────┘
```

```typescript
class ResponsiveCanvas {
  private renderer: THREE.WebGLRenderer
  private camera: THREE.PerspectiveCamera
  private container: HTMLElement

  constructor(container: HTMLElement) {
    this.container = container
    this.renderer = new THREE.WebGLRenderer({
      antialias: true,
      powerPreference: 'high-performance',
    })
    this.camera = new THREE.PerspectiveCamera(60, 1, 0.1, 1000)

    container.appendChild(this.renderer.domElement)
    this.resize()

    // ResizeObserver 比 resize 事件更精确
    const ro = new ResizeObserver(() => this.resize())
    ro.observe(container)
  }

  resize(): void {
    const rect = this.container.getBoundingClientRect()
    const width = rect.width
    const height = rect.height
    const pixelRatio = Math.min(window.devicePixelRatio, 2)

    this.renderer.setSize(width, height, false)
    this.renderer.setPixelRatio(pixelRatio)

    this.camera.aspect = width / height
    this.camera.updateProjectionMatrix()

    // 实际 canvas 分辨率
    // canvas.width = width * pixelRatio
    // canvas.height = height * pixelRatio
    // canvas.style.width = width + 'px'
    // canvas.style.height = height + 'px'
  }
}
```

## 移动端适配

### 触控事件处理

```typescript
class TouchInput {
  private touches = new Map<number, { x: number; y: number }>()
  private pinchDistance = 0
  private callbacks = {
    move: [] as Array<(dx: number, dy: number) => void>,
    pinch: [] as Array<(scale: number) => void>,
    tap: [] as Array<(x: number, y: number) => void>,
  }

  constructor(element: HTMLElement) {
    element.addEventListener('touchstart', this.onTouchStart.bind(this), { passive: false })
    element.addEventListener('touchmove', this.onTouchMove.bind(this), { passive: false })
    element.addEventListener('touchend', this.onTouchEnd.bind(this))
  }

  private onTouchStart(e: TouchEvent) {
    e.preventDefault()
    for (const touch of Array.from(e.changedTouches)) {
      this.touches.set(touch.identifier, { x: touch.clientX, y: touch.clientY })
    }
    if (e.touches.length === 2) {
      this.pinchDistance = this.getPinchDistance(e.touches)
    }
  }

  private onTouchMove(e: TouchEvent) {
    e.preventDefault()
    if (e.touches.length === 1) {
      const touch = e.touches[0]
      const prev = this.touches.get(touch.identifier)
      if (prev) {
        const dx = touch.clientX - prev.x
        const dy = touch.clientY - prev.y
        this.callbacks.move.forEach(fn => fn(dx, dy))
        this.touches.set(touch.identifier, { x: touch.clientX, y: touch.clientY })
      }
    } else if (e.touches.length === 2) {
      const newDist = this.getPinchDistance(e.touches)
      const scale = newDist / this.pinchDistance
      this.callbacks.pinch.forEach(fn => fn(scale))
      this.pinchDistance = newDist
    }
  }

  private onTouchEnd(e: TouchEvent) {
    for (const touch of Array.from(e.changedTouches)) {
      this.touches.delete(touch.identifier)
    }
  }

  private getPinchDistance(touches: TouchList): number {
    const dx = touches[0].clientX - touches[1].clientX
    const dy = touches[0].clientY - touches[1].clientY
    return Math.sqrt(dx * dx + dy * dy)
  }

  onMove(fn: (dx: number, dy: number) => void) { this.callbacks.move.push(fn) }
  onPinch(fn: (scale: number) => void) { this.callbacks.pinch.push(fn) }
  onTap(fn: (x: number, y: number) => void) { this.callbacks.tap.push(fn) }
}
```

### 陀螺仪控制相机

```typescript
class GyroCamera {
  private camera: THREE.PerspectiveCamera
  private quaternion = new THREE.Quaternion()
  private enabled = false

  constructor(camera: THREE.PerspectiveCamera) {
    this.camera = camera
  }

  async requestPermission(): Promise<boolean> {
    if (typeof DeviceOrientationEvent !== 'undefined' &&
        'requestPermission' in DeviceOrientationEvent) {
      const permission = await (DeviceOrientationEvent as any).requestPermission()
      return permission === 'granted'
    }
    return true
  }

  enable(): void {
    window.addEventListener('deviceorientation', this.onOrientation.bind(this))
    this.enabled = true
  }

  private onOrientation(event: DeviceOrientationEvent) {
    if (!this.enabled) return
    const { alpha, beta, gamma } = event
    if (alpha === null || beta === null || gamma === null) return

    const euler = new THREE.Euler(
      THREE.MathUtils.degToRad(beta),
      THREE.MathUtils.degToRad(alpha),
      -THREE.MathUtils.degToRad(gamma),
      'YXZ'
    )
    this.quaternion.setFromEuler(euler)
    this.camera.quaternion.copy(this.quaternion)
  }
}
```

### Pixel Ratio 与画质自适应

```typescript
class AdaptiveQuality {
  private currentRatio: number
  private fpsHistory: number[] = []
  private targetFPS: number

  constructor(targetFPS: number = 60) {
    this.targetFPS = targetFPS
    this.currentRatio = Math.min(window.devicePixelRatio, 2)
  }

  update(currentFPS: number, renderer: THREE.WebGLRenderer): void {
    this.fpsHistory.push(currentFPS)
    if (this.fpsHistory.length < 30) return

    const avgFPS = this.fpsHistory.reduce((a, b) => a + b) / this.fpsHistory.length
    this.fpsHistory = []

    if (avgFPS < this.targetFPS * 0.85) {
      // 帧率过低，降低分辨率
      this.currentRatio = Math.max(0.5, this.currentRatio - 0.25)
      renderer.setPixelRatio(this.currentRatio)
      console.log(`Quality down: pixelRatio=${this.currentRatio}`)
    } else if (avgFPS > this.targetFPS * 0.95 && this.currentRatio < window.devicePixelRatio) {
      // 帧率充足，提高分辨率
      this.currentRatio = Math.min(window.devicePixelRatio, this.currentRatio + 0.25)
      renderer.setPixelRatio(this.currentRatio)
    }
  }
}
```

## Web Worker 离屏计算

```typescript
// worker.ts - 数据处理 Worker
self.onmessage = (e: MessageEvent) => {
  const { type, data } = e.data

  switch (type) {
    case 'process-geojson': {
      const features = data.features
      const processed = features.map((f: any) => ({
        lat: f.geometry.coordinates[1],
        lon: f.geometry.coordinates[0],
        value: f.properties.value,
      }))
      self.postMessage({ type: 'geojson-ready', data: processed })
      break
    }
    case 'compute-noise': {
      const { width, height, scale } = data
      const noise = new Float32Array(width * height)
      for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
          noise[y * width + x] = simplex2D(x * scale, y * scale)
        }
      }
      self.postMessage({ type: 'noise-ready', data: noise }, [noise.buffer])
      break
    }
  }
}

function simplex2D(x: number, y: number): number {
  return (Math.sin(x * 1.5) * Math.cos(y * 1.5) + 1) * 0.5
}

// main.ts - 主线程
class WorkerPool {
  private workers: Worker[] = []
  private taskQueue: Array<{ resolve: Function; reject: Function; data: any }> = []
  private busy = new Set<Worker>()

  constructor(workerUrl: string, poolSize: number = navigator.hardwareConcurrency) {
    for (let i = 0; i < poolSize; i++) {
      const worker = new Worker(workerUrl, { type: 'module' })
      worker.onmessage = this.onWorkerMessage.bind(this, worker)
      this.workers.push(worker)
    }
  }

  private onWorkerMessage(worker: Worker, e: MessageEvent) {
    this.busy.delete(worker)
    const task = this.taskQueue.shift()
    task?.resolve(e.data)
    this.processQueue()
  }

  execute<T>(data: any): Promise<T> {
    return new Promise((resolve, reject) => {
      this.taskQueue.push({ resolve, reject, data })
      this.processQueue()
    })
  }

  private processQueue(): void {
    if (this.taskQueue.length === 0) return
    const idleWorker = this.workers.find(w => !this.busy.has(w))
    if (!idleWorker) return
    this.busy.add(idleWorker)
    idleWorker.postMessage(this.taskQueue[this.taskQueue.length - 1].data)
  }

  terminate(): void {
    this.workers.forEach(w => w.terminate())
  }
}
```

## 渐进式加载

```
渐进式加载策略：

阶段 1: 首屏 (< 2s)
  ├─ 加载低精度模型 (10% 面数)
  ├─ 加载低分辨率纹理 (256px)
  └─ 显示加载进度条

阶段 2: 基础场景 (2-5s)
  ├─ 加载中精度模型 (50% 面数)
  ├─ 加载中分辨率纹理 (1024px)
  └─ 启用基础光照

阶段 3: 完整场景 (5-15s, 后台)
  ├─ 加载高精度模型
  ├─ 加载高分辨率纹理 (4K)
  ├─ 启用阴影和后处理
  └─ 预加载交互资源
```

```typescript
class ProgressiveLoader {
  private stages: Array<{
    name: string
    priority: number
    load: () => Promise<void>
  }> = []
  private loadedStages = new Set<string>()

  addStage(name: string, priority: number, load: () => Promise<void>): void {
    this.stages.push({ name, priority, load })
    this.stages.sort((a, b) => a.priority - b.priority)
  }

  async loadEssential(onProgress?: (progress: number) => void): Promise<void> {
    const essential = this.stages.filter(s => s.priority <= 1)
    let loaded = 0

    for (const stage of essential) {
      await stage.load()
      this.loadedStages.add(stage.name)
      loaded++
      onProgress?.(loaded / essential.length)
    }
  }

  async loadBackground(): Promise<void> {
    const remaining = this.stages.filter(s => !this.loadedStages.has(s.name))

    for (const stage of remaining) {
      await stage.load()
      this.loadedStages.add(stage.name)
    }
  }

  loadWithPriority(minPriority: number): Promise<void> {
    const stages = this.stages.filter(
      s => s.priority <= minPriority && !this.loadedStages.has(s.name)
    )
    return Promise.all(stages.map(s => s.load().then(() => {
      this.loadedStages.add(s.name)
    }))) as any
  }
}

// 使用示例
const loader = new ProgressiveLoader()

loader.addStage('low-res-model', 0, async () => {
  const gltf = await resources.load('gltf', '/models/scene-low.glb')
  scene.add(gltf.scene)
})

loader.addStage('low-res-textures', 0, async () => {
  const tex = await resources.load('texture', '/textures/env-256.webp')
  scene.environment = tex
})

loader.addStage('high-res-model', 2, async () => {
  const gltf = await resources.load('gltf', '/models/scene-high.glb')
  // 替换低精度模型
})

// 首屏加载
await loader.loadEssential(progress => updateProgressBar(progress))
startRenderLoop()

// 后台继续加载
loader.loadBackground()
```

```typescript
import puppeteer from 'puppeteer'

async function performanceTest() {
  const browser = await puppeteer.launch({ headless: false })
  const page = await browser.newPage()

  await page.goto('http://localhost:3000', { waitUntil: 'networkidle0' })

  // 测量加载时间
  const timing = await page.evaluate(() => {
    const perf = performance.getEntriesByType('navigation')[0] as PerformanceNavigationTiming
    return {
      domContentLoaded: perf.domContentLoadedEventEnd - perf.startTime,
      loadComplete: perf.loadEventEnd - perf.startTime,
    }
  })

  // 测量 FPS
  const fps = await page.evaluate(() => {
    return new Promise<number>(resolve => {
      let frames = 0
      const start = performance.now()
      const count = () => {
        frames++
        if (performance.now() - start >= 2000) {
          resolve(frames / 2)
        } else {
          requestAnimationFrame(count)
        }
      }
      requestAnimationFrame(count)
    })
  })

  console.log(`Load: ${timing.loadComplete.toFixed(0)}ms`)
  console.log(`FPS: ${fps.toFixed(1)}`)

  if (timing.loadComplete > 3000) {
    console.error('FAIL: Load time exceeds budget')
    process.exit(1)
  }
  if (fps < 55) {
    console.error('FAIL: FPS below budget')
    process.exit(1)
  }

  await browser.close()
}
```

## 常见误区

1. **Canvas 不设置 `touch-action: none`**：移动端浏览器的默认触摸行为（滚动、缩放）会干扰 Canvas 的触控事件。必须在 CSS 中设置 `touch-action: none`，同时在 JS 中 `preventDefault()`。

2. **所有资源都走同一个 CDN 域名**：浏览器对同一域名的并发连接数有限制（HTTP/1.1 下约 6 个）。大模型和纹理文件可以分散到多个子域名（assets1.example.com, assets2.example.com），提高并行下载效率。

3. **忽略 `og:image` 导致社交分享无预览**：3D 应用的页面在社交平台分享时，如果没有 OG 标签或预览图，会显示空白或默认图标。应该用 Puppeteer 预渲染一张 1200×630 的预览图。

4. **Web Worker 中直接操作 THREE.js 对象**：Three.js 的对象（Mesh、Texture 等）不能通过 `postMessage` 传递，因为它们包含不可序列化的 WebGL 引用。Worker 中只处理纯数据（数组、JSON），结果传回主线程后构建 Three.js 对象。

## 工程建议

1. **用 Service Worker 缓存 3D 资源**：模型和纹理文件不会频繁更新，用 Service Worker 的 Cache API 缓存后，二次访问可以实现秒开。配合 content hash 文件名，更新时自动失效。

2. **实现加载骨架屏**：在 3D 场景加载完成前，显示一个 CSS 骨架屏或低质量的 Canvas 预览（如纯色背景 + Logo），避免用户面对白屏等待。

3. **分环境配置**：开发环境用本地资源和完整日志，staging 环境用 CDN 资源和性能监控，生产环境用压缩资源和错误追踪。用环境变量控制行为差异。

4. **监控真实用户的性能数据**：实验室数据（Lighthouse）只能做参考，必须采集真实用户的 RUM（Real User Monitoring）数据——FCP、LCP、FPS 在不同设备、不同网络下的分布，才能做出正确的优化决策。

## 小结

本课覆盖了 3D Web 应用从开发到上线的全流程。响应式 Canvas 适配确保在任何尺寸下都正确渲染；移动端的触控、陀螺仪、自适应画质保证移动体验；CDN 分发和 content hash 缓存策略优化加载速度；Web Worker 将重计算移出主线程；渐进式加载让用户尽快看到内容；SEO 和 OG 标签让 3D 内容可被搜索引擎和社交平台发现；性能预算防止退化；CI/CD 自动化确保每次部署都经过质量检查。

## 练习

1. 实现一个完整的响应式 3D 场景：支持窗口缩放、移动端触控旋转、双指缩放、陀螺仪控制（可选），在不同设备上测试适配效果。
2. 用 Web Worker 实现一个数据处理管线：在 Worker 中解析大型 GeoJSON 文件（> 10MB），计算聚合统计，将结果传回主线程渲染为 3D 柱状图。
3. 实现渐进式加载：准备一个 3D 场景的低精度和高精度版本，先加载低精度版本在 2 秒内显示，后台加载高精度版本并无缝替换。
4. 搭建一个完整的 CI/CD 流水线：代码推送后自动运行 lint、typecheck、build、性能测试，通过后自动部署到静态托管服务（如 Vercel/Netlify）。

### 练习一

**思路**：响应式 3D 场景需要处理三件事：窗口缩放（resize 事件更新 camera.aspect 和 renderer.setSize）、触控旋转（TouchControls 或手动处理 touchmove）、双指缩放（计算两个 touch 点的距离变化）。陀螺仪通过 DeviceOrientation API 获取。

**答案**：

```typescript
import * as THREE from 'three'
import { OrbitControls } from 'three/addons/controls/OrbitControls.js'

function createResponsiveScene() {
  const scene = new THREE.Scene()
  scene.background = new THREE.Color(0x1a1a2e)
  const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 100)
  camera.position.set(0, 3, 8)

  const renderer = new THREE.WebGLRenderer({ antialias: true })
  renderer.setSize(window.innerWidth, window.innerHeight)
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2))
  document.body.appendChild(renderer.domElement)

  const controls = new OrbitControls(camera, renderer.domElement)
  controls.enableDamping = true
  controls.enableZoom = true
  controls.enablePan = true
  controls.touches = {
    ONE: THREE.TOUCH.ROTATE,
    TWO: THREE.TOUCH.DOLLY_PAN
  }

  scene.add(new THREE.AmbientLight(0xffffff, 0.5))
  scene.add(new THREE.DirectionalLight(0xffffff, 0.8))

  const cubeGeo = new THREE.BoxGeometry(1, 1, 1)
  const cubeMat = new THREE.MeshStandardMaterial({ color: 0x4fc3f7 })
  for (let i = 0; i < 20; i++) {
    const cube = new THREE.Mesh(cubeGeo, cubeMat.clone())
    cube.material.color.setHSL(Math.random(), 0.7, 0.5)
    cube.position.set((Math.random() - 0.5) * 8, Math.random() * 3, (Math.random() - 0.5) * 8)
    scene.add(cube)
  }

  const floor = new THREE.Mesh(
    new THREE.PlaneGeometry(20, 20),
    new THREE.MeshStandardMaterial({ color: 0x333344 })
  )
  floor.rotation.x = -Math.PI / 2
  scene.add(floor)

  let gyroEnabled = false
  const gyroBtn = document.createElement('button')
  gyroBtn.textContent = '启用陀螺仪'
  gyroBtn.style.cssText = 'position:fixed;top:10px;right:10px;padding:8px 16px;z-index:10;'
  document.body.appendChild(gyroBtn)

  gyroBtn.addEventListener('click', async () => {
    if (typeof (DeviceOrientationEvent as any).requestPermission === 'function') {
      try {
        const response = await (DeviceOrientationEvent as any).requestPermission()
        if (response === 'granted') {
          enableGyroscope()
        }
      } catch (e) {
        alert('陀螺仪权限被拒绝')
      }
    } else {
      enableGyroscope()
    }
  })

  function enableGyroscope() {
    gyroEnabled = true
    controls.enabled = false
    gyroBtn.textContent = '陀螺仪已启用（点击关闭）'
    gyroBtn.onclick = () => {
      gyroEnabled = false
      controls.enabled = true
      gyroBtn.textContent = '启用陀螺仪'
    }
  }

  const gyroEuler = { alpha: 0, beta: 0, gamma: 0 }
  window.addEventListener('deviceorientation', (e) => {
    if (!gyroEnabled) return
    gyroEuler.alpha = (e.alpha || 0) * Math.PI / 180
    gyroEuler.beta = (e.beta || 0) * Math.PI / 180
    gyroEuler.gamma = (e.gamma || 0) * Math.PI / 180
  })

  window.addEventListener('resize', () => {
    camera.aspect = window.innerWidth / window.innerHeight
    camera.updateProjectionMatrix()
    renderer.setSize(window.innerWidth, window.innerHeight)
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2))
  })

  function animate() {
    requestAnimationFrame(animate)

    if (gyroEnabled) {
      const q = new THREE.Quaternion()
      const euler = new THREE.Euler(gyroEuler.beta, gyroEuler.alpha, -gyroEuler.gamma, 'YXZ')
      q.setFromEuler(euler)
      camera.quaternion.copy(q)
    }

    controls.update()
    renderer.render(scene, camera)
  }
  animate()
}

```

**要点**：
- OrbitControls 内置触控支持，`touches` 配置单指旋转、双指缩放/平移
- iOS 13+ 要求陀螺仪必须先调用 `requestPermission()` 获取用户授权
- `devicePixelRatio` 用 `Math.min(navigator.devicePixelRatio, 2)` 限制，避免 3x/4x 屏幕导致渲染负担过重
- 陀螺仪和 OrbitControls 互斥——启用陀螺仪时需禁用 controls，反之亦然

### 练习二

**思路**：Web Worker 处理大型 GeoJSON 的核心流程：主线程将文件 ArrayBuffer 发送给 Worker → Worker 解析 JSON、计算聚合统计 → Worker 将结果（Float32Array）通过 Transferable 传回主线程 → 主线程用结果创建 3D 柱状图。Transferable 避免了数据拷贝。

**答案**：

```typescript
// worker.ts
self.onmessage = (e: MessageEvent) => {
  const { buffer } = e.data as { buffer: ArrayBuffer }
  const text = new TextDecoder().decode(buffer)
  const geojson = JSON.parse(text)

  const stats = new Map<string, { count: number; totalValue: number; lat: number; lon: number }>()

  geojson.features.forEach((feature: any) => {
    const props = feature.properties
    const region = props.region || props.name || 'Unknown'
    const value = props.value || props.gdp || props.population || 0
    const coords = feature.geometry?.coordinates || [0, 0]

    if (!stats.has(region)) {
      stats.set(region, { count: 0, totalValue: 0, lat: coords[1], lon: coords[0] })
    }
    const s = stats.get(region)!
    s.count++
    s.totalValue += value
  })

  const regions = Array.from(stats.entries())
  const resultData = new Float32Array(regions.length * 5)
  regions.forEach(([name, s], i) => {
    resultData[i * 5] = s.lat
    resultData[i * 5 + 1] = s.lon
    resultData[i * 5 + 2] = s.totalValue / s.count
    resultData[i * 5 + 3] = s.count
    resultData[i * 5 + 4] = s.totalValue
  })

  const nameStrings = regions.map(([name]) => name)

  self.postMessage(
    { resultData, names: nameStrings, regionCount: regions.length },
    [resultData.buffer]
  )
}

// main.ts

async function loadDataAndRender() {
  const scene = new THREE.Scene()

  camera.position.set(0, 20, 30)

  const renderer = new THREE.WebGLRenderer({ antialias: true })
  renderer.setSize(window.innerWidth, window.innerHeight)
  document.body.appendChild(renderer.domElement)

  const controls = new OrbitControls(camera, renderer.domElement)
  controls.enableDamping = true

  scene.add(new THREE.AmbientLight(0xffffff, 0.5))
  scene.add(new THREE.DirectionalLight(0xffffff, 0.8))

  const statusEl = document.createElement('div')
  statusEl.style.cssText = 'position:fixed;top:10px;left:10px;color:#fff;font-size:14px;z-index:10;'
  statusEl.textContent = '加载中...'
  document.body.appendChild(statusEl)

  const response = await fetch('/data/large-dataset.geojson')
  const buffer = await response.arrayBuffer()
  statusEl.textContent = 'Worker 解析中...'

  const worker = new Worker(new URL('./worker.ts', import.meta.url), { type: 'module' })

  worker.onmessage = (e) => {
    const { resultData, names, regionCount } = e.data
    statusEl.textContent = `已加载 ${regionCount} 个区域`

    const maxVal = Math.max(...Array.from({ length: regionCount }, (_, i) => resultData[i * 5 + 4]))

    for (let i = 0; i < regionCount; i++) {
      const lat = resultData[i * 5]
      const lon = resultData[i * 5 + 1]
      const avgValue = resultData[i * 5 + 2]
      const totalValue = resultData[i * 5 + 4]

      const phi = (90 - lat) * Math.PI / 180
      const theta = (lon + 180) * Math.PI / 180
      const r = 5
      const x = -r * Math.sin(phi) * Math.cos(theta)
      const y = r * Math.cos(phi)
      const z = r * Math.sin(phi) * Math.sin(theta)

      const height = (totalValue / maxVal) * 3
      const geo = new THREE.CylinderGeometry(0.05, 0.05, height, 8)
      const mat = new THREE.MeshStandardMaterial({
        color: new THREE.Color().setHSL(totalValue / maxVal * 0.3, 0.8, 0.5)
      })
      const bar = new THREE.Mesh(geo, mat)
      const normal = new THREE.Vector3(x, y, z).normalize()
      bar.position.copy(normal.clone().multiplyScalar(r + height / 2))
      bar.lookAt(0, 0, 0)
      bar.rotateX(Math.PI / 2)
      bar.userData = { name: names[i], avgValue, totalValue }
      scene.add(bar)
    }

    const globe = new THREE.Mesh(
      new THREE.SphereGeometry(r, 64, 64),
      new THREE.MeshStandardMaterial({ color: 0x1a237e, roughness: 0.8 })
    )
    scene.add(globe)
  }

  worker.postMessage({ buffer }, [buffer])

  window.addEventListener('resize', () => {
    camera.aspect = window.innerWidth / window.innerHeight
    camera.updateProjectionMatrix()
    renderer.setSize(window.innerWidth, window.innerHeight)
  })

  function animate() {
    requestAnimationFrame(animate)
    controls.update()
    renderer.render(scene, camera)
  }
  animate()
}
```

**要点**：
- Transferable（第二个参数 `[buffer]`）将 ArrayBuffer 的所有权转移给目标线程，零拷贝
- Worker 中不能访问 DOM、THREE 或任何主线程对象，只能做纯计算
- 聚合统计（按区域分组求均值/总和）是典型的数据密集型任务，放在 Worker 中避免阻塞渲染
- Worker 的 `type: 'module'` 支持在现代浏览器中使用 ES Module 语法

### 练习三

**思路**：渐进式加载的核心是先加载低精度资源（低多边形模型 + 低分辨率纹理）快速显示场景，后台异步加载高精度资源，加载完毕后无缝替换。替换时需要保持相同的 Transform（位置、旋转、缩放）。

**答案**：

```typescript
import * as THREE from 'three'
import { OrbitControls } from 'three/addons/controls/OrbitControls.js'

class ProgressiveLoader {
  private scene: THREE.Scene
  private highResMeshes: Map<string, THREE.Mesh> = new Map()
  private lowResMeshes: Map<string, THREE.Mesh> = new Map()

  constructor(scene: THREE.Scene) {
    this.scene = scene
  }

  async loadLowRes(url: string, name: string): Promise<THREE.Mesh> {
    const { GLTFLoader } = await import('three/addons/loaders/GLTFLoader.js')
    const loader = new GLTFLoader()
    return new Promise((resolve, reject) => {
      loader.load(url, (gltf) => {
        const mesh = gltf.scene.children[0] as THREE.Mesh
        mesh.name = name
        this.scene.add(mesh)
        this.lowResMeshes.set(name, mesh)
        resolve(mesh)
      }, undefined, reject)
    })
  }

  async loadHighRes(url: string, name: string): Promise<void> {
    const { GLTFLoader } = await import('three/addons/loaders/GLTFLoader.js')
    const loader = new GLTFLoader()
    return new Promise((resolve, reject) => {
      loader.load(url, (gltf) => {
        const mesh = gltf.scene.children[0] as THREE.Mesh
        mesh.name = name

        const lowRes = this.lowResMeshes.get(name)
        if (lowRes) {
          mesh.position.copy(lowRes.position)
          mesh.rotation.copy(lowRes.rotation)
          mesh.scale.copy(lowRes.scale)
          mesh.visible = true
          this.scene.remove(lowRes)
          lowRes.geometry.dispose()
          if (Array.isArray(lowRes.material)) {
            lowRes.material.forEach(m => m.dispose())
          } else {
            lowRes.material.dispose()
          }
          this.lowResMeshes.delete(name)
        }

        this.scene.add(mesh)
        this.highResMeshes.set(name, mesh)
        resolve()
      }, undefined, reject)
    })
  }

  swap(name: string): void {
    const low = this.lowResMeshes.get(name)
    const high = this.highResMeshes.get(name)
    if (low && high) {
      high.position.copy(low.position)
      high.rotation.copy(low.rotation)
      high.scale.copy(low.scale)
      this.scene.remove(low)
      low.geometry.dispose()
      this.scene.add(high)
      this.lowResMeshes.delete(name)
    }
  }
}

async function createProgressiveScene() {
  const scene = new THREE.Scene()
  scene.background = new THREE.Color(0x1a1a2e)
  const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 100)
  camera.position.set(0, 3, 8)

  const renderer = new THREE.WebGLRenderer({ antialias: true })
  renderer.setSize(window.innerWidth, window.innerHeight)
  document.body.appendChild(renderer.domElement)

  const controls = new OrbitControls(camera, renderer.domElement)
  controls.enableDamping = true

  scene.add(new THREE.AmbientLight(0xffffff, 0.5))
  scene.add(new THREE.DirectionalLight(0xffffff, 0.8))

  const loader = new ProgressiveLoader(scene)

  const loadingEl = document.createElement('div')
  loadingEl.style.cssText = 'position:fixed;top:50%;left:50%;transform:translate(-50%,-50%);color:#fff;font-size:18px;z-index:10;'
  loadingEl.textContent = '加载低精度模型...'
  document.body.appendChild(loadingEl)

  const progressBar = document.createElement('div')
  progressBar.style.cssText = 'position:fixed;top:55%;left:50%;transform:translateX(-50%);width:200px;height:4px;background:#333;z-index:10;'
  const progressFill = document.createElement('div')
  progressFill.style.cssText = 'height:100%;width:0%;background:#4fc3f7;transition:width 0.3s;'
  progressBar.appendChild(progressFill)
  document.body.appendChild(progressBar)

  try {
    await loader.loadLowRes('/models/scene-low.glb', 'main-scene')
    loadingEl.textContent = '低精度模型已加载'
    progressFill.style.width = '50%'

    setTimeout(async () => {
      loadingEl.textContent = '加载高精度模型...'
      await loader.loadHighRes('/models/scene-high.glb', 'main-scene')
      loadingEl.textContent = '高精度模型已加载'
      progressFill.style.width = '100%'
      setTimeout(() => {
        loadingEl.style.display = 'none'
        progressBar.style.display = 'none'
      }, 1000)
    }, 100)
  } catch {
    const fallbackGeo = new THREE.BoxGeometry(1, 1, 1)
    const fallbackMat = new THREE.MeshStandardMaterial({ color: 0x4fc3f7 })
    scene.add(new THREE.Mesh(fallbackGeo, fallbackMat))
    loadingEl.textContent = '加载失败，显示默认模型'
  }

  window.addEventListener('resize', () => {
    camera.aspect = window.innerWidth / window.innerHeight
    camera.updateProjectionMatrix()
    renderer.setSize(window.innerWidth, window.innerHeight)
  })

  function animate() {
    requestAnimationFrame(animate)
    controls.update()
    renderer.render(scene, camera)
  }
  animate()
}

```

**要点**：
- 低精度模型应尽量小（< 100KB），保证 2 秒内加载完成
- 替换时必须复制 Transform（position/rotation/scale），否则高精度模型会跳到原点
- 低精度模型释放后要 dispose geometry 和 material，否则内存会短暂翻倍
- 进度条用 CSS transition 实现平滑动画，比 JS 动画更高效
