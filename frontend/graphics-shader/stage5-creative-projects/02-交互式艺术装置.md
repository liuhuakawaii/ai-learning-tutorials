# 交互式艺术装置

## 场景引入

在美术馆里，一面巨大的屏幕随着观众的挥手而绽放粒子花朵；一个音乐厅中，旋律的频谱驱动着三维网格的起伏变形——这些交互式艺术装置将观众从被动观看者变成了作品的共创者。本课将探索如何用 Shader 和 Web API 将鼠标、触摸、音频甚至体感设备的数据实时转化为视觉反馈，构建完整的交互式 Shader Art 作品。

## 学习目标

1. 掌握鼠标/触摸事件驱动的视觉变形技术
2. 理解 Web Audio API 的 AnalyserNode 频谱数据获取
3. 实现音频驱动的粒子系统和网格变形
4. 了解 Kinect/Leap Motion 等体感设备的接入概念
5. 构建完整的交互式 Shader Art 实时反馈循环

## 鼠标与触摸事件驱动变形

### 事件捕获与坐标转换

浏览器的鼠标/触摸事件需要转换为 Shader 可用的归一化坐标：

```
屏幕坐标 (clientX, clientY)
    │
    ├─ 归一化到 [-1, 1]
    │   mouseX = (clientX / width) * 2 - 1
    │   mouseY = -(clientY / height) * 2 + 1
    │
    └─ 传入 Uniform
        u_mouse: vec2
```

```typescript
class InteractionManager {
  mouse = { x: 0, y: 0 }
  smoothMouse = { x: 0, y: 0 }
  velocity = { x: 0, y: 0 }
  private prevMouse = { x: 0, y: 0 }

  constructor(canvas: HTMLCanvasElement) {
    canvas.addEventListener('mousemove', this.onMove.bind(this))
    canvas.addEventListener('touchmove', this.onTouch.bind(this), { passive: false })
  }

  private onMove(e: MouseEvent) {
    this.mouse.x = (e.clientX / window.innerWidth) * 2 - 1
    this.mouse.y = -(e.clientY / window.innerHeight) * 2 + 1
  }

  private onTouch(e: TouchEvent) {
    e.preventDefault()
    const touch = e.touches[0]
    this.mouse.x = (touch.clientX / window.innerWidth) * 2 - 1
    this.mouse.y = -(touch.clientY / window.innerHeight) * 2 + 1
  }

  update() {
    const smoothFactor = 0.1
    this.velocity.x = this.mouse.x - this.prevMouse.x
    this.velocity.y = this.mouse.y - this.prevMouse.y
    this.smoothMouse.x += (this.mouse.x - this.smoothMouse.x) * smoothFactor
    this.smoothMouse.y += (this.mouse.y - this.smoothMouse.y) * smoothFactor
    this.prevMouse.x = this.mouse.x
    this.prevMouse.y = this.mouse.y
  }
}
```

### 鼠标驱动的涟漪效果 Shader

```glsl
// fragment shader - 鼠标涟漪
uniform vec2 u_mouse;
uniform float u_time;
uniform float u_velocity;
varying vec2 v_uv;

void main() {
  vec2 uv = v_uv;
  float dist = distance(uv, u_mouse * 0.5 + 0.5);

  float ripple = sin(dist * 30.0 - u_time * 5.0) * 0.5 + 0.5;
  ripple *= exp(-dist * 3.0);
  ripple *= smoothstep(0.0, 0.3, u_velocity);

  vec3 color = mix(
    vec3(0.1, 0.1, 0.3),
    vec3(0.0, 0.8, 1.0),
    ripple
  );

  gl_FragColor = vec4(color, 1.0);
}
```

## 音频响应：AnalyserNode 频谱数据

### Web Audio API 基础架构

```
音频源 → AnalyserNode → 频域数据 (Uint8Array)
   │           │              │
AudioBuffer  getByte         256/512/1024
 或 <audio>  FrequencyData()  个频率 bin
                             │
                             ├─ 低频 (bass): bin[0-10]
                             ├─ 中频 (mid): bin[10-100]
                             └─ 高频 (treble): bin[100+]
```

```typescript
class AudioAnalyzer {
  private context: AudioContext
  private analyser: AnalyserNode
  private dataArray: Uint8Array
  frequencyData: Float32Array

  constructor(fftSize: number = 512) {
    this.context = new AudioContext()
    this.analyser = this.context.createAnalyser()
    this.analyser.fftSize = fftSize
    this.analyser.smoothingTimeConstant = 0.8
    this.dataArray = new Uint8Array(this.analyser.frequencyBinCount)
    this.frequencyData = new Float32Array(this.analyser.frequencyBinCount)
  }

  async connectMicrophone() {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
    const source = this.context.createMediaStreamSource(stream)
    source.connect(this.analyser)
  }

  connectAudioElement(element: HTMLAudioElement) {
    const source = this.context.createMediaElementSource(element)
    source.connect(this.analyser)
    source.connect(this.context.destination)
  }

  update() {
    this.analyser.getByteFrequencyData(this.dataArray)
    for (let i = 0; i < this.dataArray.length; i++) {
      this.frequencyData[i] = this.dataArray[i] / 255.0
    }
  }

  getBandAverage(start: number, end: number): number {
    let sum = 0
    for (let i = start; i < end; i++) sum += this.frequencyData[i]
    return sum / (end - start)
  }

  get bass() { return this.getBandAverage(0, 10) }
  get mid() { return this.getBandAverage(10, 100) }
  get treble() { return this.getBandAverage(100, 256) }
}
```

## 音频驱动粒子系统

将频谱数据作为纹理传入 GPU，粒子根据对应的频率数据运动：

```glsl
// compute shader 概念 (WebGL2 用 transformFeedback 替代)
// 粒子更新逻辑
uniform sampler2D u_frequencyTexture;
uniform float u_time;
uniform float u_bass;

void main() {
  vec2 uv = gl_FragCoord.xy / resolution;

  // 从频谱纹理采样
  float frequency = texture2D(u_frequencyTexture, vec2(uv.x, 0.0)).r;

  // 粒子位置更新
  vec3 pos = texture2D(u_positionTexture, uv).xyz;
  pos.y += frequency * 0.1;
  pos.x += sin(u_time + pos.y * 2.0) * 0.01 * u_bass;

  // 衰减与重生
  if (pos.y > 5.0) {
    pos = vec3(
      (uv.x - 0.5) * 10.0,
      -5.0,
      (uv.y - 0.5) * 10.0
    );
  }

  gl_FragColor = vec4(pos, 1.0);
}
```

```typescript
class AudioParticles {
  private mesh: THREE.Points
  private positionBuffer: Float32Array
  private velocityBuffer: Float32Array
  private count: number

  constructor(count: number = 10000) {
    this.count = count
    this.positionBuffer = new Float32Array(count * 3)
    this.velocityBuffer = new Float32Array(count * 3)

    for (let i = 0; i < count; i++) {
      this.positionBuffer[i * 3] = (Math.random() - 0.5) * 10
      this.positionBuffer[i * 3 + 1] = (Math.random() - 0.5) * 10
      this.positionBuffer[i * 3 + 2] = (Math.random() - 0.5) * 10
      this.velocityBuffer[i * 3 + 1] = Math.random() * 0.02
    }

    const geometry = new THREE.BufferGeometry()
    geometry.setAttribute('position', new THREE.BufferAttribute(this.positionBuffer, 3))
    const material = new THREE.PointsMaterial({
      size: 0.05,
      transparent: true,
      blending: THREE.AdditiveBlending,
      depthWrite: false,
    })
    this.mesh = new THREE.Points(geometry, material)
  }

  update(audio: AudioAnalyzer) {
    const bass = audio.bass
    const mid = audio.mid
    const positions = this.positionBuffer

    for (let i = 0; i < this.count; i++) {
      const idx = i * 3
      positions[idx + 1] += this.velocityBuffer[idx + 1] * (1 + bass * 5)
      positions[idx] += Math.sin(positions[idx + 1] * 0.5) * mid * 0.02

      if (positions[idx + 1] > 5) {
        positions[idx + 1] = -5
        positions[idx] = (Math.random() - 0.5) * 10
      }
    }
    (this.mesh.geometry.attributes.position as THREE.BufferAttribute).needsUpdate = true
  }
}
```

## 音频驱动网格变形

将频谱数据映射为顶点位移，让平面网格随音乐起伏：

```glsl
// vertex shader - 音频驱动变形
uniform sampler2D u_frequencyData;
uniform float u_amplitude;
uniform float u_time;
varying float v_displacement;
varying vec2 v_uv;

void main() {
  v_uv = uv;

  float freq = texture2D(u_frequencyData, vec2(uv.x, 0.0)).r;
  float displacement = freq * u_amplitude;

  displacement *= sin(uv.y * 3.14159);

  vec3 newPosition = position + normal * displacement;
  v_displacement = displacement;

  gl_Position = projectionMatrix * modelViewMatrix * vec4(newPosition, 1.0);
}
```

```glsl
// fragment shader - 根据位移着色
varying float v_displacement;
varying vec2 v_uv;
uniform float u_time;

void main() {
  vec3 lowColor = vec3(0.05, 0.0, 0.2);
  vec3 midColor = vec3(0.0, 0.6, 1.0);
  vec3 highColor = vec3(1.0, 0.2, 0.8);

  float t = clamp(v_displacement / 2.0, 0.0, 1.0);
  vec3 color = mix(lowColor, midColor, t);
  color = mix(color, highColor, smoothstep(0.5, 1.0, t));

  float glow = exp(-v_displacement * 0.5) * 0.3;
  color += vec3(glow);

  gl_FragColor = vec4(color, 1.0);
}
```

## 体感设备接入概念

### Kinect / Leap Motion 数据流

```
Kinect / Leap Motion
       │
       ├─ WebSocket / HTTP Server
       │   (中间件将设备数据转为网络协议)
       │
       ├─ 手部骨骼数据
       │   [{ x, y, z }, ...]  (25 个关节点)
       │
       └─ 深度图 / 点云
           (可直接作为纹理传入 Shader)
```

```typescript
// Leap Motion WebSocket 接入示例概念
interface LeapHand {
  palmPosition: [number, number, number]
  fingers: Array<{
    tipPosition: [number, number, number]
    direction: [number, number, number]
  }>
  grabStrength: number
  pinchStrength: number
}

class LeapMotionBridge {
  private ws: WebSocket
  private hands: LeapHand[] = []
  private uniforms: Record<string, THREE.IUniform>

  constructor(url: string, uniforms: Record<string, THREE.IUniform>) {
    this.uniforms = uniforms
    this.ws = new WebSocket(url)
    this.ws.onmessage = (e) => {
      const data = JSON.parse(e.data)
      this.hands = data.hands || []
      this.updateUniforms()
    }
  }

  private updateUniforms() {
    if (this.hands.length > 0) {
      const palm = this.hands[0].palmPosition
      this.uniforms.u_handPos.value.set(
        palm[0] / 200, palm[1] / 200, palm[2] / 200
      )
      this.uniforms.u_grab.value = this.hands[0].grabStrength
    }
  }
}
```

## 实时反馈循环设计

交互式艺术装置的核心是一个不中断的反馈循环：

```
┌──────────────────────────────────────────────────┐
│                  实时反馈循环                       │
│                                                    │
│  输入层          处理层           输出层            │
│  ┌─────┐       ┌──────┐       ┌──────┐           │
│  │鼠标 │──────▶│平滑  │──────▶│Uniform│──┐       │
│  │触摸 │       │滤波  │       │更新  │  │       │
│  │音频 │──────▶│频谱  │──────▶│      │  ▼       │
│  │体感 │       │分析  │       └──────┘ Shader   │
│  └─────┘       └──────┘                 │       │
│       ▲                               渲染       │
│       │                                 │       │
│       └───── 视觉反馈 ◀────────────────┘       │
│                                                    │
└──────────────────────────────────────────────────┘
```

```typescript
class ArtInstallation {
  private renderer: THREE.WebGLRenderer
  private scene: THREE.Scene
  private camera: THREE.PerspectiveCamera
  private clock = new THREE.Clock()
  private interaction: InteractionManager
  private audio: AudioAnalyzer
  private uniforms: Record<string, THREE.IUniform>

  constructor(canvas: HTMLCanvasElement) {
    this.renderer = new THREE.WebGLRenderer({ canvas, antialias: true })
    this.scene = new THREE.Scene()
    this.camera = new THREE.PerspectiveCamera(75, 1, 0.1, 100)
    this.camera.position.z = 3
    this.interaction = new InteractionManager(canvas)
    this.audio = new AudioAnalyzer(512)

    this.uniforms = {
      u_time: { value: 0 },
      u_mouse: { value: new THREE.Vector2() },
      u_velocity: { value: 0 },
      u_bass: { value: 0 },
      u_mid: { value: 0 },
      u_treble: { value: 0 },
      u_frequencyData: { value: null },
    }

    this.buildScene()
    this.animate()
  }

  private buildScene() {
    const geometry = new THREE.PlaneGeometry(4, 4, 256, 256)
    const material = new THREE.ShaderMaterial({
      uniforms: this.uniforms,
      vertexShader: audioVertexShader,
      fragmentShader: audioFragmentShader,
    })
    this.scene.add(new THREE.Mesh(geometry, material))
  }

  private animate = () => {
    requestAnimationFrame(this.animate)
    const elapsed = this.clock.getElapsedTime()

    this.interaction.update()
    this.audio.update()

    this.uniforms.u_time.value = elapsed
    this.uniforms.u_mouse.value.set(
      this.interaction.smoothMouse.x,
      this.interaction.smoothMouse.y
    )
    this.uniforms.u_velocity.value = Math.hypot(
      this.interaction.velocity.x,
      this.interaction.velocity.y
    )
    this.uniforms.u_bass.value = this.audio.bass
    this.uniforms.u_mid.value = this.audio.mid
    this.uniforms.u_treble.value = this.audio.treble

    this.renderer.render(this.scene, this.camera)
  }
}
```

## 常见误区

1. **直接使用原始鼠标坐标而不做平滑处理**：原始鼠标坐标跳变剧烈，直接传入 Shader 会导致视觉抖动。必须用指数移动平均（EMA）或弹簧阻尼进行平滑，同时计算速度用于动态效果强度控制。

2. **在主线程中处理音频数据的复杂计算**：频谱数据的 FFT 计算虽然由 AnalyserNode 完成，但如果需要对频谱做进一步分析（如节拍检测、频段聚类），这些计算应该放到 Worker 中，否则会阻塞渲染。

3. **忽略音频上下文的自动播放策略**：浏览器要求 AudioContext 必须在用户交互后才能 resume。应该在页面加载时就创建 AudioContext 并挂起，在首次点击时调用 `context.resume()`。

4. **体感数据未做坐标空间转换**：Kinect/Leap Motion 的坐标原点和单位与 Three.js 场景完全不同，直接使用会导致模型出现在视野外。必须先理解设备坐标系，再做映射和归一化。

## 工程建议

1. **用 GUI 面板调试参数**：集成 lil-gui 或 Tweakpane，将 Shader 的 uniform 参数暴露为滑块，可以大幅加速艺术效果的调试和迭代过程。

2. **实现录制与回放**：将输入事件（鼠标轨迹、音频时间戳）录制为 JSON，在开发时可以回放而不必每次都手动操作，也方便做自动化截图测试。

3. **模块化 Shader 代码**：将噪声函数、颜色映射、坐标变换等通用功能提取为 GLSL include 文件，通过 `#pragma include` 或构建工具注入，避免在每个 Shader 中重复编写。

4. **考虑无障碍访问**：音频驱动的视觉效果对听障用户没有意义，应提供替代交互模式（如鼠标驱动同样效果），并确保色彩对比度满足 WCAG 标准。

## 小结

本课探索了交互式艺术装置的技术栈：从鼠标/触摸事件的捕获与平滑，到 Web Audio API 的频谱分析，再到体感设备的接入概念。核心是建立一个实时反馈循环——输入信号经过处理后驱动 Shader 的 uniform 变量，产生随交互实时变化的视觉效果。音频驱动的粒子系统和网格变形是两个最常用的技术模式。

## 参考答案

### 练习一

**思路**：鼠标涟漪效果的核心是将鼠标位置和速度传入 Shader，在片元着色器中根据鼠标轨迹生成同心圆波纹。速度越快波纹越密集——通过在鼠标移动时记录多个历史位置点（trail），每个点生成一个扩散环。静止时波纹通过时间衰减自然消散。

**答案**：

```typescript
import * as THREE from 'three'

const vertexShader = `
varying vec2 v_uv;
void main() {
  v_uv = uv;
  gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
}
`

const fragmentShader = `
uniform float u_time;
uniform vec2 u_resolution;
uniform vec2 u_mouse;
uniform float u_speed;
uniform float u_trail[10];
uniform vec2 u_trailPos[10];
uniform int u_trailCount;

float ripple(vec2 uv, vec2 center, float t) {
  float d = length(uv - center);
  float wave = sin(d * 40.0 - t * 6.0) * 0.5 + 0.5;
  float ring = smoothstep(0.02, 0.0, abs(d - t * 0.3));
  return wave * ring * exp(-t * 2.0);
}

void main() {
  vec2 uv = (gl_FragCoord.xy / u_resolution) * 2.0 - 1.0;
  uv.x *= u_resolution.x / u_resolution.y;

  float effect = 0.0;
  for (int i = 0; i < 10; i++) {
    if (i >= u_trailCount) break;
    vec2 trailUV = u_trailPos[i];
    trailUV.x *= u_resolution.x / u_resolution.y;
    float t = u_time - u_trail[i];
    if (t > 0.0 && t < 2.0) {
      effect += ripple(uv, trailUV, t) * 0.5;
    }
  }

  vec3 color = vec3(0.0);
  color += effect * vec3(0.2, 0.6, 1.0);
  color += effect * effect * vec3(0.8, 0.3, 1.0);
  gl_FragColor = vec4(color, 1.0);
}
`

function createMouseRipple() {
  const scene = new THREE.Scene()
  const camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1)

  const renderer = new THREE.WebGLRenderer({ antialias: true })
  renderer.setSize(window.innerWidth, window.innerHeight)
  document.body.appendChild(renderer.domElement)

  const trailHistory: { x: number; y: number; time: number; speed: number }[] = []
  const maxTrail = 10
  let lastMouse = { x: 0, y: 0 }
  let currentSpeed = 0

  const uniforms = {
    u_time: { value: 0 },
    u_resolution: { value: new THREE.Vector2(window.innerWidth, window.innerHeight) },
    u_mouse: { value: new THREE.Vector2(0, 0) },
    u_speed: { value: 0 },
    u_trail: { value: new Float32Array(maxTrail) },
    u_trailPos: { value: Array.from({ length: maxTrail }, () => new THREE.Vector2()) },
    u_trailCount: { value: 0 }
  }

  const mat = new THREE.ShaderMaterial({
    vertexShader, fragmentShader, uniforms
  })
  scene.add(new THREE.Mesh(new THREE.PlaneGeometry(2, 2), mat))

  renderer.domElement.addEventListener('mousemove', (e) => {
    const x = e.clientX / window.innerWidth
    const y = 1 - e.clientY / window.innerHeight
    const dx = x - lastMouse.x, dy = y - lastMouse.y
    currentSpeed = Math.sqrt(dx * dx + dy * dy) * 100
    lastMouse = { x, y }

    trailHistory.push({ x, y, time: uniforms.u_time.value, speed: currentSpeed })
    if (trailHistory.length > maxTrail) trailHistory.shift()
  })

  const clock = new THREE.Clock()
  function animate() {
    requestAnimationFrame(animate)
    uniforms.u_time.value = clock.getElapsedTime()
    uniforms.u_speed.value = currentSpeed

    const now = uniforms.u_time.value
    const activeTrails = trailHistory.filter(t => now - t.time < 2.0)
    uniforms.u_trailCount.value = activeTrails.length
    activeTrails.forEach((t, i) => {
      uniforms.u_trail.value[i] = t.time
      uniforms.u_trailPos[i].value.set(t.x, t.y)
    })

    renderer.render(scene, camera)
  }
  animate()
}

createMouseRipple()
```

**要点**：
- 将鼠标移动历史存储为 trail 数组传入 Shader，每个点独立生成扩散波纹
- `exp(-t * 2.0)` 实现时间衰减，波纹在 2 秒内自然消散
- 鼠标速度通过相邻帧的位移计算，速度快时 trail 更新更频繁，波纹自然更密集
- 全屏后处理用 OrthographicCamera + PlaneGeometry(2, 2) 覆盖整个裁剪空间

### 练习二

**思路**：音频可视化需要三个环节：加载音频文件 → 用 AnalyserNode 获取频谱数据 → 将频谱映射到粒子环的视觉属性。AnalyserNode.getByteFrequencyData() 返回 0-255 的频谱数组，将其分段映射到粒子的半径、颜色和透明度。

**答案**：

```typescript
import * as THREE from 'three'

function createAudioVisualizer() {
  const scene = new THREE.Scene()
  const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000)
  camera.position.z = 5

  const renderer = new THREE.WebGLRenderer({ antialias: true })
  renderer.setSize(window.innerWidth, window.innerHeight)
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2))
  document.body.appendChild(renderer.domElement)

  const PARTICLE_COUNT = 512
  const geo = new THREE.BufferGeometry()
  const positions = new Float32Array(PARTICLE_COUNT * 3)
  const colors = new Float32Array(PARTICLE_COUNT * 3)
  const sizes = new Float32Array(PARTICLE_COUNT)

  for (let i = 0; i < PARTICLE_COUNT; i++) {
    const angle = (i / PARTICLE_COUNT) * Math.PI * 2
    positions[i * 3] = Math.cos(angle) * 2
    positions[i * 3 + 1] = Math.sin(angle) * 2
    positions[i * 3 + 2] = 0
    sizes[i] = 3
  }

  geo.setAttribute('position', new THREE.BufferAttribute(positions, 3))
  geo.setAttribute('color', new THREE.BufferAttribute(colors, 3))
  geo.setAttribute('size', new THREE.BufferAttribute(sizes, 1))

  const vertexShader = `
    attribute float size;
    varying vec3 v_color;
    void main() {
      v_color = color;
      vec4 mvPosition = modelViewMatrix * vec4(position, 1.0);
      gl_PointSize = size * (200.0 / -mvPosition.z);
      gl_Position = projectionMatrix * mvPosition;
    }
  `
  const fragmentShader = `
    varying vec3 v_color;
    void main() {
      float d = length(gl_PointCoord - 0.5);
      if (d > 0.5) discard;
      float alpha = 1.0 - smoothstep(0.3, 0.5, d);
      gl_FragColor = vec4(v_color, alpha);
    }
  `

  const mat = new THREE.ShaderMaterial({
    vertexShader, fragmentShader,
    vertexColors: true, transparent: true, depthWrite: false
  })
  const points = new THREE.Points(geo, mat)
  scene.add(points)

  const audioContext = new AudioContext()
  const analyser = audioContext.createAnalyser()
  analyser.fftSize = 1024
  const bufferLength = analyser.frequencyBinCount
  const dataArray = new Uint8Array(bufferLength)

  const fileInput = document.createElement('input')
  fileInput.type = 'file'
  fileInput.accept = 'audio/*'
  fileInput.style.cssText = 'position:fixed;top:10px;left:10px;z-index:10;'
  document.body.appendChild(fileInput)

  fileInput.addEventListener('change', async (e) => {
    const file = (e.target as HTMLInputElement).files?.[0]
    if (!file) return
    const arrayBuffer = await file.arrayBuffer()
    const audioBuffer = await audioContext.decodeAudioData(arrayBuffer)
    const source = audioContext.createBufferSource()
    source.buffer = audioBuffer
    source.connect(analyser)
    analyser.connect(audioContext.destination)
    source.start()
  })

  function animate() {
    requestAnimationFrame(animate)
    analyser.getByteFrequencyData(dataArray)

    const posAttr = geo.attributes.position as THREE.BufferAttribute
    const colAttr = geo.attributes.color as THREE.BufferAttribute
    const sizeAttr = geo.attributes.size as THREE.BufferAttribute

    for (let i = 0; i < PARTICLE_COUNT; i++) {
      const freqIndex = Math.floor((i / PARTICLE_COUNT) * bufferLength)
      const amplitude = dataArray[freqIndex] / 255

      const angle = (i / PARTICLE_COUNT) * Math.PI * 2
      const radius = 1.5 + amplitude * 2.5
      posAttr.setXYZ(i, Math.cos(angle) * radius, Math.sin(angle) * radius, 0)

      const hue = (i / PARTICLE_COUNT) * 0.3 + amplitude * 0.2
      const color = new THREE.Color().setHSL(hue, 0.8, 0.3 + amplitude * 0.5)
      colAttr.setXYZ(i, color.r, color.g, color.b)

      sizeAttr.setX(i, 2 + amplitude * 8)
    }

    posAttr.needsUpdate = true
    colAttr.needsUpdate = true
    sizeAttr.needsUpdate = true
    renderer.render(scene, camera)
  }
  animate()
}
```

### 练习三

**思路**：音频驱动地形的核心是两层位移叠加——底层用 Perlin 噪声生成基础起伏，上层叠加频谱数据。低频能量映射到大面积起伏（低频噪声的位移系数），高频能量映射到细节扰动（顶点级别的微小偏移）。

**答案**：

```typescript
import * as THREE from 'three'
import { OrbitControls } from 'three/addons/controls/OrbitControls.js'

const terrainVertex = `
uniform float u_time;
uniform float u_bass;
uniform float u_treble;
uniform float u_spectrum[32];
varying float v_elevation;
varying vec2 v_uv;

vec3 mod289(vec3 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
vec4 mod289(vec4 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
vec4 permute(vec4 x) { return mod289(((x*34.0)+1.0)*x); }
vec4 taylorInvSqrt(vec4 r) { return 1.79284291400159 - 0.85373472095314 * r; }

float snoise(vec3 v) {
  const vec2 C = vec2(1.0/6.0, 1.0/3.0);
  const vec4 D = vec4(0.0, 0.5, 1.0, 2.0);
  vec3 i = floor(v + dot(v, C.yyy));
  vec3 x0 = v - i + dot(i, C.xxx);
  vec3 g = step(x0.yzx, x0.xyz);
  vec3 l = 1.0 - g;
  vec3 i1 = min(g.xyz, l.zxy);
  vec3 i2 = max(g.xyz, l.zxy);
  vec3 x1 = x0 - i1 + C.xxx;
  vec3 x2 = x0 - i2 + C.yyy;
  vec3 x3 = x0 - D.yyy;
  i = mod289(i);
  vec4 p = permute(permute(permute(
    i.z + vec4(0.0, i1.z, i2.z, 1.0))
    + i.y + vec4(0.0, i1.y, i2.y, 1.0))
    + i.x + vec4(0.0, i1.x, i2.x, 1.0));
  float n_ = 0.142857142857;
  vec3 ns = n_ * D.wyz - D.xzx;
  vec4 j = p - 49.0 * floor(p * ns.z * ns.z);
  vec4 x_ = floor(j * ns.z);
  vec4 y_ = floor(j - 7.0 * x_);
  vec4 x = x_ * ns.x + ns.yyyy;
  vec4 y = y_ * ns.x + ns.yyyy;
  vec4 h = 1.0 - abs(x) - abs(y);
  vec4 b0 = vec4(x.xy, y.xy);
  vec4 b1 = vec4(x.zw, y.zw);
  vec4 s0 = floor(b0) * 2.0 + 1.0;
  vec4 s1 = floor(b1) * 2.0 + 1.0;
  vec4 sh = -step(h, vec4(0.0));
  vec4 a0 = b0.xzyw + s0.xzyw * sh.xxyy;
  vec4 a1 = b1.xzyw + s1.xzyw * sh.zzww;
  vec3 p0 = vec3(a0.xy, h.x);
  vec3 p1 = vec3(a0.zw, h.y);
  vec3 p2 = vec3(a1.xy, h.z);
  vec3 p3 = vec3(a1.zw, h.w);
  vec4 norm = taylorInvSqrt(vec4(dot(p0,p0),dot(p1,p1),dot(p2,p2),dot(p3,p3)));
  p0 *= norm.x; p1 *= norm.y; p2 *= norm.z; p3 *= norm.w;
  vec4 m = max(0.6 - vec4(dot(x0,x0),dot(x1,x1),dot(x2,x2),dot(x3,x3)), 0.0);
  m = m * m;
  return 42.0 * dot(m*m, vec4(dot(p0,x0),dot(p1,x1),dot(p2,x2),dot(p3,x3)));
}

void main() {
  v_uv = uv;
  float baseNoise = snoise(vec3(position.x * 0.3, position.z * 0.3, u_time * 0.1)) * 2.0;
  float bassDisplacement = u_bass * snoise(vec3(position.x * 0.1, position.z * 0.1, u_time * 0.05)) * 3.0;
  float trebleDisplacement = u_treble * snoise(vec3(position.x * 2.0, position.z * 2.0, u_time * 0.5)) * 0.3;
  float specIndex = (uv.x + uv.y) * 0.5 * 31.0;
  float specDisplacement = u_spectrum[int(floor(specIndex))] * 0.5;

  float elevation = baseNoise + bassDisplacement + trebleDisplacement + specDisplacement;
  v_elevation = elevation;
  vec3 newPos = position + normal * elevation;
  gl_Position = projectionMatrix * modelViewMatrix * vec4(newPos, 1.0);
}
`

const terrainFragment = `
varying float v_elevation;
varying vec2 v_uv;

void main() {
  vec3 lowColor = vec3(0.05, 0.1, 0.3);
  vec3 midColor = vec3(0.2, 0.5, 0.8);
  vec3 highColor = vec3(1.0, 0.9, 0.6);
  float t = (v_elevation + 2.0) / 6.0;
  vec3 color = mix(lowColor, midColor, smoothstep(0.0, 0.5, t));
  color = mix(color, highColor, smoothstep(0.5, 1.0, t));
  gl_FragColor = vec4(color, 1.0);
}
`

    uniforms.u_bass.value = (bassSum / bassEnd / 255)

    for (let i = 0; i < 32; i++) {
      const idx = Math.floor((i / 32) * dataArray.length)
      uniforms.u_spectrum.value[i] = dataArray[idx] / 255
    }

    uniforms.u_time.value = clock.getElapsedTime()
    controls.update()
    renderer.render(scene, camera)
  }
  animate()
}
```

### 练习四

**思路**：完整的 Shader Art 作品需要整合多个系统：音频输入（AnalyserNode）、鼠标输入（平滑后的 uniform）、参数控制（lil-gui）、全屏切换（Fullscreen API）、截图导出（toDataURL）。核心是建立一个统一的 uniform 对象，所有输入源都往里写，Shader 从里读。

**答案**：

```typescript
import * as THREE from 'three'
import GUI from 'lil-gui'

const vertexShader = `
varying vec2 v_uv;
void main() {
  v_uv = uv;
  gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
}
`

const fragmentShader = `
uniform float u_time;
uniform vec2 u_resolution;
uniform vec2 u_mouse;
uniform float u_bass;
uniform float u_mid;
uniform float u_treble;
uniform float u_speed;
uniform float u_scale;
uniform float u_noiseStrength;
uniform float u_colorShift;
uniform float u_glowIntensity;

varying vec2 v_uv;

vec3 mod289(vec3 x) { return x - floor(x * (1.0/289.0)) * 289.0; }
vec2 mod289(vec2 x) { return x - floor(x * (1.0/289.0)) * 289.0; }
vec3 permute(vec3 x) { return mod289(((x*34.0)+1.0)*x); }

float snoise(vec2 v) {
  const vec4 C = vec4(0.211324865405187, 0.366025403784439, -0.577350269189626, 0.024390243902439);
  vec2 i = floor(v + dot(v, C.yy));
  vec2 x0 = v - i + dot(i, C.xx);
  vec2 i1 = (x0.x > x0.y) ? vec2(1.0, 0.0) : vec2(0.0, 1.0);
  vec4 x12 = x0.xyxy + C.xxzz;
  x12.xy -= i1;
  i = mod289(i);
  vec3 p = permute(permute(i.y + vec3(0.0, i1.y, 1.0)) + i.x + vec3(0.0, i1.x, 1.0));
  vec3 m = max(0.5 - vec3(dot(x0,x0), dot(x12.xy,x12.xy), dot(x12.zw,x12.zw)), 0.0);
  m = m * m; m = m * m;
  vec3 x = 2.0 * fract(p * C.www) - 1.0;
  vec3 h = abs(x) - 0.5;
  vec3 ox = floor(x + 0.5);
  vec3 a0 = x - ox;
  m *= 1.79284291400159 - 0.85373472095314 * (a0*a0 + h*h);
  vec3 g;
  g.x = a0.x * x0.x + h.x * x0.y;
  g.yz = a0.yz * x12.xz + h.yz * x12.yw;
  return 130.0 * dot(m, g);
}

void main() {
  vec2 uv = (gl_FragCoord.xy / u_resolution) * 2.0 - 1.0;
  uv.x *= u_resolution.x / u_resolution.y;

  float mouseDist = length(uv - u_mouse);
  float mouseInfluence = smoothstep(1.5, 0.0, mouseDist);

  float n = snoise(uv * u_scale + u_time * u_speed * 0.3) * u_noiseStrength;
  n += u_bass * snoise(uv * 2.0 + u_time * 0.1) * 0.5;
  n += u_treble * snoise(uv * 8.0 + u_time * 1.0) * 0.15;
  n += mouseInfluence * 0.5;

  float pattern = sin(uv.x * 10.0 + n * 5.0) * cos(uv.y * 10.0 + n * 3.0);
  pattern = smoothstep(-0.2, 0.2, pattern);

  float hue = pattern * 0.3 + u_colorShift + u_bass * 0.1;
  vec3 col = 0.5 + 0.5 * cos(6.28318 * (hue + vec3(0.0, 0.1, 0.2)));
  col *= 0.3 + u_glowIntensity * (u_mid + u_bass * 0.5);
  col += mouseInfluence * vec3(0.3, 0.1, 0.5) * 0.5;

  gl_FragColor = vec4(col, 1.0);
}
`
```
