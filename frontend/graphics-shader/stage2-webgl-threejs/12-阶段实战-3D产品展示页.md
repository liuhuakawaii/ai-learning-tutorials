// store/useProductStore.ts
import { create } from 'zustand'

interface MaterialPreset {
  name: string
  color: string
  metalness: number
  roughness: number
}

interface ProductStore {
  activePreset: string
  presets: Record<string, MaterialPreset>
  autoRotate: boolean
  isLoaded: boolean
  setActivePreset: (name: string) => void
  toggleAutoRotate: () => void
  setLoaded: (loaded: boolean) => void
}

const defaultPresets: Record<string, MaterialPreset> = {
  '曜石黑': { name: '曜石黑', color: '#1a1a1a', metalness: 0.9, roughness: 0.1 },
  '星空银': { name: '星空银', color: '#c0c0c0', metalness: 0.8, roughness: 0.2 },
  '玫瑰金': { name: '玫瑰金', color: '#b76e79', metalness: 0.85, roughness: 0.15 },
  '深海蓝': { name: '深海蓝', color: '#1a3a5c', metalness: 0.7, roughness: 0.25 },
}

export const useProductStore = create<ProductStore>((set) => ({
  activePreset: '曜石黑',
  presets: defaultPresets,
  autoRotate: true,
  isLoaded: false,
  setActivePreset: (name) => set({ activePreset: name }),
  toggleAutoRotate: () => set((s) => ({ autoRotate: !s.autoRotate })),
  setLoaded: (loaded) => set({ isLoaded: loaded }),
}))

## 三、产品模型加载组件

```tsx
// components/Product/ProductModel.tsx
import { useRef, useEffect } from 'react'
import { useFrame } from '@react-three/fiber'
import { useGLTF } from '@react-three/drei'
import * as THREE from 'three'
import { useProductStore } from '../../store/useProductStore'

export function ProductModel({ modelPath, scale = 1 }: {
  modelPath: string
  scale?: number
}) {
  const groupRef = useRef<THREE.Group>(null!)
  const { scene, materials } = useGLTF(modelPath)
  const activePreset = useProductStore((s) => s.activePreset)
  const presets = useProductStore((s) => s.presets)
  const setLoaded = useProductStore((s) => s.setLoaded)

  useEffect(() => {
    setLoaded(true)
    scene.traverse((child) => {
      if (child instanceof THREE.Mesh) {
        child.castShadow = true
        child.receiveShadow = true
      }
    })
    return () => {
      scene.traverse((child) => {
        if (child instanceof THREE.Mesh) {
          child.geometry.dispose()
          const mats = Array.isArray(child.material)
            ? child.material : [child.material]
          mats.forEach((m) => m.dispose())
        }
      })
    }
  }, [scene, setLoaded])

  useEffect(() => {
    const preset = presets[activePreset]
    if (!preset) return
    Object.values(materials).forEach((material) => {
      if (material instanceof THREE.MeshStandardMaterial) {
        material.color.set(preset.color)
        material.metalness = preset.metalness
        material.roughness = preset.roughness
        material.needsUpdate = true
      }
    })
  }, [activePreset, presets, materials])

  useFrame((state) => {
    if (groupRef.current) {
      groupRef.current.position.y = Math.sin(state.clock.elapsedTime * 0.5) * 0.05
    }
  })

  return (
    <group ref={groupRef} scale={scale}>
      <primitive object={scene} />
    </group>
  )
}

useGLTF.preload('/models/product.glb')
```

## 八、性能优化清单

```
优化项                      实施方法                           预期收益
──────────────────────────────────────────────────────────────────────
Draw Call 减少              模型预合并、共享材质                CPU 开销降低 50%
纹理压缩                    KTX2 + Basis Universal              显存减少 6-8 倍
LOD 切换                    Drei LOD 组件                       远处面数减少 90%
阴影优化                    shadow-mapSize 限制 2048            GPU 开销降低 30%
React memo                  包裹静态子组件                      避免无意义重渲染
按需渲染                    frameloop="demand"                  空闲时 0 GPU 占用
```

上线前务必用 `@react-three/perf` 监控 Draw Call 数量和帧率。

## 常见误区

### 1. 直接用大模型不做优化

设计师给的模型可能有 500 万面，直接用会让页面卡死。上线前必须用 gltf-pipeline 或 Blender 减面、合并网格、压缩纹理。

### 2. 忽略移动端适配

移动端 GPU 性能只有桌面的 1/5。必须降低阴影分辨率、减少面数、降低 DPR，甚至关闭部分后处理效果。

### 3. 光照设置过于复杂

多盏灯光 + 阴影 + 环境贴图会严重影响性能。产品展示通常只需要 3-4 盏灯 + 一张 HDR 环境贴图。

### 4. 没有处理加载失败

模型加载可能因为网络问题失败，必须有错误边界和重试机制，不能让页面白屏。

## 工程建议

1. **模型预处理：** 使用 gltf-pipeline 压缩模型，Draco 压缩几何体，KTX2 压缩纹理
2. **渐进式加载：** 先加载低精度模型快速显示，再异步加载高精度版本
3. **帧率监控：** 开发阶段始终开启 `@react-three/perf`，确保帧率稳定在 60fps
4. **错误边界：** 用 React Error Boundary 包裹 Canvas，捕获渲染异常

### 练习一

**思路**：用 R3F 搭建完整的产品展示页面：加载 GLTF 模型，用 Zustand 管理材质状态（颜色/金属度/粗糙度），UI 面板切换材质预设，`useFrame` 实现自动旋转。

**答案**：

```tsx
import { useRef, useState, Suspense } from "react";
import { Canvas, useFrame } from "@react-three/fiber";
import { OrbitControls, Environment, useGLTF, Html } from "@react-three/drei";
import * as THREE from "three";

// 材质预设
const MATERIAL_PRESETS = [
    { name: "银色金属", color: "#cccccc", metalness: 0.9, roughness: 0.1 },
    { name: "金色金属", color: "#ffd700", metalness: 0.95, roughness: 0.05 },
    { name: "红色哑光", color: "#cc2222", metalness: 0.0, roughness: 0.8 },
    { name: "蓝色塑料", color: "#2266cc", metalness: 0.1, roughness: 0.4 },
    { name: "黑色磨砂", color: "#222222", metalness: 0.3, roughness: 0.9 },
];

interface MaterialState {
    color: string;
    metalness: number;
    roughness: number;
}

function ProductModel({
    materialState,
    autoRotate,
}: {
    materialState: MaterialState;
    autoRotate: boolean;
}) {
    const groupRef = useRef<THREE.Group>(null!);
    // 替换为实际模型路径
    const { scene } = useGLTF("/models/product.glb");

    useFrame((_, delta) => {
        if (autoRotate && groupRef.current) {
            groupRef.current.rotation.y += delta * 0.5;
        }
    });

    // 遍历模型，更新材质
    scene.traverse((child) => {
        if ((child as THREE.Mesh).isMesh) {
            const mesh = child as THREE.Mesh;
            const mat = mesh.material as THREE.MeshStandardMaterial;
            mat.color.set(materialState.color);
            mat.metalness = materialState.metalness;
            mat.roughness = materialState.roughness;
            mat.needsUpdate = true;
        }
    });

    return <primitive ref={groupRef} object={scene} />;
}

function MaterialPanel({
    presets,
    activeIndex,
    onSelect,
    autoRotate,
    onToggleRotate,
}: {
    typeof presets: typeof MATERIAL_PRESETS;
    activeIndex: number;
    onSelect: (index: number) => void;
    autoRotate: boolean;
    onToggleRotate: () => void;
}) {
    return (
        <div style={{
            position: "fixed",
            bottom: 20,
            left: "50%",
            transform: "translateX(-50%)",
            display: "flex",
            gap: 8,
            padding: 12,
            background: "rgba(0,0,0,0.85)",
            borderRadius: 12,
            alignItems: "center",
        }}>
            {presets.map((preset, i) => (
                <button
                    key={preset.name}
                    onClick={() => onSelect(i)}
                    style={{
                        padding: "8px 14px",
                        background: i === activeIndex ? "#fff" : "transparent",
                        color: i === activeIndex ? "#000" : "#fff",
                        border: "1px solid #555",
                        borderRadius: 6,
                        cursor: "pointer",
                        fontSize: 13,
                    }}
                >
                    {preset.name}
                </button>
            ))}
            <button
                onClick={onToggleRotate}
                style={{
                    padding: "8px 14px",
                    background: autoRotate ? "#4caf50" : "transparent",
                    color: "#fff",
                    border: "1px solid #555",
                    borderRadius: 6,
                    cursor: "pointer",
                    fontSize: 13,
                }}
            >
                {autoRotate ? "停止旋转" : "自动旋转"}
            </button>
        </div>
    );
}

export default function App() {
    const [materialIndex, setMaterialIndex] = useState(0);
    const [autoRotate, setAutoRotate] = useState(true);

    return (
        <>
            <Canvas camera={{ position: [3, 2, 5], fov: 45 }} shadows>
                <ambientLight intensity={0.3} />
                <directionalLight position={[5, 5, 5]} intensity={1.0} castShadow />
                <Environment preset="studio" />

                <Suspense fallback={<Html center>加载中...</Html>}>
                    <ProductModel
                        materialState={MATERIAL_PRESETS[materialIndex]}
                        autoRotate={autoRotate}
                    />
                </Suspense>

                <OrbitControls enableDamping autoRotate={false} />
                <gridHelper args={[10, 10, 0x333344, 0x222233]} />
            </Canvas>

            <MaterialPanel
                presets={MATERIAL_PRESETS}
                activeIndex={materialIndex}
                onSelect={setMaterialIndex}
                autoRotate={autoRotate}
                onToggleRotate={() => setAutoRotate((prev) => !prev)}
            />
        </>
    );
}
```

**要点**：
- `primitive` 组件直接渲染 Three.js 对象，适合加载外部模型
- `Environment` 提供 IBL 环境光，让金属材质反射出真实环境
- `Suspense` + `Html` 提供加载中的用户反馈
- 材质更新时设置 `needsUpdate = true`，否则 Three.js 不会重新编译 Shader

### 练习二

**思路**：用 `renderer.domElement.toDataURL()` 或 `gl.domElement.toDataURL()` 将当前帧渲染结果转为 PNG。R3F 中通过 `useThree` 获取渲染器，用 `preserveDrawingBuffer: true` 确保缓冲区可读。

**答案**：

```tsx
import { useRef, useCallback } from "react";
import { Canvas, useThree, useFrame } from "@react-three/fiber";
import { OrbitControls } from "@react-three/drei";
import * as THREE from "three";

function RotatingBox() {
    const ref = useRef<THREE.Mesh>(null!);
    useFrame((_, delta) => { ref.current.rotation.y += delta * 0.5; });
    return (
        <mesh ref={ref}>
            <boxGeometry args={[1.5, 1.5, 1.5]} />
            <meshStandardMaterial color="#4fc3f7" metalness={0.3} roughness={0.4} />
        </mesh>
    );
}

function ScreenshotButton() {
    const { gl, scene, camera } = useThree();

    const takeScreenshot = useCallback(() => {
        // 渲染当前帧
        gl.render(scene, camera);

        // 转为 PNG 数据 URL
        const dataURL = gl.domElement.toDataURL("image/png");

        // 创建下载链接
        const link = document.createElement("a");
        link.download = `screenshot-${Date.now()}.png`;
        link.href = dataURL;
        link.click();
    }, [gl, scene, camera]);

    return (
        <mesh position={[0, -2, 0]} onClick={takeScreenshot}>
            <boxGeometry args={[2, 0.5, 0.5]} />
            <meshStandardMaterial color="#4caf50" />
        </mesh>
    );
}

export default function App() {
    return (
        <Canvas
            camera={{ position: [3, 2, 5] }}
            gl={{ preserveDrawingBuffer: true }} // 关键：保留帧缓冲区
        >
            <ambientLight intensity={0.4} />
            <directionalLight position={[5, 5, 5]} intensity={1.0} />
            <RotatingBox />
            <ScreenshotButton />
            <OrbitControls />
        </Canvas>
    );
}
```

**要点**：
- `gl={{ preserveDrawingBuffer: true }}` 是关键配置，否则 `toDataURL()` 会返回空白图片（WebGL 默认在读取后清空缓冲区）
- 截图前必须先调用 `gl.render()` 确保当前帧已绘制到缓冲区
- `preserveDrawingBuffer` 会略微影响性能（GPU 不能使用双缓冲优化），仅在需要截图时开启
- 也可以用 `gl.domElement.toBlob()` 异步获取 Blob 对象，适合上传到服务器

### 练习三

**思路**：用 `useFrame` 和 `THREE.MathUtils.lerp` 实现点击时放大、再次点击时恢复的动画。用状态标记当前是否放大，在每帧中根据目标值插值当前缩放。

**答案**：

```tsx
import { useRef, useState } from "react";
import { Canvas, useFrame } from "@react-three/fiber";
import { OrbitControls } from "@react-three/drei";
import * as THREE from "three";

function AnimatedProduct() {
    const meshRef = useRef<THREE.Mesh>(null!);
    const [isScaled, setIsScaled] = useState(false);
    const targetScale = useRef(1);

    // 点击切换缩放状态
    const handleClick = () => {
        setIsScaled((prev) => !prev);
        targetScale.current = isScaled ? 1 : 1.4; // 放大 40%
    };

    useFrame(() => {
        if (!meshRef.current) return;

        // 平滑插值到目标缩放
        const current = meshRef.current.scale.x;
        const next = THREE.MathUtils.lerp(current, targetScale.current, 0.08);

        meshRef.current.scale.set(next, next, next);

        // 放大时轻微上浮
        const targetY = isScaled ? 0.3 : 0;
        meshRef.current.position.y = THREE.MathUtils.lerp(
            meshRef.current.position.y,
            targetY,
            0.08
        );
    });

    return (
        <mesh ref={meshRef} onClick={handleClick} castShadow>
            <boxGeometry args={[1.5, 1.5, 1.5]} />
            <meshStandardMaterial
                color={isScaled ? "#ff7043" : "#4fc3f7"}
                metalness={0.3}
                roughness={0.4}
            />
        </mesh>
    );
}

function Ground() {
    return (
        <mesh rotation={[-Math.PI / 2, 0, 0]} position={[0, -0.75, 0]} receiveShadow>
            <planeGeometry args={[10, 10]} />
            <meshStandardMaterial color="#333344" />
        </mesh>
    );
}

export default function App() {
    return (
        <Canvas camera={{ position: [3, 2, 5], fov: 50 }} shadows>
            <ambientLight intensity={0.4} />
            <directionalLight position={[5, 5, 5]} intensity={1.0} castShadow />

            <AnimatedProduct />
            <Ground />

            <OrbitControls enableDamping />

            {/* 提示文字 */}
            <div style={{
                position: "fixed",
                top: 20,
                left: "50%",
                transform: "translateX(-50%)",
                color: "#fff",
                fontFamily: "monospace",
                fontSize: 14,
                background: "rgba(0,0,0,0.6)",
                padding: "8px 16px",
                borderRadius: 6,
            }}>
                点击立方体播放缩放动画
            </div>
        </Canvas>
    );
}
```

**要点**：
- `THREE.MathUtils.lerp(current, target, 0.08)` 每帧向目标值靠近 8%，产生平滑的缓动效果
- `targetScale` 用 `useRef` 而非 `useState`，因为它的变化不需要触发重渲染（`useFrame` 每帧读取）
- 缩放和位移动画在同一个 `useFrame` 中更新，确保同步
- 放大时改变颜色和轻微上浮，增强交互反馈的视觉层次感
