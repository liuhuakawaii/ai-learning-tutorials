# 体积光与雾效

## 场景引入

清晨的森林里，阳光穿过树叶间隙形成一道道可见的光柱；雾气弥漫的山谷中，远处的山峦逐渐融入灰白色的天际。这些真实世界中的大气效果，是营造场景氛围的关键元素。在游戏中，体积光（God Rays）和雾效是最常用的大气渲染技术。本节将从最基础的指数雾开始，逐步深入到基于 Ray Marching 的体积光实现，让你掌握从简单到复杂的大气效果渲染。

## 1. 指数雾（Exponential Fog）

指数雾是最经典的雾效实现，根据物体与相机的距离进行颜色混合。

```
指数雾衰减曲线：

  雾浓度
  1.0 ┤                          ████████
      │                     █████
      │                █████
  0.5 ┤           █████
      │       ████
      │   ████
  0.0 ┤███
      └──────────────────────────────────
      0    10    20    30    40    50   距离

  exp:   fogFactor = exp(-density * distance)
  exp2:  fogFactor = exp(-(density * distance)²)
  exp2 衰减更快，适合浓雾效果
```

```glsl
// 指数雾片段着色器
uniform vec3 uFogColor;
uniform float uFogDensity;   // 0.02 ~ 0.1
uniform int uFogMode;        // 0 = exp, 1 = exp2

varying float vFogFactor;

// 在顶点着色器中计算
// float dist = length(viewPos.xyz);
// if (uFogMode == 0) {
//     vFogFactor = exp(-uFogDensity * dist);
// } else {
//     float d = uFogDensity * dist;
//     vFogFactor = exp(-d * d);
// }

void main() {
    vec4 baseColor = texture2D(uTexture, vUV);
    float fogFactor = clamp(vFogFactor, 0.0, 1.0);
    gl_FragColor = vec4(mix(uFogColor, baseColor.rgb, fogFactor), baseColor.a);
}
```

完整的雾效管理类：

```typescript
class FogSystem {
  color: THREE.Color;
  density: number;
  near: number;
  far: number;
  mode: 'exp' | 'exp2';

  constructor(options: {
    color?: THREE.Color;
    density?: number;
    near?: number;
    far?: number;
    mode?: 'exp' | 'exp2';
  } = {}) {
    this.color = options.color ?? new THREE.Color(0xcccccc);
    this.density = options.density ?? 0.02;
    this.near = options.near ?? 1;
    this.far = options.far ?? 1000;
    this.mode = options.mode ?? 'exp2';
  }

  applyToMaterial(material: THREE.ShaderMaterial): void {
    material.uniforms.uFogColor = { value: this.color };
    material.uniforms.uFogDensity = { value: this.density };
    material.uniforms.uFogMode = { value: this.mode === 'exp' ? 0 : 1 };
  }

  getFogFactor(distance: number): number {
    if (this.mode === 'exp') {
      return Math.exp(-this.density * distance);
    }
    const d = this.density * distance;
    return Math.exp(-d * d);
  }
}
```

## 2. 高度雾（Height Fog）

高度雾模拟山谷中雾气聚集在低洼处的效果，雾的浓度随高度衰减。

```
高度雾示意：

  高度 ↑
       |
  hMax ┤ ─ ─ ─ ─ ─ ─ ─ ─ ─ ─ ─ ─ ─ ─  雾浓度 = 0
       |                                  (完全透明)
       |       坡度越大
       |       雾越薄
  hMid ┤ ═══════════════════════════════  雾浓度 = 0.5
       |   ████████████████████████████
  hMin ┤ █████████████████████████████████  雾浓度 = 1.0
       | █████████████████████████████████
       └──────────────────────────────────> 水平距离
       雾主要聚集在低洼区域
```

```glsl
// 高度雾着色器
uniform float uFogHeightMin;   // 雾层底部高度
uniform float uFogHeightMax;   // 雾层顶部高度
uniform float uFogDensity;
uniform vec3 uFogColor;

varying vec3 vWorldPos;

float computeHeightFog(vec3 worldPos, vec3 cameraPos) {
    // 相机到片段的射线
    vec3 rayDir = normalize(worldPos - cameraPos);
    float rayLength = length(worldPos - cameraPos);

    // 射线与雾层的交点
    float tMin = max(0.0, (uFogHeightMin - cameraPos.y) / rayDir.y);
    float tMax = min(rayLength, (uFogHeightMax - cameraPos.y) / rayDir.y);

    if (tMin >= tMax) return 0.0; // 射线不穿过雾层

    // 积分雾浓度（简化为平均值）
    float midHeight = mix(cameraPos.y + rayDir.y * (tMin + tMax) * 0.5,
                         mix(uFogHeightMin, uFogHeightMax, 0.5), 0.5);
    float heightFactor = 1.0 - smoothstep(uFogHeightMin, uFogHeightMax, midHeight);

    float fogAmount = heightFactor * uFogDensity * (tMax - tMin);
    return 1.0 - exp(-fogAmount);
}

void main() {
    vec4 baseColor = texture2D(uTexture, vUV);
    float fog = computeHeightFog(vWorldPos, uCameraPos);
    gl_FragColor = vec4(mix(baseColor.rgb, uFogColor, fog), baseColor.a);
}
```

## 3. 体积光 Ray Marching

体积光的核心思想是：沿视线方向步进，每一步采样 shadow map 判断是否被遮挡，被照亮的部分积累光照。

```
Ray Marching 体积光原理：

  光源 (☀)
    \
     \  光线方向
      \
       \     ████████ 遮挡物
        \    ████████
         \   ████████
          \  ████████
           \ ████████
            \████████
             \
  相机 *─────────────────────> 视线方向
       |  .  .  |████████|  .  .  |
       采样点:   被遮挡    被照亮
       跳过       积累光照

  每个采样点：
  1. 从相机沿视线步进一段距离
  2. 将采样点变换到光源空间
  3. 比较采样点深度与 shadow map 深度
  4. 如果未被遮挡，累加光照贡献
```

```glsl
// 体积光 Ray Marching 片段着色器
uniform sampler2D uDepthTexture;
uniform sampler2D uShadowMap;
uniform mat4 uInverseViewProjection;
uniform mat4 uLightViewProjection;
uniform vec3 uLightPos;
uniform vec3 uLightColor;
uniform float uScattering;
uniform int uSampleCount;      // 采样步数 32~128
uniform float uMaxDistance;

varying vec2 vUV;

// 从深度缓冲重建世界坐标
vec3 reconstructWorldPos(vec2 uv, float depth) {
    vec4 clipPos = vec4(uv * 2.0 - 1.0, depth * 2.0 - 1.0, 1.0);
    vec4 worldPos = uInverseViewProjection * clipPos;
    return worldPos.xyz / worldPos.w;
}

// Shadow Map 采样
float sampleShadow(vec3 worldPos) {
    vec4 lightClipPos = uLightViewProjection * vec4(worldPos, 1.0);
    vec3 lightUV = lightClipPos.xyz / lightClipPos.w * 0.5 + 0.5;
    float shadowDepth = texture2D(uShadowMap, lightUV.xy).r;
    return step(lightUV.z - 0.005, shadowDepth); // 1 = 被照亮, 0 = 在阴影中
}

// Henyey-Greenstein 相位函数（散射方向分布）
float henyeyGreenstein(float cosTheta, float g) {
    float g2 = g * g;
    return (1.0 - g2) / (4.0 * 3.14159 * pow(1.0 + g2 - 2.0 * g * cosTheta, 1.5));
}

void main() {
    float depth = texture2D(uDepthTexture, vUV).r;
    vec3 worldPos = reconstructWorldPos(vUV, depth);
    vec3 rayDir = normalize(worldPos - uCameraPos);
    float rayLength = min(length(worldPos - uCameraPos), uMaxDistance);

    // 相位函数：前向散射为主
    vec3 toLight = normalize(uLightPos - uCameraPos);
    float cosTheta = dot(rayDir, toLight);
    float phase = henyeyGreenstein(cosTheta, uScattering);

    // Ray Marching
    float stepSize = rayLength / float(uSampleCount);
    vec3 accumulatedLight = vec3(0.0);

    for (int i = 0; i < uSampleCount; i++) {
        float t = (float(i) + 0.5) * stepSize;
        vec3 samplePos = uCameraPos + rayDir * t;

        // 采样 shadow map
        float shadow = sampleShadow(samplePos);

        // 累积光照
        accumulatedLight += shadow * uLightColor * phase * stepSize;
    }

    gl_FragColor = vec4(accumulatedLight, 1.0);
}
```

## 4. God Rays 效果

God Rays（上帝之光/光辉）是一种屏幕空间效果，通过径向模糊模拟光线散射。

```
God Rays 实现流程：

  原始场景          提亮遮挡物         径向模糊          合成
  ┌────────┐       ┌────────┐       ┌────────┐       ┌────────┐
  │  ████  │       │        │       │  ····  │       │  ████  │
  │  ████  │  ──>  │  ████  │  ──>  │ ··██·· │  ──>  │ ··██·· │
  │        │       │  ████  │       │···██···│       │···██···│
  └────────┘       └────────┘       └────────┘       └────────┘

  步骤：
  1. 渲染场景深度
  2. 根据深度判断哪些像素被光源照亮且未被遮挡
  3. 从光源位置向外做径向采样（通常 64 次）
  4. 将结果叠加到原始场景
```

```glsl
// God Rays 径向模糊片段着色器
uniform sampler2D uOccluderTexture;  // 遮挡物纹理
uniform vec2 uLightScreenPos;        // 光源屏幕坐标
uniform float uExposure;
uniform float uDecay;                // 衰减系数 0.95~0.99
uniform float uDensity;
uniform int uSamples;

varying vec2 vUV;

void main() {
    vec2 deltaTextCoord = vUV - uLightScreenPos;
    deltaTextCoord *= 1.0 / float(uSamples) * uDensity;

    vec2 uv = vUV;
    vec3 color = vec3(0.0);
    float illuminationDecay = 1.0;

    for (int i = 0; i < uSamples; i++) {
        uv -= deltaTextCoord;
        vec3 sampleColor = texture2D(uOccluderTexture, uv).rgb;
        sampleColor *= illuminationDecay;
        color += sampleColor;
        illuminationDecay *= uDecay;
    }

    gl_FragColor = vec4(color * uExposure, 1.0);
}
```

```typescript
// Three.js God Rays 后处理
class GodRaysPass {
  private renderTarget: THREE.WebGLRenderTarget;
  private material: THREE.ShaderMaterial;
  private lightScreenPos: THREE.Vector2;

  constructor(light: THREE.Light, camera: THREE.Camera) {
    this.renderTarget = new THREE.WebGLRenderTarget(
      window.innerWidth / 2,  // 降分辨率提升性能
      window.innerHeight / 2
    );

    this.material = new THREE.ShaderMaterial({
      uniforms: {
        uOccluderTexture: { value: null },
        uLightScreenPos: { value: new THREE.Vector2(0.5, 0.5) },
        uExposure: { value: 0.6 },
        uDecay: { value: 0.96 },
        uDensity: { value: 0.8 },
        uSamples: { value: 64 },
      },
      vertexShader: `varying vec2 vUV; void main() { vUV = uv; gl_Position = vec4(position, 1.0); }`,
      fragmentShader: godRaysFragShader,
    });
  }

  update(light: THREE.Light, camera: THREE.Camera): void {
    // 将光源位置投影到屏幕空间
    const pos = light.position.clone().project(camera);
    this.material.uniforms.uLightScreenPos.value.set(
      (pos.x + 1) / 2,
      (pos.y + 1) / 2
    );
  }
}
```

## 5. 介质散射模型

真实的大气散射包含两种主要机制：

```
散射类型对比：

  Rayleigh 散射（瑞利散射）          Mie 散射（米氏散射）
  ─────────────────────────         ────────────────────────
  粒子尺寸 << 波长                   粒子尺寸 ≈ 波长
  散射强度 ∝ 1/λ⁴                   散射强度与波长关系弱
  蓝光散射最强                       前向散射为主
  天空呈蓝色                         太阳周围光晕

  散射方向分布：

  Rayleigh: 对称                  Mie: 强前向散射
       *                              *
     *   *                          *   *
    *  ·  *                        * ··· *
     *   *                          *   *
       *                              *
```

```glsl
// 大气散射着色器
const float PI = 3.14159265359;

// Rayleigh 散射系数
vec3 rayleighScattering(float cosTheta) {
    return vec3(5.8e-6, 13.5e-6, 33.1e-6) *
           (1.0 + cosTheta * cosTheta) / (3.0 * PI);
}

// Mie 散射（Henyey-Greenstein 相位函数）
float miePhase(float cosTheta, float g) {
    float g2 = g * g;
    return (1.0 - g2) / (4.0 * PI * pow(1.0 + g2 - 2.0 * g * cosTheta, 1.5));
}

// 大气透射率（Beer-Lambert 定律）
vec3 transmittance(vec3 start, vec3 end, vec3 sunDir) {
    float rayLength = length(end - start);
    // 简化：假设均匀大气
    vec3 extinction = vec3(5.8e-6 + 21e-6); // Rayleigh + Mie 消光系数
    return exp(-extinction * rayLength);
}

// 完整大气散射
vec3 atmosphericScattering(vec3 rayDir, vec3 sunDir, float distance) {
    float cosTheta = dot(rayDir, sunDir);

    // Rayleigh 散射
    vec3 rayleigh = rayleighScattering(cosTheta) * vec3(0.2, 0.5, 1.0);

    // Mie 散射
    float mie = miePhase(cosTheta, 0.76) * vec3(0.5);

    // 透射率
    vec3 trans = transmittance(vec3(0.0), rayDir * distance, sunDir);

    return (rayleigh + mie) * (1.0 - trans);
}
```

## 常见误区

1. **采样步数过多导致性能崩溃**：Ray Marching 的采样步数直接决定性能。128 步在全分辨率下几乎不可接受，应该降分辨率（1/4 或 1/8）后再做 Ray Marching，最后上采样模糊。

2. **忽略相位函数**：不使用散射相位函数会导致体积光看起来均匀发光，缺乏方向感。Mie 散射的强前向散射特性是"太阳周围光晕"的关键。

3. **雾效与后处理冲突**：雾效如果在片段着色器中计算，会与后处理效果（如 HDR、Bloom）产生冲突。应该将雾效作为后处理 Pass 在色调映射之前执行。

4. **God Rays 光源位置不准确**：光源在屏幕空间的位置需要每帧重新计算，如果使用世界坐标而不做投影变换，当相机移动时效果会错位。

---

## 工程建议

1. **降分辨率渲染体积光**：体积光对分辨率不敏感，1/4 分辨率配合双线性过滤就能得到平滑效果，性能提升 16 倍。

2. **时间复用采样点**：使用蓝噪声（Blue Noise）纹理对采样点进行时间抖动（Temporal Jitter），结合 TAA 可以用 32 步达到 128 步的效果。

3. **分层雾效**：将指数雾、高度雾、体积光分别作为独立层，通过参数混合，可以快速调整场景氛围而不需要修改着色器代码。

4. **性能预算分配**：体积光通常占后处理预算的 20%~30%。如果场景已经使用了 SSAO 和 Bloom，体积光的采样步数应适当降低。

本节从最基础的指数雾开始，逐步深入到高度雾、Ray Marching 体积光、God Rays 和大气散射。雾效和体积光的核心都是沿光线方向积分光照贡献，区别在于积分方式和散射模型的复杂度。在实际项目中，降分辨率 + 模糊是体积光性能优化的关键手段，而相位函数决定了散射的视觉特征。

## 参考答案

### 练习一

**思路**：指数雾的核心是根据片段到相机的距离计算雾浓度因子，然后将场景颜色与雾颜色混合。支持 exp 和 exp2 两种模式，通过 Uniform 切换。exp 模式衰减线性，适合轻雾；exp2 模式衰减更快，适合浓雾。

**答案**：

```glsl
// 指数雾片段着色器
precision mediump float;

uniform vec3 uFogColor;
uniform float uFogDensity;
uniform int uFogMode; // 0 = exp, 1 = exp2

varying float vViewDist;

void main() {
    vec4 baseColor = texture2D(uTexture, vUV);

    float fogFactor;
    if (uFogMode == 0) {
        // exp 模式：线性衰减
        fogFactor = exp(-uFogDensity * vViewDist);
    } else {
        // exp2 模式：更快衰减
        float d = uFogDensity * vViewDist;
        fogFactor = exp(-d * d);
    }

    fogFactor = clamp(fogFactor, 0.0, 1.0);
    vec3 finalColor = mix(uFogColor, baseColor.rgb, fogFactor);
    gl_FragColor = vec4(finalColor, baseColor.a);
}
```

```typescript
// FogSystem 管理类
class ExponentialFog {
    color: THREE.Color;
    density: number;
    mode: 'exp' | 'exp2';

    constructor(options: {
        color?: THREE.Color;
        density?: number;
        mode?: 'exp' | 'exp2';
    } = {}) {
        this.color = options.color ?? new THREE.Color(0xcccccc);
        this.density = options.density ?? 0.02;
        this.mode = options.mode ?? 'exp';
    }

    applyToMaterial(material: THREE.ShaderMaterial): void {
        material.uniforms.uFogColor = { value: this.color };
        material.uniforms.uFogDensity = { value: this.density };
        material.uniforms.uFogMode = { value: this.mode === 'exp' ? 0 : 1 };
    }

    setMode(mode: 'exp' | 'exp2'): void {
        this.mode = mode;
    }
}
```

**要点**：
- `exp` 模式使用 `exp(-density * distance)`，衰减平缓，适合远景雾
- `exp2` 模式使用 `exp(-(density * distance)^2)`，衰减更快，适合浓雾/雪雾
- 雾因子应在顶点着色器中计算（逐顶点）以提升性能，或在片段着色器中计算（逐像素）以提升质量

### 练习二

**思路**：高度雾基于片段的世界坐标 Y 值计算雾浓度。低处雾浓，高处雾淡，模拟山谷/盆地的雾气效果。将高度雾与指数雾叠加，距离和高度共同影响最终雾浓度。

**答案**：

```glsl
// 高度雾 + 指数雾片段着色器
precision mediump float;

uniform vec3 uFogColor;
uniform float uFogDensity;
uniform float uFogHeightMin;    // 雾最低高度，如 -2.0
uniform float uFogHeightMax;    // 雾最高高度，如 8.0
uniform float uFogHeightFalloff; // 高度衰减系数，如 0.1
uniform float uViewDist;

varying vec3 vWorldPos;

void main() {
    vec4 baseColor = texture2D(uTexture, vUV);

    // 1. 距离雾（指数雾）
    float distFog = exp(-uFogDensity * uViewDist);

    // 2. 高度雾：低处浓度高，高处浓度低
    float height = vWorldPos.y;
    float heightFog = 1.0 - smoothstep(uFogHeightMin, uFogHeightMax, height);
    heightFog *= exp(-uFogHeightFalloff * max(0.0, height - uFogHeightMin));

    // 3. 叠加两种雾（取较大值或混合）
    float totalFog = max(1.0 - distFog, heightFog);
    totalFog = clamp(totalFog, 0.0, 1.0);

    vec3 finalColor = mix(baseColor.rgb, uFogColor, totalFog);
    gl_FragColor = vec4(finalColor, baseColor.a);
}
```

```typescript
// 高度雾配置
class HeightFog {
    color: THREE.Color;
    density: number;
    heightMin: number;
    heightMax: number;
    heightFalloff: number;

```

**要点**：
- 高度雾使用 `smoothstep` 在高度范围内平滑过渡，避免硬边
- 距离雾和高度雾的叠加方式：取 `max` 得到更浓的雾，或用加权混合
- 山谷雾效的关键是 `heightMin` 设为负值或地形最低点，让低洼处积聚雾气

### 练习三

**思路**：God Rays（光束效果）是后处理技术：从光源在屏幕上的位置出发，沿径向方向多次采样场景亮度，累加后取平均值，形成从光源向外扩散的光束。32 次采样配合指数衰减权重，近光源处贡献更大。

**答案**：

```glsl
// God Rays 后处理片段着色器
precision mediump float;

uniform sampler2D uScene;
uniform vec2 uLightScreenPos; // 光源在屏幕上的位置 (0~1)
uniform float uExposure;
uniform float uDecay;     // 衰减系数，如 0.96
uniform float uDensity;   // 密度，如 1.0
uniform float uWeight;    // 权重，如 0.4

varying vec2 vUV;

void main() {
    vec2 texCoord = vUV;
    vec2 deltaTextCoord = texCoord - uLightScreenPos;

    // 32 次径向采样
    const int NUM_SAMPLES = 32;
    deltaTextCoord *= 1.0 / float(NUM_SAMPLES) * uDensity;

    vec3 color = texture2D(uScene, texCoord).rgb;
    float illuminationDecay = 1.0;
    vec3 accum = vec3(0.0);

    for (int i = 0; i < NUM_SAMPLES; i++) {
        texCoord -= deltaTextCoord;
        vec3 sampleColor = texture2D(uScene, texCoord).rgb;

        // 指数衰减：越远离光源衰减越大
        sampleColor *= illuminationDecay * uWeight;
        accum += sampleColor;
        illuminationDecay *= uDecay;
    }

    // 将光束叠加到原场景
    color += accum * uExposure;
    gl_FragColor = vec4(color, 1.0);
}
```

```typescript
// God Rays 后处理 Pass
class GodRaysPass {
    private material: THREE.ShaderMaterial;
    private lightScreenPos: THREE.Vector2;

    constructor() {
        this.lightScreenPos = new THREE.Vector2(0.5, 0.8);
        this.material = new THREE.ShaderMaterial({
            uniforms: {
                uScene: { value: null },
                uLightScreenPos: { value: this.lightScreenPos },
                uExposure: { value: 0.3 },
                uDecay: { value: 0.96 },
                uDensity: { value: 1.0 },
                uWeight: { value: 0.4 },
            },
            vertexShader: postProcessVertexShader,
            fragmentShader: godRaysFragmentShader,
        });
    }

    updateLightPosition(light: THREE.Light, camera: THREE.Camera): void {
        // 将光源位置投影到屏幕空间
        const pos = light.position.clone().project(camera);
        this.lightScreenPos.set(
            (pos.x + 1.0) * 0.5,
            (pos.y + 1.0) * 0.5
        );
    }
}
```

**要点**：
- 采样方向从当前像素指向光源，每次步进固定偏移量
- `illuminationDecay` 指数衰减使近光源处采样权重更高，形成光束效果
- 性能优化：可先将场景降分辨率到 1/2 或 1/4 再做径向采样

### 练习四

**思路**：体积光通过 Ray Marching 从相机到片段的光路上逐步采样。每步检查该点是否被光源看到（采样 Shadow Map），如果可见则累加散射贡献。这是 Volumetric Lighting 的核心算法。

**答案**：

```glsl
// 体积光 Ray Marching 片段着色器
precision highp float;

uniform sampler2D uDepthTex;
uniform sampler2D uShadowMap;
uniform mat4 uLightViewProj;
uniform vec3 uLightDir;
uniform vec3 uCameraPos;
uniform float uScattering;      // 散射系数
uniform int uSampleCount;       // 采样步数，如 64
uniform float uMaxDist;         // 最大步进距离

varying vec2 vUV;

// 重建世界坐标
vec3 reconstructWorldPos(vec2 uv, float depth) {
    vec4 clipPos = vec4(uv * 2.0 - 1.0, depth * 2.0 - 1.0, 1.0);
    vec4 worldPos = uInvViewProj * clipPos;
    return worldPos.xyz / worldPos.w;
}

// 检查点是否在阴影中
float sampleShadow(vec3 worldPos) {
    vec4 lightClipPos = uLightViewProj * vec4(worldPos, 1.0);
    vec3 lightUV = lightClipPos.xyz / lightClipPos.w * 0.5 + 0.5;
    float shadowDepth = texture2D(uShadowMap, lightUV.xy).r;
    return step(lightUV.z - 0.005, shadowDepth);
}

void main() {
    float depth = texture2D(uDepthTex, vUV).r;
    vec3 worldPos = reconstructWorldPos(vUV, depth);
    vec3 rayDir = worldPos - uCameraPos;
    float rayLength = length(rayDir);
    rayDir = normalize(rayDir);

    // 限制步进距离
    rayLength = min(rayLength, uMaxDist);
    float stepSize = rayLength / float(uSampleCount);

    vec3 accumLight = vec3(0.0);
    vec3 currentPos = uCameraPos;

    for (int i = 0; i < 64; i++) {
        if (i >= uSampleCount) break;

        currentPos += rayDir * stepSize;

        // 检查该点是否被光源照亮
        float shadow = sampleShadow(currentPos);

        // 累加散射贡献
        accumLight += shadow * uScattering;
    }

    accumLight /= float(uSampleCount);

    // 叠加到原场景
    vec3 sceneColor = texture2D(uScene, vUV).rgb;
    vec3 finalColor = sceneColor + accumLight * vec3(1.0, 0.95, 0.8);
    gl_FragColor = vec4(finalColor, 1.0);
}
```

**要点**：
- 每步需要将世界坐标变换到光源空间采样 Shadow Map，这是体积光最耗时的部分
- 降分辨率渲染（如 1/4 分辨率）+ 双边上采样可以大幅降低采样开销
- 使用蓝噪声对采样位置进行时间抖动，配合 TAA 可以用更少步数获得平滑效果

### 练习五

**思路**：Rayleigh 散射描述大气分子对短波长光（蓝光）的散射，产生蓝天效果；Mie 散射描述气溶胶对光的散射，产生日落时的橙红色光晕。在体积光的每一步中，根据散射类型计算不同波长的散射强度，模拟日落时分的大气效果。

**答案**：

```glsl
// Rayleigh + Mie 散射体积光片段着色器
precision highp float;

uniform vec3 uSunDir;
uniform vec3 uSunColor;
uniform float uRayleighScattering;
uniform float uMieScattering;
uniform float uMieG;           // Mie 相位函数参数，如 0.76

varying vec2 vUV;

// Rayleigh 相位函数
float rayleighPhase(float cosTheta) {
    return 3.0 / (16.0 * 3.14159) * (1.0 + cosTheta * cosTheta);
}

// Henyey-Greenstein 相位函数（Mie 散射）
float henyeyGreenstein(float cosTheta, float g) {
    float g2 = g * g;
    return (1.0 - g2) / (4.0 * 3.14159 * pow(1.0 + g2 - 2.0 * g * cosTheta, 1.5));
}

void main() {
    float depth = texture2D(uDepthTex, vUV).r;
    vec3 worldPos = reconstructWorldPos(vUV, depth);
    vec3 rayDir = normalize(worldPos - uCameraPos);

    float cosTheta = dot(rayDir, uSunDir);

    // Rayleigh 散射：波长依赖，蓝光散射最强
    vec3 rayleighCoeff = uRayleighScattering * vec3(5.8e-6, 13.5e-6, 33.1e-6);
    float rayleighPhaseVal = rayleighPhase(cosTheta);

    // Mie 散射：波长无关，前向散射为主
    float miePhaseVal = henyeyGreenstein(cosTheta, uMieG);

    // 体积光 Ray Marching
    float rayLength = min(length(worldPos - uCameraPos), uMaxDist);
    float stepSize = rayLength / float(uSampleCount);

    vec3 accumRayleigh = vec3(0.0);
    vec3 accumMie = vec3(0.0);
    vec3 currentPos = uCameraPos;

    for (int i = 0; i < 64; i++) {
        if (i >= uSampleCount) break;
        currentPos += rayDir * stepSize;

        float shadow = sampleShadow(currentPos);

        // Rayleigh 累加
        accumRayleigh += shadow * rayleighCoeff * rayleighPhaseVal;
        // Mie 累加
        accumMie += shadow * vec3(uMieScattering) * miePhaseVal;
    }

    accumRayleigh /= float(uSampleCount);
    accumMie /= float(uSampleCount);

    // 日落效果：太阳低角度时 Rayleigh 散射使光呈橙红色
    vec3 sunColorTint = uSunColor * vec3(1.0, 0.6, 0.3);
    vec3 scattering = (accumRayleigh + accumMie) * sunColorTint;

    vec3 sceneColor = texture2D(uScene, vUV).rgb;
    gl_FragColor = vec4(sceneColor + scattering, 1.0);
}
```

**要点**：
- Rayleigh 散射系数与波长的四次方成反比（λ⁻⁴），所以蓝光散射最强
- Mie 散射的 Henyey-Greenstein 相位函数参数 g 控制前向/后向散射比例（g=0.76 为强前向散射）
- 日落时太阳角度低，光路更长，短波长光被散射殆尽，剩下橙红色光
