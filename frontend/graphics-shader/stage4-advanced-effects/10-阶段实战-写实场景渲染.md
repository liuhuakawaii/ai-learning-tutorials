## 2. 渲染管线执行顺序

```
渲染管线执行流程：

  Frame Start
      │
      ▼
  ┌──────────────────┐
  │ 1. 更新相机/输入  │
  └────────┬─────────┘
           ▼
  ┌──────────────────┐
  │ 2. 地形 LOD 更新  │
  └────────┬─────────┘
           ▼
  ┌──────────────────┐
  │ 3. Shadow Pass    │  ──> 生成级联阴影贴图
  │    (3 级联)       │
  └────────┬─────────┘
           ▼
  ┌──────────────────┐
  │ 4. 水面反射 Pass  │  ──> 渲染镜像场景到反射纹理
  └────────┬─────────┘
           ▼
  ┌──────────────────┐
  │ 5. G-Buffer Pass  │  ──> 写入几何信息
  │    (不透明物体)    │
  └────────┬─────────┘
           ▼
  ┌──────────────────┐
  │ 6. SSAO Pass      │  ──> 计算环境光遮蔽
  └────────┬─────────┘
           ▼
  ┌──────────────────┐
  │ 7. Deferred       │  ──> 光照计算
  │    Lighting Pass  │
  └────────┬─────────┘
           ▼
  ┌──────────────────┐
  │ 8. Forward Pass   │  ──> 透明物体（水面、粒子）
  └────────┬─────────┘
           ▼
  ┌──────────────────┐
  │ 9. Post Processing│  ──> Bloom + 色调映射 + 色彩校正
  └────────┬─────────┘
           ▼
  ┌──────────────────┐
  │ 10. 输出到屏幕    │
  └──────────────────┘
```

```typescript
// 渲染循环
render(): void {
  const delta = this.clock.getDelta();
  const elapsed = this.clock.getElapsedTime();

  // 1. 更新
  this.updateCamera(delta);
  this.terrainSystem.updateLOD(this.camera.position);
  this.particleSystem.update(delta, elapsed);

  // 2. Shadow Pass
  this.shadowSystem.update(this.camera, this.sunLight);
  for (let i = 0; i < 3; i++) {
    this.renderer.setRenderTarget(this.shadowSystem.getRenderTarget(i));
    this.renderer.clear();
    this.renderShadowCasters(this.shadowSystem.getCamera(i));
  }

  // 3. 水面反射
  this.waterSystem.updateReflection(this.renderer, this.scene, this.camera);

  // 4. G-Buffer Pass
  this.renderer.setRenderTarget(this.gBuffer.renderTarget);
  this.renderer.clear();
  this.renderOpaqueObjects();

  // 5. SSAO
  this.ssaoSystem.render(this.renderer, this.gBuffer, this.camera);

  // 6. Deferred Lighting
  this.renderer.setRenderTarget(this.lightingTarget);
  this.renderer.clear();
  this.deferredLightingPass.render(this.gBuffer, this.shadowSystem, this.ssaoSystem);

  // 7. Forward Pass（透明物体）
  this.renderer.setRenderTarget(null);
  this.waterSystem.render(this.renderer, this.camera, this.lightingTarget.texture);
  this.particleSystem.render(this.renderer, this.camera);

  // 8. Post Processing
  this.postProcessing.render(this.renderer, this.lightingTarget.texture);
}
```

### 3.1 地形 + PBR 材质

```glsl
// 地形 PBR 片段着色器
uniform sampler2D uGrassAlbedo, uGrassNormal, uGrassRoughness;
uniform sampler2D uRockAlbedo, uRockNormal, uRockRoughness;
uniform sampler2D uSnowAlbedo, uSnowNormal, uSnowRoughness;
uniform sampler2D uSplatMap;

varying vec3 vWorldPos;
varying vec3 vNormal;
varying vec2 vUV;

// PBR 光照计算函数（复用第 5 节的实现）
vec3 calculatePBR(vec3 albedo, vec3 normal, float metallic, float roughness,
                  vec3 lightDir, vec3 lightColor, vec3 viewDir);

void main() {
    vec4 splat = texture2D(uSplatMap, vUV);

    // 各层材质
    vec3 grassAlbedo = texture2D(uGrassAlbedo, vUV * 20.0).rgb;
    vec3 rockAlbedo = texture2D(uRockAlbedo, vUV * 15.0).rgb;
    vec3 snowAlbedo = texture2D(uSnowAlbedo, vUV * 25.0).rgb;

    vec3 grassNormal = texture2D(uGrassNormal, vUV * 20.0).rgb * 2.0 - 1.0;
    vec3 rockNormal = texture2D(uRockNormal, vUV * 15.0).rgb * 2.0 - 1.0;
    vec3 snowNormal = texture2D(uSnowNormal, vUV * 25.0).rgb * 2.0 - 1.0;

    // 混合材质
    vec3 albedo = grassAlbedo * splat.r + rockAlbedo * splat.g + snowAlbedo * splat.b;
    vec3 normal = normalize(
        grassNormal * splat.r + rockNormal * splat.g + snowNormal * splat.b
    );
    float roughness = 0.8 * splat.r + 0.9 * splat.g + 0.3 * splat.b;

    // 世界空间法线
    vec3 worldNormal = normalize(vNormal + normal * 0.5);

    // PBR 光照
    vec3 viewDir = normalize(uCameraPos - vWorldPos);
    vec3 color = calculatePBR(albedo, worldNormal, 0.0, roughness,
                              uSunDirection, uSunColor, viewDir);

    gl_FragColor = vec4(color, 1.0);
}
```

### 3.2 水面 + 阴影

```glsl
// 水面着色器中接收阴影
uniform sampler2D uShadowMap;
uniform mat4 uLightSpaceMatrix;

float calculateShadow(vec3 worldPos) {
    vec4 lightSpacePos = uLightSpaceMatrix * vec4(worldPos, 1.0);
    vec3 projCoords = lightSpacePos.xyz / lightSpacePos.w * 0.5 + 0.5;

    float closestDepth = texture2D(uShadowMap, projCoords.xy).r;
    float currentDepth = projCoords.z;

    float bias = max(0.005 * (1.0 - dot(vec3(0, 1, 0), uSunDirection)), 0.001);
    float shadow = currentDepth - bias > closestDepth ? 0.6 : 1.0;

    return shadow;
}

// 在水面最终合成中应用阴影
void main() {
    // ... 反射、折射、高光计算

    float shadow = calculateShadow(vWorldPos);
    color *= shadow; // 水面也受阴影影响

    gl_FragColor = vec4(color, 0.9);
}
```

```typescript
// 粒子系统集成
class ParticleSystem {
  private fireEmitter: GPUParticleEmitter;
  private smokeEmitter: GPUParticleEmitter;
  private embersEmitter: GPUParticleEmitter;

  update(delta: number, elapsed: number): void {
    // 火焰粒子
    this.fireEmitter.update({
      deltaTime: delta,
      gravity: new THREE.Vector3(0, 2, 0), // 向上飘
      wind: new THREE.Vector3(
        Math.sin(elapsed * 0.5) * 0.5,
        0,
        Math.cos(elapsed * 0.3) * 0.3
      ),
      lifeDecay: 0.8,
    });

    // 烟雾粒子
    this.smokeEmitter.update({
      deltaTime: delta,
      gravity: new THREE.Vector3(0, 0.5, 0),
      wind: new THREE.Vector3(1, 0, 0), // 风向
      lifeDecay: 0.3,
    });

    // 火星粒子
    this.embersEmitter.update({
      deltaTime: delta,
      gravity: new THREE.Vector3(0, -2, 0), // 下落
      wind: new THREE.Vector3(
        Math.sin(elapsed) * 2,
        0,
        Math.cos(elapsed * 0.7) * 1.5
      ),
      lifeDecay: 1.5,
    });
  }

  render(renderer: THREE.WebGLRenderer, camera: THREE.Camera): void {
    // 按混合模式排序渲染
    // 1. 烟雾（Alpha Blending）
    this.smokeEmitter.render(renderer, camera);

    // 2. 火焰（Additive Blending）
    this.fireEmitter.render(renderer, camera);

    // 3. 火星（Additive Blending）
    this.embersEmitter.render(renderer, camera);
  }
}
```

```typescript
class PerformanceManager {
  private frameTimeHistory: number[] = [];
  private targetFPS: number = 60;
  private currentQualityLevel: QualityLevel = 'high';

  update(deltaTime: number): void {
    this.frameTimeHistory.push(deltaTime);
    if (this.frameTimeHistory.length > 60) {
      this.frameTimeHistory.shift();
    }

    const avgFrameTime = this.frameTimeHistory.reduce((a, b) => a + b, 0)
      / this.frameTimeHistory.length;

    // 动态调整质量
    if (avgFrameTime > 0.02 && this.currentQualityLevel === 'high') {
      this.setQuality('medium');
    } else if (avgFrameTime > 0.025 && this.currentQualityLevel === 'medium') {
      this.setQuality('low');
    } else if (avgFrameTime < 0.012 && this.currentQualityLevel !== 'high') {
      this.setQuality('high');
    }
  }
```

## 常见误区

1. **各系统独立开发不考虑集成**：每个子系统单独工作良好，但集成后可能出现 Z-fighting、混合顺序错误、Pass 依赖缺失等问题。应该在开发初期就确定渲染管线的执行顺序和数据依赖关系。

2. **不做性能预算**：每个效果都追求最高质量，最终帧率不可接受。应该在项目开始时就分配每个 Pass 的时间预算，超标时主动降级。

3. **忽略移动端适配**：桌面端运行良好的方案在移动端可能完全不可用。应该设计分级质量系统，根据平台自动调整参数。

4. **后处理效果堆叠过多**：Bloom + SSAO + 体积光 + 色调映射 + 色彩校正 + 暗角——过多后处理会累积带宽消耗和延迟。应该合并 Pass，减少纹理读写次数。

---

## 工程建议

1. **模块化设计**：每个渲染系统（地形、水面、粒子、阴影等）应该是独立模块，通过统一接口与渲染管线交互。这样可以方便地启用/禁用单个模块进行调试和性能分析。

2. **数据驱动配置**：场景参数（光照、雾效、后处理强度等）应该通过 JSON 配置文件管理，而不是硬编码在着色器中。这样可以快速调整效果而不需要重新编译。

3. **渲染统计面板**：实现一个实时统计面板，显示 Draw Call 数量、三角形数量、GPU 时间、各 Pass 耗时等信息。这是性能调优的基础工具。

4. **渐进式构建**：不要一次性实现所有效果。先搭建基础渲染管线（地形 + PBR + 阴影），确认稳定后再逐步添加水面、粒子、后处理等效果。每添加一个效果都应该进行性能测试。

## 小结

本节将前面 9 节课程的技术整合为一个完整的写实场景渲染系统。核心挑战不在于单个效果的实现，而在于系统架构设计、渲染管线编排和性能调优。一个好的架构应该支持模块化开发、数据驱动配置和自适应质量调整。记住：**60fps 下每帧只有 16.6ms，每 1ms 都需要精打细算。**

**思路**：延迟渲染管线的核心是将几何信息先写入 G-Buffer（多渲染目标），再在屏幕空间统一计算光照。需要三个 Pass 按顺序执行：Shadow Pass 生成阴影贴图 → G-Buffer Pass 写入几何属性 → Deferred Lighting Pass 读取 G-Buffer 和阴影贴图计算最终颜色。关键是理解每个 Pass 的输入输出和它们之间的数据依赖。

**答案**：

```typescript
import * as THREE from 'three';

// ==================== G-Buffer 管理 ====================
class GBuffer {
  public renderTarget: THREE.WebGLRenderTarget;
  public albedoTexture: THREE.Texture;
  public normalTexture: THREE.Texture;
  public depthTexture: THREE.DepthTexture;
  public materialTexture: THREE.Texture; // metallic + roughness + AO

  constructor(width: number, height: number) {
    this.renderTarget = new THREE.WebGLRenderTarget(width, height, {
      count: 4, // 4 个颜色附件
      format: THREE.RGBAFormat,
      type: THREE.UnsignedByteType,
      minFilter: THREE.NearestFilter,
      magFilter: THREE.NearestFilter,
    });

    // 分配各通道
    this.albedoTexture = this.renderTarget.textures[0];   // RT0: albedo + shadow
    this.normalTexture = this.renderTarget.textures[1];   // RT1: world normal
    this.materialTexture = this.renderTarget.textures[2];  // RT2: metallic, roughness, AO
    this.depthTexture = this.renderTarget.depthTexture!;

    // 配置深度纹理
    this.depthTexture.format = THREE.DepthFormat;
    this.depthTexture.type = THREE.UnsignedIntType;
  }

  resize(width: number, height: number): void {
    this.renderTarget.setSize(width, height);
  }
}

// ==================== G-Buffer 材质 ====================
const gBufferVertexShader = `
  varying vec3 vWorldPos;
  varying vec3 vNormal;
  varying vec2 vUV;

  void main() {
    vec4 worldPos = modelMatrix * vec4(position, 1.0);
    vWorldPos = worldPos.xyz;
    vNormal = normalize((modelMatrix * vec4(normal, 0.0)).xyz);
    vUV = uv;
    gl_Position = projectionMatrix * viewMatrix * worldPos;
  }
`;

const gBufferFragmentShader = `
  uniform vec3 uAlbedo;
  uniform float uMetallic;
  uniform float uRoughness;
  uniform float uAO;

  varying vec3 vWorldPos;
  varying vec3 vNormal;
  varying vec2 vUV;

  layout(location = 0) out vec4 outAlbedo;
  layout(location = 1) out vec4 outNormal;
  layout(location = 2) out vec4 outMaterial;

  void main() {
    outAlbedo = vec4(uAlbedo, 1.0);
    // 法线编码到 [0,1] 范围
    outNormal = vec4(vNormal * 0.5 + 0.5, 1.0);
    outMaterial = vec4(uMetallic, uRoughness, uAO, 1.0);
  }
`;

// ==================== 级联阴影 ====================
class CascadedShadowMap {
  private shadowMaps: THREE.WebGLRenderTarget[];
  private shadowCameras: THREE.OrthographicCamera[];
  private cascadeSplits: number[];

  constructor(splits: number[], resolutions: number[]) {
    this.cascadeSplits = splits;
    this.shadowMaps = [];
    this.shadowCameras = [];

    for (let i = 0; i < splits.length; i++) {
      this.shadowMaps.push(
        new THREE.WebGLRenderTarget(resolutions[i], resolutions[i], {
          depthTexture: new THREE.DepthTexture(),
          format: THREE.RGBAFormat,
        })
      );
      this.shadowCameras.push(new THREE.OrthographicCamera(-50, 50, 50, -50, 0.1, 500));
    }
  }

  update(mainCamera: THREE.PerspectiveCamera, light: THREE.DirectionalLight): void {
    for (let i = 0; i < this.cascadeSplits.length; i++) {
      const camera = this.shadowCameras[i];
      // 根据主相机视锥裁剪当前级联范围
      const near = i === 0 ? mainCamera.near : this.cascadeSplits[i - 1];
      const far = this.cascadeSplits[i];
      // 将阴影相机放置在光源方向，覆盖当前级联范围
      camera.position.copy(light.position).normalize().multiplyScalar(far * 0.5);
      camera.lookAt(0, 0, 0);
      camera.left = -far * 0.5;
      camera.right = far * 0.5;
      camera.top = far * 0.5;
      camera.bottom = -far * 0.5;
      camera.near = 0.1;
      camera.far = far;
      camera.updateProjectionMatrix();
    }
  }

  getRenderTarget(index: number): THREE.WebGLRenderTarget {
    return this.shadowMaps[index];
  }

  getCamera(index: number): THREE.OrthographicCamera {
    return this.shadowCameras[index];
  }

  getShadowMatrices(): THREE.Matrix4[] {
    return this.shadowCameras.map((cam) => cam.matrixWorldInverse);
  }
}

// ==================== 延迟光照 Pass ====================
const deferredLightingVertex = `
  varying vec2 vUV;
  void main() {
    vUV = uv;
    gl_Position = vec4(position.xy * 2.0, 0.0, 1.0);
  }
`;

const deferredLightingFragment = `
  uniform sampler2D uAlbedoTex;
  uniform sampler2D uNormalTex;
  uniform sampler2D uMaterialTex;
  uniform sampler2D uDepthTex;
  uniform sampler2D uSSAOTex;
  uniform sampler2D uShadowMap0;
  uniform sampler2D uShadowMap1;
  uniform sampler2D uShadowMap2;
  uniform mat4 uLightSpaceMatrix0;
  uniform mat4 uLightSpaceMatrix1;
  uniform mat4 uLightSpaceMatrix2;
  uniform vec3 uSunDirection;
  uniform vec3 uSunColor;
  uniform vec3 uCameraPos;
  uniform mat4 uInverseViewProjection;

  varying vec2 vUV;

  // PBR 函数
  float distributionGGX(vec3 N, vec3 H, float roughness) {
    float a = roughness * roughness;
    float a2 = a * a;
    float NdotH = max(dot(N, H), 0.0);
    float denom = NdotH * NdotH * (a2 - 1.0) + 1.0;
    return a2 / (3.14159 * denom * denom);
  }

  float geometrySmith(vec3 N, vec3 V, vec3 L, float roughness) {
    float r = roughness + 1.0;
    float k = (r * r) / 8.0;
    float NdotV = max(dot(N, V), 0.0);
    float NdotL = max(dot(N, L), 0.0);
    float ggx1 = NdotV / (NdotV * (1.0 - k) + k);
    float ggx2 = NdotL / (NdotL * (1.0 - k) + k);
    return ggx1 * ggx2;
  }

  vec3 fresnelSchlick(float cosTheta, vec3 F0) {
    return F0 + (1.0 - F0) * pow(clamp(1.0 - cosTheta, 0.0, 1.0), 5.0);
  }

  float sampleShadow(vec3 worldPos, sampler2D shadowMap, mat4 lightMatrix) {
    vec4 ls = lightMatrix * vec4(worldPos, 1.0);
    vec3 proj = ls.xyz / ls.w * 0.5 + 0.5;
    float closest = texture(shadowMap, proj.xy).r;
    float bias = max(0.005 * (1.0 - dot(vec3(0,1,0), uSunDirection)), 0.001);
    return proj.z - bias > closest ? 0.6 : 1.0;
  }

  void main() {
    float depth = texture(uDepthTex, vUV).r;
    if (depth >= 1.0) discard;

    vec3 albedo = texture(uAlbedoTex, vUV).rgb;
    vec3 normal = normalize(texture(uNormalTex, vUV).rgb * 2.0 - 1.0);
    vec3 material = texture(uMaterialTex, vUV).rgb;
    float metallic = material.r;
    float roughness = material.g;
    float ao = material.b;
    float ssao = texture(uSSAOTex, vUV).r;

    // 从深度重建世界坐标
    vec4 clipPos = vec4(vUV * 2.0 - 1.0, depth * 2.0 - 1.0, 1.0);
    vec4 worldPos = uInverseViewProjection * clipPos;
    vec3 worldPosition = worldPos.xyz / worldPos.w;

    vec3 V = normalize(uCameraPos - worldPosition);
    vec3 L = normalize(-uSunDirection);
    vec3 H = normalize(V + L);
    vec3 F0 = mix(vec3(0.04), albedo, metallic);

    // PBR 光照
    float NDF = distributionGGX(normal, H, roughness);
    float G = geometrySmith(normal, V, L, roughness);
    vec3 F = fresnelSchlick(max(dot(H, V), 0.0), F0);
    vec3 kD = (1.0 - F) * (1.0 - metallic);
    vec3 diffuse = kD * albedo / 3.14159;
    vec3 specular = (NDF * G * F) / max(4.0 * max(dot(normal, V), 0.0) * max(dot(normal, L), 0.0), 0.001);
    float NdotL = max(dot(normal, L), 0.0);
    vec3 Lo = (diffuse + specular) * uSunColor * NdotL;

    // 级联阴影
    float shadow;
    if (length(worldPosition - uCameraPos) < 30.0) {
      shadow = sampleShadow(worldPosition, uShadowMap0, uLightSpaceMatrix0);
    } else if (length(worldPosition - uCameraPos) < 100.0) {
      shadow = sampleShadow(worldPosition, uShadowMap1, uLightSpaceMatrix1);
    } else {
      shadow = sampleShadow(worldPosition, uShadowMap2, uLightSpaceMatrix2);
    }

    // 环境光
    vec3 ambient = vec3(0.03) * albedo * ao * ssao;

    vec3 color = ambient + Lo * shadow;
    gl_FragColor = vec4(color, 1.0);
  }
`;

// ==================== 渲染管线主类 ====================
class DeferredRenderPipeline {
  private renderer: THREE.WebGLRenderer;
  private gBuffer: GBuffer;
  private shadowMap: CascadedShadowMap;
  private lightingTarget: THREE.WebGLRenderTarget;
  private ssaoTarget: THREE.WebGLRenderTarget;

  private gBufferMaterial: THREE.ShaderMaterial;
  private lightingMaterial: THREE.ShaderMaterial;
  private lightingQuad: THREE.Mesh;

  constructor(renderer: THREE.WebGLRenderer) {
    this.renderer = renderer;
    const size = renderer.getSize(new THREE.Vector2());

    this.gBuffer = new GBuffer(size.x, size.y);
    this.shadowMap = new CascadedShadowMap([30, 100, 300], [2048, 1024, 512]);

    this.lightingTarget = new THREE.WebGLRenderTarget(size.x, size.y);
    this.ssaoTarget = new THREE.WebGLRenderTarget(size.x, size.y, {
      format: THREE.RedFormat,
    });

    this.gBufferMaterial = new THREE.ShaderMaterial({
      vertexShader: gBufferVertexShader,
      fragmentShader: gBufferFragmentShader,
    });

    this.lightingMaterial = new THREE.ShaderMaterial({
      vertexShader: deferredLightingVertex,
      fragmentShader: deferredLightingFragment,
      uniforms: {
        uAlbedoTex: { value: this.gBuffer.albedoTexture },
        uNormalTex: { value: this.gBuffer.normalTexture },
        uMaterialTex: { value: this.gBuffer.materialTexture },
        uDepthTex: { value: this.gBuffer.depthTexture },
        uSSAOTex: { value: this.ssaoTarget.texture },
        uSunDirection: { value: new THREE.Vector3(0.5, 1, 0.3).normalize() },
        uSunColor: { value: new THREE.Color(1.0, 0.95, 0.9) },
        uCameraPos: { value: new THREE.Vector3() },
      },
    });

    const quad = new THREE.PlaneGeometry(2, 2);
    this.lightingQuad = new THREE.Mesh(quad, this.lightingMaterial);
  }

  render(scene: THREE.Scene, camera: THREE.Camera): void {
    const light = scene.children.find(
      (c) => c instanceof THREE.DirectionalLight
    ) as THREE.DirectionalLight;

    // 1. Shadow Pass
    this.shadowMap.update(camera as THREE.PerspectiveCamera, light);
    for (let i = 0; i < 3; i++) {
      this.renderer.setRenderTarget(this.shadowMap.getRenderTarget(i));
      this.renderer.clear();
      this.renderer.render(scene, this.shadowMap.getCamera(i));
    }

    // 2. G-Buffer Pass
    this.renderer.setRenderTarget(this.gBuffer.renderTarget);
    this.renderer.clear();
    scene.traverse((obj) => {
      if ((obj as THREE.Mesh).isMesh) {
        const mesh = obj as THREE.Mesh;
        const prevMat = mesh.material;
        mesh.material = this.gBufferMaterial;
        this.renderer.render(mesh, camera);
        mesh.material = prevMat;
      }
    });

    // 3. SSAO Pass（简化，实际需要单独的 SSAO 着色器）
    this.renderer.setRenderTarget(this.ssaoTarget);
    this.renderer.clear();
    this.renderer.setClearColor(0xffffff);
    // ... 使用 SSAO 着色器渲染全屏四边形

    // 4. Deferred Lighting Pass
    this.lightingMaterial.uniforms.uCameraPos.value.copy(
      (camera as THREE.PerspectiveCamera).position
    );
    this.renderer.setRenderTarget(this.lightingTarget);
    this.renderer.clear();
    // 使用正交相机渲染全屏四边形
    const orthoCamera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);
    this.renderer.render(
      new THREE.Scene().add(this.lightingQuad) as unknown as THREE.Scene,
      orthoCamera
    );
  }
}
```

**要点**：
- G-Buffer 使用多渲染目标（MRT），一次 Draw Call 写入 albedo、normal、material 三个纹理
- 级联阴影按距离选择不同分辨率的阴影贴图，近处精度高、远处精度低
- 延迟光照 Pass 使用全屏四边形 + 正交相机，逐像素读取 G-Buffer 计算 PBR
- 从深度缓冲重建世界坐标需要 inverse(viewProjection) 矩阵

### 练习二

**思路**：地形系统的核心是高度图驱动的顶点位移 + Splat Map 驱动的多材质混合。在顶点着色器中采样高度图设置地形起伏，在片段着色器中根据 Splat Map 的 RGB 通道分别混合草地、岩石、雪地三种材质的 albedo 和 normal 贴图，最后用 PBR 光照模型计算最终颜色。

**答案**：

```glsl
// ==================== 地形顶点着色器 ====================
uniform sampler2D uHeightMap;
uniform float uHeightScale;
uniform float uTerrainSize;

varying vec3 vWorldPos;
varying vec3 vNormal;
varying vec2 vUV;

void main() {
    vUV = uv;

    // 从高度图采样位移
    float height = texture2D(uHeightMap, uv).r * uHeightScale;
    vec3 displacedPos = position;
    displacedPos.y += height;

    // 计算法线（使用相邻采样差分）
    float texelSize = 1.0 / 512.0;
    float hL = texture2D(uHeightMap, uv + vec2(-texelSize, 0.0)).r * uHeightScale;
    float hR = texture2D(uHeightMap, uv + vec2(texelSize, 0.0)).r * uHeightScale;
    float hD = texture2D(uHeightMap, uv + vec2(0.0, -texelSize)).r * uHeightScale;
    float hU = texture2D(uHeightMap, uv + vec2(0.0, texelSize)).r * uHeightScale;

    vec3 calcNormal = normalize(vec3(hL - hR, 2.0 * texelSize * uTerrainSize, hD - hU));

    vWorldPos = (modelMatrix * vec4(displacedPos, 1.0)).xyz;
    vNormal = normalize((modelMatrix * vec4(calcNormal, 0.0)).xyz);

    gl_Position = projectionMatrix * viewMatrix * vec4(vWorldPos, 1.0);
}

// ==================== 地形片段着色器 ====================
uniform sampler2D uGrassAlbedo, uGrassNormal, uGrassRoughness;
uniform sampler2D uRockAlbedo, uRockNormal, uRockRoughness;
uniform sampler2D uSnowAlbedo, uSnowNormal, uSnowRoughness;
uniform sampler2D uSplatMap;
uniform vec3 uSunDirection;
uniform vec3 uSunColor;
uniform vec3 uCameraPos;

varying vec3 vWorldPos;
varying vec3 vNormal;
varying vec2 vUV;

// PBR 辅助函数
float D_GGX(float NdotH, float roughness) {
    float a = roughness * roughness;
    float a2 = a * a;
    float denom = NdotH * NdotH * (a2 - 1.0) + 1.0;
    return a2 / (3.14159 * denom * denom);
}

vec3 fresnelSchlick(float cosTheta, vec3 F0) {
    return F0 + (1.0 - F0) * pow(1.0 - cosTheta, 5.0);
}

vec3 calculatePBR(vec3 albedo, vec3 N, float metallic, float roughness,
                  vec3 lightDir, vec3 lightColor, vec3 viewDir) {
    vec3 H = normalize(viewDir + lightDir);
    float NdotL = max(dot(N, lightDir), 0.0);
    float NdotV = max(dot(N, viewDir), 0.0);
    float NdotH = max(dot(N, H), 0.0);
    float HdotV = max(dot(H, viewDir), 0.0);

    vec3 F0 = mix(vec3(0.04), albedo, metallic);
    vec3 F = fresnelSchlick(HdotV, F0);
    float D = D_GGX(NdotH, roughness);

    vec3 kD = (1.0 - F) * (1.0 - metallic);
    vec3 diffuse = kD * albedo / 3.14159;
    vec3 specular = D * F / max(4.0 * NdotV * NdotL, 0.001);

    return (diffuse + specular) * lightColor * NdotL;
}

void main() {
    // 采样 Splat Map（R=草地 G=岩石 B=雪地）
    vec4 splat = texture2D(uSplatMap, vUV);

    // 多尺度 UV 避免纹理拉伸
    vec2 grassUV = vUV * 20.0;
    vec2 rockUV = vUV * 15.0;
    vec2 snowUV = vUV * 25.0;

    // 采样各层材质
    vec3 grassAlbedo = texture2D(uGrassAlbedo, grassUV).rgb;
    vec3 rockAlbedo = texture2D(uRockAlbedo, rockUV).rgb;
    vec3 snowAlbedo = texture2D(uSnowAlbedo, snowUV).rgb;

    vec3 grassNormal = normalize(texture2D(uGrassNormal, grassUV).rgb * 2.0 - 1.0);
    vec3 rockNormal = normalize(texture2D(uRockNormal, rockUV).rgb * 2.0 - 1.0);
    vec3 snowNormal = normalize(texture2D(uSnowNormal, snowUV).rgb * 2.0 - 1.0);

    float grassRough = texture2D(uGrassRoughness, grassUV).r;
    float rockRough = texture2D(uRockRoughness, rockUV).r;
    float snowRough = texture2D(uSnowRoughness, snowUV).r;

    // 按 Splat Map 权重混合
    vec3 albedo = grassAlbedo * splat.r + rockAlbedo * splat.g + snowAlbedo * splat.b;
    vec3 blendedNormal = normalize(
        grassNormal * splat.r + rockNormal * splat.g + snowNormal * splat.b
    );
    float roughness = grassRough * splat.r + rockRough * splat.g + snowRough * splat.b;

    // 将细节法线与地形法线混合
    vec3 worldNormal = normalize(vNormal + blendedNormal * 0.5);

    // PBR 光照
    vec3 viewDir = normalize(uCameraPos - vWorldPos);
    vec3 color = calculatePBR(albedo, worldNormal, 0.0, roughness,
                              uSunDirection, uSunColor, viewDir);

    // 简单环境光
    color += albedo * 0.03;

    gl_FragColor = vec4(color, 1.0);
}
```

**要点**：
- Splat Map 的 RGB 通道分别控制草地、岩石、雪地的权重，通过美术工具生成
- 法线通过高度图差分计算，避免额外的法线贴图开销
- 多尺度 UV（20x、15x、25x）防止大面积地形的纹理拉伸
- 细节法线与地形法线混合时需要 normalize，否则会导致法线长度不为 1

### 练习三

**思路**：水面渲染分三步：1）用反射相机渲染镜像场景到反射纹理；2）用折射相机渲染正常场景到折射纹理；3）在水面着色器中用 Fresnel 系数混合反射和折射，并叠加 Gerstner 波模拟真实波浪。Gerstner 波相比正弦波能产生尖锐的波峰和圆滑的波谷，更接近真实水面。

**答案**：

```glsl
// ==================== Gerstner 波顶点着色器 ====================
uniform float uTime;
uniform vec4 uWaveParams0; // dirX, dirZ, amplitude, wavelength
uniform vec4 uWaveParams1;
uniform vec4 uWaveParams2;

varying vec3 vWorldPos;
varying vec3 vNormal;
varying vec2 vUV;

vec3 gerstnerWave(vec4 params, vec3 pos, inout vec3 tangent, inout vec3 binormal) {
    vec2 dir = normalize(params.xy);
    float amplitude = params.z;
    float wavelength = params.w;
    float k = 2.0 * 3.14159 / wavelength;
    float c = sqrt(9.8 / k);
    float f = k * (dot(dir, pos.xz) - c * uTime);
    float a = amplitude;
    float steepness = 0.5; // 陡峭度

    vec3 displacement;
    displacement.x = steepness * a * dir.x * cos(f);
    displacement.y = a * sin(f);
    displacement.z = steepness * a * dir.y * cos(f);

    // 累加切线和副切线
    tangent += vec3(
        -steepness * dir.x * dir.x * a * k * sin(f),
        dir.x * a * k * cos(f),
        -steepness * dir.x * dir.y * a * k * sin(f)
    );
    binormal += vec3(
        -steepness * dir.x * dir.y * a * k * sin(f),
        dir.y * a * k * cos(f),
        -steepness * dir.y * dir.y * a * k * sin(f)
    );

    return displacement;
}

void main() {
    vUV = uv;
    vec3 pos = position;
    vec3 tangent = vec3(1.0, 0.0, 0.0);
    vec3 binormal = vec3(0.0, 0.0, 1.0);

    // 叠加三组 Gerstner 波
    pos += gerstnerWave(uWaveParams0, position, tangent, binormal);
    pos += gerstnerWave(uWaveParams1, position, tangent, binormal);
    pos += gerstnerWave(uWaveParams2, position, tangent, binormal);

    // 法线由切线和副切线叉积得到
    vNormal = normalize(cross(binormal, tangent));
    vWorldPos = (modelMatrix * vec4(pos, 1.0)).xyz;

    gl_Position = projectionMatrix * viewMatrix * vec4(vWorldPos, 1.0);
}
```

```glsl
// ==================== 水面片段着色器 ====================
uniform sampler2D uReflectionTex;
uniform sampler2D uRefractionTex;
uniform sampler2D uDudvMap;
uniform sampler2D uNormalMap;
uniform float uTime;
uniform vec3 uCameraPos;
uniform vec3 uSunDirection;
uniform vec3 uSunColor;

varying vec3 vWorldPos;
varying vec3 vNormal;
varying vec2 vUV;

void main() {
    // DuDv 扰动
    vec2 distortedUV = vUV + texture2D(uDudvMap, vUV * 8.0 + uTime * 0.02).rg * 0.03;
    vec2 dudv = (texture2D(uDudvMap, distortedUV).rg * 2.0 - 1.0) * 0.02;

    // 法线贴图
    vec3 normalMap = texture2D(uNormalMap, distortedUV * 4.0).rgb * 2.0 - 1.0;
    vec3 normal = normalize(vNormal + normalMap * 0.3);

    // 反射和折射 UV
    vec2 reflectUV = vec2(vUV.x, 1.0 - vUV.y) + dudv;
    vec2 refractUV = vUV + dudv;
    vec3 reflectColor = texture2D(uReflectionTex, reflectUV).rgb;
    vec3 refractColor = texture2D(uRefractionTex, refractUV).rgb;

    // Fresnel（视角越掠射反射越强）
    vec3 viewDir = normalize(uCameraPos - vWorldPos);
    float fresnel = pow(1.0 - max(dot(viewDir, normal), 0.0), 3.0);
    fresnel = clamp(fresnel, 0.1, 0.95);

    // 混合反射和折射
    vec3 color = mix(refractColor, reflectColor, fresnel);

    // 镜面高光（太阳反射）
    vec3 lightDir = normalize(-uSunDirection);
    vec3 halfDir = normalize(viewDir + lightDir);
    float specular = pow(max(dot(normal, halfDir), 0.0), 256.0);
    color += uSunColor * specular * 1.5;

    // 水色调
    color = mix(color, vec3(0.0, 0.15, 0.2), 0.15);

    // 深度雾化（远处水面融入环境）
```

```typescript
interface ParticleConfig {

class GPUParticleSystem {
  private positionBuffer: THREE.BufferAttribute;
  private velocityBuffer: THREE.BufferAttribute;
  private lifeBuffer: THREE.BufferAttribute;
  private geometry: THREE.BufferGeometry;
  private material: THREE.ShaderMaterial;
  private points: THREE.Points;
  private config: ParticleConfig;

  constructor(config: ParticleConfig) {
    this.config = config;
    const n = config.count;

    // 初始化粒子数据
    const positions = new Float32Array(n * 3);
    const velocities = new Float32Array(n * 3);
    const lifes = new Float32Array(n); // x=currentLife, y=maxLife, z=size

    for (let i = 0; i < n; i++) {
      this.resetParticle(i, positions, velocities, lifes);
    }

    this.geometry = new THREE.BufferGeometry();
    this.positionBuffer = new THREE.BufferAttribute(positions, 3);
    this.velocityBuffer = new THREE.BufferAttribute(velocities, 3);
    this.lifeBuffer = new THREE.BufferAttribute(lifes, 3);

    this.geometry.setAttribute('aPosition', this.positionBuffer);
    this.geometry.setAttribute('aVelocity', this.velocityBuffer);
    this.geometry.setAttribute('aLife', this.lifeBuffer);

    this.material = new THREE.ShaderMaterial({
      uniforms: {
        uTime: { value: 0 },
        uDelta: { value: 0 },
        uGravity: { value: new THREE.Vector3(0, 2, 0) },
        uWind: { value: new THREE.Vector3(0, 0, 0) },
        uColorStart: { value: new THREE.Color(1.0, 0.6, 0.1) },
        uColorEnd: { value: new THREE.Color(1.0, 0.2, 0.0) },
        uTexture: { value: this.createParticleTexture() },
      },
      vertexShader: particleVertexShader,
      fragmentShader: particleFragmentShader,
      transparent: true,
      depthWrite: false,
      blending: config.blendMode,
    });

    this.points = new THREE.Points(this.geometry, this.material);
    this.points.frustumCulled = false;
  }

  private resetParticle(
    i: number, pos: Float32Array, vel: Float32Array, life: Float32Array
  ): void {
    const cfg = this.config;
    const ep = cfg.emitterPosition;

    // 位置：发射器附近随机偏移
    pos[i * 3] = ep.x + (Math.random() - 0.5) * 1.0;
    pos[i * 3 + 1] = ep.y + Math.random() * 0.5;
    pos[i * 3 + 2] = ep.z + (Math.random() - 0.5) * 1.0;

    // 速度：在配置范围内随机
    vel[i * 3] = cfg.velocityRange[0].x + Math.random() * (cfg.velocityRange[1].x - cfg.velocityRange[0].x);
    vel[i * 3 + 1] = cfg.velocityRange[0].y + Math.random() * (cfg.velocityRange[1].y - cfg.velocityRange[0].y);
    vel[i * 3 + 2] = cfg.velocityRange[0].z + Math.random() * (cfg.velocityRange[1].z - cfg.velocityRange[0].z);

    // 生命：currentLife=random, maxLife, size
    const maxLife = cfg.lifetime * (0.5 + Math.random() * 0.5);
    life[i * 3] = Math.random() * maxLife;
    life[i * 3 + 1] = maxLife;
    life[i * 3 + 2] = cfg.sizeRange[0] + Math.random() * (cfg.sizeRange[1] - cfg.sizeRange[0]);
  }

  update(delta: number, elapsed: number, wind: THREE.Vector3): void {
    this.material.uniforms.uDelta.value = delta;
    this.material.uniforms.uTime.value = elapsed;
    this.material.uniforms.uWind.value.copy(wind);
  }

  getObject(): THREE.Points {
    return this.points;
  }

  private createParticleTexture(): THREE.Texture {
    const canvas = document.createElement('canvas');
    canvas.width = 64;
    canvas.height = 64;
    const ctx = canvas.getContext('2d')!;
    const gradient = ctx.createRadialGradient(32, 32, 0, 32, 32, 32);
    gradient.addColorStop(0, 'rgba(255,255,255,1)');
    gradient.addColorStop(0.5, 'rgba(255,255,255,0.5)');
    gradient.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = gradient;
    ctx.fillRect(0, 0, 64, 64);
    const tex = new THREE.CanvasTexture(canvas);
    return tex;
  }
}
```

```glsl
// ==================== 粒子顶点着色器 ====================
attribute vec3 aPosition;
attribute vec3 aVelocity;
attribute vec3 aLife; // currentLife, maxLife, size

uniform float uTime;
uniform float uDelta;
uniform vec3 uGravity;
uniform vec3 uWind;

varying float vLifeRatio;
varying float vSize;

void main() {
    // 在 GPU 上更新粒子位置
    float currentLife = aLife.x + uDelta;
    float maxLife = aLife.y;

    // 超过生命周期则重置（通过取模循环）
    if (currentLife > maxLife) {
        currentLife = mod(currentLife, maxLife);
    }

    vLifeRatio = currentLife / maxLife;
    vSize = aLife.z;

    // 物理更新
    vec3 pos = aPosition + aVelocity * currentLife
             + uGravity * currentLife * currentLife * 0.5
             + uWind * currentLife;

    vec4 mvPosition = modelViewMatrix * vec4(pos, 1.0);

    // 越大的粒子越近时更大
    gl_PointSize = vSize * (200.0 / -mvPosition.z);
    // 用生命比缩放：出生时从小变大，死亡时缩小
    gl_PointSize *= smoothstep(0.0, 0.1, vLifeRatio) * smoothstep(1.0, 0.7, vLifeRatio);

    gl_Position = projectionMatrix * mvPosition;
}
```

```glsl
uniform sampler2D uTexture;

varying float vSize;

    if (length(center) > 0.5) discard;

    vec3 color = mix(uColorStart, uColorEnd, vLifeRatio);
    // 快消亡时变暗
    color *= 1.0 - vLifeRatio * 0.5;

    // 透明度：出生渐入，消亡渐出
    float alpha = smoothstep(0.0, 0.1, vLifeRatio) * smoothstep(1.0, 0.8, vLifeRatio);
    alpha *= 0.8;

    // 采样软粒子纹理
    float texAlpha = texture2D(uTexture, gl_PointCoord).a;

    gl_FragColor = vec4(color, alpha * texAlpha);
```

### 练习五

**思路**：后处理管线按顺序执行：先用亮度提取 + 高斯模糊生成 Bloom 效果，再做 ACES 色调映射将 HDR 压缩到 LDR，最后用色彩校正微调饱和度和对比度。关键是 Bloom 需要两次下采样+模糊+上采样的乒乓缓冲，色调映射和色彩校正可以合并到一个 Pass 中减少纹理读写。

**答案**：

```typescript
// ==================== 后处理管线 ====================
class PostProcessingPipeline {
  private bloomThresholdPass: FullScreenPass;
  private blurPasses: FullScreenPass[];
  private compositePass: FullScreenPass;

  private bloomPingPong: [THREE.WebGLRenderTarget, THREE.WebGLRenderTarget];
  private bloomMips: THREE.WebGLRenderTarget[];

  private config: {
    bloomThreshold: number;
    bloomIntensity: number;
    bloomRadius: number;
    exposure: number;
    saturation: number;
    contrast: number;
  };

  constructor(width: number, height: number) {
    this.config = {
      bloomThreshold: 0.8,
      bloomIntensity: 0.5,
      bloomRadius: 0.4,
      exposure: 1.0,
      saturation: 1.1,
      contrast: 1.05,
    };

    // Bloom 乒乓缓冲
    const halfW = Math.floor(width / 2);
    const halfH = Math.floor(height / 2);
    this.bloomPingPong = [
      new THREE.WebGLRenderTarget(halfW, halfH, { type: THREE.HalfFloatType }),
      new THREE.WebGLRenderTarget(halfW, halfH, { type: THREE.HalfFloatType }),
    ];

    // 多级 Bloom 缓冲（逐步降采样）
    this.bloomMips = [];
    let mipW = halfW, mipH = halfH;
    for (let i = 0; i < 5; i++) {
      mipW = Math.max(1, Math.floor(mipW / 2));
      mipH = Math.max(1, Math.floor(mipH / 2));
      this.bloomMips.push(
        new THREE.WebGLRenderTarget(mipW, mipH, { type: THREE.HalfFloatType })
      );
    }

    this.bloomThresholdPass = new FullScreenPass(bloomThresholdShader);
    this.compositePass = new FullScreenPass(compositeShader);
  }

  render(renderer: THREE.WebGLRenderer, hdrTexture: THREE.Texture): void {
    // 1. 亮度提取
    this.bloomThresholdPass.uniforms.uTexture.value = hdrTexture;
    this.bloomThresholdPass.uniforms.uThreshold.value = this.config.bloomThreshold;
    this.bloomThresholdPass.render(renderer, this.bloomMips[0]);

    // 2. 逐级降采样模糊
    for (let i = 1; i < this.bloomMips.length; i++) {
      this.downsample(renderer, this.bloomMips[i - 1].texture, this.bloomMips[i]);
    }

    // 3. 逐级上采样混合
    for (let i = this.bloomMips.length - 2; i >= 0; i--) {
      this.upsample(renderer, this.bloomMips[i + 1].texture, this.bloomMips[i]);
    }

    // 4. 最终合成：HDR + Bloom + 色调映射 + 色彩校正
    this.compositePass.uniforms.uHDRTex.value = hdrTexture;
    this.compositePass.uniforms.uBloomTex.value = this.bloomMips[0].texture;
    this.compositePass.uniforms.uBloomIntensity.value = this.config.bloomIntensity;
    this.compositePass.uniforms.uExposure.value = this.config.exposure;
    this.compositePass.uniforms.uSaturation.value = this.config.saturation;
    this.compositePass.uniforms.uContrast.value = this.config.contrast;
    this.compositePass.render(renderer, null);
  }

  private downsample(
    renderer: THREE.WebGLRenderer, input: THREE.Texture, target: THREE.WebGLRenderTarget
  ): void {
    // 13-tap Kawase 降采样
    const texelSize = new THREE.Vector2(1.0 / target.width, 1.0 / target.height);
    this.bloomThresholdPass.uniforms.uTexture.value = input;
    this.bloomThresholdPass.uniforms.uTexelSize.value = texelSize;
    this.bloomThresholdPass.render(renderer, target);
  }

  private upsample(
    renderer: THREE.WebGLRenderer, input: THREE.Texture, target: THREE.WebGLRenderTarget
  ): void {
    this.bloomThresholdPass.uniforms.uTexture.value = input;
    this.bloomThresholdPass.render(renderer, target);
  }
}
```

```glsl
// bloomThresholdShader
const bloomThresholdShader = {
  vertex: `
    varying vec2 vUV;
    void main() {
      vUV = uv;
      gl_Position = vec4(position.xy * 2.0, 0.0, 1.0);
    }
  `,
  fragment: `
    uniform sampler2D uTexture;
    uniform float uThreshold;
    varying vec2 vUV;

    void main() {
      vec3 color = texture2D(uTexture, vUV).rgb;
      // 亮度计算（人眼对绿色最敏感）
      float brightness = dot(color, vec3(0.2126, 0.7152, 0.0722));
      // 软阈值：保留超过阈值的部分，带平滑过渡
      float soft = brightness - uThreshold;
      soft = clamp(soft, 0.0, 1.0);
      soft = soft * soft * (3.0 - 2.0 * soft); // smoothstep
      color *= soft;
      gl_FragColor = vec4(color, 1.0);
    }
  `,
  uniforms: {
    uTexture: null,
    uThreshold: 0.8,
    uTexelSize: new THREE.Vector2(1, 1),
```

```glsl
// ==================== 最终合成着色器 ====================
// compositeShader
const compositeShader = {
  vertex: `
    varying vec2 vUV;
    void main() {
      vUV = uv;
      gl_Position = vec4(position.xy * 2.0, 0.0, 1.0);
    }
  `,
  fragment: `
    uniform sampler2D uHDRTex;
    uniform sampler2D uBloomTex;
    uniform float uBloomIntensity;
    uniform float uExposure;
    uniform float uSaturation;
    uniform float uContrast;

    varying vec2 vUV;

    // ACES 色调映射
    vec3 ACESFilm(vec3 x) {
      float a = 2.51;
      float b = 0.03;
      float c = 2.43;
      float d = 0.59;
      float e = 0.14;
      return clamp((x * (a * x + b)) / (x * (c * x + d) + e), 0.0, 1.0);
    }

    void main() {
      vec3 hdr = texture2D(uHDRTex, vUV).rgb;
      vec3 bloom = texture2D(uBloomTex, vUV).rgb;

      // 叠加 Bloom
      vec3 color = hdr + bloom * uBloomIntensity;

      // 曝光
      color *= uExposure;

      // ACES 色调映射
      color = ACESFilm(color);

      // 饱和度调整
      float lum = dot(color, vec3(0.2126, 0.7152, 0.0722));
      color = mix(vec3(lum), color, uSaturation);

      // 对比度调整
      color = (color - 0.5) * uContrast + 0.5;

      // Gamma 校正
      color = pow(color, vec3(1.0 / 2.2));

      gl_FragColor = vec4(color, 1.0);
    }
  `,
  uniforms: {
    uHDRTex: null,
    uBloomTex: null,
    uBloomIntensity: 0.5,
    uExposure: 1.0,
    uSaturation: 1.1,
    uContrast: 1.05,
  },
};
```

**要点**：
- Bloom 采用多级降采样 + 逐级上采样方案（Kawase blur），比单次大核模糊性能好且效果更自然
- ACES 色调映射将 HDR 亮度压缩到 [0,1]，保留暗部细节的同时处理高光溢出
- 色彩校正用 luminance 做基准：`mix(lum, color, saturation)` 让饱和度调整不会改变亮度
- 最终必须做 Gamma 校正（`pow(color, 1/2.2)`），否则显示器上的颜色会偏暗

### 练习六

**思路**：性能监控面板需要实时采集各 Pass 的 GPU 耗时和帧率。使用 `renderer.info` 获取 Draw Call 数量和三角形数量，用时间戳差值计算帧时间。自适应质量系统根据平均帧时间自动切换高/中/低三档配置，每档调整阴影分辨率、SSAO 采样数、粒子数量、水面反射分辨率等参数。

**答案**：

```typescript
// ==================== 性能监控面板 ====================
interface RenderStats {

class PerformanceMonitor {
  private container: HTMLElement;
  private statsEl: HTMLElement;
  private frameTimeHistory: number[] = [];
  private passTimings: Map<string, number> = new Map();
  private lastTime: number = performance.now();

  constructor(parentContainer: HTMLElement) {
    this.container = document.createElement('div');
    this.container.style.cssText = `
      position: fixed; top: 10px; right: 10px; z-index: 9999;
      background: rgba(0,0,0,0.85); color: #0f0; padding: 12px 16px;
      font-family: 'Courier New', monospace; font-size: 13px;
      border-radius: 6px; min-width: 220px; line-height: 1.6;
      pointer-events: none;
    `;
    parentContainer.appendChild(this.container);
    this.statsEl = document.createElement('div');
    this.container.appendChild(this.statsEl);
  }

  beginPass(name: string): void {
    this.passTimings.set(`${name}_start`, performance.now());
  }

  endPass(name: string): void {
    const start = this.passTimings.get(`${name}_start`);
    if (start !== undefined) {
      this.passTimings.set(name, performance.now() - start);
    }
  }

  update(renderer: THREE.WebGLRenderer): void {
    const now = performance.now();
    const frameTime = now - this.lastTime;
    this.lastTime = now;

    this.frameTimeHistory.push(frameTime);
    if (this.frameTimeHistory.length > 120) {
      this.frameTimeHistory.shift();
    }

    const avgFrameTime = this.frameTimeHistory.reduce((a, b) => a + b, 0)
      / this.frameTimeHistory.length;
    const fps = 1000 / avgFrameTime;
    const info = renderer.info;

    // 渲染面板
    let html = `<b>Performance</b><br>`;
    html += `FPS: ${fps.toFixed(1)} (${avgFrameTime.toFixed(2)}ms)<br>`;
    html += `Draw Calls: ${info.render.calls}<br>`;
    html += `Triangles: ${info.render.triangles}<br>`;
    html += `Textures: ${info.memory.textures}<br>`;
    html += `Geometries: ${info.memory.geometries}<br>`;
    html += `<b>Pass Timings</b><br>`;

    for (const [name, time] of this.passTimings.entries()) {
      if (!name.endsWith('_start')) {
        const color = time > 3 ? '#f44' : time > 2 ? '#ff0' : '#0f0';
        html += `<span style="color:${color}">${name}: ${time.toFixed(2)}ms</span><br>`;
      }
    }

    this.statsEl.innerHTML = html;

    // 重置每帧计数
    renderer.info.reset();
  }

  getAverageFrameTime(): number {
    if (this.frameTimeHistory.length === 0) return 16.6;
    return this.frameTimeHistory.reduce((a, b) => a + b, 0)
      / this.frameTimeHistory.length;
  }
}

      bloomEnabled: false,
      ssaoEnabled: false,
      postProcessQuality: 'quarter',

  onQualityChange(callback: (settings: QualitySettings) => void): void {
    this.callbacks.push(callback);
  }

  update(monitor: PerformanceMonitor): void {
    // 冷却期内不切换（避免频繁抖动）
    if (this.transitionCooldown > 0) {
      this.transitionCooldown--;
      return;
    }

    const avgFrameTime = monitor.getAverageFrameTime();

    // 判断逻辑：连续 60 帧超标才降级，恢复阈值更宽松
    if (avgFrameTime > 20 && this.currentLevel === 'high') {
      this.setQuality('medium');
      this.transitionCooldown = 120; // 2 秒冷却
    } else if (avgFrameTime > 25 && this.currentLevel === 'medium') {
      this.setQuality('low');
      this.transitionCooldown = 120;
    } else if (avgFrameTime < 14 && this.currentLevel === 'low') {
      this.setQuality('medium');
      this.transitionCooldown = 180; // 3 秒冷却，避免反复切换
    } else if (avgFrameTime < 14 && this.currentLevel === 'medium') {
      this.setQuality('high');
      this.transitionCooldown = 180;
    }
  }

  private setQuality(level: QualityLevel): void {
    this.currentLevel = level;
    const settings = this.settings[level];
    console.log(`[Quality] 切换到 ${level} 模式`);
    for (const cb of this.callbacks) {
      cb(settings);
    }
  }

  getCurrentSettings(): QualitySettings {
    return this.settings[this.currentLevel];
  }

  getCurrentLevel(): QualityLevel {
    return this.currentLevel;
  }
}
```

- `renderer.info.reset()` 必须每帧调用，否则 Draw Call 数会累加
- 自适应质量需要设置冷却期（120-180 帧），防止在性能边界反复切换（抖动）

### 练习七

**思路**：优化到 60fps 需要系统化地找到瓶颈。先用性能面板记录各 Pass 耗时，确定是 CPU bound（Draw Call 过多）还是 GPU bound（着色器复杂度/带宽）。然后针对性优化：视锥剔除减少 Draw Call、LOD 降低远处面数、合并纹理减少绑定、降采样阴影/SSAO 减少计算量、合并后处理 Pass 减少带宽。

```typescript
// ==================== 优化分析与执行 ====================

// 1. 性能分析工具
class Profiler {
  private frameTimings: Map<string, number[]> = new Map();
  private sampleCount: number = 120; // 采样 2 秒

  recordPass(name: string, timeMs: number): void {
    if (!this.frameTimings.has(name)) {
      this.frameTimings.set(name, []);
    }
    const arr = this.frameTimings.get(name)!;
    arr.push(timeMs);
    if (arr.length > this.sampleCount) arr.shift();
  }

  getReport(): string {
    const total = this.getAverageTotal();
    let report = `\n===== 性能分析报告 (平均 ${total.toFixed(2)}ms / ${(1000/total).toFixed(1)}fps) =====\n`;
    report += `${'Pass'.padEnd(20)} ${'平均(ms)'.padEnd(10)} ${'占比'.padEnd(8)} ${'瓶颈等级'}\n`;
    report += '-'.repeat(55) + '\n';

    const entries = Array.from(this.frameTimings.entries())
      .map(([name, times]) => ({
        name,
        avg: times.reduce((a, b) => a + b, 0) / times.length,
      }))
      .sort((a, b) => b.avg - a.avg);

    for (const { name, avg } of entries) {
      const pct = (avg / total * 100).toFixed(1);
      const level = avg > 3 ? '🔴 严重' : avg > 2 ? '🟡 需优化' : '🟢 正常';
      report += `${name.padEnd(20)} ${avg.toFixed(2).padEnd(10)} ${(pct + '%').padEnd(8)} ${level}\n`;
    }
    return report;
  }

  private getAverageTotal(): number {
    let total = 0;
    for (const times of this.frameTimings.values()) {
      if (times.length > 0) {
        total += times.reduce((a, b) => a + b, 0) / times.length;
      }
    }
    return total;
  }

  getBottleneck(): { name: string; avg: number } | null {
    let worst = { name: '', avg: 0 };
    for (const [name, times] of this.frameTimings.entries()) {
      const avg = times.reduce((a, b) => a + b, 0) / times.length;
      if (avg > worst.avg) worst = { name, avg };
    }
    return worst.name ? worst : null;
  }
}

// 2. 视锥剔除优化
class FrustumCuller {
  private frustum = new THREE.Frustum();
  private projScreenMatrix = new THREE.Matrix4();

  update(camera: THREE.PerspectiveCamera): void {
    this.projScreenMatrix.multiplyMatrices(
      camera.projectionMatrix, camera.matrixWorldInverse
    );
    this.frustum.setFromProjectionMatrix(this.projScreenMatrix);
  }

  cullObjects(objects: THREE.Object3D[]): THREE.Object3D[] {
    return objects.filter((obj) => {
      if (!obj.geometry) return true;
      // 使用包围球测试
      if (!obj.geometry.boundingSphere) obj.geometry.computeBoundingSphere();
      const sphere = obj.geometry.boundingSphere!.clone();
      sphere.applyMatrix4(obj.matrixWorld);
      return this.frustum.intersectsSphere(sphere);
    });
  }
}

// 3. Draw Call 合并优化
class BatchOptimizer {
  // 合并相同材质的静态网格
  static mergeStaticMeshes(meshes: THREE.Mesh[]): THREE.Mesh {
    const geometries: THREE.BufferGeometry[] = [];
    const baseMaterial = meshes[0].material as THREE.Material;

    for (const mesh of meshes) {
      const geo = mesh.geometry.clone();
      geo.applyMatrix4(mesh.matrixWorld);
      geometries.push(geo);
    }

    const merged = THREE.BufferGeometryUtils.mergeGeometries(geometries);
    return new THREE.Mesh(merged, baseMaterial);
  }

  // 实例化渲染相同模型
  static createInstancedMesh(
    template: THREE.Mesh, positions: THREE.Vector3[], count: number
  ): THREE.InstancedMesh {
    const instanced = new THREE.InstancedMesh(
      template.geometry, template.material, count
    );
    const matrix = new THREE.Matrix4();
    for (let i = 0; i < count; i++) {
      matrix.setPosition(positions[i]);
      instanced.setMatrixAt(i, matrix);
    }
    instanced.instanceMatrix.needsUpdate = true;
    return instanced;
  }
}

function optimizeTo60fps(
  renderer: RealisticSceneRenderer,
  profiler: Profiler
): void {
  // 第一步：记录基线性能
  console.log('=== 优化前基线 ===');
  console.log(profiler.getReport());

  // 第二步：根据瓶颈执行针对性优化
  const bottleneck = profiler.getBottleneck();
  if (!bottleneck) return;

  console.log(`最大瓶颈: ${bottleneck.name} (${bottleneck.avg.toFixed(2)}ms)`);

  const optimizations: Record<string, () => void> = {
    // 阴影过慢 → 降低级联分辨率 + 缩小渲染距离
    'Shadow': () => {
      renderer.shadowSystem.setResolutions([1024, 512, 256]);
      renderer.shadowSystem.setMaxDistance(200);
      console.log('  → 阴影分辨率降至 [1024, 512, 256]');
    },

    // G-Buffer 过慢 → 合并静态网格 + 视锥剔除 + 纹理压缩
    'G-Buffer': () => {
      renderer.enableFrustumCulling(true);
      renderer.mergeStaticGeometry();
      console.log('  → 启用视锥剔除 + 合并静态网格');
    },

    // SSAO 过慢 → 减少采样数 + 降分辨率
    'SSAO': () => {
      renderer.ssaoSystem.setSamples(16);
      renderer.ssaoSystem.setResolution(0.5);
      console.log('  → SSAO 采样降至 16，分辨率降至 50%');
    },

    // 水面过慢 → 降低反射分辨率 + 减少波浪组数
    'Water': () => {
      renderer.waterSystem.setReflectionRes(512);
      renderer.waterSystem.setWaveCount(2);
      console.log('  → 水面反射分辨率降至 512');
    },

    // 粒子过慢 → 减少粒子数量 + 用更简单的着色器
    'Particles': () => {
      renderer.particleSystem.setScale(0.5);
      console.log('  → 粒子数量减半');
    },

    // 后处理过慢 → 合并 Pass + 降分辨率
    'PostProcess': () => {
      renderer.postProcessing.setResolution(0.75);
      renderer.postProcessing.mergePasses(true);
      console.log('  → 后处理分辨率降至 75%，合并 Pass');
    },

    // 地形过慢 → 增大 LOD 偏移 + 减少 Chunk 数量
    'Terrain': () => {
      renderer.terrainSystem.setLODBias(1);
      renderer.terrainSystem.setMaxChunks(64);
      console.log('  → 地形 LOD 偏移 +1，最大 Chunk 数 64');
    },
  };

  if (optimizations[bottleneck.name]) {
    optimizations[bottleneck.name]();
  }

  // 第三步：通用优化
  console.log('\n=== 执行通用优化 ===');

  // 减少状态切换：按材质排序透明物体
  renderer.sortTransparentObjects(true);
  console.log('  → 透明物体按距离排序');

  // 纹理图集：合并小纹理
  renderer.enableTextureAtlas(true);
  console.log('  → 启用纹理图集');

  // 对象池：复用粒子对象
  renderer.particleSystem.enableObjectPool(true);
  console.log('  → 粒子对象池已启用');

  // 第四步：验证优化效果
  console.log('\n=== 优化后结果 ===');
  console.log(profiler.getReport());
}
```

**要点**：
- 优化前必须先做 profiling，不要盲目猜测瓶颈——Shadow 看似简单但高分辨率下可能比 G-Buffer 更耗时
- 视锥剔除是最有效的优化：看不见的东西根本不应该提交给 GPU

- SSAO 和阴影可以降分辨率渲染再上采样，人眼几乎察觉不到差异
- 优化是迭代过程：修一个瓶颈后重新 profile，可能暴露新的瓶颈
