## 后处理管线

```typescript
import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js'
import { RenderPass } from 'three/addons/postprocessing/RenderPass.js'
import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js'
import { SSAOPass } from 'three/addons/postprocessing/SSAOPass.js'
import { OutputPass } from 'three/addons/postprocessing/OutputPass.js'

const composer = new EffectComposer(renderer)

// 渲染 pass
composer.addPass(new RenderPass(scene, camera))

// SSAO
const ssaoPass = new SSAOPass(scene, camera, width, height)
ssaoPass.kernelRadius = 0.5
composer.addPass(ssaoPass)

// Bloom
const bloomPass = new UnrealBloomPass(
  new THREE.Vector2(width, height),
  0.3,  // strength
  0.4,  // radius
  0.85  // threshold
)
composer.addPass(bloomPass)

// 色调映射
composer.addPass(new OutputPass())

// 渲染循环
function animate() {
  composer.render()
  requestAnimationFrame(animate)
}
```

## 程序化地形

```typescript
function createTerrain(size: number, segments: number) {
  const geometry = new THREE.PlaneGeometry(size, size, segments, segments)
  const positions = geometry.attributes.position

  // 用噪声生成高度
  for (let i = 0; i < positions.count; i++) {
    const x = positions.getX(i)
    const z = positions.getZ(i)
    const height = noise2D(x * 0.02, z * 0.02) * 10
      + noise2D(x * 0.05, z * 0.05) * 3
      + noise2D(x * 0.1, z * 0.1) * 1
    positions.setZ(i, height)
  }

  geometry.computeVertexNormals()

  const material = new THREE.MeshStandardMaterial({
    color: 0x3a7d3a,
    roughness: 0.9,
    metalness: 0.0
  })

  return new THREE.Mesh(geometry, material)
}
```

## 你可能踩的坑

**坑一：后处理管线顺序**

Pass 的顺序很重要：RenderPass → SSAO → Bloom → OutputPass。

**坑二：PBR 材质的贴图颜色空间**

albedo 贴图用 sRGB，normal/roughness/metalness 用 Linear。

**坑三：阴影分辨率和性能**

CSM 级联阴影的每一级都需要渲染一次场景。级数越多性能越差。
