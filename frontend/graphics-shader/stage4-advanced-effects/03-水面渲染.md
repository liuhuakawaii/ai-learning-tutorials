# 水面渲染

## 场景引入

游戏中的河流、湖泊、海洋是玩家最常见的场景元素之一。一个逼真的水面需要同时处理反射、折射、波浪、泡沫等多种视觉效果。从《塞尔达传说：旷野之息》的风格化水面到《战地》系列的写实海洋，水面渲染一直是实时图形学中最具挑战性的课题。本节将从反射和折射的基本原理开始，逐步实现一个包含波浪、焦散和泡沫的完整水面 Shader。

## 学习目标

- 掌握平面反射（Planar Reflection）和 SSR 的实现方法
- 理解折射效果的 GrabPass 实现
- 学会多层法线贴图叠加实现水面扰动
- 掌握 Gerstner 波的数学原理和着色器实现
- 理解焦散（Caustics）的近似方法
- 能够实现完整的水面渲染 Shader

## 1. 水面反射

水面反射有多种实现方式，各有优劣：

```
反射实现方式对比：

  方式               原理                    优点              缺点
  ─────────────────────────────────────────────────────────────────────
  Planar Reflection  镜像相机渲染            质量最高          仅限平面
  SSR                屏幕空间光线追踪        无需额外渲染      信息缺失
  Cubemap            预烘焙环境贴图          性能好            不实时
  Hybrid             Cubemap + SSR 修正      兼顾质量和性能    实现复杂
```

Planar Reflection 原理：

```
Planar Reflection 原理：

                    相机
                     *
                    /|
                   / |
                  /  |  视线
                 /   |
                /    |
  ─────────────*─────*───────────── 水面
                \    |
                 \   |  镜像视线
                  \  |
                   \ |
                    \|
                     * 镜像相机

  步骤：
  1. 计算水面平面的镜像矩阵
  2. 用镜像矩阵变换相机
  3. 从镜像相机渲染场景到纹理
  4. 将反射纹理采样到水面着色器
```

```typescript
// Three.js Planar Reflection 实现
import * as THREE from 'three';

class PlanarReflection {
  private reflectCamera: THREE.PerspectiveCamera;
  private renderTarget: THREE.WebGLRenderTarget;
  private mirrorMatrix: THREE.Matrix4;

  constructor(
    private plane: THREE.Plane,
    private resolution: number = 1024
  ) {
    this.reflectCamera = new THREE.PerspectiveCamera();
    this.renderTarget = new THREE.WebGLRenderTarget(resolution, resolution);
    this.mirrorMatrix = new THREE.Matrix4();
  }

  update(mainCamera: THREE.Camera): void {
    // 构建镜像矩阵
    const normal = this.plane.normal;
    const d = this.plane.constant;

    this.mirrorMatrix.set(
      1 - 2 * normal.x * normal.x, -2 * normal.x * normal.y,    -2 * normal.x * normal.z,    -2 * normal.x * d,
      -2 * normal.x * normal.y,    1 - 2 * normal.y * normal.y, -2 * normal.y * normal.z,    -2 * normal.y * d,
      -2 * normal.x * normal.z,    -2 * normal.y * normal.z,    1 - 2 * normal.z * normal.z, -2 * normal.z * d,
      0, 0, 0, 1
    );

    // 镜像相机
    this.reflectCamera.copy(mainCamera as THREE.PerspectiveCamera);
    this.reflectCamera.applyMatrix4(this.mirrorMatrix);

    // 翻转裁剪面（防止水面背面被剔除）
    // 注意：需要设置 gl_ClipDistance 或使用 stencil buffer
  }

  render(renderer: THREE.WebGLRenderer, scene: THREE.Scene): void {
    renderer.setRenderTarget(this.renderTarget);
    renderer.render(scene, this.reflectCamera);
    renderer.setRenderTarget(null);
  }

  getTexture(): THREE.Texture {
    return this.renderTarget.texture;
  }
}
```

## 2. 折射效果

折射通过 GrabPass 获取水面背后的场景纹理，然后根据法线进行偏移采样。

```
折射原理：

  入射光线
      \
       \  θ₁
        \
  ───────*────────────── 水面
        / \
       /   \  折射光线
      / θ₂  \
     /       \

  Snell 定律: n₁ * sin(θ₁) = n₂ * sin(θ₂)
  水的折射率 n₂ ≈ 1.33

  实际实现中不精确计算折射方向，
  而是用法线对 GrabPass 纹理做 UV 偏移
```

```glsl
// 折射片段着色器（Three.js 中使用 ScreenGrabPass）
uniform sampler2D uRefractionTexture;  // 屏幕抓取纹理
uniform sampler2D uNormalMap;
uniform float uRefractionStrength;
uniform float uIOR;  // 折射率 1.33

varying vec2 vUV;
varying vec4 vScreenPos;

void main() {
    vec3 normal = texture2D(uNormalMap, vUV * 5.0).rgb * 2.0 - 1.0;

    // 屏幕空间 UV
    vec2 screenUV = vScreenPos.xy / vScreenPos.w * 0.5 + 0.5;

    // 法线偏移 UV（模拟折射）
    vec2 refractOffset = normal.xy * uRefractionStrength;
    vec2 refractUV = screenUV + refractOffset;

    // 边界限制
    refractUV = clamp(refractUV, 0.001, 0.999);

    // 深度相关的折射强度（水越深折射越明显）
    float sceneDepth = texture2D(uDepthTexture, screenUV).r;
    float waterDepth = texture2D(uDepthTexture, refractUV).r;
    float depthFactor = clamp((waterDepth - sceneDepth) * 50.0, 0.0, 1.0);

    vec3 refractColor = texture2D(uRefractionTexture, refractUV).rgb;

    // 水下颜色衰减（深度吸收）
    vec3 waterTint = vec3(0.1, 0.3, 0.4);
    refractColor = mix(refractColor, waterTint, depthFactor * 0.5);

    gl_FragColor = vec4(refractColor, 1.0);
}
```

## 3. 法线扰动（多层法线贴图叠加）

水面的波纹通过叠加多层法线贴图实现，每层使用不同的缩放、速度和方向。

```
多层法线叠加：

  层 1（大波浪）     层 2（中波纹）     层 3（小涟漪）     合成结果
  ┌──────────┐     ┌──────────┐     ┌──────────┐     ┌──────────┐
  │  ≈≈≈≈≈≈  │     │  ∿∿∿∿∿∿  │     │  ∽∽∽∽∽∽  │     │  复杂水面 │
  │  ≈≈≈≈≈≈  │  +  │  ∿∿∿∿∿∿  │  +  │  ∽∽∽∽∽∽  │  =   波纹   │
  │  ≈≈≈≈≈≈  │     │  ∿∿∿∿∿∿  │     │  ∽∽∽∽∽∽  │     │          │
  └──────────┘     └──────────┘     └──────────┘     └──────────┘
  UV scale: 0.5     UV scale: 1.0     UV scale: 2.0
  speed: 0.05       speed: 0.1        speed: 0.2
```

```glsl
// 多层法线贴图叠加
uniform sampler2D uNormalMap;
uniform float uTime;
uniform float uNormalStrength;

vec3 computeWaterNormal(vec2 uv) {
    // 第 1 层：大波浪
    vec2 uv1 = uv * 0.5 + uTime * vec2(0.03, 0.02);
    vec3 n1 = texture2D(uNormalMap, uv1).rgb * 2.0 - 1.0;

    // 第 2 层：中波纹（反向流动，增加复杂度）
    vec2 uv2 = uv * 1.0 + uTime * vec2(-0.05, 0.03);
    vec3 n2 = texture2D(uNormalMap, uv2).rgb * 2.0 - 1.0;

    // 第 3 层：小涟漪
    vec2 uv3 = uv * 2.5 + uTime * vec2(0.08, -0.04);
    vec3 n3 = texture2D(uNormalMap, uv3).rgb * 2.0 - 1.0;

    // 叠加（在切线空间中混合）
    vec3 normal = normalize(
        n1 * 0.5 +
        n2 * 0.35 +
        n3 * 0.15
    );

    // 控制法线强度
    normal.xy *= uNormalStrength;
    normal = normalize(normal);

    return normal;
}
```

## 4. Gerstner 波

Gerstner 波比简单的正弦波更真实，它能产生尖锐的波峰和宽阔的波谷。

```
正弦波 vs Gerstner 波：

  正弦波：
  ──────╲    ╱──────╲    ╱──────
        ╲  ╱        ╲  ╱
         ╲╱          ╲╱
  对称、圆滑，不像真实水面

  Gerstner 波：
  ──────╲  ╱────────╲  ╱──────
         ╲╱          ╲╱
  尖峰、宽谷，更接近真实水面
```

```glsl
// Gerstner 波实现
struct GerstnerWave {
    vec2 direction;   // 波浪方向（归一化）
    float amplitude;  // 振幅
    float wavelength; // 波长
    float speed;      // 速度
    float steepness;  // 陡峭度 0~1
};

// 单个 Gerstner 波计算
vec3 gerstnerWave(vec3 pos, GerstnerWave wave, float time) {
    float k = 2.0 * 3.14159 / wave.wavelength;  // 波数
    float c = wave.speed;                         // 相速度
    float a = wave.amplitude;
    float steepness = wave.steepness;

    float f = k * (dot(wave.direction, pos.xz) - c * time);
    float sinF = sin(f);
    float cosF = cos(f);

    // Gerstner 波位移
    float qa = steepness * a;
    return vec3(
        wave.direction.x * qa * cosF,   // x 位移（水平）
        a * sinF,                         // y 位移（垂直）
        wave.direction.y * qa * cosF    // z 位移（水平）
    );
}

// 多层 Gerstner 波叠加
vec3 gerstnerWaves(vec3 pos, float time) {
    GerstnerWave waves[4] = GerstnerWave[4](
        GerstnerWave(vec2(1.0, 0.0), 0.3, 8.0, 1.0, 0.5),
        GerstnerWave(vec2(0.7, 0.7), 0.2, 5.0, 1.2, 0.4),
        GerstnerWave(vec2(-0.3, 0.8), 0.15, 3.0, 0.8, 0.3),
        GerstnerWave(vec2(0.5, -0.5), 0.1, 2.0, 1.5, 0.2)
    );

    vec3 displacement = vec3(0.0);
    for (int i = 0; i < 4; i++) {
        displacement += gerstnerWave(pos, waves[i], time);
    }
    return displacement;
}

// Gerstner 波法线计算
vec3 gerstnerNormal(vec3 pos, float time) {
    GerstnerWave waves[4] = GerstnerWave[4](/* 同上 */);

    vec3 normal = vec3(0.0, 1.0, 0.0);
    for (int i = 0; i < 4; i++) {
        float k = 2.0 * 3.14159 / waves[i].wavelength;
        float f = k * (dot(waves[i].direction, pos.xz) - waves[i].speed * time);
        float steepness = waves[i].steepness;

        normal.x -= waves[i].direction.x * waves[i].amplitude * k * cos(f) * steepness;
        normal.z -= waves[i].direction.y * waves[i].amplitude * k * cos(f) * steepness;
        normal.y -= steepness * waves[i].amplitude * k * sin(f);
    }
    return normalize(normal);
}
```

## 5. 焦散（Caustics）近似

焦散是光线通过水面折射后在水底形成的光影图案。

```
焦散形成原理：

  入射光线    水面波纹    折射汇聚    水底焦散
    │ │ │    ╲  │  ╱      ╲ │ ╱      ·  █  ·
    │ │ │     ╲ │ ╱        ╲│╱       █ ██ █
    │ │ │      ╲│╱          │        ·  █  ·
  ───────────────────────────────────────── 水底
                                     光斑图案
```

```glsl
// 简化焦散效果（基于法线扰动）
uniform sampler2D uCausticsTexture;
uniform float uTime;
uniform float uCausticsScale;
uniform float uCausticsSpeed;

vec3 computeCaustics(vec3 worldPos, vec3 waterNormal) {
    // 使用两层焦散纹理，反向运动产生干涉
    vec2 causticsUV1 = worldPos.xz * uCausticsScale + uTime * uCausticsSpeed;
    vec2 causticsUV2 = worldPos.xz * uCausticsScale * 1.3 - uTime * uCausticsSpeed * 0.7;

    float caustics1 = texture2D(uCausticsTexture, causticsUV1).r;
    float caustics2 = texture2D(uCausticsTexture, causticsUV2).r;

    // 取两个纹理的最小值，产生更锐利的焦散图案
    float caustics = min(caustics1, caustics2) * 2.0;

    // 根据法线强度调整焦散强度（波浪越大焦散越明显）
    float normalStrength = length(waterNormal.xz);
    caustics *= smoothstep(0.0, 0.5, normalStrength);

    // 焦散颜色（略带蓝绿色）
    return vec3(0.8, 0.95, 1.0) * caustics;
}
```

## 6. 泡沫效果

泡沫出现在波浪的边缘和碰撞处。

```glsl
// 泡沫效果
uniform sampler2D uFoamTexture;
uniform float uFoamThreshold;   // 泡沫生成阈值
uniform float uFoamScale;

vec3 computeFoam(vec3 worldPos, vec3 normal, float depth) {
    // 基于波浪陡峭度生成泡沫
    float steepness = 1.0 - normal.y;
    float foamMask = smoothstep(uFoamThreshold, uFoamThreshold + 0.1, steepness);

    // 浅水区域泡沫更多
    foamMask += smoothstep(0.5, 0.0, depth) * 0.5;
    foamMask = clamp(foamMask, 0.0, 1.0);

    // 采样泡沫纹理
    vec2 foamUV = worldPos.xz * uFoamScale;
    foamUV += normal.xz * 0.1; // 随波浪移动
    vec4 foamColor = texture2D(uFoamTexture, foamUV + uTime * 0.05);

    // 泡沫颜色（白色，略带透明）
    return vec3(1.0) * foamMask * foamColor.r;
}
```

## 7. 完整水面 Shader

将所有效果整合到一个完整的水面着色器中：

```glsl
// 完整水面片段着色器
uniform sampler2D uReflectionTexture;
uniform sampler2D uRefractionTexture;
uniform sampler2D uNormalMap;
uniform sampler2D uDepthTexture;
uniform sampler2D uCausticsTexture;
uniform sampler2D uFoamTexture;
uniform sampler2D uDUDVMap;  // 扰动贴图

uniform float uTime;
uniform vec3 uCameraPos;
uniform vec3 uSunDirection;
uniform vec3 uSunColor;
uniform vec3 uWaterColor;
uniform float uFresnelPower;
uniform float uNormalStrength;
uniform float uRefractionStrength;

varying vec3 vWorldPos;
varying vec4 vScreenPos;
varying vec2 vUV;

void main() {
    vec2 screenUV = vScreenPos.xy / vScreenPos.w * 0.5 + 0.5;

    // 水深
    float sceneDepth = texture2D(uDepthTexture, screenUV).r;
    float waterDepth = length(vWorldPos - uCameraPos);
    float depth = abs(sceneDepth - waterDepth);
    depth = clamp(depth * 10.0, 0.0, 1.0);

    // 法线计算（多层叠加）
    vec3 normal = computeWaterNormal(vUV);

    // 扰动 UV（使用 DUDV 贴图）
    vec2 distortion = texture2D(uDUDVMap, vUV * 3.0 + uTime * 0.02).rg * 2.0 - 1.0;
    vec2 distortedUV = vUV + distortion * 0.05;

    // 反射
    vec3 viewDir = normalize(uCameraPos - vWorldPos);
    vec3 reflectDir = reflect(-viewDir, normal);
    vec2 reflectUV = screenUV + normal.xz * 0.05;
    reflectUV = clamp(reflectUV, 0.001, 0.999);
    vec3 reflection = texture2D(uReflectionTexture, reflectUV).rgb;

    // 折射
    vec2 refractUV = screenUV - normal.xz * uRefractionStrength;
    refractUV = clamp(refractUV, 0.001, 0.999);
    vec3 refraction = texture2D(uRefractionTexture, refractUV).rgb;
    refraction = mix(refraction, uWaterColor, depth * 0.3);

    // Fresnel（菲涅尔效应）
    float fresnel = pow(1.0 - max(dot(viewDir, normal), 0.0), uFresnelPower);
    fresnel = clamp(fresnel, 0.05, 0.95);

    // 高光（Blinn-Phong）
    vec3 halfDir = normalize(viewDir + uSunDirection);
    float specular = pow(max(dot(normal, halfDir), 0.0), 256.0);
    vec3 specularColor = uSunColor * specular * 2.0;

    // 焦散
    vec3 caustics = computeCaustics(vWorldPos, normal) * (1.0 - depth);

    // 泡沫
    vec3 foam = computeFoam(vWorldPos, normal, depth);

    // 最终合成
    vec3 color = mix(refraction, reflection, fresnel);
    color += specularColor;
    color += caustics * 0.3;
    color = mix(color, foam, foam.r); // 泡沫覆盖

    // 水下雾效
    color = mix(uWaterColor * 0.5, color, exp(-depth * 2.0));

    gl_FragColor = vec4(color, 0.9);
}
```

## 常见误区

1. **反射 UV 不做扰动**：直接使用屏幕 UV 采样反射纹理会导致水面看起来像镜子。必须用法线对 UV 进行扰动，才能产生水面波纹的扭曲感。

2. **法线贴图只用一层**：单层法线贴图的水面看起来是规则重复的波纹，非常假。至少叠加 2~3 层不同缩放和速度的法线贴图，才能产生自然的水面效果。

3. **忽略 Fresnel 效应**：水面的反射强度与观察角度相关——正对水面时折射为主，掠射角时反射为主。不使用 Fresnel 会导致反射和折射比例固定，缺乏真实感。

4. **焦散效果不做深度衰减**：焦散应该随水深衰减，深水区焦散消失。不做深度衰减会导致整个水面都有焦散，不符合物理规律。

---

## 工程建议

1. **反射分辨率降级**：Planar Reflection 的渲染目标可以使用屏幕分辨率的一半，配合模糊后几乎看不出差异，但性能提升 4 倍。

2. **Gerstner 波性能优化**：Gerstner 波的 sin/cos 计算较重。对于远处的水面，可以切换为简单的正弦波或直接使用法线贴图近似。

3. **深度预计算**：水面深度（水面到水底的距离）可以在主渲染 Pass 中预计算并存储在单独的 Render Target 中，避免在水面着色器中重复采样深度纹理并做反投影。

4. **移动端适配**：移动端应减少法线贴图层数（2 层即可），使用低分辨率反射（1/4），焦散和泡沫使用简化的数学函数而非纹理采样。

## 小结

本节介绍了水面渲染的核心技术：反射（Planar Reflection）、折射（GrabPass + 法线偏移）、Gerstner 波、多层法线扰动、焦散和泡沫。水面渲染的难点在于将这些独立效果正确整合，其中 Fresnel 效应控制反射/折射比例，深度控制效果衰减，法线控制视觉细节。一个高质量的水面 Shader 通常需要 200~300 行着色器代码。

## 参考答案

### 练习一

**思路**：平面反射的核心是构建一个相对于水面的镜像相机。步骤：1) 计算水面平面的镜像矩阵；2) 用镜像矩阵变换主相机的视图矩阵；3) 从镜像相机的视角渲染场景到 RenderTarget；4) 在水面着色器中采样反射纹理。

**答案**：

```typescript
import * as THREE from 'three';

class PlanarReflection {
    private reflectCamera: THREE.PerspectiveCamera;
    private renderTarget: THREE.WebGLRenderTarget;
    private mirrorMatrix: THREE.Matrix4;

    constructor(private resolution: number = 1024) {
        this.reflectCamera = new THREE.PerspectiveCamera();
        this.renderTarget = new THREE.WebGLRenderTarget(resolution, resolution, {
            format: THREE.RGBAFormat,
            type: THREE.UnsignedByteType,
        });
        this.mirrorMatrix = new THREE.Matrix4();
    }

    update(mainCamera: THREE.PerspectiveCamera, waterPlane: THREE.Plane): void {
        // 构建镜像矩阵：沿水面法线翻转
        const normal = waterPlane.normal;
        const d = waterPlane.constant;

        this.mirrorMatrix.set(
            1 - 2 * normal.x * normal.x, -2 * normal.x * normal.y,    -2 * normal.x * normal.z,    -2 * normal.x * d,
            -2 * normal.x * normal.y,    1 - 2 * normal.y * normal.y, -2 * normal.y * normal.z,    -2 * normal.y * d,
            -2 * normal.x * normal.z,    -2 * normal.y * normal.z,    1 - 2 * normal.z * normal.z, -2 * normal.z * d,
            0,                            0,                            0,                            1
        );

        // 用镜像矩阵变换相机
        this.reflectCamera.copy(mainCamera);
        this.reflectCamera.applyMatrix4(this.mirrorMatrix);

        // 翻转近裁剪面（镜像后相机朝向反转）
        this.reflectCamera.projectionMatrix.copy(mainCamera.projectionMatrix);
    }

    render(renderer: THREE.WebGLRenderer, scene: THREE.Scene): void {
        // 裁剪水面以下的物体（使用 oblique clipping plane）
        renderer.setRenderTarget(this.renderTarget);
        renderer.render(scene, this.reflectCamera);
        renderer.setRenderTarget(null);
    }

    getTexture(): THREE.Texture {
        return this.renderTarget.texture;
    }

    dispose(): void {
        this.renderTarget.dispose();
    }
}
```

**要点**：
- 镜像矩阵是沿水面法线的反射矩阵，将相机翻转到水面另一侧
- 需要设置 oblique near clipping plane 裁剪水面以下的物体，避免渲染水面下方的几何体
- 反射纹理分辨率通常为屏幕分辨率的 1/2 或 1/4，平衡质量和性能

### 练习二

**思路**：折射通过采样屏幕空间的背景纹理（GrabPass 或深度纹理重建），并使用法线贴图对 UV 进行偏移来模拟光线穿过水面时的弯曲效果。偏移量由法线扰动和折射强度控制。

**答案**：

```glsl
// 折射效果片段着色器
precision mediump float;

uniform sampler2D uRefractionTex;   // 屏幕背景纹理
uniform sampler2D uNormalMap;       // 法线贴图
uniform sampler2D uDepthTex;        // 深度纹理
uniform float uRefractionStrength;  // 折射强度
uniform float uTime;
uniform vec3 uWaterColor;           // 水体颜色
uniform float uWaterDepth;          // 水深

varying vec2 vUV;
varying vec4 vScreenPos;

void main() {
    // 采样法线贴图并扰动 UV
    vec2 uv1 = vUV * 3.0 + vec2(uTime * 0.05, uTime * 0.03);
    vec2 uv2 = vUV * 5.0 - vec2(uTime * 0.03, uTime * 0.05);
    vec3 normal1 = texture2D(uNormalMap, uv1).rgb * 2.0 - 1.0;
    vec3 normal2 = texture2D(uNormalMap, uv2).rgb * 2.0 - 1.0;
    vec3 normal = normalize(normal1 + normal2);

    // 屏幕空间 UV
    vec2 screenUV = vScreenPos.xy / vScreenPos.w * 0.5 + 0.5;

    // 折射偏移：法线的 xy 分量偏移屏幕 UV
    vec2 refractOffset = normal.xy * uRefractionStrength;
    vec2 refractUV = screenUV + refractOffset;
    refractUV = clamp(refractUV, 0.0, 1.0);

    // 采样折射纹理
    vec3 refractColor = texture2D(uRefractionTex, refractUV).rgb;

    // 计算水深（水面深度与水底深度的差值）
    float sceneDepth = texture2D(uDepthTex, refractUV).r;
    float waterDepth = linearizeDepth(sceneDepth) - linearizeDepth(gl_FragCoord.z);
    waterDepth = max(waterDepth, 0.0);

    // 水体颜色随深度衰减
    float depthFactor = 1.0 - exp(-waterDepth * uWaterDepth);
    vec3 finalColor = mix(refractColor, uWaterColor, depthFactor * 0.5);

    gl_FragColor = vec4(finalColor, 0.9);
}
```

**要点**：
- 折射偏移量不宜过大，通常 0.01~0.05 单位 UV 偏移
- 两层法线贴图以不同速度和方向滚动，增加水面扰动的细节
- 深度差控制水体颜色的混合比例：浅处透明，深处呈水体本色

### 练习三

**思路**：Gerstner 波比正弦波更真实，波峰更尖锐、波谷更平坦。每层波由方向、波长、振幅、速度四个参数定义。叠加 3 层不同参数的波浪，在顶点着色器中计算顶点位移。

**答案**：

```glsl
// Gerstner 波顶点着色器
precision highp float;

struct GerstnerWave {
    vec2 direction;   // 波浪方向（归一化）
    float wavelength; // 波长
    float amplitude;  // 振幅
    float speed;      // 速度
};

uniform GerstnerWave uWaves[3];
uniform float uTime;

attribute vec3 aPosition;
attribute vec2 aUV;

varying vec3 vWorldPos;
varying vec3 vNormal;
varying vec2 vUV;

vec3 gerstnerWave(GerstnerWave wave, vec3 pos, float time, inout vec3 tangent, inout vec3 binormal) {
    float k = 6.28318 / wave.wavelength;  // 波数 = 2π / 波长
    float c = wave.speed;                   // 相速度
    float a = wave.amplitude;
    vec2 d = normalize(wave.direction);

    float f = k * (dot(d, pos.xz) - c * time);

    // Gerstner 波位移
    vec3 displacement;
    displacement.x = d.x * a * cos(f);
    displacement.y = a * sin(f);
    displacement.z = d.y * a * cos(f);

    // 切线和副切线（用于计算法线）
    tangent += vec3(
        -d.x * d.x * a * k * sin(f),
        d.x * a * k * cos(f),
        -d.x * d.y * a * k * sin(f)
    );
    binormal += vec3(
        -d.x * d.y * a * k * sin(f),
        d.y * a * k * cos(f),
        -d.y * d.y * a * k * sin(f)
    );

    return displacement;
}

void main() {
    vec3 pos = aPosition;
    vec3 tangent = vec3(1.0, 0.0, 0.0);
    vec3 binormal = vec3(0.0, 0.0, 1.0);

    // 叠加 3 层 Gerstner 波
    for (int i = 0; i < 3; i++) {
        pos += gerstnerWave(uWaves[i], aPosition, uTime, tangent, binormal);
    }

    // 法线 = 切线 × 副切线
    vNormal = normalize(cross(binormal, tangent));
    vWorldPos = pos;
    vUV = aUV;

    gl_Position = projectionMatrix * viewMatrix * vec4(pos, 1.0);
}
```

```typescript
// 波浪参数配置
const waveUniforms = [
    { direction: new THREE.Vector2(1, 0), wavelength: 8.0, amplitude: 0.3, speed: 2.0 },
    { direction: new THREE.Vector2(0.7, 0.7), wavelength: 4.0, amplitude: 0.15, speed: 1.5 },
    { direction: new THREE.Vector2(-0.3, 0.95), wavelength: 2.0, amplitude: 0.05, speed: 3.0 },
];
```

**要点**：
- Gerstner 波的位移公式同时包含水平和垂直分量，产生波峰尖锐的效果
- 法线通过切线和副切线的叉积计算，每层波都贡献切线/副切线的变化
- 三层波叠加时，大波长低振幅的波控制整体形态，小波长高振幅的波增加细节

### 练习四

**思路**：Fresnel 效应描述了反射率随观察角度变化的物理现象：正面看水面（小角度）折射为主，掠角看水面（大角度）反射为主。使用 Schlick 近似公式计算 Fresnel 系数，然后用它混合反射和折射颜色。

**答案**：

```glsl
// Fresnel 效应片段着色器
precision mediump float;

uniform sampler2D uReflectionTex;
uniform sampler2D uRefractionTex;
uniform sampler2D uNormalMap;
uniform float uFresnelBias;     // 基础反射率，如 0.02
uniform float uFresnelScale;    // 缩放，如 1.0
uniform float uFresnelPower;    // 指数，如 5.0
uniform float uTime;

varying vec3 vWorldPos;
varying vec3 vNormal;
varying vec3 vViewDir;
varying vec4 vScreenPos;

void main() {
    // 多层法线扰动
    vec2 uv1 = vUV * 3.0 + vec2(uTime * 0.05, uTime * 0.03);
    vec2 uv2 = vUV * 6.0 - vec2(uTime * 0.02, uTime * 0.04);
    vec3 normal1 = texture2D(uNormalMap, uv1).rgb * 2.0 - 1.0;
    vec3 normal2 = texture2D(uNormalMap, uv2).rgb * 2.0 - 1.0;
    vec3 normal = normalize(normal1 + normal2);

    vec3 viewDir = normalize(vViewDir);
    float NdotV = max(dot(normal, viewDir), 0.0);

    // Schlick Fresnel 近似
    float fresnel = uFresnelBias + uFresnelScale * pow(1.0 - NdotV, uFresnelPower);

    // 采样反射和折射纹理
    vec2 screenUV = vScreenPos.xy / vScreenPos.w * 0.5 + 0.5;

    // 反射 UV：法线偏移 + 翻转
    vec2 reflectUV = screenUV + normal.xy * 0.03;
    reflectUV.x = 1.0 - reflectUV.x; // 水平翻转
    vec3 reflectionColor = texture2D(uReflectionTex, reflectUV).rgb;

    // 折射 UV：法线偏移
    vec2 refractUV = screenUV + normal.xy * 0.02;
    vec3 refractionColor = texture2D(uRefractionTex, refractUV).rgb;

    // Fresnel 混合
    vec3 finalColor = mix(refractionColor, reflectionColor, fresnel);

    gl_FragColor = vec4(finalColor, 1.0);
}
```

**要点**：
- Schlick 近似公式：`F = F0 + (1 - F0) * (1 - NdotV)^5`，其中 F0 是基础反射率（水面约 0.02）
- Fresnel 使掠角反射增强，正面折射增强，这是水面看起来"远处有倒影、近处能看透"的原因
- 可以使用 `pow(1.0 - NdotV, power)` 的变体来控制 Fresnel 过渡的锐利程度

### 练习五

**思路**：将前面练习中的反射、折射、Gerstner 波、Fresnel、泡沫和高光整合为一个完整的水面 Shader。关键在于各效果的混合权重：Fresnel 控制反射/折射比例，深度控制泡沫和水体颜色衰减，法线控制高光形状。

**答案**：

```glsl
// 完整水面片段着色器
precision highp float;

uniform sampler2D uReflectionTex;
uniform sampler2D uRefractionTex;
uniform sampler2D uNormalMap;
uniform sampler2D uDepthTex;
uniform sampler2D uFoamTex;

uniform float uTime;
uniform vec3 uWaterColor;
uniform vec3 uWaterDeepColor;
uniform float uFresnelPower;
uniform float uRefractionStrength;
uniform float uSpecularPower;
uniform vec3 uLightDir;
uniform vec3 uLightColor;
uniform float uFoamThreshold;    // 泡沫深度阈值
uniform float uWaveSteepness;    // 泡沫锐度

varying vec3 vWorldPos;
varying vec3 vNormal;
varying vec3 vViewDir;
varying vec4 vScreenPos;
varying float vWaveHeight;

void main() {
    vec3 viewDir = normalize(vViewDir);

    // 1. 多层法线扰动
    vec2 uv1 = vUV * 3.0 + vec2(uTime * 0.05, uTime * 0.03);
    vec2 uv2 = vUV * 6.0 - vec2(uTime * 0.02, uTime * 0.04);
    vec3 n1 = texture2D(uNormalMap, uv1).rgb * 2.0 - 1.0;
    vec3 n2 = texture2D(uNormalMap, uv2).rgb * 2.0 - 1.0;
    vec3 normal = normalize(n1 * 0.6 + n2 * 0.4);

    // 2. Fresnel
    float NdotV = max(dot(normal, viewDir), 0.0);
    float fresnel = 0.02 + 0.98 * pow(1.0 - NdotV, uFresnelPower);

    // 3. 反射
    vec2 screenUV = vScreenPos.xy / vScreenPos.w * 0.5 + 0.5;
    vec2 reflectUV = screenUV + normal.xy * 0.03;
    reflectUV.x = 1.0 - reflectUV.x;
    vec3 reflection = texture2D(uReflectionTex, reflectUV).rgb;

    // 4. 折射
    vec2 refractUV = screenUV + normal.xy * uRefractionStrength;
    vec3 refraction = texture2D(uRefractionTex, refractUV).rgb;

    // 5. 深度与水体颜色
    float sceneDepth = texture2D(uDepthTex, refractUV).r;
    float waterDepth = linearizeDepth(sceneDepth) - linearizeDepth(gl_FragCoord.z);
    waterDepth = max(waterDepth, 0.0);
    float depthFactor = 1.0 - exp(-waterDepth * 0.5);
    refraction = mix(refraction, uWaterDeepColor, depthFactor * 0.6);

    // 6. Fresnel 混合反射/折射
    vec3 color = mix(refraction, reflection, fresnel);

    // 7. 镜面高光（Blinn-Phong）
    vec3 halfVec = normalize(viewDir + normalize(uLightDir));
    float spec = pow(max(dot(normal, halfVec), 0.0), uSpecularPower);
    color += uLightColor * spec * 0.8;

    // 8. 泡沫（基于波浪高度）
    float foamFactor = smoothstep(uFoamThreshold, uFoamThreshold + uWaveSteepness, vWaveHeight);
    vec3 foamColor = texture2D(uFoamTex, vUV * 4.0 + uTime * 0.1).rgb;
    color = mix(color, foamColor, foamFactor * 0.7);

    gl_FragColor = vec4(color, 0.95);
}
```

**要点**：
- 完整水面 Shader 的混合顺序：法线 → Fresnel → 反射/折射混合 → 水体颜色 → 高光 → 泡沫
- 泡沫出现在波峰处，使用 Gerstner 波的 Y 位移作为阈值判断
- 深度控制两个效果：水体颜色衰减（浅处透明深处蓝）和泡沫生成（岸边泡沫）
