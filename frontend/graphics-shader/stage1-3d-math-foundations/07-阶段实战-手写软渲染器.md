# 阶段实战：手写软渲染器——自己就是 GPU

## 现象

你学了向量、矩阵、光照模型。现在不依赖 WebGL，只用 Canvas 2D 的像素操作，从零实现一个 3D 软渲染器。这个项目让你彻底理解 GPU 渲染管线——因为你自己就是 GPU。

## 渲染流水线

```
顶点数据
    │
    ▼
顶点变换（模型→世界→观察→裁剪→屏幕）
    │
    ▼
三角形光栅化（确定哪些像素被覆盖）
    │
    ▼
逐像素着色（插值属性，计算光照）
    │
    ▼
帧缓冲（屏幕）
```

```typescript
type Vec3 = [number, number, number]
type Vec4 = [number, number, number, number]
type Mat4 = Float32Array

// 矩阵运算
function mat4Multiply(a: Mat4, b: Mat4): Mat4 {
  const out = new Float32Array(16)
  for (let c = 0; c < 4; c++) for (let r = 0; r < 4; r++)
    out[c*4+r] = a[r]*b[c*4] + a[4+r]*b[c*4+1] + a[8+r]*b[c*4+2] + a[12+r]*b[c*4+3]
  return out
}

function mat4MulVec4(m: Mat4, v: Vec4): Vec4 {
  return [
    m[0]*v[0]+m[4]*v[1]+m[8]*v[2]+m[12]*v[3],
    m[1]*v[0]+m[5]*v[1]+m[9]*v[2]+m[13]*v[3],
    m[2]*v[0]+m[6]*v[1]+m[10]*v[2]+m[14]*v[3],
    m[3]*v[0]+m[7]*v[1]+m[11]*v[2]+m[15]*v[3],
  ]
}

// 透视投影矩阵
function mat4Perspective(fovY: number, aspect: number, near: number, far: number): Mat4 {
  const f = 1/Math.tan(fovY/2), nf = 1/(near-far)
  const m = new Float32Array(16)
  m[0]=f/aspect; m[5]=f; m[10]=(far+near)*nf; m[11]=-1; m[14]=2*far*near*nf
  return m
}

// LookAt 矩阵
function mat4LookAt(eye: Vec3, target: Vec3, up: Vec3): Mat4 {
  const z = vec3Normalize(vec3Sub(eye, target))
  const x = vec3Normalize(vec3Cross(up, z))
  const y = vec3Cross(z, x)
  const m = new Float32Array(16)
  m[0]=x[0]; m[4]=x[1]; m[8]=x[2];  m[12]=-vec3Dot(x,eye)
  m[1]=y[0]; m[5]=y[1]; m[9]=y[2];  m[13]=-vec3Dot(y,eye)
  m[2]=z[0]; m[6]=z[1]; m[10]=z[2]; m[14]=-vec3Dot(z,eye)
  m[3]=0; m[7]=0; m[11]=0; m[15]=1
  return m
}
```

## 三角形光栅化

```typescript
function barycentric(v0: Vec3, v1: Vec3, v2: Vec3, px: number, py: number): [number,number,number]|null {
  const ab = [v1[0]-v0[0], v1[1]-v0[1]]
  const ac = [v2[0]-v0[0], v2[1]-v0[1]]
  const ap = [px-v0[0], py-v0[1]]
  const denom = ab[0]*ac[1] - ab[1]*ac[0]
  if (Math.abs(denom) < 1e-10) return null
  const beta = (ap[0]*ac[1] - ap[1]*ac[0]) / denom
  const gamma = (ab[0]*ap[1] - ab[1]*ap[0]) / denom
  const alpha = 1 - beta - gamma
  if (alpha < 0 || beta < 0 || gamma < 0) return null
  return [alpha, beta, gamma]
}
```

## 深度缓冲

```typescript
class FrameBuffer {
  colorBuffer: Uint8ClampedArray
  depthBuffer: Float32Array

  constructor(public width: number, public height: number) {
    this.colorBuffer = new Uint8ClampedArray(width * height * 4)
    this.depthBuffer = new Float32Array(width * height)
  }

  clear(r: number, g: number, b: number) {
    for (let i = 0; i < this.width * this.height; i++) {
      this.colorBuffer[i*4]=r; this.colorBuffer[i*4+1]=g
      this.colorBuffer[i*4+2]=b; this.colorBuffer[i*4+3]=255
      this.depthBuffer[i] = Infinity
    }
  }

  setPixel(x: number, y: number, z: number, r: number, g: number, b: number): boolean {
    const idx = y * this.width + x
    if (z >= this.depthBuffer[idx]) return false
    this.depthBuffer[idx] = z
    const ci = idx * 4
    this.colorBuffer[ci] = Math.min(255, Math.max(0, Math.round(r*255)))
    this.colorBuffer[ci+1] = Math.min(255, Math.max(0, Math.round(g*255)))
    this.colorBuffer[ci+2] = Math.min(255, Math.max(0, Math.round(b*255)))
    this.colorBuffer[ci+3] = 255
    return true
  }
}
```

## Blinn-Phong 光照

```typescript
function blinnPhong(
  normal: Vec3, lightPos: Vec3, cameraPos: Vec3, worldPos: Vec3
): { diffuse: number; specular: number } {
  const L = vec3Normalize(vec3Sub(lightPos, worldPos))
  const V = vec3Normalize(vec3Sub(cameraPos, worldPos))
  const H = vec3Normalize([L[0]+V[0], L[1]+V[1], L[2]+V[2]])
  const diffuse = Math.max(0, vec3Dot(normal, L))
  const specular = Math.pow(Math.max(0, vec3Dot(normal, H)), 32)
  return { diffuse, specular }
}
```

## 你可能踩的坑

**坑一：忘记透视校正插值**

透视投影下直接线性插值会导致纹理扭曲。需要对 1/w 做插值。

**坑二：深度值范围混淆**

NDC 空间 z ∈ [-1,1]，映射到屏幕空间应归一化到 [0,1]。

**坑三：包围盒未裁剪到屏幕**

三角形超出屏幕时包围盒可能为负数，导致数组越界。

function createSphere(segments: number, rings: number) {
  const vertices = [], indices = []
  for (let ring = 0; ring <= rings; ring++) {
    const phi = (ring / rings) * Math.PI
    for (let seg = 0; seg <= segments; seg++) {
      const theta = (seg / segments) * 2 * Math.PI
      const x = Math.sin(phi)*Math.cos(theta)
      const y = Math.cos(phi)
      const z = Math.sin(phi)*Math.sin(theta)
      vertices.push({ position: [x,y,z], normal: [x,y,z], color: [(x+1)*0.5,(y+1)*0.5,(z+1)*0.5] })
    }
  }
  // ... 生成三角形索引
  return { vertices, indices }
}
