# 写作助手

## 场景引入

在任何需要输入文本的场景——写邮件、发社交媒体、编辑文档——AI 写作助手都能提供帮助。它可以检查语法错误、改写表达方式、生成内容大纲、扩展简短描述。

本课将讲解如何在浏览器扩展中实现一个写作助手，核心挑战是如何在任意输入框中无缝集成 AI 功能。

## 学习目标

- 实现跨输入框的写作助手功能
- 掌握语法检查和文本改写的 AI 集成方式
- 构建浮动工具栏 UI
- 处理文本替换和光标位置管理

## 核心架构

```
用户在任意输入框输入文本
  → 选中文本或触发快捷键
  → 浮动工具栏显示（改写/扩写/缩写/检查语法）
  → 调用 LLM API
  → 替换选中文本或插入结果
```

## 浮动工具栏

```javascript
// content/writing-assistant.js
class WritingAssistant {
  constructor() {
    this.toolbar = null;
    this.activeElement = null;
    this.init();
  }

  init() {
    document.addEventListener('mouseup', this.handleSelection.bind(this));
    document.addEventListener('keydown', this.handleKeydown.bind(this));
  }

  handleSelection(event) {
    const target = event.target;
    if (!this.isEditable(target)) return;

    const selectedText = this.getSelectedText(target);
    if (selectedText.length < 5) {
      this.hideToolbar();
      return;
    }

    this.activeElement = target;
    this.showToolbar(event.clientX, event.clientY, selectedText);
  }

  handleKeydown(event) {
    if (event.ctrlKey && event.shiftKey && event.key === 'A') {
      event.preventDefault();
      const target = document.activeElement;
      if (this.isEditable(target)) {
        this.activeElement = target;
        const rect = target.getBoundingClientRect();
        const text = this.getSelectedText(target) || this.getAllText(target);
        this.showToolbar(rect.left, rect.bottom, text);
      }
    }
  }

  isEditable(element) {
    if (element.tagName === 'TEXTAREA') return true;
    if (element.tagName === 'INPUT' && element.type === 'text') return true;
    if (element.contentEditable === 'true') return true;
    return false;
  }

  getSelectedText(element) {
    if (element.tagName === 'TEXTAREA' || element.tagName === 'INPUT') {
      return element.value.substring(element.selectionStart, element.selectionEnd);
    }
    return window.getSelection().toString();
  }

  getAllText(element) {
    if (element.tagName === 'TEXTAREA' || element.tagName === 'INPUT') {
      return element.value;
    }
    return element.innerText;
  }

  showToolbar(x, y, text) {
    if (!this.toolbar) {
      this.createToolbar();
    }

    this.currentText = text;
    this.toolbar.style.left = `${x}px`;
    this.toolbar.style.top = `${y + 10}px`;
    this.toolbar.style.display = 'block';
  }

  hideToolbar() {
    if (this.toolbar) {
      this.toolbar.style.display = 'none';
    }
  }

  createToolbar() {
    const container = document.createElement('div');
    container.id = 'writing-assistant-toolbar';

    const shadow = container.attachShadow({ mode: 'closed' });
    shadow.innerHTML = `
      <style>
        .toolbar {
          position: absolute;
          display: none;
          background: white;
          border-radius: 8px;
          box-shadow: 0 2px 12px rgba(0,0,0,0.15);
          padding: 8px;
          z-index: 2147483647;
          font-family: -apple-system, BlinkMacSystemFont, sans-serif;
        }
        .btn-group {
          display: flex;
          gap: 4px;
        }
        .btn {
          padding: 6px 12px;
          border: none;
          border-radius: 6px;
          background: #f5f5f5;
          cursor: pointer;
          font-size: 12px;
          color: #333;
          white-space: nowrap;
          transition: background 0.15s;
        }
        .btn:hover {
          background: #e8e8e8;
        }
        .btn.primary {
          background: #1a73e8;
          color: white;
        }
        .btn.primary:hover {
          background: #1557b0;
        }
        .result-panel {
          display: none;
          margin-top: 8px;
          padding-top: 8px;
          border-top: 1px solid #eee;
        }
        .result-panel.show {
          display: block;
        }
        .result-text {
          font-size: 13px;
          line-height: 1.5;
          max-height: 200px;
          overflow-y: auto;
          padding: 8px;
          background: #f9f9f9;
          border-radius: 4px;
          white-space: pre-wrap;
        }
        .result-actions {
          display: flex;
          gap: 4px;
          margin-top: 8px;
        }
        .loading {
          text-align: center;
          color: #999;
          padding: 8px;
        }
      </style>
      <div class="toolbar">
        <div class="btn-group">
          <button class="btn" data-action="rewrite">改写</button>
          <button class="btn" data-action="expand">扩写</button>
          <button class="btn" data-action="shorten">缩写</button>
          <button class="btn" data-action="grammar">语法检查</button>
          <button class="btn" data-action="formal">正式化</button>
          <button class="btn" data-action="casual">口语化</button>
        </div>
        <div class="result-panel">
          <div class="result-text"></div>
          <div class="result-actions">
            <button class="btn primary" data-action="apply">应用</button>
            <button class="btn" data-action="copy">复制</button>
            <button class="btn" data-action="dismiss">取消</button>
          </div>
        </div>
      </div>
    `;

    shadow.querySelectorAll('.btn').forEach(btn => {
      btn.addEventListener('click', (e) => {
        const action = e.target.dataset.action;
        this.handleAction(action);
      });
    });

    document.body.appendChild(container);
    this.toolbar = container;
    this.shadow = shadow;
  }

  async handleAction(action) {
    if (action === 'apply') {
      this.applyResult();
      return;
    }
    if (action === 'copy') {
      navigator.clipboard.writeText(this.currentResult);
      return;
    }
    if (action === 'dismiss') {
      this.hideToolbar();
      return;
    }

    const resultPanel = this.shadow.querySelector('.result-panel');
    const resultText = this.shadow.querySelector('.result-text');

    resultPanel.classList.add('show');
    resultText.innerHTML = '<div class="loading">处理中...</div>';

    try {
      const result = await this.processAction(action, this.currentText);
      this.currentResult = result;
      resultText.textContent = result;
    } catch (error) {
      resultText.innerHTML = `<div style="color:#e74c3c">${error.message}</div>`;
    }
  }

  async processAction(action, text) {
    const prompts = {
      rewrite: `改写以下文本，保持原意但使用不同的表达方式：\n\n${text}`,
      expand: `将以下文本扩展为更详细的描述：\n\n${text}`,
      shorten: `将以下文本精简为更简洁的表达：\n\n${text}`,
      grammar: `检查以下文本的语法错误，返回修正后的文本：\n\n${text}`,
      formal: `将以下文本改写为正式的书面语：\n\n${text}`,
      casual: `将以下文本改写为轻松的口语风格：\n\n${text}`
    };

    return new Promise((resolve, reject) => {
      chrome.runtime.sendMessage({
        type: 'LLM_GENERATE',
        prompt: prompts[action]
      }, (response) => {
        if (response.error) reject(new Error(response.error));
        else resolve(response.text);
      });
    });
  }

  applyResult() {
    if (!this.activeElement || !this.currentResult) return;

    const element = this.activeElement;

    if (element.tagName === 'TEXTAREA' || element.tagName === 'INPUT') {
      const start = element.selectionStart;
      const end = element.selectionEnd;
      const value = element.value;

      element.value = value.substring(0, start) + this.currentResult + value.substring(end);
      element.selectionStart = start;
      element.selectionEnd = start + this.currentResult.length;
      element.dispatchEvent(new Event('input', { bubbles: true }));
    } else if (element.contentEditable === 'true') {
      const selection = window.getSelection();
      if (selection.rangeCount > 0) {
        const range = selection.getRangeAt(0);
        range.deleteContents();
        range.insertNode(document.createTextNode(this.currentResult));
      }
    }

    this.hideToolbar();
  }
}

new WritingAssistant();
```

## Background LLM 服务

```javascript
// background.js
chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
  if (message.type === 'LLM_GENERATE') {
    generateText(message.prompt).then(sendResponse).catch(error => {
      sendResponse({ error: error.message });
    });
    return true;
  }
});

async function generateText(prompt) {
  const { apiKey, baseUrl, model } = await chrome.storage.local.get([
    'apiKey', 'baseUrl', 'model'
  ]);

  if (!apiKey) {
    throw new Error('请先在扩展设置中配置 API Key');
  }

  const response = await fetch(`${baseUrl || 'https://api.openai.com/v1'}/chat/completions`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`
    },
    body: JSON.stringify({
      model: model || 'gpt-4o-mini',
      messages: [
        {
          role: 'system',
          content: '你是一个专业的写作助手。只返回处理后的文本，不要添加解释或前缀。'
        },
        { role: 'user', content: prompt }
      ],
      max_tokens: 2000,
      temperature: 0.7
    })
  });

  if (!response.ok) {
    throw new Error(`API 调用失败: ${response.status}`);
  }

  const data = await response.json();
  return data.choices[0].message.content.trim();
}
```

## 语法检查高亮

```javascript
// content/grammar-checker.js
class GrammarChecker {
  constructor() {
    this.markers = [];
  }

  async checkGrammar(element) {
    const text = this.getText(element);
    if (text.length < 10) return;

    const errors = await this.getGrammarErrors(text);
    this.highlightErrors(element, errors);
  }

  async getGrammarErrors(text) {
    return new Promise((resolve, reject) => {
      chrome.runtime.sendMessage({
        type: 'GRAMMAR_CHECK',
        text
      }, (response) => {
        if (response.error) reject(new Error(response.error));
        else resolve(response.errors);
      });
    });
  }

  highlightErrors(element, errors) {
    this.clearMarkers();

    errors.forEach(error => {
      const range = this.findTextRange(element, error.text);
      if (!range) return;

      const marker = document.createElement('span');
      marker.className = 'grammar-error-marker';
      marker.style.cssText = `
        border-bottom: 2px wavy #e74c3c;
        cursor: pointer;
        position: relative;
      `;
      marker.title = `${error.message}\n建议: ${error.suggestion}`;

      try {
        range.surroundContents(marker);
        this.markers.push(marker);

        marker.addEventListener('click', () => {
          if (confirm(`修正为: "${error.suggestion}"?`)) {
            marker.textContent = error.suggestion;
            marker.style.borderBottom = '2px solid #4caf50';
          }
        });
      } catch {}
    });
  }

  getText(element) {
    if (element.tagName === 'TEXTAREA' || element.tagName === 'INPUT') {
      return element.value;
    }
    return element.innerText;
  }

  findTextRange(element, searchText) {
    const text = this.getText(element);
    const index = text.indexOf(searchText);
    if (index === -1) return null;

    if (element.tagName === 'TEXTAREA' || element.tagName === 'INPUT') {
      element.focus();
      element.setSelectionRange(index, index + searchText.length);
      return null;
    }

    const walker = document.createTreeWalker(element, NodeFilter.SHOW_TEXT);
    let currentPos = 0;

    while (walker.nextNode()) {
      const node = walker.currentNode;
      const nodeEnd = currentPos + node.textContent.length;

      if (index >= currentPos && index < nodeEnd) {
        const range = document.createRange();
        range.setStart(node, index - currentPos);
        range.setEnd(node, Math.min(index + searchText.length - currentPos, node.textContent.length));
        return range;
      }

      currentPos = nodeEnd;
    }

    return null;
  }

  clearMarkers() {
    this.markers.forEach(marker => {
      const parent = marker.parentNode;
      while (marker.firstChild) {
        parent.insertBefore(marker.firstChild, marker);
      }
      parent.removeChild(marker);
    });
    this.markers = [];
  }
}
```

## 常见误区

### 误区一：直接操作 input/textarea 的 value

修改 `input.value` 不会触发 React/Vue 等框架的状态更新。需要同时触发 `input` 事件：

```javascript
element.value = newValue;
element.dispatchEvent(new Event('input', { bubbles: true }));
```

### 误区二：忽略 contentEditable 的复杂性

`contentEditable` 元素的 DOM 结构可能很复杂（嵌套 span、br 等），直接插入文本可能破坏格式。使用 `document.execCommand` 或 Selection API 更安全。

### 误区三：在所有输入框中都显示工具栏

某些输入框（如搜索框、密码框）不适合显示写作助手。应该过滤这些场景。

## 工程建议

1. **智能触发**：只在用户选中文本时显示工具栏，不要干扰正常输入
2. **快捷键支持**：提供键盘快捷键，提高效率
3. **结果缓存**：相同文本的处理结果应该缓存
4. **上下文感知**：根据输入框类型（邮件、评论、文档）调整 AI 的语气
5. **隐私保护**：明确告知用户文本会被发送到 AI API

## 小结

写作助手扩展的核心是浮动工具栏、LLM API 集成和文本替换。通过检测用户选中的文本，提供改写、扩写、语法检查等功能，并将结果无缝应用到输入框中。

```javascript
class GmailAssistant {
  constructor() {
    this.observeGmail();
  }

  observeGmail() {
    const observer = new MutationObserver(() => {
      this.injectReplyButton();
    });

    observer.observe(document.body, {
      childList: true,
      subtree: true
    });
  }

  injectReplyButton() {
    const replyBoxes = document.querySelectorAll('[aria-label="Message Body"]');
    replyBoxes.forEach(box => {
      if (box.dataset.aiInjected) return;
      box.dataset.aiInjected = 'true';

      const toolbar = box.closest('.aO7')?.querySelector('.gU');
      if (!toolbar) return;

      const btn = document.createElement('button');
      btn.textContent = 'AI 回复';
      btn.style.cssText = `
        padding: 4px 12px;
        background: #1a73e8;
        color: white;
        border: none;
        border-radius: 4px;
        cursor: pointer;
        margin-left: 8px;
      `;

      btn.addEventListener('click', async () => {
        const emailThread = this.extractEmailThread();
        const reply = await this.generateReply(emailThread);
        box.innerHTML = reply.replace(/\n/g, '<br>');
      });

      toolbar.appendChild(btn);
    });
  }

  extractEmailThread() {
    const messages = document.querySelectorAll('.a3s');
    return [...messages].map(msg => msg.innerText).join('\n---\n');
  }

  async generateReply(thread) {
    return new Promise((resolve, reject) => {
      chrome.runtime.sendMessage({
        type: 'LLM_GENERATE',
        prompt: `根据以下邮件内容生成一个合适的回复：\n\n${thread}`
      }, (response) => {
        if (response.error) reject(new Error(response.error));
        else resolve(response.text);
      });
    });
  }
}
```

```javascript
function analyzeWriting(text) {
  const sentences = text.split(/[。！？.!?]+/).filter(s => s.trim());
  const words = text.split(/[\s,，。！？.!?]+/).filter(w => w.trim());
  const paragraphs = text.split(/\n\s*\n/).filter(p => p.trim());

  const wordFreq = {};
  words.forEach(word => {
    if (word.length > 1) {
      wordFreq[word] = (wordFreq[word] || 0) + 1;
    }
  });

  const topWords = Object.entries(wordFreq)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 10);

  const avgSentenceLength = words.length / sentences.length;
  const readability = avgSentenceLength < 15 ? '易读' :
                     avgSentenceLength < 25 ? '适中' : '较难';

  return {
    characters: text.length,
    words: words.length,
    sentences: sentences.length,
    paragraphs: paragraphs.length,
    avgSentenceLength: Math.round(avgSentenceLength),
    readability,
    topWords
  };
}
```
