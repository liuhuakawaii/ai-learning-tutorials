# 智能搜索

## 场景引入

传统搜索是关键词匹配：你输入"JavaScript 异步"，搜索引擎返回包含这些关键词的页面。但很多时候，你不知道该用什么关键词，或者你的需求用语言描述比用关键词更准确。

"我想找一个能处理并发请求、支持取消、有重试机制的 HTTP 库"——这种需求用关键词搜索很难表达清楚，但用自然语言描述就很清晰。语义搜索（Semantic Search）正是为此而生。

本课将讲解如何在浏览器扩展中实现语义搜索，包括本地向量搜索和远程 API 搜索两种方案。

## 学习目标

- 理解语义搜索的基本原理
- 实现基于向量的本地语义搜索
- 集成远程语义搜索 API
- 构建知识图谱的搜索界面

## 语义搜索基础

### 向量嵌入

语义搜索的核心是将文本转换为向量（Embedding），然后通过向量相似度来衡量语义相关性。

```
"JavaScript 异步编程" → [0.23, -0.15, 0.87, ...]  (1536 维向量)
"Promise 和 async/await" → [0.21, -0.13, 0.85, ...]  (相似向量)
"做红烧肉的步骤" → [-0.45, 0.67, -0.12, ...]  (不相似向量)
```

两个向量之间的余弦相似度（Cosine Similarity）越接近 1，语义越相似。

### 本地 vs 远程

| 方案 | 优点 | 缺点 |
|------|------|------|
| 本地向量搜索 | 无需网络、隐私安全、无 API 费用 | 模型体积大、精度较低 |
| 远程 API 搜索 | 精度高、模型更新及时 | 需要网络、有 API 费用 |

## 远程 API 方案

### 生成 Embedding

```javascript
// services/embedding.js
class EmbeddingService {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.cache = new Map();
  }

  async getEmbedding(text) {
    const cacheKey = text.substring(0, 100);
    if (this.cache.has(cacheKey)) {
      return this.cache.get(cacheKey);
    }

    const response = await fetch('https://api.openai.com/v1/embeddings', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: JSON.stringify({
        model: 'text-embedding-3-small',
        input: text.substring(0, 8000)
      })
    });

    if (!response.ok) {
      throw new Error(`Embedding API 错误: ${response.status}`);
    }

    const data = await response.json();
    const embedding = data.data[0].embedding;

    this.cache.set(cacheKey, embedding);
    return embedding;
  }

  async getBatchEmbeddings(texts) {
    const response = await fetch('https://api.openai.com/v1/embeddings', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: JSON.stringify({
        model: 'text-embedding-3-small',
        input: texts.map(t => t.substring(0, 8000))
      })
    });

    const data = await response.json();
    return data.data.map(d => d.embedding);
  }
}
```

### 向量相似度计算

```javascript
// utils/vector.js
function cosineSimilarity(a, b) {
  let dotProduct = 0;
  let normA = 0;
  let normB = 0;

  for (let i = 0; i < a.length; i++) {
    dotProduct += a[i] * b[i];
    normA += a[i] * a[i];
    normB += b[i] * b[i];
  }

  return dotProduct / (Math.sqrt(normA) * Math.sqrt(normB));
}

function findSimilar(queryEmbedding, documents, topK = 5) {
  const results = documents.map(doc => ({
    ...doc,
    similarity: cosineSimilarity(queryEmbedding, doc.embedding)
  }));

  results.sort((a, b) => b.similarity - a.similarity);
  return results.slice(0, topK);
}
```

## 构建本地知识库

### 索引页面内容

```javascript
// content/indexer.js
class PageIndexer {
  constructor() {
    this.dbName = 'semantic-search-db';
    this.storeName = 'pages';
  }

  async initDB() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this.dbName, 1);

      request.onupgradeneeded = (event) => {
        const db = event.target.result;
        if (!db.objectStoreNames.contains(this.storeName)) {
          const store = db.createObjectStore(this.storeName, { keyPath: 'url' });
          store.createIndex('timestamp', 'timestamp');
        }
      };

      request.onsuccess = (event) => resolve(event.target.result);
      request.onerror = (event) => reject(event.target.error);
    });
  }

  async indexPage(url, title, content, embedding) {
    const db = await this.initDB();

    return new Promise((resolve, reject) => {
      const transaction = db.transaction([this.storeName], 'readwrite');
      const store = transaction.objectStore(this.storeName);

      store.put({
        url,
        title,
        content: content.substring(0, 5000),
        embedding,
        timestamp: Date.now()
      });

      transaction.oncomplete = () => resolve();
      transaction.onerror = (event) => reject(event.target.error);
    });
  }

  async getAllPages() {
    const db = await this.initDB();

    return new Promise((resolve, reject) => {
      const transaction = db.transaction([this.storeName], 'readonly');
      const store = transaction.objectStore(this.storeName);
      const request = store.getAll();

      request.onsuccess = () => resolve(request.result);
      request.onerror = (event) => reject(event.target.error);
    });
  }

  async search(queryEmbedding, topK = 10) {
    const pages = await this.getAllPages();
    return findSimilar(queryEmbedding, pages, topK);
  }
}
```

### 自动索引浏览过的页面

```javascript
// background.js
class AutoIndexer {
  constructor() {
    this.embeddingService = null;
    this.pageIndexer = new PageIndexer();
    this.setupListeners();
  }

  setupListeners() {
    chrome.tabs.onUpdated.addListener(async (tabId, changeInfo, tab) => {
      if (changeInfo.status === 'complete' && tab.url) {
        await this.indexPageIfNeeded(tab);
      }
    });
  }

  async indexPageIfNeeded(tab) {
    if (tab.url.startsWith('chrome://') || tab.url.startsWith('chrome-extension://')) {
      return;
    }

    const { settings = {} } = await chrome.storage.local.get('settings');
    if (!settings.autoIndex) return;

    try {
      const response = await chrome.tabs.sendMessage(tab.id, {
        type: 'EXTRACT_CONTENT'
      });

      if (!response || !response.text || response.text.length < 100) return;

      if (!this.embeddingService) {
        const { apiKey } = await chrome.storage.local.get('apiKey');
        if (!apiKey) return;
        this.embeddingService = new EmbeddingService(apiKey);
      }

      const chunks = this.splitIntoChunks(response.text, 500);
      const embeddings = await this.embeddingService.getBatchEmbeddings(chunks);

      for (let i = 0; i < chunks.length; i++) {
        await this.pageIndexer.indexPage(
          `${tab.url}#chunk_${i}`,
          tab.title,
          chunks[i],
          embeddings[i]
        );
      }

      console.log(`已索引: ${tab.title} (${chunks.length} 个分块)`);
    } catch (error) {
      console.error('索引失败:', error);
    }
  }

  splitIntoChunks(text, chunkSize) {
    const sentences = text.match(/[^。！？.!?]+[。！？.!?]+/g) || [text];
    const chunks = [];
    let currentChunk = '';

    for (const sentence of sentences) {
      if (currentChunk.length + sentence.length > chunkSize) {
        if (currentChunk) chunks.push(currentChunk);
        currentChunk = sentence;
      } else {
        currentChunk += sentence;
      }
    }

    if (currentChunk) chunks.push(currentChunk);
    return chunks;
  }
}
```

## 搜索 UI

```javascript
// popup/search.js
class SemanticSearchUI {
  constructor() {
    this.pageIndexer = new PageIndexer();
    this.embeddingService = null;
    this.init();
  }

  async init() {
    const { apiKey } = await chrome.storage.local.get('apiKey');
    if (apiKey) {
      this.embeddingService = new EmbeddingService(apiKey);
    }

    document.getElementById('search-input').addEventListener('input',
      this.debounce(this.handleSearch.bind(this), 500)
    );
  }

  debounce(fn, delay) {
    let timer;
    return function(...args) {
      clearTimeout(timer);
      timer = setTimeout(() => fn.apply(this, args), delay);
    };
  }

  async handleSearch(event) {
    const query = event.target.value.trim();
    if (query.length < 3) {
      this.clearResults();
      return;
    }

    if (!this.embeddingService) {
      this.showError('请先配置 API Key');
      return;
    }

    this.showLoading();

    try {
      const queryEmbedding = await this.embeddingService.getEmbedding(query);
      const results = await this.pageIndexer.search(queryEmbedding, 10);
      this.renderResults(results, query);
    } catch (error) {
      this.showError(error.message);
    }
  }

  renderResults(results, query) {
    const container = document.getElementById('search-results');

    if (results.length === 0) {
      container.innerHTML = '<div class="empty">没有找到相关内容</div>';
      return;
    }

    container.innerHTML = results.map(result => `
      <div class="result-item">
        <div class="result-title">${result.title}</div>
        <div class="result-url">${new URL(result.url).hostname}</div>
        <div class="result-snippet">${this.highlightQuery(result.content, query)}</div>
        <div class="result-score">相关度: ${(result.similarity * 100).toFixed(1)}%</div>
      </div>
    `).join('');

    container.querySelectorAll('.result-item').forEach((item, index) => {
      item.addEventListener('click', () => {
        chrome.tabs.create({ url: results[index].url.split('#')[0] });
      });
    });
  }

  highlightQuery(text, query) {
    const snippet = text.substring(0, 200);
    return snippet;
  }

  showLoading() {
    document.getElementById('search-results').innerHTML =
      '<div class="loading">搜索中...</div>';
  }

  showError(message) {
    document.getElementById('search-results').innerHTML =
      `<div class="error">${message}</div>`;
  }

  clearResults() {
    document.getElementById('search-results').innerHTML = '';
  }
}

new SemanticSearchUI();
```

## 知识图谱

将索引的页面组织成知识图谱，展示页面之间的语义关系。

```javascript
// popup/knowledge-graph.js
class KnowledgeGraph {
  constructor() {
    this.pageIndexer = new PageIndexer();
    this.canvas = document.getElementById('graph-canvas');
    this.ctx = this.canvas.getContext('2d');
  }

  async render() {
    const pages = await this.pageIndexer.getAllPages();
    if (pages.length < 2) return;

    const similarities = this.calculateSimilarityMatrix(pages);
    const positions = this.layoutNodes(pages, similarities);

    this.drawGraph(positions, similarities, pages);
  }

  calculateSimilarityMatrix(pages) {
    const matrix = [];
    for (let i = 0; i < pages.length; i++) {
      matrix[i] = [];
      for (let j = 0; j < pages.length; j++) {
        matrix[i][j] = i === j ? 1 :
          cosineSimilarity(pages[i].embedding, pages[j].embedding);
      }
    }
    return matrix;
  }

  layoutNodes(pages, similarities) {
    const width = this.canvas.width;
    const height = this.canvas.height;
    const centerX = width / 2;
    const centerY = height / 2;
    const radius = Math.min(width, height) * 0.35;

    return pages.map((page, i) => {
      const angle = (2 * Math.PI * i) / pages.length;
      return {
        x: centerX + radius * Math.cos(angle),
        y: centerY + radius * Math.sin(angle),
        page
      };
    });
  }

  drawGraph(positions, similarities, pages) {
    const ctx = this.ctx;
    ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    for (let i = 0; i < positions.length; i++) {
      for (let j = i + 1; j < positions.length; j++) {
        if (similarities[i][j] > 0.7) {
          ctx.beginPath();
          ctx.moveTo(positions[i].x, positions[i].y);
          ctx.lineTo(positions[j].x, positions[j].y);
          ctx.strokeStyle = `rgba(26, 115, 232, ${similarities[i][j]})`;
          ctx.lineWidth = similarities[i][j] * 3;
          ctx.stroke();
        }
      }
    }

    positions.forEach((pos, i) => {
      ctx.beginPath();
      ctx.arc(pos.x, pos.y, 20, 0, 2 * Math.PI);
      ctx.fillStyle = '#1a73e8';
      ctx.fill();

      ctx.fillStyle = 'white';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.font = '10px sans-serif';
      ctx.fillText(String(i + 1), pos.x, pos.y);

      ctx.fillStyle = '#333';
      ctx.font = '11px sans-serif';
      const title = pos.page.title.substring(0, 15);
      ctx.fillText(title, pos.x, pos.y + 30);
    });
  }
}
```

## 常见误区

### 误区一：语义搜索不需要关键词

语义搜索和关键词搜索各有优势。最佳方案是混合搜索（Hybrid Search）：同时使用关键词匹配和语义相似度。

### 误区二：Embedding 模型越大越好

对于浏览器扩展场景，`text-embedding-3-small`（1536 维）已经足够。更大的模型会增加 API 费用和存储开销。

### 误区三：所有内容都需要索引

只索引有意义的内容（文章正文、文档），不要索引导航栏、广告等噪音内容。

## 工程建议

1. **分块策略**：将长文本分成 200-500 字的块，每块有重叠，避免信息断裂
2. **增量索引**：只索引新页面，不要重复索引已索引的页面
3. **缓存 Embedding**：Embedding 计算有成本，结果应该缓存
4. **隐私考虑**：明确告知用户哪些页面会被索引
5. **存储限制**：IndexedDB 有存储限制，定期清理旧数据

## 小结

智能搜索扩展通过向量嵌入将文本转换为语义向量，然后通过向量相似度实现语义级别的搜索。结合本地索引和远程 API，可以构建一个强大的个人知识搜索引擎。

```javascript
async function hybridSearch(query, topK = 10) {
  const pages = await pageIndexer.getAllPages();

  const keywordResults = pages
    .filter(page =>
      page.title.toLowerCase().includes(query.toLowerCase()) ||
      page.content.toLowerCase().includes(query.toLowerCase())
    )
    .map(page => ({
      ...page,
      keywordScore: (
        (page.title.toLowerCase().includes(query.toLowerCase()) ? 2 : 0) +
        (page.content.toLowerCase().includes(query.toLowerCase()) ? 1 : 0)
      )
    }));

  const queryEmbedding = await embeddingService.getEmbedding(query);
  const semanticResults = findSimilar(queryEmbedding, pages, topK * 2);

  const merged = new Map();

  keywordResults.forEach(r => {
    merged.set(r.url, {
      ...r,
      keywordScore: r.keywordScore,
      semanticScore: 0
    });
  });

  semanticResults.forEach(r => {
    if (merged.has(r.url)) {
      merged.get(r.url).semanticScore = r.similarity;
    } else {
      merged.set(r.url, {
        ...r,
        keywordScore: 0,
        semanticScore: r.similarity
      });
    }
  });

  const results = [...merged.values()].map(r => ({
    ...r,
    combinedScore: r.keywordScore * 0.4 + r.semanticScore * 0.6
  }));

  results.sort((a, b) => b.combinedScore - a.combinedScore);
  return results.slice(0, topK);
}
```

```javascript
async function getSearchSuggestions(partialQuery) {
  const { searchHistory = [] } = await chrome.storage.local.get('searchHistory');

  const historyMatches = searchHistory
    .filter(h => h.query.toLowerCase().startsWith(partialQuery.toLowerCase()))
    .sort((a, b) => b.count - a.count)
    .slice(0, 3)
    .map(h => h.query);

  if (partialQuery.length >= 2 && embeddingService) {
    const queryEmbedding = await embeddingService.getEmbedding(partialQuery);
    const pages = await pageIndexer.search(queryEmbedding, 3);

    const contentSuggestions = pages
      .filter(p => p.similarity > 0.7)
      .map(p => p.title.substring(0, 30));

    return [...new Set([...historyMatches, ...contentSuggestions])].slice(0, 5);
  }

  return historyMatches.slice(0, 5);
}
```

```javascript
async function exportKnowledgeBase(format = 'json') {
  const pages = await pageIndexer.getAllPages();

  if (format === 'json') {
    const exportData = {
      version: '1.0',
      exportDate: new Date().toISOString(),
      pages: pages.map(p => ({
        url: p.url,
        title: p.title,
        content: p.content,
        timestamp: p.timestamp
      }))
    };

    downloadJSON(exportData, 'knowledge-base.json');
  } else if (format === 'markdown') {
    let md = '# 我的知识库\n\n';
    md += `导出时间: ${new Date().toLocaleString()}\n\n`;

    const grouped = {};
    pages.forEach(page => {
      const domain = new URL(page.url).hostname;
      if (!grouped[domain]) grouped[domain] = [];
      grouped[domain].push(page);
    });

    for (const [domain, domainPages] of Object.entries(grouped)) {
      md += `## ${domain}\n\n`;
      domainPages.forEach(page => {
        md += `### ${page.title}\n`;
        md += `> ${page.url}\n\n`;
        md += `${page.content.substring(0, 500)}\n\n`;
      });
    }

    downloadText(md, 'knowledge-base.md');
  }
}

function downloadJSON(data, filename) {
  const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
}

function downloadText(text, filename) {
  const blob = new Blob([text], { type: 'text/markdown' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
}
```
