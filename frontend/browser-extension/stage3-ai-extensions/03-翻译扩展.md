# 翻译扩展

## 场景引入

翻译是浏览器扩展最经典的功能之一。从早期的 Google Translate 扩展到现在的沉浸式翻译，翻译扩展已经从简单的"整页替换"进化到了"双语对照"、"划词翻译"、"PDF 翻译"等多种形态。

本课将讲解如何构建一个功能完整的翻译扩展，涵盖划词翻译、全页翻译和双语对照三种模式。

## 学习目标

- 实现划词翻译功能
- 构建全页翻译和双语对照模式
- 掌握翻译 API 的集成和缓存策略
- 处理翻译中的性能和体验优化

```javascript
// content.js
class SelectionTranslator {
  constructor() {
    this.popup = null;
    this.isVisible = false;
    this.debounceTimer = null;
    this.init();
  }

  init() {
    document.addEventListener('mouseup', this.handleMouseUp.bind(this));
    document.addEventListener('mousedown', this.handleMouseDown.bind(this));
    document.addEventListener('keydown', this.handleKeyDown.bind(this));
  }

  handleMouseUp(event) {
    if (this.popup && this.popup.contains(event.target)) return;

    const selectedText = window.getSelection().toString().trim();
    if (selectedText.length < 2 || selectedText.length > 500) {
      return;
    }

    clearTimeout(this.debounceTimer);
    this.debounceTimer = setTimeout(() => {
      this.showTranslation(event.clientX, event.clientY, selectedText);
    }, 300);
  }

  handleMouseDown(event) {
    if (this.popup && !this.popup.contains(event.target)) {
      this.hidePopup();
    }
  }

  handleKeyDown(event) {
    if (event.key === 'Escape') {
      this.hidePopup();
    }
  }

  async showTranslation(x, y, text) {
    if (!this.popup) {
      this.createPopup();
    }

    this.positionPopup(x, y);
    this.showLoading();

    try {
      const result = await this.translate(text);
      this.showResult(text, result);
    } catch (error) {
      this.showError(error.message);
    }
  }

  async translate(text) {
    const { targetLang = 'zh-CN' } = await chrome.storage.local.get('targetLang');

    return new Promise((resolve, reject) => {
      chrome.runtime.sendMessage({
        type: 'TRANSLATE',
        text,
        targetLang
      }, (response) => {
        if (chrome.runtime.lastError) {
          reject(new Error(chrome.runtime.lastError.message));
        } else if (response.error) {
          reject(new Error(response.error));
        } else {
          resolve(response);
        }
      });
    });
  }

  createPopup() {
    const container = document.createElement('div');
    container.id = 'selection-translator';

    const shadow = container.attachShadow({ mode: 'closed' });
    shadow.innerHTML = `

          z-index: 2147483647;
          font-family: -apple-system, BlinkMacSystemFont, sans-serif;
          font-size: 14px;
          line-height: 1.5;
        }
        .original {
          color: #666;
          font-size: 12px;
          margin-bottom: 8px;
          padding-bottom: 8px;
          border-bottom: 1px solid #eee;
        }
        .translated {
          color: #333;
          font-weight: 500;
        }
        .loading {
          color: #999;
          text-align: center;
          padding: 8px;
        }
        .error {
          color: #e74c3c;
          font-size: 12px;
        }
        .actions {
          display: flex;
          gap: 8px;
          margin-top: 8px;
          padding-top: 8px;
          border-top: 1px solid #eee;
        }
        .actions button {
          background: none;
          border: 1px solid #ddd;
          border-radius: 4px;
          padding: 4px 8px;
          cursor: pointer;
          font-size: 12px;
        }
        .actions button:hover {
          background: #f5f5f5;
        }
      </style>
      <div class="popup">
        <div class="content"></div>
      </div>
    `;

    document.body.appendChild(container);
    this.popup = container;
    this.shadow = shadow;
  }

  positionPopup(x, y) {
    const popup = this.shadow.querySelector('.popup');
    const rect = this.popup.getBoundingClientRect();

    let left = x + window.scrollX;
    let top = y + window.scrollY + 20;

    if (left + 400 > window.innerWidth) {
      left = window.innerWidth - 410;
    }

    if (top + 200 > window.innerHeight + window.scrollY) {
      top = y + window.scrollY - 220;
    }

    popup.style.left = `${left}px`;
    popup.style.top = `${top}px`;
    popup.style.display = 'block';
    this.isVisible = true;
  }

  showLoading() {
    const content = this.shadow.querySelector('.content');
    content.innerHTML = '<div class="loading">翻译中...</div>';
  }

  showResult(original, result) {
    const content = this.shadow.querySelector('.content');
    content.innerHTML = `
      <div class="original">${original.substring(0, 100)}</div>
      <div class="translated">${result.translatedText}</div>
      ${result.alternatives ? `
        <div class="alternatives">
          ${result.alternatives.map(alt => `<div class="alt">${alt}</div>`).join('')}
        </div>
      ` : ''}

        <button class="copy-btn">复制</button>
        <button class="replace-btn">替换</button>
        <button class="speak-btn">朗读</button>
      </div>
    `;

    content.querySelector('.copy-btn').addEventListener('click', () => {
      navigator.clipboard.writeText(result.translatedText);
    });

    content.querySelector('.replace-btn').addEventListener('click', () => {
      this.replaceSelection(result.translatedText);
      this.hidePopup();
    });

    content.querySelector('.speak-btn').addEventListener('click', () => {
      this.speak(result.translatedText, result.targetLang);
    });
  }

  showError(message) {
    const content = this.shadow.querySelector('.content');
    content.innerHTML = `<div class="error">${message}</div>`;
  }

  hidePopup() {
    if (this.popup) {
      const popup = this.shadow.querySelector('.popup');
      popup.style.display = 'none';
      this.isVisible = false;
    }
  }

  replaceSelection(text) {
    const selection = window.getSelection();
    if (selection.rangeCount > 0) {
      const range = selection.getRangeAt(0);
      range.deleteContents();
      range.insertNode(document.createTextNode(text));
    }
  }

  speak(text, lang) {
    const utterance = new SpeechSynthesisUtterance(text);
    utterance.lang = lang || 'zh-CN';
    speechSynthesis.speak(utterance);
  }
}
```

```javascript
// content/page-translator.js
class PageTranslator {
  constructor() {
    this.originalNodes = new Map();
    this.isTranslated = false;
    this.batchSize = 10;
  }

  async translatePage(targetLang) {
    if (this.isTranslated) {
      this.restoreOriginal();
      return;
    }

    const textNodes = this.extractTranslatableNodes();
    const batches = this.createBatches(textNodes, this.batchSize);

    for (const batch of batches) {
      const texts = batch.map(node => node.textContent.trim());
      const results = await this.translateBatch(texts, targetLang);

      batch.forEach((node, index) => {
        this.originalNodes.set(node, node.textContent);
        node.textContent = results[index];
      });
    }

    this.isTranslated = true;
  }

  extractTranslatableNodes() {
    const walker = document.createTreeWalker(
      document.body,
      NodeFilter.SHOW_TEXT,
      {
        acceptNode(node) {
          if (!node.textContent.trim()) return NodeFilter.FILTER_REJECT;
          if (node.parentElement.closest('script, style, code, pre, [data-no-translate]')) {
            return NodeFilter.FILTER_REJECT;
          }
          if (node.textContent.trim().length < 3) return NodeFilter.FILTER_REJECT;
          return NodeFilter.FILTER_ACCEPT;
        }
      }
    );

    const nodes = [];
    while (walker.nextNode()) {
      nodes.push(walker.currentNode);
    }
    return nodes;
  }

  createBatches(items, size) {
    const batches = [];
    for (let i = 0; i < items.length; i += size) {
      batches.push(items.slice(i, i + size));
    }
    return batches;
  }

  async translateBatch(texts, targetLang) {
    return new Promise((resolve, reject) => {
      chrome.runtime.sendMessage({
        type: 'TRANSLATE_BATCH',
        texts,
        targetLang
      }, (response) => {
        if (chrome.runtime.lastError) {
          reject(new Error(chrome.runtime.lastError.message));
        } else if (response.error) {
          reject(new Error(response.error));
        } else {
          resolve(response.translations);
        }
      });
    });
  }

  restoreOriginal() {
    for (const [node, originalText] of this.originalNodes) {
      node.textContent = originalText;
    }
    this.originalNodes.clear();
    this.isTranslated = false;
  }
}
```

```javascript
// content/bilingual.js
class BilingualTranslator {
  constructor() {
    this.isActive = false;
  }

  async enable(targetLang) {
    if (this.isActive) return;

    const paragraphs = document.querySelectorAll('p, li, h1, h2, h3, h4, h5, h6, td, th, blockquote');
    const translatableElements = [...paragraphs].filter(el => {
      const text = el.textContent.trim();
      return text.length > 10 && !el.closest('script, style, code, pre');
    });

    const texts = translatableElements.map(el => el.textContent.trim());
    const translations = await this.translateBatch(texts, targetLang);

    translatableElements.forEach((el, index) => {
      if (translations[index] && translations[index] !== texts[index]) {
        const translationDiv = document.createElement('div');
        translationDiv.className = 'bilingual-translation';
        translationDiv.style.cssText = `
          color: #1a73e8;
          font-size: 0.9em;
          margin-top: 4px;
          padding: 4px 8px;
          background: rgba(26, 115, 232, 0.05);
          border-left: 3px solid #1a73e8;
          border-radius: 0 4px 4px 0;
        `;
        translationDiv.textContent = translations[index];
        el.appendChild(translationDiv);
      }
    });

    this.isActive = true;
  }

  disable() {
    document.querySelectorAll('.bilingual-translation').forEach(el => el.remove());
    this.isActive = false;
  }

  async translateBatch(texts, targetLang) {
    return new Promise((resolve, reject) => {
      chrome.runtime.sendMessage({
        type: 'TRANSLATE_BATCH',
        texts,
        targetLang
      }, (response) => {
        if (response.error) reject(new Error(response.error));
        else resolve(response.translations);
      });
    });
  }
}
```

## Background 翻译服务

```javascript
// background.js
class TranslationService {
  constructor() {
    this.cache = new Map();
    this.cacheMaxSize = 1000;
  }

  getCacheKey(text, targetLang) {
    return `${targetLang}:${text.substring(0, 100)}`;
  }

  async translate(text, targetLang) {
    const cacheKey = this.getCacheKey(text, targetLang);
    if (this.cache.has(cacheKey)) {
      return this.cache.get(cacheKey);
    }

    const { translationProvider, apiKey } = await chrome.storage.local.get([
      'translationProvider', 'apiKey'
    ]);

    let result;
    switch (translationProvider) {
      case 'openai':
        result = await this.translateWithOpenAI(text, targetLang, apiKey);
        break;
      case 'google':
        result = await this.translateWithGoogle(text, targetLang);
        break;
      default:
        result = await this.translateWithOpenAI(text, targetLang, apiKey);
    }

    this.cache.set(cacheKey, result);
    if (this.cache.size > this.cacheMaxSize) {
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
    }

    return result;
  }

  async translateWithOpenAI(text, targetLang, apiKey) {
    const langNames = { 'zh-CN': '中文', 'en': '英文', 'ja': '日文', 'ko': '韩文' };
    const langName = langNames[targetLang] || targetLang;

    const response = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${apiKey}`
      },
      body: JSON.stringify({
        model: 'gpt-4o-mini',
        messages: [
          {
            role: 'system',
            content: `你是专业翻译。将用户输入的文本翻译成${langName}。只返回翻译结果，不要解释。保持原文格式。`
          },
          { role: 'user', content: text }
        ],
        max_tokens: 2000,
        temperature: 0.3
      })
    });

    const data = await response.json();
    return {
      translatedText: data.choices[0].message.content,
      targetLang,
      provider: 'openai'
    };
  }

  async translateWithGoogle(text, targetLang) {
    const url = `https://translate.googleapis.com/translate_a/single?client=gtx&sl=auto&tl=${targetLang}&dt=t&q=${encodeURIComponent(text)}`;

    const response = await fetch(url);
    const data = await response.json();

    return {
      translatedText: data[0].map(item => item[0]).join(''),
      targetLang,
      provider: 'google'
    };
  }

  async translateBatch(texts, targetLang) {
    const results = await Promise.all(
      texts.map(text => this.translate(text, targetLang))
    );
    return results.map(r => r.translatedText);
  }
}

const translationService = new TranslationService();

chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
  if (message.type === 'TRANSLATE') {
    translationService.translate(message.text, message.targetLang)
      .then(sendResponse)
      .catch(error => sendResponse({ error: error.message }));
    return true;
  }

  if (message.type === 'TRANSLATE_BATCH') {
    translationService.translateBatch(message.texts, message.targetLang)
      .then(translations => sendResponse({ translations }))
      .catch(error => sendResponse({ error: error.message }));
    return true;
  }
});
```

## 常见误区

### 误区一：全页翻译会替换所有文本

需要过滤掉代码块、导航栏、按钮文字等不应翻译的内容。使用 `data-no-translate` 属性标记不应翻译的元素。

### 误区二：不缓存翻译结果

翻译 API 调用有成本，相同文本的翻译结果应该缓存。使用 LRU 缓存策略控制内存占用。

### 误区三：一次性翻译整个页面

大量文本应该分批翻译，避免 API 调用超时或超出 token 限制。同时分批翻译可以实现"渐进式翻译"的用户体验。

### 误区四：忽略翻译质量

机器翻译可能产生错误。提供"查看原文"功能，让用户可以对照查看。

## 工程建议

1. **多翻译源**：支持多个翻译 API（OpenAI、Google、DeepL），让用户选择
2. **智能语言检测**：自动检测源语言，不需要用户手动选择
3. **翻译缓存**：缓存翻译结果，减少 API 调用
4. **性能优化**：分批翻译、懒翻译（只翻译可视区域）
5. **用户控制**：提供开关、目标语言选择、翻译模式切换

## 小结

翻译扩展的核心是内容提取、API 集成和 DOM 操作。划词翻译需要精确获取选中文本并展示浮动面板，全页翻译需要遍历 DOM 节点并批量处理，双语对照需要在原文下方插入翻译结果。

```javascript
async function extractPDFText() {
  const pdfEmbed = document.querySelector('embed[type="application/pdf"]');
  if (!pdfEmbed) return null;

  // 使用 pdf.js 库
  const pdfjsLib = await import('pdfjs-dist');
  const pdf = await pdfjsLib.getDocument(pdfEmbed.src).promise;
  const texts = [];

  for (let i = 1; i <= pdf.numPages; i++) {
    const page = await pdf.getPage(i);
    const content = await page.getTextContent();
    texts.push(content.items.map(item => item.str).join(' '));
  }

  return texts;
}
```

**答案**：

```javascript
class TranslationMemory {
  constructor() {
    this.storageKey = 'translation_memory';
    this.maxEntries = 5000;
  }

  async get(text, sourceLang, targetLang) {
    const memory = await this.loadMemory();
    const key = this.generateKey(text, sourceLang, targetLang);
    return memory[key] || null;
  }

  async set(text, translation, sourceLang, targetLang) {
    const memory = await this.loadMemory();
    const key = this.generateKey(text, sourceLang, targetLang);

    memory[key] = {
      translation,
      timestamp: Date.now(),
      useCount: (memory[key]?.useCount || 0) + 1
    };

    const entries = Object.entries(memory);
    if (entries.length > this.maxEntries) {
      entries.sort((a, b) => a[1].timestamp - b[1].timestamp);
      const toRemove = entries.slice(0, entries.length - this.maxEntries);
      toRemove.forEach(([k]) => delete memory[k]);
    }

    await chrome.storage.local.set({ [this.storageKey]: memory });
  }

  generateKey(text, sourceLang, targetLang) {
    const normalized = text.trim().toLowerCase().substring(0, 200);
    return `${sourceLang}:${targetLang}:${normalized}`;
  }

  async loadMemory() {
    const { [this.storageKey]: memory = {} } = await chrome.storage.local.get(this.storageKey);
    return memory;
  }
}
```

```javascript
class GlossaryAwareTranslator {
  constructor() {
    this.glossaryKey = 'translation_glossary';
  }

  async loadGlossary() {
    const { [this.glossaryKey]: glossary = {} } = await chrome.storage.local.get(this.glossaryKey);
    return glossary;
  }

  async applyGlossary(text, targetLang) {
    const glossary = await this.loadGlossary();
    let result = text;

    for (const [term, translations] of Object.entries(glossary)) {
      if (translations[targetLang]) {
        const regex = new RegExp(term, 'gi');
        result = result.replace(regex, translations[targetLang]);
      }
    }

    return result;
  }

  async translateWithGlossary(text, targetLang) {
    const glossaryApplied = await this.applyGlossary(text, targetLang);

    const glossary = await this.loadGlossary();
    const glossaryTerms = Object.entries(glossary)
      .filter(([_, translations]) => translations[targetLang])
      .map(([term, translations]) => `${term} → ${translations[targetLang]}`)
      .join('\n');

    const prompt = glossaryTerms
      ? `请翻译以下文本。使用以下术语表：\n${glossaryTerms}\n\n${text}`
      : text;

    return this.callTranslationAPI(prompt, targetLang);
  }

  async addGlossaryTerm(sourceTerm, targetTerm, targetLang) {
    const glossary = await this.loadGlossary();

    if (!glossary[sourceTerm]) {
      glossary[sourceTerm] = {};
    }
    glossary[sourceTerm][targetLang] = targetTerm;

    await chrome.storage.local.set({ [this.glossaryKey]: glossary });
  }
}
```
