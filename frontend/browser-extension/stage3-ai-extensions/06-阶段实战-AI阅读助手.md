# 阶段实战：AI 阅读助手

## 场景引入

阶段三学习了页面内容提取、AI 摘要、翻译、写作助手和智能搜索。现在将这些能力综合起来，构建一个完整的 AI 阅读助手扩展。

这个扩展是"阅读者的瑞士军刀"：一键提取文章内容、生成摘要、翻译段落、记录笔记、语义搜索已读文章。它覆盖了 AI 浏览器扩展的核心场景。

## 学习目标

- 综合运用内容提取、AI 摘要、翻译、笔记等能力
- 构建完整的 Side Panel 交互界面
- 实现阅读历史管理和语义搜索
- 掌握 AI 扩展的端到端开发流程

## 功能规划

```
AI 阅读助手
├── 一键提取文章内容（Readability）
├── AI 摘要（简洁/详细/要点）
├── 划词翻译
```

```javascript
// content/extractor.js
class ArticleExtractor {
  constructor() {
    this.setupMessageListener();
  }

  setupMessageListener() {
    chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
      if (message.type === 'EXTRACT_ARTICLE') {
        sendResponse(this.extract());
      }
      if (message.type === 'EXTRACT_SELECTION') {
        sendResponse({
          text: window.getSelection().toString().trim(),
          url: window.location.href,
          title: document.title
        });
      }
      return false;
    });
  }

  extract() {
    try {
      const Readability = window.__Readability;
      if (Readability) {
        const doc = document.cloneNode(true);
        const reader = new Readability(doc);
        const article = reader.parse();
        if (article) {
          return {
            title: article.title,
            content: article.textContent,
            html: article.content,
            excerpt: article.excerpt,
            byline: article.byline,
            url: window.location.href,
            siteName: article.siteName
          };
        }
      }
    } catch (error) {
      console.error('Readability 提取失败:', error);
    }

    return this.fallbackExtract();
  }

  fallbackExtract() {
    const title = document.title;
    const content = document.body.innerText;
    return {
      title,
      content: content.substring(0, 15000),
      url: window.location.href
    };
  }
}

new ArticleExtractor();
```

```javascript
class PageAnnotator {
  constructor() {
    this.setupContextMenu();
    this.setupMessageListener();
  }

  setupContextMenu() {
    document.addEventListener('mouseup', () => {
      const selectedText = window.getSelection().toString().trim();
      if (selectedText.length > 0) {
        chrome.runtime.sendMessage({
          type: 'SELECTION_AVAILABLE',
          text: selectedText
        }).catch(() => {});
      }
    });
  }

  setupMessageListener() {
    chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
      if (message.type === 'HIGHLIGHT_SELECTION') {
        this.highlightSelection(message.color);
        sendResponse({ success: true });
      }
      return false;
    });
  }

  highlightSelection(color = '#ffeb3b') {
    const selection = window.getSelection();
    if (selection.rangeCount === 0) return;

    const range = selection.getRangeAt(0);
    const span = document.createElement('span');
    span.style.backgroundColor = color;
    span.style.borderRadius = '2px';
    span.style.padding = '1px 0';
    span.dataset.annotationTime = Date.now();

    try {
      range.surroundContents(span);
    } catch {
      const fragment = range.extractContents();
      span.appendChild(fragment);
      range.insertNode(span);
    }

    selection.removeAllRanges();
  }
}

new PageAnnotator();
```

```javascript
  switch (message.type) {
    case 'LLM_SUMMARIZE':
      llmService.summarize(message.text, message.options)
        .then(sendResponse)
        .catch(error => sendResponse({ error: error.message }));
      return true;

    case 'LLM_TRANSLATE':
      llmService.translate(message.text, message.targetLang)
        .then(sendResponse)
        .catch(error => sendResponse({ error: error.message }));
      return true;

    case 'SAVE_READING_HISTORY':
      storageService.saveReadingHistory(message.data).then(sendResponse);
      return true;

    case 'GET_READING_HISTORY':
      storageService.getReadingHistory().then(sendResponse);
      return true;

    case 'SEARCH_KNOWLEDGE':
      searchKnowledge(message.query).then(sendResponse);
      return true;
  }
});

async function searchKnowledge(query) {
  const pages = await storageService.getAllPages();
  if (pages.length === 0) return [];

  const { apiKey } = await chrome.storage.local.get('apiKey');
  if (!apiKey) {
    return pages.filter(p =>
      p.title.toLowerCase().includes(query.toLowerCase()) ||
      p.content.toLowerCase().includes(query.toLowerCase())
    ).slice(0, 10);
  }

  const embedding = await embeddingService.getEmbedding(query, apiKey);
  return findSimilar(embedding, pages, 10);
}

function findSimilar(queryEmbedding, documents, topK) {
  const results = documents.map(doc => ({
    ...doc,
    similarity: cosineSimilarity(queryEmbedding, doc.embedding || [])
  }));

  results.sort((a, b) => b.similarity - a.similarity);
  return results.slice(0, topK);
}

function cosineSimilarity(a, b) {
  if (!a.length || !b.length || a.length !== b.length) return 0;
  let dot = 0, normA = 0, normB = 0;
  for (let i = 0; i < a.length; i++) {
    dot += a[i] * b[i];
    normA += a[i] * a[i];
    normB += b[i] * b[i];
  }
  return dot / (Math.sqrt(normA) * Math.sqrt(normB));
}

// 侧边栏连接
chrome.runtime.onConnect.addListener((port) => {
  if (port.name === 'reader-panel') {
    port.onMessage.addListener(async (message) => {
      switch (message.type) {
        case 'EXTRACT_PAGE':
          const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });
          chrome.tabs.sendMessage(tab.id, { type: 'EXTRACT_ARTICLE' }, (response) => {

          port.postMessage({ type: 'SUMMARY_RESULT', summary: result });
          break;
      }
    });
  }
});
```

```javascript
// sidepanel.js
let port = null;
let currentArticle = null;

document.addEventListener('DOMContentLoaded', () => {
  connectToBackground();
  setupTabs();
  setupActions();
});

function connectToBackground() {
  port = chrome.runtime.connect({ name: 'reader-panel' });

  port.onMessage.addListener((message) => {
    switch (message.type) {
      case 'ARTICLE_DATA':
        handleArticleData(message.data);
        break;
      case 'SUMMARY_RESULT':
        showSummary(message.summary);
        break;
    }
  });
}

function setupTabs() {
  document.querySelectorAll('.tab').forEach(tab => {
    tab.addEventListener('click', () => {
      document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
      document.querySelectorAll('.tab-content').forEach(c => c.classList.add('hidden'));

      tab.classList.add('active');
      document.getElementById(`${tab.dataset.tab}-tab`).classList.remove('hidden');

      if (tab.dataset.tab === 'notes') loadNotes();
      if (tab.dataset.tab === 'history') loadHistory();
    });
  });
}

function setupActions() {
  document.getElementById('extract-btn').addEventListener('click', () => {
    port.postMessage({ type: 'EXTRACT_PAGE' });
  });

  document.getElementById('summarize-btn').addEventListener('click', () => {
    if (!currentArticle) return;
    const style = document.getElementById('summary-style').value;

    document.getElementById('summary-result').classList.remove('hidden');
    document.getElementById('summary-result').textContent = '正在生成摘要...';

    port.postMessage({
      type: 'SUMMARIZE',
      text: currentArticle.content,
      options: { style }
    });
  });

  document.getElementById('search-input').addEventListener('input',
    debounce(handleSearch, 500)
  );
}

function handleArticleData(data) {
  if (!data) return;
  currentArticle = data;

  document.getElementById('article-info').classList.remove('hidden');
  document.getElementById('summary-controls').classList.remove('hidden');
  document.getElementById('article-title').textContent = data.title;
  document.getElementById('article-excerpt').textContent =
    data.content.substring(0, 200) + '...';

  chrome.runtime.sendMessage({
    type: 'SAVE_READING_HISTORY',
    data: {
      url: data.url,
      title: data.title,
      excerpt: data.content.substring(0, 300),
      readAt: new Date().toISOString()
    }
  });
}

function showSummary(summary) {
  document.getElementById('summary-result').textContent = summary;
}

async function loadNotes() {
  const { notes = [] } = await chrome.storage.local.get('notes');
  const container = document.getElementById('notes-list');

  container.innerHTML = notes.slice(0, 50).map(note => `
    <div class="note-item" data-url="${note.url}">
      <div class="note-text">${note.text.substring(0, 100)}</div>
      <div class="note-url">${note.title || note.url}</div>
    </div>
  `).join('');

  container.querySelectorAll('.note-item').forEach(item => {
    item.addEventListener('click', () => {
      chrome.tabs.create({ url: item.dataset.url });
    });
  });
}

async function loadHistory() {
  const { readingHistory = [] } = await chrome.storage.local.get('readingHistory');
  const container = document.getElementById('history-list');

  container.innerHTML = readingHistory.slice(0, 50).map(item => `
    <div class="history-item" data-url="${item.url}">
      <div class="history-title">${item.title}</div>
      <div class="history-url">${item.url}</div>
    </div>
  `).join('');

  container.querySelectorAll('.history-item').forEach(item => {
    item.addEventListener('click', () => {
      chrome.tabs.create({ url: item.dataset.url });
    });
  });

async function handleSearch(event) {
  const query = event.target.value.trim();
  if (query.length < 3) return;

  const container = document.getElementById('search-results');
  container.innerHTML = '<div>搜索中...</div>';

  try {
    const results = await new Promise((resolve, reject) => {
      chrome.runtime.sendMessage({
        type: 'SEARCH_KNOWLEDGE',
        query
      }, (response) => {
        if (chrome.runtime.lastError) reject(chrome.runtime.lastError);
        else resolve(response);
      });
    });

    container.innerHTML = results.map(item => `
      <div class="search-result" data-url="${item.url}">
        <div class="result-title">${item.title}</div>
        <div class="result-url">${item.url}</div>
        ${item.similarity ? `<div class="similarity-score">相关度: ${(item.similarity * 100).toFixed(1)}%</div>` : ''}
      </div>
    `).join('');

    container.querySelectorAll('.search-result').forEach(item => {
      item.addEventListener('click', () => {
        chrome.tabs.create({ url: item.dataset.url });
      });
    });
  } catch (error) {
    container.innerHTML = `<div style="color:#e74c3c">${error.message}</div>`;
  }
}

function debounce(fn, delay) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}
```

## 工程建议

1. **Readability 加载**：通过 manifest 的 `web_accessible_resources` 或在 content script 中动态加载 Readability.js
2. **API Key 配置**：在 popup 或 sidepanel 的设置页面中配置，存储在 chrome.storage.local
3. **错误降级**：当 AI API 不可用时，降级到基础功能（内容提取、笔记）
4. **数据导出**：支持导出阅读历史和笔记为 Markdown/JSON

## 常见误区

### 误区一：在 Side Panel 中执行 Content Script 操作

Side Panel 是独立的页面，不能直接操作当前标签页的 DOM。必须通过消息传递。

### 误区二：忽略页面切换

用户切换标签页时，Side Panel 需要更新显示内容。监听 `chrome.tabs.onActivated` 事件。

### 误区三：大量数据存储在 chrome.storage.local

chrome.storage.local 默认限制 5MB。大量阅读历史和笔记应该使用 IndexedDB。

## 小结

AI 阅读助手综合了阶段三的所有核心能力，是一个完整的 AI 浏览器扩展项目。它展示了如何将内容提取、AI 处理、用户交互和数据管理整合到一个扩展中。

```javascript
async function generateReadingReport(period = 'week') {
  const { readingHistory = [] } = await chrome.storage.local.get('readingHistory');

  const now = new Date();
  const cutoff = new Date();
  if (period === 'day') cutoff.setDate(now.getDate() - 1);
  else cutoff.setDate(now.getDate() - 7);

  const periodHistory = readingHistory.filter(
    item => new Date(item.readAt) >= cutoff
  );

  const domains = {};
  periodHistory.forEach(item => {
    const domain = new URL(item.url).hostname;
    domains[domain] = (domains[domain] || 0) + 1;
  });

  const topDomains = Object.entries(domains)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 5);

  return {
    period,
    totalArticles: periodHistory.length,
    totalWords: periodHistory.reduce((sum, item) => sum + (item.wordCount || 0), 0),
    topDomains,
    articles: periodHistory
  };
}
```

```javascript
class ArticleReader {
  constructor() {
    this.synth = window.speechSynthesis;
    this.utterance = null;
    this.isPlaying = false;
  }

  speak(text, rate = 1.0, lang = 'zh-CN') {
    this.stop();

    this.utterance = new SpeechSynthesisUtterance(text);
    this.utterance.lang = lang;
    this.utterance.rate = rate;

    this.utterance.onend = () => {
      this.isPlaying = false;
      chrome.runtime.sendMessage({ type: 'SPEECH_ENDED' });
    };

    this.synth.speak(this.utterance);
    this.isPlaying = true;
  }

  pause() {
    this.synth.pause();
  }

  resume() {
    this.synth.resume();
  }

  stop() {
    this.synth.cancel();
    this.isPlaying = false;
  }
}
```

```javascript
function wrapText(ctx, text, x, y, maxWidth, lineHeight) {
  const words = text.split('');
  let line = '';
  let currentY = y;

  for (const char of words) {
    const testLine = line + char;
    if (ctx.measureText(testLine).width > maxWidth) {
      ctx.fillText(line, x, currentY);
      line = char;
      currentY += lineHeight;
    } else {
      line = testLine;
    }
  }
  ctx.fillText(line, x, currentY);
}
```
