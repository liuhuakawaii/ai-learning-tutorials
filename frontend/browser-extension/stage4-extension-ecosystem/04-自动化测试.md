# 自动化测试

## 场景引入

手动测试扩展是一件痛苦的事：打开浏览器、加载扩展、点击图标、检查功能、修改代码、重新加载、再测试……每次修改都要重复这个过程。

自动化测试可以解放你的双手。通过 Puppeteer 或 Playwright，你可以模拟用户操作、验证扩展功能、在 CI/CD 中自动运行测试。

本课将讲解如何为浏览器扩展构建自动化测试套件。

## 学习目标

- 掌握使用 Puppeteer 测试扩展的方法
- 学会编写 E2E 测试用例
- 理解扩展测试的特殊挑战
- 配置 CI/CD 自动化测试

### 启动带扩展的浏览器

```javascript
// tests/setup.js
const puppeteer = require('puppeteer');
const path = require('path');

async function launchBrowserWithExtension(extensionPath) {
  const browser = await puppeteer.launch({
    headless: false, // 扩展测试需要有头模式
    args: [
      `--disable-extensions-except=${extensionPath}`,
      `--load-extension=${extensionPath}`,
      '--no-sandbox'
    ]
  });

  return browser;
}

async function getExtensionId(browser) {
  const targets = await browser.targets();
  const backgroundTarget = targets.find(
    t => t.type() === 'service_worker' || t.type() === 'background_page'
  );

  if (!backgroundTarget) {
    throw new Error('找不到扩展的 Service Worker');
  }

  const url = backgroundTarget.url();
  const [, , extensionId] = url.split('/');
  return extensionId;
}

module.exports = { launchBrowserWithExtension, getExtensionId };
```

## 基础测试

### 测试 Popup

```javascript
// tests/popup.test.js
const puppeteer = require('puppeteer');
const { launchBrowserWithExtension, getExtensionId } = require('./setup');

describe('Popup 测试', () => {
  let browser;
  let extensionId;

  beforeAll(async () => {
    const extensionPath = path.resolve(__dirname, '../dist');
    browser = await launchBrowserWithExtension(extensionPath);
    extensionId = await getExtensionId(browser);
  });

  afterAll(async () => {
    await browser.close();
  });

  test('Popup 能正常打开', async () => {
    const page = await browser.newPage();
    await page.goto(`chrome-extension://${extensionId}/popup/popup.html`);

    const title = await page.title();
    expect(title).toBeTruthy();
  });

  test('Popup 显示正确的内容', async () => {
    const page = await browser.newPage();
    await page.goto(`chrome-extension://${extensionId}/popup/popup.html`);

    await page.waitForSelector('.container');
    const text = await page.$eval('.container', el => el.textContent);
    expect(text).toBeTruthy();
  });

  test('点击按钮触发操作', async () => {
    const page = await browser.newPage();
    await page.goto(`chrome-extension://${extensionId}/popup/popup.html`);

    await page.waitForSelector('#action-btn');
    await page.click('#action-btn');

    await page.waitForSelector('.result');
    const result = await page.$eval('.result', el => el.textContent);
    expect(result).toBeTruthy();
  });
});
```

### 测试 Background Service Worker

```javascript
// tests/background.test.js
describe('Background Service Worker 测试', () => {
  let browser;
  let extensionId;

  beforeAll(async () => {
    const extensionPath = path.resolve(__dirname, '../dist');
    browser = await launchBrowserWithExtension(extensionPath);
    extensionId = await getExtensionId(browser);
  });

  afterAll(async () => {
    await browser.close();
  });

  test('Service Worker 响应消息', async () => {
    const page = await browser.newPage();
    await page.goto('https://example.com');

    const response = await page.evaluate(async (extId) => {
      return new Promise((resolve) => {
        chrome.runtime.sendMessage(extId, { type: 'PING' }, (response) => {
          resolve(response);
        });
      });
    }, extensionId);

    expect(response).toEqual({ pong: true });
  });

  test('存储功能正常', async () => {
    const page = await browser.newPage();
    await page.goto(`chrome-extension://${extensionId}/popup/popup.html`);

    await page.evaluate(async () => {
      await chrome.storage.local.set({ testKey: 'testValue' });
    });

    const value = await page.evaluate(async () => {
      const { testKey } = await chrome.storage.local.get('testKey');
      return testKey;
    });

    expect(value).toBe('testValue');
  });
});
```

// tests/user-interactions.test.js
describe('用户交互测试', () => {
  test('模拟选中文本', async () => {

    await page.keyboard.down('Shift');
    await page.keyboard.press('KeyH');
    await page.keyboard.up('Shift');
    await page.keyboard.up('Control');

    await page.waitForTimeout(1000);

## CI/CD 配置

### GitHub Actions

```yaml
# .github/workflows/test.yml
name: Extension Tests

on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v3

      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '18'

      - name: Install dependencies
        run: npm ci

      - name: Build extension
        run: npm run build

      - name: Run tests
        run: npm test
        env:
          CI: true
```

## 测试策略

### 测试金字塔

```
        E2E 测试（少量）
       /            \
    集成测试（适量）
   /                \
  单元测试（大量）
```

- **单元测试**：测试独立的函数和模块
- **集成测试**：测试组件之间的交互
- **E2E 测试**：测试完整的用户流程

### 单元测试

```javascript
// tests/unit/utils.test.js
const { formatDate, truncateText, isValidUrl } = require('../../src/utils');

describe('工具函数', () => {
  test('formatDate 格式化日期', () => {
    const date = new Date('2024-01-15');
    expect(formatDate(date)).toBe('2024-01-15');
  });

  test('truncateText 截断文本', () => {
    const text = '这是一段很长的文本内容';
    expect(truncateText(text, 5)).toBe('这是一段很...');
  });

  test('isValidUrl 验证 URL', () => {
    expect(isValidUrl('https://example.com')).toBe(true);
    expect(isValidUrl('not-a-url')).toBe(false);
  });
});
```

## 常见误区

### 误区一：只在 Chrome 上测试

如果扩展支持多浏览器，必须在所有目标浏览器上测试。

### 误区二：忽略无头模式的差异

某些扩展功能在无头模式下可能不正常。关键测试应该使用有头模式。

### 误区三：测试不稳定就忽略

不稳定的测试（flaky tests）会降低团队对测试的信任。应该修复或删除不稳定的测试。

### 误区四：不测试错误场景

只测试正常流程是不够的。必须测试错误处理、边界情况和异常输入。

## 工程建议

1. **有头模式**：扩展测试必须使用有头模式（`headless: false`）
2. **等待策略**：使用 `waitForSelector` 而非固定延时
3. **测试隔离**：每个测试用例应该独立，不依赖其他测试的状态
4. **CI 集成**：在 CI/CD 中自动运行测试
5. **覆盖率**：关注代码覆盖率，但不要追求 100%

## 小结

自动化测试是扩展质量保障的关键。通过 Puppeteer 可以模拟用户操作、验证扩展功能、在 CI/CD 中自动运行。合理的测试策略（单元测试 + 集成测试 + E2E 测试）可以显著提高扩展的可靠性。

describe('错误场景测试', () => {
  test('无网络连接时的降级', async () => {
    const page = await browser.newPage();
    await page.setOfflineMode(true);
    await page.goto(`chrome-extension://${extensionId}/popup.html`);

    const error = await page.waitForSelector('.error-message');
    const text = await error.evaluate(el => el.textContent);
    expect(text).toContain('网络');

  test('API 错误处理', async () => {
    await page.setRequestInterception(true);
    page.on('request', (request) => {
      if (request.url().includes('api.example.com')) {
        request.respond({ status: 500 });
      } else {
        request.continue();
      }
    });

    await page.goto(`chrome-extension://${extensionId}/popup.html`);
    await page.click('#fetch-btn');

    const error = await page.waitForSelector('.api-error');
    expect(error).toBeTruthy();
  });

  test('权限拒绝后的提示', async () => {
    // 模拟权限拒绝
    const page = await browser.newPage();
    await page.goto(`chrome-extension://${extensionId}/popup.html`);

    await page.evaluate(async () => {
      chrome.permissions.request = () => Promise.resolve(false);
    });

    await page.click('#request-permission-btn');

    const message = await page.waitForSelector('.permission-denied');
    expect(message).toBeTruthy();
  });
});

### 练习三

**答案**：

```yaml
# .github/workflows/test.yml
name: Test Extension

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v3

      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '18'
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Lint
        run: npm run lint

      - name: Unit tests
        run: npm run test:unit

      - name: Build
        run: npm run build

      - name: E2E tests
        run: npm run test:e2e

      - name: Upload artifacts
        if: failure()
        uses: actions/upload-artifact@v3
        with:
          name: test-screenshots
          path: tests/screenshots/
```
