## 高亮核心

```javascript
// content/highlighter.js
class TextHighlighter {
  highlightRange(range, color, annotationId) {
    const span = document.createElement('span')
    span.className = 'ext-highlight'
    span.setAttribute('data-annotation-id', annotationId)
    span.style.backgroundColor = color
    span.style.cursor = 'pointer'

    try {
      range.surroundContents(span)
    } catch (e) {
      // 跨元素选区：extractContents 处理
      const fragment = range.extractContents()
      const wrapper = document.createElement('span')
      wrapper.className = 'ext-highlight'
      wrapper.setAttribute('data-annotation-id', annotationId)
      wrapper.style.backgroundColor = color
      wrapper.appendChild(fragment)
      range.insertNode(wrapper)
    }
  }

  removeHighlight(id) {
    document.querySelectorAll(`[data-annotation-id="${id}"]`).forEach(el => {
      const parent = el.parentNode
      while (el.firstChild) parent.insertBefore(el.firstChild, el)
      parent.removeChild(el)
      parent.normalize()
    })
  }

  getXPathForRange(range) {
    const container = range.startContainer
    const element = container.nodeType === Node.TEXT_NODE
      ? container.parentNode : container
    return {
      xpath: this.getXPath(element),
      startOffset: range.startOffset,
      textSnippet: container.textContent.substring(
        Math.max(0, range.startOffset - 20),
        range.startOffset + range.toString().length + 20
      )
    }
  }

  getXPath(element) {
    if (element === document.body) return '/html/body'
    let index = 1
    let sibling = element.previousSibling
    while (sibling) {
      if (sibling.nodeType === 1 && sibling.tagName === element.tagName) index++
      sibling = sibling.previousSibling
    }
    return `${this.getXPath(element.parentNode)}/${element.tagName.toLowerCase()}[${index}]`
  }

  restoreHighlight(annotation) {
    try {
      const result = document.evaluate(annotation.xpath, document, null, 9, null)
      const element = result.singleNodeValue
      if (!element) return false

      const walker = document.createTreeWalker(element, NodeFilter.SHOW_TEXT)
      while (walker.nextNode()) {
        const offset = walker.currentNode.textContent.indexOf(annotation.selectedText)
        if (offset !== -1) {
          const range = document.createRange()
          range.setStart(walker.currentNode, offset)
          range.setEnd(walker.currentNode, offset + annotation.selectedText.length)
          this.highlightRange(range, annotation.color, annotation.id)
          return true
        }
      }
    } catch { return false }
  }
}
```

## 操作菜单

```javascript
// content/menu.js
class SelectionMenu {
  constructor(onAction) {
    this.menu = document.createElement('div')
    this.menu.className = 'ext-selection-menu'
    this.menu.style.display = 'none'

    const colors = ['#ffeb3b', '#4caf50', '#2196f3', '#ff9800', '#e91e63']
    colors.forEach(color => {
      const btn = document.createElement('button')
      btn.className = 'ext-color-btn'
      btn.style.backgroundColor = color
      btn.onclick = (e) => { e.stopPropagation(); onAction('highlight', { color }); this.hide() }
      this.menu.appendChild(btn)
    })

    const noteBtn = document.createElement('button')
    noteBtn.className = 'ext-action-btn'
    noteBtn.textContent = '📝 笔记'
    noteBtn.onclick = (e) => { e.stopPropagation(); onAction('note', {}); this.hide() }
    this.menu.appendChild(noteBtn)

    document.body.appendChild(this.menu)
  }

  show(x, y) {
    this.menu.style.left = `${x}px`
    this.menu.style.top = `${y - 60}px`
    this.menu.style.display = 'flex'
  }

  hide() { this.menu.style.display = 'none' }
}
```

## 主控制器

```javascript
// content/content.js
(async function() {
  const highlighter = new TextHighlighter()
  const url = window.location.href
  const storageKey = `annotations_${url}`

  let annotations = (await chrome.storage.local.get(storageKey))[storageKey] || []

  // 恢复高亮
  annotations.forEach(ann => highlighter.restoreHighlight(ann))

  const menu = new SelectionMenu((action, data) => {
    const selection = window.getSelection()
    const range = selection.getRangeAt(0)
    const annotation = {
      id: `ann_${Date.now()}_${Math.random().toString(36).slice(2, 9)}`,
      selectedText: selection.toString().trim(),
      color: data.color || '#ffeb3b',
      note: '',
      url,
      ...highlighter.getXPathForRange(range),
      createdAt: new Date().toISOString()
    }

    if (action === 'note') {
      annotation.note = prompt('笔记:') || ''
      if (!annotation.note) return
    }

    highlighter.highlightRange(range, annotation.color, annotation.id)
    annotations.push(annotation)
    chrome.storage.local.set({ [storageKey]: annotations })
    selection.removeAllRanges()
  })

  document.addEventListener('mouseup', (e) => {
    if (e.target.closest('.ext-selection-menu')) return
    const text = window.getSelection().toString().trim()
    if (text.length > 0) {
      const range = window.getSelection().getRangeAt(0)
      const rect = range.getBoundingClientRect()
      menu.show(rect.left + rect.width / 2 - 75, rect.top + window.scrollY)
    }
  })

  document.addEventListener('mousedown', (e) => {
    if (!e.target.closest('.ext-selection-menu')) menu.hide()
  })
})()
```

## 你可能踩的坑

**坑一：range.surroundContents 跨节点失败**

选区跨越多个 DOM 节点时会抛异常。必须用 `extractContents` + 手动插入。

**坑二：XPath 恢复依赖页面结构不变**

SPA 路由切换后页面结构可能变化。更健壮的方案结合文本指纹匹配。

**坑三：大量高亮恢复性能差**

数百个高亮恢复时会卡。用 `requestIdleCallback` 分批恢复。

```javascript
function exportToMarkdown(annotations) {
  let md = `# 网页标注\n\n导出时间: ${new Date().toLocaleString()}\n\n`
  md += `## ${window.location.hostname}\n> ${window.location.href}\n\n`

  annotations.forEach(ann => {
    md += `- <mark style="background:${ann.color}">${ann.selectedText}</mark>\n`
    if (ann.note) md += `  - 笔记: ${ann.note}\n`
  })

  const blob = new Blob([md], { type: 'text/markdown' })
  const a = document.createElement('a')
  a.href = URL.createObjectURL(blob)
  a.download = `annotations-${new Date().toISOString().slice(0, 10)}.md`
  a.click()
}
```
