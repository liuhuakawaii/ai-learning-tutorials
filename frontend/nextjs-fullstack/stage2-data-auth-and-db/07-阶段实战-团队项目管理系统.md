# 阶段实战：团队项目管理系统

## 项目目标

构建一个团队项目管理系统：登录注册、团队空间、项目 CRUD、成员邀请、权限控制。综合运用 Prisma、Auth.js、Server Actions、Zod 验证。

## 功能清单

1. 用户注册/登录（Auth.js）
2. 创建/编辑/删除项目
3. 邀请成员加入项目
4. 角色权限：OWNER > ADMIN > MEMBER
5. 表单验证（Zod）
6. 文件上传

## 数据模型

```prisma
model User {
  id       String  @id @default(cuid())
  email    String  @unique
  name     String?
  password String  // hashed
  role     Role    @default(MEMBER)
  projects ProjectMember[]
}

model Project {
  id          String  @id @default(cuid())
  name        String
  description String?
  ownerId     String
  owner       User    @relation(fields: [ownerId], references: [id])
  members     ProjectMember[]
}

model ProjectMember {
  id        String @id @default(cuid())
  userId    String
  projectId String
  role      Role   @default(MEMBER)
  user      User   @relation(fields: [userId], references: [id])
  project   Project @relation(fields: [projectId], references: [id], onDelete: Cascade)
  @@unique([userId, projectId])
}
```

## 认证（Auth.js）

```typescript
// auth.ts
import NextAuth from 'next-auth'
import Credentials from 'next-auth/providers/credentials'
import { prisma } from '@/lib/prisma'
import { compare } from 'bcryptjs'

export const { handlers, auth, signIn, signOut } = NextAuth({
  providers: [
    Credentials({
      credentials: { email: {}, password: {} },
      authorize: async ({ email, password }) => {
        const user = await prisma.user.findUnique({ where: { email: email as string } })
        if (!user) return null
        const valid = await compare(password as string, user.password)
        return valid ? user : null
      },
    }),
  ],
  callbacks: {
    session: ({ session, token }) => {
      session.user.id = token.sub!
      return session
    },
  },
})
```

## 权限检查

```typescript
// lib/auth.ts
import { auth } from '@/auth'
import { prisma } from './prisma'

export async function requireAuth() {
  const session = await auth()
  if (!session?.user) throw new Error('Unauthorized')
  return session.user
}

export async function requireProjectRole(projectId: string, minRole: Role = 'MEMBER') {
  const user = await requireAuth()
  const member = await prisma.projectMember.findUnique({
    where: { userId_projectId: { userId: user.id, projectId } },
  })
  if (!member) throw new Error('Not a member')

  const hierarchy = { OWNER: 3, ADMIN: 2, MEMBER: 1 }
  if (hierarchy[member.role] < hierarchy[minRole]) throw new Error('Insufficient permissions')

  return { user, member }
}
```

## Server Actions

```typescript
// app/(admin)/projects/actions.ts
'use server'

import { revalidatePath } from 'next/cache'
import { redirect } from 'next/navigation'
import { z } from 'zod'
import { prisma } from '@/lib/prisma'
import { requireAuth, requireProjectRole } from '@/lib/auth'

const projectSchema = z.object({
  name: z.string().min(2, '项目名至少 2 个字').max(50),
  description: z.string().max(200).optional(),
})

export async function createProject(formData: FormData) {
  const user = await requireAuth()
  const parsed = projectSchema.safeParse({
    name: formData.get('name'),
    description: formData.get('description'),
  })
  if (!parsed.success) return { error: parsed.error.errors[0].message }

  const project = await prisma.project.create({
    data: {
      ...parsed.data,
      ownerId: user.id,
      members: { create: { userId: user.id, role: 'OWNER' } },
    },
  })
  revalidatePath('/admin/projects')
  redirect(`/admin/projects/${project.id}`)
}

export async function deleteProject(projectId: string) {
  await requireProjectRole(projectId, 'OWNER')
  await prisma.project.delete({ where: { id: projectId } })
  revalidatePath('/admin/projects')
}

export async function inviteMember(projectId: string, email: string, role: Role = 'MEMBER') {
  await requireProjectRole(projectId, 'ADMIN')
  const user = await prisma.user.findUnique({ where: { email } })
  if (!user) return { error: '用户不存在' }

  await prisma.projectMember.create({
    data: { userId: user.id, projectId, role },
  })
  revalidatePath(`/admin/projects/${projectId}`)
}
```

## 文件上传

```typescript
// app/api/upload/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { writeFile } from 'fs/promises'
import { join } from 'path'
import { requireAuth } from '@/lib/auth'

export async function POST(req: NextRequest) {
  await requireAuth()
  const formData = await req.formData()
  const file = formData.get('file') as File
  if (!file) return NextResponse.json({ error: 'No file' }, { status: 400 })

  const bytes = await file.arrayBuffer()
  const buffer = Buffer.from(bytes)
  const filename = `${Date.now()}-${file.name}`
  await writeFile(join(process.cwd(), 'public/uploads', filename), buffer)

  return NextResponse.json({ url: `/uploads/${filename}` })
}
```

  ADMIN:  ['manage_members', 'edit_project', 'view_project'],
  MEMBER: ['edit_project', 'view_project'],

function canDo(role: Role, action: string) {
  return permissions[role]?.includes(action) ?? false
}
