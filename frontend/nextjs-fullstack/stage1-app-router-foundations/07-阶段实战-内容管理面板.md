# 阶段实战：内容管理面板

## 项目目标

构建一个简易 CMS 面板：文章增删改查、草稿/发布状态切换、加载态和空状态、表单错误提示。综合运用 App Router 路由布局、Server Components、Client Components、Server Actions。

## 项目结构

```
app/
├── layout.tsx                  ← 根布局
├── (admin)/                    ← 后台路由组
│   ├── layout.tsx              ← 后台布局（侧边栏）
│   ├── loading.tsx             ← 加载态
│   ├── page.tsx                → /admin（Dashboard）
│   ├── posts/
│   │   ├── page.tsx            → /admin/posts（文章列表）
│   │   ├── loading.tsx
│   │   ├── new/page.tsx        → /admin/posts/new
│   │   └── [id]/
│   │       ├── page.tsx        → /admin/posts/123
│   │       └── edit/page.tsx   → /admin/posts/123/edit

    ├── EmptyState.tsx

```

`(admin)` 是路由分组——不参与 URL 路径，只为后台页面提供统一的 layout。

## 数据模型

```typescript
// lib/db.ts
export interface Post {
  id: string
  title: string
  content: string
  status: 'draft' | 'published'
  createdAt: string
  updatedAt: string
}

let posts: Post[] = [
  { id: '1', title: '第一篇文章', content: '内容...', status: 'published', createdAt: '2024-01-15T10:00:00Z', updatedAt: '2024-01-15T10:00:00Z' },
  { id: '2', title: 'Next.js 学习笔记', content: 'App Router 是...', status: 'draft', createdAt: '2024-01-16T14:30:00Z', updatedAt: '2024-01-16T14:30:00Z' },
]

export const db = {
  post: {
    findMany: async (options?: { status?: string }) => {
      await new Promise(r => setTimeout(r, 300))
      return options?.status ? posts.filter(p => p.status === options.status) : [...posts]
    },
    findUnique: async (id: string) => {
      await new Promise(r => setTimeout(r, 200))
      return posts.find(p => p.id === id) || null
    },
    create: async (data: Omit<Post, 'id' | 'createdAt' | 'updatedAt'>) => {
      const post = { ...data, id: String(Date.now()), createdAt: new Date().toISOString(), updatedAt: new Date().toISOString() }
      posts.push(post)
      return post
    },
    update: async (id: string, data: Partial<Post>) => {
      const i = posts.findIndex(p => p.id === id)
      if (i === -1) throw new Error('Not found')
      posts[i] = { ...posts[i], ...data, updatedAt: new Date().toISOString() }
      return posts[i]
    },
    delete: async (id: string) => { posts = posts.filter(p => p.id !== id) },
  },
}
```

## Server Actions

```typescript
// app/(admin)/posts/actions.ts
'use server'

import { revalidatePath } from 'next/cache'
import { redirect } from 'next/navigation'
import { db } from '@/lib/db'

export async function createPost(formData: FormData) {
  const title = formData.get('title') as string
  const content = formData.get('content') as string
  const status = formData.get('status') as 'draft' | 'published'

  if (!title || title.length < 2) return { error: '标题至少 2 个字' }

  const post = await db.post.create({ title, content: content || '', status: status || 'draft' })
  revalidatePath('/admin/posts')
  redirect(`/admin/posts/${post.id}`)
}

export async function updatePost(postId: string, formData: FormData) {
  const title = formData.get('title') as string
  const content = formData.get('content') as string
  const status = formData.get('status') as 'draft' | 'published'

  if (!title || title.length < 2) return { error: '标题至少 2 个字' }

  await db.post.update(postId, { title, content: content || '', status })
  revalidatePath('/admin/posts')
  redirect(`/admin/posts/${postId}`)
}

export async function deletePost(postId: string) {
  await db.post.delete(postId)
  revalidatePath('/admin/posts')
}
```

`revalidatePath` 清除列表页缓存，否则用户看不到最新数据。

## 表单组件

```tsx
// components/PostForm.tsx
'use client'

import { useActionState } from 'react'

export function PostForm({ post, action }: {
  post?: { title: string; content: string; status: string }
  action: (formData: FormData) => Promise<{ error?: string }>
}) {
  const [state, formAction, isPending] = useActionState(action, null)

  return (
    <form action={formAction} className="space-y-4">
      <div>
        <label htmlFor="title" className="block text-sm font-medium mb-1">标题</label>
        <input id="title" name="title" defaultValue={post?.title} required
          className="w-full px-3 py-2 border rounded" />
      </div>
      <div>
        <label htmlFor="content" className="block text-sm font-medium mb-1">内容</label>
        <textarea id="content" name="content" rows={10} defaultValue={post?.content}
          className="w-full px-3 py-2 border rounded" />
      </div>
      <div>
        <label htmlFor="status" className="block text-sm font-medium mb-1">状态</label>
        <select id="status" name="status" defaultValue={post?.status || 'draft'}
          className="w-full px-3 py-2 border rounded">
          <option value="draft">草稿</option>
          <option value="published">发布</option>
        </select>
      </div>
      {state?.error && <p className="text-red-500">{state.error}</p>}
      <button type="submit" disabled={isPending}
        className="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600 disabled:opacity-50">
        {isPending ? '保存中...' : '保存'}
      </button>
    </form>
  )
}
```
