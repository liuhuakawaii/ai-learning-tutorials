# 网络请求

## 场景引入

App 需从后端 API 获取列表、提交表单、上传图片。移动网络不稳定，请求需在页面切换时取消，失败后需自动重试。这节课系统学习网络请求方案。

## 学习目标

- 掌握 fetch 和 Axios 封装
- 实现请求/响应拦截器
- 设计离线缓存策略
- 完善错误处理与重试
- 管理加载状态与请求取消

## 一、fetch 基础

```typescript
// utils/request.ts
const BASE_URL = 'https://api.example.com';

interface ApiResponse<T> { code: number; data: T; message: string; }

export async function get<T>(path: string): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    headers: { 'Content-Type': 'application/json', Accept: 'application/json' },
  });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  const result: ApiResponse<T> = await res.json();
  if (result.code !== 0) throw new Error(result.message || '请求失败');
  return result.data;
}

export async function post<T>(path: string, body: unknown): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
  });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  const result: ApiResponse<T> = await res.json();
  if (result.code !== 0) throw new Error(result.message || '请求失败');
  return result.data;
}
```

```typescript
// utils/http.ts
import axios, { AxiosInstance, AxiosError, InternalAxiosRequestConfig } from 'axios';
import AsyncStorage from '@react-native-async-storage/async-storage';

const http: AxiosInstance = axios.create({ baseURL: 'https://api.example.com', timeout: 15000 });

http.interceptors.request.use(async (config: InternalAxiosRequestConfig) => {
  const token = await AsyncStorage.getItem('auth_token');
  if (token && config.headers) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

http.interceptors.response.use(
  (response) => {
    const { code, data, message } = response.data;
    if (code !== 0) throw new Error(message || '业务错误');
    return data;
  },
  async (error: AxiosError) => {
    if (error.response) {
      const s = error.response.status;
      if (s === 401) await AsyncStorage.removeItem('auth_token');
      if (s === 403) throw new Error('没有权限');
      if (s === 404) throw new Error('资源不存在');
      if (s >= 500) throw new Error('服务器繁忙');
    }
    if (error.code === 'ECONNABORTED') throw new Error('请求超时');
    if (!error.response) throw new Error('网络连接失败');
    throw error;
  }
);

export const httpGet = <T>(url: string, config?: any) => http.get<unknown, T>(url, config);
export const httpPost = <T>(url: string, data?: unknown, config?: any) => http.post<unknown, T>(url, data, config);
export default http;
```

## 三、离线缓存

```typescript
// utils/cache.ts
import AsyncStorage from '@react-native-async-storage/async-storage';

interface CacheEntry<T> { data: T; timestamp: number; ttl: number; }
const PREFIX = '@cache:';
const DEFAULT_TTL = 5 * 60 * 1000;

export const cache = {
  async get<T>(key: string): Promise<T | null> {
    const raw = await AsyncStorage.getItem(`${PREFIX}${key}`);
    if (!raw) return null;
    const entry: CacheEntry<T> = JSON.parse(raw);
    if (Date.now() - entry.timestamp > entry.ttl) { await AsyncStorage.removeItem(`${PREFIX}${key}`); return null; }
    return entry.data;
  },
  async set<T>(key: string, data: T, ttl = DEFAULT_TTL) {
    await AsyncStorage.setItem(`${PREFIX}${key}`, JSON.stringify({ data, timestamp: Date.now(), ttl }));
  },
  async clear() {
    const keys = await AsyncStorage.getAllKeys();
    await AsyncStorage.multiRemove(keys.filter((k) => k.startsWith(PREFIX)));
  },
};

import { httpGet } from './http';

export async function cachedGet<T>(url: string, opts: { ttl?: number; forceRefresh?: boolean } = {}): Promise<T> {
  const { ttl, forceRefresh = false } = opts;
  const ck = `get:${url}`;
  if (!forceRefresh) { const c = await cache.get<T>(ck); if (c !== null) return c; }
  const data = await httpGet<T>(url);
  await cache.set(ck, data, ttl);
  return data;
}
```

## 四、错误处理与重试

```typescript
// utils/retry.ts
export async function withRetry<T>(fn: () => Promise<T>, opts: { maxRetries?: number; delay?: number; backoff?: 'fixed' | 'exponential' } = {}): Promise<T> {
  const { maxRetries = 3, delay = 1000, backoff = 'exponential' } = opts;
  let lastErr: Error;
  for (let i = 0; i <= maxRetries; i++) {
    try { return await fn(); }
    catch (e) {
      lastErr = e as Error;
      if (i === maxRetries) throw lastErr;
      const status = (lastErr as any).status;
      if (status >= 400 && status < 500) throw lastErr; // 不重试客户端错误
      await new Promise((r) => setTimeout(r, backoff === 'exponential' ? delay * Math.pow(2, i) : delay));
    }
  }
  throw lastErr!;
}

// 使用：withRetry(() => articleApi.getList(1), { maxRetries: 2 })
```

## 五、加载状态管理

```typescript
// hooks/useAsyncState.ts
import { useState, useCallback } from 'react';

interface State<T> { data: T | null; isLoading: boolean; error: string | null; }

export function useAsyncState<T>(init: T | null = null) {
  const [state, setState] = useState<State<T>>({ data: init, isLoading: false, error: null });

  const execute = useCallback(async (fn: () => Promise<T>) => {
    setState((p) => ({ ...p, isLoading: true, error: null }));
    try {
      const data = await fn();
      setState({ data, isLoading: false, error: null });
      return data;
    } catch (e) {
      setState((p) => ({ ...p, isLoading: false, error: (e as Error).message }));
      throw e;
    }
  }, []);

  return { ...state, execute };
}
```

## 六、请求取消

```typescript
import { useState, useEffect, useRef } from 'react';
import { useDebounce } from '@/hooks/useDebounce';

interface Result { id: string; title: string; }

export default function SearchWithCancel() {
  const [keyword, setKeyword] = useState('');
  const [results, setResults] = useState<Result[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const debounced = useDebounce(keyword, 300);
  const abortRef = useRef<AbortController | null>(null);

  useEffect(() => {
    if (!debounced.trim()) { setResults([]); return; }
    abortRef.current?.abort();
    const ctrl = new AbortController();
    abortRef.current = ctrl;

    (async () => {
      setIsLoading(true);
      try {
        const res = await fetch(`https://api.example.com/search?q=${encodeURIComponent(debounced)}`, { signal: ctrl.signal });
        if (!res.ok) throw new Error('搜索失败');
        setResults(await res.json());
      } catch (e) {
        if ((e as Error).name !== 'AbortError') console.error(e);
      } finally { setIsLoading(false); }
    })();

    return () => ctrl.abort();
  }, [debounced]);

  // TextInput + FlatList 渲染
}
```

## 常见误区

1. **忘记取消请求**：组件卸载后更新状态导致内存泄漏
2. **缓存不过期**：永久缓存导致数据不一致
3. **重试无限次**：应限制最大次数
4. **混用 fetch 和 axios**：应统一网络层
5. **渲染函数中发请求**：应在 useEffect 或事件中

---

## 工程建议

1. **统一网络层**：封装 http 模块
2. **API 模块化**：按业务拆分 articleApi、userApi
3. **缓存分层**：配置长缓存，用户数据短缓存，实时不缓存
4. **环境配置**：环境变量区分 API 地址

---

## 小结

- fetch 轻量需手动封装，Axios 内置拦截器/超时/取消
- 请求拦截器附加 Token，响应拦截器统一处理错误
- AsyncStorage 实现缓存，需设计过期策略
- AbortController 取消请求避免泄漏
- withRetry 指数退避重试，不重试 4xx

import { FlatList, RefreshControl } from 'react-native';
import { cachedGet } from '@/utils/cachedRequest';

export default function CachedList() {
  const [data, setData] = useState<any[]>([]);
  const [refreshing, setRefreshing] = useState(false);

  const load = useCallback(async (force = false) => {
    const res = await cachedGet<{ articles: any[] }>('/articles', { ttl: 600000, forceRefresh: force });
    setData(res.articles);
    setRefreshing(false);
  }, []);

  useEffect(() => { load(); }, [load]);

  return (
    <FlatList data={data} keyExtractor={(i) => i.id}
      renderItem={({ item }) => <Text style={{ padding: 16 }}>{item.title}</Text>}
      refreshControl={<RefreshControl refreshing={refreshing} onRefresh={() => { setRefreshing(true); load(true); }} />} />
  );
}
