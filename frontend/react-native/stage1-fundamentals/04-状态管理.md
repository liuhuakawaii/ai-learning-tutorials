# 状态管理

## 场景引入

App 有登录状态需多页面共享，购物车需跨页同步，服务器数据需缓存刷新。`useState` 管理组件内部状态没问题，但跨组件共享时层层传递 props 会变乱麻。这节课学 Zustand 管客户端状态、TanStack Query 管服务端状态——社区最推荐的组合。

## 学习目标

- 理解 React Context 的局限
- 掌握 Zustand（create/selector/中间件）
- 使用 TanStack React Query 管理服务端状态
- 实现状态持久化

## 一、React Context 的局限

```typescript
import { createContext, useContext, useState, ReactNode } from 'react';

interface ThemeCtx { isDark: boolean; toggle: () => void; }
const ThemeContext = createContext<ThemeCtx | null>(null);

export function ThemeProvider({ children }: { children: ReactNode }) {
  const [isDark, setIsDark] = useState(false);
  return (
    <ThemeContext.Provider value={{ isDark, toggle: () => setIsDark(p => !p) }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error('useTheme 必须在 ThemeProvider 内使用');
  return ctx;
}
```

**问题**：Provider 的 value 变化时所有消费者重新渲染。适合低频全局配置，不适合高频更新或复杂逻辑。

```typescript
interface CounterState { count: number; increment: () => void; decrement: () => void; reset: () => void; }
```

### 2.2 异步操作

```typescript
// stores/user.ts
import { create } from 'zustand';

interface User { id: string; name: string; email: string; }

interface UserState {
  user: User | null; isLoading: boolean; error: string | null;
  login: (email: string, password: string) => Promise<void>;
  logout: () => void;
}

export const useUserStore = create<UserState>((set) => ({
  user: null, isLoading: false, error: null,
  login: async (email, password) => {
    set({ isLoading: true, error: null });
    try {
      const res = await fetch('https://api.example.com/login', {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password }),
      });
      if (!res.ok) throw new Error('登录失败');
      set({ user: await res.json(), isLoading: false });
    } catch (e) { set({ error: (e as Error).message, isLoading: false }); }
  },
  logout: () => set({ user: null, error: null }),
}));
```

### 2.3 持久化中间件

```typescript
// stores/todo.ts
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';
import AsyncStorage from '@react-native-async-storage/async-storage';

interface Todo { id: string; text: string; completed: boolean; }

interface TodoState {
  todos: Todo[];
  addTodo: (text: string) => void;
  toggleTodo: (id: string) => void;
  removeTodo: (id: string) => void;
}

export const useTodoStore = create<TodoState>()(
  persist(
    (set) => ({
      todos: [],
      addTodo: (text) => set((s) => ({ todos: [...s.todos, { id: Date.now().toString(), text, completed: false }] })),
      toggleTodo: (id) => set((s) => ({ todos: s.todos.map((t) => t.id === id ? { ...t, completed: !t.completed } : t) })),
      removeTodo: (id) => set((s) => ({ todos: s.todos.filter((t) => t.id !== id) })),
    }),
    { name: 'todo-storage', storage: createJSONStorage(() => AsyncStorage) }
  )
);
```

### 2.4 购物车（派生值）

```typescript
import { create } from 'zustand';

interface CartItem { id: string; name: string; price: number; quantity: number; }

interface CartState {
  items: CartItem[];
  addItem: (item: Omit<CartItem, 'quantity'>) => void;
  removeItem: (id: string) => void;
  updateQuantity: (id: string, qty: number) => void;
  clearCart: () => void;
  totalPrice: () => number;
  totalCount: () => number;
}

export const useCartStore = create<CartState>((set, get) => ({
  items: [],
  addItem: (item) => set((s) => {
    const existing = s.items.find((i) => i.id === item.id);
    if (existing) return { items: s.items.map((i) => i.id === item.id ? { ...i, quantity: i.quantity + 1 } : i) };
    return { items: [...s.items, { ...item, quantity: 1 }] };
  }),
  removeItem: (id) => set((s) => ({ items: s.items.filter((i) => i.id !== id) })),
  updateQuantity: (id, qty) => set((s) => ({
    items: qty <= 0 ? s.items.filter((i) => i.id !== id) : s.items.map((i) => i.id === id ? { ...i, quantity: qty } : i),
  })),
  clearCart: () => set({ items: [] }),
  totalPrice: () => get().items.reduce((s, i) => s + i.price * i.quantity, 0),
  totalCount: () => get().items.reduce((s, i) => s + i.quantity, 0),
}));
```

```typescript
// hooks/useArticles.ts
import { useInfiniteQuery, useQuery } from '@tanstack/react-query';

interface Article { id: string; title: string; summary: string; }
interface ArticlesResponse { articles: Article[]; page: number; hasMore: boolean; }

async function fetchArticles(page: number): Promise<ArticlesResponse> {
  const res = await fetch(`https://api.example.com/articles?page=${page}&limit=20`);
  if (!res.ok) throw new Error('获取失败');
  return res.json();
}

export function useInfiniteArticles() {
  return useInfiniteQuery({
    queryKey: ['articles', 'infinite'],
    queryFn: ({ pageParam = 1 }) => fetchArticles(pageParam),
    getNextPageParam: (last) => last.hasMore ? last.page + 1 : undefined,
    initialPageParam: 1,
  });
}
```

### 3.3 无限滚动

```typescript
import { View, Text, FlatList, ActivityIndicator, StyleSheet } from 'react-native';
import { useInfiniteArticles } from '@/hooks/useArticles';

export default function NewsFeed() {
  const { data, fetchNextPage, hasNextPage, isFetchingNextPage, isLoading, isError, error, refetch, isRefetching } = useInfiniteArticles();

  if (isLoading) return <ActivityIndicator style={styles.center} size="large" />;
  if (isError) return (
    <View style={styles.center}>
      <Text style={{ color: '#FF3B30' }}>{error.message}</Text>
      <Text style={{ color: '#007AFF', marginTop: 8 }} onPress={() => refetch()}>重试</Text>
    </View>
  );

  const articles = data?.pages.flatMap((p) => p.articles) ?? [];

  return (
    <FlatList data={articles} keyExtractor={(item) => item.id}
      renderItem={({ item }) => (
        <View style={styles.item}>
          <Text style={styles.title}>{item.title}</Text>
          <Text style={styles.summary} numberOfLines={2}>{item.summary}</Text>
        </View>
      )}
      onEndReached={() => { if (hasNextPage && !isFetchingNextPage) fetchNextPage(); }}
      onEndReachedThreshold={0.5} refreshing={isRefetching} onRefresh={refetch}
      ListFooterComponent={isFetchingNextPage ? <ActivityIndicator style={{ padding: 16 }} /> : null}
    />
  );
}

const styles = StyleSheet.create({
  center: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  item: { padding: 16, borderBottomWidth: 1, borderBottomColor: '#f0f0f0' },
  title: { fontSize: 17, fontWeight: '600' },
  summary: { fontSize: 14, color: '#666', marginTop: 6 },
});
```

### 3.4 Mutation（乐观更新）

```typescript
import { useMutation, useQueryClient } from '@tanstack/react-query';

export function useToggleFavorite() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: ({ id, isFav }: { id: string; isFav: boolean }) =>
      fetch(`https://api.example.com/articles/${id}/favorite`, { method: isFav ? 'DELETE' : 'POST' }),
    onMutate: async ({ id }) => {
      await qc.cancelQueries({ queryKey: ['articles'] });
      qc.setQueriesData({ queryKey: ['articles'] }, (old: any) => {
        if (!old) return old;
        return { ...old, pages: old.pages.map((p: any) => ({
          ...p, articles: p.articles.map((a: any) => a.id === id ? { ...a, isFavorite: !a.isFavorite } : a),
        }))};
      });
    },
    onSettled: () => qc.invalidateQueries({ queryKey: ['articles'] }),
  });
}
```

## 四、状态分层

| 层级 | 工具 | 示例 |
|------|------|------|
| 组件本地 | useState | 表单输入、UI 状态 |
| 客户端全局 | Zustand | 认证、购物车、主题 |
| 服务端缓存 | React Query | 文章列表、用户信息 |

---

## 常见误区

1. **所有状态都放 Zustand**：局部状态用 useState
2. **Query 和 Store 职责混淆**：服务端数据用 React Query
3. **不用 selector**：`useStore(s => s)` 等于没用
4. **staleTime 默认 0**：每次挂载都重新请求
5. **Mutation 后不 invalidate**：页面显示旧数据

---

## 工程建议

1. **Store 按领域拆分**：useUserStore、useCartStore
2. **Query Key 数组形式**：`['articles', { page, category }]`
3. **离线优先**：React Query 支持 networkMode: 'offlineFirst'
4. **TypeScript**：Zustand 和 React Query 类型支持完善

---

## 小结

- Context 适合低频全局配置，不适合复杂状态
- Zustand：create 定义 store，useStore(selector) 使用，persist 持久化
- React Query：缓存、后台刷新、无限滚动、乐观更新
- 分层：本地 useState → 全局 Zustand → 服务端 React Query

### 练习一

```typescript
// stores/theme.ts
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { useColorScheme } from 'react-native';

type Mode = 'light' | 'dark' | 'system';
const colors = {
  light: { bg: '#F2F2F7', surface: '#FFF', text: '#000', primary: '#007AFF' },
  dark: { bg: '#000', surface: '#1C1C1E', text: '#FFF', primary: '#0A84FF' },
};

interface State { mode: Mode; setMode: (m: Mode) => void; getColors: (sys: 'light' | 'dark' | null) => typeof colors.light; }

export const useThemeStore = create<State>()(
  persist(
    (set, get) => ({
      mode: 'system',
      setMode: (m) => set({ mode: m }),
      getColors: (sys) => { const m = get().mode; return (m === 'system' ? (sys ?? 'light') : m) === 'dark' ? colors.dark : colors.light; },
    }),
    { name: 'theme', storage: createJSONStorage(() => AsyncStorage) }
  )
);

export function useThemeColors() { return useThemeStore((s) => s.getColors(useColorScheme())); }
```

### 练习三

```typescript
// hooks/useDebounce.ts
import { useState, useEffect } from 'react';
export function useDebounce<T>(v: T, d: number): T {
  const [dv, setDv] = useState(v);
  useEffect(() => { const t = setTimeout(() => setDv(v), d); return () => clearTimeout(t); }, [v, d]);
  return dv;
}

// hooks/useSearch.ts
import { useQuery } from '@tanstack/react-query';
export function useSearch(kw: string) {
  return useQuery({
    queryKey: ['search', kw.trim()],
    queryFn: async () => { const r = await fetch(`https://api.example.com/search?q=${encodeURIComponent(kw)}`); if (!r.ok) throw new Error('失败'); return r.json(); },
    enabled: kw.trim().length >= 2, staleTime: 120000,
  });
}
```

`enabled` 避免空关键词请求，`useDebounce` 减少请求频率。
