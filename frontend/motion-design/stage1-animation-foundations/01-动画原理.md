# 动画原理：帧、缓动与时间

## 从一个实验开始

打开 Chrome DevTools 的 Performance 面板，录制下面两段动画：

```html
<div id="box-a" style="width:60px;height:60px;background:#3b82f6;border-radius:8px;position:absolute;top:40px;left:0;"></div>
<div id="box-b" style="width:60px;height:60px;background:#ef4444;border-radius:8px;position:absolute;top:120px;left:0;"></div>

<script>
function animateA() {
  const el = document.getElementById('box-a')
  let x = 0
  function step() {
    x += 2
    el.style.left = x + 'px'         // 触发 Layout → Paint → Composite
    if (x < 600) requestAnimationFrame(step)
  }
  requestAnimationFrame(step)
}

function animateB() {
  const el = document.getElementById('box-b')
  let x = 0
  function step() {
    x += 2
    el.style.transform = `translateX(${x}px)`  // 只触发 Composite
    if (x < 600) requestAnimationFrame(step)
  }
  requestAnimationFrame(step)
}

animateA()
animateB()
</script>
```

录制完成后看火焰图：蓝色方块的每帧耗时明显高于红色方块。两者做的事情"一样"——让方块向右移动，但性能差距可以达到 5-10 倍。

这个实验揭示了动画原理的第一课：**不是所有"看起来一样的动画"在底层代价相同**。

## 渲染管线：每帧 16.67ms 里发生了什么

浏览器把代码变成屏幕像素，每一帧经历五个阶段：

```
JavaScript → Style → Layout → Paint → Composite
  执行逻辑    计算样式  计算位置   绘制像素  GPU合成
```

每个阶段的代价不同：

| 属性示例 | 触发阶段 | 代价 |
|----------|----------|------|
| `el.style.left = '100px'` | Layout → Paint → Composite | 最高 |
| `el.style.backgroundColor = 'red'` | Paint → Composite | 中等 |
| `el.style.transform = 'translateX(100px)'` | Composite only | 最低 |
| `el.style.opacity = '0.5'` | Composite only | 最低 |

动画应该尽量只操作 `transform` 和 `opacity`。这不是"最佳实践"，而是物理约束——60fps 下每帧只有 16.67ms，触发 Layout 意味着浏览器要重新计算所有受影响元素的几何位置。

## 帧率感知实验

人眼对不同类型的运动敏感度不同。在页面上同时运行三个动画，观察哪个最先让你觉得"卡"：

```html
<div id="pos" style="width:40px;height:40px;background:#3b82f6;border-radius:50%;position:absolute;top:40px;left:0;transition:none;"></div>
<div id="opa" style="width:40px;height:40px;background:#10b981;border-radius:50%;position:absolute;top:100px;left:200px;"></div>
<div id="col" style="width:40px;height:40px;background:#f59e0b;border-radius:50%;position:absolute;top:160px;left:200px;"></div>
<script>
// 位置变化 — 最敏感
let t0 = performance.now()
function moveLoop(ts) {
  const p = ((ts - t0) % 2000) / 2000
  document.getElementById('pos').style.transform = `translateX(${Math.sin(p * Math.PI * 2) * 200 + 200}px)`
  requestAnimationFrame(moveLoop)
}
requestAnimationFrame(moveLoop)

// 透明度变化 — 中等敏感
function opaLoop(ts) {
  const p = ((ts - t0) % 2000) / 2000
  document.getElementById('opa').style.opacity = String(0.3 + 0.7 * Math.abs(Math.sin(p * Math.PI)))
  requestAnimationFrame(opaLoop)
}
requestAnimationFrame(opaLoop)

// 颜色变化 — 最不敏感
function colLoop(ts) {
  const p = ((ts - t0) % 2000) / 2000
  const r = Math.round(245 * Math.abs(Math.sin(p * Math.PI)))
  document.getElementById('col').style.backgroundColor = `rgb(${r},158,11)`
  requestAnimationFrame(colLoop)
}
requestAnimationFrame(colLoop)
</script>
```

位移动画在低帧率下最容易暴露问题。这就是为什么性能优化的重点放在位移相关的属性上。

## 缓动函数：为什么 linear 看起来不对

现实世界里没有东西会瞬间起步、匀速运动、瞬间停下。缓动函数是对惯性的数学模拟。

```html
<canvas id="easing-canvas" width="500" height="200"></canvas>
<script>
const canvas = document.getElementById('easing-canvas')
const ctx = canvas.getContext('2d')
const W = canvas.width, H = canvas.height, P = 40

const easings = {
  'linear':    { fn: t => t,                              color: '#94a3b8' },
  'ease-in':   { fn: t => t * t,                           color: '#ef4444' },
  'ease-out':  { fn: t => t * (2 - t),                     color: '#22c55e' },
  'ease-in-out': { fn: t => t < 0.5 ? 2*t*t : -1+(4-2*t)*t, color: '#3b82f6' },
}

// 坐标轴
ctx.strokeStyle = '#334155'; ctx.lineWidth = 1
ctx.beginPath(); ctx.moveTo(P, 10); ctx.lineTo(P, H-P); ctx.lineTo(W-P, H-P); ctx.stroke()

// 曲线
Object.entries(easings).forEach(([name, { fn, color }]) => {
  ctx.beginPath(); ctx.strokeStyle = color; ctx.lineWidth = 2.5
  for (let i = 0; i <= 100; i++) {
    const t = i / 100, y = fn(t)
    const x = P + t * (W - 2*P), yPos = (H - P) - y * (H - 2*P - 10)
    i === 0 ? ctx.moveTo(x, yPos) : ctx.lineTo(x, yPos)
  }
  ctx.stroke()
})

// 图例
let ly = 15
Object.entries(easings).forEach(([name, { color }]) => {
  ctx.fillStyle = color; ctx.fillRect(W - P - 110, ly, 14, 14)
  ctx.fillStyle = '#e2e8f0'; ctx.font = '12px sans-serif'; ctx.fillText(name, W - P - 90, ly + 12)
  ly += 22
})
</script>
```

横轴是时间（0→1），纵轴是进度（0→1）。linear 是直线，ease-in 开头慢（加速起步），ease-out 结尾慢（减速停下），ease-in-out 两头慢中间快。

核心认知：**缓动函数是 `f(t) → [0,1]` 的映射**，输入线性时间进度，输出实际进度。

## requestAnimationFrame vs setTimeout

```javascript
// 差：不与刷新率同步，后台不暂停
setInterval(() => { el.style.transform = `translateX(${x}px)` }, 16)

// 好：与浏览器刷新率同步，后台标签页自动暂停
requestAnimationFrame(() => { el.style.transform = `translateX(${x}px)` })
```

rAF 的三个优势：
1. 自动匹配屏幕刷新率（60Hz = 每 16.67ms 调用一次）
2. 后台标签页自动暂停
3. 浏览器可以批量优化多个回调

## 时间轴思维：编排多个动画

复杂动画是多个元素在不同时间点的组合。用 Promise 链实现串行编排：

```javascript
function animate(el, props, duration, easing = 'ease-out') {
  return new Promise(resolve => {
    el.style.transition = Object.keys(props)
      .map(p => `${p} ${duration}ms ${easing}`).join(', ')
    el.addEventListener('transitionend', () => resolve(), { once: true })
    Object.entries(props).forEach(([p, v]) => el.style[p] = v)
  })
}

async function pageEntrance() {
  const header = document.querySelector('.header')
  const sidebar = document.querySelector('.sidebar')
  const content = document.querySelector('.content')

  // 阶段 1：并行入场
  await Promise.all([
    animate(header, { opacity: '1', transform: 'translateY(0)' }, 400),
    animate(sidebar, { opacity: '1', transform: 'translateX(0)' }, 400),
  ])
  // 阶段 2：内容入场
  await animate(content, { opacity: '1' }, 300)
}
```

交错入场（stagger）是列表动画的常见模式：

```javascript
async function staggerFadeIn(items, delay = 80) {
  items.forEach(el => { el.style.opacity = '0'; el.style.transform = 'translateY(16px)' })
  const promises = Array.from(items).map((el, i) =>
    new Promise(resolve => setTimeout(() => {
      animate(el, { opacity: '1', transform: 'translateY(0)' }, 300).then(resolve)
    }, i * delay))
  )
  await Promise.all(promises)
}
```

## 时长选择

| 场景 | 时长 | 缓动 |
|------|------|------|
| 按钮悬停 | 150-200ms | ease-out |
| Tab 切换 | 200-300ms | ease-in-out |
| 元素入场 | 300-400ms | ease-out |
| 元素出场 | 200-300ms | ease-in（比入场快） |
| 页面转场 | 300-500ms | ease-in-out |

出场比入场快——用户已经知道它是什么，快速消失不打断注意力。

### 练习一

```javascript
;(function createFPSMonitor() {
  const el = document.createElement('div')
  Object.assign(el.style, {
    position: 'fixed', top: '8px', left: '8px', background: 'rgba(0,0,0,0.75)',
    color: '#0f0', padding: '4px 10px', font: '12px monospace', borderRadius: '4px', zIndex: '99999',
  })
  document.body.appendChild(el)

  let last = performance.now(), frames = 0
  function tick(now) {
    frames++
    if (now - last >= 1000) {
      const fps = Math.round(frames * 1000 / (now - last))
      el.textContent = `FPS: ${fps}`
      el.style.color = fps >= 55 ? '#0f0' : fps >= 30 ? '#ff0' : '#f00'
      frames = 0; last = now
    }
    requestAnimationFrame(tick)
  }
  requestAnimationFrame(tick)
})()
```

### 练习二

```html
<canvas id="race" width="600" height="180"></canvas>
<script>
const c = document.getElementById('race'), ctx = c.getContext('2d')
const W = c.width, dur = 2000
const easings = [
  { name: 'linear',    fn: t => t,                              color: '#94a3b8' },
  { name: 'ease-in',   fn: t => t * t,                           color: '#ef4444' },
  { name: 'ease-out',  fn: t => t * (2 - t),                     color: '#22c55e' },
  { name: 'ease-in-out', fn: t => t < 0.5 ? 2*t*t : -1+(4-2*t)*t, color: '#3b82f6' },
]
const t0 = performance.now()
function draw(ts) {
  ctx.clearRect(0, 0, W, 180)
  const elapsed = (ts - t0) % (dur * 2)
  const t = elapsed < dur ? elapsed / dur : 2 - elapsed / dur
  easings.forEach((e, i) => {
    const y = 25 + i * 40, x = 30 + e.fn(t) * (W - 80)
    ctx.beginPath(); ctx.arc(x, y, 12, 0, Math.PI * 2)
    ctx.fillStyle = e.color; ctx.fill()
    ctx.fillStyle = '#e2e8f0'; ctx.font = '12px sans-serif'
    ctx.fillText(e.name, 8, y + 4)
  })
  requestAnimationFrame(draw)
}
requestAnimationFrame(draw)
</script>
```

### 练习三

```javascript
function animate(el, props, duration, easing = 'ease-out') {
  return new Promise(resolve => {
    el.style.transition = Object.keys(props)
      .map(p => `${p} ${duration}ms ${easing}`).join(', ')
    const onEnd = () => { el.removeEventListener('transitionend', onEnd); resolve() }
    el.addEventListener('transitionend', onEnd)
    Object.entries(props).forEach(([p, v]) => el.style[p] = v)
  })
}

function wait(ms) { return new Promise(r => setTimeout(r, ms)) }

async function entrance() {
  const title = document.querySelector('.title')
  const subtitle = document.querySelector('.subtitle')
  title.style.opacity = '0'; title.style.transform = 'translateY(30px)'
  subtitle.style.opacity = '0'

  await Promise.all([
    animate(title, { opacity: '1' }, 400),
    animate(title, { transform: 'translateY(0)' }, 400),
  ])
  await wait(500)
  await animate(subtitle, { opacity: '1' }, 300)
}
```
