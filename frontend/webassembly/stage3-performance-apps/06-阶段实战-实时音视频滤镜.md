# 阶段实战：实时音视频滤镜

## 做什么

用 WASM + SIMD 实现实时视频滤镜：摄像头捕获 → 逐帧处理 → Canvas 渲染。支持灰度、模糊、边缘检测等滤镜，目标 30fps。

## 技术栈

- WebRTC（摄像头捕获）
- Canvas 2D（渲染）
- WASM + SIMD（滤镜处理）
- Web Worker（避免阻塞主线程）

## Canvas 渲染循环

```javascript
function renderLoop(video, canvas, wasmModule) {
  const ctx = canvas.getContext('2d')
  canvas.width = video.videoWidth
  canvas.height = video.videoHeight

  function frame() {
    ctx.drawImage(video, 0, 0)
    const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height)

    // WASM 处理
    const ptr = wasmModule._malloc(imageData.data.length)
    new Uint8Array(wasmModule.HEAP8.buffer, ptr, imageData.data.length).set(imageData.data)
    wasmModule._apply_filter(ptr, imageData.data.length, canvas.width, canvas.height)
    imageData.data.set(new Uint8Array(wasmModule.HEAP8.buffer, ptr, imageData.data.length))
    wasmModule._free(ptr)

    ctx.putImageData(imageData, 0, 0)
    requestAnimationFrame(frame)
  }
  frame()
}
```

## WASM 滤镜库

```c
// video_filter.c
#include <emscripten/emscripten.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

EMSCRIPTEN_KEEPALIVE
void grayscale(unsigned char* pixels, int length) {
    for (int i = 0; i < length; i += 4) {
        unsigned char gray = (unsigned char)(0.299*pixels[i] + 0.587*pixels[i+1] + 0.114*pixels[i+2]);
        pixels[i] = gray; pixels[i+1] = gray; pixels[i+2] = gray;
    }
}

EMSCRIPTEN_KEEPALIVE
void edge_detect(unsigned char* pixels, int width, int height) {
    int len = width * height * 4;
    unsigned char* gray = (unsigned char*)malloc(len);
    unsigned char* result = (unsigned char*)malloc(len);

    // 灰度化
    for (int i = 0; i < len; i += 4) {
        unsigned char g = (unsigned char)(0.299*pixels[i]+0.587*pixels[i+1]+0.114*pixels[i+2]);
        gray[i] = gray[i+1] = gray[i+2] = g; gray[i+3] = 255;
    }

    // Sobel
    int gxm[] = {-1,0,1,-2,0,2,-1,0,1};
    int gym[] = {-1,-2,-1,0,0,0,1,2,1};
    for (int y = 1; y < height-1; y++) {
        for (int x = 1; x < width-1; x++) {
            float gx = 0, gy = 0;
            for (int ky = -1; ky <= 1; ky++) {
                for (int kx = -1; kx <= 1; kx++) {
                    int val = gray[((y+ky)*width+(x+kx))*4];
                    gx += val * gxm[(ky+1)*3+(kx+1)];
                    gy += val * gym[(ky+1)*3+(kx+1)];
                }
            }
            unsigned char edge = (unsigned char)fminf(sqrtf(gx*gx+gy*gy), 255);
            int idx = (y*width+x)*4;
            result[idx] = result[idx+1] = result[idx+2] = edge;
            result[idx+3] = 255;
        }
    }
    memcpy(pixels, result, len);
    free(gray); free(result);
}
```

## Web Worker 处理

把 WASM 处理放到 Worker 中，避免阻塞主线程：

```javascript
// worker.js
importScripts('video_filter.js')

let wasmModule

self.onmessage = async function(e) {
  if (e.data.type === 'init') {
    wasmModule = await createVideoFilterModule()
    self.postMessage({ type: 'ready' })
  } else if (e.data.type === 'process') {
    const { pixels, width, height, filter } = e.data
    const ptr = wasmModule._malloc(pixels.length)
    new Uint8Array(wasmModule.HEAP8.buffer, ptr, pixels.length).set(pixels)

    switch (filter) {
      case 'grayscale': wasmModule._grayscale(ptr, pixels.length); break
      case 'edge': wasmModule._edge_detect(ptr, width, height); break
    }

    const result = new Uint8Array(wasmModule.HEAP8.buffer, ptr, pixels.length)
    self.postMessage({ type: 'result', pixels: result }, [result.buffer])
    wasmModule._free(ptr)
  }
}
```
