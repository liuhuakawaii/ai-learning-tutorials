# 04 - 游戏引擎

## 场景引入

你在做 2D 平台跳跃游戏，场景有 500 个实体（玩家、敌人、子弹、道具）。JavaScript 用对象数组管理，每帧遍历做更新和碰撞检测，低端手机只能跑到 30fps。ECS 架构把数据按类型连续存储，配合 WASM 线性内存，可以实现极高的缓存命中率。

## 学习目标

- 理解 ECS 架构设计思想
- 掌握用 Rust 实现 ECS 核心
- 学会固定时间步长的游戏循环
- 掌握碰撞检测和状态机

## ECS 架构

ECS 用组合替代继承：

```
Entity 0: [Position, Velocity, Sprite, PlayerControl, Health]
Entity 1: [Position, Velocity, Sprite, EnemyAI, Health]
Entity 2: [Position, Velocity, Sprite, Bullet]
```

- **Entity** — 只是一个 ID（u32）
- **Component** — 纯数据，按类型连续存储
- **System** — 纯逻辑，遍历具有特定组件组合的实体

## Rust ECS 核心

```rust
use wasm_bindgen::prelude::*;
use std::collections::HashMap;

pub type Entity = u32;
pub const TAG_PLAYER: u32 = 1 << 0;
pub const TAG_ENEMY: u32 = 1 << 1;
pub const TAG_BULLET: u32 = 1 << 2;
pub const TAG_SOLID: u32 = 1 << 4;

#[wasm_bindgen]
pub struct World {
    next_entity: Entity,
    entities: Vec<Entity>,
    pos_x: Vec<f32>, pos_y: Vec<f32>,
    vel_x: Vec<f32>, vel_y: Vec<f32>,
    collider_w: Vec<f32>, collider_h: Vec<f32>,
    sprite_id: Vec<u16>,
    tag_masks: Vec<u32>,
    entity_index: HashMap<Entity, usize>,
}

#[wasm_bindgen]
impl World {
    #[wasm_bindgen(constructor)]
    pub fn new() -> Self {
        World {
            next_entity: 0, entities: vec![],
            pos_x: vec![], pos_y: vec![], vel_x: vec![], vel_y: vec![],
            collider_w: vec![], collider_h: vec![], sprite_id: vec![],
            tag_masks: vec![], entity_index: HashMap::new(),
        }
    }

    pub fn spawn(&mut self, x: f32, y: f32, sprite: u16, tag: u32) -> Entity {
        let e = self.next_entity; self.next_entity += 1;
        let idx = self.entities.len();
        self.entities.push(e); self.pos_x.push(x); self.pos_y.push(y);
        self.vel_x.push(0.0); self.vel_y.push(0.0);
        self.collider_w.push(16.0); self.collider_h.push(16.0);
        self.sprite_id.push(sprite); self.tag_masks.push(tag);
        self.entity_index.insert(e, idx); e
    }

    // 交换移除：保持数组紧凑
    pub fn despawn(&mut self, entity: Entity) {
        if let Some(&idx) = self.entity_index.get(&entity) {
            let last = self.entities.len() - 1;
            if idx != last {
                let last_e = self.entities[last];
                self.pos_x[idx]=self.pos_x[last]; self.pos_y[idx]=self.pos_y[last];
                self.vel_x[idx]=self.vel_x[last]; self.vel_y[idx]=self.vel_y[last];
                self.collider_w[idx]=self.collider_w[last]; self.collider_h[idx]=self.collider_h[last];
                self.sprite_id[idx]=self.sprite_id[last]; self.tag_masks[idx]=self.tag_masks[last];
                self.entity_index.insert(last_e, idx);
            }
            self.entities.pop(); self.pos_x.pop(); self.pos_y.pop();
            self.vel_x.pop(); self.vel_y.pop(); self.collider_w.pop();
            self.collider_h.pop(); self.sprite_id.pop(); self.tag_masks.pop();
            self.entity_index.remove(&entity);
        }
    }

    pub fn entity_count(&self) -> usize { self.entities.len() }
    pub fn pos_x_ptr(&self) -> *const f32 { self.pos_x.as_ptr() }
    pub fn pos_y_ptr(&self) -> *const f32 { self.pos_y.as_ptr() }
    pub fn sprite_id_ptr(&self) -> *const u16 { self.sprite_id.as_ptr() }

    pub fn update_positions(&mut self, dt: f32) {
        for i in 0..self.entities.len() {
            self.pos_x[i] += self.vel_x[i] * dt;
            self.pos_y[i] += self.vel_y[i] * dt;
        }
    }

    pub fn set_velocity(&mut self, entity: Entity, vx: f32, vy: f32) {
        if let Some(&idx) = self.entity_index.get(&entity) { self.vel_x[idx] = vx; self.vel_y[idx] = vy; }
    }

    pub fn apply_gravity(&mut self, gravity: f32, dt: f32) {
        for i in 0..self.entities.len() {
            if self.tag_masks[i] & TAG_SOLID == 0 { self.vel_y[i] += gravity * dt; }
        }
    }
}
```

## 固定时间步长游戏循环

```typescript
export class GameLoop {
  private accumulator = 0;
  private lastTime = 0;
  private running = false;
  private readonly FIXED_DT = 1.0 / 60.0;

  constructor(private world: any, private input: any) {}

  start() { this.running = true; this.lastTime = performance.now(); requestAnimationFrame(this.tick.bind(this)); }
  stop() { this.running = false; }

  private tick(now: number) {
    if (!this.running) return;
    let frameTime = (now - this.lastTime) / 1000;
    if (frameTime > 0.25) frameTime = 0.25;
    this.lastTime = now;
    this.accumulator += frameTime;
    while (this.accumulator >= this.FIXED_DT) {
      this.fixedUpdate(this.FIXED_DT);
      this.accumulator -= this.FIXED_DT;
    }
    this.render(this.accumulator / this.FIXED_DT);
    requestAnimationFrame(this.tick.bind(this));
  }

  private fixedUpdate(dt: number) {
    this.world.apply_gravity(980.0, dt);
    this.world.update_positions(dt);
    this.checkCollisions();
  }
  private checkCollisions() { /* 碰撞检测 */ }
  private render(alpha: number) { /* 插值渲染 */ }
}
```

## AABB 碰撞检测

```rust
use wasm_bindgen::prelude::*;

#[wasm_bindgen]
pub struct CollisionResult { pub entity_a: u32, pub entity_b: u32, pub overlap_x: f32, pub overlap_y: f32 }

#[wasm_bindgen]
pub fn detect_collisions(
    pos_x: &[f32], pos_y: &[f32], col_w: &[f32], col_h: &[f32],
    tag_masks: &[u32], mask_a: u32, mask_b: u32,
) -> Vec<CollisionResult> {
    let mut results = vec![];
    for i in 0..pos_x.len() {
        if tag_masks[i] & mask_a == 0 { continue; }
        for j in (i+1)..pos_x.len() {
            if tag_masks[j] & mask_b == 0 { continue; }
            let ox = (col_w[i] + col_w[j]) / 2.0 - (pos_x[i] - pos_x[j]).abs();
            let oy = (col_h[i] + col_h[j]) / 2.0 - (pos_y[i] - pos_y[j]).abs();
            if ox > 0.0 && oy > 0.0 {
                results.push(CollisionResult { entity_a: i as u32, entity_b: j as u32, overlap_x: ox, overlap_y: oy });
            }
        }
    }
    results
}
```

## 状态机

```rust
use wasm_bindgen::prelude::*;

#[wasm_bindgen]
#[derive(Clone, Copy, PartialEq)]
pub enum PlayerState { Idle, Running, Jumping, Falling, Attacking, Hurt }

#[wasm_bindgen]
pub struct StateMachine {
    current: PlayerState, state_time: f32,
    sprite_starts: Vec<u16>, frame_counts: Vec<u8>, frame_durations: Vec<f32>,
}

#[wasm_bindgen]
impl StateMachine {
    #[wasm_bindgen(constructor)]
    pub fn new() -> Self {
        StateMachine {
            current: PlayerState::Idle, state_time: 0.0,
            sprite_starts: vec![0, 4, 10, 13, 16, 21],
            frame_counts: vec![4, 6, 3, 3, 5, 2],
            frame_durations: vec![0.15, 0.10, 0.08, 0.08, 0.06, 0.20],
        }
    }
    pub fn transition(&mut self, new_state: PlayerState) {
        if self.current != new_state { self.current = new_state; self.state_time = 0.0; }
    }
    pub fn update(&mut self, dt: f32) -> u16 {
        self.state_time += dt;
        let idx = self.current as usize;
        let frame = (self.state_time / self.frame_durations[idx]) as u16;
        self.sprite_starts[idx] + frame.min(self.frame_counts[idx] as u16 - 1)
    }
    pub fn current_state(&self) -> PlayerState { self.current }
}
```

## 常见误区

1. **用 JS 对象存储组件** — 对象数组内存不连续，ECS 的核心优势就是数据连续存储
2. **用可变时间步长做物理** — 可变 dt 导致物理不稳定（弹球穿墙），必须用固定步长
3. **每帧创建新实体** — 应从对象池获取，避免 GC 压力
4. **在 WASM 中播放音频** — 音频必须用 Web Audio API，WASM 只负责游戏逻辑

## 工程建议

1. **SoA 布局是 ECS 灵魂** — 同类数据连续存储，缓存预取效率最高
2. **交换移除保持紧凑** — 删除实体用最后一个元素填补空洞
3. **空间哈希加速碰撞** — 500 实体暴力检测 O(n²)，空间哈希降到 O(n·k)
4. **固定步长 + 插值渲染** — 物理用固定 dt，渲染用插值消除抖动

## 小结

ECS 的关键不是"组件"概念，而是**按类型连续存储**带来的缓存友好性——这正是 WASM 线性内存的优势。固定时间步长保证物理稳定，AABB 碰撞配合空间哈希加速，状态机管理动画逻辑。

### 练习一

```rust
const MAX: usize = 16;

pub struct Quadtree {
    bounds: (f32, f32, f32, f32),
    entities: Vec<usize>,
    children: Option<Box<[Quadtree; 4]>>,
}

impl Quadtree {
    pub fn new(x: f32, y: f32, w: f32, h: f32) -> Self {
        Quadtree { bounds: (x, y, w, h), entities: vec![], children: None }
    }
    pub fn insert(&mut self, idx: usize, px: f32, py: f32) {
        let (x, y, w, h) = self.bounds;
        if px < x || px >= x + w || py < y || py >= y + h { return; }
        if let Some(ref mut ch) = self.children {
            for c in ch.iter_mut() { c.insert(idx, px, py); } return;
        }
        self.entities.push(idx);
        if self.entities.len() > MAX { self.subdivide(); }
    }
    fn subdivide(&mut self) {
        let (x, y, w, h) = self.bounds;
        let (hw, hh) = (w / 2.0, h / 2.0);
        self.children = Some(Box::new([
            Quadtree::new(x, y, hw, hh), Quadtree::new(x+hw, y, hw, hh),
            Quadtree::new(x, y+hh, hw, hh), Quadtree::new(x+hw, y+hh, hw, hh),
        ]));
        // 重新插入已有实体
    }
}
```

**要点**：四叉树适合不均匀分布，空间哈希适合均匀分布。

```rust
pub struct JumpController {
    jump_force: f32, coyote_time: f32, jump_buffer: f32,
    coyote_timer: f32, buffer_timer: f32,
}

impl JumpController {
    pub fn update(&mut self, dt: f32, jump_pressed: bool, grounded: bool) -> f32 {
        if grounded { self.coyote_timer = self.coyote_time; } else { self.coyote_timer -= dt; }
        if jump_pressed { self.buffer_timer = self.jump_buffer; } else { self.buffer_timer -= dt; }
        if self.buffer_timer > 0.0 && self.coyote_timer > 0.0 {
            self.coyote_timer = 0.0; self.buffer_timer = 0.0; return self.jump_force;
        }
        0.0
    }
}
```

**要点**：土狼时间和跳跃缓冲让玩家感觉"游戏理解了我的意图"。

### 练习三

```rust
pub struct ParticleSystem {
    px: Vec<f32>, py: Vec<f32>, vx: Vec<f32>, vy: Vec<f32>,
    life: Vec<f32>, count: usize,
}

impl ParticleSystem {
    pub fn emit(&mut self, x: f32, y: f32, n: usize, speed: f32) {
        for _ in 0..n {
            let a = js_sys::Math::random() as f32 * std::f32::consts::TAU;
            self.px.push(x); self.py.push(y);
            self.vx.push(a.cos() * speed); self.vy.push(a.sin() * speed);
            self.life.push(1.0); self.count += 1;
        }
    }
    pub fn update(&mut self, dt: f32) {
        let mut i = 0;
        while i < self.count {
            self.life[i] -= dt;
            if self.life[i] <= 0.0 {
                self.count -= 1; let l = self.count;
                self.px[i]=self.px[l]; self.py[i]=self.py[l];
                self.vx[i]=self.vx[l]; self.vy[i]=self.vy[l]; self.life[i]=self.life[l];
                continue;
            }
            self.vy[i] += 200.0 * dt;
            self.px[i] += self.vx[i] * dt; self.py[i] += self.vy[i] * dt;
            i += 1;
        }
    }
}
```

**要点**：粒子系统用独立 SoA 存储，交换移除保证活跃粒子连续排列。
