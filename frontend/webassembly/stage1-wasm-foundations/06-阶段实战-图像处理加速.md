# 06 - 阶段实战：图像处理加速

## 场景引入

你正在开发在线图片编辑器。用户上传 4000×3000 的照片，点击"灰度"按钮。纯 JavaScript 逐像素处理需要 180ms，用户能感到延迟。同时应用灰度+模糊+锐化三个滤镜，延迟累积到 500ms 以上。

这是 WASM 最典型的用武之地：逐像素的数学运算密集且独立，非常适合用 WASM 加速。

## 学习目标

- 理解图像滤镜算法原理（灰度、模糊、锐化）
- 用 C 实现像素处理并编译为 WASM
- 对比 JavaScript 和 WASM 的处理性能
- 掌握 Canvas 与 WASM 的数据交互
- 学会进行基准测试和性能分析

## 图像滤镜算法

**灰度**：`gray = 0.299×R + 0.587×G + 0.114×B`（基于人眼敏感度的加权公式）

**高斯模糊 3×3 核**：
```
1/16 × [1 2 1]
        [2 4 2]
        [1 2 1]
```

**锐化拉普拉斯算子**：
```
[ 0 -1  0]
[-1  5 -1]
[ 0 -1  0]
```

## C 实现：像素处理库

```c
// image_filter.c
#include <emscripten/emscripten.h>
#include <stdlib.h>
#include <string.h>

EMSCRIPTEN_KEEPALIVE
void grayscale(unsigned char* pixels, int length) {
    for (int i = 0; i < length; i += 4) {
        unsigned char gray = (unsigned char)(0.299*pixels[i] + 0.587*pixels[i+1] + 0.114*pixels[i+2]);
        pixels[i] = gray;
        pixels[i+1] = gray;
        pixels[i+2] = gray;
    }
}

static void apply_kernel(unsigned char* src, unsigned char* dst,
    int width, int height, const float* kernel, int ks) {
    int half = ks / 2;
    for (int y = 0; y < height; y++) {
        for (int x = 0; x < width; x++) {
            for (int c = 0; c < 3; c++) {
                float sum = 0;
                for (int ky = -half; ky <= half; ky++) {
                    for (int kx = -half; kx <= half; kx++) {
                        int px = x+kx < 0 ? 0 : (x+kx >= width ? width-1 : x+kx);
                        int py = y+ky < 0 ? 0 : (y+ky >= height ? height-1 : y+ky);
                        sum += src[(py*width+px)*4+c] * kernel[(ky+half)*ks+(kx+half)];
                    }
                }
                if (sum < 0) sum = 0;
                if (sum > 255) sum = 255;
                dst[(y*width+x)*4+c] = (unsigned char)sum;
            }
            dst[(y*width+x)*4+3] = src[(y*width+x)*4+3];
        }
    }
}

EMSCRIPTEN_KEEPALIVE
void gaussian_blur(unsigned char* pixels, int width, int height) {
    int len = width * height * 4;
    unsigned char* tmp = (unsigned char*)malloc(len);
    float kernel[] = {1./16,2./16,1./16, 2./16,4./16,2./16, 1./16,2./16,1./16};
    apply_kernel(pixels, tmp, width, height, kernel, 3);
    memcpy(pixels, tmp, len);
    free(tmp);
}

EMSCRIPTEN_KEEPALIVE
void sharpen(unsigned char* pixels, int width, int height) {
    int len = width * height * 4;
    unsigned char* tmp = (unsigned char*)malloc(len);
    float kernel[] = {0,-1,0, -1,5,-1, 0,-1,0};
    apply_kernel(pixels, tmp, width, height, kernel, 3);
    memcpy(pixels, tmp, len);
    free(tmp);
}
```

## JavaScript 对比实现

```javascript
function grayscaleJS(imageData) {
  const d = imageData.data;
  for (let i = 0; i < d.length; i += 4) {
    const g = 0.299*d[i] + 0.587*d[i+1] + 0.114*d[i+2];
    d[i] = d[i+1] = d[i+2] = g;
  }
}

function convolutionJS(imageData, kernel, ks) {
  const { data, width, height } = imageData;
  const result = new Uint8ClampedArray(data.length);
  const half = Math.floor(ks / 2);
  for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
      for (let c = 0; c < 3; c++) {
        let sum = 0;
        for (let ky = -half; ky <= half; ky++) {
          for (let kx = -half; kx <= half; kx++) {
            const px = Math.min(Math.max(x+kx,0), width-1);
            const py = Math.min(Math.max(y+ky,0), height-1);
            sum += data[(py*width+px)*4+c] * kernel[(ky+half)*ks+(kx+half)];
          }
        }
        result[(y*width+x)*4+c] = Math.min(Math.max(sum,0), 255);
      }
      result[(y*width+x)*4+3] = data[(y*width+x)*4+3];
    }
  }
  imageData.data.set(result);
}
```

<body>

    let originalData, wasmModule;

    async function init() {
      wasmModule = await createImageFilterModule();
      const img = new Image();
      img.onload = () => {
        const canvas = document.getElementById('canvas');
        canvas.width = img.width;
        canvas.height = img.height;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0);
        originalData = ctx.getImageData(0, 0, img.width, img.height);

      img.src = 'test.jpg';
    }

    function benchmark(name, fn, iterations = 5) {
      const times = [];
      for (let i = 0; i < iterations; i++) {
        const start = performance.now();
        fn();
        times.push(performance.now() - start);
      }
      return `${name}: ${(times.reduce((a,b)=>a+b)/iterations).toFixed(2)}ms`;
    }

    document.getElementById('btnGrayJS').onclick = () => {
      const img = new ImageData(new Uint8ClampedArray(originalData.data), originalData.width, originalData.height);
      const r = benchmark('灰度 JS', () => grayscaleJS(img));
      document.getElementById('canvas').getContext('2d').putImageData(img, 0, 0);
      document.getElementById('results').textContent = r;
    };

    document.getElementById('btnGrayWasm').onclick = () => {
      const img = new ImageData(new Uint8ClampedArray(originalData.data), originalData.width, originalData.height);

        const ptr = wasmModule._malloc(img.data.length);

        wasmModule._free(ptr);
      });
      document.getElementById('canvas').getContext('2d').putImageData(img, 0, 0);
      document.getElementById('results').textContent = r;
    };

## 性能基准测试

| 操作 | 图片尺寸 | JavaScript | WASM | 加速比 |
|------|----------|-----------|------|--------|
| 灰度 | 1920×1080 | 12ms | 3ms | 4× |
| 灰度 | 4000×3000 | 65ms | 15ms | 4.3× |
| 高斯模糊 | 1920×1080 | 120ms | 25ms | 4.8× |
| 高斯模糊 | 4000×3000 | 550ms | 110ms | 5× |

## 常见误区

1. **"WASM 加速适用于所有 Web 任务"** — 不是，像素处理适合，DOM 操作不适合。
2. **"数据传递开销可忽略"** — 对小图片可以，4000×3000（48MB）则需要几毫秒。
3. **"边界处理不重要"** — 边界像素处理方式直接影响输出质量。

## 工程建议

- 用 Web Worker 运行 WASM，避免阻塞主线程
- 预分配 WASM 内存，重复使用，避免频繁 malloc/free
- 如果 GPU 可用，WebGL 着色器比 WASM 更快
- 大图片可以分块处理，先显示低分辨率预览

## 小结

- 图像滤镜是 WASM 最典型的应用场景，计算密集且数据独立
- C 实现的像素处理编译为 WASM 可获得 4-5 倍加速
- Canvas 的 `getImageData`/`putImageData` 配合 WASM 内存实现数据交互
- 基准测试需多次运行取平均值

## 参考答案

### 练习一

**思路**：逐像素读 R/G/B，加 amount 后 clamp 到 0-255。

```c
#include <emscripten/emscripten.h>

EMSCRIPTEN_KEEPALIVE
void brightness(unsigned char* pixels, int length, int amount) {
    for (int i = 0; i < length; i += 4) {
        int r = pixels[i] + amount;
        pixels[i] = r < 0 ? 0 : (r > 255 ? 255 : r);
        int g = pixels[i+1] + amount;
        pixels[i+1] = g < 0 ? 0 : (g > 255 ? 255 : g);
        int b = pixels[i+2] + amount;
        pixels[i+2] = b < 0 ? 0 : (b > 255 ? 255 : b);
    }
}
```

```javascript
function brightnessJS(imageData, amount) {
  const d = imageData.data;
  for (let i = 0; i < d.length; i += 4) {
    d[i] = Math.min(Math.max(d[i]+amount, 0), 255);
    d[i+1] = Math.min(Math.max(d[i+1]+amount, 0), 255);
    d[i+2] = Math.min(Math.max(d[i+2]+amount, 0), 255);
  }
}
// 1920×1080: JS ~8ms, WASM ~2ms
```

**要点**：C 中 `int` 运算比 JS 的 `number` 更直接。Alpha 通道不参与亮度调整。

### 练习二

**思路**：先灰度化，再分别计算水平和垂直梯度 Gx、Gy，取模 √(Gx²+Gy²)。

```c
#include <emscripten/emscripten.h>
#include <math.h>
#include <stdlib.h>
#include <string.h>

EMSCRIPTEN_KEEPALIVE
void sobel_edge(unsigned char* pixels, int width, int height) {
    int len = width * height * 4;
    unsigned char* gray = (unsigned char*)malloc(len);
    unsigned char* result = (unsigned char*)malloc(len);

    for (int i = 0; i < len; i += 4) {
        unsigned char g = (unsigned char)(0.299*pixels[i]+0.587*pixels[i+1]+0.114*pixels[i+2]);
        gray[i] = gray[i+1] = gray[i+2] = g;
        gray[i+3] = 255;
    }

    int gxm[] = {-1,0,1,-2,0,2,-1,0,1};
    int gym[] = {-1,-2,-1,0,0,0,1,2,1};

    for (int y = 0; y < height; y++) {
        for (int x = 0; x < width; x++) {
            float gx = 0, gy = 0;
            for (int ky = -1; ky <= 1; ky++) {
                for (int kx = -1; kx <= 1; kx++) {
                    int px = x+kx<0?0:(x+kx>=width?width-1:x+kx);
                    int py = y+ky<0?0:(y+ky>=height?height-1:y+ky);
                    int val = gray[(py*width+px)*4];
                    int ki = (ky+1)*3+(kx+1);
                    gx += val*gxm[ki]; gy += val*gym[ki];
                }
            }
            unsigned char edge = (unsigned char)sqrtf(gx*gx+gy*gy);
            if (edge > 255) edge = 255;
            int idx = (y*width+x)*4;
            result[idx] = result[idx+1] = result[idx+2] = edge;
            result[idx+3] = 255;
        }
    }

    memcpy(pixels, result, len);
    free(gray); free(result);
}
```

**要点**：先灰度化再做 Sobel，避免对三通道分别计算。边界用最近像素填充。
