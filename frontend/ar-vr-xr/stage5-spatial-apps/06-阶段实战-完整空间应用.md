```typescript
// src/core/AppArchitecture.ts

interface SpatialApp {
  platform: PlatformDetector;
  sessionManager: XRSessionManager;
  mode: 'flat' | 'ar' | 'vr';
}

class SpatialWorkstation {
  private platform: PlatformDetector;
  private sessionManager: XRSessionManager;
  private sceneGraph: SceneGraph;
  private networkManager: SyncManager;
  private aiEngine: SceneAnalyzer;
  private uiLayer: SpatialUILayer;

  constructor(canvas: HTMLCanvasElement) {
    this.platform = new PlatformDetector();
    this.sessionManager = new XRSessionManager(canvas);
    this.sceneGraph = new SceneGraph();
    this.networkManager = new SyncManager();
    this.aiEngine = new SceneAnalyzer();
    this.uiLayer = new SpatialUILayer();

    this.init();
  }

  private async init(): Promise<void> {
    await this.platform.detect();
    await this.setupModeSpecific();
    this.startRenderLoop();
  }

  private async setupModeSpecific(): Promise<void> {
    if (this.platform.supportsAR) {
      this.setupARMode();
    }
    if (this.platform.supportsVR) {
      this.setupVRMode();
    }
    this.setupFlatMode();
  }

  private setupARMode(): void {
    this.sessionManager.onSessionStart('immersive-ar', () => {
      this.sceneGraph.enablePlaneDetection();
      this.sceneGraph.enableHitTesting();
      this.uiLayer.showAROverlay();
    });
  }

  private setupVRMode(): void {
    this.sessionManager.onSessionStart('immersive-vr', () => {
      this.sceneGraph.enableTeleport();
      this.sceneGraph.enableGrab();
      this.uiLayer.showVRPanel();
    });
  }

  private setupFlatMode(): void {
    this.sceneGraph.enableOrbitControls();
    this.uiLayer.showDesktopUI();
  }

  private startRenderLoop(): void {
    const renderer = this.sessionManager.getRenderer();
    const animate = (time: number, frame?: XRFrame) => {
      this.sceneGraph.update(time, frame);
      this.networkManager.poll();
      renderer.render(this.sceneGraph.scene, this.sceneGraph.camera);
    };
    this.sessionManager.setAnimationLoop(animate);
  }
}
```

核心设计原则：
- 模式无关：核心逻辑不依赖 AR/VR/Flat 模式
- 渐进增强：每个模式添加额外能力，而非替代
- 单一数据源：场景状态统一管理，不同模式共享同一份数据

## 2. AR/VR 模式切换

用户应该可以随时在 AR 和 VR 之间切换，场景内容保持同步：

```typescript
// src/core/ModeSwitcher.ts

import * as THREE from 'three';

interface ModeState {
  current: 'flat' | 'ar' | 'vr';
  sceneContent: Map<string, SpatialObject>;
  cameraPose: THREE.Pose;
}

class ModeSwitcher {
  private state: ModeState;
  private sessionManager: XRSessionManager;
  private contentPreserver: ContentPreserver;

  constructor(sessionManager: XRSessionManager) {
    this.sessionManager = sessionManager;
    this.contentPreserver = new ContentPreserver();
    this.state = {
      current: 'flat',
      sceneContent: new Map(),
      cameraPose: { position: new THREE.Vector3(), quaternion: new THREE.Quaternion() }
    };
  }

  async switchTo(mode: 'ar' | 'vr'): Promise<void> {
    this.preserveCurrentContent();

    if (this.state.current !== 'flat') {
      await this.sessionManager.endSession();
    }

    const sessionType = mode === 'ar' ? 'immersive-ar' : 'immersive-vr';
    const features: XRSessionInit = {
      requiredFeatures: ['local-floor'],
      optionalFeatures: mode === 'ar'
        ? ['hit-test', 'plane-detection', 'anchors', 'mesh-detection']
        : ['hand-tracking', 'bounded-floor']
    };

    await this.sessionManager.startSession(sessionType, features);
    this.restoreContent(mode);
    this.state.current = mode;
  }

  private preserveCurrentContent(): void {
    this.state.sceneContent = this.contentPreserver.serialize();
  }

  private restoreContent(mode: 'ar' | 'vr'): void {
    const content = this.state.sceneContent;
    content.forEach((obj, id) => {
      if (mode === 'ar') {
        this.placeInAR(obj);
      } else {
        this.placeInVR(obj);
      }
    });
  }

  private placeInAR(obj: SpatialObject): void {
    const surface = this.findNearestSurface(obj.position);
    if (surface) {
      obj.position.copy(surface.point);
      obj.position.y += 0.01;
    }
  }

  private placeInVR(obj: SpatialObject): void {
    const roomCenter = new THREE.Vector3(0, 1.2, -1.5);
    const offset = obj.position.clone().sub(roomCenter);
    obj.position.copy(roomCenter).add(offset.multiplyScalar(0.5));
  }

  private findNearestSurface(pos: THREE.Vector3): THREE.Vector3 | null {
    const raycaster = new THREE.Raycaster(
      pos.clone().add(new THREE.Vector3(0, 2, 0)),
      new THREE.Vector3(0, -1, 0)
    );
    const intersects = raycaster.intersectObjects(
      this.sessionManager.getPlanes()
    );
    return intersects.length > 0 ? intersects[0].point : null;
  }
}
```

## 3. 内容管理系统

空间应用需要一个统一的内容管理系统来处理 3D 资产的加载、缓存和空间布局：

```typescript
// src/core/ContentManager.ts

import * as THREE from 'three';
import { GLTFLoader } from 'three/addons/loaders/GLTFLoader.js';
import { DRACOLoader } from 'three/addons/loaders/DRACOLoader.js';

interface ContentItem {
  id: string;
  type: 'model' | 'screen' | 'widget';
  source: string;
  transform: THREE.Matrix4;
  metadata: Record<string, unknown>;
}

class ContentManager {
  private items: Map<string, ContentItem> = new Map();
  private gltfLoader: GLTFLoader;
  private dracoLoader: DRACOLoader;
  private objectPool: Map<string, THREE.Object3D[]> = new Map();

  constructor() {
    this.dracoLoader = new DRACOLoader();
    this.dracoLoader.setDecoderPath('/draco/');
    this.gltfLoader = new GLTFLoader();
    this.gltfLoader.setDRACOLoader(this.dracoLoader);
  }

  async addModel(
    id: string,
    url: string,
    position: THREE.Vector3,
    options?: { scale?: number; animate?: boolean }
  ): Promise<THREE.Object3D> {
    const cached = this.objectPool.get(url);
    let object: THREE.Object3D;

    if (cached && cached.length > 0) {
      object = cached[0].clone();
    } else {
      const gltf = await this.gltfLoader.loadAsync(url);
      object = gltf.scene;
      if (!this.objectPool.has(url)) {
        this.objectPool.set(url, []);
      }
      this.objectPool.get(url)!.push(object.clone());
    }

    object.position.copy(position);
    if (options?.scale) object.scale.setScalar(options.scale);

    const item: ContentItem = {
      id,
      type: 'model',
      source: url,
      transform: object.matrixWorld.clone(),
      metadata: { options }
    };
    this.items.set(id, item);

    return object;
  }

  addVirtualScreen(
    id: string,
    position: THREE.Vector3,
    size: { width: number; height: number },
    contentUrl?: string
  ): THREE.Group {
    const group = new THREE.Group();

    const geometry = new THREE.PlaneGeometry(size.width, size.height);
    const material = new THREE.MeshStandardMaterial({
      color: 0x1a1a2e,
      roughness: 0.1,
      metalness: 0.8,
      side: THREE.DoubleSide
    });
    const screen = new THREE.Mesh(geometry, material);
    screen.name = 'screen-frame';
    group.add(screen);

    if (contentUrl) {
      const textureLoader = new THREE.TextureLoader();
      textureLoader.load(contentUrl, (texture) => {
        const displayGeometry = new THREE.PlaneGeometry(
          size.width * 0.95,
          size.height * 0.95
        );
        const displayMaterial = new THREE.MeshBasicMaterial({
          map: texture,
          side: THREE.DoubleSide
        });
        const display = new THREE.Mesh(displayGeometry, displayMaterial);
        display.position.z = 0.01;
        display.name = 'screen-content';
        group.add(display);
      });
    }

    group.position.copy(position);
    group.lookAt(new THREE.Vector3(0, position.y, 0));

    this.items.set(id, {
      id,
      type: 'screen',
      source: contentUrl || '',
      transform: group.matrixWorld.clone(),
      metadata: { size }
    });

    return group;
  }

  removeItem(id: string): void {
    this.items.delete(id);
  }

  getItems(): ContentItem[] {
    return Array.from(this.items.values());
  }

  serialize(): string {
    return JSON.stringify(Array.from(this.items.values()));
  }

  deserialize(data: string): void {
    const items: ContentItem[] = JSON.parse(data);
    this.items.clear();
    items.forEach(item => this.items.set(item.id, item));
  }
}
```

## 4. 多人协作同步

实现基于 WebSocket 的多人实时同步：

```typescript
// src/network/CollaborationSync.ts

interface SyncMessage {
  type: 'join' | 'leave' | 'update' | 'cursor' | 'chat';
  userId: string;
  timestamp: number;
  payload: unknown;
}

interface UserState {
  id: string;
  name: string;
  avatar: THREE.Object3D;
  headPose: { position: THREE.Vector3; quaternion: THREE.Quaternion };
  handPoses: { left: THREE.Matrix4; right: THREE.Matrix4 } | null;
  cursorPosition: THREE.Vector3;
}

class CollaborationSync {
  private ws: WebSocket | null = null;
  private users: Map<string, UserState> = new Map();
  private localUserId: string;
  private scene: THREE.Scene;
  private updateThrottle = 16;
  private lastUpdate = 0;

  constructor(scene: THREE.Scene) {
    this.scene = scene;
    this.localUserId = crypto.randomUUID();
  }

  async connect(roomId: string, serverUrl: string): Promise<void> {
    return new Promise((resolve, reject) => {
      this.ws = new WebSocket(`${serverUrl}/room/${roomId}`);

      this.ws.onopen = () => {
        this.send({ type: 'join', userId: this.localUserId, timestamp: Date.now(), payload: {} });
        resolve();
      };

      this.ws.onmessage = (event) => {
        const msg = JSON.parse(event.data) as SyncMessage;
        this.handleMessage(msg);
      };

      this.ws.onerror = (error) => reject(error);
    });
  }

  private handleMessage(msg: SyncMessage): void {
    if (msg.userId === this.localUserId) return;

    switch (msg.type) {
      case 'join':
        this.addRemoteUser(msg.userId, msg.payload as { name: string });
        break;
      case 'leave':
        this.removeRemoteUser(msg.userId);
        break;
      case 'update':
        this.updateRemoteUser(msg.userId, msg.payload as Partial<UserState>);
        break;
      case 'cursor':
        this.updateRemoteCursor(msg.userId, msg.payload as { position: THREE.Vector3 });
        break;
    }
  }

  updateLocalState(headPose: { position: THREE.Vector3; quaternion: THREE.Quaternion }): void {
    const now = Date.now();
    if (now - this.lastUpdate < this.updateThrottle) return;
    this.lastUpdate = now;

    this.send({
      type: 'update',
      userId: this.localUserId,
      timestamp: now,
      payload: { headPose }
    });
  }

  private addRemoteUser(userId: string, info: { name: string }): void {
    const avatar = this.createAvatar(info.name);
    this.scene.add(avatar);
    this.users.set(userId, {
      id: userId,
      name: info.name,
      avatar,
      headPose: { position: new THREE.Vector3(), quaternion: new THREE.Quaternion() },
      handPoses: null,
      cursorPosition: new THREE.Vector3()
    });
  }

  private removeRemoteUser(userId: string): void {
    const user = this.users.get(userId);
    if (user) {
      this.scene.remove(user.avatar);
      this.users.delete(userId);
    }
  }

  private updateRemoteUser(userId: string, data: Partial<UserState>): void {
    const user = this.users.get(userId);
    if (!user) return;

    if (data.headPose) {
      user.avatar.position.copy(data.headPose.position);
      user.avatar.quaternion.copy(data.headPose.quaternion);
    }
  }

  private createAvatar(name: string): THREE.Group {
    const group = new THREE.Group();

    const headGeo = new THREE.SphereGeometry(0.12, 16, 16);
    const headMat = new THREE.MeshStandardMaterial({
      color: new THREE.Color().setHSL(Math.random(), 0.7, 0.5)
    });
    const head = new THREE.Mesh(headGeo, headMat);
    head.position.y = 1.7;
    group.add(head);

    const bodyGeo = new THREE.CylinderGeometry(0.15, 0.12, 0.6, 8);
    const body = new THREE.Mesh(bodyGeo, headMat);
    body.position.y = 1.3;
    group.add(body);

    return group;
  }

  private send(msg: SyncMessage): void {
    if (this.ws?.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify(msg));
    }
  }

  disconnect(): void {
    this.send({ type: 'leave', userId: this.localUserId, timestamp: Date.now(), payload: {} });
    this.ws?.close();
    this.users.forEach(user => this.scene.remove(user.avatar));
    this.users.clear();
  }
}
```

## 5. AI 环境理解

集成 AI 能力进行环境理解和自动内容布局：

```typescript
// src/ai/SpatialAI.ts

interface SceneAnalysis {
  surfaces: SurfaceInfo[];
  objects: DetectedObject[];
  suggestedLayout: LayoutSuggestion[];
  lightingEstimate: LightingInfo;
}

interface SurfaceInfo {
  id: string;
  type: 'floor' | 'wall' | 'table' | 'ceiling';
  bounds: THREE.Box3;
  normal: THREE.Vector3;
  area: number;
}

interface DetectedObject {
  label: string;
  confidence: number;
  bounds: THREE.Box3;
  position: THREE.Vector3;
}

interface LayoutSuggestion {
  contentType: 'screen' | 'model' | 'widget';
  position: THREE.Vector3;
  orientation: THREE.Quaternion;
  reason: string;
}

class SpatialAI {
  private scene: THREE.Scene;
  private analysisCache: SceneAnalysis | null = null;

  constructor(scene: THREE.Scene) {
    this.scene = scene;
  }

  async analyzeEnvironment(
    planes: XRPlane[],
    meshes: XRMesh[]
  ): Promise<SceneAnalysis> {
    const surfaces = this.classifySurfaces(planes);
    const objects = await this.detectObjects(meshes);
    const lighting = this.estimateLighting();
    const layout = this.generateLayout(surfaces, objects);

    this.analysisCache = {
      surfaces,
      objects,
      suggestedLayout: layout,
      lightingEstimate: lighting
    };

    return this.analysisCache;
  }

  private classifySurfaces(planes: XRPlane[]): SurfaceInfo[] {
    return planes.map((plane, index) => {
      const geometry = plane.polygon;
      const normal = this.calculateNormal(geometry);
      const type = this.classifySurfaceType(normal, plane.semanticLabel);
      const bounds = this.calculateBounds(geometry);

      return {
        id: `surface-${index}`,
        type,
        bounds,
        normal: new THREE.Vector3(normal.x, normal.y, normal.z),
        area: this.calculateArea(geometry)
      };
    });
  }

  private classifySurfaceType(
    normal: DOMPointReadOnly,
    semanticLabel?: string
  ): SurfaceInfo['type'] {
    if (semanticLabel) {
      if (semanticLabel.includes('floor')) return 'floor';
      if (semanticLabel.includes('wall')) return 'wall';
      if (semanticLabel.includes('table') || semanticLabel.includes('desk')) return 'table';
      if (semanticLabel.includes('ceiling')) return 'ceiling';
    }

    const ny = normal.y;
    if (ny > 0.8) return 'floor';
    if (ny < -0.8) return 'ceiling';
    if (Math.abs(normal.z) > 0.8 || Math.abs(normal.x) > 0.8) return 'wall';
    return 'table';
  }

  private generateLayout(
    surfaces: SurfaceInfo[],
    objects: DetectedObject[]
  ): LayoutSuggestion[] {
    const suggestions: LayoutSuggestion[] = [];

    const tables = surfaces.filter(s => s.type === 'table');
    const walls = surfaces.filter(s => s.type === 'wall');
    const floor = surfaces.find(s => s.type === 'floor');

    if (tables.length > 0) {
      const table = tables[0];
      suggestions.push({
        contentType: 'model',
        position: new THREE.Vector3(
          table.bounds.getCenter(new THREE.Vector3()).x,
          table.bounds.max.y + 0.02,
          table.bounds.getCenter(new THREE.Vector3()).z
        ),
        orientation: new THREE.Quaternion(),
        reason: '桌面是放置 3D 模型的理想位置'
      });
    }

    if (walls.length > 0) {
      const wall = walls[0];
      const wallCenter = wall.bounds.getCenter(new THREE.Vector3());
      suggestions.push({
        contentType: 'screen',
        position: new THREE.Vector3(wallCenter.x, wallCenter.y, wallCenter.z),
        orientation: new THREE.Quaternion().setFromUnitVectors(
          new THREE.Vector3(0, 0, 1),
          wall.normal
        ),
        reason: '墙面适合挂载虚拟屏幕'
      });
    }

    if (floor) {
      const floorCenter = floor.bounds.getCenter(new THREE.Vector3());
      suggestions.push({
        contentType: 'widget',
        position: new THREE.Vector3(floorCenter.x, floorCenter.y + 0.01, floorCenter.z),
        orientation: new THREE.Quaternion(),
        reason: '地面可放置交互式 widget'
      });
    }

    return suggestions;
  }

  private calculateNormal(polygon: DOMPointReadOnly[]): DOMPointReadOnly {
    if (polygon.length < 3) return { x: 0, y: 1, z: 0, w: 1 } as DOMPointReadOnly;
    const a = polygon[0];
    const b = polygon[1];
    const c = polygon[2];
    const ab = { x: b.x - a.x, y: b.y - a.y, z: b.z - a.z };
    const ac = { x: c.x - a.x, y: c.y - a.y, z: c.z - a.z };
    return {
      x: ab.y * ac.z - ab.z * ac.y,
      y: ab.z * ac.x - ab.x * ac.z,
      z: ab.x * ac.y - ab.y * ac.x,
      w: 1
    } as DOMPointReadOnly;
  }

  private calculateBounds(polygon: DOMPointReadOnly[]): THREE.Box3 {
    const box = new THREE.Box3();
    polygon.forEach(p => box.expandByPoint(new THREE.Vector3(p.x, p.y, p.z)));
    return box;
  }

  private calculateArea(polygon: DOMPointReadOnly[]): number {
    if (polygon.length < 3) return 0;
    let area = 0;
    for (let i = 1; i < polygon.length - 1; i++) {
      const v1 = new THREE.Vector3().subVectors(
        new THREE.Vector3(polygon[i].x, polygon[i].y, polygon[i].z),
        new THREE.Vector3(polygon[0].x, polygon[0].y, polygon[0].z)
      );
      const v2 = new THREE.Vector3().subVectors(
        new THREE.Vector3(polygon[i + 1].x, polygon[i + 1].y, polygon[i + 1].z),
        new THREE.Vector3(polygon[0].x, polygon[0].y, polygon[0].z)
      );
      area += v1.cross(v2).length() / 2;
    }
    return area;
  }

  private async detectObjects(meshes: XRMesh[]): Promise<DetectedObject[]> {
    return meshes.map((mesh, index) => ({
      label: mesh.semanticLabel || `object-${index}`,
      confidence: 0.8,
      bounds: this.calculateBounds(mesh.vertices),
      position: new THREE.Vector3()
    }));
  }

  private estimateLighting(): LightingInfo {
    return {
      ambientIntensity: 0.4,
      directionalIntensity: 0.8,
      direction: new THREE.Vector3(0.5, 1, 0.3)
    };
  }

  getAnalysis(): SceneAnalysis | null {
    return this.analysisCache;
  }
}

interface LightingInfo {
  ambientIntensity: number;
  directionalIntensity: number;
  direction: THREE.Vector3;
}
```

## 6. 性能监控与自适应

空间应用需要严格的性能管理：

```typescript
// src/core/PerformanceManager.ts

interface PerformanceMetrics {
  fps: number;
  frameTime: number;
  gpuTime: number;
  drawCalls: number;
  triangles: number;
  textureMemory: number;
}

type QualityLevel = 'low' | 'medium' | 'high' | 'ultra';

class PerformanceManager {
  private metrics: PerformanceMetrics;
  private targetFPS = 90;
  private qualityLevel: QualityLevel = 'high';
  private frameTimeHistory: number[] = [];
  private maxHistory = 60;
  private adaptiveEnabled = true;

  constructor() {
    this.metrics = {
      fps: 0,
      frameTime: 0,
      gpuTime: 0,
      drawCalls: 0,
      triangles: 0,
      textureMemory: 0
    };
  }

  beginFrame(): number {
    return performance.now();
  }

  endFrame(startTime: number): void {
    const frameTime = performance.now() - startTime;
    this.frameTimeHistory.push(frameTime);
    if (this.frameTimeHistory.length > this.maxHistory) {
      this.frameTimeHistory.shift();
    }

    const avgFrameTime = this.frameTimeHistory.reduce((a, b) => a + b, 0) / this.frameTimeHistory.length;
    this.metrics.frameTime = avgFrameTime;
    this.metrics.fps = 1000 / avgFrameTime;

    if (this.adaptiveEnabled) {
      this.adaptQuality();
    }
  }

  private adaptQuality(): void {
    const avgFPS = this.metrics.fps;
    const target = this.targetFPS;

    if (avgFPS < target * 0.85 && this.qualityLevel !== 'low') {
      this.decreaseQuality();
    } else if (avgFPS > target * 1.1 && this.qualityLevel !== 'ultra') {
      this.increaseQuality();
    }
  }

  private decreaseQuality(): void {
    const levels: QualityLevel[] = ['ultra', 'high', 'medium', 'low'];
    const current = levels.indexOf(this.qualityLevel);
    if (current < levels.length - 1) {
      this.qualityLevel = levels[current + 1];
      this.applyQualitySettings();
    }
  }

  private increaseQuality(): void {
    const levels: QualityLevel[] = ['low', 'medium', 'high', 'ultra'];
    const current = levels.indexOf(this.qualityLevel);
    if (current < levels.length - 1) {
      this.qualityLevel = levels[current + 1];
      this.applyQualitySettings();
    }
  }

  private applyQualitySettings(): void {
    const settings: Record<QualityLevel, { pixelRatio: number; shadowMap: number; antialias: boolean }> = {
      low: { pixelRatio: 0.75, shadowMap: 512, antialias: false },
      medium: { pixelRatio: 1.0, shadowMap: 1024, antialias: false },
      high: { pixelRatio: 1.0, shadowMap: 2048, antialias: true },
      ultra: { pixelRatio: 1.5, shadowMap: 4096, antialias: true }
    };
    const s = settings[this.qualityLevel];
    console.log(`Quality adjusted to ${this.qualityLevel}:`, s);
  }

  getMetrics(): PerformanceMetrics {
    return { ...this.metrics };
  }

  getQualityLevel(): QualityLevel {
    return this.qualityLevel;
  }

  setTargetFPS(fps: number): void {
    this.targetFPS = fps;
  }
}
```

## 常见误区

1. **试图做一个"什么都能做"的应用** — 空间计算应用应该聚焦核心场景，不是所有功能都需要 AR 和 VR 双模式支持

2. **忽视模式切换时的内容丢失** — 从 AR 切到 VR 时，如果场景内容没有妥善保存和迁移，用户会丢失所有工作

3. **多人同步不做冲突处理** — 当两个用户同时操作同一个物体时，没有冲突解决机制会导致状态不一致

4. **AI 布局建议不考虑用户意图** — AI 应该是辅助工具，不是决策者。用户应该能覆盖 AI 的建议

5. **性能优化太晚** — 空间应用的性能问题不像 Web 页面可以"稍后优化"，低帧率在 XR 中会导致眩晕

6. **忽视降级策略** — 不是所有设备都支持 AR 和 VR，必须有 2D 模式的降级方案

7. **过度依赖网络** — 多人协作依赖网络是合理的，但单人功能不应该因为断网而不可用

## 工程建议

1. **从"最小可用体验"开始** — 先在 2D 模式下把核心功能跑通，再添加 AR/VR 模式

2. **场景状态与渲染分离** — 用一个统一的状态模型，渲染层只是状态的可视化表达

3. **内容管线标准化** — 所有 3D 资产统一用 glTF 格式，启用 Draco 压缩，控制模型大小

4. **网络协议用二进制** — 高频同步数据（头部姿态、手部位置）用 Float32Array 而非 JSON

5. **AI 推理异步化** — 环境分析是耗时操作，不要阻塞渲染线程，结果缓存后增量更新

6. **测试用 WebXR Emulator** — 没有实体设备时，用 Chrome 的 WebXR Emulator 扩展进行开发和测试

7. **性能预算写进 CI** — 用 puppeteer + WebXR polyfill 跑自动化性能测试，帧率低于阈值就报警

8. **逐步构建场景** — 不要一次性加载所有 3D 资产，用 LOD 和流式加载控制内存占用

## 小结

本课我们从头到尾构建了一个完整的空间协作工作站，涵盖了：

- **架构设计**：模式无关的核心架构，支持 AR/VR/Flat 三种模式
- **模式切换**：AR 和 VR 之间的无缝切换，场景内容保持同步
- **内容管理**：统一的 3D 资产管理系统，支持对象池和缓存
- **多人协作**：基于 WebSocket 的实时同步，包含虚拟形象和状态管理
- **AI 集成**：环境理解、表面分类、智能布局建议
- **性能管理**：自适应质量调整，确保帧率达标

空间计算开发的核心挑战不在于单个技术点，而在于如何把这些技术有机地组合成一个流畅的用户体验。从架构设计的第一天起，就要考虑模式切换、内容同步、性能约束和降级策略。

```typescript
import * as THREE from 'three';

interface WidgetData {
  id: string;
  type: 'calendar' | 'weather' | 'tasklist';
  data: Record<string, unknown>;
}

abstract class InteractiveWidget extends THREE.Group {
  protected data: WidgetData;
  protected panel: THREE.Mesh;
  protected isInteractive = true;

  constructor(widgetData: WidgetData) {
    super();
    this.data = widgetData;

    const geometry = new THREE.PlaneGeometry(0.3, 0.2);
    const material = new THREE.MeshStandardMaterial({
      color: 0x2a2a3e,
      roughness: 0.3,
      metalness: 0.5,
      side: THREE.DoubleSide
    });
    this.panel = new THREE.Mesh(geometry, material);
    this.add(this.panel);

    this.userData.type = 'interactive-widget';
    this.userData.widgetId = widgetData.id;
  }

  abstract render(): void;
  abstract handleInteraction(event: XRInteractionEvent): void;

  updateData(newData: Record<string, unknown>): void {
    this.data.data = { ...this.data.data, ...newData };
    this.render();
  }

  setFollowMode(enabled: boolean): void {
    this.userData.followUser = enabled;
  }

  setFixedSurface(surface: SurfaceInfo): void {
    this.position.copy(surface.bounds.getCenter(new THREE.Vector3()));
    this.position.y += 0.02;
    const normal = surface.normal;
    this.lookAt(this.position.clone().add(normal));
  }
}

class CalendarWidget extends InteractiveWidget {
  render(): void {
    const canvas = document.createElement('canvas');
    canvas.width = 512;
    canvas.height = 340;
    const ctx = canvas.getContext('2d')!;
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, 512, 340);
    ctx.fillStyle = '#ffffff';
    ctx.font = 'bold 24px sans-serif';
    ctx.fillText(this.data.data.title as string || 'Calendar', 20, 40);

    ctx.font = '16px sans-serif';
    days.forEach((day, i) => {
      const x = 20 + (i % 7) * 68;
      const y = 80 + Math.floor(i / 7) * 40;
      ctx.fillText(day, x, y);
    });

    const texture = new THREE.CanvasTexture(canvas);
    (this.panel.material as THREE.MeshBasicMaterial).map = texture;
    (this.panel.material as THREE.MeshBasicMaterial).needsUpdate = true;
  }

  handleInteraction(event: XRInteractionEvent): void {

    if (uv) {
      const col = Math.floor(uv.x * 7);
```

### 练习二

**思路**：用 IndexedDB 存储序列化的场景状态和 3D 资产 Blob，断网时从缓存读取，恢复网络后用时间戳合并冲突。

**答案**：

```typescript
// src/core/OfflineCache.ts

const DB_NAME = 'spatial-showcase';
const DB_VERSION = 1;

class OfflineCache {
  private db: IDBDatabase | null = null;

  async init(): Promise<void> {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(DB_NAME, DB_VERSION);
      request.onupgradeneeded = (event) => {
        const db = (event.target as IDBOpenDBRequest).result;
        if (!db.objectStoreNames.contains('assets')) {
          db.createObjectStore('assets', { keyPath: 'url' });
        }
        if (!db.objectStoreNames.contains('scene-state')) {
          db.createObjectStore('scene-state', { keyPath: 'id' });
        }
        if (!db.objectStoreNames.contains('pending-changes')) {
          db.createObjectStore('pending-changes', { keyPath: 'id', autoIncrement: true });
        }
      };
      request.onsuccess = (event) => {
        this.db = (event.target as IDBOpenDBRequest).result;
        resolve();
      };
      request.onerror = () => reject(request.error);
    });
  }

  async cacheAsset(url: string, blob: Blob): Promise<void> {
    const tx = this.db!.transaction('assets', 'readwrite');
    tx.objectStore('assets').put({ url, blob, timestamp: Date.now() });
    return new Promise((resolve) => { tx.oncomplete = () => resolve(); });
  }

  async getAsset(url: string): Promise<Blob | null> {
    const tx = this.db!.transaction('assets', 'readonly');
    const request = tx.objectStore('assets').get(url);
    return new Promise((resolve) => {
      request.onsuccess = () => resolve(request.result?.blob || null);
      request.onerror = () => resolve(null);
    });
  }

  async saveSceneState(state: string): Promise<void> {
    const tx = this.db!.transaction('scene-state', 'readwrite');
    tx.objectStore('scene-state').put({ id: 'current', state, timestamp: Date.now() });
    return new Promise((resolve) => { tx.oncomplete = () => resolve(); });
  }

  async loadSceneState(): Promise<string | null> {
    const tx = this.db!.transaction('scene-state', 'readonly');
    const request = tx.objectStore('scene-state').get('current');
    return new Promise((resolve) => {
      request.onsuccess = () => resolve(request.result?.state || null);
      request.onerror = () => resolve(null);
    });
  }

  async addPendingChange(change: unknown): Promise<void> {
    const tx = this.db!.transaction('pending-changes', 'readwrite');
    tx.objectStore('pending-changes').add({ change, timestamp: Date.now() });
    return new Promise((resolve) => { tx.oncomplete = () => resolve(); });
  }

  async getPendingChanges(): Promise<unknown[]> {
    const tx = this.db!.transaction('pending-changes', 'readonly');
    const request = tx.objectStore('pending-changes').getAll();
    return new Promise((resolve) => {
      request.onsuccess = () => resolve(request.result || []);
      request.onerror = () => resolve([]);
    });
  }

  async clearPendingChanges(): Promise<void> {
    const tx = this.db!.transaction('pending-changes', 'readwrite');
    tx.objectStore('pending-changes').clear();
    return new Promise((resolve) => { tx.oncomplete = () => resolve(); });
  }
}
```

**要点**：
- 用三个 object store 分别存储资产、场景状态和待同步变更
- 资产缓存用 URL 作为 key，直接存 Blob 减少序列化开销
- 待同步变更队列确保网络恢复后不丢失操作

### 练习三

**思路**：用 THREE.Sprite + canvas 渲染文字标注，通过 WebXR 的 hit-test 锚定到真实平面，标注作为同步对象加入多人同步系统。

**答案**：

```typescript
// src/ar/SpatialAnnotation.ts

import * as THREE from 'three';

interface Annotation {
  id: string;
  text: string;
  position: THREE.Vector3;
  normal: THREE.Vector3;
  creator: string;
  timestamp: number;
  object: THREE.Group;
}

class SpatialAnnotationSystem {
  private annotations: Map<string, Annotation> = new Map();
  private scene: THREE.Scene;
  private syncManager: CollaborationSync;

  constructor(scene: THREE.Scene, syncManager: CollaborationSync) {
    this.scene = scene;
    this.syncManager = syncManager;
  }

  async addAnnotation(
    hitResult: XRRayHitResult,
    text: string,
    userId: string
  ): Promise<Annotation> {
    const pose = hitResult.getPose(this.session.referenceSpace!)!;
    const position = new THREE.Vector3(
      pose.transform.position.x,
      pose.transform.position.y,
      pose.transform.position.z
    );
    const normal = new THREE.Vector3(0, 1, 0);

    const object = this.createAnnotationObject(text, position);
    this.scene.add(object);

    const annotation: Annotation = {
      id: crypto.randomUUID(),
      text,
      position,
      normal,
      creator: userId,
      timestamp: Date.now(),
      object
    };

    this.annotations.set(annotation.id, annotation);
    this.syncManager.updateLocalState({ position, quaternion: new THREE.Quaternion() });

    return annotation;
  }

  private createAnnotationObject(text: string, position: THREE.Vector3): THREE.Group {
    const group = new THREE.Group();

    const canvas = document.createElement('canvas');
    canvas.width = 256;
    canvas.height = 64;
    const ctx = canvas.getContext('2d')!;
    ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
    ctx.roundRect(0, 0, 256, 64, 8);
    ctx.fill();
    ctx.fillStyle = '#ffffff';
    ctx.font = '20px sans-serif';
    ctx.fillText(text, 16, 40);

    const texture = new THREE.CanvasTexture(canvas);
    const spriteMaterial = new THREE.SpriteMaterial({ map: texture });
    const sprite = new THREE.Sprite(spriteMaterial);
    sprite.scale.set(0.2, 0.05, 1);
    sprite.position.y = 0.05;
    group.add(sprite);

    const pinGeometry = new THREE.ConeGeometry(0.008, 0.04, 8);
    const pinMaterial = new THREE.MeshStandardMaterial({ color: 0xff4444 });
    const pin = new THREE.Mesh(pinGeometry, pinMaterial);
    pin.position.y = 0.02;
    group.add(pin);

    group.position.copy(position);
    group.userData.type = 'annotation';

    return group;
  }

  editAnnotation(id: string, newText: string): void {
    const annotation = this.annotations.get(id);
    if (!annotation) return;
    annotation.text = newText;
    this.scene.remove(annotation.object);
    annotation.object = this.createAnnotationObject(newText, annotation.position);
    this.scene.add(annotation.object);
  }

  removeAnnotation(id: string): void {
    const annotation = this.annotations.get(id);
    if (!annotation) return;
    this.scene.remove(annotation.object);
    this.annotations.delete(id);
  }

  getAnnotations(): Annotation[] {
    return Array.from(this.annotations.values());
  }

  private session: XRSession | null = null;
  setSession(session: XRSession): void {
    this.session = session;
  }
}
```

**要点**：
- 用 Canvas 纹理 + Sprite 渲染文字标注，保证始终面向用户
- 用小锥体作为"图钉"视觉提示，增强空间定位感
- 标注数据结构独立于渲染对象，方便序列化和同步
