# 阶段实战：AR 导航应用

## 场景引入

你站在大型商场入口，打开 AR 导航，屏幕上浮现出发光箭头指引方向。沿箭头走，它实时更新方向，经过扶梯时变成"上楼"图标，到达目标时变成"到达"标志。这就是本阶段的实战项目——综合运用 GPS、视觉定位、LiDAR、3D 渲染构建完整 AR 导航应用。

## 学习目标

- 整合多种追踪技术实现多源定位融合
- 实现路径规划与 AR 箭头导航
- 构建完整的导航 UI 系统

## 系统架构

```
UI 层 → 渲染层 → 追踪层 → 服务层 → 平台层
```

| 场景 | 主定位源 | 辅助源 | 精度 |
|------|----------|--------|------|
| 室外行走 | GPS | IMU | 5-10m |
| 室内行走 | IMU + 视觉里程计 | 地图匹配 | 1-3m |
| 目标到达 | 图像标记 / LiDAR | - | <0.5m |

## 核心实现

### 定位融合管理器

```javascript
class PositionFusionManager {
  constructor() {
    this.sources = new Map();
    this.fusedPosition = { latitude: 0, longitude: 0, altitude: 0 };
    this.kalman = { x: null, y: null, P: 1, Q: 0.01, R: 5 };
  }

  registerSource(name, weight) {
    this.sources.set(name, { position: null, lastUpdate: 0, weight, accuracy: Infinity });
  }

  updateSource(name, position, accuracy) {
    const s = this.sources.get(name);
    if (!s) return;
    s.position = position; s.lastUpdate = performance.now(); s.accuracy = accuracy;
    this.fusePositions();
  }

  fusePositions() {
    const now = performance.now();
    let totalW = 0, lat = 0, lon = 0, alt = 0;
    for (const [, s] of this.sources) {
      if (!s.position || now - s.lastUpdate > 5000) continue;
      const w = s.weight * (1 / s.accuracy);
      lat += s.position.latitude * w; lon += s.position.longitude * w;
      alt += (s.position.altitude || 0) * w; totalW += w;
    }
    if (totalW === 0) return;
    const raw = { latitude: lat/totalW, longitude: lon/totalW, altitude: alt/totalW };
    const k = this.kalman;
    if (k.x === null) { k.x = raw.latitude; k.y = raw.longitude; }
    else {
      k.P += k.Q; const gain = k.P / (k.P + k.R);
      k.x += gain * (raw.latitude - k.x); k.y += gain * (raw.longitude - k.y);
      k.P *= (1 - gain);
    }
    this.fusedPosition = { latitude: k.x, longitude: k.y, altitude: raw.altitude };
  }

  getPosition() { return { ...this.fusedPosition }; }
}
```

### 路径规划服务

```javascript
class NavigationService {
  constructor(apiEndpoint) {
    this.apiEndpoint = apiEndpoint; this.route = null; this.currentStep = 0;
  }

  async planRoute(start, end) {
    const res = await fetch(`${this.apiEndpoint}/route`, {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ start: { lat: start.latitude, lon: start.longitude }, end: { lat: end.latitude, lon: end.longitude } })
    });
    const data = await res.json();
    this.route = {
      totalDistance: data.distance, totalTime: data.duration,
      steps: data.steps.map((s, i) => ({ index: i, instruction: s.instruction, distance: s.distance, endPoint: s.end, maneuverType: s.maneuver }))
    };
    this.currentStep = 0; return this.route;
  }

  updateProgress(userPos) {
    if (!this.route) return null;
    const step = this.route.steps[this.currentStep];
    if (!step) return null;
    const dist = GeoMath.distance(userPos.latitude, userPos.longitude, step.endPoint.lat, step.endPoint.lon);
    if (dist < 5) {
      this.currentStep++;
      if (this.currentStep >= this.route.steps.length) return { status: 'arrived' };
      return { status: 'step_changed', step: this.route.steps[this.currentStep] };
    }
    return {
      status: 'navigating', step, distanceToNext: dist,
      progress: (this.currentStep + (1 - dist / step.distance)) / this.route.steps.length,
      heading: GeoMath.bearing(userPos.latitude, userPos.longitude, step.endPoint.lat, step.endPoint.lon)
    };
  }

  getETA() {
    if (!this.route) return 0;
    let r = 0; for (let i = this.currentStep; i < this.route.steps.length; i++) r += this.route.steps[i].distance / 1.4;
    return r;
  }
}
```

### AR 箭头渲染器

```javascript
import * as THREE from 'three';

class ARArrowRenderer {
  constructor(scene) {
    this.scene = scene;
    this.arrowGroup = new THREE.Group();
    this.scene.add(this.arrowGroup);
    this.animTime = 0;
    this.buildArrow();
  }

  buildArrow() {
    const shape = new THREE.Shape();
    shape.moveTo(0, 0.5); shape.lineTo(-0.3, -0.2); shape.lineTo(-0.1, -0.2);
    shape.lineTo(-0.1, -0.5); shape.lineTo(0.1, -0.5); shape.lineTo(0.1, -0.2);
    shape.lineTo(0.3, -0.2); shape.closePath();
    const geo = new THREE.ExtrudeGeometry(shape, { depth: 0.05, bevelSize: 0.02 });
    geo.rotateX(-Math.PI / 2);
    this.arrowMesh = new THREE.Mesh(geo, new THREE.MeshStandardMaterial({
      color: 0x2196f3, emissive: 0x1565c0, emissiveIntensity: 0.5, transparent: true, opacity: 0.9
    }));
    this.arrowGroup.add(this.arrowMesh);
  }

  update(position, heading, distanceToNext) {
    this.arrowGroup.position.copy(position); this.arrowGroup.position.y += 0.1;
    this.arrowGroup.rotation.y = -THREE.MathUtils.degToRad(heading);
    this.arrowGroup.scale.setScalar(Math.max(0.5, Math.min(1.5, distanceToNext / 10)));
    this.animTime += 0.016;
    this.arrowMesh.position.y = Math.sin(this.animTime * 3) * 0.02;
  }

  setArrived() { this.arrowMesh.material.color.set(0x4caf50); this.arrowMesh.material.emissive.set(0x2e7d32); }
  setVisible(v) { this.arrowGroup.visible = v; }
}
```

### 导航 UI 系统

```javascript
class NavigationUI {
  constructor() {
    this.els = {
      instruction: document.getElementById('nav-instruction'),
      distance: document.getElementById('nav-distance'),
      eta: document.getElementById('nav-eta'),
      progress: document.getElementById('nav-progress'),
      status: document.getElementById('status')
    };
  }

  updateNavigation(state) {
    if (!state) return;
    if (state.status === 'navigating' || state.status === 'step_changed') this.showNavigation(state);
    else if (state.status === 'arrived') this.showArrival();
  }

  showNavigation(state) {
    const maneuvers = { straight: '直行', turn_left: '左转', turn_right: '右转', elevator: '乘坐电梯' };
    this.els.instruction.textContent = maneuvers[state.step.maneuverType] || state.step.instruction;
    const d = state.distanceToNext;
    this.els.distance.textContent = d >= 1000 ? `${(d/1000).toFixed(1)} 公里` : `${Math.round(d)} 米`;
    this.els.progress.style.width = `${state.progress * 100}%`;
  }

  showArrival() {
    this.els.instruction.textContent = '您已到达目的地';
    this.els.distance.textContent = ''; this.els.progress.style.width = '100%';
  }

  updateETA(seconds) {
    if (seconds < 60) this.els.eta.textContent = '不到 1 分钟';
    else if (seconds < 3600) this.els.eta.textContent = `约 ${Math.ceil(seconds/60)} 分钟`;
    else this.els.eta.textContent = `约 ${Math.floor(seconds/3600)} 小时 ${Math.ceil((seconds%3600)/60)} 分钟`;
  }

  updateStatus(status) {
    const labels = { locating: '正在定位...', navigating: '导航中', arrived: '已到达' };
    this.els.status.textContent = labels[status] || status;
  }
}
```

## 主应用整合

```javascript
class ARNavigationApp {
  constructor() {
    this.positionManager = new PositionFusionManager();
    this.navService = new NavigationService('/api/navigation');
    this.arrowRenderer = null; this.navUI = new NavigationUI();
    this.scene = new THREE.Scene(); this.isNavigating = false;
  }

  async init() {
    const renderer = new THREE.WebGLRenderer({ canvas: document.getElementById('ar-canvas'), alpha: true });
    renderer.xr.enabled = true; renderer.setSize(window.innerWidth, window.innerHeight);
    const session = await navigator.xr.requestSession('immersive-ar', {
      requiredFeatures: ['local'], optionalFeatures: ['geo', 'dom-overlay'],
      domOverlay: { root: document.getElementById('ui') }
    });
    await renderer.xr.setSession(session);

    this.arrowRenderer = new ARArrowRenderer(this.scene);
    this.positionManager.registerSource('gps', 0.4);
    this.positionManager.registerSource('visual', 0.4);
    this.positionManager.registerSource('imu', 0.2);

    const locator = new GeoLocator();
    await locator.getCurrentPosition(); locator.startWatching();
    locator.onUpdate(pos => this.positionManager.updateSource('gps', pos, pos.accuracy));

    document.getElementById('start-nav').addEventListener('click', async () => {
      await this.navService.planRoute(this.positionManager.getPosition(), this.getSelectedDestination());
      this.isNavigating = true; this.arrowRenderer.setVisible(true); this.navUI.updateStatus('navigating');
    });
    document.getElementById('stop-nav').addEventListener('click', () => {
      this.isNavigating = false; this.arrowRenderer.setVisible(false); this.navUI.updateStatus('locating');
    });

    renderer.setAnimationLoop(() => {
      if (this.isNavigating) {
        const pos = this.positionManager.getPosition();
        const state = this.navService.updateProgress(pos);
        if (state) {
          this.navUI.updateNavigation(state);
          if (state.status === 'navigating') {
            const local = GeoMath.geoToLocal(pos.latitude, pos.longitude, state.step.endPoint.lat, state.step.endPoint.lon);
            this.arrowRenderer.update(new THREE.Vector3(local.x, 0, local.z), state.heading, state.distanceToNext);
          }
          if (state.status === 'arrived') {
            this.arrowRenderer.setArrived();
            setTimeout(() => { this.isNavigating = false; this.arrowRenderer.setVisible(false); }, 3000);
          }
          this.navUI.updateETA(this.navService.getETA());
        }
      }
      renderer.render(this.scene, this.camera);
    });
  }

  getSelectedDestination() {
    const el = document.querySelector('.destination.selected');
    return { latitude: parseFloat(el.dataset.lat), longitude: parseFloat(el.dataset.lon) };
  }
}

const app = new ARNavigationApp();
app.init();
```

## 常见误区

### 1. GPS 导航足够精确

GPS 5-15 米误差会导致箭头指向错误。必须用视觉标记或 IMU 做姿态补偿，GPS 只负责粗定位。

### 2. 忽视室内定位

大型商场、机场的导航主要在室内。需要 WiFi 指纹、蓝牙信标或图像标记等室内方案。

### 3. 箭头方向不需要平滑

GPS 方位角在低速时噪声极大。必须使用低通滤波器平滑方向数据。

## 工程建议

1. **分层定位**：室外 GPS，室内外过渡 WiFi，室内视觉标记 + IMU
2. **路径缓存**：离线缓存已规划路径，网络中断时继续导航
3. **电量管理**：省电模式降低 GPS 更新频率
4. **无障碍设计**：视障用户语音导航，听障用户振动反馈
5. **容错设计**：定位丢失时显示"正在重新定位"

## 小结

AR 导航应用综合运用多源定位融合、路径规划、AR 渲染、UI 交互等技术。关键在于理解不同定位源的精度特性、实现平滑的状态机、提供优雅的降级体验。

## 参考答案

### 练习一

**思路**：IndexedDB 存储路径数据，LRU 淘汰策略，坐标截断到小数点后 3 位实现模糊匹配。

**答案**：

```javascript
class RouteCache {
  constructor(maxEntries = 10) { this.maxEntries = maxEntries; this.db = null; }

  async init() {
    this.db = await new Promise((resolve, reject) => {
      const req = indexedDB.open('ar-nav-routes', 1);
      req.onupgradeneeded = (e) => {
        const db = e.target.result;
        const store = db.createObjectStore('routes', { keyPath: 'id' });
        store.createIndex('startKey', 'startKey');
        store.createIndex('lastUsed', 'lastUsed');
      };
      req.onsuccess = () => resolve(req.result); req.onerror = () => reject(req.error);
    });
  }

  key(lat, lon) { return `${lat.toFixed(3)}_${lon.toFixed(3)}`; }

  async cacheRoute(route) {
    const entry = {
      id: `${this.key(route.start.latitude, route.start.longitude)}__${this.key(route.end.latitude, route.end.longitude)}`,
      startKey: this.key(route.start.latitude, route.start.longitude),
      endKey: this.key(route.end.latitude, route.end.longitude),
      steps: route.steps, totalDistance: route.totalDistance, lastUsed: Date.now()
    };
    const tx = this.db.transaction('routes', 'readwrite');
    tx.objectStore('routes').put(entry);
    await new Promise(r => tx.oncomplete = r);
    await this.enforceLimits();
  }

  async findRoute(startLat, startLon, endLat, endLon) {
    const tx = this.db.transaction('routes', 'readonly');
    const results = await new Promise(r => {
      const req = tx.objectStore('routes').index('startKey').getAll(this.key(startLat, startLon));
      req.onsuccess = () => r(req.result);
    });
    return results.find(r => r.endKey === this.key(endLat, endLon)) || null;
  }

  async enforceLimits() {
    const tx = this.db.transaction('routes', 'readwrite');
    const store = tx.objectStore('routes');
    const all = await new Promise(r => {
      const req = store.index('lastUsed').getAll(); req.onsuccess = () => r(req.result);
    });
    all.sort((a, b) => a.lastUsed - b.lastUsed);
    while (all.length > this.maxEntries) store.delete(all.shift().id);
  }
}
```

**要点**：IndexedDB 比 localStorage 更适合大容量路径数据；LRU 策略确保缓存不无限增长；坐标截断实现"附近"匹配。

### 练习二

**思路**：Web Speech API 的 SpeechSynthesis 播报、SpeechRecognition 识别。音量通过麦克风 AnalyserNode 自适应调整。

**答案**：

```javascript
class VoiceNavigation {
  constructor() {
    this.synth = window.speechSynthesis; this.recognition = null;
    this.analyser = null; this.lastSpokenDist = Infinity;
    this.thresholds = [100, 30, 10]; this.callbacks = { onSearch: [] };
  }

  async init() {
    const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (SR) {
      this.recognition = new SR(); this.recognition.lang = 'zh-CN';
      this.recognition.onresult = (e) => this.callbacks.onSearch.forEach(cb => cb(e.results[0][0].transcript));
    }
    try {
      const ctx = new AudioContext();
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      this.analyser = ctx.createAnalyser(); this.analyser.fftSize = 256;
      ctx.createMediaStreamSource(stream).connect(this.analyser);
    } catch (e) { /* 麦克风不可用 */ }
  }

  onSearch(callback) { this.callbacks.onSearch.push(callback); }
  startListening() { this.recognition?.start(); }

  speak(text) {
    this.synth.cancel();
    const u = new SpeechSynthesisUtterance(text); u.lang = 'zh-CN';
    if (this.analyser) {
      const data = new Uint8Array(this.analyser.frequencyBinCount);
      this.analyser.getByteFrequencyData(data);
      u.volume = Math.min(1.0, 0.8 + data.reduce((s, v) => s + v, 0) / data.length / 255 * 0.5);
    }
    this.synth.speak(u);
  }

  checkNavigation(distance, step) {
    const texts = { turn_left: '左转', turn_right: '右转', straight: '直行' };
    for (const t of this.thresholds) {
      if (distance <= t && this.lastSpokenDist > t) {
        const action = texts[step.maneuverType] || '';
        if (t === 100) this.speak(`前方 ${Math.round(distance)} 米${action}`);
        else if (t === 30) this.speak(`${Math.round(distance)} 米后${action}`);
        else if (t === 10) this.speak(`即将${action}`);
        break;
      }
    }
    this.lastSpokenDist = distance;
  }

  speakArrival(name) { this.speak(`您已到达${name || '目的地'}`); }
}
```

**要点**：三级距离阈值提供渐进式语音提示；环境噪声检测自适应调整音量；`synth.cancel()` 避免语音堆积。
