## 场景管理器

场景管理器是应用核心，负责初始化 Three.js、管理渲染循环和协调子系统。

```typescript
import * as THREE from 'three'

class SceneManager {
  readonly renderer: THREE.WebGLRenderer
  readonly scene: THREE.Scene
  readonly camera: THREE.PerspectiveCamera
  private clock = new THREE.Clock()
  private updateCallbacks: ((dt: number) => void)[] = []

  constructor(container: HTMLElement) {
    this.renderer = new THREE.WebGLRenderer({ antialias: true })
    this.renderer.setSize(container.clientWidth, container.clientHeight)
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2))
    this.renderer.shadowMap.enabled = true
    this.renderer.shadowMap.type = THREE.PCFSoftShadowMap
    this.renderer.outputColorSpace = THREE.SRGBColorSpace
    this.renderer.toneMapping = THREE.ACESFilmicToneMapping
    container.appendChild(this.renderer.domElement)

    this.scene = new THREE.Scene()
    this.scene.background = new THREE.Color(0x87ceeb)
    this.scene.fog = new THREE.Fog(0x87ceeb, 20, 50)

    this.camera = new THREE.PerspectiveCamera(
      60, container.clientWidth / container.clientHeight, 0.1, 100
    )

    window.addEventListener('resize', () => {
      this.camera.aspect = container.clientWidth / container.clientHeight
      this.camera.updateProjectionMatrix()
      this.renderer.setSize(container.clientWidth, container.clientHeight)
    })
  }

  onUpdate(callback: (dt: number) => void) {
    this.updateCallbacks.push(callback)
  }

  start() {
    this.renderer.setAnimationLoop(() => {
      const dt = this.clock.getDelta()
      for (const cb of this.updateCallbacks) cb(dt)
      this.renderer.render(this.scene, this.camera)
    })
  }
}
```

## 轨道相机控制

轨道相机允许用户拖拽旋转视角、滚轮缩放。核心是用球坐标系（Spherical）管理相机位置。

```typescript
class OrbitCamera {
  private spherical = new THREE.Spherical(5, Math.PI / 3, 0)
  private target = new THREE.Vector3(0, 1, 0)
  private dragging = false
  private lastMouse = { x: 0, y: 0 }

  constructor(private camera: THREE.PerspectiveCamera, el: HTMLElement) {
    this.update()
    el.addEventListener('mousedown', (e) => { this.dragging = true; this.lastMouse = { x: e.clientX, y: e.clientY } })
    el.addEventListener('mousemove', (e) => {
      if (!this.dragging) return
      this.spherical.theta -= (e.clientX - this.lastMouse.x) * 0.005
      this.spherical.phi = Math.max(0.1, Math.min(Math.PI - 0.1, this.spherical.phi - (e.clientY - this.lastMouse.y) * 0.005))
      this.lastMouse = { x: e.clientX, y: e.clientY }
      this.update()
    })
    el.addEventListener('mouseup', () => this.dragging = false)
    el.addEventListener('wheel', (e) => {
      this.spherical.radius = Math.max(1, Math.min(20, this.spherical.radius + e.deltaY * 0.01))
      this.update()
    })
  }

  private update() {
    const pos = new THREE.Vector3().setFromSpherical(this.spherical)
    this.camera.position.copy(this.target).add(pos)
    this.camera.lookAt(this.target)
  }
}
```

## 光照系统

三层光照：环境光（基础照明）+ 方向光（模拟太阳，产生阴影）+ 半球光（模拟天空反射）。

```typescript
function setupLighting(scene: THREE.Scene): THREE.DirectionalLight {
  // 环境光
  scene.add(new THREE.AmbientLight(0x404040, 0.5))

  // 主光源（带阴影）
  const sun = new THREE.DirectionalLight(0xffffff, 1.2)
  sun.position.set(10, 15, 10)
  sun.castShadow = true
  sun.shadow.mapSize.set(2048, 2048)
  sun.shadow.camera.near = 0.5
  sun.shadow.camera.far = 50
  const d = 15
  sun.shadow.camera.left = -d
  sun.shadow.camera.right = d
  sun.shadow.camera.top = d
  sun.shadow.camera.bottom = -d
  sun.shadow.bias = -0.001
  scene.add(sun)

  // 半球光（天空+地面反射）
  scene.add(new THREE.HemisphereLight(0x87ceeb, 0x362e1a, 0.4))

  return sun
}
```

## 输入管理与射线交互

通过射线检测实现鼠标点击和悬停交互。

```typescript
interface Interactable {
  mesh: THREE.Object3D
  onSelect?(): void
  onHover?(hovering: boolean): void
}

class InputManager {
  private raycaster = new THREE.Raycaster()
  private mouse = new THREE.Vector2()
  private items: Interactable[] = []
  private hovered: Interactable | null = null

  constructor(private renderer: THREE.WebGLRenderer, private camera: THREE.PerspectiveCamera) {
    const el = renderer.domElement
    el.addEventListener('click', () => { this.updateMouse(event as MouseEvent); this.hitTest()?.onSelect?.() })
    el.addEventListener('mousemove', (e) => {
      this.updateMouse(e)
      const hit = this.hitTest()
      if (this.hovered && this.hovered !== hit) { this.hovered.onHover?.(false); el.style.cursor = 'default' }
      if (hit && hit !== this.hovered) { hit.onHover?.(true); el.style.cursor = 'pointer' }
      this.hovered = hit
    })
  }

  register(item: Interactable) { this.items.push(item) }

  private updateMouse(e: MouseEvent) {
    const r = this.renderer.domElement.getBoundingClientRect()
    this.mouse.x = ((e.clientX - r.left) / r.width) * 2 - 1
    this.mouse.y = -((e.clientY - r.top) / r.height) * 2 + 1
  }

  private hitTest(): Interactable | null {
    this.raycaster.setFromCamera(this.mouse, this.camera)
    const hits = this.raycaster.intersectObjects(this.items.map(i => i.mesh), true)
    if (!hits.length) return null
    return this.items.find(i => i.mesh === hits[0].object || i.mesh.children.includes(hits[0].object)) ?? null
  }
}
```

## 可交互立方体

```typescript
function createInteractiveBox(color: number): { group: THREE.Group } & Interactable {
  const group = new THREE.Group()
  const mat = new THREE.MeshStandardMaterial({ color, roughness: 0.3, metalness: 0.1 })
  const mesh = new THREE.Mesh(new THREE.BoxGeometry(0.5, 0.5, 0.5), mat)
  mesh.castShadow = true
  mesh.receiveShadow = true
  group.add(mesh)

  let selected = false
  const originalColor = color

  return {
    group,
    mesh,
    onSelect() {
      selected = !selected
      mat.color.setHex(selected ? 0xff6b6b : originalColor)
      group.scale.setScalar(selected ? 1.1 : 1)
    },
    onHover(hovering) {
      if (selected) return
      mat.color.setHex(hovering ? 0x6bb5ff : originalColor)
      group.scale.setScalar(hovering ? 1.05 : 1)
    }
  }
}
```

## 带网格纹理的地面

```typescript
function createGround(size: number = 20): THREE.Mesh {
  const canvas = document.createElement('canvas')
  canvas.width = canvas.height = 512
  const ctx = canvas.getContext('2d')!
  ctx.fillStyle = '#808080'
  ctx.fillRect(0, 0, 512, 512)
  ctx.strokeStyle = '#666666'
  ctx.lineWidth = 1
  const gs = 512 / 10
  for (let i = 0; i <= 10; i++) {
    ctx.beginPath(); ctx.moveTo(i * gs, 0); ctx.lineTo(i * gs, 512); ctx.stroke()
    ctx.beginPath(); ctx.moveTo(0, i * gs); ctx.lineTo(512, i * gs); ctx.stroke()
  }
  const texture = new THREE.CanvasTexture(canvas)
  const ground = new THREE.Mesh(
    new THREE.PlaneGeometry(size, size),
    new THREE.MeshStandardMaterial({ map: texture, roughness: 0.8 })
  )
  ground.rotation.x = -Math.PI / 2
  ground.receiveShadow = true
  return ground
}
```

## 常见误区

1. **"所有代码写在一个文件里"** —— 3D 应用复杂度增长快，从一开始就用模块化架构
2. **"不用管阴影"** —— XR 中没有阴影的物体会"飘"在空中，破坏沉浸感
3. **"Three.js 默认配置就够用"** —— 启用 `SRGBColorSpace` 和 `ACESFilmicToneMapping` 画质提升明显

## 工程建议

1. 用 `renderer.info` 从开发初期就监控 draw call 和三角形数量
2. 用 `castShadow`/`receiveShadow` 精确控制阴影计算范围
3. 使用 `MeshStandardMaterial`，物理正确且在不同光照下表现一致
4. 输入逻辑和渲染逻辑分离，InputManager 只管检测，交互响应由物体自己处理
5. 为每个对象实现资源释放，Three.js 不会自动释放 GPU 资源

## 参考答案

### 练习一

**答案**：
```typescript
// 在 SceneManager 的 onUpdate 回调中
let time = 0
engine.onUpdate((dt) => {
  time += dt
  boxes.forEach((box, i) => {
    const phase = time * Math.PI + i * 0.5 // 错开相位
    box.group.scale.y = 1 + Math.sin(phase) * 0.05
  })
})
```

**要点**：`Math.sin` 参数用 `time * Math.PI` 实现 2 秒周期。随机相位让各立方体呼吸节奏不同步，视觉更自然。幅度 0.05（5%）足够明显但不夸张。

### 练习二

**答案**：
```typescript
function createGridGround(size: number = 20, divisions: number = 20): THREE.Group {
  const group = new THREE.Group()

  const grid = new THREE.GridHelper(size, divisions, 0x888888, 0x666666)
  grid.position.y = 0.001 // 避免 Z-fighting
  group.add(grid)

  const ground = new THREE.Mesh(
    new THREE.PlaneGeometry(size, size),
    new THREE.MeshStandardMaterial({ color: 0x808080, roughness: 0.9 })
  )
  ground.rotation.x = -Math.PI / 2
  ground.receiveShadow = true
  group.add(ground)

  return group
}
```

**要点**：`GridHelper` 不支持阴影接收，需要单独创建地面平面。`position.y = 0.001` 避免网格线与地面重叠导致闪烁。

**常见错误写法**：只用 `GridHelper` 不加地面平面。`GridHelper` 是线段渲染，`receiveShadow` 不生效，阴影无法显示。
