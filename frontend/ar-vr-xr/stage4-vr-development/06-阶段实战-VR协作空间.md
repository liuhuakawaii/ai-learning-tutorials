## 白板系统（CRDT）

多人同时画画会产生冲突。使用 CRDT 保证最终一致性：

```typescript
class WhiteboardCRDT {
  private strokes = new Map<string, Stroke>()
  private tombstones = new Set<string>()
  private vectorClock = new Map<string, number>()

  addStroke(stroke: Stroke) { this.strokes.set(stroke.id, stroke); this.incrementClock(stroke.authorId) }

  deleteStroke(strokeId: string, authorId: string) {
    this.tombstones.add(strokeId); this.strokes.delete(strokeId); this.incrementClock(authorId)
  }

  merge(remote: { strokes: Stroke[]; tombstones: string[] }) {
    for (const stroke of remote.strokes) {
      if (!this.tombstones.has(stroke.id)) {
        const existing = this.strokes.get(stroke.id)
        if (!existing || stroke.timestamp > existing.timestamp) this.strokes.set(stroke.id, stroke)
      }
    }
    for (const id of remote.tombstones) { this.tombstones.add(id); this.strokes.delete(id) }
  }

  export() { return { strokes: Array.from(this.strokes.values()), tombstones: Array.from(this.tombstones) } }
  getStrokes() { return Array.from(this.strokes.values()) }
  private incrementClock(peerId: string) { this.vectorClock.set(peerId, (this.vectorClock.get(peerId) ?? 0) + 1) }
}
```

### 3D 白板渲染

```typescript
class WhiteboardRenderer {
  private mesh: THREE.Mesh
  private canvas: HTMLCanvasElement
  private ctx: CanvasRenderingContext2D
  private texture: THREE.CanvasTexture

  constructor(width: number, height: number, resolution = 2048) {
    this.canvas = document.createElement('canvas')
    this.canvas.width = resolution
    this.canvas.height = Math.round(resolution * (height / width))
    this.ctx = this.canvas.getContext('2d')!
    this.texture = new THREE.CanvasTexture(this.canvas)
    this.texture.minFilter = THREE.LinearMipMapLinearFilter
    this.mesh = new THREE.Mesh(
      new THREE.PlaneGeometry(width, height),
      new THREE.MeshStandardMaterial({ map: this.texture, roughness: 0.8 })
    )
  }

  fullRedraw(strokes: Stroke[]) {
    const ctx = this.ctx, w = this.canvas.width, h = this.canvas.height
    ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, w, h)
    ctx.strokeStyle = '#eeeeee'; ctx.lineWidth = 1
    for (let x = 0; x < w; x += 40) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, h); ctx.stroke() }
    for (let y = 0; y < h; y += 40) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(w, y); ctx.stroke() }
    strokes.forEach(s => this.drawStroke(ctx, s, w, h))
    this.texture.needsUpdate = true
  }

  drawStroke(ctx: CanvasRenderingContext2D, stroke: Stroke, w: number, h: number) {
    if (stroke.points.length < 2) return
    ctx.strokeStyle = stroke.color; ctx.lineWidth = stroke.width * (w / 2)
    ctx.lineCap = 'round'; ctx.lineJoin = 'round'; ctx.beginPath()
    const [fx, fy] = stroke.points[0]; ctx.moveTo((fx + 1) * w / 2, (1 - fy) * h / 2)
    for (let i = 1; i < stroke.points.length; i++) {
      const [px, py] = stroke.points[i]; ctx.lineTo((px + 1) * w / 2, (1 - py) * h / 2)
    }
    ctx.stroke(); this.texture.needsUpdate = true
  }

  getMesh() { return this.mesh }
}
```

## 3D 模型系统

```typescript
class ModelManager {
  private models = new Map<string, THREE.Group>()
  private loader: GLTFLoader

  constructor(private scene: THREE.Scene) { this.loader = new GLTFLoader() }

  async importModel(url: string, position: THREE.Vector3, placedBy: string): Promise<string> {
    const id = crypto.randomUUID()
    const gltf = await this.loader.loadAsync(url)
    const model = gltf.scene
    const box = new THREE.Box3().setFromObject(model)
    const size = box.getSize(new THREE.Vector3())
    model.scale.setScalar(0.5 / Math.max(size.x, size.y, size.z))
    const center = box.getCenter(new THREE.Vector3())
    model.position.sub(center.multiplyScalar(model.scale.x)).add(position)
    model.userData.modelId = id
    model.traverse(c => { if (c instanceof THREE.Mesh) { c.castShadow = true; c.receiveShadow = true } })
    this.scene.add(model); this.models.set(id, model)
    return id
  }

  removeModel(id: string) { const m = this.models.get(id); if (m) { this.scene.remove(m); this.models.delete(id) } }
}
```

## 房间服务端

```typescript
import { WebSocketServer, WebSocket } from 'ws'

class RoomServer {
  private rooms = new Map<string, { participants: Map<string, { ws: WebSocket; name: string }>; whiteboard: any; models: any[] }>()

  constructor(port: number) {
    const wss = new WebSocketServer({ port })
    wss.on('connection', (ws) => {
      ws.on('message', (data) => {
        const msg = JSON.parse(data.toString())
        if (msg.type === 'join_room') this.joinRoom(ws, msg)
        else if (msg.type === 'peer_update') this.broadcast(msg.roomId, msg, msg.peerId)
        else if (msg.type === 'whiteboard_stroke') { this.findRoom(ws)?.whiteboard.strokes.push(msg.stroke); this.broadcast(msg.roomId, { type: 'whiteboard_stroke', stroke: msg.stroke }) }
      })
      ws.on('close', () => this.disconnect(ws))
    })
  }

  private joinRoom(ws: WebSocket, msg: any) {
    let room = this.rooms.get(msg.roomId)
    if (!room) { room = { participants: new Map(), whiteboard: { strokes: [], tombstones: [] }, models: [] }; this.rooms.set(msg.roomId, room) }
    const id = crypto.randomUUID(); room.participants.set(id, { ws, name: msg.userName })
    ws.send(JSON.stringify({ type: 'room_state', participantId: id, whiteboard: room.whiteboard, models: room.models }))
    this.broadcast(msg.roomId, { type: 'peer_join', peerId: id, userName: msg.userName }, id)
  }

  private broadcast(roomId: string, msg: any, excludeId?: string) {
    const room = this.rooms.get(roomId); if (!room) return
    const data = JSON.stringify(msg)
    room.participants.forEach((p, id) => { if (id !== excludeId && p.ws.readyState === WebSocket.OPEN) p.ws.send(data) })
  }

  private findRoom(ws: WebSocket) {
    for (const room of this.rooms.values()) for (const p of room.participants.values()) if (p.ws === ws) return room
    return null
  }

  private disconnect(ws: WebSocket) {
    for (const [roomId, room] of this.rooms) {
      for (const [id, p] of room.participants) {
        if (p.ws === ws) { room.participants.delete(id); this.broadcast(roomId, { type: 'peer_leave', peerId: id }); if (room.participants.size === 0) this.rooms.delete(roomId); return }
      }
    }
  }
}
```

## 客户端集成

```typescript
class VRCollaborationSpace {
  private network: VRNetworkManager
  private voiceChat: SpatialVoiceChat
  private whiteboard: WhiteboardCRDT
  private whiteboardRenderer: WhiteboardRenderer
  private modelManager: ModelManager
  private avatars = new Map<string, VRAvatar>()

  constructor(private renderer: THREE.WebGLRenderer, private scene: THREE.Scene, cameraRig: THREE.Group) {
    this.network = new VRNetworkManager('wss://your-server.com')
    this.voiceChat = new SpatialVoiceChat()
    this.whiteboard = new WhiteboardCRDT()
    this.whiteboardRenderer = new WhiteboardRenderer(3, 2)
    this.modelManager = new ModelManager(scene)
  }

  async joinRoom(roomId: string, userName: string) {
    await this.network.connect(roomId, userName)
    await this.voiceChat.startMicrophone()
    this.setupRoom()
    this.setupNetworkHandlers()
  }

  private setupRoom() {
    const wb = this.whiteboardRenderer.getMesh(); wb.position.set(0, 1.5, -3); this.scene.add(wb)
    const table = new THREE.Mesh(new THREE.BoxGeometry(2, 0.75, 1), new THREE.MeshStandardMaterial({ color: 0x8B4513 }))
    table.position.set(0, 0.375, -1.5); this.scene.add(table)
    this.scene.add(new THREE.AmbientLight(0xffffff, 0.4))
    const dir = new THREE.DirectionalLight(0xffffff, 0.6); dir.position.set(5, 10, 5); dir.castShadow = true; this.scene.add(dir)
  }

  private setupNetworkHandlers() {
    this.network.on('peer_join', (msg: any) => {
      const avatar = new VRAvatar(msg.peerId, msg.userName)
      this.avatars.set(msg.peerId, avatar); this.scene.add(avatar.getGroup())
      this.voiceChat.addPeerAudio(msg.peerId, new MediaStream())
    })
    this.network.on('peer_leave', (msg: any) => {
      const a = this.avatars.get(msg.peerId)
      if (a) { a.dispose(); this.scene.remove(a.getGroup()); this.avatars.delete(msg.peerId) }
      this.voiceChat.removePeerAudio(msg.peerId)
    })
    this.network.on('peer_update', (msg: any) => {
      const a = this.avatars.get(msg.peerId)
      if (a) a.updatePose(msg.headPose, msg.leftHand, msg.rightHand, Date.now())
      this.voiceChat.updatePeerPosition(msg.peerId, msg.headPose.position)
    })
    this.network.on('whiteboard_stroke', (msg: any) => {
      this.whiteboard.addStroke(msg.stroke)
      this.whiteboardRenderer.drawStroke((this.whiteboardRenderer as any).ctx, msg.stroke, 2048, Math.round(2048 * 2 / 3))
    })
  }

  update(dt: number) {
    this.network.update(dt)
    const renderTime = performance.now()
    this.avatars.forEach(a => a.render(renderTime))
    const camera = this.renderer.xr.getCamera()
    const pos = new THREE.Vector3(), fwd = new THREE.Vector3()
    camera.getWorldPosition(pos); camera.getWorldDirection(fwd)
    this.voiceChat.updateListener(pos.toArray() as any, fwd.toArray() as any, [0, 1, 0])
  }
}
```

## 性能优化策略

| 领域 | 策略 |
|------|------|
| 网络 | 姿态压缩 float32→int16 减少 50% 带宽；只发送变化字段；笔画批量发送 |
| 渲染 | 远程 Avatar 用低精度模型；白板用 Mipmap；重复物体用 Instancing |
| 音频 | 超过 10 米的用户静音；限制同时活跃音频流为 8 个 |

## 常见误区

1. **同步所有状态**：白板每笔都同步会浪费带宽，应本地先绘制再批量发送
2. **忽视断线重连**：VR 用户可能因电量、网络突然断线，必须实现自动重连
3. **不做权限控制**：任何人都能删除他人内容会导致混乱

## 工程建议

1. **状态快照**：每 30 秒自动保存房间状态
2. **冲突解决**：白板用 CRDT，模型用所有权锁
3. **性能预算**：每房间最多 8 人，超过时使用分区
4. **测试策略**：用多个浏览器标签模拟多用户
5. **无障碍**：为非 VR 用户提供 2D 网页端接入

### 练习一

**思路**：用 IndexedDB 存储房间状态，退出前自动保存，进入时检查是否有未完成的房间。

```typescript
class RoomPersistence {
  private async openDB(): Promise<IDBDatabase> {
    return new Promise((resolve, reject) => {
      const req = indexedDB.open('vr-collab-space', 1)
      req.onerror = () => reject(req.error)
      req.onsuccess = () => resolve(req.result)
      req.onupgradeneeded = () => req.result.createObjectStore('rooms', { keyPath: 'id' })
    })
  }

  async saveRoom(roomId: string, state: any) {
    const db = await this.openDB()
    db.transaction('rooms', 'readwrite').objectStore('rooms').put({ id: roomId, ...state, savedAt: Date.now() })
    db.close()
  }

  async loadRoom(roomId: string): Promise<any | null> {
    const db = await this.openDB()
    return new Promise((resolve, reject) => {
      const req = db.transaction('rooms', 'readonly').objectStore('rooms').get(roomId)
      req.onsuccess = () => { db.close(); resolve(req.result ?? null) }
      req.onerror = () => { db.close(); reject(req.error) }
    })
  }
}
```

### 练习二

**思路**：用 Web Speech API 的 SpeechRecognition 接口，识别结果显示为 3D 文字精灵跟随 avatar 移动。

```typescript
class VoiceToText {
  private recognition: SpeechRecognition | null = null

  constructor(private scene: THREE.Scene) {
    const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
    if (SR) { this.recognition = new SR(); this.recognition.continuous = true; this.recognition.interimResults = true; this.recognition.lang = 'zh-CN' }
  }

  start(peerId: string, avatar: VRAvatar) {
    if (!this.recognition) return
    this.recognition.onresult = (event: SpeechRecognitionEvent) => {
      const last = event.results[event.results.length - 1]
      this.showText(avatar, last[0].transcript, last.isFinal)
    }
    this.recognition.start()
  }

  private showText(avatar: VRAvatar, text: string, isFinal: boolean) {
    const canvas = document.createElement('canvas'); canvas.width = 512; canvas.height = 128
    const ctx = canvas.getContext('2d')!
    ctx.fillStyle = isFinal ? 'rgba(0,0,0,0.7)' : 'rgba(0,0,0,0.4)'; ctx.fillRect(0, 0, 512, 128)
    ctx.fillStyle = isFinal ? '#ffffff' : '#cccccc'; ctx.font = '28px "PingFang SC", sans-serif'
    ctx.textAlign = 'center'; ctx.fillText(text.length > 20 ? text.slice(-20) + '...' : text, 256, 72)
    const sprite = new THREE.Sprite(new THREE.SpriteMaterial({ map: new THREE.CanvasTexture(canvas), transparent: true }))
    sprite.scale.set(0.4, 0.1, 1); sprite.position.y = 0.4
    ;(avatar as any).head.add(sprite)
    setTimeout(() => sprite.parent?.remove(sprite), 5000)
  }

  stop() { this.recognition?.stop() }
}
```
