## 一、邀请系统设计

### 1.1 邀请流程

```
1. 管理员输入被邀请人邮箱
2. 系统生成邀请链接
3. 发送邀请邮件
4. 被邀请人点击链接
5. 被邀请人登录/注册
6. 自动加入团队
```

### 1.2 数据模型

```prisma
model Invitation {
  id        String   @id @default(cuid())
  email     String
  token     String   @unique
  role      Role     @default(MEMBER)
  status    InvitationStatus @default(PENDING)
  expiresAt DateTime
  createdAt DateTime @default(now())

  team     Team   @relation(fields: [teamId], references: [id], onDelete: Cascade)
  teamId   String
  inviter  User   @relation(fields: [inviterId], references: [id])
  inviterId String

  @@unique([email, teamId])
}

enum InvitationStatus {
  PENDING
  ACCEPTED
  EXPIRED
  REVOKED
}
```

## 二、创建邀请

### 2.1 生成邀请链接

```tsx
// lib/invitation.ts
import crypto from 'crypto'
import { prisma } from './prisma'

export async function createInvitation({
  email,
  teamId,
  inviterId,
  role = 'MEMBER',
}: {
  email: string
  teamId: string
  inviterId: string
  role?: 'ADMIN' | 'MEMBER'
}) {
  // 检查是否已有待处理的邀请
  const existing = await prisma.invitation.findFirst({
    where: {
      email,
      teamId,
      status: 'PENDING',
    }
  })

  if (existing) {
    // 如果未过期，返回现有邀请
    if (existing.expiresAt > new Date()) {
      return { invitation: existing, isNew: false }
    }
    // 如果已过期，更新状态
    await prisma.invitation.update({
      where: { id: existing.id },
      data: { status: 'EXPIRED' }
    })
  }

  // 生成唯一 token
  const token = crypto.randomBytes(32).toString('hex')

  // 设置过期时间（7 天）
  const expiresAt = new Date()
  expiresAt.setDate(expiresAt.getDate() + 7)

  const invitation = await prisma.invitation.create({
    data: {
      email,
      token,
      role,
      expiresAt,
      teamId,
      inviterId,
    }
  })

  return { invitation, isNew: true }
}
```

### 2.2 Server Action

```tsx
// app/teams/[teamId]/actions.ts
'use server'

import { z } from 'zod'
import { revalidatePath } from 'next/cache'
import { prisma } from '@/lib/prisma'
import { requireTeamRole } from '@/lib/permissions'
import { createInvitation } from '@/lib/invitation'
import { sendInvitationEmail } from '@/lib/email'

const InviteSchema = z.object({
  email: z.string().email('邮箱格式不正确'),
  role: z.enum(['ADMIN', 'MEMBER']).default('MEMBER'),
})

export async function inviteMember(teamId: string, prevState: any, formData: FormData) {
  const { userId } = await requireTeamRole(teamId, 'ADMIN')

  const validated = InviteSchema.safeParse({
    email: formData.get('email'),
    role: formData.get('role'),
  })

  if (!validated.success) {
    return { errors: validated.error.flatten().fieldErrors }
  }

  const { email, role } = validated.data

  // 检查用户是否已是成员
  const existingUser = await prisma.user.findUnique({
    where: { email }
  })

  if (existingUser) {
    const isMember = await prisma.membership.findUnique({
      where: {
        userId_teamId: { userId: existingUser.id, teamId }
      }
    })

    if (isMember) {
      return { error: '该用户已是团队成员' }
    }
  }

  // 创建邀请
  const { invitation, isNew } = await createInvitation({
    email,
    teamId,
    inviterId: userId,
    role,
  })

  // 获取团队信息
  const team = await prisma.team.findUnique({
    where: { id: teamId }
  })

  // 发送邀请邮件
  const inviteUrl = `${process.env.NEXT_PUBLIC_URL}/invitations/${invitation.token}`

  await sendInvitationEmail({
    to: email,
    teamName: team!.name,
    inviteUrl,
    expiresInDays: 7,
  })

  revalidatePath(`/teams/${teamId}/members`)

  return {
    success: true,
    message: isNew ? '邀请已发送' : '邀请已重新发送'
  }
}
```

## 三、接受邀请

### 3.1 邀请页面

```tsx
// app/invitations/[token]/page.tsx
import { notFound, redirect } from 'next/navigation'
import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'
import { AcceptButton } from './AcceptButton'

export default async function InvitationPage({
  params
}: {
  params: Promise<{ token: string }>
}) {
  const { token } = await params

  // 查找邀请
  const invitation = await prisma.invitation.findUnique({
    where: { token },
    include: {
      team: { select: { name: true } },
      inviter: { select: { name: true } },
    }
  })

  if (!invitation) {
    notFound()
  }

  // 检查是否过期
  if (invitation.expiresAt < new Date()) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="text-center">
          <h1 className="text-2xl font-bold mb-4">邀请已过期</h1>
          <p className="text-gray-600">请联系团队管理员重新发送邀请</p>
        </div>
      </div>
    )
  }

  // 检查状态
  if (invitation.status !== 'PENDING') {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="text-center">
          <h1 className="text-2xl font-bold mb-4">邀请已处理</h1>
          <p className="text-gray-600">此邀请已被接受或撤销</p>
        </div>
      </div>
    )
  }

  // 检查是否已登录
  const user = await getCurrentUser()

  if (!user) {
    // 保存 token 到 URL，登录后回来
    redirect(`/login?redirect=/invitations/${token}`)
  }

  // 检查邮箱是否匹配
  if (user.email !== invitation.email) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="text-center">
          <h1 className="text-2xl font-bold mb-4">邮箱不匹配</h1>
          <p className="text-gray-600">
            此邀请发送给 {invitation.email}，
            但你当前登录的是 {user.email}
          </p>
        </div>
      </div>
    )
  }

  return (
    <div className="min-h-screen flex items-center justify-center">
      <div className="max-w-md w-full p-8 bg-white rounded-lg shadow">
        <h1 className="text-2xl font-bold mb-4">团队邀请</h1>
        <p className="mb-6">
          <strong>{invitation.inviter.name}</strong> 邀请你加入{' '}
          <strong>{invitation.team.name}</strong>
        </p>
        <AcceptButton token={token} />
      </div>
    </div>
  )
}
```

### 3.2 接受邀请的 Action

```tsx
// app/invitations/[token]/actions.ts
'use server'

import { prisma } from '@/lib/prisma'
import { getCurrentUser } from '@/lib/auth'
import { revalidatePath } from 'next/cache'
import { redirect } from 'next/navigation'

export async function acceptInvitation(token: string) {
  const user = await getCurrentUser()

  if (!user) {
    redirect(`/login?redirect=/invitations/${token}`)
  }

  const invitation = await prisma.invitation.findUnique({
    where: { token }
  })

  if (!invitation) {
    return { error: '邀请不存在' }
  }

  if (invitation.status !== 'PENDING') {
    return { error: '邀请已处理' }
  }

  if (invitation.expiresAt < new Date()) {
    return { error: '邀请已过期' }
  }

  if (invitation.email !== user.email) {
    return { error: '邮箱不匹配' }
  }

  // 检查是否已是成员
  const existingMember = await prisma.membership.findUnique({
    where: {
      userId_teamId: {
        userId: user.id,
        teamId: invitation.teamId
      }
    }
  })

  if (existingMember) {
    // 更新邀请状态
    await prisma.invitation.update({
      where: { id: invitation.id },
      data: { status: 'ACCEPTED' }
    })
    return { success: true, message: '你已是团队成员' }
  }

  // 添加成员并更新邀请状态
  await prisma.$transaction([
    prisma.membership.create({
      data: {
        userId: user.id,
        teamId: invitation.teamId,
        role: invitation.role,
      }
    }),
    prisma.invitation.update({
      where: { id: invitation.id },
      data: { status: 'ACCEPTED' }
    })
  ])

  revalidatePath(`/teams/${invitation.teamId}`)
  redirect(`/teams/${invitation.teamId}`)
}
```

### 3.3 接受按钮

```tsx
// app/invitations/[token]/AcceptButton.tsx
'use client'

import { useTransition } from 'react'
import { acceptInvitation } from './actions'

export function AcceptButton({ token }: { token: string }) {
  const [isPending, startTransition] = useTransition()

  function handleAccept() {
    startTransition(async () => {
      const result = await acceptInvitation(token)
      if (result?.error) {
        alert(result.error)
      }
    })
  }

  return (
    <button
      onClick={handleAccept}
      disabled={isPending}
      className="w-full py-3 bg-blue-500 text-white rounded hover:bg-blue-600 disabled:opacity-50"
    >
      {isPending ? '处理中...' : '接受邀请'}
    </button>
  )
}
```

### 4.1 使用 Resend

```tsx
// lib/email.ts
import { Resend } from 'resend'

const resend = new Resend(process.env.RESEND_API_KEY)

export async function sendInvitationEmail({
  to,
  teamName,
  inviteUrl,
  expiresInDays,
}: {
  to: string
  teamName: string
  inviteUrl: string
  expiresInDays: number
}) {
  await resend.emails.send({
    from: 'Your App <noreply@yourdomain.com>',

    `,
  })
}
```

## 五、撤销邀请

```tsx
// app/teams/[teamId]/members/actions.ts
'use server'

import { prisma } from '@/lib/prisma'
import { requireTeamRole } from '@/lib/permissions'
import { revalidatePath } from 'next/cache'

export async function revokeInvitation(teamId: string, invitationId: string) {
  await requireTeamRole(teamId, 'ADMIN')

  await prisma.invitation.update({
    where: { id: invitationId },
    data: { status: 'REVOKED' }
  })

  revalidatePath(`/teams/${teamId}/members`)
}
```

## 六、邀请列表

```tsx
// app/teams/[teamId]/members/page.tsx
import { prisma } from '@/lib/prisma'
import { requireTeamRole } from '@/lib/permissions'
import { InviteForm } from './InviteForm'
import { InvitationActions } from './InvitationActions'

export default async function MembersPage({
  params
}: {
  params: Promise<{ teamId: string }>
}) {
  const { teamId } = await params
  const { role } = await requireTeamRole(teamId, 'MEMBER')

  const [members, invitations] = await Promise.all([
    prisma.membership.findMany({
      where: { teamId },
      include: { user: { select: { id: true, name: true, email: true } } }
    }),
    prisma.invitation.findMany({
      where: { teamId, status: 'PENDING' },
      orderBy: { createdAt: 'desc' }
    })
  ])

  const canManage = role === 'OWNER' || role === 'ADMIN'

  return (
    <div>
      <h2 className="text-2xl font-bold mb-6">成员管理</h2>

      {/* 邀请表单 */}
      {canManage && (
        <div className="mb-8">
          <h3 className="font-bold mb-4">邀请新成员</h3>
          <InviteForm teamId={teamId} />
        </div>
      )}

      {/* 待处理邀请 */}
      {invitations.length > 0 && (
        <div className="mb-8">
          <h3 className="font-bold mb-4">待处理邀请</h3>
          <ul className="space-y-2">
            {invitations.map(inv => (
              <li key={inv.id} className="flex justify-between items-center p-3 bg-gray-50 rounded">
                <div>
                  <span>{inv.email}</span>
                  <span className="text-sm text-gray-500 ml-2">({inv.role})</span>
                </div>
                {canManage && (
                  <InvitationActions invitationId={inv.id} teamId={teamId} />
                )}
              </li>
            ))}
          </ul>
        </div>
      )}

      {/* 成员列表 */}
      <div>
        <h3 className="font-bold mb-4">团队成员</h3>
        <ul className="space-y-2">
          {members.map(member => (
            <li key={member.id} className="flex justify-between items-center p-3 bg-white border rounded">
              <div>
                <span>{member.user.name}</span>
                <span className="text-sm text-gray-500 ml-2">({member.role})</span>
              </div>
            </li>
          ))}
        </ul>
      </div>
    </div>
  )
}
```
